% TOC number-width refinement for multi-digit chapter.section entries.
% The standard book-class number box is too narrow for entries such as 17.10.
\makeatletter
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3.3em}}
\renewcommand*\l@subsection{\@dottedtocline{2}{4.8em}{4.2em}}
\makeatother
\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{3}

% Wider TOC number boxes for double-digit chapter/section numbers.
% The book class defaults are too narrow for entries such as 17.10 and 33.10.
\makeatletter
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3.3em}}
\renewcommand*\l@subsection{\@dottedtocline{2}{4.8em}{4.2em}}
\makeatother
