\section*{Theory: Plane cubics and the Cremona involution}

\subsection*{Plane cubics}

Assume $k$ is algebraically closed.

A \emph{plane cubic} is a curve
\[
C\subset \mathbb P^2
\]
defined by a single homogeneous polynomial of degree $3$:
\[
C=Z(F),
\qquad
F\in k[x,y,z] \text{ homogeneous of degree }3.
\]
An affine cubic in $\mathbb A^2$ given by $f(x,y)$ has projective closure obtained by homogenization:
\[
F(x,y,z)=z^3 f\!\left(\frac{x}{z},\frac{y}{z}\right).
\]

\subsubsection*{Classification by reducibility and singularities}

Every plane cubic falls into one of the following types.

\paragraph{Smooth cubic.}
If $C$ is smooth, then it is irreducible and has genus
\[
g(C)=1.
\]
After choosing a base point $O\in C$, it carries the structure of an elliptic curve via the chord--tangent law.
A line meets $C$ in $3$ points counting multiplicity.

\paragraph{Singular irreducible cubic.}
If $C$ is irreducible but singular, then it is rational:
\[
g(C)=0.
\]
The typical singularities are a node or a cusp, and $C$ admits a rational parametrization.

\paragraph{Reducible cubic.}
If $F$ factors, then $C$ is reducible.
The possibilities are:
\[
\text{three lines (with multiplicities),}
\]
or
\[
\text{a line plus a conic (with multiplicities).}
\]
For example,
\[
Z(xyz)=Z(x)\cup Z(y)\cup Z(z),
\qquad
Z\bigl(z(xy-z^2)\bigr)=Z(z)\cup Z(xy-z^2).
\]

\subsubsection*{Bezout's theorem for cubics}

If $C=Z(F)$ and $D=Z(G)$ are plane curves in $\mathbb P^2$ with no common component, then
\[
\sum_{p\in C\cap D} I_p(C,D)=\deg(F)\deg(G).
\]
In particular, a line meets a cubic in
\[
3
\]
points counting multiplicity, and a conic meets a cubic in
\[
6
\]
points counting multiplicity.

\subsubsection*{Weierstrass form}

Assume $\mathrm{char}(k)\neq 2,3$.
Every smooth plane cubic is projectively equivalent to a Weierstrass model
\[
y^2 z = x^3 + a x z^2 + b z^3,
\]
with discriminant $\Delta\neq 0$.
In this form, the group law has the explicit inversion
\[
(x:y:z)\longmapsto (x:-y:z).
\]

\subsection*{The standard Cremona involution}

Define the rational map
\[
\sigma:\mathbb P^2 \dashrightarrow \mathbb P^2,
\qquad
(x:y:z)\longmapsto (yz:zx:xy).
\]
It is undefined exactly at the three base points
\[
p_1=(1:0:0),
\qquad
p_2=(0:1:0),
\qquad
p_3=(0:0:1).
\]
Moreover, it contracts the coordinate lines:
\[
Z(x)\ \text{to}\ p_1,
\qquad
Z(y)\ \text{to}\ p_2,
\qquad
Z(z)\ \text{to}\ p_3.
\]
The map is birational, and it is an involution as a rational map:
\[
\sigma\circ \sigma = \mathrm{id}.
\]

\subsubsection*{Resolution by blow-ups}

Let
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
be the blow-up of $\mathbb P^2$ at the three base points $p_1,p_2,p_3$.
Then $\sigma$ lifts to a morphism
\[
\widetilde{\sigma}:\widetilde{\mathbb P^2}\to \mathbb P^2.
\]
In the Picard group of $\widetilde{\mathbb P^2}$, if $H$ denotes the pullback of the class of a line
and $E_1,E_2,E_3$ are the exceptional divisors, then
\[
\sigma^{*}(H)=2H-E_1-E_2-E_3,
\]
\[
\sigma^{*}(E_1)=H-E_2-E_3,
\qquad
\sigma^{*}(E_2)=H-E_1-E_3,
\qquad
\sigma^{*}(E_3)=H-E_1-E_2.
\]

\subsubsection*{Effect on degrees of plane curves}

Let $C\subset \mathbb P^2$ be a curve of degree $d$.
Assume $C$ passes through the base points with multiplicities
\[
m_1,
\qquad
m_2,
\qquad
m_3.
\]
Then the strict transform of $C$ under the Cremona involution has degree
\[
d' = 2d - m_1 - m_2 - m_3.
\]
For example, a general line has $d=1$ and $m_1=m_2=m_3=0$, hence
\[
d'=2,
\]
so it maps to a conic through the three base points.
A conic through all three base points has $d=2$ and $m_1=m_2=m_3=1$, hence
\[
d'=1,
\]
so it maps to a line.

\subsubsection*{Remark on cubics}

For a cubic $d=3$, if it passes simply through $p_1,p_2,p_3$, then
\[
d'=2\cdot 3 - 1 - 1 - 1 = 3,
\]
so it transforms to another cubic.
Higher multiplicities at the base points can decrease the degree after transformation.

\subsection*{What is a cubic}

A \emph{cubic} is an algebraic set defined by a polynomial equation of total degree $3$.

\paragraph{Affine case.}
A cubic hypersurface in $\mathbb A^n$ is
\[
X=Z(f)\subset \mathbb A^n,
\qquad
f\in k[x_1,\dots,x_n],
\qquad
\deg(f)=3.
\]

\paragraph{Projective case.}
A cubic hypersurface in $\mathbb P^n$ is
\[
X=Z(F)\subset \mathbb P^n,
\qquad
F\in k[x_0,\dots,x_n] \text{ homogeneous},
\qquad
\deg(F)=3.
\]

\paragraph{Curve versus surface.}
A hypersurface in $\mathbb A^n$ (or $\mathbb P^n$) typically has dimension $n-1$.
Hence:
\[
\text{cubic in } \mathbb A^2 \text{ or } \mathbb P^2 \text{ is a curve},
\]
\[
\text{cubic in } \mathbb A^3 \text{ or } \mathbb P^3 \text{ is a surface}.
\]

\section*{Affine cubic versus projective cubic: are they the same?}

Let $k$ be a field.

\subsubsection*{Affine hypersurfaces}

A \emph{cubic affine hypersurface} in $\mathbb A^n$ is the zero locus of a polynomial of total degree at most $3$:
\[
X_{\mathrm{aff}}
=
Z(f)
\subset
\mathbb A^n,
\qquad
f\in k[x_1,\dots,x_n],
\qquad
\deg(f)\le 3.
\]

\subsubsection*{Projective hypersurfaces}

A \emph{cubic projective hypersurface} in $\mathbb P^n$ is the zero locus of a homogeneous polynomial of degree $3$:
\[
X_{\mathrm{proj}}
=
Z(F)
\subset
\mathbb P^n,
\qquad
F\in k[x_0,x_1,\dots,x_n] \text{ homogeneous},
\qquad
\deg(F)=3.
\]
In projective space, equations must be homogeneous.

\subsubsection*{Homogenization and projective closure}

Given $f\in k[x_1,\dots,x_n]$ with $\deg(f)\le 3$, its homogenization of degree $3$ is the homogeneous polynomial
\[
F(x_0,x_1,\dots,x_n)
=
x_0^{3}\, f\!\left(\frac{x_1}{x_0},\dots,\frac{x_n}{x_0}\right).
\]
Equivalently, each monomial
\[
x_1^{a_1}\cdots x_n^{a_n}
\]
of total degree
\[
a_1+\cdots+a_n=d\le 3
\]
is replaced by
\[
x_1^{a_1}\cdots x_n^{a_n}x_0^{3-d}.
\]

The projective hypersurface
\[
X_{\mathrm{proj}}=Z(F)\subset \mathbb P^n
\]
is called the \emph{projective closure} of
\[
X_{\mathrm{aff}}=Z(f)\subset \mathbb A^n.
\]

\subsubsection*{Affine chart and points at infinity}

Consider the standard affine chart
\[
U_0=\{x_0\neq 0\}\subset \mathbb P^n,
\]
with coordinates
\[
x_i'=\frac{x_i}{x_0}.
\]
On this chart we have
\[
F(1,x_1,\dots,x_n)=f(x_1,\dots,x_n).
\]
Hence
\[
X_{\mathrm{proj}}\cap U_0
\cong
X_{\mathrm{aff}}.
\]
So the affine hypersurface is the part of the projective hypersurface lying in the chart $x_0\neq 0$.

However, $X_{\mathrm{proj}}$ may contain \emph{additional points at infinity}, namely
\[
X_{\mathrm{proj}}\cap \{x_0=0\}.
\]
Thus, in general,
\[
X_{\mathrm{proj}}
=
\overline{X_{\mathrm{aff}}}^{\ \mathrm{proj}}
=
X_{\mathrm{aff}}
\cup
\bigl(X_{\mathrm{proj}}\cap \{x_0=0\}\bigr),
\]
where $\overline{X_{\mathrm{aff}}}^{\ \mathrm{proj}}$ denotes the projective closure.

\subsubsection*{Example}

In $\mathbb A^2$ consider the affine cubic curve
\[
f(x,y)=y-x^3.
\]
Homogenizing to degree $3$ gives a projective cubic in $\mathbb P^2$:
\[
F(x,y,z)=yz^2-x^3.
\]
On the affine chart $z=1$,
\[
F(x,y,1)=y-x^3,
\]
so this recovers the affine curve.
At infinity $z=0$,
\[
F(x,y,0)=-x^3,
\]
so $x=0$, and the projective closure contains the point
\[
(0:1:0).
\]
Hence the projective cubic is the affine cubic together with an extra point at infinity.


\subsection*{Why projective closures can be larger although homogeneous polynomials are more restrictive}

Let $k$ be a field.

\subsubsection*{Homogeneous polynomials are a smaller class}

A polynomial
\[
f\in k[x_1,\dots,x_n]
\]
is called \emph{homogeneous of degree $d$} if every monomial occurring in $f$ has total degree $d$.
For example, in $k[x,y,z]$ the polynomial
\[
x^3+x^2y+yz^2
\]
is homogeneous of degree $3$, whereas
\[
x^3+y
\]
is not homogeneous, because the monomials have degrees $3$ and $1$.

Thus, as polynomials, homogeneous degree-$3$ polynomials form a \emph{more restrictive} class than arbitrary degree-$3$ polynomials.

\subsubsection*{Why the geometric set becomes larger in projective space}

The apparent paradox comes from comparing objects in \emph{different ambient spaces}.
Affine space $\mathbb A^n$ sits inside projective space $\mathbb P^n$ as the standard chart
\[
U_0=\{x_0\neq 0\}\subset \mathbb P^n,
\]
via
\[
(x_1,\dots,x_n)\longmapsto (1:x_1:\cdots:x_n).
\]
Projective space also contains the hyperplane at infinity
\[
H_\infty=\{x_0=0\}\subset \mathbb P^n,
\]
which has no affine counterpart.

Given an affine equation
\[
f(x_1,\dots,x_n)=0
\]
one forms its homogenization $F$ and considers the projective hypersurface
\[
Z(F)\subset \mathbb P^n.
\]
On the affine chart $U_0$ one recovers the original affine hypersurface:
\[
Z(F)\cap U_0
\cong
Z(f)\subset \mathbb A^n.
\]
However, $Z(F)$ may also meet the hyperplane at infinity $H_\infty$.
Hence, in general,
\[
Z(F)
=
\bigl(Z(F)\cap U_0\bigr)\ \cup\ \bigl(Z(F)\cap H_\infty\bigr),
\]
so $Z(F)$ can be larger than the affine set $Z(f)$ because it may contain additional points at infinity.

\subsubsection*{What points at infinity are added}

Assume $\deg(f)\le 3$ and write $f$ as a sum of homogeneous parts:
\[
f
=
f_3+f_2+f_1+f_0,
\]
where $f_d$ is homogeneous of degree $d$.

The homogenization of degree $3$ is
\[
F(x_0,x_1,\dots,x_n)
=
f_3(x_1,\dots,x_n)
+
x_0 f_2(x_1,\dots,x_n)
+
x_0^2 f_1(x_1,\dots,x_n)
+
x_0^3 f_0.
\]
Restricting to the hyperplane at infinity $H_\infty=\{x_0=0\}$ gives
\[
F(0,x_1,\dots,x_n)
=
f_3(x_1,\dots,x_n).
\]
Therefore the points at infinity of the projective closure are exactly the projective zeros of the top-degree part:
\[
Z(F)\cap H_\infty
=
Z\bigl(f_3\bigr)\subset \mathbb P^{n-1}.
\]
So the geometry at infinity is controlled by the leading homogeneous term of $f$.

\subsubsection*{Example}

Consider the affine cubic curve in $\mathbb A^2$:
\[
f(x,y)=y-x^3.
\]
Its top-degree part is
\[
f_3(x,y)=-x^3.
\]
Homogenizing to degree $3$ in $\mathbb P^2$ yields
\[
F(x,y,z)=yz^2-x^3.
\]
On the affine chart $z=1$,
\[
F(x,y,1)=y-x^3,
\]
so $Z(F)\cap\{z\neq 0\}$ is the original affine curve.

At infinity $z=0$,
\[
F(x,y,0)=-x^3,
\]
so
\[
x=0,
\]
and the projective closure contains the point
\[
(0:1:0).
\]
Thus the projective curve is larger than the affine curve by the additional point at infinity.

\subsubsection*{Summary}

Homogeneous polynomials are more restrictive as equations, but projective space contains additional points (the hyperplane at infinity).
When one passes from an affine hypersurface $Z(f)\subset \mathbb A^n$ to its projective closure $Z(F)\subset \mathbb P^n$, one typically adds solutions at infinity, which is why $Z(F)$ can be larger than $Z(f)$.


\section*{Picard group and blow-ups (plane case, with applications to Cremona)}

\subsection*{Divisors and the Picard group}

Let $X$ be a smooth irreducible variety over an algebraically closed field $k$.

\subsubsection*{Weil divisors}

A \emph{Weil divisor} on $X$ is a finite formal integer linear combination of irreducible codimension-$1$ subvarieties:
\[
D=\sum_i n_i V_i,
\qquad
n_i\in \mathbb Z.
\]
The group of Weil divisors is denoted
\[
\mathrm{Div}(X).
\]

A divisor is called \emph{effective} if
\[
n_i\ge 0
\]
for all $i$, and we write
\[
D\ge 0.
\]

\subsubsection*{Principal divisors}

If $f\in k(X)^\times$ is a nonzero rational function on $X$, then its divisor of zeros and poles is a divisor
\[
\mathrm{div}(f)\in \mathrm{Div}(X),
\]
called a \emph{principal divisor}.
Two divisors are \emph{linearly equivalent} if their difference is principal:
\[
D\sim D'
\quad\Longleftrightarrow\quad
D-D'=\mathrm{div}(f)
\text{ for some } f\in k(X)^\times.
\]

\subsubsection*{Picard group}

The \emph{Picard group} of $X$ is
\[
\mathrm{Pic}(X)
=
\{\text{line bundles on }X\}/\cong
\]
with group law given by tensor product.

On a smooth variety, one has the fundamental identification
\[
\mathrm{Pic}(X)
\cong
\mathrm{Div}(X)\big/\sim,
\]
so the Picard group may be viewed as divisors modulo linear equivalence.

\subsection*{Picard group of projective space}

For projective space,
\[
\mathrm{Pic}(\mathbb P^n)\cong \mathbb Z.
\]
A generator is the class of a hyperplane divisor $H$.
Thus every line bundle is of the form
\[
\mathcal O_{\mathbb P^n}(d),
\qquad
d\in\mathbb Z,
\]
and divisor classes are multiples of $H$:
\[
\mathrm{Pic}(\mathbb P^n)=\mathbb Z\cdot H.
\]

In particular, on $\mathbb P^2$ we write $H$ for the class of a line.

\subsection*{Blow-up at a point: geometric idea}

Let $X$ be a smooth surface and let $p\in X$ be a point.
The \emph{blow-up of $X$ at $p$} is a morphism
\[
\pi:\widetilde X\to X
\]
such that:
\begin{enumerate}
	\item $\pi$ is an isomorphism over $X\setminus\{p\}$,
	\item the fiber $\pi^{-1}(p)$ is a copy of $\mathbb P^1$, called the \emph{exceptional divisor}.
\end{enumerate}

Intuitively, blowing up replaces the point $p$ by the space of directions through $p$.
On a surface, directions are parametrized by $\mathbb P^1$, which explains why the exceptional fiber is $\mathbb P^1$.

\subsubsection*{Local model in affine coordinates}

In local affine coordinates $(u,v)$ near $p=(0,0)\in \mathbb A^2$, the blow-up
\[
\mathrm{Bl}_0(\mathbb A^2)
\subset
\mathbb A^2\times \mathbb P^1
\]
is given by the incidence relation
\[
u b = v a
\]
in coordinates $(u,v)\in \mathbb A^2$ and $(a:b)\in \mathbb P^1$.
The projection
\[
\pi(u,v,(a:b))=(u,v)
\]
is the blow-down map.
The exceptional divisor is
\[
E=\pi^{-1}(0,0)\cong \mathbb P^1.
\]

\subsection*{Picard group under blow-up of $\mathbb P^2$}

Let
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
be the blow-up of $\mathbb P^2$ at one point $p$.
Let $E$ be the exceptional divisor.

\subsubsection*{Generators and structure}

The Picard group becomes rank $2$:
\[
\mathrm{Pic}(\widetilde{\mathbb P^2})
\cong
\mathbb Z\cdot H \ \oplus\ \mathbb Z\cdot E,
\]
where $H$ denotes the pullback of the class of a line:
\[
H=\pi^{*}(\text{line class}).
\]

If we blow up $r$ distinct points $p_1,\dots,p_r$, then
\[
\mathrm{Pic}(\widetilde{\mathbb P^2})
\cong
\mathbb Z\cdot H \ \oplus\ \mathbb Z\cdot E_1 \ \oplus\ \cdots\ \oplus\ \mathbb Z\cdot E_r.
\]

\subsubsection*{Intersection numbers}

On a smooth surface, divisor classes have an intersection product.
For the blow-up of $\mathbb P^2$ at $r$ points one has
\[
H^2=1,
\qquad
H\cdot E_i=0,
\qquad
E_i\cdot E_j=0 \text{ for } i\neq j,
\qquad
E_i^2=-1.
\]
The key geometric meaning is:
\begin{itemize}
	\item $H^2=1$ because two general lines meet in one point.
	\item $E_i^2=-1$ reflects that $E_i$ is an exceptional curve of the first kind.
\end{itemize}

\subsection*{Strict transforms and multiplicities}

Let $C\subset \mathbb P^2$ be a plane curve of degree $d$.
Assume $C$ passes through $p_i$ with multiplicity $m_i$.
After blowing up at the points $p_i$, the class of the strict transform $\widetilde C$ in the Picard group is
\[
[\widetilde C]
=
dH-\sum_{i=1}^r m_i E_i.
\]

This formula records two pieces of data:
\begin{enumerate}
	\item the degree $d$ of the original curve,
	\item how much of $C$ is concentrated at the blown-up points, via the multiplicities $m_i$.
\end{enumerate}

\subsection*{Cremona involution via blow-ups (why Picard matters)}

Let
\[
\sigma:\mathbb P^2\dashrightarrow \mathbb P^2
\]
be the standard Cremona transformation
\[
(x:y:z)\longmapsto (yz:zx:xy),
\]
with base points
\[
p_1=(1:0:0),
\qquad
p_2=(0:1:0),
\qquad
p_3=(0:0:1).
\]
Let
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
be the blow-up at $p_1,p_2,p_3$, with exceptional divisors $E_1,E_2,E_3$ and line class $H$.

Then $\sigma$ lifts to a morphism on $\widetilde{\mathbb P^2}$, and its action on divisor classes is encoded by
\[
\sigma^{*}(H)=2H-E_1-E_2-E_3,
\]
\[
\sigma^{*}(E_1)=H-E_2-E_3,
\qquad
\sigma^{*}(E_2)=H-E_1-E_3,
\qquad
\sigma^{*}(E_3)=H-E_1-E_2.
\]

\subsubsection*{Degree change formula}

Let $C\subset \mathbb P^2$ be a curve of degree $d$, with multiplicities $m_1,m_2,m_3$ at the base points.
The strict transform of $\sigma(C)$ has degree
\[
d' = 2d - m_1 - m_2 - m_3.
\]
This is obtained by tracking the class
\[
[\widetilde C]=dH-m_1E_1-m_2E_2-m_3E_3
\]
under the pullback action on $\mathrm{Pic}(\widetilde{\mathbb P^2})$.

\subsubsection*{Example}

A general line has $d=1$ and $m_1=m_2=m_3=0$, hence
\[
d'=2,
\]
so it maps to a conic.
A conic through all three base points has $d=2$ and $m_1=m_2=m_3=1$, hence
\[
d'=1,
\]
so it maps to a line.

\section*{Picard group and blow-ups (plane case)}

\subsection*{Divisors and the Picard group}

Let $X$ be a smooth irreducible variety over an algebraically closed field $k$.

\subsubsection*{Weil divisors}

A \emph{Weil divisor} on $X$ is a finite formal integer linear combination of irreducible codimension-$1$ subvarieties:
\[
D=\sum_i n_i V_i,
\qquad
n_i\in \mathbb Z.
\]
The group of Weil divisors is denoted
\[
\mathrm{Div}(X).
\]

\subsubsection*{Principal divisors and linear equivalence}

If $f\in k(X)^\times$ is a nonzero rational function on $X$, then $f$ defines a divisor of zeros and poles
\[
\mathrm{div}(f)\in \mathrm{Div}(X),
\]
called a \emph{principal divisor}.
Two divisors are \emph{linearly equivalent} if their difference is principal:
\[
D\sim D'
\quad\Longleftrightarrow\quad
D-D'=\mathrm{div}(f)
\text{ for some } f\in k(X)^\times.
\]

\subsubsection*{Picard group}

The \emph{Picard group} of $X$ is the group of line bundles on $X$ up to isomorphism:
\[
\mathrm{Pic}(X)
=
\{\text{line bundles on }X\}/\cong,
\]
with group law given by tensor product.
On a smooth variety there is an identification
\[
\mathrm{Pic}(X)\cong \mathrm{Div}(X)\big/\sim,
\]
so $\mathrm{Pic}(X)$ can be viewed as divisor classes modulo linear equivalence.

\subsection*{Picard group of $\mathbb P^2$}

For projective space one has
\[
\mathrm{Pic}(\mathbb P^n)\cong \mathbb Z.
\]
A generator is the hyperplane class $H$.
In particular,
\[
\mathrm{Pic}(\mathbb P^2)=\mathbb Z\cdot H,
\]
and every divisor class on $\mathbb P^2$ is of the form $dH$.

\subsection{Blow-up of a point on a surface}

Let $X$ be a smooth surface and let $p\in X$.
The \emph{blow-up of $X$ at $p$} is a morphism
\[
\pi:\widetilde X\to X
\]
such that:
\begin{enumerate}
	\item $\pi$ is an isomorphism over $X\setminus\{p\}$,
	\item the fiber $\pi^{-1}(p)$ is a curve
	\[
	E=\pi^{-1}(p)\cong \mathbb P^1,
	\]
	called the \emph{exceptional divisor}.
\end{enumerate}
Geometrically, blowing up replaces the point $p$ by the set of tangent directions through $p$.

\subsubsection{Local model in affine coordinates}

In local affine coordinates $(u,v)$ near $p=(0,0)\in \mathbb A^2$, the blow-up
\[
\mathrm{Bl}_0(\mathbb A^2)
\subset
\mathbb A^2\times \mathbb P^1
\]
is given by
\[
u b = v a
\]
in coordinates $(u,v)\in \mathbb A^2$ and $(a:b)\in \mathbb P^1$.
The projection
\[
\pi(u,v,(a:b))=(u,v)
\]
is the blow-down map, and the exceptional divisor is $E=\{u=v=0\}\cong \mathbb P^1$.

\subsection*{Picard group after blowing up $\mathbb P^2$}

Let
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
be the blow-up of $\mathbb P^2$ at $r$ distinct points $p_1,\dots,p_r$, with exceptional divisors $E_1,\dots,E_r$.
Let $H$ denote the pullback of the class of a line.

Then
\[
\mathrm{Pic}(\widetilde{\mathbb P^2})
\cong
\mathbb Z\cdot H\ \oplus\ \mathbb Z\cdot E_1\ \oplus\ \cdots\ \oplus\ \mathbb Z\cdot E_r.
\]

\subsubsection*{Intersection numbers}

On the blow-up of $\mathbb P^2$ at $r$ points, the intersection product satisfies
\[
H^2=1,
\qquad
H\cdot E_i=0,
\qquad
E_i\cdot E_j=0 \text{ for } i\neq j,
\qquad
E_i^2=-1.
\]

\subsection*{Strict transforms and multiplicities}

Let $C\subset \mathbb P^2$ be a plane curve of degree $d$.
Assume $C$ passes through $p_i$ with multiplicity $m_i$.
Then the divisor class of the strict transform $\widetilde C\subset \widetilde{\mathbb P^2}$ is
\[
[\widetilde C]
=
dH-\sum_{i=1}^r m_i E_i.
\]

\subsection*{Remark: connection to Cremona}

For the standard Cremona involution with base points $p_1,p_2,p_3$, one blows up these three points to obtain
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2.
\]
The Cremona map lifts to a morphism on $\widetilde{\mathbb P^2}$ and acts linearly on
\[
\mathrm{Pic}(\widetilde{\mathbb P^2}).
\]
Tracking a curve class
\[
dH-m_1E_1-m_2E_2-m_3E_3
\]
under this action yields the degree change formula
\[
d' = 2d - m_1 - m_2 - m_3.
\]

\section*{Concrete divisors, codimension, and line bundles}

\subsection*{Why divisors use codimension one}

Let $X$ be an irreducible variety.
A \emph{prime divisor} on $X$ means an irreducible closed subvariety of codimension $1$.
A \emph{Weil divisor} is a finite formal integer combination of prime divisors:
\[
D=\sum_i n_i V_i,
\qquad
n_i\in \mathbb Z.
\]

The reason codimension $1$ appears is that zeros and poles of a rational function occur along hypersurfaces.
For a nonzero rational function $f\in k(X)^\times$ one defines the \emph{principal divisor}
\[
\mathrm{div}(f)=\sum_{V} \operatorname{ord}_{V}(f)\,V,
\]
where the sum runs over prime divisors $V\subset X$ and $\operatorname{ord}_{V}(f)$ is the order of vanishing (negative for poles) along $V$.
Higher codimension sets do not contribute to $\mathrm{div}(f)$.

In particular:
\[
\dim(X)=1 \Longrightarrow \text{prime divisors are points},
\]
\[
\dim(X)=2 \Longrightarrow \text{prime divisors are curves}.
\]

\subsection*{Example: divisors on $\mathbb P^1$}

On $X=\mathbb P^1$, prime divisors are points.
Using the affine coordinate $t=x/y$, consider
\[
f(t)=\frac{t-a}{t-b}\in k(t)^\times.
\]
Then $f$ has a simple zero at $t=a$ and a simple pole at $t=b$, hence
\[
\mathrm{div}(f)=(a)-(b).
\]
One has
\[
\mathrm{Pic}(\mathbb P^1)\cong \mathbb Z,
\]
and the degree of a divisor
\[
D=\sum_i n_i(p_i)
\]
is
\[
\deg(D)=\sum_i n_i.
\]
Every principal divisor has degree $0$.

\subsection*{Example: divisors on $\mathbb P^2$}

On $X=\mathbb P^2$, prime divisors are irreducible plane curves.
Let $(x:y:z)$ be homogeneous coordinates and consider the rational function
\[
f=\frac{x}{z}\in k(\mathbb P^2)^\times.
\]
The zero locus of $f$ is the line $Z(x)$ and the pole locus is the line $Z(z)$, so
\[
\mathrm{div}\!\left(\frac{x}{z}\right)=Z(x)-Z(z).
\]
Thus $Z(x)\sim Z(z)$, and in fact all lines are linearly equivalent.
One has
\[
\mathrm{Pic}(\mathbb P^2)\cong \mathbb Z,
\]
generated by the class $H$ of a line.

\subsection*{Example: blow-up of $\mathbb P^2$ at a point}

Let
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
be the blow-up of $\mathbb P^2$ at a point $p$.
The exceptional divisor is
\[
E=\pi^{-1}(p)\cong \mathbb P^1.
\]
The Picard group becomes
\[
\mathrm{Pic}(\widetilde{\mathbb P^2})\cong \mathbb Z\cdot H \oplus \mathbb Z\cdot E,
\]
where $H$ is the pullback of the class of a line.

If $C\subset \mathbb P^2$ is a curve of degree $d$ and has multiplicity $m$ at $p$, then the class of its strict transform satisfies
\[
[\widetilde C]=dH-mE.
\]
For example, a line not passing through $p$ has strict transform class $H$, while a line passing through $p$ has strict transform class $H-E$.

\subsection*{Line bundles are not projective lines}

A \emph{line bundle} on $X$ is a rank-$1$ vector bundle.
Equivalently, it is a morphism
\[
\pi:L\to X
\]
such that for each $x\in X$ the fiber $L_x$ is a $1$-dimensional $k$-vector space and locally
\[
L|_{U}\cong U\times k.
\]
Thus a line bundle is a family of $1$-dimensional vector spaces parametrized by $X$.

The variety $\mathbb P^1$ is a projective line, not a line bundle.
A $\mathbb P^1$-bundle arises as the projectivization $\mathbb P(E)$ of a rank-$2$ vector bundle $E$.
If $L$ is a line bundle (rank $1$), then its projectivization has fibers $\mathbb P^0$ (a point), so
\[
\mathbb P(L)\cong X.
\]

\subsection*{One blow-up of $\mathbb P^2$: concrete divisor classes and computations}

Let $p\in \mathbb P^2$ be a point and let
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
be the blow-up of $\mathbb P^2$ at $p$.
Let
\[
E=\pi^{-1}(p)\cong \mathbb P^1
\]
be the exceptional divisor.
Let $H$ denote the pullback of the class of a line in $\mathbb P^2$.

\subsubsection*{Picard group and intersection pairing}

One has
\[
\mathrm{Pic}(\widetilde{\mathbb P^2})\cong \mathbb Z\cdot H \oplus \mathbb Z\cdot E,
\]
and the intersection numbers satisfy
\[
H^2=1,
\qquad
H\cdot E=0,
\qquad
E^2=-1.
\]

For divisor classes
\[
D=aH-bE,
\qquad
D'=cH-dE,
\]
the intersection product is
\[
D\cdot D' = ac-bd.
\]

\subsubsection*{Strict transform class, degree, and multiplicity}

Let $C\subset \mathbb P^2$ be a plane curve of degree $d$, and assume $C$ has multiplicity $m$ at $p$.
Then the divisor class of the strict transform $\widetilde C\subset \widetilde{\mathbb P^2}$ is
\[
[\widetilde C]=dH-mE.
\]

Moreover, degree and multiplicity can be recovered as intersection numbers:
\[
\widetilde C\cdot H = d,
\qquad
\widetilde C\cdot E = m.
\]
Indeed,
\[
(dH-mE)\cdot H = dH^2-m(E\cdot H)=d,
\]
and
\[
(dH-mE)\cdot E = d(H\cdot E)-mE^2=m.
\]

\subsubsection*{Examples}

\begin{enumerate}
	\item A line $L$ not passing through $p$ has strict transform class
	\[
	[\widetilde L]=H,
	\]
	and
	\[
	\widetilde L^{\,2}=H^2=1.
	\]
	
	\item A line $L$ passing through $p$ has multiplicity $m=1$ at $p$, hence
	\[
	[\widetilde L]=H-E.
	\]
	Its self-intersection is
	\[
	(H-E)^2=H^2-2(H\cdot E)+E^2=1-1=0,
	\]
	and it meets the exceptional divisor once:
	\[
	(H-E)\cdot E = 1.
	\]
	
	\item A conic $Q$ not passing through $p$ has class
	\[
	[\widetilde Q]=2H,
	\]
	so
	\[
	\widetilde Q^{\,2}=(2H)^2=4.
	\]
	
	\item A conic $Q$ passing through $p$ simply has multiplicity $m=1$, hence
	\[
	[\widetilde Q]=2H-E.
	\]
	Then
	\[
	(2H-E)^2 = 4-1 = 3,
	\qquad
	(2H-E)\cdot E = 1.
	\]
	
	\item A conic $Q$ with multiplicity $m=2$ at $p$ has class
	\[
	[\widetilde Q]=2H-2E,
	\]
	so
	\[
	(2H-2E)^2=4-4=0,
	\qquad
	(2H-2E)\cdot E = 2.
	\]
\end{enumerate}

\subsection*{The tautological line bundle on projective space}

Let $V$ be a finite-dimensional $k$-vector space, $\dim(V)=n+1$.
Projective space $\mathbb P(V)$ is the set of $1$-dimensional linear subspaces $\ell\subset V$.
A point of $\mathbb P(V)$ is written $[\ell]$.

\subsubsection*{The tautological bundle $\mathcal O_{\mathbb P(V)}(-1)$}

There is a canonical rank-$1$ subbundle of the trivial bundle $\mathbb P(V)\times V$:
\[
\mathcal O_{\mathbb P(V)}(-1)\subset \mathbb P(V)\times V,
\]
defined by
\[
\mathcal O_{\mathbb P(V)}(-1)
=
\{\,([\ell],v)\in \mathbb P(V)\times V \mid v\in \ell\,\}.
\]
Its fiber over a point $[\ell]\in \mathbb P(V)$ is the line itself:
\[
\mathcal O_{\mathbb P(V)}(-1)_{[\ell]}=\ell.
\]
This is why it is called the tautological line bundle.

\subsubsection*{The principal $k^\times$-bundle and the relation to $\mathcal O(-1)$}

There is a natural quotient map
\[
V\setminus\{0\}\longrightarrow \mathbb P(V),
\qquad
v\longmapsto [kv],
\]
whose fibers are $\ell\setminus\{0\}$.
After choosing a basis of $\ell$, one identifies $\ell\setminus\{0\}$ with $k^\times$.
This makes $V\setminus\{0\}\to \mathbb P(V)$ a principal $k^\times$-bundle.
The line bundle $\mathcal O_{\mathbb P(V)}(-1)$ can be viewed as the associated line bundle obtained by adjoining the zero vector in each fiber, but the intrinsic definition above is standard.

\subsubsection*{The hyperplane bundle $\mathcal O(1)$ and the Picard group}

Define the dual line bundle
\[
\mathcal O_{\mathbb P(V)}(1)=\mathcal O_{\mathbb P(V)}(-1)^{\vee}.
\]
On $\mathbb P^n=\mathbb P(k^{n+1})$, one has
\[
\mathrm{Pic}(\mathbb P^n)\cong \mathbb Z,
\]
and the class of $\mathcal O(1)$ is a generator.
Hyperplanes are precisely the zero loci of global sections of $\mathcal O(1)$, and more generally
\[
\mathcal O(d)=\mathcal O(1)^{\otimes d}
\]
corresponds to degree-$d$ hypersurfaces.

\subsubsection*{Local trivializations on standard charts}

Let $[x_0:\cdots:x_n]$ be homogeneous coordinates on $\mathbb P^n$ and set
\[
U_i=\{x_i\neq 0\}.
\]
On each $U_i$, the bundle $\mathcal O(-1)$ is trivial.
On overlaps $U_i\cap U_j$, the transition function for $\mathcal O(-1)$ is
\[
g_{ij}=\frac{x_j}{x_i}.
\]
For the dual bundle $\mathcal O(1)$ the transition function is
\[
g_{ij}^{-1}=\frac{x_i}{x_j}.
\]

\subsubsection*{Connection to blow-ups}

If
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
is the blow-up at a point, then the line class $H$ used in Picard computations on $\widetilde{\mathbb P^2}$ is the pullback of the hyperplane bundle:
\[
H=\pi^{*}\mathcal O_{\mathbb P^2}(1).
\]


\subsection*{From divisors to line bundles: $\mathcal O(D)$ and zero loci of sections}

Let $X$ be a smooth irreducible variety over an algebraically closed field $k$.
On a smooth variety, divisor classes and line bundles are closely linked.

\subsubsection*{Cartier divisors and local equations}

A \emph{Cartier divisor} $D$ on $X$ can be described by an open cover $\{U_i\}$ and rational functions
\[
f_i\in k(X)^\times
\]
such that on overlaps $U_i\cap U_j$ one has
\[
f_i/f_j\in \mathcal O_X^\times(U_i\cap U_j).
\]
Intuitively, each $f_i$ is a local defining equation for the divisor on $U_i$, and the compatibility says that two local equations differ by a nowhere-vanishing regular function.

On a smooth variety every Weil divisor is Cartier, so the construction below applies to Weil divisors as well.

\subsubsection*{Definition of the line bundle $\mathcal O(D)$}

Given a Cartier divisor $D$ represented by local equations $\{(U_i,f_i)\}$, define a line bundle $\mathcal O(D)$ by gluing trivial bundles on each $U_i$ with transition functions
\[
g_{ij}=\frac{f_i}{f_j}\in \mathcal O_X^\times(U_i\cap U_j).
\]
Equivalently, one may define the associated invertible sheaf by
\[
\mathcal O(D)(U)=\{\,h\in k(X)\mid h f_i \in \mathcal O_X(U\cap U_i)\ \text{for all } i\,\}.
\]
This construction depends only on the linear equivalence class of $D$.

\subsubsection*{Linear equivalence corresponds to isomorphic line bundles}

If
\[
D\sim D',
\]
then
\[
D-D'=\mathrm{div}(\varphi)
\]
for some $\varphi\in k(X)^\times$.
In this case one has a canonical isomorphism of line bundles
\[
\mathcal O(D)\cong \mathcal O(D').
\]
Hence the assignment $D\mapsto \mathcal O(D)$ factors through divisor classes and gives the identification
\[
\mathrm{Pic}(X)\cong \mathrm{Div}(X)\big/\sim.
\]

\subsubsection*{Sections and divisors of zeros and poles}

A rational section $s$ of $\mathcal O(D)$ can be viewed locally as a rational function multiplied by the local trivializing section.
Its divisor of zeros and poles, denoted $\mathrm{div}(s)$, is a Weil divisor.
There is a fundamental relation:
\[
\mathrm{div}(s)+D\ge 0
\quad\Longleftrightarrow\quad
s \text{ is a regular section of } \mathcal O(D).
\]
In particular, if $s$ is a nonzero regular section of $\mathcal O(D)$, then its zero locus defines an effective divisor and one has
\[
Z(s)=\mathrm{div}(s)+D.
\]
Thus divisors can be realized as zero loci of sections of associated line bundles.

\subsubsection*{Example: $\mathbb P^n$, hyperplanes, and $\mathcal O(1)$}

Let $H\subset \mathbb P^n$ be a hyperplane.
Then the associated line bundle is the hyperplane bundle:
\[
\mathcal O_{\mathbb P^n}(H)\cong \mathcal O_{\mathbb P^n}(1).
\]
A hyperplane $H=Z(L)$ is given by a nonzero linear form $L$, and $L$ defines a global section of $\mathcal O(1)$ whose zero locus is exactly $H$:
\[
Z(L)=H.
\]
More generally, for $d\ge 1$,
\[
\mathcal O_{\mathbb P^n}(d)=\mathcal O_{\mathbb P^n}(1)^{\otimes d},
\]
and homogeneous polynomials of degree $d$ define global sections of $\mathcal O(d)$ whose zero loci are degree-$d$ hypersurfaces.

\subsubsection*{Connection to blow-ups}

If
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
is the blow-up at a point, then the class $H$ used in divisor computations on $\widetilde{\mathbb P^2}$ is the pullback of the hyperplane bundle:
\[
\mathcal O_{\widetilde{\mathbb P^2}}(H)\cong \pi^{*}\mathcal O_{\mathbb P^2}(1).
\]

\subsection*{Projective bundles $\mathbb P(E)\to X$}

Let $X$ be a variety and let
\[
p:E\to X
\]
be a vector bundle (locally free sheaf) of rank $r$.
For each point $x\in X$, the fiber $E_x$ is an $r$-dimensional $k$-vector space.

\subsubsection*{Definition of the projective bundle}

The \emph{projective bundle} associated to $E$ is a morphism
\[
\pi:\mathbb P(E)\to X
\]
whose fiber over $x$ is the projective space of the fiber:
\[
\pi^{-1}(x)\cong \mathbb P(E_x)\cong \mathbb P^{r-1}.
\]
Thus:
\[
\mathrm{rank}(E)=2 \Longrightarrow \mathbb P(E)\to X \text{ is a } \mathbb P^1\text{-bundle},
\]
\[
\mathrm{rank}(E)=1 \Longrightarrow \mathbb P(E)\to X \text{ has fibers } \mathbb P^0 \text{ and } \mathbb P(E)\cong X.
\]

\subsubsection*{Local triviality}

Projective bundles are locally trivial in the Zariski topology.
If $E$ is trivial on an open set $U\subset X$, meaning
\[
E|_{U}\cong U\times k^{r},
\]
then
\[
\mathbb P(E)|_{U}\cong U\times \mathbb P^{r-1}.
\]

\subsubsection*{Tautological line bundle and the universal exact sequence}

There are two common conventions:
\begin{itemize}
	\item $\mathbb P(E)$ parameterizes lines in the fibers $E_x$,
	\item $\mathbb P(E)$ parameterizes $1$-dimensional quotients of the fibers $E_x$.
\end{itemize}

Using the quotient convention, there is a canonical surjection on $\mathbb P(E)$ producing the \emph{universal quotient line bundle}
\[
\pi^{*}E \twoheadrightarrow \mathcal O_{\mathbb P(E)}(1),
\]
and a short exact sequence
\[
0\to \mathcal S \to \pi^{*}E \to \mathcal O_{\mathbb P(E)}(1)\to 0,
\]
where $\mathcal S$ is a rank $r-1$ subbundle.
The dual line bundle
\[
\mathcal O_{\mathbb P(E)}(-1)=\mathcal O_{\mathbb P(E)}(1)^{\vee}
\]
is often called the tautological line bundle.

\subsubsection*{Picard group of a projective bundle}

Under mild hypotheses (for example $X$ integral and $E$ locally free), one has an isomorphism
\[
\mathrm{Pic}(\mathbb P(E))
\cong
\mathrm{Pic}(X)\ \oplus\ \mathbb Z\cdot [\mathcal O_{\mathbb P(E)}(1)].
\]
Concretely, every line bundle on $\mathbb P(E)$ can be written uniquely as
\[
\pi^{*}L \otimes \mathcal O_{\mathbb P(E)}(m),
\qquad
L\in \mathrm{Pic}(X),
\qquad
m\in \mathbb Z.
\]

\subsubsection*{Example: blow-up exceptional divisor}

If
\[
\pi:\widetilde X\to X
\]
is the blow-up of a smooth variety $X$ at a point $p$, then the exceptional divisor is a projective space:
\[
E=\pi^{-1}(p)\cong \mathbb P^{\dim(X)-1}.
\]
In particular, for the blow-up of $\mathbb P^2$ at a point one has
\[
E\cong \mathbb P^1.
\]
Geometrically, $E$ parameterizes tangent directions at the blown-up point.

\subsection*{Concrete examples: sections of $\mathcal O(1)$ and $\mathcal O(d)$}

Let $[x_0:\cdots:x_n]$ be homogeneous coordinates on $\mathbb P^n$.

\subsubsection*{Hyperplanes as zero loci of sections of $\mathcal O(1)$}

Let
\[
L=a_0x_0+\cdots+a_nx_n
\]
be a nonzero linear form.
The hyperplane is
\[
H=Z(L)\subset \mathbb P^n.
\]
The polynomial $L$ defines a global section of $\mathcal O_{\mathbb P^n}(1)$, and its vanishing locus as a section is exactly $H$:
\[
Z(L)=H.
\]

On the standard chart
\[
U_i=\{x_i\neq 0\},
\]
the section is represented by the regular function
\[
\frac{L}{x_i}.
\]
Hence $H\cap U_i$ is cut out by one equation
\[
\frac{L}{x_i}=0
\]
in the affine coordinates on $U_i$.

\subsubsection*{Degree-$d$ hypersurfaces as zero loci of sections of $\mathcal O(d)$}

Let
\[
F\in k[x_0,\dots,x_n]
\]
be homogeneous of degree $d\ge 1$.
Then $F$ defines a global section of
\[
\mathcal O_{\mathbb P^n}(d),
\qquad
\mathcal O_{\mathbb P^n}(d)=\mathcal O_{\mathbb P^n}(1)^{\otimes d},
\]
and its zero locus is the degree-$d$ hypersurface
\[
Z(F)\subset \mathbb P^n.
\]
On the chart $U_i$ the section is represented by the regular function
\[
\frac{F}{x_i^{d}}.
\]

\subsubsection*{A principal divisor: why hyperplanes are linearly equivalent}

Consider the rational function on $\mathbb P^n$
\[
f=\frac{x_0}{x_1}.
\]
Its divisor of zeros and poles is
\[
\mathrm{div}(f)=Z(x_0)-Z(x_1).
\]
Hence
\[
Z(x_0)\sim Z(x_1),
\]
so all hyperplanes have the same divisor class in $\mathrm{Pic}(\mathbb P^n)$, corresponding to $\mathcal O(1)$.

\subsection*{Concrete projective bundle example via polynomials: blow-up of $\mathbb P^2$ at a point}

Let
\[
p=(0:0:1)\in \mathbb P^2.
\]
The blow-up of $\mathbb P^2$ at $p$ can be realized as a closed subvariety of $\mathbb P^2\times \mathbb P^1$:
\[
\widetilde{\mathbb P^2}
=
\bigl\{((x:y:z),(s:t))\in \mathbb P^2\times\mathbb P^1 \mid xt-ys=0\bigr\}.
\]
The projection
\[
\pi:\widetilde{\mathbb P^2}\to \mathbb P^2
\]
is the blow-down map.
The exceptional divisor is
\[
E=\pi^{-1}(p)=\{((0:0:1),(s:t))\}\cong \mathbb P^1.
\]

The second projection
\[
\rho:\widetilde{\mathbb P^2}\to \mathbb P^1,
\qquad
((x:y:z),(s:t))\longmapsto (s:t),
\]
exhibits $\widetilde{\mathbb P^2}$ as a $\mathbb P^1$-bundle over $\mathbb P^1$.
Indeed, for fixed $(s:t)$ the fiber is
\[
\rho^{-1}(s:t)
=
\{(x:y:z)\in \mathbb P^2 \mid tx-sy=0\},
\]
which is a projective line in $\mathbb P^2$, hence isomorphic to $\mathbb P^1$.
