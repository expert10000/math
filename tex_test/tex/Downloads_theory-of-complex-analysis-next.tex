	\section*{Principal-branch integral of $\Log^2$ along a curve from $i$ to $-i$}
	
	\paragraph{Problem.}
	Let $\gamma:[0,1]\to\mathbb C$ be a $C^1$ curve with $\gamma(0)=i$, $\gamma(1)=-i$,
	such that $\gamma$ does not intersect $(-\infty,0]$. Compute
	\[
	\int_{\gamma}\Log^2(z)\,dz,
	\]
	where $\Log$ denotes the principal branch.
	
	\paragraph{Background.}
	We use $\Log z=\ln|z|+i\Arg z$ with branch cut $(-\infty,0]$ and $\Arg z\in(-\pi,\pi)$.
	On any simply connected set avoiding the cut, $\Log^2 z$ is holomorphic, hence the integral
	is path independent. Seek an antiderivative of the form $F(z)=z\,G(\Log z)$. Then
	\[
	F'(z)=G(\Log z)+G'(\Log z)=\Log^2 z
	\iff
	G'(w)+G(w)=w^2.
	\]
	A polynomial solution is $G(w)=w^2-2w+2$, since $(2w-2)+(w^2-2w+2)=w^2$.
	Thus
	\[
	F(z)=z\big(\Log^2 z-2\Log z+2\big),\qquad F'(z)=\Log^2 z.
	\]
	
	\paragraph{Evaluation.}
	Using $\Log(i)=i\frac{\pi}{2}$, $\Log(-i)=-i\frac{\pi}{2}$,
	\[
	\begin{aligned}
		F(i)&=i\!\left(\left(i\frac{\pi}{2}\right)^{2}-2\left(i\frac{\pi}{2}\right)+2\right)
		=-\frac{i\pi^{2}}{4}+\pi+2i,\\[2mm]
		F(-i)&=-i\!\left(\left(-i\frac{\pi}{2}\right)^{2}-2\left(-i\frac{\pi}{2}\right)+2\right)
		=\frac{i\pi^{2}}{4}+\pi-2i.
	\end{aligned}
	\]
	Therefore
	\[
	\boxed{\ \displaystyle \int_{\gamma}\Log^{2}z\,dz
		=F(-i)-F(i)= i\!\left(\frac{\pi^{2}}{2}-4\right). \ }
	\]
	
	\paragraph{Remark.}
	More generally, for $n\in\mathbb N$ one may integrate $\Log^{n}z$ by solving
	$G'+G=w^{n}$ and taking $F(z)=z\,G(\Log z)$.
	
	
\section*{Integral of $e^{e^{it}}$ over one period}

\paragraph{Problem.}
Compute $\displaystyle \int_{0}^{2\pi} e^{\,e^{it}}\,dt$.

\paragraph{Path method.}
Let $z=e^{it}$ so that $|z|=1$ and $dt=\frac{dz}{iz}$. Then
\[
\int_{0}^{2\pi} e^{e^{it}}\,dt
=\frac1i\oint_{|z|=1}\frac{e^{z}}{z}\,dz.
\]
By Cauchy’s integral formula for $f(z)=e^{z}$,
\[
\oint_{|z|=1}\frac{e^{z}}{z}\,dz=2\pi i\,f(0)=2\pi i.
\]
Hence
\[
\boxed{\displaystyle \int_{0}^{2\pi} e^{\,e^{it}}\,dt=2\pi }.
\]

\paragraph{Remark (series check).}
Since $e^{e^{it}}=\sum_{n=0}^\infty \frac{e^{int}}{n!}$, the integral over $[0,2\pi]$
kills all terms with $n\neq0$ and equals $2\pi$.


\section*{Cauchy’s Theorem from Green’s Theorem}

\paragraph{Setup.}
Let $C$ be a positively oriented simple closed $C^1$ curve bounding $D\subset\mathbb R^2$.
Assume $f=u+iv$ is holomorphic on an open set containing $\overline D$ and that
$u_x,u_y,v_x,v_y$ are continuous on $D$.

\paragraph{Tools.}
Green’s theorem: $\displaystyle \int_C (L\,dx+M\,dy)=\iint_D (M_x-L_y)\,dx\,dy$.
For holomorphic $f=u+iv$, the Cauchy--Riemann equations hold:
$u_x=v_y$, $u_y=-v_x$.
Also $dz=dx+i\,dy$, hence
\[
\int_C f\,dz=\int_C (u\,dx-v\,dy)+i\int_C (v\,dx+u\,dy).
\]

\paragraph{Proof.}
Apply Green to each real 1--form.

\emph{Real part:} with $L=u$, $M=-v$,
\[
\int_C (u\,dx-v\,dy)=\iint_D\big[(-v)_x-u_y\big]\,dx\,dy
=\iint_D (-v_x-u_y)\,dx\,dy=0
\]
since $u_y=-v_x$.

\emph{Imaginary part:} with $L=v$, $M=u$,
\[
\int_C (v\,dx+u\,dy)=\iint_D (u_x-v_y)\,dx\,dy=0
\]
since $u_x=v_y$.

Therefore $\displaystyle \int_C f(z)\,dz=0$.

\paragraph{Bonus (why not this proof?).}
This argument requires $C^1$ regularity of $u,v$ and invokes Green’s theorem from vector calculus.
In complex analysis we prefer Goursat’s approach, which proves Cauchy’s theorem without assuming
continuity of partial derivatives (only complex differentiability), and it applies to piecewise
$C^1$ curves and more general domains. It is thus strictly more general and self-contained within
complex analysis.

\section*{Cauchy’s Theorem from Green’s Theorem — extended background}

\paragraph{Holomorphic integrand as a 1--form.}
For $z=x+iy$ and $f=u+iv$,
\[
f\,dz=(u\,dx-v\,dy)+i\,(v\,dx+u\,dy),
\]
so $\displaystyle \int_C f\,dz$ splits into two real line integrals.

\paragraph{Green as Stokes in 2D.}
Green’s theorem is the special case of Stokes for 1--forms:
\[
\oint_{\partial D}(L\,dx+M\,dy)=\iint_D (M_x-L_y)\,dx\wedge dy
=\iint_D d(L\,dx+M\,dy).
\]
Thus a closed 1--form (one with vanishing exterior derivative) integrates to $0$ on any closed curve.

\paragraph{CR $\Rightarrow$ closedness.}
If $f$ is holomorphic, the Cauchy--Riemann equations hold:
$u_x=v_y$ and $u_y=-v_x$. Then
\[
d(u\,dx-v\,dy)=(-v_x-u_y)\,dx\wedge dy=0,
\qquad
d(v\,dx+u\,dy)=(u_x-v_y)\,dx\wedge dy=0.
\]
Applying Green to each form gives
\[
\int_C (u\,dx-v\,dy)=0,
\qquad
\int_C (v\,dx+u\,dy)=0,
\]
hence $\displaystyle \int_C f\,dz=0$.

\paragraph{Complex-coordinate proof (optional).}
With $dz=dx+i\,dy$, $d\bar z=dx-i\,dy$ and
$\partial_{\bar z}=\tfrac12(\partial_x+i\partial_y)$,
one has
\[
d(f\,dz)=(\partial_{\bar z} f)\, d\bar z\wedge dz.
\]
If $f$ is holomorphic, $\partial_{\bar z}f=0$, so by Stokes
\[
\oint_{\partial D} f\,dz=\iint_D d(f\,dz)=0.
\]

\paragraph{Why not use this as the main proof?}
This argument assumes $u,v\in C^1$ and invokes Green’s theorem.
Goursat’s theorem proves Cauchy’s theorem without any continuity of
partials, for piecewise $C^1$ curves, and stays within complex methods,
which is both more general and conceptually central to the theory.

\section*{Green’s Theorem: deeper background and the ``calc'' to Cauchy}

\paragraph{Green/Stokes in several forms.}
Let $D\subset\mathbb R^2$ have positively oriented, piecewise $C^1$ boundary $\partial D$ and let
$L,M\in C^1(\overline D)$. Then
\[
\oint_{\partial D} (L\,dx+M\,dy)
=
\iint_D \big(M_x-L_y\big)\,dx\,dy.
\]
Equivalently, for the $1$--form $\alpha=L\,dx+M\,dy$,
\[
\oint_{\partial D}\alpha=\iint_D d\alpha,\qquad
d\alpha=(M_x-L_y)\,dx\wedge dy.
\]
If $D$ has holes, orient the outer boundary counterclockwise and each inner boundary clockwise;
the sum of the boundary integrals equals the same area integral.

\paragraph{Regularity.}
We assume $L,M\in C^1$ and piecewise $C^1$ boundary; in our application this will follow from
$u_x,u_y,v_x,v_y$ being continuous.

\paragraph{Complex $1$--forms and $\partial_{\bar z}$.}
With $z=x+iy$, $dz=dx+i\,dy$, $d\bar z=dx-i\,dy$, $\partial_{\bar z}=\tfrac12(\partial_x+i\partial_y)$,
\[
d(f\,dz)=(\partial_{\bar z}f)\,d\bar z\wedge dz.
\]
Thus $d(f\,dz)=0$ iff $\partial_{\bar z}f=0$, i.e. $f$ is holomorphic (CR).

\paragraph{Cauchy from Green: the calculation.}
Let $f=u+iv$ be holomorphic with $u_x,u_y,v_x,v_y$ continuous on $\overline D$. Since
\[
\int_{\partial D} f\,dz
=\int_{\partial D} (u\,dx-v\,dy)
+i\int_{\partial D} (v\,dx+u\,dy),
\]
apply Green twice:

\emph{Real part:} with $L=u$, $M=-v$,
\[
\int_{\partial D} (u\,dx-v\,dy)
=\iint_D (-v_x-u_y)\,dx\,dy=0
\quad(\text{since }u_y=-v_x).
\]

\emph{Imaginary part:} with $L=v$, $M=u$,
\[
\int_{\partial D} (v\,dx+u\,dy)
=\iint_D (u_x-v_y)\,dx\,dy=0
\quad(\text{since }u_x=v_y).
\]

Therefore
\[
\boxed{\ \displaystyle \int_{\partial D} f(z)\,dz=0.\ }
\]

\paragraph{Remarks.}
On multiply connected regions, the sum of boundary integrals (with orientations) vanishes.
Goursat’s theorem proves the same result without assuming the continuity of the first partials,
which is why it is preferred as the foundational proof in complex analysis.

\section*{Practical 1-forms and calculations (plane and beyond)}

\subsection*{A. Plane examples}

\paragraph{A1. Area as a line integral.}
For $\alpha = x\,dy$ or $\tilde\alpha=\tfrac12(x\,dy-y\,dx)$,
\[
d\alpha=d\tilde\alpha=dx\wedge dy,\qquad
\oint_{\partial D} x\,dy = \iint_D 1\,dx\,dy = \operatorname{Area}(D),
\quad
\oint_{\partial D} \tfrac12(x\,dy-y\,dx) = \operatorname{Area}(D).
\]
\emph{Ellipse:} For $C:\, x^2/a^2+y^2/b^2=1$, $\displaystyle \oint_C \tfrac12(x\,dy-y\,dx)=\pi ab$.

\paragraph{A2. Polynomial field via Green.}
Let $\omega=(x^2-y)\,dx+(x+y^2)\,dy$. Then
\[
d\omega=(x+y^2)_x-(x^2-y)_y=2.
\]
On the unit disk, $\displaystyle \oint_{x^2+y^2=1} \omega = \iint_{x^2+y^2\le1} 2\,dx\,dy=2\pi$.

\paragraph{A3. Closed vs exact.}
A 1-form $\omega=P\,dx+Q\,dy$ is closed if $P_y=Q_x$. On simply connected domains, closed $\Rightarrow$ exact.

\emph{Exact:} $\omega=2x\,dx+2y\,dy=d(x^2+y^2)$, hence $\oint\omega=0$.

\emph{Closed but not exact (punctured plane):}
\[
\eta=\frac{-y}{x^2+y^2}\,dx+\frac{x}{x^2+y^2}\,dy,\qquad d\eta=0 \text{ on }\mathbb R^2\!\setminus\!\{0\}.
\]
Yet $\displaystyle \oint_{x^2+y^2=1} \eta = 2\pi$.

\subsection*{B. Complex-coordinate viewpoint}

\paragraph{B1. $dz$, $d\bar z$ and $\partial_{\bar z}$.}
With $z=x+iy$, $dz=dx+i\,dy$, $d\bar z=dx-i\,dy$ and $\partial_{\bar z}=\tfrac12(\partial_x+i\partial_y)$,
\[
d(f\,dz)=(\partial_{\bar z}f)\,d\bar z\wedge dz.
\]
If $f$ is holomorphic, then $d(f\,dz)=0$ and $\oint_{\partial D} f\,dz=0$.

\paragraph{B2. Singularities (glimpse of residues).}
For $f(z)=\frac{1}{z-a}$ and $C$ a small CCW circle about $a$,
\[
\oint_C f(z)\,dz=2\pi i.
\]

\paragraph{B3. Multiply connected domains.}
If $D$ is an annulus with CCW outer boundary $C_{\mathrm{out}}$ and clockwise inner boundary $C_{\mathrm{in}}$,
then for holomorphic $f$ on $D$,
\[
\oint_{C_{\mathrm{out}}} f\,dz+\oint_{C_{\mathrm{in}}} f\,dz=0.
\]

\subsection*{C. Polygon area (shoelace via Green).}
For vertices $(x_k,y_k)$ in CCW order and $(x_{n+1},y_{n+1})=(x_1,y_1)$,
\[
\operatorname{Area}(D)=\oint_{\partial D} x\,dy
=\frac12\sum_{k=1}^n \big(x_k y_{k+1}-x_{k+1}y_k\big).
\]

\paragraph{Tips.}
To evaluate $\oint_C (P\,dx+Q\,dy)$: check $M_x-L_y$. If simple, use Green; if $0$, try a potential or inspect topology (holes).
In complex settings, convert to $\oint \frac{f(z)}{z-a}\,dz$ when possible for Cauchy/Residue; otherwise Cauchy’s theorem gives $0$.

\section*{A) Warm-ups on the plane}

\subsection*{A1) Area from a line integral}

Take the $1$-form
\[
\alpha = x\,dy,
\]
or the symmetrized area form
\[
\tilde\alpha = \tfrac{1}{2}\,\bigl(x\,dy - y\,dx\bigr).
\]

Then
\[
d\alpha = dx \wedge dy,
\]
and
\[
d\tilde\alpha = dx \wedge dy.
\]

By Green's theorem,
\[
\oint_{\partial D} x\,dy \;=\; \iint_{D} 1\,dx\,dy,
\]
hence
\[
\oint_{\partial D} x\,dy \;=\; \operatorname{Area}(D).
\]

Also,
\[
\oint_{\partial D} \tfrac{1}{2}\,\bigl(x\,dy - y\,dx\bigr) \;=\; \iint_{D} dx \wedge dy,
\]
so
\[
\oint_{\partial D} \tfrac{1}{2}\,\bigl(x\,dy - y\,dx\bigr) \;=\; \operatorname{Area}(D).
\]

\paragraph{Example (ellipse).}
For the ellipse
\[
C:\quad \frac{x^{2}}{a^{2}} + \frac{y^{2}}{b^{2}} = 1,
\]
we have
\[
\oint_{C} \tfrac{1}{2}\,\bigl(x\,dy - y\,dx\bigr) \;=\; \pi a b.
\]
(You can parametrize $x=a\cos t$, $y=b\sin t$ and compute directly, or just invoke Green's theorem.)

\subsection*{A2) Circulation and flux of a polynomial vector field}

Let
\[
\omega \;=\; (x^{2} - y)\,dx \;+\; (x + y^{2})\,dy.
\]

Then
\[
d\omega \;=\; \bigl(\partial_{x}(x + y^{2}) - \partial_{y}(x^{2} - y)\bigr)\,dx\wedge dy,
\]
so
\[
d\omega \;=\; \bigl(1 - (-1)\bigr)\,dx\wedge dy \;=\; 2\,dx\wedge dy.
\]

Over the unit circle $C: x^{2} + y^{2} = 1$ (and unit disk),
\[
\oint_{C} \omega \;=\; \iint_{x^{2} + y^{2} \le 1} 2\,dx\,dy,
\]
hence
\[
\oint_{C} \omega \;=\; 2\,\pi.
\]

\subsection*{A3) Exact vs.\ closed: quick tests}

For a $1$-form $\omega = P\,dx + Q\,dy$,
\[
\omega \ \text{is closed} \iff P_{y} = Q_{x}.
\]
If $D$ is simply connected, then
\[
\text{closed} \;\Rightarrow\; \text{exact}, \quad \text{i.e.}\ \exists\,\phi\ \text{with}\ \omega = d\phi.
\]

\paragraph{Exact example.}
\[
\omega \;=\; 2x\,dx + 2y\,dy \;=\; d(x^{2} + y^{2}).
\]
Therefore any loop integral satisfies
\[
\oint \omega \;=\; 0,
\]
either by Green (since $d\omega = 0$) or by the fundamental theorem for line integrals.

\paragraph{Closed but not exact on the punctured plane.}
Define
\[
\eta \;=\; \frac{-y}{x^{2} + y^{2}}\,dx \;+\; \frac{x}{x^{2} + y^{2}}\,dy.
\]
Then
\[
d\eta \;=\; 0 \quad \text{on } \mathbb{R}^{2}\setminus\{0\},
\]
but on the unit circle $|z|=1$,
\[
\oint_{|z|=1} \eta \;=\; 2\pi.
\]
This nonzero loop integral reflects the topological obstruction (the ``hole'' at the origin). It is the real-variable analogue of
\[
\oint \frac{1}{z}\,dz \;=\; 2\pi i.
\]

\section*{B) Complex-coordinate computations}

\subsection*{B1) Using $dz$, $d\bar z$ and $\partial_{\bar z}$}

For smooth $f$,
\[
d\bigl(f\,dz\bigr) \;=\; \bigl(\partial_{\bar z} f\bigr)\, d\bar z \wedge dz.
\]

If $f$ is holomorphic, that is $\partial_{\bar z} f = 0$, then the $1$-form $f\,dz$ is closed, and by Stokes/Green we have
\[
\oint_{\partial D} f\,dz \;=\; 0,
\]
whenever $f$ is holomorphic on a region containing $D$.

\paragraph{Example.}
Let $f(z)=z^{2}$ and $C$ any closed $C^{1}$ loop avoiding singularities (there are none). Then
\[
\oint_{C} z^{2}\,dz \;=\; 0.
\]

\subsection*{B2) Holomorphic with an isolated singularity (residues peek)}

Let
\[
f(z) \;=\; \frac{1}{z - a},
\]
and let $C$ be a positively oriented circle centered at $a$. Then
\[
\oint_{C} f(z)\,dz \;=\; 2\pi i.
\]

In real-form language, the real part of $\frac{1}{z}\,dz$ is exactly the $\eta$ from A3:
\[
\Re\!\left(\frac{1}{z}\,dz\right) \;=\; \eta,
\]
while
\[
\Im\!\left(\frac{1}{z}\,dz\right) \;=\; -\,\frac{x}{x^{2}+y^{2}}\,dx \;-\; \frac{y}{x^{2}+y^{2}}\,dy.
\]
Together these give the complex integral $2\pi i$.

\subsection*{B3) Multiply connected domain and orientations}

Let $D$ be an annulus with outer boundary $C_{\text{out}}$ oriented counterclockwise and inner boundary $C_{\text{in}}$ oriented clockwise. If $f$ is holomorphic on $D$, then
\[
\oint_{C_{\text{out}}} f\,dz \;+\; \oint_{C_{\text{in}}} f\,dz \;=\; 0.
\]

For example, for $f(z)=z$ each integral is zero. For $f(z)=1/z$ the above identity does not apply (there is a singularity inside), and each circle integral equals $\pm 2\pi i$ according to the orientation.

\section*{C) Quick polygon area (shoelace via Green)}

Given a simple polygon with vertices $(x_{k},y_{k})$ in counterclockwise order (for $k=1,\dots,n$), and setting $(x_{n+1},y_{n+1})=(x_{1},y_{1})$, the area can be written as
\[
\operatorname{Area}(D) \;=\; \oint_{\partial D} x\,dy,
\]
and also as the shoelace sum
\[
\operatorname{Area}(D) \;=\; \frac{1}{2}\,\sum_{k=1}^{n} \Bigl(x_{k}\,y_{k+1} \;-\; x_{k+1}\,y_{k}\Bigr).
\]
This is just Green's theorem with the choice $\alpha = x\,dy$.

\section*{Further tips}

To decide a line integral
\[
\oint_{C} \bigl(P\,dx + Q\,dy\bigr),
\]
first compute
\[
M_{x} - L_{y} \;=\; \partial_{x}Q \;-\; \partial_{y}P.
\]
If $M_{x} - L_{y}$ is constant or simple, use Green's theorem immediately. If $M_{x} - L_{y}=0$, try to find a potential $\phi$ (exactness) or check the topology of the domain for holes that can obstruct exactness.

In complex problems, it is often useful to rewrite integrals in the Cauchy/Residue style
\[
\frac{1}{i}\,\oint \frac{f(z)}{z-a}\,dz,
\]
and apply Cauchy’s integral formula or the residue theorem when singularities are present. If there are no singularities in the region bounded by the curve, then by Cauchy’s theorem
\[
\oint f(z)\,dz \;=\; 0.
\]


\section*{Goal}
Explain why
\[
\operatorname{Area}(D)
\;=\;
\frac{1}{2}\,\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr),
\]
including three independent derivations:
(1) averaging two Green identities,
(2) an exterior-calculus choice of a $1$-form with \(d\alpha = dx\wedge dy\),
(3) a parametrization check leading to the shoelace formula.

Throughout, \(D\subset\mathbb R^2\) is a region with positively oriented (counterclockwise) boundary \(\partial D\).

\section*{1) Green’s theorem by ``averaging''}

Green’s theorem states that for smooth \(L(x,y),M(x,y)\),
\[
\oint_{\partial D}\!\bigl(L\,dx + M\,dy\bigr)
\;=\;
\iint_{D}\!\bigl(M_x - L_y\bigr)\,dx\,dy.
\]

\subsection*{First choice \((L,M)=(0,x)\)}
Then
\[
M_x - L_y \;=\; \partial_x(x) - \partial_y(0) \;=\; 1,
\]
so
\[
\oint_{\partial D} x\,dy
\;=\;
\iint_D 1\,dx\,dy
\;=\;
\operatorname{Area}(D).
\]

\subsection*{Second choice \((L,M)=(-y,0)\)}
Then
\[
M_x - L_y \;=\; \partial_x(0) - \partial_y(-y) \;=\; 1,
\]
hence
\[
-\;\oint_{\partial D} y\,dx
\;=\;
\iint_D 1\,dx\,dy
\;=\;
\operatorname{Area}(D).
\]

\subsection*{Average the two identities}
Add the last two equalities and divide by \(2\):
\[
\operatorname{Area}(D)
\;=\;
\frac12\,\oint_{\partial D} x\,dy
\;+\;
\frac12\,\Bigl(-\oint_{\partial D} y\,dx\Bigr)
\;=\;
\frac12\,\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr).
\]

\paragraph{Interpretation.}
The factor \(\tfrac12\) is the average of two \emph{already correct} area-from-boundary formulas:
\(\;\displaystyle \oint x\,dy\) and \(-\displaystyle\oint y\,dx\).

\section*{2) Exterior calculus: pick \(\alpha\) with \(d\alpha=dx\wedge dy\)}

We seek a $1$-form \(\alpha\) of the form
\[
\alpha \;=\; a\,x\,dy \;+\; b\,(-y\,dx),
\]
with constants \(a,b\in\mathbb R\), such that \(d\alpha\) equals the area $2$-form \(dx\wedge dy\).

\subsection*{Compute \(d\alpha\)}
Using \(d(f\,\beta)=df\wedge\beta + f\,d\beta\), \(d(dx)=d(dy)=0\), and \(dy\wedge dx=-dx\wedge dy\):
\[
d\alpha
\;=\;
a\,dx\wedge dy
\;+\;
b\,\bigl(-\,dy\wedge dx\bigr)
\;=\;
(a+b)\,dx\wedge dy.
\]

\subsection*{Match the area $2$-form}
We require
\[
d\alpha \;=\; dx\wedge dy,
\qquad\Longleftrightarrow\qquad
a+b \;=\; 1.
\]
A particularly symmetric choice is \(a=b=\tfrac12\), yielding
\[
\alpha \;=\; \frac12\,(x\,dy - y\,dx),
\qquad
d\alpha \;=\; dx\wedge dy.
\]

\subsection*{Apply Stokes/Green}
By Stokes,
\[
\oint_{\partial D} \alpha
\;=\;
\iint_D d\alpha
\;=\;
\iint_D dx\wedge dy
\;=\;
\operatorname{Area}(D),
\]
i.e.
\[
\operatorname{Area}(D)
\;=\;
\frac12\,\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr).
\]

\paragraph{Interpretation.}
The factor \(\tfrac12\) comes from the symmetric coefficients \(a=b=\tfrac12\) that make
\(d\alpha = dx\wedge dy\).

\section*{3) Parametrization check and the shoelace formula}

Let the boundary \(C=\partial D\) be parametrized (CCW) by \(t\mapsto (x(t),y(t))\), \(t\in[a,b]\).
Then
\[
dx \;=\; x'(t)\,dt,
\qquad
dy \;=\; y'(t)\,dt.
\]
Therefore,
\[
\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr)
\;=\;
\int_{a}^{b}\!\bigl(x(t)\,y'(t) - y(t)\,x'(t)\bigr)\,dt.
\]
By Green (or a divergence argument), \(\tfrac12\) of this integral equals the area:
\[
\operatorname{Area}(D)
\;=\;
\frac12\,\int_{a}^{b}\!\bigl(x\,y' - y\,x'\bigr)\,dt
\;=\;
\frac12\,\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr).
\]

\subsection*{Polygonal boundary: the shoelace}
If \(C\) is a simple polygon with vertices \((x_k,y_k)\) in CCW order for \(k=1,\dots,n\), and
\((x_{n+1},y_{n+1})=(x_1,y_1)\), then
\[
\operatorname{Area}(D)
\;=\;
\frac12\,\sum_{k=1}^{n}
\Bigl(
x_k\,y_{k+1} \;-\; x_{k+1}\,y_k
\Bigr),
\]
which is exactly the discrete version of
\(\;\tfrac12\oint (x\,dy-y\,dx)\;\) (the ``shoelace formula'').

\paragraph{Geometric intuition for the factor \(\tfrac12\).}
Along each small edge element, the signed quantity \(x\,dy - y\,dx\) is the oriented area
of the infinitesimal parallelogram spanned by \((x,y)\) and \((dx,dy)\).
Each such parallelogram is twice the area of the corresponding ``triangle strip''
between the edge and the origin, hence the factor \(\tfrac12\).

\section*{Summary (which form to use?)}
For a CCW-oriented boundary,
\[
\boxed{%
	\operatorname{Area}(D)
	\;=\;
	\oint_{\partial D} x\,dy
	\;=\;
	-\;\oint_{\partial D} y\,dx
	\;=\;
	\frac12\,\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr).
}
\]
The symmetric form \(\;\tfrac12\oint(x\,dy - y\,dx)\;\) is rotation-invariant in look and
often numerically more stable (errors in the two terms tend to cancel).
	
	
\section*{Conventions}
We write a $1$-form on the plane as $\omega=L\,dx+M\,dy$. For a positively oriented (CCW) boundary
$C=\partial D$, Green's theorem says
\[
\oint_{C} (L\,dx+M\,dy)
\;=\;
\iint_{D} \bigl(\partial_x M - \partial_y L\bigr)\,dx\,dy.
\]
For complex line integrals on a circle, we often parametrize $z=Re^{it}$, $dz=iRe^{it}\,dt$.

\section*{1) Straight segment: direct parametrization}
\textbf{Compute} $\displaystyle \int_{C}(y\,dx+x\,dy)$ for the segment from $(0,0)$ to $(1,2)$.

\paragraph{Parametrization.}
Let $r(t)=(x(t),y(t))=(t,2t)$, $t\in[0,1]$. Then $dx=dt$ and $dy=2\,dt$.

\paragraph{Integrand.}
\[
y\,dx+x\,dy
=

=

\paragraph{Integral.}

\int_{C}(y\,dx+x\,dy)
=

=
2.

\section*{2) ``Angle'' $1$-form around a circle (gives $2\pi$)}

\begin{itemize}[leftmargin=1.25em]
	\item If the field is given as $\omega=L\,dx+M\,dy$, compute $M_x-L_y$.
	\begin{itemize}
		\item If $M_x-L_y$ is constant or simple, jump to Green's theorem.
		\item If $M_x-L_y=0$ on a simply connected domain, try a potential $\phi$ with $\omega=d\phi$ (exactness).
		\item If there is a hole (punctures), expect nonzero loop integrals even when $M_x-L_y=0$.
	\end{itemize}

x(t)=t,\qquad y(t)=2t,\qquad t\in[0,1].

\omega \;=\; y\,dx + x\,dy.

Since

the form \(\omega\) is exact with potential \(\phi(x,y)=xy\).

\section*{Line segment $C$, the $1$-form, and why the integral is ``work''}

\paragraph{What is $C$?}
$C$ is the path of integration. Here it is the straight line segment from $(0,0)$ to $(1,2)$.
A convenient parametrization is
\[
x(t)=t,\qquad y(t)=2t,\qquad t\in[0,1],
\]
so that the image is exactly the portion of the line $y=2x$ joining the two endpoints.
Orientation: increasing $t$ runs from $(0,0)$ to $(1,2)$.

\paragraph{What is the $1$-form?}
A (differential) $1$-form on the plane has the shape
\[
\omega \;=\; P(x,y)\,dx \;+\; Q(x,y)\,dy.
\]
Here
\[
\omega \;=\; y\,dx \;+\; x\,dy.
\]

\paragraph{Why does it represent ``work''?}
A vector field $\mathbf F=(P,Q)$ does work along a path $C$ given by the line integral
\[
\int_C \mathbf F\cdot d\mathbf r
\;=\;
\int_C \bigl(P\,dx + Q\,dy\bigr).
\]
Thus integrating the $1$-form $P\,dx+Q\,dy$ is the same as computing the mechanical work
of $\mathbf F=(P,Q)$. In our case $\mathbf F=(y,x)$, so
\[
\int_C (y\,dx+x\,dy) \;=\; \int_C \mathbf F\cdot d\mathbf r.
\]

\paragraph{Key simplification (exactness).}
Note that
\[
d(xy) \;=\; x\,dy + y\,dx \;=\; \omega,
\]
so $\omega$ is \emph{exact} with potential $\phi(x,y)=xy$ (i.e.\ $\omega=d\phi$).
By the fundamental theorem for line integrals,
\[
\int_C \omega \;=\; \phi(\text{end}) - \phi(\text{start}),
\]
which depends only on the endpoints, not on the particular path.

\paragraph{Evaluation for this $C$.}
With start $A=(0,0)$ and end $B=(1,2)$,
\[
\int_C (y\,dx+x\,dy)
\;=\;
\phi(B)-\phi(A)
\;=\;
(xy)\Big|_{(0,0)}^{(1,2)}
\;=\;
(1\cdot 2) - (0\cdot 0)
\;=\; 2.
\]
Therefore, the value $2$ would be the same for \emph{any} path from $(0,0)$ to $(1,2)$,
not just the straight segment.

%======================== Residue/Winding Numbers Problem ========================%
\section*{Problem}
For $\gamma(a,r)$ the positively oriented circle centered at $a$ of radius $r>0$, compute
\[
\int_{\gamma(0,1)} \frac{5z^{2}-8}{z^{3}-2z^{2}}\,dz.
\]
Then describe closed paths $\gamma$ for which $\displaystyle \int_{\gamma} f(z)\,dz$ equals
\textbf{(a)} $14\pi i$ and \textbf{(b)} $-2\pi i$, where
\[
f(z)=\frac{5z^{2}-8}{z^{3}-2z^{2}}.
\]

\section*{Background / Plan}
Factor the denominator:
\[
z^{3}-2z^{2}=z^{2}(z-2),
\]
so $f$ has an order-$2$ pole at $z=0$ and a simple pole at $z=2$.

Use partial fractions to read off residues. For any closed path $\gamma$,
\[
\int_{\gamma} f(z)\,dz
\;=\;
2\pi i\!\!\sum_{a\in\{0,2\}}\! n(\gamma,a)\,\operatorname{Res}(f;a),
\]
where $n(\gamma,a)$ is the winding number of $\gamma$ about $a$.

\section*{Partial Fractions and Residues}
Seek constants $A,B,C$ with
\[
\frac{5z^{2}-8}{z^{2}(z-2)}
\;=\;
\frac{A}{z-2}+\frac{B}{z}+\frac{C}{z^{2}}.
\]
Clearing denominators gives
\[
5z^{2}-8
=
A z^{2} + B z(z-2) + C(z-2)
=
(A+B)z^{2} + (-2B+C)z - 2C.
\]
Match coefficients:
\[
A+B=5,\qquad -2B+C=0,\qquad -2C=-8.
\]
Hence $C=4$, $B=2$, $A=3$, so
\[
f(z)=\frac{3}{z-2}+\frac{2}{z}+\frac{4}{z^{2}}.
\]
Therefore
\[
\operatorname{Res}(f;0)=2,
\qquad
\operatorname{Res}(f;2)=3.
\]
(The $4/z^{2}$ term contributes no residue at $0$.)

\section*{Integral over $\gamma(0,1)$}
The unit circle about the origin encloses $z=0$ but not $z=2$. Thus
\[
\int_{\gamma(0,1)} f(z)\,dz
=
2\pi i\cdot \operatorname{Res}(f;0)
=
2\pi i\cdot 2
=
\boxed{4\pi i}.
\]

\section*{Paths giving prescribed values}
For a general closed curve $\gamma$ with winding numbers
\[
n_0:=n(\gamma,0),\qquad n_2:=n(\gamma,2),
\]
the residue theorem gives
\[
\int_{\gamma} f(z)\,dz
=
2\pi i\,(2n_0+3n_2).
\]

\paragraph{(a) $14\pi i$.}
Require $2n_0+3n_2=7$. Examples (infinitely many):
\[
n_0=2,\ n_2=1
\quad\text{(two CCW loops around $0$ and one CCW loop around $2$),}
\]
\[
n_0=5,\ n_2=-1
\quad\text{(five CCW around $0$, one CW around $2$).}
\]

\paragraph{(b) $-2\pi i$.}
Require $2n_0+3n_2=-1$. Examples:
\[
n_0=1,\ n_2=-1
\quad\text{(one CCW around $0$, one CW around $2$),}
\]
\[
n_0=4,\ n_2=-3
\quad\text{(four CCW around $0$, three CW around $2$).}
\]
Any integer pair $(n_0,n_2)$ solving the linear Diophantine equation yields a valid $\gamma$
(e.g.\ a concatenation of small circles around $0$ and $2$ with the prescribed orientations).

%===================== What Green's Theorem Actually Says (LaTeX) =====================%
\section*{What Green's Theorem Actually Says}

Let $D\subset\mathbb{R}^2$ be a region with positively oriented (counterclockwise) boundary
$C=\partial D$, and let $L,M$ be $C^1$ functions on an open set containing $\overline{D}$.
Green's theorem states
\[
\oint_{C}\!\bigl(L\,dx+M\,dy\bigr)
\;=\;
\iint_{D}\!\bigl(M_{x}-L_{y}\bigr)\,dx\,dy.
\]
\emph{Interpretation:} the curve in the line integral must be \emph{the boundary of the region}
appearing in the double integral.

\bigskip

\paragraph{1) $D$ = unit disk, $C$ = its boundary (unit circle).}
Green applies directly:
\[
\oint_{|z|=1}\!\bigl(L\,dx+M\,dy\bigr)
\;=\;
\iint_{\{|z|\le 1\}}\!\bigl(M_{x}-L_{y}\bigr)\,dx\,dy.
\]

\bigskip

\paragraph{2) $D$ = unit disk, but $C$ is a larger circle (radius $R>1$).}
Here $C$ is \emph{not} the boundary of $D$. The boundary of the unit disk is the \emph{unit}
circle, not the larger one. To compare the two line integrals, use the multiply connected version
on the annulus
\[
A=\{\,1\le |z|\le R\,\}.
\]
Let $C_{1}$ be the unit circle oriented CCW and $C_{R}$ the circle of radius $R$ oriented CCW.
Then Stokes/Green on $A$ gives
\[
\oint_{C_{R}}\!\bigl(L\,dx+M\,dy\bigr)
\;-\;
\oint_{C_{1}}\!\bigl(L\,dx+M\,dy\bigr)
\;=\;
\iint_{A}\!\bigl(M_{x}-L_{y}\bigr)\,dx\,dy.
\]
\emph{Conclusion:} the two line integrals are equal iff the area integral over the annulus is zero—
e.g.\ if $M_{x}-L_{y}\equiv 0$ on $A$ (the $1$-form is closed there). Otherwise, they differ by
\[
\iint_{A}\!\bigl(M_{x}-L_{y}\bigr)\,dx\,dy,
\]
the integral of the curl over the region between the circles.

\bigskip

\paragraph{3) Complex-analysis corollary.}
Let $f=u+iv$ be holomorphic on a region containing the annulus between two closed curves
(with no singularities inside). The Cauchy–Riemann equations imply the $1$-forms
\[
u\,dx - v\,dy,
\qquad
v\,dx + u\,dy
\]
are \emph{closed}, i.e.\ their curls vanish:
\[
\partial_x(v)-\partial_y(u)=0,
\qquad
\partial_x(u)+\partial_y(v)=0.
\]
Hence their line integrals are the same on any two homologous curves (e.g.\ the unit circle and a
larger circle) provided the region between them contains no singularities of $f$. This is the
path-independence you use when applying Cauchy’s theorem.

%===================== Green in 2D: what's what? (LaTeX) =====================%
\section*{Green in 2D: what's what?}

\paragraph{Region $D$.}
The (closed) disk of radius $R$ centered at the origin:
\[
D \;=\; \{(x,y)\in\mathbb{R}^2 : x^2+y^2 \le R^2\}.
\]

\paragraph{Boundary $C=\partial D$.}
The positively oriented circle:
\[
C:\quad x=R\cos t,\ \ y=R\sin t,\qquad t\in[0,2\pi].
\]

\paragraph{Vector field / function.}
Let $\mathbf F=(L,M)$ with $L=L(x,y)$, $M=M(x,y)$.

\paragraph{Associated $1$-form.}
\[
\alpha \;=\; L\,dx + M\,dy.
\]

\paragraph{Green’s theorem.}
\[
\underbrace{\oint_{C}\bigl(L\,dx+M\,dy\bigr)}_{\text{line integral (circulation)}}
\;=\;
\underbrace{\iint_{D}\bigl(M_{x}-L_{y}\bigr)\,dx\,dy}_{\text{scalar curl (``rotation'')}}.
\]

\subsection*{What each side measures}
\[
\oint_{C}\bigl(L\,dx+M\,dy\bigr)
\;=\;
\int_{C} \mathbf F\cdot \mathbf T\,ds
\quad\text{(total tangential circulation/work along $C$),}
\]
\[
\iint_{D}\bigl(M_{x}-L_{y}\bigr)\,dA
\quad\text{(total vorticity / scalar curl over the interior).}
\]
\emph{Green says: circulation around the edge equals total curl inside.}

%--------------------------------------------------------------------%
\subsection*{Example 1 — constant curl, easy numbers}
Take $\mathbf F=(-y,x)$. Then $L=-y$, $M=x$ and
\[
M_x - L_y \;=\; \partial_x(x) - \partial_y(-y) \;=\; 1 - (-1) \;=\; 2 \quad(\text{constant}).
\]
Area integral:
\[
\iint_{D} (M_x-L_y)\,dA
\;=\;
\iint_{D} 2\,dA
\;=\; 2\,\pi R^2.
\]
Line integral (check by parametrizing $C$):
\[
dx=-R\sin t\,dt,\qquad dy=R\cos t\,dt,
\]
\[
L\,dx+M\,dy
=
(-y)\,dx + (x)\,dy
=
(-R\sin t)(-R\sin t)\,dt + (R\cos t)(R\cos t)\,dt
=
R^2\,dt,
\]
\[
\oint_{C} (L\,dx+M\,dy)
=
\int_{0}^{2\pi} R^2\,dt
=
2\pi R^2.
\]
Both sides match: $2\pi R^2$.

%--------------------------------------------------------------------%
\subsection*{Example 2 — gradient field (curl $=0$), circulation $=0$}
Let $\phi(x,y)=x^2+y^3$ and $\mathbf F=\nabla\phi=(2x,\,3y^2)$.
Then
\[
M_x - L_y
=
\partial_x(3y^2)-\partial_y(2x)
=
0-0
=
0,
\]
so
\[
\oint_{C} (2x\,dx+3y^2\,dy)
=
\iint_{D} 0\,dA
=
0.
\]
\emph{Interpretation:} exact/gradient fields do no net work around closed curves.

%--------------------------------------------------------------------%
\subsection*{Example 3 — area from a $1$-form (what $\tfrac12(x\,dy-y\,dx)$ measures)}
Set
\[
\alpha \;=\; \frac12\,(x\,dy - y\,dx).
\]
Then
\[
d\alpha
=
\frac12\,(dx\wedge dy + dy\wedge dx)
=
dx\wedge dy.
\]
Hence, by Stokes/Green,
\[
\oint_{C} \alpha
=
\iint_{D} dx\wedge dy
=
\operatorname{Area}(D)
=
\pi R^2.
\]
Therefore,
\[
\boxed{\ \frac12 \oint_{C} (x\,dy - y\,dx)\;=\;\operatorname{Area}(D)\;=\;\pi R^2\ }.
\]

%--------------------------------------------------------------------%
\subsection*{Takeaways}
\begin{itemize}
	\item Pick $\mathbf F=(L,M)$ $\Rightarrow$ $1$-form $L\,dx+M\,dy$.
	\item Circulation (line integral on $C$) equals total curl over $D$.
	\item If $\;M_x-L_y\equiv 0$ on $D$, then $\displaystyle \oint_{C}(L\,dx+M\,dy)=0$ (path-independence on simply connected $D$).
	\item The special $1$-form $\tfrac12(x\,dy - y\,dx)$ integrates to the enclosed area.
\end{itemize}

%===================== Problem: Mixed Complex/Real Integral via Cauchy =====================%
\section*{Problem}
\[
I=\oint_{\gamma(0,1)} \frac{\Re z}{\,2z - i\,}\,dz,
\qquad
J=\int_{0}^{2\pi} \frac{\cos^{2}\theta}{\,5-4\sin\theta\,}\,d\theta.
\]
Evaluate $I$ using only Cauchy’s Integral Formula. Then put $z=e^{i\theta}$ on $|z|=1$ to relate $I$ and $J$ and determine $J$.

\paragraph{Background idea.}
$\Re z$ is not holomorphic, but on $|z|=1$ we may use
\[
\Re z=\frac12\,(z+\bar z)=\frac12\,(z+z^{-1}),
\]
so the integrand becomes a sum of rational (meromorphic) terms and Cauchy/Residue applies.

%------------------------------------------------------------------------------------------%
\subsection*{Step 1 — Compute $I$ (Cauchy only)}
On $|z|=1$ write
\[
I=\frac12\oint_{|z|=1}\frac{z+z^{-1}}{\,2z-i\,}\,dz
=\frac12\left(\underbrace{\oint_{|z|=1}\frac{z}{\,2z-i\,}\,dz}_{=:I_1}
+\underbrace{\oint_{|z|=1}\frac{1}{\,z(2z-i)\,}\,dz}_{=:I_2}\right).
\]

\paragraph{First piece.}
\[
\frac{z}{\,2z-i\,}=\frac12\left(1+\frac{i}{\,2z-i\,}\right),
\qquad
\oint_{|z|=1} 1\,dz=0.
\]
Hence
\[
I_1=\frac{i}{2}\oint_{|z|=1}\frac{dz}{\,2z-i\,}.
\]
There is a simple pole at $z=\dfrac{i}{2}$ inside the unit circle, with residue $\dfrac{1}{2}$.
Therefore
\[
I_1=\frac{i}{2}\cdot 2\pi i\cdot \frac{1}{2}=\frac{i\cdot 2\pi i}{4}=-\frac{\pi}{2}.
\]

\paragraph{Second piece.}
Partial fractions:
\[
\frac{1}{z(2z-i)}=\frac{A}{z}+\frac{B}{2z-i},
\qquad
1=A(2z-i)+Bz.
\]
Equating coefficients gives $A=i$, $B=-2i$. Thus
\[
I_2
=i\oint_{|z|=1}\frac{dz}{z}
-2i\oint_{|z|=1}\frac{dz}{\,2z-i\,}
=
i(2\pi i)-2i\bigl(2\pi i\cdot \tfrac12\bigr)
=
-2\pi+2\pi=0.
\]

\paragraph{Conclusion for $I$.}
\[
I=\frac12\,(I_1+I_2)=\frac12\left(-\frac{\pi}{2}+0\right)=\boxed{-\frac{\pi}{4}}.
\]

%------------------------------------------------------------------------------------------%
\subsection*{Step 2 — Express $I$ as a $\theta$–integral and read off $J$}
On $|z|=1$ set $z=e^{i\theta}$, $dz=i e^{i\theta}\,d\theta$, and $\Re z=\cos\theta$:
\[
I=\int_{0}^{2\pi} \frac{\cos\theta\; i e^{i\theta}}{\,2e^{i\theta}-i\,}\,d\theta.
\]
Multiply numerator and denominator by the conjugate along the circle, $2e^{-i\theta}+i$:
\[
\frac{i e^{i\theta}}{\,2e^{i\theta}-i\,}
=
\frac{2i - e^{i\theta}}{\,5+4i\sin\theta\,},
\]
so
\[
I
=
\int_{0}^{2\pi}\frac{\cos\theta\,\bigl(2i-e^{i\theta}\bigr)}{\,5+4i\sin\theta\,}\,d\theta
=
\int_{0}^{2\pi}\frac{-\cos^{2}\theta}{\,5-4\sin\theta\,}\,d\theta
\quad(\text{the imaginary part integrates to }0 \text{ over a full period}).
\]
Therefore
\[
I=-\int_{0}^{2\pi}\frac{\cos^{2}\theta}{\,5-4\sin\theta\,}\,d\theta
\;=\;
-\;J.
\]

\paragraph{Final values.}
From Step 1, $I=-\dfrac{\pi}{4}$, hence
\[
\boxed{J=\frac{\pi}{4}}.
\]

%===================== The little engine behind "area via boundary integral" =====================%
\section*{The 1-form and its exterior derivative}

Start with the $1$-form
\[
\alpha \;=\; \frac12\,(x\,dy - y\,dx).
\]

\paragraph{Rules to use.}
\begin{align*}
	\text{(Product rule)}\qquad & d(f\,\beta) \;=\; df\wedge \beta \;+\; f\,d\beta,\\[2pt]
	\text{(Nilpotence)}\qquad & d^2=0 \ \Rightarrow\ d(dx)=d(dy)=0,\\[2pt]
	\text{(Antisymmetry)}\qquad & dy\wedge dx \;=\; -\,dx\wedge dy,\quad dx\wedge dx=0,\ dy\wedge dy=0.
\end{align*}

\paragraph{Compute $d\alpha$ step by step.}
\begin{align*}
	d\alpha
	&= \frac12\bigl(d(x\,dy) - d(y\,dx)\bigr)\\[4pt]
	&= \frac12\Bigl( dx\wedge dy + x\,d(dy) \;-\; dy\wedge dx - y\,d(dx)\Bigr) \quad \text{(product rule)}\\[4pt]
	&= \frac12\bigl(dx\wedge dy - dy\wedge dx\bigr) \qquad\qquad\qquad\ \ \ \ \text{(since $d(dy)=d(dx)=0$)}\\[4pt]
	&= \frac12\bigl(dx\wedge dy - (-dx\wedge dy)\bigr) \quad \text{(use $dy\wedge dx=-dx\wedge dy$)}\\[4pt]
	&= \frac12\bigl(dx\wedge dy + dx\wedge dy\bigr)\\[4pt]
	&= dx\wedge dy.
\end{align*}

Thus $d\alpha = dx\wedge dy$, the \emph{area $2$-form} on the plane.

\paragraph{Stokes/Green gives area from a boundary integral.}
For a region $D$ with positively oriented boundary $\partial D$,
\[
\oint_{\partial D} \alpha
\;=\;
\iint_D d\alpha
\;=\;
\iint_D dx\wedge dy
\;=\;
\operatorname{Area}(D),
\]
i.e.
\[
\boxed{\ \operatorname{Area}(D)\;=\;\frac12\;\oint_{\partial D}\!\bigl(x\,dy - y\,dx\bigr)\ }.
\]

\paragraph{Geometric meaning.}
The $1$-form $\alpha=\tfrac12(x\,dy-y\,dx)$ is a symmetric “area-measuring” form on the boundary.
The identity $d\alpha=dx\wedge dy$ says its infinitesimal curl equals unit area density. Integrating
$d\alpha$ over $D$ gives area; Stokes moves this to the boundary integral of $\alpha$.

%===================== Area via Boundary Integral: Worked Examples =====================%
\section*{Area via \(\displaystyle \frac12\oint_{\partial D}(x\,dy-y\,dx)\): Worked Examples}

We use
\[
\operatorname{Area}(D)=\frac{1}{2}\oint_{\partial D} \bigl(x\,dy - y\,dx\bigr).
\]

%------------------------------------------------------------------------%
\subsection*{Example 1 — Disk of radius \(R\)}
Parametrize the boundary circle \(C\):
\[
x=R\cos t,\qquad y=R\sin t,\qquad t\in[0,2\pi],
\]
so that
\[
dx=-R\sin t\,dt,\qquad dy=R\cos t\,dt.
\]
Then
\[
\frac12\bigl(x\,dy-y\,dx\bigr)
=
\frac12\Bigl(R\cos t\cdot R\cos t - R\sin t\cdot(-R\sin t)\Bigr)\,dt
=
\frac{R^{2}}{2}\,dt.
\]
Integrate:
\[
\operatorname{Area}
=
\int_{0}^{2\pi} \frac{R^{2}}{2}\,dt
=
\pi R^{2}.
\]

%------------------------------------------------------------------------%
\subsection*{Example 2 — Triangle with vertices \((0,0)\to(2,1)\to(1,3)\to(0,0)\)}
Integrate \(\tfrac12(x\,dy-y\,dx)\) edge by edge (CCW orientation).

\paragraph{Edge 1: \((0,0)\to(2,1)\).}
Parametrize \(x=2t\), \(y=t\), \(t\in[0,1]\). Then \(dx=2\,dt\), \(dy=dt\), and
\[
\frac12(x\,dy-y\,dx)
=
\frac12\,(2t\cdot dt - t\cdot 2\,dt)
=
0
\quad\Rightarrow\quad \text{contribution }=0.
\]

\paragraph{Edge 2: \((2,1)\to(1,3)\).}
Parametrize \(x=2-s\), \(y=1+2s\), \(s\in[0,1]\). Then \(dx=-ds\), \(dy=2\,ds\), and
\[
\frac12(x\,dy-y\,dx)
=
\frac12\Bigl((2-s)\cdot 2 - (1+2s)\cdot(-1)\Bigr)\,ds
=
\frac{5}{2}\,ds.
\]
Contribution:
\[
\int_{0}^{1}\frac{5}{2}\,ds=\frac{5}{2}=2.5.
\]

\paragraph{Edge 3: \((1,3)\to(0,0)\).}
Parametrize \(x=1-t\), \(y=3-3t\), \(t\in[0,1]\). Then \(dx=-dt\), \(dy=-3\,dt\), and
\[
\frac12(x\,dy-y\,dx)
=
\frac12\Bigl((1-t)(-3) - (3-3t)(-1)\Bigr)\,dt
=
0
\quad\Rightarrow\quad \text{contribution }=0.
\]

\paragraph{Total area.}
\[
\operatorname{Area}=0+2.5+0=\boxed{2.5}.
\]
(Checks the shoelace formula:
\(\displaystyle \tfrac12\bigl|0\cdot1+2\cdot3+1\cdot0-(2\cdot0+1\cdot1+0\cdot3)\bigr|
=\tfrac12|6-1|=2.5\).)

%===================== Circles, monomials, and a sine-power integral =====================%
\section*{Problem}
Let $\gamma(0,r)$ be the positively oriented circle $\lvert z\rvert=r>0$.
For each integer $k$, evaluate
\[
\oint_{\gamma(0,r)} z^{k}\,dz,
\]
making clear any special cases.

Using $\displaystyle \sin\theta=\frac{e^{i\theta}-e^{-i\theta}}{2i}$, rewrite
\[
\int_{0}^{2\pi}\sin^{2n}\theta\,d\theta
\]
as a contour integral around $\gamma(0,1)$ and deduce
\[
\int_{0}^{2\pi}\sin^{2n}\theta\,d\theta \;=\; \frac{2\pi}{4^n}\binom{2n}{n}.
\]

\section*{Solution}
\subsection*{1) The monomial integral on a circle}
Parametrize $z=re^{i\theta}$, $dz=i r e^{i\theta}\,d\theta$:
\[
\oint_{\gamma(0,r)} z^k\,dz
= i r^{k+1}\!\int_{0}^{2\pi} e^{i(k+1)\theta}\,d\theta
=
\begin{cases}
	0, & k\neq -1,\\[4pt]
	2\pi i, & k=-1.
\end{cases}
\]
\emph{Special case:} $k=-1$ is the only nonzero value, and it is independent of $r$.

\subsection*{2) The even power sine integral via a contour integral}
On $|z|=1$ let $z=e^{i\theta}$, so $d\theta=\dfrac{dz}{iz}$ and
\[
\sin\theta=\frac{z-1/z}{2i}.
\]
Hence
\[
\int_{0}^{2\pi}\sin^{2n}\theta\,d\theta
=
\oint_{|z|=1}
\frac{(z-1/z)^{2n}}{(2i)^{2n}}\cdot \frac{dz}{iz}.
\]
Expand $(z-1/z)^{2n}$:
\[
(z-1/z)^{2n}
=\sum_{m=0}^{2n}\binom{2n}{m}(-1)^m\,z^{\,2n-2m}.
\]
Therefore
\[
\frac{(z-1/z)^{2n}}{(2i)^{2n}}\cdot \frac{1}{iz}
=
\sum_{m=0}^{2n}
\binom{2n}{m}\frac{(-1)^m}{(2i)^{2n}i}\;z^{\,2n-2m-1}.
\]
Only the $z^{-1}$ term contributes to the contour integral; this occurs when
$2n-2m-1=-1$, i.e.\ $m=n$. The residue at $z=0$ is thus
\[
\operatorname{Res}_{z=0}
=
\binom{2n}{n}\frac{(-1)^n}{(2i)^{2n}i}.
\]
By the residue theorem,
\[
\int_{0}^{2\pi}\sin^{2n}\theta\,d\theta
=
2\pi i \cdot \binom{2n}{n}\frac{(-1)^n}{(2i)^{2n}i}
=
2\pi\,\binom{2n}{n}\,\frac{(-1)^n}{(2i)^{2n}}.
\]
Since $(2i)^{2n}=(4i^{2})^{n}=(-4)^{n}=(-1)^n4^n$, the $(-1)^n$ cancels and we obtain
\[
\boxed{\ \displaystyle
	\int_{0}^{2\pi}\sin^{2n}\theta\,d\theta
	=
	\frac{2\pi}{4^n}\binom{2n}{n}\ }.
\]

%===================== Complex Power Maps w = z^k (k=1,2,3,4) =====================%
\section*{Problem}
Describe the complex power maps
\[
w=z^{k},\qquad k=1,2,3,4,
\]
with emphasis on geometry: action on radii/angles, circles/rays, winding, critical points,
conformality, and behavior on the unit circle.

\section*{Universal facts (all $k\in\mathbb N$)}
Write $z=re^{i\theta}$ (polar form). Then
\[
w=z^{k}=r^{k}e^{ik\theta}.
\]
Hence, for every $k$:
\begin{itemize}
	\item \textbf{Radius scaling:} $\lvert w\rvert=\lvert z\rvert^{k}=r^{k}$. Circles $\lvert z\rvert=r$ map to circles $\lvert w\rvert=r^{k}$.
	\item \textbf{Angle multiplication:} $\arg w=k\,\arg z \pmod{2\pi}$. Rays $\theta=\theta_{0}$ map to rays $\phi=k\theta_{0}$.
	\item \textbf{Winding of the unit circle:} $\lvert z\rvert=1 \mapsto \lvert w\rvert=1$, wrapped $k$ times (as $\theta:0\to 2\pi$ gives $k\theta:0\to 2k\pi$).
	\item \textbf{Preimages:} For $w\neq 0$, exactly $k$ distinct preimages (the $k$th roots of $w$), spaced by angles $2\pi/k$.
	\item \textbf{Conformality:} Holomorphic and angle-preserving wherever $z\neq 0$.
	\item \textbf{Critical set:} $(z^{k})'=k z^{k-1}$ vanishes iff $z=0$. Thus $z=0$ is a critical point of multiplicity $k-1$, locally a $(k:1)$ folding.
	\item \textbf{Orientation:} Preserved away from $0$ (Jacobian $=\lvert k z^{k-1}\rvert^{2}>0$ for $z\neq 0$).
\end{itemize}

\section*{Case-by-case descriptions}
\paragraph{$k=1$ (identity).}
$w=z$. Radii and angles unchanged. No critical points. Global bijection. One preimage for each $w$.

\paragraph{$k=2$ (doubling angle).}
$w=z^{2}$. Angles \(\phi=2\theta\); $\lvert w\rvert=r^{2}$.
The unit circle winds twice. Critical behavior: derivative $2z$ vanishes at $0$ (local $2{:}1$).
Rays map to rays with doubled angle; any line through the origin maps to itself with angle doubled.
Real axis ($\theta=0,\pi$) and imaginary axis ($\theta=\pm\pi/2$) both map to the nonnegative real axis
(squaring removes sign).

\paragraph{$k=3$ (tripling angle).}
$w=z^{3}$. Angles \(\phi=3\theta\); $\lvert w\rvert=r^{3}$.
Unit circle wraps three times. Critical: $3z^{2}$, order-$2$ at $0$ (local $3{:}1$).
Preimages are rotated by $120^\circ$: if $z$ maps to $w$, then $ze^{2\pi i/3}$ and $ze^{4\pi i/3}$ do too.
Real axis maps to real axis with sign preserved $(\pm x)^{3}=\pm x^{3}$; imaginary axis maps to imaginary axis.

\paragraph{$k=4$ (quadrupling angle).}
$w=z^{4}$. Angles \(\phi=4\theta\); $\lvert w\rvert=r^{4}$.
Unit circle wraps four times. Critical: $4z^{3}$, order-$3$ at $0$ (local $4{:}1$).
Both real and imaginary axes map to the nonnegative real axis (even power discards sign).
Preimages exhibit $90^\circ$ rotational symmetry.

\section*{Behavior on the unit circle and line integrals}
On $\lvert z\rvert=1$ one has $z=e^{i\theta}\mapsto w=e^{ik\theta}$: same circle, $k$-fold wrap.
For circle integrals,
\[
\oint_{\lvert z\rvert=1} z^{k}\,dz=
\begin{cases}
	0, & k\neq -1,\\[2pt]
	2\pi i, & k=-1,
\end{cases}
\]
since $z^{k}$ is holomorphic on and inside the circle for $k\neq -1$, and $\oint dz/z=2\pi i$.

\section*{Visual heuristics}
\[
\text{“Multiply the angle, power the radius.”}
\]
Near $0$, $z^{k}$ collapses many directions into one (derivative $0$ at the origin);
on the unit circle it is the pure phase map $e^{i\theta}\mapsto e^{ik\theta}$, producing $k$ windings.

%======================= Integral with Cauchy / residues =======================%
\section*{Problem}
Let $a\in\mathbb R$ with $a>1$. Using Cauchy’s Integral Formula (residue theorem) and
the holomorphic function
\[
f(z)=\frac{z^{\,n}}{z-a},
\]
compute
\[
I_n(a):=\int_0^{2\pi}\frac{\cos(n\theta)}{\,1-2a\cos\theta+a^2\,}\,d\theta .
\]

\section*{Setup: unit circle substitution}
On $|z|=1$ let $z=e^{i\theta}$, so $d\theta=\dfrac{dz}{iz}$ and
\[
\cos(n\theta)=\tfrac12\!\left(z^{\,n}+z^{-n}\right),\qquad
1-2a\cos\theta+a^2=\frac{(z-a)(1-az)}{z}.
\]
Hence
\[
I_n(a)
=\frac{1}{2i}\oint_{|z|=1}\frac{z^{\,n}+z^{-n}}{(z-a)(1-az)}\,dz.
\]
Write
\[
g(z):=\frac{z^{\,n}+z^{-n}}{(z-a)(1-az)}.
\]
For $a>1$, the poles of $g$ \emph{inside} $|z|=1$ are:
\[
z=\frac1a\quad\text{(simple)},\qquad
z=0\ \text{(present only when }n\ge1\text{)}.
\]
(The pole at $z=a$ lies outside.)

By the residue theorem,
\[
I_n(a)=\frac{1}{2i}\cdot 2\pi i\sum_{\text{inside}}\Res(g)=\pi\sum_{\text{inside}}\Res(g).
\]

\section*{Residues}
\paragraph{At $z=\tfrac1a$.}
Since $1-az$ vanishes simply,
\[
\Res_{z=1/a} g
=\lim_{z\to 1/a}\frac{z^{\,n}+z^{-n}}{(z-a)}\cdot\frac{1}{-a}
=\frac{a^{-n}+a^{\,n}}{a^2-1}.
\]

\paragraph{At $z=0$ (only if $n\ge1$).}
Use the double geometric series
\[
\frac{1}{(z-a)(1-az)}
=-\frac1a\,\frac{1}{1-z/a}\,\frac{1}{1-az}
=-\frac1a\sum_{m\ge0}\Big(\frac{z}{a}\Big)^m\sum_{\ell\ge0}(az)^\ell .
\]
Multiplying by $z^{-n}$, the coefficient of $z^{-1}$ (the residue) arises from
$m+\ell=n-1$:
\[
\Res_{z=0} g
=-\frac1a\sum_{m=0}^{n-1}a^{\,\ell-m}
=-\frac1a\sum_{m=0}^{n-1}a^{\,n-1-2m}
=-\frac{a^{\,n}-a^{-n}}{a^2-1}.
\]

\section*{Sum and value of the integral}
For $n\ge1$,
\[
\sum \Res(g)=\frac{a^{-n}+a^{\,n}}{a^2-1}-\frac{a^{\,n}-a^{-n}}{a^2-1}
=\frac{2a^{-n}}{a^2-1}.
\]
Therefore
\[
\boxed{\ I_n(a)=\pi\cdot\frac{2a^{-n}}{a^2-1}
	=\frac{2\pi}{a^2-1}\,a^{-n},\qquad a>1,\ n\ge1.\ }
\]

For $n=0$ there is no pole at $0$, so
\[
\boxed{\ I_0(a)=\pi\cdot\frac{2}{a^2-1}=\frac{2\pi}{a^2-1}. \ }
\]

\section*{Checks}
\[
I_0(a)=\int_0^{2\pi}\frac{d\theta}{1-2a\cos\theta+a^2}
=\frac{2\pi}{a^2-1},\qquad
I_1(a)=\frac{2\pi}{a^2-1}\cdot\frac1a,\quad
I_2(a)=\frac{2\pi}{a^2-1}\cdot\frac1{a^2},\ \ldots
\]
These match the Poisson–kernel/Fourier identity
\(
\displaystyle \frac{1}{1-2a\cos\theta+a^2}
=\frac{1}{a^2-1}\Big(1+2\sum_{m\ge1}a^{-m}\cos m\theta\Big)
\)
for $a>1$.

I_k(a)=\frac{1}{2i}\oint_{|z|=1}\frac{z^{\,k}+z^{-k}}{(z-a)(1-az)}\,dz.
\]
For $a>1$, the poles inside $|z|=1$ are $z=\frac1a$ (simple) and, when $k\ge1$, also $z=0$ from $z^{-k}$.
A short residue computation gives
\[
\sum_{\text{inside}}\Res
\left(\frac{z^{\,k}+z^{-k}}{(z-a)(1-az)}\right)
=\frac{2a^{-k}}{a^2-1},
\]
so
\[
\boxed{\ I_k(a)=\frac{2\pi}{a^2-1}\,a^{-k},\qquad a>1,\ k\in\mathbb N.\ }
\]
For $k=0$ there is no pole at $0$, and we get
\[
\boxed{\ I_0(a)=\frac{2\pi}{a^2-1}. \ }
\]

\section*{Explicit values for $k=1,2,3,4$}
\[
\begin{aligned}
	I_1(a) &= \frac{2\pi}{a^2-1}\cdot \frac{1}{a},\\[4pt]
	I_2(a) &= \frac{2\pi}{a^2-1}\cdot \frac{1}{a^{2}},\\[4pt]
	I_3(a) &= \frac{2\pi}{a^2-1}\cdot \frac{1}{a^{3}},\\[4pt]
	I_4(a) &= \frac{2\pi}{a^2-1}\cdot \frac{1}{a^{4}}.
\end{aligned}
\]

\section*{Why the pattern is so clean}
\subsection*{(1) Poisson kernel / Fourier viewpoint}
Set $r=\frac{1}{a}\in(0,1)$. Then
\[
\frac{1}{1-2a\cos\theta+a^2}
=\frac{1}{a^2}\cdot \frac{1}{1-2r\cos\theta+r^2}
=\frac{1}{a^2-1}\Big(1+2\sum_{m\ge1} r^{m}\cos(m\theta)\Big).
\]
Multiply by $\cos(k\theta)$ and integrate over $[0,2\pi]$; orthogonality leaves only the $m=k$ term:
\[
I_k(a)=\frac{1}{a^2-1}\cdot 2r^{\,k}\int_{0}^{2\pi}\cos^2(k\theta)\,d\theta
=\frac{2\pi}{a^2-1}\,r^{\,k}
=\frac{2\pi}{a^2-1}\,a^{-k}.
\]
For $k=0$ only the constant term contributes, giving $I_0(a)=\frac{2\pi}{a^2-1}$.

\subsection*{(2) Contour / residue viewpoint}
With $z=e^{i\theta}$,
\[
I_k(a)=\frac{1}{2i}\oint_{|z|=1}\frac{z^{\,k}+z^{-k}}{(z-a)(1-az)}\,dz,
\]
and the only interior poles are at $z=1/a$ and (if $k\ge1$) $z=0$. Their residues sum to $\dfrac{2a^{-k}}{a^2-1}$,
so $I_k(a)=\dfrac{2\pi}{a^2-1}\,a^{-k}$ again.

\section*{Interpretation as $k$ varies}
\begin{itemize}
	\item \textbf{Exponential decay in $k$:} $I_k(a)\sim C\,a^{-k}$—higher harmonics are damped as the parameter $a>1$
	pushes the pole farther from the unit circle.
	\item \textbf{Monotone in $a$:} For fixed $k\ge1$, $I_k(a)$ decreases with $a$.
	\item \textbf{Base case $k=0$:} Recovers the classical identity
	$\displaystyle \int_{0}^{2\pi}\frac{d\theta}{a^2-2a\cos\theta+1}=\frac{2\pi}{a^2-1}$.
\end{itemize}

%===================== A compact deep-dive:  F_n(z)=z^n/(z-a),  a>1 =====================%
\section*{Setup}
Fix $a\in\mathbb R$ with $a>1$. For $n\in\mathbb N$, consider
\[
F_n(z)=\frac{z^{\,n}}{z-a}.
\]
We record zeros/poles, partial fraction form, residue at $a$, behavior at $\infty$,
critical points, and a brief geometric note.

\section*{Shared facts (all $n\ge1$)}
\begin{itemize}
	\item \textbf{Meromorphic on $\widehat{\mathbb C}$:} a simple pole at $z=a$; a zero of order $n$ at $z=0$.
	\item \textbf{Residue at $a$:}
	\[
	\Res_{z=a} F_n=\lim_{z\to a}(z-a)\frac{z^{\,n}}{z-a}=a^{n}.
	\]
	\item \textbf{Polynomial part (long division):}
	\[
	\frac{z^{\,n}}{z-a}
	= z^{n-1}+a z^{n-2}+\cdots + a^{n-2}z + a^{n-1} \;+\; \frac{a^{n}}{z-a}.
	\]
	\item \textbf{Growth at $\infty$:} $F_n(z)\sim z^{\,n-1}$ as $|z|\to\infty$.
	Thus $\infty$ is a critical point of multiplicity $n-2$ when $n\ge2$ (removable for $n=1$).
	\item \textbf{Derivative / finite critical points:}
	\[
	F_n'(z)=\frac{z^{\,n-1}\big((n-1)z-na\big)}{(z-a)^2}.
	\]
	Hence, aside from the pole, the finite critical points are
	\[
	z=0 \ \text{(multiplicity $n-1$ if $n\ge2$)},\qquad
	z=\frac{n}{\,n-1\,}a \ \text{(simple, for $n\ge2$)}.
	\]
	(This matches Riemann--Hurwitz: total branching $=2n-2$.)
\end{itemize}

\section*{Case-by-case}

\paragraph{$n=1:\; F_1(z)=\dfrac{z}{z-a}$ \;(Möbius).}
\begin{itemize}
	\item Zeros/poles: simple zero at $0$; simple pole at $a$.
	\item Residue: $\Res_{a}F_1=a$.
	\item Partial fraction: $F_1(z)=1+\dfrac{a}{z-a}$.
	\item Growth: $F_1(\infty)=1$ (removable at $\infty$).
	\item Critical points: none in $\mathbb C$ ($F_1'(z)=-a/(z-a)^2\neq0$).
	\item Geometry: Möbius map; sends lines/circles to lines/circles; conformal except at $z=a$.
\end{itemize}

\paragraph{$n=2:\; F_2(z)=\dfrac{z^{2}}{z-a}$.}
\begin{itemize}
	\item Zeros/poles: double zero at $0$; simple pole at $a$.
	\item Residue: $\Res_{a}F_2=a^{2}$.
	\item Partial fraction: $F_2(z)=z+a+\dfrac{a^{2}}{z-a}$.
	\item Growth: $F_2(z)\sim z$ as $|z|\to\infty$ (simple pole at $\infty$).
	\item Critical points from $F_2'(z)=\dfrac{z(2z-2a)}{(z-a)^2}$:
	\ $z=0$ (order $1$), $z=2a$ (simple).
	\item Geometry: degree-$2$ rational map (generically $2{:}1$); near $a$, behaves like $a^{2}/(z-a)$.
\end{itemize}

\paragraph{$n=3:\; F_3(z)=\dfrac{z^{3}}{z-a}$.}
\begin{itemize}
	\item Zeros/poles: triple zero at $0$; simple pole at $a$.
	\item Residue: $\Res_{a}F_3=a^{3}$.
	\item Partial fraction: $F_3(z)=z^{2}+az+a^{2}+\dfrac{a^{3}}{z-a}$.
	\item Growth: $F_3(z)\sim z^{2}$ (double pole at $\infty$).
	\item Critical points from $F_3'(z)=\dfrac{z^{2}(2z-3a)}{(z-a)^2}$:
	\ $z=0$ (order $2$), $z=\tfrac{3}{2}a$ (simple); plus one at $\infty$ (order $1$), totaling $4=2n-2$.
	\item Geometry: degree-$3$ branched covering; a generic value has three preimages (with multiplicity).
\end{itemize}

\paragraph{$n=4:\; F_4(z)=\dfrac{z^{4}}{z-a}$.}
\begin{itemize}
	\item Zeros/poles: quadruple zero at $0$; simple pole at $a$.
	\item Residue: $\Res_{a}F_4=a^{4}$.
	\item Partial fraction: $F_4(z)=z^{3}+az^{2}+a^{2}z+a^{3}+\dfrac{a^{4}}{z-a}$.
	\item Growth: $F_4(z)\sim z^{3}$ (triple pole at $\infty$).
	\item Critical points from $F_4'(z)=\dfrac{z^{3}(3z-4a)}{(z-a)^2}$:
	\ $z=0$ (order $3$), $z=\tfrac{4}{3}a$ (simple); plus a critical point at $\infty$ (order $2$);
	total branching $6=2n-2$.
	\item Geometry: degree-$4$ map; generic value has four preimages; stronger ``folding'' near $0$.
\end{itemize}

\section*{Unit-circle snapshot ($|z|=1$, $a>1$)}
\[
F_n(e^{it})=\frac{e^{int}}{e^{it}-a}.
\]
The denominator never vanishes on $|z|=1$; the magnitude
$\big|F_n(e^{it})\big|=1/|e^{it}-a|$ is independent of $n$, while the phase winds $n$ times as $t:0\to2\pi$.
This is exactly why, in integrals against $\cos(n\theta)$, a clean factor $a^{-n}$ appears.

%===================== Winding numbers from an oriented curve =====================%
\section*{Problem}
Consider the oriented closed curve $\gamma$ shown. Compute the winding number
\[
\Ind(\gamma;p)=\frac{1}{2\pi i}\oint_{\gamma}\frac{dz}{z-p}
\]
around each marked point $p\in\{a,b,c,d,e\}$.

\section*{Background: what the winding number is and how to read it}
\begin{itemize}
	\item \textbf{Meaning.} $\Ind(\gamma;p)$ counts, with sign, how many times $\gamma$
	winds around $p$: $+1$ for a counterclockwise (CCW) loop, $-1$ for a clockwise (CW) loop,
	and sums over multiple loops.
	\item \textbf{Ray test (practical rule).} Draw a ray from $p$ to infinity (e.g.\ horizontally to the right).
	Count intersections of $\gamma$ with the ray:
	each upward crossing contributes $+1$, each downward crossing contributes $-1$;
	the sum equals $\Ind(\gamma;p)$. Any generic ray avoiding vertices gives the same integer.
	\item \textbf{Reading orientation on a small lobe.}
	\begin{itemize}
		\item On a \emph{CCW} lobe: right edge arrows go up; along the top they go left; left edge down; bottom right.
		\item On a \emph{CW} lobe: right edge arrows go down; along the top they go right; left edge up; bottom left.
	\end{itemize}
\end{itemize}

\section*{Reading the figure (qualitative description)}
\begin{itemize}
	\item Point $a$ lies inside a right-hand lobe with arrows up on the right edge and left along the top:
	a CCW loop. Hence $\Ind(\gamma;a)=+1$.
	\item Point $b$ lies in a smaller right–middle lobe that is CCW. Hence $\Ind(\gamma;b)=+1$.
	\item Point $c$ lies in a left interior oval whose overall orientation is clockwise. Hence $\Ind(\gamma;c)=-1$.
	\item Point $d$ is not encircled by any lobe (strands pass on both sides with no net wrap). Hence $\Ind(\gamma;d)=0$.
	\item Point $e$ is inside a bottom oval oriented clockwise (top edge arrows run right). Hence $\Ind(\gamma;e)=-1$.
\end{itemize}

\section*{Answer (collected)}
\[
\boxed{
	\Ind(\gamma;a)=+1,\qquad
	\Ind(\gamma;b)=+1,\qquad
	\Ind(\gamma;c)=-1,\qquad
	\Ind(\gamma;d)=0,\qquad
	\Ind(\gamma;e)=-1.
}
\]

\section*{Problem}
Give a compact profile of the meromorphic function
\[
F_p(z)=\frac{1}{z-p}\qquad(p\in\C),
\]
and explain why the winding number
\[
\Ind(\gamma;p)=\frac{1}{2\pi i}\oint_{\gamma}\frac{dz}{z-p}
\]
is constant on connected components of $\C\setminus\gamma$.

\section*{The function \(F_p(z)=1/(z-p)\)}

\subsection*{Algebraic/analytic data}
\begin{itemize}[leftmargin=2em]
	\item \textbf{Domain.} Holomorphic on $\C\setminus\{p\}$.
	\item \textbf{Zeros.} None in $\C$; on $\Chat$ there is a simple zero at $\infty$.
	\item \textbf{Pole.} Simple pole at $z=p$ with principal part $\dfrac{1}{z-p}$.
	\item \textbf{Residue.} $\Res_{z=p} F_p = 1$.
	\item \textbf{Laurent at \(p\).} \( \displaystyle \frac{1}{z-p} \) (pure principal part).
	\item \textbf{Power series at \(z_0\neq p\).} If $|z-z_0|<|z_0-p|$, then
	\[
	\frac{1}{z-p}=\frac{1}{z_0-p}\sum_{n=0}^{\infty}\!\left(\frac{z-z_0}{\,z_0-p\,}\right)^{\!n}.
	\]
	\item \textbf{Derivatives.} \( \displaystyle F_p'(z)=-\frac{1}{(z-p)^2},\quad
	F_p''(z)=\frac{2}{(z-p)^3},\ \ldots \)
	Hence no finite critical points (derivative never $0$); branching only at $\infty$.
\end{itemize}

\subsection*{Geometry / mapping properties (Möbius transform)}
\begin{itemize}[leftmargin=2em]
	\item \textbf{Decomposition.} \(F_p = \big(z\mapsto 1/z\big)\circ \big(z\mapsto z-p\big)\).
	\item \textbf{Conformality.} Holomorphic and locally orientation-preserving on $\C\setminus\{p\}$.
	\item \textbf{Circles/lines.} Möbius maps send generalized circles to generalized circles.
	Any circle/line passing through $p$ maps to a line; otherwise to a circle not through $0$.
	\item \textbf{Polar description.} If \(z-p=\rho e^{i\theta}\), then
	\[
	w=F_p(z)=\frac{1}{\rho}e^{-i\theta}.
	\]
	Radii invert ($\rho\mapsto 1/\rho$) and angles negate ($\theta\mapsto -\theta$).
	The circle $|z-p|=\rho$ maps to $|w|=1/\rho$.
\end{itemize}

\subsection*{Behavior at \(\infty\)}
As $|z|\to\infty$, \(F_p(z)\to 0\). On $\Chat$, $w=0$ is the value at $\infty$ and $F_p$ has a simple zero at $\infty$.
Being a degree-$1$ rational map, $F_p$ is a biholomorphism of the sphere.

\section*{Why \(\Ind(\gamma;p)\) is constant on components of \(\C\setminus\gamma\)}

\subsection*{(A) Homotopy invariance (Cauchy)}
If $\gamma_0,\gamma_1$ are closed curves homotopic in $\C\setminus\{p\}$, then by Cauchy's theorem
\[
\oint_{\gamma_0}\frac{dz}{z-p}=\oint_{\gamma_1}\frac{dz}{z-p},
\]
so $\Ind(\gamma;p)$ depends only on the homotopy class of $\gamma$ in $\C\setminus\{p\}$.
Hence $p\mapsto \Ind(\gamma;p)$ is locally constant on each connected component of $\C\setminus\gamma$.

\subsection*{(B) Argument--change formula}
Choose a branch of $\Log(z-p)$ along $\gamma$. Since $\dfrac{d}{dz}\Log(z-p)=\dfrac{1}{z-p}$,
\[
\oint_{\gamma}\frac{dz}{z-p}
=\oint_{\gamma} d\,\Log(z-p)
=\Delta_{\gamma}\Log(z-p)
=i\,\Delta_{\gamma}\Arg(z-p),
\]
and therefore
\[
\boxed{\ \Ind(\gamma;p)=\frac{1}{2\pi}\,\Delta_{\gamma}\Arg\big(\gamma(t)-p\big)\ }.
\]
This is the rigorous form of the “ray test” for counting signed crossings.

\subsection*{(C) Via the map \(w=1/(z-p)\)}
Under $w=F_p(z)$ we have $\Arg\,w=-\Arg(z-p)$, hence
\[
\Delta_{\gamma}\Arg(z-p)=-\,\Delta_{w(\gamma)}\Arg(w)
\quad\Rightarrow\quad
\oint_{\gamma}\frac{dz}{z-p}=-\oint_{w(\gamma)}\frac{dw}{w}
=-\,2\pi i\,\Ind\big(w(\gamma);0\big)
=2\pi i\,\Ind(\gamma;p).
\]
Thus the index is the signed number of turns of $w(\gamma)$ about $0$.

\section*{TL;DR}
\[
F_p(z)=\frac{1}{z-p}:\ \text{meromorphic, simple pole at }p\ (\Res=1),\ \text{simple zero at }\infty,
\]
\[
F_p'(z)=-\frac{1}{(z-p)^2},\ \text{Möbius (translation+inversion), circles/lines }\to\ \text{circles/lines}.
\]
Along a loop around $p$: radius inverts, angle negates; the total change in $\Arg(z-p)$ equals $2\pi\cdot\Ind$.
Hence $\Ind(\gamma;p)$ is constant on components of $\C\setminus\gamma$.


\begin{problem}
	Let $f(z)=e^{-z^{2}}$. Using an appropriate contour and the Cauchy--Goursat theorem, show that for $b>0$,
	\[
	\int_{0}^{\infty} e^{-x^{2}}\cos(2bx)\,dx=\frac{\sqrt{\pi}}{2}\,e^{-b^{2}}.
	\]
	You may use $\int_{0}^{\infty} e^{-x^{2}}\,dx=\sqrt{\pi}/2$.
\end{problem}

\begin{solution}
	\noindent\textbf{Step 1: The rectangle.}\;
	Consider the rectangle with vertices $-R,\,R,\,R+ib,\,-R+ib$ oriented counterclockwise. 
	Since $e^{-z^{2}}$ is entire, Cauchy--Goursat gives
	\[
	\int_{-R}^{R} e^{-x^{2}}\,dx
	+\int_{\text{right}} e^{-z^{2}}\,dz
	-\int_{-R}^{R} e^{-(x+ib)^{2}}\,dx
	+\int_{\text{left}} e^{-z^{2}}\,dz
	=0 .
	\]
	
	\medskip
	\noindent\textbf{Step 2: Vertical sides vanish.}\;
	On $z=\pm R+iy$ with $0\le y\le b$,
	\[
	|e^{-z^{2}}|=e^{-(R^{2}-y^{2})}\le e^{-R^{2}}e^{b^{2}},
	\]
	so both vertical integrals tend to $0$ as $R\to\infty$.
	
	\medskip
	\noindent\textbf{Step 3: Shifted top edge.}\;
	Letting $R\to\infty$ yields
	\[
	\int_{-\infty}^{\infty} e^{-x^{2}}\,dx
	=\int_{-\infty}^{\infty} e^{-(x+ib)^{2}}\,dx .
	\]
	Since $e^{-(x+ib)^{2}}=e^{-x^{2}+b^{2}-i\,2bx}
	= e^{b^{2}}\,e^{-x^{2}}\,e^{-i2bx}$, we get
	\[
	\int_{-\infty}^{\infty} e^{-x^{2}} e^{\,i2bx}\,dx
	= e^{-b^{2}} \int_{-\infty}^{\infty} e^{-x^{2}}\,dx
	= \sqrt{\pi}\,e^{-b^{2}} .
	\]
	
	\medskip
	\noindent\textbf{Step 4: Take real parts and halve.}\;
	Because the integrand is even,
	\[
	\setlength{\jot}{6pt}
	\begin{aligned}
		\int_{0}^{\infty} e^{-x^{2}}\cos(2bx)\,dx
		&= \frac{1}{2}\int_{-\infty}^{\infty} e^{-x^{2}}\cos(2bx)\,dx \\[2pt]
		&= \frac{1}{2}\,\Re\!\left(\int_{-\infty}^{\infty} e^{-x^{2}} e^{\,i2bx}\,dx\right) \\[2pt]
		&= \frac{1}{2}\,\Re\!\big(\sqrt{\pi}\,e^{-b^{2}}\big)
		= \frac{\sqrt{\pi}}{2}\,e^{-b^{2}} .
	\end{aligned}
	\]
	\hfill$\square$
\end{solution}

\subsection*{Background on the function $f(z)=e^{-z^{2}}$}

\paragraph{Domain and analyticity.}
The function $f(z)=e^{-z^{2}}$ is entire (holomorphic on all of $\mathbb C$) with global power series
\[
e^{-z^{2}}=\sum_{n=0}^{\infty}\frac{(-1)^n}{n!}\,z^{2n}.
\]

\paragraph{Derivative, critical set, and ODE.}
We have
\[
f'(z)=-2z\,e^{-z^{2}}=-2z\,f(z),
\]
so the unique critical point is $z=0$ (and the corresponding critical value is $f(0)=1$).
The function solves the first-order linear ODE $f'(z)+2z\,f(z)=0$.

\paragraph{Zeros and singularities.}
The exponential never vanishes, so $f$ has no zeros in $\mathbb C$. Being entire, it has no poles; its only singularity is the essential singularity at infinity. Its growth is anisotropic: for $z=re^{i\theta}$,
\[
|f(z)|=e^{-\operatorname{Re}(z^{2})}=e^{-r^{2}\cos(2\theta)},
\]
which decays when $|\theta|<\pi/4$ and explodes when $\pi/4<|\theta|<3\pi/4$.

\paragraph{Symmetries.}
Since the Taylor coefficients are real, $f(\overline{z})=\overline{f(z)}$.
Moreover $f$ is even: $f(-z)=f(z)$.

\paragraph{Modulus and phase; characteristic curves.}
Writing $z=x+iy$,
\[
e^{-z^{2}}=e^{-(x^{2}-y^{2})}\,e^{-i\,2xy}.
\]
Hence
\[
|f(z)|=e^{-(x^{2}-y^{2})},\qquad \arg f(z)=-2xy.
\]
Therefore:
\begin{itemize}
	\item Level sets of constant modulus $|f|=\rho$ are the hyperbolas $x^{2}-y^{2}=-\log \rho$ (rectangular hyperbolas with asymptotes $y=\pm x$).
	\item Level sets of constant argument $\arg f=\theta$ are the hyperbolas $xy=-\tfrac{\theta}{2}$, orthogonal to the modulus curves.
\end{itemize}
On the diagonals $y=\pm x$ we have $|f|=1$ and $\arg f=\mp 2x^{2}$. On the real axis $f(x)=e^{-x^{2}}\in(0,1]$; on the imaginary axis $f(iy)=e^{y^{2}}>1$.

\paragraph{Conformality.}
The map is conformal everywhere except at $z=0$, where $f'(0)=0$.

\paragraph{Order and Fourier connection.}
The function is an entire function of order $2$. The Gaussian is (up to normalization) an eigenfunction of the Fourier transform:
\[
\int_{-\infty}^{\infty}e^{-x^{2}}e^{i2bx}\,dx=\sqrt{\pi}\,e^{-b^{2}},
\]
and the standard rectangle-shift contour argument provides a classical complex-analytic proof.

\subsection*{Background on $g_b(x)=e^{-x^{2}}\cos(2bx)$}

\paragraph{Domain and smoothness.}
$g_b$ is defined and $C^\infty$ on all of $\mathbb R$ (and entire as a function of $x\in\mathbb C$).

\paragraph{Symmetry.}
$g_b$ is even: $g_b(-x)=g_b(x)$.

\paragraph{Envelope and oscillation.}
$\lvert g_b(x)\rvert\le e^{-x^{2}}$, so the Gaussian envelope controls the size. 
Oscillation frequency scales linearly with $b$; as $b\to 0$, $g_b\to e^{-x^{2}}$.

\paragraph{Zeros and critical points.}
Zeros occur when $\cos(2bx)=0$, i.e.
\[
x=\frac{(k+\tfrac12)\pi}{2b},\qquad k\in\mathbb Z,\ b>0.
\]
Differentiating,
\[
g_b'(x)=e^{-x^{2}}\!\big(-2x\cos(2bx)-2b\sin(2bx)\big)=0
\quad\Longleftrightarrow\quad
\tan(2bx)=-\frac{x}{b},
\]
so local extrema lie near $x\approx \frac{k\pi}{2b}$, shifted slightly toward the origin.

\paragraph{Fourier/Gaussian connection.}
With $h_b(x)=e^{-x^{2}}e^{i2bx}$, one has $g_b(x)=\Re h_b(x)$ and
\[
\int_{-\infty}^{\infty} h_b(x)\,dx=\sqrt{\pi}\,e^{-b^{2}},
\]
whence
\[
\int_{0}^{\infty} e^{-x^{2}}\cos(2bx)\,dx=\frac{\sqrt{\pi}}{2}\,e^{-b^{2}}
\quad\text{(even integrand).}
\]

\paragraph{Integrability and $L^{2}$ size.}
$g_b\in L^{1}(\mathbb R)\cap L^{2}(\mathbb R)$, and
\[
\int_{-\infty}^{\infty} g_b(x)\,dx=\sqrt{\pi}\,e^{-b^{2}},\qquad
\int_{-\infty}^{\infty} g_b(x)^{2}\,dx=\frac{\sqrt{\pi/2}}{2}\,\big(1+e^{-2b^{2}}\big).
\]

\paragraph{Differential relation (via complex parent).}
For $h_b(x)=e^{-x^{2}}e^{i2bx}$ one has $h_b'(x)=(-2x+i2b)h_b(x)$ and $g_b=\Re h_b$.

	\section*{Theory notes: $\sinh x$ and $\cos x$ (with plots)}
	
	\subsection*{Definitions and core identities}
	\[
	\sinh x=\frac{e^{x}-e^{-x}}{2}, \qquad 
	\cos x=\frac{e^{ix}+e^{-ix}}{2}=\Re\!\left(e^{ix}\right).
	\]
	Useful complex links:
	\[
	\sinh(ix)=i\sin x,\qquad \cosh(ix)=\cos x.
	\]
	Pythagorean-type identity (hyperbolic):
	\[
	\cosh^2 x-\sinh^2 x=1.
	\]
	
	\subsection*{Basic properties of $\sinh x$}
	\begin{itemize}
		\item Odd and strictly increasing on $\mathbb{R}$; range $(-\infty,\infty)$.
		\item Derivatives: $\displaystyle \frac{d}{dx}\sinh x=\cosh x\ge 1$ (with equality only at $x=0$), 
		$\displaystyle \frac{d^2}{dx^2}\sinh x=\sinh x$.
		\item Asymptotics: $\sinh x\sim \tfrac12 e^{x}$ as $x\to+\infty$ and $\sinh x\sim -\tfrac12 e^{-|x|}$ as $x\to-\infty$.
		\item Series (entire function):
		\[
		\sinh x=\sum_{n=0}^{\infty}\frac{x^{2n+1}}{(2n+1)!}.
		\]
		\item Inverse and derivative:
		\[
		\operatorname{arsinh}\,y=\ln\!\bigl(y+\sqrt{y^2+1}\,\bigr), 
		\qquad \frac{d}{dy}\operatorname{arsinh} y=\frac{1}{\sqrt{1+y^2}}.
		\]
		\item Addition/angle-doubling:
		\[
		\sinh(a\pm b)=\sinh a\,\cosh b\pm \cosh a\,\sinh b,\qquad 
		\sinh(2x)=2\sinh x\,\cosh x.
		\]
		\item Primitives: $\displaystyle \int \sinh x\,dx=\cosh x+C$.
	\end{itemize}
	
	\subsection*{Basic properties of $\cos x$}
	\begin{itemize}
		\item Even and $2\pi$-periodic; range $[-1,1]$.
		\item Zeros at $x=\tfrac{\pi}{2}+\pi k$; extrema: $\cos x=1$ at $x=2\pi k$, $\cos x=-1$ at $x=(2k+1)\pi$.
		\item Derivatives/ODE: $\displaystyle \frac{d}{dx}\cos x=-\sin x$, \quad $\displaystyle \frac{d^2}{dx^2}\cos x=-\cos x$ (solves $y''+y=0$).
		\item Series (entire function):
		\[
		\cos x=\sum_{n=0}^{\infty}\frac{(-1)^n x^{2n}}{(2n)!}.
		\]
		\item Bounds and local shape (for small $x$): $1-\tfrac{x^2}{2}\le \cos x\le 1$; 
		more sharply, $1-\tfrac{x^2}{2}\le \cos x\le 1-\tfrac{x^2}{2}+\tfrac{x^4}{24}$.
		\item Addition/angle-doubling:
		\[
		\cos(a\pm b)=\cos a\,\cos b\mp \sin a\,\sin b,\qquad 
		\cos(2x)=2\cos^2 x-1=1-2\sin^2 x.
		\]
		\item Mean value over a period: $\displaystyle \frac{1}{2\pi}\int_{0}^{2\pi}\cos x\,dx=0$.
	\end{itemize}

\section*{Setup and notation}
Let $z=x+iy\in\mathbb{C}$ and $w=re^{i\varphi}\in\mathbb{C}\setminus\{0\}$ be the polar form of $w$.
Consider the holomorphic map
\[
w=e^{z}=e^{x+iy}=e^{x}\,e^{iy}.
\]
We write $\Arg$ for the principal argument (with a branch cut along the negative real axis) and $\log$ for the principal branch of the logarithm.

\section*{Coordinate relations: Cartesian $\leftrightarrow$ polar}
From $w=e^{x}e^{iy}$ we immediately obtain
\[
\boxed{\,r=|w|=e^{x}\,},\qquad
\boxed{\,\varphi=\arg w\equiv y\pmod{2\pi}}.
\]
The inverse map (a multivalued logarithm) is
\[
x=\ln r,\qquad y=\varphi+2\pi k,\ k\in\mathbb{Z},\qquad
z=\ln r + i(\varphi+2\pi k).
\]
After restricting to a horizontal strip $y\in(\beta,\beta+2\pi)$ we obtain a single-valued branch $z=\ln r + i\varphi$ for $\varphi\in(\beta,\beta+2\pi)$.

\section*{Conformality, scale, and area element}
Since $(e^{z})'=e^{z}$, the map is conformal and the local scale factor is
\[
\lambda(z)=|e^{z}|=e^{x}=r.
\]
For infinitesimal displacements: $dr=r\,dx$ and $d\varphi=dy$.
Line elements scale by $\lambda$ and area elements by $\lambda^{2}$:
\[
\mathrm{d}A_{w}=\lambda(z)^{2}\,\mathrm{d}A_{z}=|e^{z}|^{2}\,\mathrm{d}x\,\mathrm{d}y
= r^{2}\,\mathrm{d}x\,\mathrm{d}y.
\]

\section*{Level sets and orthogonality}
Writing $w=e^{z}=e^{x}(\cos y+i\sin y)=:u(x,y)+iv(x,y)$, we have
\[
u(x,y)=e^{x}\cos y,\qquad v(x,y)=e^{x}\sin y,
\]
\[
r=e^{x},\qquad \varphi=y\ (\text{mod }2\pi).
\]
The level sets $x=\text{const}$ and $y=\text{const}$ (the rectangular grid in the $z$-plane) map to:
\begin{itemize}
	\item $x=a\ \Rightarrow\ r=e^{a}$: concentric circles of radius $e^{a}$,
	\item $y=b\ \Rightarrow\ \varphi=b$: rays from the origin at angle $b$.
\end{itemize}
These two families are orthogonal both in the $z$- and $w$-planes (conformality preserves angles).

\section*{Images of lines and circles}
\subsection*{Lines}
\begin{itemize}
	\item \textbf{Vertical:} $x=a$ map to circles $\{\,|w|=e^{a}\,\}$.
	\item \textbf{Horizontal:} $y=b$ map to rays $\{\,\arg w=b\,\}$.
	\item \textbf{Oblique:} $y=mx+b$ map to logarithmic spirals
	\[
	w=e^{x}e^{i(mx+b)}\ \Rightarrow\ \arg w=b+m\ln|w|\quad(\text{log spiral}).
	\]
\end{itemize}

\subsection*{Circles in $z$ and their images}
A general circle in the $z$-plane $z(\theta)=a+\rho e^{i\theta}$ yields
\[
w(\theta)=e^{a}\,e^{\rho\cos\theta}\,e^{i(\Im a+\rho\sin\theta)}.
\]
The radius $|w|$ oscillates between $e^{\Re a-\rho}$ and $e^{\Re a+\rho}$ and the angle varies as $\Im a+\rho\sin\theta$. In general this is \emph{not} a circle (it is a smooth closed curve inside the annulus), except for the limiting case of a “circle through $\infty$” (i.e.\ a vertical line), which maps to a concentric circle in $w$.

\section*{Images of basic regions}
\begin{itemize}
	\item \textbf{Horizontal strips} $y\in(\beta,\beta+2\pi)$: the image is $\mathbb{C}\setminus\{0\}$ minus a single ray at argument $\beta$ (a slit plane).
	\item \textbf{Vertical strips} $x\in(a,b)$: the image is an \emph{annulus}
	\[
	A=\{\, e^{a}<|w|<e^{b}\,\}.
	\]
	\item \textbf{Rectangle} $R=\{a<x<b,\ \gamma<y<\delta\}$: the image is a \emph{sector of an annulus}
	\[
	\{\, e^{a}<|w|<e^{b},\ \gamma<\arg w<\delta \,\}.
	\]
	\item \textbf{Half-plane} $x>a$: the image is the \emph{exterior} of the disk $\{|w|>e^{a}\}$.
	\item \textbf{Disk} $\{|w|<R\}$: the preimage is the half-plane $x<\ln R$.
\end{itemize}

\section*{Inverse branch: the logarithm}
The inverse of $e^{z}$ is the multivalued logarithm:
\[
\log w=\ln|w|+i(\arg w + 2\pi k),\quad k\in\mathbb{Z}.
\]
Upon choosing a principal branch (e.g.\ cutting along $\arg w=\beta$), we get a single-valued holomorphic function $\Log_{\beta}$ on $\mathbb{C}\setminus\{re^{i\beta}:r\ge 0\}$ with
\[
\Log_{\beta}(e^{z})=z \quad \text{for } \Im z\in(\beta,\beta+2\pi).
\]

\section*{“Two grids” $\to$ a polar grid}
The rectangular grid in $z$—the families $x=\text{const}$ and $y=\text{const}$—maps exactly to the polar grid in $w$: concentric circles $r=\text{const}$ and rays $\varphi=\text{const}$.
\[
\{x=\text{const}\}\ \mapsto\ \{r=\text{const}\},\qquad
\{y=\text{const}\}\ \mapsto\ \{\varphi=\text{const}\}.
\]
This is a textbook instance of conformality: angles between the families are preserved, and the scale grows like $r$ as one moves away from $0$.

\section*{Metric and harmonic features}
\begin{itemize}
	\item \textbf{Lengths:} a small segment of length $\ell$ in $z$ maps to a segment of length $\approx r\,\ell$.
	\item \textbf{Areas:} a small area $S$ maps to an area $\approx r^{2}S$.
	\item \textbf{Cauchy--Riemann:} $u=e^{x}\cos y$, $v=e^{x}\sin y$ satisfy the C--R equations; $u$ and $v$ are harmonic and conjugate, hence their level sets are orthogonal.
\end{itemize}

\section*{Classical correspondences}
\begin{enumerate}
	\item \textbf{Rays} $\{\arg w=\varphi_{0}\}$ $\leftrightarrow$ \textbf{horizontal lines} $\{y=\varphi_{0}+2\pi k\}$.
	\item \textbf{Concentric circles} $\{|w|=R\}$ $\leftrightarrow$ \textbf{vertical lines} $\{x=\ln R\}$.
	\item \textbf{Annulus} $R_{1}<|w|<R_{2}$ $\leftrightarrow$ \textbf{vertical strip} $\ln R_{1}<x<\ln R_{2}$.
	\item \textbf{Sector} $\{\alpha<\arg w<\beta\}$ $\leftrightarrow$ \textbf{horizontal strip} $\{\alpha<y<\beta\}$.
\end{enumerate}

\section*{Remark on the Riemann sphere}
After compactification (adding the point $\infty$), a vertical line $x=a$ is a circle through $\infty$; its image under $e^{z}$ is the circle $\{|w|=e^{a}\}$ in the $w$-plane (not passing through $\infty$). This explains why \emph{only} those “circles through $\infty$” in $z$ become circles in $w$.

\section*{Counterexample: a circle in $z$ does not map to a circle in $w$}
For $z(\theta)=a+\rho e^{i\theta}$ we have
\[
|w(\theta)|=e^{\Re a+\rho\cos\theta},\qquad
\arg w(\theta)=\Im a+\rho\sin\theta,
\]
which does not satisfy a circle equation in $(u,v)$; the radius varies exponentially, producing a “rippled” closed curve within the annulus $e^{\Re a-\rho}\le |w|\le e^{\Re a+\rho}$.

\bigskip

\noindent\textbf{Summary.} The map $w=e^{z}$ is conformal, converting the rectangular grid to the polar grid and realizing the natural change $(x,y)\mapsto(r,\varphi)=(e^{x},y)$. Vertical lines go to concentric circles, horizontal lines to rays, oblique lines to logarithmic spirals, and rectangles to annular sectors. Only “circles through $\infty$” in $z$ (i.e.\ vertical lines) become circles in $w$; generic circles do not.
