\chapter*{Fourier Conventions}
\addcontentsline{toc}{chapter}{Fourier Conventions}
\label{front:vol03-fourier-conventions}

Fourier formulas are normalization-sensitive. Volume III therefore fixes one
whole-space convention and uses it throughout Fourier transforms,
tempered-distribution transforms, fundamental solutions, and transform methods
for PDE.

\paragraph{Whole-space transform.}
For \(f\in L^1(\mathbb R^n)\),
\[
\boxed{
\widehat f(\xi)
=\int_{\mathbb R^n} f(x)e^{-2\pi i x\cdot\xi}\,dx
}
\]
and, whenever an inversion theorem applies,
\[
\boxed{
f(x)=\int_{\mathbb R^n}\widehat f(\xi)e^{2\pi i x\cdot\xi}\,d\xi.
}
\]
The same normalization extends to \(\mathcal S'(\mathbb R^n)\) by duality.

\paragraph{Derivative and Laplacian multipliers.}
\[
\widehat{\partial_j f}(\xi)
=2\pi i\xi_j\widehat f(\xi),
\qquad
\widehat{\Delta f}(\xi)
=-4\pi^2|\xi|^2\widehat f(\xi),
\]
so
\[
\boxed{\widehat{-\Delta f}(\xi)=4\pi^2|\xi|^2\widehat f(\xi).}
\]

\paragraph{Translation, modulation, scaling, convolution.}
For suitable functions,
\[
\widehat{f(\,\cdot-a\,)}(\xi)
=e^{-2\pi i a\cdot\xi}\widehat f(\xi),
\qquad
\widehat{e^{2\pi i b\cdot x}f(x)}(\xi)=\widehat f(\xi-b),
\]
\[
\widehat{f(a\,\cdot)}(\xi)=|a|^{-n}\widehat f(\xi/a)
\quad (a\ne0),
\qquad
\widehat{f*g}=\widehat f\,\widehat g.
\]

\paragraph{Gaussian and heat kernel.}
\[
\widehat{e^{-\pi|x|^2}}(\xi)=e^{-\pi|\xi|^2}.
\]
For
\[
K_t(x)=(4\pi t)^{-n/2}e^{-|x|^2/(4t)}
\]
one has
\[
\boxed{\widehat K_t(\xi)=e^{-4\pi^2t|\xi|^2}.}
\]
Thus the heat equation multiplier is \(e^{-4\pi^2t|\xi|^2}\).

\paragraph{Periodic Fourier series.}
For real \(2\pi\)-periodic functions,
\[
a_n=\frac1\pi\int_{-\pi}^{\pi}f(x)\cos(nx)\,dx,\qquad
b_n=\frac1\pi\int_{-\pi}^{\pi}f(x)\sin(nx)\,dx,
\]
with
\[
\frac{a_0}{2}+\sum_{n\ge1}(a_n\cos(nx)+b_n\sin(nx)).
\]
This is the periodic-series normalization; it is compatible with, but not a
replacement for, the whole-space transform convention.

\paragraph{Convention audit rule.}
Any formula elsewhere in Volume III involving differentiation, the Laplacian,
Dirac masses, Gaussians, heat kernels, or PDE multipliers must reduce to these
identities. A Fourier-normalization change is therefore global, not local.
