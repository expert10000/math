\chapter{Green Functions}
\label{ch:iii25}

Green functions adapt fundamental solutions to domains and boundary conditions. They separate the singular response to a point source from a correction chosen to enforce the boundary data.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct Green functions from fundamental solutions plus boundary-correction terms in elementary domains;
\item verify the defining differential equation and boundary conditions separately;
\item derive representation formulas for solutions using Green functions;
\item interpret symmetry and positivity properties when the operator and domain permit them;
\item distinguish free-space kernels from domain-dependent Green functions;
\item check jump or derivative conditions that determine one-dimensional Green functions uniquely;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
For the general representation in this chapter use a real bounded smooth domain and $L=-\Delta$ with homogeneous Dirichlet conditions. Its Green function satisfies $-\Delta_xG(x,y)=\delta_y$ and vanishes in $x$ on the boundary. Existence and smoothness off the pole are standard Dirichlet Green-function theorems, with full proofs deferred. If $f\in C_c^\infty(\Omega)$ and $u$ is the classical zero-boundary solution, $u(x)=\int_\Omega G(x,y)f(y)\,dy$. For smooth boundary data $g$, the added term is $-\int_{\partial\Omega}g(y)\partial_{\nu_y}G(x,y)\,dS_y$. For a general non-self-adjoint operator one must use the adjoint equation in the integration variable; an inverse must exist. For complex self-adjoint kernels symmetry is Hermitian, $G(x,y)=\overline{G(y,x)}$.

\section{Green function definition}
For a domain operator, $G(x,y)$ solves the differential equation in $x$ with a point source at $y$ and prescribed boundary behavior.

\section{Singular plus regular decomposition}
Typically $G(x,y)=E(x-y)-H(x,y)$, where $E$ is a whole-space fundamental solution and $H$ corrects the boundary values.

\section{Representation formula}
Solutions are expressed by integrating the forcing against $G$, plus boundary terms when nonhomogeneous data are present.

\section{Symmetry}
Self-adjoint elliptic problems often yield $G(x,y)=G(y,x)$.

Maximum principles can imply nonnegativity of Dirichlet Green functions for positive elliptic operators.

\section{One-dimensional construction}
For second-order ODEs, Green kernels are assembled from two homogeneous solutions matched continuously with a derivative jump.

\begin{example}[Dirichlet Green function for minus second derivative]\label{ex:iii25-expand-01}
For
\[
-u''=f\quad\text{on }(0,1),\qquad u(0)=u(1)=0,
\]
the Green kernel is
\[
G(x,\xi)=
\begin{cases}
x(1-\xi),&x\le\xi,\\
\xi(1-x),&x\ge\xi.
\end{cases}
\]
It is continuous at \(x=\xi\), vanishes at both endpoints, and its derivative
jump satisfies
\[
-\bigl(G_x(\xi^+,\xi)-G_x(\xi^-,\xi)\bigr)=1.
\]
These three checks certify the point-source equation and boundary conditions.
\end{example}

Changing Dirichlet to Neumann conditions changes the correction and may require compatibility conditions.

\section{Operator inverse viewpoint}
The Green kernel is the integral kernel of the inverse or resolvent of the boundary-value operator.

\section{Core structural results}

\begin{theorem}[Green representation]\label{thm:iii25-01}
For the real Dirichlet Laplacian and smooth data specified above, the classical zero-boundary solution satisfies $u(x)=\int_\Omega G(x,y)f(y)dy$.
\begin{proof}
Proof sketch: excise a ball around the pole and apply the second Green identity using symmetry of this real Laplace kernel. The outer boundary terms vanish for homogeneous data, the small sphere term tends to u(x), and the volume integral tends to the displayed integral since the kernel singularity is locally integrable. The Green-function existence theorem is used, not proved by this calculation.
\end{proof}
\end{theorem}

\begin{theorem}[Symmetry for self-adjoint problems]\label{thm:iii25-02}
For the real Dirichlet Laplacian in the smooth-domain setting above, $G(x,y)=G(y,x)$.
\begin{proof}
Apply Green's identity to the two Green functions with poles at $x$ and $y$, excise small neighborhoods of the poles, and pass to the limit; boundary terms vanish by the shared boundary condition.
\end{proof}
\end{theorem}

\begin{theorem}[Derivative jump in one dimension]\label{thm:iii25-03}
For $-u''=f$ on an interval, the Green kernel is continuous across $x=y$ and its first derivative has the jump required to create a Dirac mass.
\begin{proof}
Integrating the equation $-G_{xx}=\delta_y$ across a small interval around $y$ gives $-(G_x(y^+)-G_x(y^-))=1$, while absence of a delta-prime term forces continuity.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[One-dimensional Dirichlet Green function]\label{ex:iii25-worked-01}
For \(-u''=f\) on \((0,L)\) with \(u(0)=u(L)=0\), the Green kernel is
\[
G(x,y)=
\begin{cases}
x(L-y)/L,&x\le y,\\
y(L-x)/L,&x\ge y.
\end{cases}
\]
It is continuous at \(x=y\), vanishes at the endpoints, and its \(x\)-derivative
has jump \(-1\), so \(-\partial_x^2G(\cdot,y)=\delta_y\).
\end{example}

\begin{example}[Green representation]\label{ex:iii25-worked-02}
With the preceding kernel,
\[
u(x)=\int_0^LG(x,y)f(y)\,dy
\]
satisfies the endpoint conditions automatically. Differentiating
distributionally in \(x\) gives
\[
-u''(x)=\int_0^L\delta_y(x)f(y)\,dy=f(x).
\]
\end{example}

\begin{example}[Symmetry reflects self-adjointness]\label{ex:iii25-worked-03}
For the Dirichlet Laplacian,
\[
G(x,y)=G(y,x).
\]
At the operator level this is the kernel form of
\(\langle L^{-1}f,g\rangle=\langle f,L^{-1}g\rangle\), inherited from the
self-adjointness of the boundary-value problem.
\end{example}

\begin{example}[Positivity gives comparison]\label{ex:iii25-worked-04}
If \(G(x,y)\ge0\) and \(f\ge0\), then the representation
\(u(x)=\int G(x,y)f(y)\,dy\) gives \(u\ge0\). Thus positivity of a Green kernel
encodes a maximum/comparison principle for the inverse operator.
\end{example}

\section{Solved dossiers}
Each dossier combines several ideas from the chapter in a longer argument. The aim is to make hypotheses, calculations, and proof strategy explicit.

\begin{problem}[One-dimensional Dirichlet Green function]\label{prob:iii25-dossier-01}
Derive or verify the Dirichlet Green kernel for \(-u''\) on \((0,L)\).
\end{problem}
\begin{solution}
Solve the homogeneous equation on each side of \(y\), impose endpoint
conditions and continuity, then choose the derivative jump \(-1\) to create
the Dirac mass.
\end{solution}

\begin{problem}[Green representation]\label{prob:iii25-dossier-02}
Verify the Green representation solves the one-dimensional Dirichlet problem.
\end{problem}
\begin{solution}
Boundary values follow from the kernel. Applying
\(-\partial_x^2\) under the integral gives the delta identity and returns
\(f\).
\end{solution}

\begin{problem}[Symmetry reflects self-adjointness]\label{prob:iii25-dossier-03}
Explain why a Green kernel for a self-adjoint problem is expected to be symmetric.
\end{problem}
\begin{solution}
The inverse of a self-adjoint invertible operator is self-adjoint; writing
the inverse as an integral operator forces kernel symmetry (under standard
regularity assumptions).
\end{solution}

\begin{problem}[Positivity gives comparison]\label{prob:iii25-dossier-04}
Show how a nonnegative Green kernel implies positivity preservation.
\end{problem}
\begin{solution}
The integral of the product of two nonnegative functions is nonnegative at
every \(x\).
\end{solution}

\begin{problem}[Derivative jump]\label{prob:iii25-dossier-05}
For \(-u''\), what derivative jump must \(G(\cdot,y)\) have at \(x=y\)?
\end{problem}
\begin{solution}
Integrating \(-G''=\delta_y\) across a small interval gives
\(G_x(y^+)-G_x(y^-)=-1\).
\end{solution}

\begin{problem}[Boundary adaptation]\label{prob:iii25-dossier-06}
How does a Green function differ from a whole-space fundamental solution?
\end{problem}
\begin{solution}
It includes boundary-correction terms so that, in addition to the delta
equation, it satisfies the prescribed boundary conditions.
\end{solution}

\begin{problem}[Non-self-adjoint warning]\label{prob:iii25-dossier-07}
Must a Green kernel be symmetric for a non-self-adjoint operator?
\end{problem}
\begin{solution}
No. Symmetry is tied to self-adjointness and compatible boundary
conditions.
\end{solution}

\begin{problem}[Kernel singularity]\label{prob:iii25-dossier-08}
Where is a Green function singular?
\end{problem}
\begin{solution}
Typically along the diagonal \(x=y\), with the same local singularity as a
fundamental solution; boundary terms modify only the global behavior.
\end{solution}

Continuity removes that unwanted singularity.

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Free-space kernel]\label{exr:iii25-expand-10}
Why does a fundamental solution generally violate domain boundary conditions?
\end{exercise}
\begin{hint}
It knows only the operator and source, not the boundary.
\end{hint}
\begin{solution}
A correction term is needed to form G.
\end{solution}

\begin{exercise}[Neumann compatibility]\label{exr:iii25-expand-11}
Why can a pure Neumann Green problem require a zero-mean forcing condition?
\end{exercise}
\begin{hint}
Constants lie in the kernel of the Neumann Laplacian.
\end{hint}
\begin{solution}
For $-\Delta$ with zero Neumann data, integration yields $\int_\Omega f=0$. A normalized Neumann Green function instead solves $-\Delta_xG(x,y)=\delta_y-1/|\Omega|$, $\partial_\nu G=0$, and $\int_\Omega G(x,y)\,dx=0$ on a connected smooth bounded domain. A pure delta source alone is incompatible with zero flux.
\end{solution}

\begin{exercise}[Symmetry can fail]\label{exr:iii25-expand-12}
Why can Green symmetry fail for non-self-adjoint problems?
\end{exercise}
\begin{hint}
The inverse operator need not be self-adjoint.
\end{hint}
\begin{solution}
Its kernel has no reason to be symmetric.
\end{solution}

\begin{exercise}[Eigenfunction expansion]\label{exr:iii25-expand-16}
Explain formally why a self-adjoint Green kernel can be expanded as a sum phi\_k(x)phi\_k(y)/lambda\_k.
\end{exercise}
\begin{hint}
Expand f in eigenfunctions and invert each scalar eigenvalue.
\end{hint}
\begin{solution}
Assume an invertible real self-adjoint operator with a complete orthonormal eigenbasis and eigenvalues $\lambda_k\ne0$. Then $L^{-1}f=\sum_k\lambda_k^{-1}\langle f,\phi_k\rangle\phi_k$ in L2. Reading this as a kernel is formal unless kernel convergence is established; a Hilbert--Schmidt inverse gives an L2 kernel expansion. For complex eigenfunctions the second factor is conjugated.
\end{solution}
