\chapter{Sturm–Liouville Green Kernels}
\label{ch:iii26}

Sturm--Liouville Green kernels combine boundary-value ODE theory, self-adjointness, Wronskians, and spectral expansions. They are the one-dimensional prototype of resolvent kernels for elliptic operators.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct Green kernels for second-order Sturm--Liouville boundary-value problems from two homogeneous solutions;
\item compute the Wronskian and use it to normalize the kernel;
\item verify continuity of the kernel and the prescribed derivative jump across the diagonal;
\item check boundary conditions in each variable before accepting a proposed kernel;
\item derive solution representations by integrating the forcing term against the kernel;
\item distinguish regular, periodic, Dirichlet, and Neumann kernel constructions by their boundary conditions;
\end{itemize}

\section{Wronskian structure}
A weighted Wronskian is constant for homogeneous solutions and controls the denominator of the Green kernel.

\begin{example}[Wronskian normalization for the Dirichlet Laplacian]\label{ex:iii26-expand-01}
For \(-y''\) on \((0,1)\) with Dirichlet conditions, choose
\(u(x)=x\) satisfying the left condition and \(v(x)=1-x\) satisfying the right
condition. Their Wronskian is
\[
W=u v'-u'v=x(-1)-1(1-x)=-1,
\]
constant as expected. The Green kernel is proportional to
\(u(x_<)v(x_>)\), and the constant Wronskian supplies exactly the normalization
needed for the derivative jump.
\end{example}

\section{Matching conditions}
Continuity at the source and a derivative jump enforce the Dirac equation.


% BEGIN VOL03-EXPANSION III26-example-02
\begin{example}[Matching conditions determine the kernel]\label{ex:iii26-expand-02}
A second-order Green kernel is built from one homogeneous branch on each side of
\(x=\xi\). Two conditions meet at the source: continuity of \(G\), which
prevents a delta-prime term, and the flux jump
\[
-p(\xi)\bigl(G_x(\xi^+,\xi)-G_x(\xi^-,\xi)\bigr)=1,
\]
which creates the desired Dirac mass. Boundary conditions determine the outer
normalizations. Together these conditions uniquely fix the nonresonant kernel.
\end{example}

\begin{example}[Resonance blocks inversion]\label{ex:iii26-expand-03}
Suppose \(L\phi=\lambda_0\phi\) and one tries to invert \(L-\lambda_0\).
The eigenfunction coefficient equation becomes
\[
(\lambda_0-\lambda_0)c=\langle f,\phi\rangle.
\]
A solution can exist only if \(\langle f,\phi\rangle=0\). Even then the
solution is nonunique because any multiple of \(\phi\) may be added. Thus the
pole of the Green resolvent at an eigenvalue is the analytic signature of
resonance.
\end{example}

\section{Core structural results}

\begin{theorem}[Green-kernel construction]\label{thm:iii26-01}
Let $u$ satisfy the left boundary condition and $v$ the right one. Away from eigenvalues, $G(x,\xi)$ is proportional to $u(x_<)v(x_>)$ divided by the weighted Wronskian.
\begin{proof}
Each branch solves the homogeneous equation and the boundary condition on its side. Continuity and the prescribed flux jump at $x=\xi$ determine the common normalization by the Wronskian.
\end{proof}
\end{theorem}

\begin{theorem}[Symmetry]\label{thm:iii26-02}
For a regular self-adjoint Sturm--Liouville problem, $G(x,\xi)=G(\xi,x)$.
\begin{proof}
The product formula is symmetric after swapping $x$ and $\xi$, or equivalently follows from self-adjointness of the inverse.
\end{proof}
\end{theorem}

\begin{theorem}[Spectral resolvent expansion]\label{thm:iii26-03}
If $\{\phi_n\}$ is an orthonormal eigenbasis and $\lambda$ avoids the spectrum, then the Green kernel has formal expansion $G_\lambda(x,\xi)=\sum_n\phi_n(x)\phi_n(\xi)/(\lambda_n-\lambda)$.
\begin{proof}
Expand the forcing in the eigenbasis, solve each scalar coefficient equation, and read off the integral kernel of the resulting inverse.
\end{proof}
\end{theorem}

What product form is used for x<xi?

\end{solution}

\begin{exercise}[Continuity]\label{exr:iii26-expand-03}
What matching condition prevents a delta-prime?
\end{exercise}
\begin{hint}
Keep the kernel value continuous.
\end{hint}
\begin{solution}
G(xi-,xi)=G(xi+,xi).
\end{solution}

\begin{exercise}[Flux jump]\label{exr:iii26-expand-04}
What creates the delta source?
\end{exercise}
\begin{hint}
A prescribed derivative/flux jump.
\end{hint}
\begin{solution}
The weighted derivative jump equals the inverse leading coefficient normalization.
\end{solution}

\begin{exercise}[Resonance]\label{exr:iii26-expand-05}
What happens to the resolvent at an eigenvalue?
\end{exercise}
\begin{hint}
A spectral denominator vanishes.
\end{hint}
\begin{solution}
The inverse fails unless forcing satisfies compatibility, and solutions are nonunique.
\end{solution}

The derivative cancels to zero.

\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
The qG term vanishes in the shrinking interval and the pG' flux contributes the unit mass.

\end{hint}

\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Eigenvalue denominator]\label{exr:iii26-expand-10}
Why is the spectral Green expansion invalid at an eigenvalue?
\end{exercise}
\begin{hint}
A denominator lambda\_k-lambda becomes zero.
\end{hint}
\begin{solution}
The resolvent does not exist as a bounded inverse there.
\end{solution}

\begin{exercise}[Nonself-adjoint symmetry]\label{exr:iii26-expand-11}
Why should one not expect a symmetric Green kernel for a nonself-adjoint problem?
\end{exercise}
\begin{hint}
The inverse operator is not self-adjoint.
\end{hint}
\begin{solution}
Kernel symmetry encodes self-adjointness.
\end{solution}

\begin{exercise}[Wrong boundary solution]\label{exr:iii26-expand-12}
What fails if both homogeneous factors satisfy the same endpoint condition?
\end{exercise}
\begin{hint}
The other boundary condition is not enforced.
\end{hint}

The piecewise kernel will generally violate the required right or left boundary data.
\end{solution}
