\chapter{Convergence Theorems}
\label{ch:iii04}

Monotone convergence, Fatou's lemma, and dominated convergence govern the passage of limits through integrals. Together they separate the roles of positivity, one-sided control, and integrable domination.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item apply monotone, Fatou, and dominated convergence theorems after verifying their distinct hypotheses;
\item construct dominating functions suitable for dominated-convergence arguments;
\item use Fatou's lemma to obtain one-sided integral inequalities from pointwise limits;
\item distinguish pointwise, almost-everywhere, and norm convergence in examples where the conclusions differ;
\item build counterexamples showing why domination or monotonicity cannot simply be omitted;
\item choose the weakest convergence theorem that actually applies to a given sequence of measurable functions;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
All functions in the convergence theorems are measurable on a fixed measure space; signed functions are real-valued. Limits and inequalities may hold a.e. after removal of one measurable null set. MCT and Fatou allow infinite nonnegative integrals; DCT requires a single nonnegative $g\in L^1$ dominating every $|f_n|$ a.e., and also gives $\|f_n-f\|_1\to0$.

\section{Monotone convergence}
Increasing nonnegative sequences allow direct interchange of pointwise limit and integral.


% BEGIN VOL03-EXPANSION III04-example-01
\begin{example}[Monotone convergence for a nonnegative series]\label{ex:iii04-expand-01}
Let
\[
f_n=\sum_{k=1}^n2^{-k}1_{[0,k]}.
\]
Then \(0\le f_n\uparrow f\). By MCT,
\[
\int f=\lim_n\sum_{k=1}^n k2^{-k}=2.
\]
No separate dominator is needed: positivity and monotonicity are the decisive
hypotheses.
\end{example}
% END VOL03-EXPANSION III04-example-01
\section{Fatou's lemma}
The integral of a nonnegative liminf cannot exceed the liminf of the integrals.

\section{Dominated convergence}
Almost-everywhere convergence under a single integrable absolute bound permits limit-integral interchange for signed functions.


% BEGIN VOL03-EXPANSION III04-example-02
\begin{example}[Dominated convergence with an explicit dominator]\label{ex:iii04-expand-02}
On \([0,1]\), take \(f_n(x)=x/(1+nx)\). Then \(f_n(x)\to0\) for every \(x\),
and \(0\le f_n(x)\le x\). Since \(x\) is integrable on \([0,1]\), DCT gives
\(\int f_n\to0\). The proof is complete only after the dominating function has
been named and checked to be integrable.
\end{example}
% END VOL03-EXPANSION III04-example-02
\section{Almost-everywhere convergence}
Null exceptional sets do not affect the integral but still require measurable representatives.

\section{Convergence in measure}
Convergence in measure means $\mu(|f_n-f|>\varepsilon)\to0$ for each $\varepsilon>0$. A.e. convergence implies this on finite-measure spaces, but not on arbitrary infinite-measure spaces.

\section{$L^1$ convergence}
Norm convergence controls the integral of absolute error and implies convergence in measure.

\section{Series applications}
MCT handles nonnegative series through increasing partial sums; DCT handles dominated signed series.

\section{Counterexamples}
Moving spikes show that pointwise convergence alone does not control integrals.


% BEGIN VOL03-EXPANSION III04-example-03
\begin{example}[A moving spike defeats pointwise convergence]\label{ex:iii04-expand-03}
Set \(f_n=n1_{(0,1/n)}\) on \([0,1]\). Then \(f_n\to0\) almost everywhere,
but every integral equals \(1\). The mass concentrates into a shrinking region,
so there is no single integrable dominator. This is the canonical warning that
pointwise or almost-everywhere convergence alone does not control integrals.
\end{example}

\section{Core structural results}

\begin{theorem}[Monotone convergence theorem]\label{thm:iii04-01}
If $0\le f_n\uparrow f$, then $\int f_n\uparrow\int f$.
\begin{proof}
Let $L=\lim_n\int f_n\le\int f$. For a nonnegative simple $s\le f$ and $0<c<1$, the sets $A_n=\{f_n\ge cs\}$ increase to $X$. Hence $\int f_n\ge c\int_{A_n}s\to c\int s$ by continuity from below on each of the finitely many level sets. Thus $L\ge c\int s$. Take the supremum over $s$ and let $c\uparrow1$; this also covers infinite integrals.
\end{proof}
\end{theorem}

\begin{theorem}[Fatou's lemma]\label{thm:iii04-02}
For nonnegative measurable $f_n$, $\int\liminf f_n\le\liminf\int f_n$.
\begin{proof}
Set $g_N=\inf_{n\ge N}f_n$. Then $g_N\uparrow\liminf f_n$ and $\int g_N\le\inf_{n\ge N}\int f_n$; apply MCT and let $N\to\infty$.
\end{proof}
\end{theorem}

\begin{theorem}[Dominated convergence theorem]\label{thm:iii04-03}
If $f_n\to f$ a.e. and $|f_n|\le g\in L^1$, then $\int f_n\to\int f$.
\begin{proof}
Apply Fatou to the nonnegative sequences $g\pm f_n$ to obtain matching upper and lower bounds; also $|f|\le g$ a.e., so $f$ is integrable.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Monotone convergence on expanding indicators]\label{ex:iii04-worked-01}
Let \(f_n=\mathbf1_{[0,n]}\) on \(\mathbb R\). Then \(f_n\uparrow1\)
pointwise and
\[
\int f_n\,dx=n\uparrow\infty=\int1\,dx.
\]
This is monotone convergence with an infinite limiting integral.
\end{example}

\begin{example}[A moving spike defeats naive dominated convergence]\label{ex:iii04-worked-02}
On \((0,1)\), let \(f_n=n\mathbf1_{(0,1/n)}\). Then \(f_n(x)\to0\) for
every \(x>0\), but
\[
\int_0^1 f_n\,dx=1.
\]
There is no single integrable dominator. Pointwise convergence alone therefore
does not justify passage of the limit through the integral.
\end{example}

\begin{example}[Fatou catches lost mass]\label{ex:iii04-worked-03}
For nonnegative measurable \(f_n\),
\[
\int\liminf f_n\le\liminf\int f_n.
\]
For the moving spikes \(n\mathbf1_{(0,1/n)}\), the left side is \(0\) and the
right side is \(1\), showing that Fatou permits strict inequality when mass
concentrates and disappears pointwise.
\end{example}

\begin{example}[Dominated convergence with a genuine dominator]\label{ex:iii04-worked-04}
Let \(f_n(x)=x^n\) on \([0,1]\). Then \(f_n\to0\) on \([0,1)\) and the
exceptional point \(x=1\) has measure zero. Since \(|f_n|\le1\in L^1[0,1]\),
dominated convergence gives
\[
\int_0^1x^n\,dx\to0,
\]
consistent with \(1/(n+1)\to0\).
\end{example}

\section{Solved dossiers}
Each dossier combines several ideas from the chapter in a longer argument. The aim is to make hypotheses, calculations, and proof strategy explicit.

\begin{problem}[Monotone convergence on expanding indicators]\label{prob:iii04-dossier-01}
Verify the monotone convergence theorem for \(f_n=\mathbf1_{[0,n]}\).
\end{problem}
\begin{solution}
The functions increase pointwise to \(1\), and their integrals are \(n\),
which increase to \(\infty=\int_{\mathbb R}1\,dx\).
\end{solution}

\begin{problem}[A moving spike defeats naive dominated convergence]\label{prob:iii04-dossier-02}
Give a pointwise-convergent sequence whose integrals do not converge to the integral of the pointwise limit.
\end{problem}
\begin{solution}
Take \(f_n=n\mathbf1_{(0,1/n)}\). Then \(f_n\to0\) pointwise on \((0,1]\),
while every integral equals \(1\).
\end{solution}

\begin{problem}[Fatou catches lost mass]\label{prob:iii04-dossier-03}
Apply Fatou's lemma to the moving-spike sequence \(n\mathbf1_{(0,1/n)}\).
\end{problem}
\begin{solution}
The pointwise liminf is \(0\), so the left side is \(0\); each integral is
\(1\), so the right side is \(1\).
\end{solution}

\begin{problem}[Dominated convergence with a genuine dominator]\label{prob:iii04-dossier-04}
Use dominated convergence to compute \(\lim_n\int_0^1x^n\,dx\).
\end{problem}
\begin{solution}
The functions converge to \(0\) a.e. and are dominated by \(1\in L^1\), so
the limit is \(0\).
\end{solution}

\begin{problem}[Monotone convergence from below]\label{prob:iii04-dossier-05}
If \(0\le f_n\uparrow f\), why can no integrable domination hypothesis be required?
\end{problem}
\begin{solution}
Monotonicity prevents cancellation and makes the integrals increase; the
theorem is built directly into the definition of the nonnegative integral via
simple approximation.
\end{solution}

\begin{problem}[Reverse Fatou under domination]\label{prob:iii04-dossier-06}
Suppose \(|f_n|\le g\in L^1\). Explain how Fatou applied to \(g-f_n\) helps prove dominated convergence.
\end{problem}
\begin{solution}
Since \(g-f_n\ge0\), Fatou bounds the liminf of its integrals. Rearranging
yields a limsup bound for \(\int f_n\); applying the same idea to \(-f_n\)
gives convergence.
\end{solution}

\begin{problem}[Convergence in \(L^1\) implies integral convergence]\label{prob:iii04-dossier-07}
Prove \(\|f_n-f\|_1\to0\) implies \(\int f_n\to\int f\).
\end{problem}
\begin{solution}
Use
\(\left|\int(f_n-f)\right|\le\int|f_n-f|=\|f_n-f\|_1\to0\).
\end{solution}

\begin{problem}[Pointwise convergence is not \(L^1\) convergence]\label{prob:iii04-dossier-08}
Use the moving spike to show pointwise convergence need not imply \(L^1\) convergence.
\end{problem}
\begin{solution}
For \(f_n=n\mathbf1_{(0,1/n)}\), the pointwise limit is \(0\) but
\(\|f_n\|_1=1\) for all \(n\).
\end{solution}

\begin{problem}[Monotone convergence theorem proof dossier]\label{prob:iii04-dossier-09}
Prove the following structural statement and identify the step where its main hypothesis is used: If $0\le f_n\uparrow f$, then $\int f_n\uparrow\int f$.

If nonnegative f\_n increases to f and integral f\_n tends to 7, what is integral f?

If nonnegative f\_n have liminf of integrals equal to 5, what upper bound follows for integral liminf f\_n?

The integrals converge to integral f.

\subsection*{Challenge problems}

\begin{exercise}[Scheffe insight]\label{exr:iii04-expand-15}
If nonnegative integrable f\_n tend to f almost everywhere and their integrals converge to the finite integral of f, what stronger convergence follows?
\end{exercise}
\begin{hint}
Compare min(f\_n,f).
\end{hint}
\begin{solution}
MCT/DCT gives $\int\min(f_n,f)\to\int f$ since $\min(f_n,f)\le f\in L^1$. Therefore $\|f_n-f\|_1=\int f_n+\int f-2\int\min(f_n,f)\to0$.
\end{solution}

\begin{exercise}[Uniform integrability]\label{exr:iii04-expand-16}
Why does uniform integrability address the moving-spike failure?
\end{exercise}
\begin{hint}
It prevents substantial integral mass on tiny sets.
\end{hint}
\begin{solution}
Combined with convergence in measure, it can force L1 convergence.
\end{solution}
