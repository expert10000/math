\chapter{Convergence Theorems}
\label{ch:iii04}

Monotone convergence, Fatou's lemma, and dominated convergence govern the passage of limits through integrals. Together they separate the roles of positivity, one-sided control, and integrable domination.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item apply monotone, Fatou, and dominated convergence theorems after verifying their distinct hypotheses;
\item construct dominating functions suitable for dominated-convergence arguments;
\item use Fatou's lemma to obtain one-sided integral inequalities from pointwise limits;
\item distinguish pointwise, almost-everywhere, and norm convergence in examples where the conclusions differ;
\item build counterexamples showing why domination or monotonicity cannot simply be omitted;
\item choose the weakest convergence theorem that actually applies to a given sequence of measurable functions;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\section{Monotone convergence}
Increasing nonnegative sequences allow direct interchange of pointwise limit and integral.


% BEGIN VOL03-EXPANSION III04-example-01
\begin{example}[Monotone convergence for a nonnegative series]\label{ex:iii04-expand-01}
Let
\[
f_n=\sum_{k=1}^n2^{-k}1_{[0,k]}.
\]
Then \(0\le f_n\uparrow f\). By MCT,
\[
\int f=\lim_n\sum_{k=1}^n k2^{-k}=2.
\]
No separate dominator is needed: positivity and monotonicity are the decisive
hypotheses.
\end{example}

\section{Dominated convergence}
Almost-everywhere convergence under a single integrable absolute bound permits limit-integral interchange for signed functions.

\begin{example}[Dominated convergence with an explicit dominator]\label{ex:iii04-expand-02}
On \([0,1]\), take \(f_n(x)=x/(1+nx)\). Then \(f_n(x)\to0\) for every \(x\),
and \(0\le f_n(x)\le x\). Since \(x\) is integrable on \([0,1]\), DCT gives
\(\int f_n\to0\). The proof is complete only after the dominating function has
been named and checked to be integrable.
\end{example}

\section{Counterexamples}
Moving spikes show that pointwise convergence alone does not control integrals.

\begin{example}[A moving spike defeats pointwise convergence]\label{ex:iii04-expand-03}
Set \(f_n=n1_{(0,1/n)}\) on \([0,1]\). Then \(f_n\to0\) almost everywhere,
but every integral equals \(1\). The mass concentrates into a shrinking region,
so there is no single integrable dominator. This is the canonical warning that
pointwise or almost-everywhere convergence alone does not control integrals.
\end{example}

\section{Core structural results}

\begin{theorem}[Monotone convergence theorem]\label{thm:iii04-01}
If $0\le f_n\uparrow f$, then $\int f_n\uparrow\int f$.
\begin{proof}
The integrals increase and are bounded above by $\int f$; every simple function below $f$ is eventually captured up to an arbitrarily small factor on most of its level sets, forcing equality of the limit.
\end{proof}
\end{theorem}

\begin{theorem}[Fatou's lemma]\label{thm:iii04-02}
For nonnegative measurable $f_n$, $\int\liminf f_n\le\liminf\int f_n$.
\begin{proof}
Set $g_N=\inf_{n\ge N}f_n$. Then $g_N\uparrow\liminf f_n$ and $\int g_N\le\inf_{n\ge N}\int f_n$; apply MCT and let $N\to\infty$.
\end{proof}
\end{theorem}

\begin{theorem}[Dominated convergence theorem]\label{thm:iii04-03}
If $f_n\to f$ a.e. and $|f_n|\le g\in L^1$, then $\int f_n\to\int f$.
\begin{proof}
Apply Fatou to the nonnegative sequences $g\pm f_n$ to obtain matching upper and lower bounds; also $|f|\le g$ a.e., so $f$ is integrable.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[MCT limit]\label{exr:iii04-expand-01}
If nonnegative f\_n increases to f and integral f\_n tends to 7, what is integral f?
\end{exercise}
\begin{hint}
Apply MCT.
\end{hint}
\begin{solution}
It is 7.
\end{solution}

\begin{exercise}[Fatou]\label{exr:iii04-expand-02}
If nonnegative f\_n have liminf of integrals equal to 5, what upper bound follows for integral liminf f\_n?
\end{exercise}
\begin{hint}
Use Fatou.
\end{hint}
\begin{solution}
At most 5.
\end{solution}

\begin{exercise}[DCT]\label{exr:iii04-expand-03}
If f\_n tends to f almost everywhere and |f\_n| is bounded by an L1 function, what happens to integrals?
\end{exercise}
\begin{hint}
Apply DCT.
\end{hint}
\begin{solution}
The integrals converge to integral f.
\end{solution}

\begin{exercise}[L1 to measure]\label{exr:iii04-expand-04}
What does L1 convergence imply?
\end{exercise}
\begin{hint}
Use Chebyshev.
\end{hint}
\begin{solution}
Convergence in measure.
\end{solution}

\begin{exercise}[Series MCT]\label{exr:iii04-expand-05}
Why can a nonnegative series be integrated term by term?
\end{exercise}
\begin{hint}
Partial sums increase.
\end{hint}
\begin{solution}
MCT applies to the increasing partial sums.
\end{solution}

\subsection*{Proofs}

Prove Fatou using tail infima.
\end{exercise}
\begin{hint}
Define g\_N as the infimum of the tail.
\end{hint}
\begin{solution}
g\_N increases to liminf f\_n; apply MCT and compare integrals.
\end{solution}

\begin{exercise}[DCT integrability]\label{exr:iii04-expand-07}
Show the DCT limit is integrable.
\end{exercise}
\begin{hint}
The pointwise limit is bounded by the dominator almost everywhere.
\end{hint}
\begin{solution}
Integral |f| is at most integral g.
\end{solution}

\begin{exercise}[L1 implies measure]\label{exr:iii04-expand-08}
Prove L1 convergence implies convergence in measure.
\end{exercise}
\begin{hint}
Apply Chebyshev to the difference.
\end{hint}
\begin{solution}
Bad-set measure is at most L1 error divided by epsilon.
\end{solution}

\begin{exercise}[MCT series]\label{exr:iii04-expand-09}
Explain why MCT is the natural theorem for nonnegative sums.
\end{exercise}
\begin{hint}
Partial sums form an increasing sequence.
\end{hint}
\begin{solution}
Their pointwise limit is the full series, so MCT interchanges limit and integral.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Moving spike]\label{exr:iii04-expand-10}
Give almost-everywhere convergence without integral convergence.
\end{exercise}
\begin{hint}
Use n on (0,1/n).
\end{hint}
\begin{solution}
It tends to zero almost everywhere while every integral is 1.
\end{solution}

\begin{exercise}[No dominator]\label{exr:iii04-expand-11}
Why can the moving spike not have one L1 dominator?
\end{exercise}
\begin{hint}
Any dominator must be large arbitrarily close to zero.
\end{hint}
\begin{solution}
Its lower envelope behaves like a nonintegrable 1/x scale.
\end{solution}

\begin{exercise}[Signed monotone caveat]\label{exr:iii04-expand-12}
Why is nonnegativity structural in standard MCT?
\end{exercise}
\begin{hint}
Signed infinite masses can create undefined subtraction.
\end{hint}
\begin{solution}
Additional lower-integrability hypotheses are needed for safe signed variants.
\end{solution}

\subsection*{Applications and investigations}

Why is DCT useful for limits of expectations?
\end{exercise}
\begin{hint}
Expectations are integrals.
\end{hint}
\begin{solution}
An integrable dominating random variable lets the limit pass through expectation.
\end{solution}

\begin{exercise}[Domain truncation]\label{exr:iii04-expand-14}
How can MCT justify expanding-domain approximations of a nonnegative integral?
\end{exercise}
\begin{hint}
Use f times indicators of increasing sets.
\end{hint}
\begin{solution}
The truncated functions increase pointwise to f.
\end{solution}

If nonnegative f\_n tend to f almost everywhere and their integrals converge to integral f, what stronger convergence is plausible?
\end{exercise}
\begin{hint}
Compare min(f\_n,f).
\end{hint}
\begin{solution}
Under these hypotheses one obtains L1 convergence.

\end{hint}

\end{solution}
