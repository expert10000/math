\chapter{Support and Singular Distributions}
\label{ch:iii17}

Support records where a distribution can actually detect test functions. Singular distributions concentrate action on lower-dimensional or discrete sets and provide canonical models of sources, jumps, and point interactions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item determine the support of regular and singular distributions from their action on test functions;
\item compute supports of Dirac masses, derivatives of Dirac masses, and finite linear combinations;
\item distinguish singular support from ordinary support in representative examples;
\item use localization by test functions to prove that a distribution vanishes on an open set;
\item analyze how multiplication by smooth functions changes support;
\item construct distributions concentrated on lower-dimensional or discrete sets in elementary settings;
\end{itemize}

\section{Support of a distribution}
The support is the complement of the largest open set on which the distribution vanishes on every test function.

\section{Compact support}
Compactly supported distributions can act on broader classes of smooth functions by inserting a cutoff equal to one near the support.

\section{Dirac support}
The support of $\delta_{x_0}$ is the singleton $\{x_0\}$.

\section{Derivatives of Dirac}
$\delta^{(k)}$ acts by signed evaluation of the $k$th derivative of the test function.

\section{Measures as distributions}
Locally finite signed or complex measures define distributions by integration.

\section{Jump singularities}
Distributional derivatives of piecewise smooth functions include Dirac masses weighted by jump sizes.

\section{Surface distributions}
Integration over smooth hypersurfaces defines distributions supported on those hypersurfaces.

\section{Localization}
Multiplying by test functions or cutoffs isolates singular behavior near chosen points or sets.

\section{Core structural results}

\begin{theorem}[Support of a derivative]\label{thm:iii17-01}
$\operatorname{supp}(\partial^\alpha T)\subseteq\operatorname{supp}T$.
\begin{proof}
If $T$ vanishes on an open set, then it vanishes on derivatives of all test functions supported there, so the derivative distribution also vanishes there.
\end{proof}
\end{theorem}

\begin{theorem}[Dirac derivatives]\label{thm:iii17-02}
$\delta_0^{(k)}(\phi)=(-1)^k\phi^{(k)}(0)$.
\begin{proof}
Apply the definition of distributional derivative repeatedly to the point-evaluation functional.
\end{proof}
\end{theorem}

\begin{theorem}[Jump formula]\label{thm:iii17-03}
If a piecewise $C^1$ function has jump $J$ at $x_0$, its distributional derivative equals the classical derivative away from the jump plus $J\delta_{x_0}$.
\begin{proof}
Integrate by parts separately on the two sides of the jump; the internal boundary terms combine to the jump size times $\phi(x_0)$.
\end{proof}
\end{theorem}
