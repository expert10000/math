\chapter{Support and Singular Distributions}
\label{ch:iii17}

Support records where a distribution can actually detect test functions. Singular distributions concentrate action on lower-dimensional or discrete sets and provide canonical models of sources, jumps, and point interactions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item determine the support of regular and singular distributions from their action on test functions;
\item compute supports of Dirac masses, derivatives of Dirac masses, and finite linear combinations;
\item distinguish singular support from ordinary support in representative examples;
\item use localization by test functions to prove that a distribution vanishes on an open set;
\item analyze how multiplication by smooth functions changes support;
\item construct distributions concentrated on lower-dimensional or discrete sets in elementary settings;
\end{itemize}

\section{Support of a distribution}
The support is the complement of the largest open set on which the distribution vanishes on every test function.

\section{Compact support}
Compactly supported distributions can act on broader classes of smooth functions by inserting a cutoff equal to one near the support.

\section{Dirac support}
The support of $\delta_{x_0}$ is the singleton $\{x_0\}$.

\section{Derivatives of Dirac}
$\delta^{(k)}$ acts by signed evaluation of the $k$th derivative of the test function.


% BEGIN VOL03-EXPANSION III17-example-01
\begin{example}[Support survives Dirac differentiation]\label{ex:iii17-expand-01}
The derivative of Dirac acts by
\[
\delta_0'(\phi)=-\phi'(0).
\]
If \(\phi\) is supported in an open set avoiding \(0\), then all derivatives
of \(\phi\) vanish near \(0\), so \(\delta_0'(\phi)=0\). Conversely a test
function supported near \(0\) can have nonzero derivative at \(0\). Therefore
\[
\operatorname{supp}\delta_0'=\{0\}.
\]
Differentiation can change singular order without enlarging support.
\end{example}

\section{Measures as distributions}
Locally finite signed or complex measures define distributions by integration.

\section{Jump singularities}
Distributional derivatives of piecewise smooth functions include Dirac masses weighted by jump sizes.


% BEGIN VOL03-EXPANSION III17-example-02
\begin{example}[A jump produces a singular source]\label{ex:iii17-expand-02}
Let
\[
f(x)=
\begin{cases}
x,&x<0,\\
x+3,&x>0.
\end{cases}
\]
The classical derivative is \(1\) away from zero, while the jump size is
\(3\). The distributional derivative is therefore
\[
Df=1+3\delta_0.
\]
Integrating by parts on the two half-lines verifies that the internal boundary
terms combine to \(3\phi(0)\).
\end{example}

\section{Surface distributions}
Integration over smooth hypersurfaces defines distributions supported on those hypersurfaces.


% BEGIN VOL03-EXPANSION III17-example-03
\begin{example}[A hypersurface measure as a distribution]\label{ex:iii17-expand-03}
For a smooth compact hypersurface \(\Sigma\subset\mathbb R^n\), define
\[
T_\Sigma(\phi)=\int_\Sigma \phi\,dS.
\]
This is linear and continuous on test functions, and it vanishes whenever
\(\phi\) is supported away from \(\Sigma\). Hence
\[
\operatorname{supp}T_\Sigma\subseteq\Sigma,
\]
with equality when every point of \(\Sigma\) is detected by a local nonnegative
test function.
\end{example}

\section{Core structural results}

\begin{theorem}[Support of a derivative]\label{thm:iii17-01}
$\operatorname{supp}(\partial^\alpha T)\subseteq\operatorname{supp}T$.
\begin{proof}
If $T$ vanishes on an open set, then it vanishes on derivatives of all test functions supported there, so the derivative distribution also vanishes there.
\end{proof}
\end{theorem}

\begin{theorem}[Dirac derivatives]\label{thm:iii17-02}
$\delta_0^{(k)}(\phi)=(-1)^k\phi^{(k)}(0)$.
\begin{proof}
Apply the definition of distributional derivative repeatedly to the point-evaluation functional.
\end{proof}
\end{theorem}

\begin{theorem}[Jump formula]\label{thm:iii17-03}
If a piecewise $C^1$ function has jump $J$ at $x_0$, its distributional derivative equals the classical derivative away from the jump plus $J\delta_{x_0}$.
\begin{proof}
Integrate by parts separately on the two sides of the jump; the internal boundary terms combine to the jump size times $\phi(x_0)$.
\end{proof}
\end{theorem}

What is the support of delta\_a?

It only evaluates at a.

\begin{exercise}[Jump weight]\label{exr:iii17-expand-03}
A piecewise function jumps upward by 5 at x=2. What singular term appears in its derivative?
\end{exercise}
\begin{hint}
Use the jump formula.
\end{hint}
\begin{solution}
5 delta\_2.
\end{solution}

\begin{exercise}[Smooth multiplier support]\label{exr:iii17-expand-04}
How does supp(aT) compare with supp(T)?
\end{exercise}
\begin{hint}
If a test lies where T vanishes, so does aT.
\end{hint}
\begin{solution}
supp(aT) is contained in supp(T), and also in the closure of where a is nonzero.
\end{solution}

\begin{exercise}[Surface support]\label{exr:iii17-expand-05}
Where is the support of integration over a smooth curve?
\end{exercise}
\begin{hint}
The functional only sees test functions on the curve.
\end{hint}
\begin{solution}
It is the curve, under the usual nondegeneracy.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Derivative support]\label{exr:iii17-expand-06}
Prove supp(DT) is contained in supp(T).
\end{exercise}
\begin{hint}
Test on an open set where T vanishes.
\end{hint}
\begin{solution}
Derivatives of test functions remain supported there, so DT also vanishes.
\end{solution}

\begin{exercise}[Dirac derivative formula]\label{exr:iii17-expand-07}
Prove the formula for delta derivatives.
\end{exercise}
\begin{hint}
Induct from the distributional derivative definition.
\end{hint}
\begin{solution}
Each derivative introduces one minus sign and one test derivative.
\end{solution}

\begin{exercise}[Jump formula]\label{exr:iii17-expand-08}
Derive the jump contribution for a piecewise C1 function.
\end{exercise}
\begin{hint}
Integrate by parts on each side of the jump.
\end{hint}
\begin{solution}
Internal boundary terms leave jump size times the test value.
\end{solution}

\begin{exercise}[Localization]\label{exr:iii17-expand-09}
Prove a distribution vanishing on every test function supported in U is zero on U.
\end{exercise}
\begin{hint}
This is the definition of vanishing on an open set.
\end{hint}
\begin{solution}
The support is contained in the complement of U.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Support not pointwise support]\label{exr:iii17-expand-10}
Why is distribution support not defined by pointwise nonzero values?
\end{exercise}
\begin{hint}
A general distribution has no pointwise values.
\end{hint}
\begin{solution}
Support is defined through vanishing on open sets of test functions.
\end{solution}

\begin{exercise}[Derivative support equality not guaranteed]\label{exr:iii17-expand-11}
Why can supp(DT) be strictly smaller than supp(T)?
\end{exercise}
\begin{hint}
Differentiation may annihilate constants.
\end{hint}
\begin{solution}
For the constant distribution on a connected domain, derivative is zero while the original support is the whole domain.
\end{solution}

\begin{exercise}[Smooth factor can shrink support]\label{exr:iii17-expand-12}
Give a way multiplication by a smooth function can reduce support.
\end{exercise}
\begin{hint}
Choose the multiplier to vanish near part of the support.
\end{hint}
\begin{solution}
The product distribution then cannot detect test functions in that vanished region.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Point source]\label{exr:iii17-expand-13}
Why does support distinguish a point source from a distributed source?
\end{exercise}
\begin{hint}
It records the smallest closed set where action occurs.
\end{hint}
\begin{solution}
Dirac support is a point, while regular forcing may occupy a region.
\end{solution}

\begin{exercise}[Interface PDE]\label{exr:iii17-expand-14}
Why do surface distributions model interface forces?
\end{exercise}
\begin{hint}
They act only on test values along a hypersurface.
\end{hint}
\begin{solution}
They encode lower-dimensional sources in ambient PDEs.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Singular support]\label{exr:iii17-expand-15}
Explain the difference between support and singular support for a piecewise smooth function with one jump.
\end{exercise}
\begin{hint}
Support records where the function/distribution is nonzero; singular support records where it is not smooth.
\end{hint}
\begin{solution}
The singular support may be just the jump point even if ordinary support is much larger.
\end{solution}

\begin{exercise}[Derivative hierarchy]\label{exr:iii17-expand-16}
Show all derivatives of delta\_0 have support {0}.
\end{exercise}
\begin{hint}
Use derivative support inclusion and a test function with prescribed kth derivative at zero.
\end{hint}
\begin{solution}
The support cannot enlarge and is nonempty because suitable local tests give nonzero action.
\end{solution}
