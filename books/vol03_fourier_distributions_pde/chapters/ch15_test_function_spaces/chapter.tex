\chapter{Test-Function Spaces}
\label{ch:iii15}

Test-function spaces provide compactly supported smooth probes. Their topology is stronger and more local than the Schwartz topology, and distributions are defined as continuous linear functionals on these probes.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a smooth function has compact support inside the prescribed open set;
\item construct test functions by rescaling, translating, multiplying, and patching standard bumps;
\item track support under multiplication, differentiation, and coordinate changes;
\item distinguish convergence in test-function space from pointwise or uniform convergence alone;
\item build cutoff functions equal to one on a compact set and supported in a larger open set;
\item explain why compact support eliminates boundary terms in distributional integration by parts;
\end{itemize}

\section{Smooth compact support}
$C_c^\infty(\Omega)$ consists of infinitely differentiable functions whose supports are compact subsets of the open set $\Omega$.

\section{Bump functions}
Smooth cutoffs allow localization without sharp discontinuities.


% BEGIN VOL03-EXPANSION III15-example-01
\begin{example}[A normalized one-dimensional bump]\label{ex:iii15-expand-01}
Define
\[
\eta(x)=
\begin{cases}
\exp\!\left(-\frac{1}{1-x^2}\right),& |x|<1,\\
0,& |x|\ge1.
\end{cases}
\]
Every derivative tends to zero as \(x\to\pm1\), so the zero extension is
\(C^\infty\) and has support \([-1,1]\). Translating and scaling,
\[
\eta_{a,r}(x)=\eta\!\left(\frac{x-a}{r}\right),
\]
produces a smooth probe supported in \([a-r,a+r]\). This is the basic
localization building block used throughout distribution theory.
\end{example}

\section{Support control}
A sequence converging in the test-function topology eventually has support inside one fixed compact set.


% BEGIN VOL03-EXPANSION III15-example-02
\begin{example}[Why fixed support is part of test-function convergence]\label{ex:iii15-expand-02}
Let \(\phi_j(x)=\eta(x-j)\), where \(\eta\) is a fixed nonzero bump. Every
derivative has the same sup norm for all \(j\), but the supports drift to
infinity. Thus \(\phi_j\) does not converge to zero in
\(C_c^\infty(\mathbb R)\), even though on each fixed compact set it is
eventually zero. Test-function convergence requires eventual containment of all
supports in one compact set together with uniform convergence of every
derivative there.
\end{example}
% END VOL03-EXPANSION III15-example-02

\section{Partitions of unity}

\begin{example}[Patching local data with a partition of unity]\label{ex:iii15-expand-03}
Suppose \(U_1,U_2\) cover a compact set and smooth cutoffs
\(\psi_1,\psi_2\ge0\) satisfy \(\psi_1+\psi_2>0\) on that set. Define
\[
\chi_j=\frac{\psi_j}{\psi_1+\psi_2}.
\]
Then \(\chi_1+\chi_2=1\), each \(\chi_j\) is smooth, and its support stays
inside the corresponding localization region after the cutoffs are chosen
subordinate to the cover. A global object can therefore be decomposed into
local pieces and reconstructed by summation.
\end{example}

\section{Core structural results}

\begin{theorem}[Existence of bump functions]\label{thm:iii15-01}
Given compact $K\subset U$ with $U$ open in $\mathbb R^n$, there is $\phi\in C_c^\infty(U)$ with $0\le\phi\le1$ and $\phi=1$ near $K$.
\begin{proof}
Convolve the indicator of a slightly enlarged neighborhood of $K$ with a sufficiently small smooth mollifier, then truncate the geometric construction so the support stays inside $U$.
\end{proof}
\end{theorem}

\begin{theorem}[Partition of unity]\label{thm:iii15-02}
Every open cover of a paracompact smooth Euclidean domain admits a smooth locally finite partition of unity subordinate to it.
\begin{proof}
Refine the cover by locally finite relatively compact sets, choose bump functions on the refinement, and normalize their locally finite positive sum.
\end{proof}
\end{theorem}

\begin{theorem}[Test-function convergence]\label{thm:iii15-03}
If $\phi_j\to0$ in $C_c^\infty(\Omega)$, then all supports eventually lie in one compact $K\Subset\Omega$ and every derivative converges uniformly on $K$.
\begin{proof}
This is the defining sequential behavior of the standard LF topology and is precisely the control needed for distributional continuity.
\end{proof}
\end{theorem}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Scaled support]\label{exr:iii15-expand-01}
A bump eta is supported in [-1,1]. What is the support of eta((x-a)/r) for r>0?
\end{exercise}
\begin{hint}
Solve |(x-a)/r|<=1.
\end{hint}
\begin{solution}
It is contained in [a-r,a+r].
\end{solution}

\begin{exercise}[Derivative support]\label{exr:iii15-expand-02}
How does differentiation affect support of a smooth compactly supported function?
\end{exercise}
\begin{hint}
A derivative vanishes wherever the function is identically zero on a neighborhood.
\end{hint}
\begin{solution}
The derivative support is contained in the original support.
\end{solution}

\begin{exercise}[Product support]\label{exr:iii15-expand-03}
Where is the support of phi psi?
\end{exercise}
\begin{hint}
A nonzero product requires both factors nonzero.
\end{hint}
\begin{solution}
It is contained in supp(phi) intersect supp(psi).
\end{solution}

\begin{exercise}[Translation]\label{exr:iii15-expand-04}
Does translating a test function preserve test-function status?
\end{exercise}
\begin{hint}
Smoothness and compactness are invariant under translation.
\end{hint}
\begin{solution}
Yes; support is translated by the same vector.
\end{solution}

\begin{exercise}[Compact cutoff]\label{exr:iii15-expand-05}
What should a cutoff for K compactly contained in U satisfy?
\end{exercise}
\begin{hint}
Recall the bump theorem.
\end{hint}
\begin{solution}
It should be smooth, compactly supported in U, and equal to one near K.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Boundary terms]\label{exr:iii15-expand-06}
Explain why integration by parts against a test function has no boundary term at infinity.
\end{exercise}
\begin{hint}
The test function and all derivatives vanish outside a compact set.
\end{hint}
\begin{solution}
The ordinary boundary contribution is zero outside that compact support.
\end{solution}

\begin{exercise}[Product closure]\label{exr:iii15-expand-07}
Prove the product of two test functions is a test function.
\end{exercise}
\begin{hint}
Use smoothness and compact support.
\end{hint}
\begin{solution}
The product is smooth and its support lies in the compact intersection of the supports.
\end{solution}

\begin{exercise}[Fixed-support criterion]\label{exr:iii15-expand-08}
Explain why a sequence with supports escaping to infinity cannot converge in test-function space.
\end{exercise}
\begin{hint}
Test convergence requires one common eventual compact support.
\end{hint}
\begin{solution}
No such compact set can contain all escaping supports.
\end{solution}

\begin{exercise}[Partition identity]\label{exr:iii15-expand-09}
Prove a locally finite partition of unity can be multiplied into a test function and summed back to it.
\end{exercise}
\begin{hint}
Only finitely many terms meet the compact support.
\end{hint}
\begin{solution}
The sum is locally finite and phi times the partition sums to phi.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Pointwise not test convergence]\label{exr:iii15-expand-10}
Give a sequence tending pointwise to zero that does not converge in test-function topology.
\end{exercise}
\begin{hint}
Translate one fixed bump to infinity.
\end{hint}
\begin{solution}
The supports fail the fixed-compact requirement.
\end{solution}

\begin{exercise}[Uniform alone insufficient]\label{exr:iii15-expand-11}
Why does uniform convergence of phi\_j to zero not imply test-function convergence?
\end{exercise}
\begin{hint}
Derivatives also need uniform control and supports must remain fixed.
\end{hint}
\begin{solution}
A rapidly oscillating sequence can have small amplitude but large derivatives.
\end{solution}

\begin{exercise}[Compact support not preserved by arbitrary composition]\label{exr:iii15-expand-12}
Why can composing a compactly supported function with a nonproper map lose compact support?
\end{exercise}
\begin{hint}
The inverse image of a compact support may be noncompact.
\end{hint}
\begin{solution}
Support control under coordinate changes needs a proper/local diffeomorphism type hypothesis.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Localization]\label{exr:iii15-expand-13}
Why multiply a distribution by a cutoff equal to one near a point?
\end{exercise}
\begin{hint}
The cutoff removes distant behavior without changing local action.
\end{hint}
\begin{solution}
It isolates the distribution near the point or compact region.
\end{solution}

\begin{exercise}[Numerical patching]\label{exr:iii15-expand-14}
Why are partitions of unity useful in local finite-element or manifold constructions?
\end{exercise}
\begin{hint}
Local basis functions can be assembled into a global object.
\end{hint}
\begin{solution}
The partition ensures compatibility and exact reconstruction.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Nested cutoffs]\label{exr:iii15-expand-15}
Given K compactly contained in U, explain how to choose cutoffs chi and psi with chi=1 near K and psi=1 near supp(chi), both supported in U.
\end{exercise}
\begin{hint}
Use two nested compact neighborhoods between K and the boundary of U.
\end{hint}
\begin{solution}
Apply the bump theorem twice to the nested neighborhoods.
\end{solution}

\begin{exercise}[Test topology versus Schwartz]\label{exr:iii15-expand-16}
Why is every compactly supported smooth function Schwartz after zero extension, yet the test-function topology is not the Schwartz topology?
\end{exercise}
\begin{hint}
Schwartz convergence allows supports to spread if weighted seminorms shrink, while test convergence fixes eventual compact support.
\end{hint}
\begin{solution}
The underlying sets overlap, but the topologies encode different localization behavior and have different duals.
\end{solution}
