\chapter{Test-Function Spaces}
\label{ch:iii15}

Test-function spaces provide compactly supported smooth probes. Their topology is stronger and more local than the Schwartz topology, and distributions are defined as continuous linear functionals on these probes.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a smooth function has compact support inside the prescribed open set;
\item construct test functions by rescaling, translating, multiplying, and patching standard bumps;
\item track support under multiplication, differentiation, and coordinate changes;
\item distinguish convergence in test-function space from pointwise or uniform convergence alone;
\item build cutoff functions equal to one on a compact set and supported in a larger open set;
\item explain why compact support eliminates boundary terms in distributional integration by parts;
\end{itemize}

\section{Smooth compact support}
$C_c^\infty(\Omega)$ consists of infinitely differentiable functions whose supports are compact subsets of the open set $\Omega$.

\section{Bump functions}
Smooth cutoffs allow localization without sharp discontinuities.

\section{Support control}

\section{Derivative control}

\section{Partitions of unity}

\section{Dual space preview}
The continuous dual of $C_c^\infty(\Omega)$ is the space of distributions on $\Omega$.

\section{Core structural results}

\begin{theorem}[Existence of bump functions]\label{thm:iii15-01}
Given compact $K\subset U$ with $U$ open in $\mathbb R^n$, there is $\phi\in C_c^\infty(U)$ with $0\le\phi\le1$ and $\phi=1$ near $K$.
\begin{proof}
Convolve the indicator of a slightly enlarged neighborhood of $K$ with a sufficiently small smooth mollifier, then truncate the geometric construction so the support stays inside $U$.
\end{proof}
\end{theorem}

\begin{theorem}[Partition of unity]\label{thm:iii15-02}
Every open cover of a paracompact smooth Euclidean domain admits a smooth locally finite partition of unity subordinate to it.
\begin{proof}
Refine the cover by locally finite relatively compact sets, choose bump functions on the refinement, and normalize their locally finite positive sum.
\end{proof}
\end{theorem}

\begin{theorem}[Test-function convergence]\label{thm:iii15-03}
If $\phi_j\to0$ in $C_c^\infty(\Omega)$, then all supports eventually lie in one compact $K\Subset\Omega$ and every derivative converges uniformly on $K$.
\begin{proof}
This is the defining sequential behavior of the standard LF topology and is precisely the control needed for distributional continuity.
\end{proof}
\end{theorem}
