\chapter{Test-Function Spaces}
\label{ch:iii15}

Test-function spaces provide compactly supported smooth probes. Their topology is stronger and more local than the Schwartz topology, and distributions are defined as continuous linear functionals on these probes.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a smooth function has compact support inside the prescribed open set;
\item construct test functions by rescaling, translating, multiplying, and patching standard bumps;
\item track support under multiplication, differentiation, and coordinate changes;
\item distinguish convergence in test-function space from pointwise or uniform convergence alone;
\item build cutoff functions equal to one on a compact set and supported in a larger open set;
\item explain why compact support eliminates boundary terms in distributional integration by parts;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
Let $\Omega\subset\mathbb R^n$ be open. On a fixed compact $K\Subset\Omega$, $\mathcal D_K$ has seminorms $\max_{|\alpha|\le m}\|D^\alpha\phi\|_\infty$. The usual topology on $\mathcal D(\Omega)=C_c^\infty(\Omega)$ is the locally convex inductive limit over a compact exhaustion. A linear functional is continuous exactly when, for each $K$, its absolute value is bounded by $C_K\max_{|\alpha|\le m_K}\|D^\alpha\phi\|_\infty$ on $\mathcal D_K$. The fixed-support criterion for convergent sequences below is a standard LF-space theorem, not a definition of the entire topology; its topological proof is deferred. Translations of test functions on $\Omega$ are allowed only if the translated support remains compactly inside $\Omega$ (automatically on $\mathbb R^n$).

\section{Smooth compact support}
$C_c^\infty(\Omega)$ consists of infinitely differentiable functions whose supports are compact subsets of the open set $\Omega$.

\section{Fixed-support spaces and the test-function topology}
For each compact \(K\Subset\Omega\), define
\[
\mathcal D_K(\Omega)
=
\{\varphi\in C_c^\infty(\Omega):
\operatorname{supp}\varphi\subset K\}.
\]
Its Fr\'echet topology is generated by
\[
p_m(\varphi)
=
\max_{|\alpha|\le m}\sup_{x\in K}|D^\alpha\varphi(x)|,
\qquad m=0,1,2,\ldots.
\]
The test-function space
\[
\mathcal D(\Omega)=C_c^\infty(\Omega)
\]
carries the usual locally convex inductive-limit topology obtained from these
fixed-support spaces. Thus topology is not optional decoration: it is the
structure that makes ``continuous linear functional'' in the definition of a
distribution meaningful.

A sequence \(\varphi_j\to\varphi\) in \(\mathcal D(\Omega)\) is eventually
supported in one compact \(K\Subset\Omega\) and satisfies
\[
p_m(\varphi_j-\varphi)\to0
\qquad\text{for every }m.
\]
This sequence criterion is useful, but the topology itself is the
inductive-limit topology, not merely a rule about sequences.

\section{Bump functions}
Smooth cutoffs allow localization without sharp discontinuities.

\begin{example}[A one-dimensional bump]\label{ex:iii15-expand-01}
Define
\[
\eta(x)=
\begin{cases}
\exp\!\left(-\frac{1}{1-x^2}\right),& |x|<1,\\

\[
\eta_{a,r}(x)=\eta\!\left(\frac{x-a}{r}\right),
\]
produces a smooth probe supported in \([a-r,a+r]\). This is the basic
localization building block used throughout distribution theory.
\end{example}

\section{Support control}
A sequence converging in the test-function topology eventually has support inside one fixed compact set.

\begin{example}[Why fixed support is part of test-function convergence]\label{ex:iii15-expand-02}
Let \(\phi_j(x)=\eta(x-j)\), where \(\eta\) is a fixed nonzero bump. Every
derivative has the same sup norm for all \(j\), but the supports drift to
infinity. Thus \(\phi_j\) does not converge to zero in
\(C_c^\infty(\mathbb R)\), even though on each fixed compact set it is
eventually zero. Test-function convergence requires eventual containment of all
supports in one compact set together with uniform convergence of every
derivative there.
\end{example}
% END VOL03-EXPANSION III15-example-02
\section{Derivative control}
Within that compact set, every derivative converges uniformly.

\section{Partitions of unity}
Locally finite smooth partitions subordinate to open covers allow local constructions to be patched globally.


% BEGIN VOL03-EXPANSION III15-example-03
\begin{example}[Patching local data with a partition of unity]\label{ex:iii15-expand-03}
Suppose \(U_1,U_2\) cover a compact set and smooth cutoffs
\(\psi_1,\psi_2\ge0\) satisfy \(\psi_1+\psi_2>0\) on that set. Define
\[
\chi_j=\frac{\psi_j}{\psi_1+\psi_2}.
\]
These quotients are defined only on the open neighborhood where the denominator is positive. There \(\chi_1+\chi_2=1\), each \(\chi_j\) is smooth, and its relative support stays
inside the corresponding localization region after the cutoffs are chosen
subordinate to the cover. A global object can therefore be decomposed into
local pieces and reconstructed by summation.
\end{example}

\section{Localization}
Multiplying by a test function isolates distributional behavior in a chosen region.

\section{Comparison with Schwartz space}
Compact support implies rapid decay after extension by zero, but the topologies and intended duals differ.

\section{Dual space preview}
The continuous dual of $C_c^\infty(\Omega)$ is the space of distributions on $\Omega$.

\section{Core structural results}

\begin{theorem}[Existence of bump functions]\label{thm:iii15-01}
Given compact $K\subset U$ with $U$ open in $\mathbb R^n$, there is $\phi\in C_c^\infty(U)$ with $0\le\phi\le1$ and $\phi=1$ near $K$.
\begin{proof}
Convolve the indicator of a slightly enlarged neighborhood of $K$ with a sufficiently small smooth mollifier, then truncate the geometric construction so the support stays inside $U$.
\end{proof}
\end{theorem}

\begin{theorem}[Partition of unity]\label{thm:iii15-02}
Every open cover of a paracompact smooth Euclidean domain admits a smooth locally finite partition of unity subordinate to it.
\begin{proof}
Proof sketch of the standard smooth partition theorem: use a locally finite refinement with closures subordinate to the original cover and a further shrinking that still covers the domain. Choose bumps positive on the shrunken sets and supported in the refinement, then normalize their locally finite positive sum. The topological refinement/shrinking theorem is used with its full proof deferred.
\end{proof}
\end{theorem}

\begin{theorem}[Test-function convergence]\label{thm:iii15-03}
If $\phi_j\to0$ in $C_c^\infty(\Omega)$, then all supports eventually lie in one compact $K\Subset\Omega$ and every derivative converges uniformly on $K$.
\begin{proof}
Standard theorem, proof deferred: convergent sequences in this strict LF test-function space lie in one fixed-support space and converge in its derivative seminorms. The converse follows from continuity of the inclusion of that fixed-support space.
\end{proof}
\end{theorem}

\begin{theorem}[Continuity criterion for linear functionals]\label{thm:iii15-04}
Let \(T:\mathcal D(\Omega)\to\mathbb C\) be linear. Then \(T\) is continuous
if and only if for every compact \(K\Subset\Omega\) there exist an integer
\(m_K\ge0\) and a constant \(C_K>0\) such that
\[
|T(\varphi)|
\le
C_K
\max_{|\alpha|\le m_K}
\sup_{x\in K}|D^\alpha\varphi(x)|
\]
for every \(\varphi\in\mathcal D_K(\Omega)\).
\begin{proof}
Restriction of a continuous functional to each Fr\'echet space
\(\mathcal D_K\) is continuous. A continuous linear functional on
\(\mathcal D_K\) is dominated by finitely many defining seminorms, hence by one
\(p_{m_K}\). Conversely, the locally convex inductive-limit topology is the
final topology for the inclusions of the fixed-support spaces, so continuity
of all restrictions gives continuity of \(T\).
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[A standard bump function]\label{ex:iii15-worked-01}
Define
\[
\eta(x)=
\begin{cases}
\exp\!\bigl(-1/(1-|x|^2)\bigr),&|x|<1,\\
0,&|x|\ge1.
\end{cases}
\]
Every derivative tends to zero faster than any power as \(|x|\uparrow1\), so
\(\eta\in C_c^\infty(\mathbb R^n)\). This is the basic localizer used
throughout distribution theory.
\end{example}

\begin{example}[Fixed-support test functions form a Fréchet space]\label{ex:iii15-worked-02}
For compact \(K\Subset\Omega\), set
\[
\mathcal D_K=\{\varphi\in C_c^\infty(\Omega):\operatorname{supp}\varphi\subset K\}
\]
and
\[
p_m(\varphi)=\max_{|\alpha|\le m}\sup_K|D^\alpha\varphi|.
\]
Convergence in \(\mathcal D_K\) means all derivatives converge uniformly on the
same compact support.
\end{example}

\begin{example}[Scaling a test function]\label{ex:iii15-worked-03}
If \(\varphi\in C_c^\infty(\mathbb R^n)\), define
\(\varphi_\varepsilon(x)=\varepsilon^{-n}\varphi(x/\varepsilon)\). Then
\[
\operatorname{supp}\varphi_\varepsilon
=\varepsilon\,\operatorname{supp}\varphi,\qquad
D^\alpha\varphi_\varepsilon
=\varepsilon^{-n-|\alpha|}(D^\alpha\varphi)(x/\varepsilon).
\]
The formula makes support localization and derivative growth explicit.
\end{example}

\begin{example}[Partition-of-unity localization]\label{ex:iii15-worked-04}
If \(\{\chi_j\}\subset C_c^\infty(\Omega)\) is locally finite and
\(\sum_j\chi_j=1\), then for \(\varphi\in C_c^\infty(\Omega)\),
\[
\varphi=\sum_j\chi_j\varphi,
\]
and only finitely many terms meet \(\operatorname{supp}\varphi\). Thus global
test functions can be decomposed into finitely many local pieces on each
compact support.
\end{example}

\section{Solved dossiers}
Each dossier combines several ideas from the chapter in a longer argument. The aim is to make hypotheses, calculations, and proof strategy explicit.

\begin{problem}[A standard bump function]\label{prob:iii15-dossier-01}
Construct a nonzero smooth compactly supported function.
\end{problem}
\begin{solution}
Use the standard exponential bump
\(\exp(-1/(1-|x|^2))\) inside the unit ball and \(0\) outside; all derivatives
match smoothly at the boundary.
\end{solution}

\begin{problem}[Fixed-support test functions form a Fréchet space]\label{prob:iii15-dossier-02}
Describe convergence in \(\mathcal D_K\).
\end{problem}
\begin{solution}
The supports stay inside \(K\), and for every derivative order \(m\), the
corresponding seminorm \(p_m(\varphi_n-\varphi)\) tends to zero.
\end{solution}

\begin{problem}[Scaling a test function]\label{prob:iii15-dossier-03}
Compute the support and derivative scaling of a rescaled test function.
\end{problem}
\begin{solution}
The support scales by \(\varepsilon\), and the \(\alpha\)-derivative gains
the factor \(\varepsilon^{-n-|\alpha|}\).
\end{solution}

\begin{problem}[Partition-of-unity localization]\label{prob:iii15-dossier-04}
Explain why a locally finite partition of unity gives a finite decomposition of any fixed test function.
\end{problem}
\begin{solution}
A compact support meets only finitely many members of a locally finite
family; on that compact set the partition sums to one.
\end{solution}

\begin{problem}[Support under multiplication]\label{prob:iii15-dossier-05}
If \(\chi,\varphi\in C_c^\infty\), where is \(\operatorname{supp}(\chi\varphi)\)?
\end{problem}
\begin{solution}
It is contained in
\(\operatorname{supp}\chi\cap\operatorname{supp}\varphi\).
\end{solution}

\begin{problem}[Test-function translation]\label{prob:iii15-dossier-06}
Show translating a test function preserves membership in \(C_c^\infty\).
\end{problem}
\begin{solution}
Smoothness is preserved, and the compact support is translated to another
compact set.
\end{solution}

\begin{problem}[Why one sup norm is insufficient]\label{prob:iii15-dossier-07}
Why does the test-function topology control all derivatives, not just \(\|\varphi\|_\infty\)?
\end{problem}
\begin{solution}
Distributions may differentiate test functions. Continuity must therefore
control derivative behavior as well as function values.
\end{solution}

\begin{problem}[Nested supports]\label{prob:iii15-dossier-08}
If \(K_1\subset K_2\), explain the inclusion \(\mathcal D_{K_1}\subset\mathcal D_{K_2}\).
\end{problem}
\begin{solution}
Any function supported in \(K_1\) is automatically supported in the larger
compact \(K_2\).
\end{solution}

\subsection*{Standard computations}

\begin{exercise}[Scaled support]\label{exr:iii15-expand-01}
A bump eta is supported in [-1,1]. What is the support of eta((x-a)/r) for r>0?
\end{exercise}
\begin{hint}
Solve |(x-a)/r|<=1.
\end{hint}
\begin{solution}
It is contained in [a-r,a+r].
\end{solution}

\begin{exercise}[Derivative support]\label{exr:iii15-expand-02}
How does differentiation affect support of a smooth compactly supported function?
\end{exercise}
\begin{hint}
A derivative vanishes wherever the function is identically zero on a neighborhood.
\end{hint}
\begin{solution}
The derivative support is contained in the original support.
\end{solution}

\begin{exercise}[Product support]\label{exr:iii15-expand-03}
Where is the support of phi psi?
\end{exercise}
\begin{hint}
A nonzero product requires both factors nonzero.
\end{hint}
\begin{solution}
It is contained in supp(phi) intersect supp(psi).
\end{solution}

\begin{exercise}[Translation]\label{exr:iii15-expand-04}
Does translating a test function preserve test-function status?
\end{exercise}
\begin{hint}
Smoothness and compactness are invariant under translation.
\end{hint}
\begin{solution}
Yes; support is translated by the same vector.
\end{solution}

\begin{exercise}[Compact cutoff]\label{exr:iii15-expand-05}
What should a cutoff for K compactly contained in U satisfy?
\end{exercise}
\begin{hint}
Recall the bump theorem.
\end{hint}
\begin{solution}
It should be smooth, compactly supported in U, and equal to one near K.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Boundary terms]\label{exr:iii15-expand-06}
Explain why integration by parts against a test function has no boundary term at infinity.
\end{exercise}
\begin{hint}
The test function and all derivatives vanish outside a compact set.
\end{hint}
\begin{solution}
The ordinary boundary contribution is zero outside that compact support.
\end{solution}

\begin{exercise}[Product closure]\label{exr:iii15-expand-07}
Prove the product of two test functions is a test function.
\end{exercise}
\begin{hint}
Use smoothness and compact support.
\end{hint}
\begin{solution}
The product is smooth and its support lies in the compact intersection of the supports.
\end{solution}

\begin{exercise}[Fixed-support criterion]\label{exr:iii15-expand-08}
Explain why a sequence with supports escaping to infinity cannot converge in test-function space.
\end{exercise}
\begin{hint}
Test convergence requires one common eventual compact support.
\end{hint}
\begin{solution}
No such compact set can contain all escaping supports.
\end{solution}

\begin{exercise}[Partition identity]\label{exr:iii15-expand-09}
Prove a locally finite partition of unity can be multiplied into a test function and summed back to it.
\end{exercise}
\begin{hint}
Only finitely many terms meet the compact support.
\end{hint}
\begin{solution}
The sum is locally finite and phi times the partition sums to phi.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Pointwise not test convergence]\label{exr:iii15-expand-10}
Give a sequence tending pointwise to zero that does not converge in test-function topology.
\end{exercise}
\begin{hint}
Translate one fixed bump to infinity.
\end{hint}
\begin{solution}
The supports fail the fixed-compact requirement.
\end{solution}

\begin{exercise}[Uniform alone insufficient]\label{exr:iii15-expand-11}
Why does uniform convergence of phi\_j to zero not imply test-function convergence?
\end{exercise}
\begin{hint}
Derivatives also need uniform control and supports must remain fixed.
\end{hint}
\begin{solution}
A rapidly oscillating sequence can have small amplitude but large derivatives.
\end{solution}

\begin{exercise}[Compact support not preserved by arbitrary composition]\label{exr:iii15-expand-12}
Why can composing a compactly supported function with a nonproper map lose compact support?
\end{exercise}
\begin{hint}
The inverse image of a compact support may be noncompact.
\end{hint}
\begin{solution}
If $\phi(0)\ne0$ and $F(x)=0$ for all $x\in\mathbb R$, then $\phi\circ F$ is a nonzero constant and has noncompact support. A proper smooth map is a sufficient condition for compact support of the pullback; a local diffeomorphism alone is not sufficient.
\end{solution}
