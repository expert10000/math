\chapter{Fundamental Solutions}
\label{ch:iii24}

A fundamental solution is a distributional inverse kernel for a differential operator. Once one is known, convolution converts a constant-coefficient PDE into an explicit representation formula.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a candidate fundamental solution satisfies \(L\Phi=\delta\) in the distributional sense;
\item derive fundamental solutions from Fourier multipliers in constant-coefficient examples;
\item compute solutions of forced equations by convolution with a fundamental solution when the operation is justified;
\item track singularity, scaling, and dimensional dependence of standard fundamental solutions;
\item distinguish a fundamental solution from a Green function adapted to boundary conditions;
\item use distributional differentiation to check formulas that are singular at the source point;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\section{Definition}
A fundamental solution $E$ for $P(D)$ satisfies $P(D)E=\delta_0$ in distributions.

\section{Convolution solution}
When convolution is legitimate, $u=E*f$ satisfies $P(D)u=f$.

\section{Laplacian kernels}
The Laplacian has logarithmic fundamental solution in two dimensions and Newtonian power kernels in dimensions at least three.


% BEGIN VOL03-EXPANSION III24-example-01
\begin{example}[Why the Newtonian power has the right scaling]\label{ex:iii24-expand-01}
For \(n\ge3\), the candidate Laplace fundamental solution has form
\[
E(x)=c_n|x|^{2-n}.
\]
Away from zero, direct radial differentiation gives \(\Delta E=0\). Its radial
flux through a sphere of radius \(r\) is proportional to
\[
r^{n-1}\partial_r(r^{2-n}),
\]
which is independent of \(r\). Choosing \(c_n\) so that this flux equals one
normalizes the Dirac mass at the origin.
\end{example}

\begin{example}[Distributional verification by punctured integration]\label{ex:iii24-expand-02}
To verify \(-\Delta E=\delta_0\), integrate on
\(\Omega_\varepsilon=\mathbb R^n\setminus B_\varepsilon(0)\). Twice applying
integration by parts transfers derivatives from \(E\) to a test function.
The volume term vanishes because \(E\) is harmonic away from zero; the outer
boundary vanishes by compact support, while the inner sphere flux tends to
\(\phi(0)\). The singular source is exactly the limiting boundary flux.
\end{example}

\section{Nonuniqueness}
Adding any homogeneous solution to a fundamental solution yields another fundamental solution.


% BEGIN VOL03-EXPANSION III24-example-03
\begin{example}[Fundamental solutions are not unique]\label{ex:iii24-expand-03}
If \(E\) satisfies \(P(D)E=\delta_0\) and \(h\) satisfies the homogeneous
equation \(P(D)h=0\), then
\[
P(D)(E+h)=\delta_0.
\]
Thus \(E+h\) is another fundamental solution. Boundary conditions or growth
requirements are what select a particular Green kernel or physically relevant
representative from this affine family.
\end{example}

\section{Regularity away from the pole}
Fundamental solutions are smooth wherever the source point is absent.

\section{Core structural results}

\begin{theorem}[Convolution inversion]\label{thm:iii24-01}
If $P(D)E=\delta_0$ and the relevant convolutions exist, then $P(D)(E*f)=f$.
\begin{proof}
Constant-coefficient differentiation commutes with convolution, so $P(D)(E*f)=(P(D)E)*f=\delta_0*f=f$.
\end{proof}
\end{theorem}

\begin{theorem}[Newtonian fundamental solution]\label{thm:iii24-02}
For $n\ge3$, a constant multiple of $|x|^{2-n}$ is a fundamental solution of $-\Delta$.
\begin{proof}
The function is harmonic away from zero. Integrate against a test function on the complement of a small ball, integrate by parts, and choose the constant so the shrinking boundary flux tends to the test value at zero.
\end{proof}
\end{theorem}

\begin{theorem}[Smoothness off the singularity]\label{thm:iii24-03}
If $P(D)E=\delta_0$, then $P(D)E=0$ on every open set avoiding the origin.
\begin{proof}
Test functions supported away from zero annihilate $\delta_0$, so the distribution solves the homogeneous equation there; standard regularity for the explicit classical kernels gives smoothness.
\end{proof}
\end{theorem}

\begin{exercise}[Radial harmonicity]\label{exr:iii24-expand-07}

\end{solution}

\end{hint}
\begin{solution}
The powers cancel to r\textasciicircum{}0.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Fundamental versus Green]\label{exr:iii24-expand-10}
Why is a free-space fundamental solution not automatically a Green function on a bounded domain?
\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
A boundary-correction term is needed.
\end{solution}

\begin{exercise}[Uniqueness failure]\label{exr:iii24-expand-11}
Why is a fundamental solution not unique?
\end{exercise}
\begin{hint}
Add any homogeneous solution.
\end{hint}
\begin{solution}
The operator kills the added term.
\end{solution}

\begin{exercise}[Classical failure at pole]\label{exr:iii24-expand-12}
Why should one not demand the PDE hold classically at the source point?
\end{exercise}
\begin{hint}
The kernel is singular there.
\end{hint}
\begin{solution}
The correct equation is distributional and encodes the point source.
\end{solution}
