\chapter{Fundamental Solutions}
\label{ch:iii24}

A fundamental solution is a distributional inverse kernel for a differential operator. Once one is known, convolution converts a constant-coefficient PDE into an explicit representation formula.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a candidate fundamental solution satisfies \(L\Phi=\delta\) in the distributional sense;
\item derive fundamental solutions from Fourier multipliers in constant-coefficient examples;
\item compute solutions of forced equations by convolution with a fundamental solution when the operation is justified;
\item track singularity, scaling, and dimensional dependence of standard fundamental solutions;
\item distinguish a fundamental solution from a Green function adapted to boundary conditions;
\item use distributional differentiation to check formulas that are singular at the source point;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
Here $P(D)$ has constant coefficients. A sufficient convolution hypothesis is $E\in\mathcal D'(\mathbb R^n)$ and $f\in\mathcal E'(\mathbb R^n)$ (compactly supported distribution); in particular $f\in C_c^\infty$ gives a smooth convolution. For the elliptic examples fix the operator $-\Delta$. With $\sigma_{n-1}=|S^{n-1}|$, choose $E(x)=|x|^{2-n}/((n-2)\sigma_{n-1})$ for $n\ge3$, $E(x)=-(2\pi)^{-1}\log|x|$ for $n=2$, and $E(x)=-|x|/2$ for $n=1$. All satisfy $-\Delta E=\delta_0$. The causal heat fundamental solution is $1_{t>0}G_t(x)$ and satisfies $(\partial_t-\Delta)E=\delta_{(0,0)}$ in spacetime.

\section{Definition}
A fundamental solution $E$ for $P(D)$ satisfies $P(D)E=\delta_0$ in distributions.

\section{Convolution solution}
When convolution is legitimate, $u=E*f$ satisfies $P(D)u=f$.

\section{Laplacian kernels}
The Laplacian has logarithmic fundamental solution in two dimensions and Newtonian power kernels in dimensions at least three.

\begin{example}[Why the Newtonian power has the right scaling]\label{ex:iii24-expand-01}
For \(n\ge3\), the candidate Laplace fundamental solution has form
\[
E(x)=c_n|x|^{2-n}.
\]
Away from zero, direct radial differentiation gives \(\Delta E=0\). Its radial
flux through a sphere of radius \(r\) is proportional to
\[
r^{n-1}\partial_r(r^{2-n}),
\]
which is independent of \(r\). Choosing \(c_n=1/((n-2)\sigma_{n-1})\) so that the negative of the outward flux equals one
normalizes the Dirac mass at the origin.
\end{example}

\begin{example}[Distributional verification by punctured integration]\label{ex:iii24-expand-02}
To verify \(-\Delta E=\delta_0\), integrate on
\(\Omega_\varepsilon=\mathbb R^n\setminus B_\varepsilon(0)\). Twice applying
integration by parts transfers derivatives from \(E\) to a test function.
The volume term vanishes because \(E\) is harmonic away from zero; the outer
boundary vanishes by compact support, while the inner sphere flux tends to
\(\phi(0)\). The singular source is exactly the limiting boundary flux.
\end{example}

\section{Nonuniqueness}
Adding any homogeneous solution to a fundamental solution yields another fundamental solution.


% BEGIN VOL03-EXPANSION III24-example-03
\begin{example}[Fundamental solutions are not unique]\label{ex:iii24-expand-03}
If \(E\) satisfies \(P(D)E=\delta_0\) and \(h\) satisfies the homogeneous
equation \(P(D)h=0\), then
\[
P(D)(E+h)=\delta_0.
\]
Thus \(E+h\) is another fundamental solution. Boundary conditions or growth
requirements are what select a particular Green kernel or physically relevant
representative from this affine family.
\end{example}
% END VOL03-EXPANSION III24-example-03
\section{Regularity away from the pole}
For the explicit elliptic Laplace kernels, the fundamental solution is smooth away from the pole. A general constant-coefficient fundamental solution need only solve the homogeneous equation there; wave kernels can remain singular on a characteristic cone away from the source.

\section{Core structural results}

\begin{theorem}[Convolution inversion]\label{thm:iii24-01}
If $P(D)E=\delta_0$ and the relevant convolutions exist, then $P(D)(E*f)=f$.
\begin{proof}
Constant-coefficient differentiation commutes with convolution, so $P(D)(E*f)=(P(D)E)*f=\delta_0*f=f$.
\end{proof}
\end{theorem}

\begin{theorem}[Newtonian fundamental solution]\label{thm:iii24-02}
For $n\ge3$, a constant multiple of $|x|^{2-n}$ is a fundamental solution of $-\Delta$.
\begin{proof}
The function is harmonic away from zero. Integrate against a test function on the complement of a small ball, integrate by parts, and choose the constant so the shrinking boundary flux tends to the test value at zero.
\end{proof}
\end{theorem}

\begin{theorem}[Homogeneous equation off the source]\label{thm:iii24-03}
If $P(D)E=\delta_0$, then $P(D)E=0$ on every open set avoiding the origin.
\begin{proof}
Test functions supported away from zero annihilate $\delta_0$, so the distribution solves the homogeneous equation there. No smoothness for a general $P(D)$ follows from this argument.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[One-dimensional Laplace fundamental solution]\label{ex:iii24-worked-01}
In one dimension,
\[
\Phi(x)=-\frac12|x|
\]
satisfies
\[
-\Phi''=\delta_0
\]
distributionally because \(D^2|x|=2\delta_0\). Thus a singular source is
encoded by a kink rather than an ordinary twice differentiable function.
\end{example}

\begin{example}[Convolution solves constant-coefficient equations formally]\label{ex:iii24-worked-02}
If \(L\Phi=\delta_0\) and convolution is justified, then
\[
L(\Phi*f)=(L\Phi)*f=\delta_0*f=f.
\]
Thus a fundamental solution converts inversion of \(L\) into convolution.
\end{example}

\begin{example}[Heat kernel is a time-dependent fundamental solution]\label{ex:iii24-worked-03}
The heat kernel
\[
K_t(x)=(4\pi t)^{-n/2}e^{-|x|^2/(4t)}
\]
satisfies \((\partial_t-\Delta)K=0\) for \(t>0\), has mass \(1\), and converges
to \(\delta_0\) as \(t\downarrow0\). Therefore
\(u(t)=K_t*f\) recovers initial data and solves the heat equation.
\end{example}

\begin{example}[Singularity strength matches dimension]\label{ex:iii24-worked-04}
For the Laplacian in dimensions \(n\ge3\), the classical fundamental
solution has the form \(c_n|x|^{2-n}\); in two dimensions it is logarithmic.
The singularity is exactly strong enough that the flux through small spheres
has a nonzero limit, producing the Dirac mass.
\end{example}

\section{Solved dossiers}
Each dossier combines several ideas from the chapter in a longer argument. The aim is to make hypotheses, calculations, and proof strategy explicit.

\begin{problem}[One-dimensional Laplace fundamental solution]\label{prob:iii24-dossier-01}
Verify a one-dimensional fundamental solution for \(-d^2/dx^2\).
\end{problem}
\begin{solution}
Since \(D^2|x|=2\delta_0\), the function
\(\Phi=-|x|/2\) satisfies \(-\Phi''=\delta_0\).
\end{solution}

\begin{problem}[Convolution solves constant-coefficient equations formally]\label{prob:iii24-dossier-02}
Explain why convolution with a fundamental solution solves \(Lu=f\).
\end{problem}
\begin{solution}
Constant-coefficient differentiation commutes with convolution, so
\(L(\Phi*f)=(L\Phi)*f=\delta_0*f=f\).
\end{solution}

\begin{problem}[Heat kernel is a time-dependent fundamental solution]\label{prob:iii24-dossier-03}
List the three properties that make the heat kernel a fundamental solution for the Cauchy problem.
\end{problem}
\begin{solution}
It solves the PDE for positive time, has unit mass, and tends to the Dirac
mass as \(t\downarrow0\).
\end{solution}

\begin{problem}[Singularity strength matches dimension]\label{prob:iii24-dossier-04}
Explain why the Laplace fundamental solution changes from a power law to a logarithm in dimension two.
\end{problem}
\begin{solution}
The radial harmonic ODE integrates to \(r^{2-n}\) for \(n\ne2\), while the
borderline exponent at \(n=2\) integrates to \(\log r\).
\end{solution}

\begin{problem}[Distributional verification]\label{prob:iii24-dossier-05}
How is \(L\Phi=\delta_0\) checked rigorously?
\end{problem}
\begin{solution}
Pair with a test function, integrate by parts away from the singularity,
and compute the limiting boundary flux around the singular point.
\end{solution}

\begin{problem}[Translation invariance]\label{prob:iii24-dossier-06}
If \(\Phi\) is a fundamental solution at \(0\), what solves a point source at \(a\)?
\end{problem}
\begin{solution}
\(\Phi(x-a)\), because constant-coefficient operators commute with
translations.
\end{solution}

\begin{problem}[Nonuniqueness]\label{prob:iii24-dossier-07}
Why is a fundamental solution not generally unique?
\end{problem}
\begin{solution}
One may add any global homogeneous solution \(h\) with \(Lh=0\).
\end{solution}

\begin{problem}[Fourier construction]\label{prob:iii24-dossier-08}
How can Fourier transform formally construct a fundamental solution?
\end{problem}
\begin{solution}
If \(L\) has multiplier \(p(\xi)\), solve
\(p(\xi)\widehat\Phi=1\) distributionally, so formally
\(\widehat\Phi=1/p(\xi)\).
\end{solution}

\begin{exercise}[Scaling]\label{exr:iii24-expand-15}
Derive the homogeneity degree of the Laplacian fundamental solution from operator scaling.
\end{exercise}
\begin{hint}
