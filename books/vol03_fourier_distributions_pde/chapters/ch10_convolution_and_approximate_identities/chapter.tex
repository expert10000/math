\chapter{Convolution and Approximate Identities}
\label{ch:iii10}

Convolution averages one function against translates of another. It interacts cleanly with translation, integration, differentiation, and Fourier transformation, and approximate identities use it to smooth and recover functions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute convolutions of elementary functions and determine their support;
\item prove basic algebraic properties of convolution under the required integrability hypotheses;
\item use Young-type estimates to control convolution norms;
\item show how approximate identities recover functions in the appropriate topology;
\item distinguish smoothing by convolution from pointwise multiplication;
\item construct mollifier-based approximations while tracking support, normalization, and convergence mode;
\end{itemize}

\section{Definition of convolution}
For suitable functions, $(f*g)(x)=\int f(x-y)g(y)dy$.

\section{Commutativity and associativity}
Under absolute-integrability hypotheses, changes of variables and Fubini prove $f*g=g*f$ and associativity.

\section{Young inequalities}
Convolution maps certain $L^p$ pairs into controlled $L^r$ spaces.

\section{Translation viewpoint}
Convolution is an average of translates of one factor weighted by the other.

\section{Approximate identities}
Kernels concentrating mass near zero recover functions in norm or pointwise under appropriate hypotheses.

\section{Smoothing}
Convolution with a smooth compactly supported kernel produces smooth functions.

\section{Differentiation}
Derivatives pass to the smooth kernel: $D^\alpha(f*\phi)=f*D^\alpha\phi$ under standard assumptions.

\section{Support}
For compact supports, $\operatorname{supp}(f*g)$ lies in the Minkowski sum of the two supports.

\section{Core structural results}

\begin{theorem}[Commutativity]\label{thm:iii10-01}
If the convolution integral is absolutely meaningful, then $f*g=g*f$.
\begin{proof}
Substitute $z=x-y$ in the defining integral.
\end{proof}
\end{theorem}

\begin{theorem}[Young $L^1$ inequality]\label{thm:iii10-02}
If $f,g\in L^1(\mathbb R^n)$, then $f*g\in L^1$ and $\|f*g\|_1\le\|f\|_1\|g\|_1$.
\begin{proof}
Use Tonelli on $|f(x-y)g(y)|$ and translation invariance of Lebesgue measure.
\end{proof}
\end{theorem}

\begin{theorem}[Approximate-identity recovery]\label{thm:iii10-03}
If $\phi_\varepsilon$ is an approximate identity and $f\in L^p$, $1\le p<\infty$, then $f*\phi_\varepsilon\to f$ in $L^p$ under standard normalization.
\begin{proof}
Write the difference as an average of translations $\tau_yf-f$, use continuity of translations in $L^p$, concentrate kernel mass near zero, and control the far tail by kernel mass.
\end{proof}
\end{theorem}
