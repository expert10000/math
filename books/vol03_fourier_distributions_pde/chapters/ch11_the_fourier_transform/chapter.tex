\chapter{The Fourier Transform}
\label{ch:iii11}

The Fourier transform converts translation and differentiation into multiplication, convolution into products, and spatial localization into frequency information. It is the central transform chapter of Volume III and is developed with extra structural depth.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Fourier transforms of elementary functions under the chapter's normalization convention;
\item derive translation, modulation, scaling, and differentiation identities from the transform definition;
\item identify when the transform is continuous, bounded, or decaying from integrability and regularity assumptions;
\item compute inverse transforms in examples where an inversion theorem applies;
\item use convolution-multiplication duality to simplify transform calculations;
\item check transform formulas by dimensional, symmetry, or inverse-transform consistency tests rather than memorization alone;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\section{Definition on $L^1$}
Fix the convention $\widehat f(\xi)=\int_{\mathbb R^n}f(x)e^{-2\pi i x\cdot\xi}dx$ for integrable functions.

\section{Boundedness and continuity}
The transform of an $L^1$ function is bounded and uniformly continuous, with $\|\widehat f\|_\infty\le\|f\|_1$.

\section{Riemann--Lebesgue lemma}
Fourier transforms of $L^1$ functions vanish at infinity.

\section{Translation and modulation}
Translation in physical space creates a phase factor in frequency; modulation shifts the transform.


% BEGIN VOL03-EXPANSION III11-example-01
\begin{example}[Translation versus modulation]\label{ex:iii11-expand-01}
If \(g(x)=f(x-a)\), then
\[
\widehat g(\xi)=e^{-2\pi ia\xi}\widehat f(\xi).
\]
Translation changes phase but not transform magnitude. By contrast multiplying
\(f\) by \(e^{2\pi ibx}\) shifts the transform to
\(\widehat f(\xi-b)\). These two dual identities are best remembered together.
\end{example}
% END VOL03-EXPANSION III11-example-01
\section{Scaling}
Dilation in space inversely dilates frequency and contributes the Jacobian factor.


% BEGIN VOL03-EXPANSION III11-example-02
\begin{example}[Scaling rule]\label{ex:iii11-expand-02}
For \(g(x)=f(ax)\),
\[
\widehat g(\xi)=|a|^{-1}\widehat f(\xi/a).
\]
The Jacobian factor and reciprocal frequency scaling are both essential.
Checking the formula at \(\xi=0\) immediately verifies the factor:
\(\widehat g(0)=|a|^{-1}\int f\).
\end{example}
% END VOL03-EXPANSION III11-example-02
\section{Differentiation and moments}
Differentiation becomes multiplication by $2\pi i\xi$, while multiplication by $x$ becomes differentiation in frequency.


% BEGIN VOL03-EXPANSION III11-example-03
\begin{example}[Differentiation becomes multiplication]\label{ex:iii11-expand-03}
Under sufficient decay,
\[
\widehat{f'}(\xi)=2\pi i\xi\,\widehat f(\xi).
\]
Integration by parts proves the identity. Thus constant-coefficient
differential operators become algebraic multipliers in frequency, which is the
basic reason Fourier transform is powerful for PDE.
\end{example}
% END VOL03-EXPANSION III11-example-03
\section{Convolution theorem}
Convolution transforms into pointwise multiplication, and products transform into convolution when hypotheses permit.

\section{Inversion}
Under suitable integrability or Schwartz hypotheses, applying the inverse transform reconstructs the original function.

\section{Symmetry and reality}
Even, odd, real, and radial structures impose corresponding symmetry on the transform.

\section{Uncertainty viewpoint}
Strong concentration in space conflicts with strong concentration in frequency, foreshadowing quantitative uncertainty principles.

\section{Core structural results}

\begin{theorem}[$L^1$ bound]\label{thm:iii11-01}
For $f\in L^1$, $|\widehat f(\xi)|\le\|f\|_1$ for every $\xi$.
\begin{proof}
Take absolute values inside the integral; the complex exponential has modulus one.
\end{proof}
\end{theorem}

\begin{theorem}[Riemann--Lebesgue lemma]\label{thm:iii11-02}
If $f\in L^1(\mathbb R^n)$, then $\widehat f(\xi)\to0$ as $|\xi|\to\infty$.
\begin{proof}
First prove it for smooth compactly supported functions by integration by parts or oscillatory cancellation, then approximate an arbitrary $L^1$ function in $L^1$ and use the uniform transform bound.
\end{proof}
\end{theorem}

\begin{theorem}[Convolution theorem]\label{thm:iii11-03}
If $f,g\in L^1$, then $\widehat{f*g}=\widehat f\,\widehat g$.
\begin{proof}
Insert the convolution integral, use Fubini justified by absolute integrability, change variables $u=x-y$, and factor the two resulting integrals.
\end{proof}
\end{theorem}

\begin{theorem}[Fourier inversion on Schwartz space]\label{thm:iii11-04}
For $f\in\mathcal S(\mathbb R^n)$, inverse transformation recovers $f$.
\begin{proof}
Regularize with a Gaussian multiplier, use Fubini to identify the regularized inverse as convolution with an approximate identity, and pass to the limit in the Schwartz or uniform topology.
\end{proof}
\end{theorem}
