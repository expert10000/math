\chapter{Weak Derivatives}
\label{ch:iii20}

Weak derivatives extend differentiation to $L^1_{\mathrm{loc}}$ functions by testing against compactly supported smooth functions. They are the bridge from distribution theory to Sobolev spaces and weak PDE formulations.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a locally integrable function has a proposed weak derivative by testing the defining integration-by-parts identity;
\item compute weak derivatives of piecewise smooth functions and identify jump contributions when they occur distributionally;
\item distinguish weak differentiability from classical differentiability;
\item prove uniqueness of weak derivatives up to almost-everywhere equality;
\item use approximation by smooth functions to justify weak-derivative identities;
\item construct functions with weak derivatives that fail to have classical derivatives at exceptional points;
\end{itemize}

\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}

\section{Weak derivative definition}
A function $g$ is the weak derivative $D_jf$ if $\int fD_j\phi=-\int g\phi$ for every test function.


% BEGIN VOL03-EXPANSION III20-example-01
\begin{example}[Weak derivative of absolute value]\label{ex:iii20-expand-01}
Let \(f(x)=|x|\). Its classical derivative fails at zero, but define
\[
g(x)=
\begin{cases}
-1,&x<0,\\
1,&x>0.
\end{cases}
\]
For every test function \(\phi\), integration by parts on the two half-lines
gives
\[
\int |x|\phi'(x)\,dx=-\int g(x)\phi(x)\,dx.
\]
Hence \(g\) is the weak derivative. A single classical nondifferentiability
does not prevent weak differentiability.
\end{example}
% END VOL03-EXPANSION III20-example-01
\section{Agreement with classical derivatives}
Classically differentiable functions have the same weak derivative.

\section{Piecewise smooth functions}
Jumps usually prevent the weak derivative from being represented by an ordinary function because the distributional derivative contains Dirac masses.


% BEGIN VOL03-EXPANSION III20-example-02
\begin{example}[A jump has no ordinary weak derivative]\label{ex:iii20-expand-02}
For Heaviside \(H\), the distributional derivative is \(\delta_0\). If an
ordinary \(g\in L^1_{\rm loc}\) were a weak derivative, its associated regular
distribution would have to equal \(\delta_0\), impossible. Thus a jump can
prevent membership in \(W^{1,p}\) even though the function is piecewise smooth:
the missing derivative is singular rather than an ordinary function.
\end{example}

\section{Approximation}
Mollification converts weak derivatives into classical derivatives and commutes with differentiation on interior regions.


% BEGIN VOL03-EXPANSION III20-example-03
\begin{example}[Mollification commutes with weak differentiation]\label{ex:iii20-expand-03}
If \(D_jf=g\) weakly and \(\rho_\varepsilon\) is a mollifier, then in an
interior region
\[
D_j(f*\rho_\varepsilon)=g*\rho_\varepsilon.
\]
The left side is now a classical derivative of a smooth function. This identity
lets one prove estimates first for mollified approximants while retaining exact
control of the weak derivative.
\end{example}

\section{Core structural results}

\begin{theorem}[Uniqueness of weak derivative]\label{thm:iii20-01}
If $g,h\in L^1_{\mathrm{loc}}$ both satisfy the weak derivative identity for $f$, then $g=h$ almost everywhere.
\begin{proof}
Their difference integrates to zero against every test function; the fundamental lemma of distributions implies the associated regular distribution is zero, hence the functions agree a.e.
\end{proof}
\end{theorem}

\begin{theorem}[Classical implies weak]\label{thm:iii20-02}
If $f\in C^1$, then its weak derivative equals its classical derivative.
\begin{proof}
Integrate by parts against compactly supported test functions; boundary terms vanish.
\end{proof}
\end{theorem}

\begin{theorem}[Mollifier commutation]\label{thm:iii20-03}
For locally integrable $f$ with weak derivative $D_jf$, $D_j(f*\rho_\varepsilon)=(D_jf)*\rho_\varepsilon$ where defined.
\begin{proof}
Differentiate the smooth convolution and transfer the derivative from the mollifier to $f$ using the weak derivative identity.
\end{proof}
\end{theorem}

\begin{hint}

\end{hint}
\begin{solution}
A Dirac mass is not represented by an Lp function.
\end{solution}

\end{solution}

\end{solution}

The singularity type determines whether the derivative remains an ordinary function.
\end{solution}
