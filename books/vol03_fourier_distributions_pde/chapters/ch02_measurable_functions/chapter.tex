\chapter{Measurable Functions}
\label{ch:iii02}

Measurable functions are those whose inverse images respect the measurable structure. Their closure under algebra, limits, and monotone simple approximation makes them the natural domain of Lebesgue integration.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify measurability of a function from inverse images of Borel sets or from level-set criteria;
\item prove that algebraic combinations and pointwise limits of measurable functions are measurable under the stated hypotheses;
\item construct simple-function approximations to nonnegative measurable functions;
\item compute measurable level sets for explicit functions and use them to analyze distribution functions;
\item distinguish almost-everywhere equality from pointwise equality in measure-theoretic arguments;
\item produce examples of nonmeasurable behavior by identifying where the measurable-structure hypotheses fail;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\section{Inverse-image definition}
A real-valued function is measurable when inverse images of Borel sets are measurable.

\section{Ray criteria}
It is enough to test sets of the form $\{f<a\}$ or $\{f>a\}$ for all real $a$ because such rays generate the Borel sigma-algebra.


% BEGIN VOL03-EXPANSION III02-example-01
\begin{example}[Measurability from level sets]\label{ex:iii02-expand-01}
For \(f(x)=x^2\), the sublevel set \(\{f<a\}\) is empty when \(a\le0\) and
equals \((-\sqrt a,\sqrt a)\) when \(a>0\). Every such set is Borel, so the
ray criterion proves \(f\) measurable. Testing a generating family of Borel
sets is enough; one need not compute the inverse image of every Borel set.
\end{example}
% END VOL03-EXPANSION III02-example-01
\section{Algebraic operations}
Sums, products, absolute values, maxima, and minima of measurable real-valued functions remain measurable.

\section{Pointwise limits}
Countable suprema, infima, limsup, liminf, and pointwise limits of measurable functions are measurable.

\section{Simple functions}
A simple measurable function has finite range and measurable level sets; it is the finite-data model for integration.

\section{Approximation from below}
Every nonnegative measurable function is the increasing pointwise limit of nonnegative simple functions obtained by dyadic quantization and truncation.


% BEGIN VOL03-EXPANSION III02-example-02
\begin{example}[Dyadic simple approximation]\label{ex:iii02-expand-02}
For \(f\ge0\), truncate at height \(n\) and replace values by lower dyadic
endpoints of mesh \(2^{-n}\). The resulting \(\phi_n\) is simple and measurable,
satisfies \(0\le\phi_n\le f\), and increases pointwise to \(f\). On the region
where \(f\le n\), the quantization error is below \(2^{-n}\). This is the
constructive bridge from measurable functions to Lebesgue integration.
\end{example}
% END VOL03-EXPANSION III02-example-02
\section{Almost-everywhere equivalence}
On complete measure spaces, changing a measurable function on a null set preserves measurability.


% BEGIN VOL03-EXPANSION III02-example-03
\begin{example}[Almost-everywhere modification]\label{ex:iii02-expand-03}
Assume the measure space is complete and \(f=g\) outside a null set \(N\).
For every Borel set \(B\), the symmetric difference between
\(f^{-1}(B)\) and \(g^{-1}(B)\) is contained in \(N\). Completeness makes that
difference measurable, hence \(g^{-1}(B)\) is measurable whenever
\(f^{-1}(B)\) is. Thus measurability survives null-set modification.
\end{example}
% END VOL03-EXPANSION III02-example-03
\section{Composition}
Composing a measurable map with a continuous or Borel measurable scalar map preserves measurability.

\section{Core structural results}

\begin{theorem}[Pointwise limit theorem]\label{thm:iii02-01}
A pointwise limit of measurable real-valued functions is measurable.
\begin{proof}
Represent the limit using limsup and liminf, each built from countable suprema and infima of measurable functions.
\end{proof}
\end{theorem}

\begin{theorem}[Algebra closure]\label{thm:iii02-02}
If $f$ and $g$ are measurable, then $f+g$ and $fg$ are measurable.
\begin{proof}
The map $x\mapsto(f(x),g(x))$ is measurable into $\mathbb R^2$, and addition and multiplication are continuous Borel maps.
\end{proof}
\end{theorem}

\begin{theorem}[Simple approximation]\label{thm:iii02-03}
If $f\ge0$ is measurable, there exist simple $\phi_n\uparrow f$.
\begin{proof}
Truncate at height $n$ and replace values by the lower endpoints of dyadic intervals of width $2^{-n}$; measurability of the level sets gives simple functions and the mesh tends to zero.
\end{proof}
\end{theorem}

\section{Graded supplementary exercises}

\subsection*{Standard computations}

\begin{exercise}[Indicator]\label{exr:iii02-expand-01}
Why is the indicator of a measurable set measurable?
\end{exercise}
\begin{hint}
Inspect threshold sets.
\end{hint}
\begin{solution}
Every threshold set is empty, the set, its complement, or the whole space.
\end{solution}

\begin{exercise}[Absolute value]\label{exr:iii02-expand-02}
If f is measurable, what about |f|?
\end{exercise}
\begin{hint}
Compose with a continuous map.
\end{hint}
\begin{solution}
It is measurable.
\end{solution}

\begin{exercise}[Maximum]\label{exr:iii02-expand-03}
If f and g are measurable, what about max(f,g)?
\end{exercise}
\begin{hint}
Use algebra and absolute value.
\end{hint}
\begin{solution}
max(f,g)=(f+g+|f-g|)/2, so it is measurable.
\end{solution}

\begin{exercise}[Pointwise limit]\label{exr:iii02-expand-04}
What can be said about a pointwise limit of measurable real functions?
\end{exercise}
\begin{hint}
Use the pointwise limit theorem.
\end{hint}
\begin{solution}
The limit is measurable.
\end{solution}

\begin{exercise}[Simple range]\label{exr:iii02-expand-05}
What distinguishes a simple function by its range?
\end{exercise}
\begin{hint}
Recall the definition.
\end{hint}
\begin{solution}
Its range is finite.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Ray criterion]\label{exr:iii02-expand-06}
Why does checking all sublevel rays suffice?
\end{exercise}
\begin{hint}
Rays generate the Borel sigma-algebra.
\end{hint}
\begin{solution}
Inverse images preserve complements and countable unions, extending measurability from generators.
\end{solution}

\begin{exercise}[Supremum]\label{exr:iii02-expand-07}
Prove the supremum of countably many measurable functions is measurable.
\end{exercise}
\begin{hint}
Write a superlevel set as a union.
\end{hint}
\begin{solution}
{sup f\_n > a} is the union of {f\_n > a}.
\end{solution}

\begin{exercise}[Limsup]\label{exr:iii02-expand-08}
Prove limsup of measurable functions is measurable.
\end{exercise}
\begin{hint}
Use tail suprema and countable infimum.
\end{hint}
\begin{solution}
limsup is inf\_N sup\_{n>=N} f\_n, built from measurable operations.
\end{solution}

\begin{exercise}[Simple approximation]\label{exr:iii02-expand-09}
Explain why dyadic lower approximants converge upward.
\end{exercise}
\begin{hint}
Refine mesh and raise truncation level.
\end{hint}
\begin{solution}
The approximants never exceed f, become finer, and their pointwise error tends to zero.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Nonmeasurable indicator]\label{exr:iii02-expand-10}
If A is nonmeasurable, why is its indicator nonmeasurable?
\end{exercise}
\begin{hint}
Use a threshold such as 1/2.
\end{hint}
\begin{solution}
The superlevel set is exactly A.
\end{solution}

\begin{exercise}[Incomplete null change]\label{exr:iii02-expand-11}
Why can null-set modification fail in an incomplete space?
\end{exercise}
\begin{hint}
A subset of a null set can be nonmeasurable.
\end{hint}
\begin{solution}
The modified function can acquire a nonmeasurable level set.
\end{solution}

\begin{exercise}[Pointwise intuition]\label{exr:iii02-expand-12}
Why does pointwise regularity alone not imply measurability?
\end{exercise}
\begin{hint}
Measurability concerns inverse images and the chosen sigma-algebra.
\end{hint}
\begin{solution}
A function can be defined using a nonmeasurable set, producing nonmeasurable level sets.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Random variables]\label{exr:iii02-expand-13}
Why must a random variable be measurable?
\end{exercise}
\begin{hint}
Events such as X<a need probabilities.
\end{hint}
\begin{solution}
Measurability ensures their inverse images are measurable events.
\end{solution}

\begin{exercise}[Quantization]\label{exr:iii02-expand-14}
Why do simple functions matter computationally?
\end{exercise}
\begin{hint}
They use finitely many values and measurable regions.
\end{hint}
\begin{solution}
They approximate measurable functions while making integrals finite sums.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Composition]\label{exr:iii02-expand-15}
Prove continuous h composed with measurable f is measurable.
\end{exercise}
\begin{hint}
Take inverse images.
\end{hint}
\begin{solution}
(h o f)\textasciicircum{}(-1)(B)=f\textasciicircum{}(-1)(h\textasciicircum{}(-1)(B)), and h\textasciicircum{}(-1)(B) is Borel.
\end{solution}

\begin{exercise}[Tail envelope]\label{exr:iii02-expand-16}
Show tail suprema decrease to limsup.
\end{exercise}
\begin{hint}
Removing early terms cannot raise a supremum.
\end{hint}
\begin{solution}
The decreasing sequence sup\_{k>=n} f\_k converges by definition to limsup f\_n.
\end{solution}
