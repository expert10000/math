\chapter{Weak Boundary-Value Problems}
\label{ch:iii23}

\section{Core structural results}

\begin{theorem}[Lax--Milgram theorem]\label{thm:iii23-01}
If $V$ is a Hilbert space, $a$ is bounded and coercive, and $F\in V^*$, then there is a unique $u\in V$ with $a(u,v)=F(v)$ for all $v\in V$.
\begin{proof}
Use Riesz representation to write $a(u,v)=\langle Au,v\rangle$ for a bounded operator $A$. Coercivity makes $A$ bounded below, hence injective with closed range; orthogonality to the range would contradict coercivity for the adjoint, so the range is all of $V$.
\end{proof}
\end{theorem}

\begin{theorem}[Energy uniqueness]\label{thm:iii23-02}
If $a$ is coercive and $a(u,v)=0$ for all $v$, then $u=0$.
\begin{proof}
Choose $v=u$. Then $0=a(u,u)\ge c\|u\|_V^2$, hence $u=0$.
\end{proof}
\end{theorem}

\begin{theorem}[Galerkin stability]\label{thm:iii23-03}
For coercive $a$, the Galerkin solution $u_h$ satisfies $\|u_h\|_V\le c^{-1}\|F\|_{V^*}$.
\begin{proof}
Test the discrete equation with $u_h$ and combine coercivity with the dual norm bound.
\end{proof}
\end{theorem}
