\chapter{Weak Boundary-Value Problems}
\label{ch:iii23}

Weak boundary-value problems replace pointwise differential equations by integral identities tested against admissible functions. This is the natural formulation for rough coefficients, Sobolev solutions, and variational existence theory.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item derive the weak formulation of a boundary-value problem by multiplying by a test function and integrating by parts;
\item identify the natural trial and test spaces for elementary weak PDE problems;
\item verify boundedness and coercivity of bilinear forms in representative examples;
\item apply an existence theorem such as Lax--Milgram only after checking its hypotheses;
\item interpret weak boundary conditions and distinguish them from pointwise boundary equalities;
\item recover additional regularity or classical equations from a weak solution when justified;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
Use real Hilbert spaces and real bounded bilinear forms. Boundedness means $|a(u,v)|\le M\|u\|_V\|v\|_V$; coercivity means $a(v,v)\ge\alpha\|v\|_V^2$ with $\alpha>0$. For the Dirichlet examples let $\Omega$ be bounded Lipschitz, $V=H_0^1(\Omega)$, $A\in L^\infty$ uniformly positive definite, $c\in L^\infty$ nonnegative, and $F\in V^*=H^{-1}$ (in particular $f\in L^2$ gives $F(v)=\int fv$). These assumptions yield a unique weak solution, not automatically a C2 solution. Pure Neumann Poisson with data $\partial_\nu u=g$ on a connected bounded smooth domain requires $\int_\Omega f+\int_{\partial\Omega}g=0$ and a normalization modulo constants. Galerkin stability assumes finite-dimensional subspaces of $V$; convergence also requires approximation of $V$ by those subspaces.

\section{Strong and weak formulations}
A classical solution satisfies the PDE pointwise with the required continuous derivatives; a strong solution may satisfy it a.e. with Sobolev derivatives. A weak solution satisfies the integrated identity in the stated energy space.

\begin{example}[Deriving the weak Poisson problem]\label{ex:iii23-expand-01}
For
\[
-\Delta u=f\quad\text{in }\Omega,\qquad u=0\quad\text{on }\partial\Omega,
\]
multiply by \(v\in C_c^\infty(\Omega)\) and integrate by parts:
\[
\int_\Omega \nabla u\cdot\nabla v
=\int_\Omega f v.
\]
The weak problem asks for \(u\in H_0^1(\Omega)\) satisfying this identity for
every \(v\in H_0^1(\Omega)\). The boundary condition is encoded in the space,
not imposed pointwise.
\end{example}

\section{Energy spaces}
Sobolev spaces such as $H_0^1(\Omega)$ encode square-integrable weak gradients and homogeneous boundary conditions.

\section{Bilinear forms}
Second-order elliptic equations are represented by bilinear forms such as $a(u,v)=\int_\Omega A\nabla u\cdot\nabla v+cuv$.

\section{Linear functionals}
Forcing terms act as continuous linear functionals on the chosen energy space.

\section{Coercivity}
A coercive estimate $a(v,v)\ge c\|v\|_V^2$ supplies control of the solution norm.

\begin{example}[Poincare turns Poisson energy into coercivity]\label{ex:iii23-expand-02}
For the Poisson bilinear form
\[
a(u,v)=\int_\Omega\nabla u\cdot\nabla v
\]
on \(H_0^1(\Omega)\),
\[
a(v,v)=\|\nabla v\|_2^2.
\]
Poincare gives \(\|v\|_2\le C\|\nabla v\|_2\), hence the gradient norm is
equivalent to the full \(H^1\) norm. Therefore \(a(v,v)\) controls
\(\|v\|_{H^1}^2\) up to a constant, providing Lax--Milgram coercivity.
\end{example}
% END VOL03-EXPANSION III23-example-02
\section{Weak boundary conditions}
Dirichlet conditions are built into $H_0^1$; Neumann data typically arise through boundary terms in integration by parts.

\section{Galerkin approximation}
Finite-dimensional subspaces convert the weak problem into linear algebra and support constructive approximation.

\begin{remark}[Discretization error versus algebraic solver residual]
For the exact Galerkin solution \(u_h\in V_h\),
\[
a(u_h,v_h)=F(v_h)
\qquad\text{for every }v_h\in V_h.
\]
After choosing a basis this becomes a linear system \(Ac=b\).  A numerical
linear solver returns an approximate coefficient vector \(\widetilde c\), so
the algebraic residual
\[
r=b-A\widetilde c
\]
measures failure to satisfy the \emph{discrete} equations.

That residual is not the same thing as the error to the exact PDE solution.
There are at least two distinct contributions:
\[
u-\widetilde u_h
=
(u-u_h)+(u_h-\widetilde u_h),
\]
namely discretization error and algebraic solver error.  Moreover, the
Euclidean norm of \(r\) depends on the chosen basis and scaling.

At the continuous variational level define
\[
R_{\widetilde u_h}(v)
=
F(v)-a(\widetilde u_h,v).
\]
If \(a\) is coercive with constant \(\alpha>0\), then
\[
\|u-\widetilde u_h\|_V
\le
\alpha^{-1}\|R_{\widetilde u_h}\|_{V^*}.
\]
Thus a residual becomes a rigorous solution-error bound only after it is
interpreted in the correct dual norm and combined with the stability constant.
A residual tested only on \(V_h\) need not measure approximation error outside
the discrete space.
\end{remark}

\begin{example}[Galerkin turns the weak PDE into linear algebra]\label{ex:iii23-expand-03}
Choose a finite-dimensional space \(V_h=\operatorname{span}\{\phi_1,\ldots,\phi_m\}\)
inside the energy space and write
\[
u_h=\sum_{j=1}^m c_j\phi_j.
\]
Testing with each basis function gives
\[
\sum_j a(\phi_j,\phi_i)c_j=F(\phi_i),\qquad i=1,\ldots,m.
\]
Thus the stiffness matrix \(A_{ij}=a(\phi_j,\phi_i)\) is the finite-dimensional
representation of the weak operator. Coercivity makes the discrete system
stable and nonsingular.
\end{example}

\section{Uniqueness by energy}
Testing the difference of two solutions against itself often proves uniqueness from positivity or coercivity.

\section{Core structural results}

\begin{theorem}[Lax--Milgram theorem]\label{thm:iii23-01}
If $V$ is a Hilbert space, $a$ is bounded and coercive, and $F\in V^*$, then there is a unique $u\in V$ with $a(u,v)=F(v)$ for all $v\in V$.
\begin{proof}
Use Riesz representation to write $a(u,v)=\langle Au,v\rangle$ for a bounded operator $A$. Coercivity makes $A$ bounded below, hence injective with closed range; orthogonality to the range would contradict coercivity for the adjoint, so the range is all of $V$.
\end{proof}
\end{theorem}

\begin{theorem}[Energy uniqueness]\label{thm:iii23-02}
If $a$ is coercive and $a(u,v)=0$ for all $v$, then $u=0$.
\begin{proof}
Choose $v=u$. Then $0=a(u,u)\ge c\|u\|_V^2$, hence $u=0$.
\end{proof}
\end{theorem}

\begin{theorem}[Galerkin stability]\label{thm:iii23-03}
For coercive $a$, the Galerkin solution $u_h$ satisfies $\|u_h\|_V\le c^{-1}\|F\|_{V^*}$.
\begin{proof}
Test the discrete equation with $u_h$ and combine coercivity with the dual norm bound.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Weak Poisson formulation]\label{ex:iii23-worked-01}
For
\[
-\Delta u=f\quad\text{in }\Omega,\qquad u|_{\partial\Omega}=0,
\]
multiply by \(v\in C_c^\infty(\Omega)\) and integrate by parts:
\[
\int_\Omega\nabla u\cdot\nabla v
=\int_\Omega fv.
\]
This identity motivates seeking \(u\in H_0^1(\Omega)\) and testing against all
\(v\in H_0^1(\Omega)\).
\end{example}

\begin{example}[Coercivity comes from Poincaré]\label{ex:iii23-worked-02}
On \(H_0^1(\Omega)\), let
\[
a(u,v)=\int_\Omega\nabla u\cdot\nabla v.
\]
Then
\[
a(u,u)=\|\nabla u\|_2^2.
\]
Poincaré controls \(\|u\|_2\) by \(\|\nabla u\|_2\), so \(a(u,u)\) controls the
full \(H^1\) norm. This is the coercivity needed for Lax--Milgram.
\end{example}

\begin{example}[The load functional must be bounded]\label{ex:iii23-worked-03}
If \(f\in L^2(\Omega)\) and \(v\in H_0^1(\Omega)\), then
\[
|F(v)|=\left|\int fv\right|
\le\|f\|_2\|v\|_2
\le C\|f\|_2\|\nabla v\|_2.
\]
Thus \(F\) is continuous on the energy space.
\end{example}

\begin{example}[Energy uniqueness]\label{ex:iii23-worked-04}
If \(u_1,u_2\) are weak solutions with the same data, their difference
\(w\) satisfies
\[
\int_\Omega\nabla w\cdot\nabla v=0
\]
for every \(v\in H_0^1\). Taking \(v=w\) gives \(\|\nabla w\|_2=0\); Poincaré
then gives \(w=0\).
\end{example}

\section{Solved dossiers}
Each dossier combines several ideas from the chapter in a longer argument. The aim is to make hypotheses, calculations, and proof strategy explicit.

\begin{problem}[Weak Poisson formulation]\label{prob:iii23-dossier-01}
Derive the weak formulation of the homogeneous Dirichlet Poisson problem.
\end{problem}
\begin{solution}
Multiply by a test function, integrate the Laplacian by parts, and absorb
the zero-boundary condition into the space \(H_0^1(\Omega)\).
\end{solution}

\begin{problem}[Coercivity comes from Poincaré]\label{prob:iii23-dossier-02}
Explain why the Dirichlet energy form is coercive on \(H_0^1(\Omega)\).
\end{problem}
\begin{solution}
Poincaré inequality makes the gradient norm equivalent to the full
\(H^1\) norm on \(H_0^1\), hence
\(a(u,u)\ge c\|u\|_{H^1}^2\).
\end{solution}

\begin{problem}[The load functional must be bounded]\label{prob:iii23-dossier-03}
Show \(F(v)=\int_\Omega fv\) is bounded on \(H_0^1(\Omega)\) for \(f\in L^2\).
\end{problem}
\begin{solution}
Use Cauchy--Schwarz followed by Poincaré.
\end{solution}

\begin{problem}[Energy uniqueness]\label{prob:iii23-dossier-04}
Prove uniqueness of the homogeneous Dirichlet weak Poisson solution.
\end{problem}
\begin{solution}
Test the difference equation with the difference itself; zero energy and
Poincaré force the difference to vanish.
\end{solution}

\begin{problem}[Lax--Milgram data]\label{prob:iii23-dossier-05}
State the three ingredients needed to apply Lax--Milgram.
\end{problem}
\begin{solution}
A Hilbert space, a bounded coercive bilinear form, and a bounded linear
functional.
\end{solution}

\begin{problem}[Zero boundary meaning]\label{prob:iii23-dossier-06}
Why should \(u=0\) on \(\partial\Omega\) be represented by \(H_0^1(\Omega)\) rather than pointwise values?
\end{problem}
\begin{solution}
Sobolev functions are equivalence classes and may lack pointwise boundary
values; \(H_0^1\) is the intrinsic closure-based zero-boundary space.
\end{solution}

\begin{problem}[Neumann warning]\label{prob:iii23-dossier-07}
Why is coercivity different for pure Neumann problems?
\end{problem}
\begin{solution}
Constants have zero gradient, so the Dirichlet energy is not coercive on
all \(H^1\); one works modulo constants or imposes a normalization and
compatibility condition.
\end{solution}

\begin{problem}[Energy estimate]\label{prob:iii23-dossier-08}
What a priori estimate follows from Lax--Milgram for Poisson?
\end{problem}
\begin{solution}
The solution norm is bounded by the load norm divided by the coercivity
constant, e.g. \(\|u\|_{H_0^1}\le C\|F\|_{(H_0^1)^*}\).
\end{solution}

\begin{exercise}[Cea preview]\label{exr:iii23-expand-16}
Why should a coercive Galerkin solution be close to the best approximation in V\_h?
\end{exercise}
\begin{hint}

\end{hint}

Galerkin orthogonality gives $a(u-u_h,v_h)=0$. For any $w_h\in V_h$, coercivity and boundedness imply $\alpha\|u-u_h\|_V^2\le a(u-u_h,u-w_h)\le M\|u-u_h\|_V\|u-w_h\|_V$. Divide unless the error is zero, then take the infimum: the constant is $M/\alpha$.
\end{solution}

\begin{exercise}[Residual bound under coercivity]\label{exr:iii23-numerical-residual-01}
Suppose the weak problem has coercivity constant
\[
\alpha=0.2
\]
and a computed approximation \(\widetilde u_h\) has a rigorously estimated
continuous residual
\[
\|R_{\widetilde u_h}\|_{V^*}\le 3\times10^{-4}.
\]
Use coercivity to bound \(\|u-\widetilde u_h\|_V\).  Why would the same
conclusion not follow merely from a small Euclidean residual of one chosen
stiffness-matrix representation?
\end{exercise}
\begin{hint}
Apply
\(\|u-\widetilde u_h\|_V\le\alpha^{-1}\|R_{\widetilde u_h}\|_{V^*}\).
Then distinguish a basis-dependent algebraic residual from the continuous
dual residual.
\end{hint}
\begin{solution}
Since \(\alpha^{-1}=5\),
\[
\|u-\widetilde u_h\|_V
\le
5(3\times10^{-4})
=
1.5\times10^{-3}.
\]
This estimate uses the residual as a functional on the full energy space and
the coercivity constant that converts residual size into solution-error size.
A small Euclidean vector residual \(b-A\widetilde c\) depends on the basis and
matrix scaling and, without a norm-equivalence/stability argument, does not by
itself give the displayed \(V\)-norm bound or separate discretization from
solver error.
\end{solution}
