\chapter{Fourier Series}
\label{ch:iii09}

Fourier series decompose periodic functions into orthogonal trigonometric modes. Coefficients are projections, partial sums are finite-dimensional approximations, and convergence depends on both function regularity and the geometry of the trigonometric system.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Fourier coefficients of elementary periodic functions using symmetry before integration;
\item derive partial sums and identify their relation to convolution with Dirichlet-type kernels;
\item apply convergence criteria at continuity points and jump discontinuities with the correct limiting value;
\item use orthogonality to prove coefficient identities and energy estimates;
\item distinguish pointwise, uniform, and \(L^2\) convergence of Fourier series in representative examples;
\item construct examples illustrating slow convergence or Gibbs-type behavior near jumps;
\end{itemize}

\section{Periodic functions}
A $2\pi$-periodic function can be viewed as a function on the circle, making trigonometric modes natural coordinates.

\section{Orthogonality}
Distinct sine and cosine modes are orthogonal on a full period.

\section{Fourier coefficients}
Coefficients are obtained by integrating against the corresponding orthogonal modes.

\section{Partial sums}
The $N$th Fourier partial sum is the orthogonal projection onto trigonometric polynomials of degree at most $N$.

\section{Bessel inequality}
Finite coefficient energy cannot exceed the $L^2$ energy of the function.

\section{Dirichlet kernel}
Partial sums can be written as convolution with the Dirichlet kernel, making convergence an approximate-identity problem with oscillatory kernels.

\section{Pointwise convergence}
Piecewise smooth functions converge to their value at continuity points and to midpoint values at jumps under classical Dirichlet hypotheses.

\section{Gibbs phenomenon}
Near jumps, overshoot persists in relative height while its spatial width shrinks.

\section{Core structural results}

\begin{theorem}[Orthogonality relations]\label{thm:iii09-01}
Distinct trigonometric modes are orthogonal in $L^2[-\pi,\pi]$.
\begin{proof}
Use product-to-sum identities; every nonconstant integer-frequency sine or cosine integrates to zero over a full period.
\end{proof}
\end{theorem}

\begin{theorem}[Bessel inequality]\label{thm:iii09-02}
The square-sum of Fourier coefficients over any finite set of orthonormal modes is bounded by $\|f\|_2^2$.
\begin{proof}
Subtract the finite orthogonal projection from $f$ and expand the nonnegative squared norm using orthogonality.
\end{proof}
\end{theorem}

\begin{theorem}[Dirichlet convergence principle]\label{thm:iii09-03}
Under standard piecewise smooth hypotheses, Fourier partial sums converge at $x$ to $(f(x^-)+f(x^+))/2$.
\begin{proof}
Express the partial sum by the Dirichlet kernel, split the integral into a small neighborhood and its complement, use local one-sided regularity near zero and oscillatory cancellation away from zero.
\end{proof}
\end{theorem}
