\chapter{Spectral and Transform Methods for PDE}
\label{ch:iii28}

Spectral and transform methods solve linear PDEs by diagonalizing the operator. Fourier transforms diagonalize translation-invariant operators, while eigenfunction expansions diagonalize boundary-value operators on bounded domains.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item diagonalize translation-invariant or separable PDE operators by Fourier or eigenfunction transforms;
\item derive transformed algebraic or ordinary differential equations from PDEs;
\item solve model heat, wave, or elliptic equations by inverse transform under the stated hypotheses;
\item interpret spectral multipliers as frequency-dependent propagation, damping, or resolvent factors;
\item compare Fourier-transform, Fourier-series, and eigenfunction-expansion methods according to the domain and boundary conditions;
\item check transformed solutions by substituting back into the original PDE and verifying initial or boundary data;
\end{itemize}

\section{Core structural results}

\begin{theorem}[Heat transform solution]\label{thm:iii28-01}
For $u_t=\Delta u$ on $\mathbb R^n$ with initial data $u_0$, $\widehat u(t,\xi)=e^{-4\pi^2t|\xi|^2}\widehat u_0(\xi)$.
\begin{proof}
Fourier transform converts the PDE into the scalar ODE $\partial_t\widehat u=-4\pi^2|\xi|^2\widehat u$ for each frequency; solve the ODE with the transformed initial data.
\end{proof}
\end{theorem}

\begin{theorem}[Spectral heat expansion]\label{thm:iii28-02}
If $L\phi_k=\lambda_k\phi_k$ on a bounded self-adjoint problem, then $u(t)=\sum_ke^{-t\lambda_k}\langle u_0,\phi_k\rangle\phi_k$ solves $u_t+Lu=0$.
\begin{proof}
Expand the initial data in the eigenbasis; each coefficient satisfies the scalar ODE $c_k'+\lambda_kc_k=0$.
\end{proof}
\end{theorem}

\begin{theorem}[Resolvent diagonalization]\label{thm:iii28-03}
For $\lambda$ off the spectrum, $(L-\lambda)^{-1}$ multiplies each eigenfunction coefficient by $(\lambda_k-\lambda)^{-1}$.
\begin{proof}
Apply $L-\lambda$ to the eigenfunction expansion and solve the scalar coefficient equations independently.
\end{proof}
\end{theorem}
