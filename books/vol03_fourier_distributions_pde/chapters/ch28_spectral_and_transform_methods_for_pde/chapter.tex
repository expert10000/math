\chapter{Spectral and Transform Methods for PDE}
\label{ch:iii28}

Spectral and transform methods solve linear PDEs by diagonalizing the operator. Fourier transforms diagonalize translation-invariant operators, while eigenfunction expansions diagonalize boundary-value operators on bounded domains.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item diagonalize translation-invariant or separable PDE operators by Fourier or eigenfunction transforms;
\item derive transformed algebraic or ordinary differential equations from PDEs;
\item solve model heat, wave, or elliptic equations by inverse transform under the stated hypotheses;
\item interpret spectral multipliers as frequency-dependent propagation, damping, or resolvent factors;
\item compare Fourier-transform, Fourier-series, and eigenfunction-expansion methods according to the domain and boundary conditions;
\item check transformed solutions by substituting back into the original PDE and verifying initial or boundary data;
\end{itemize}

\paragraph{Mathematical setting.}
Whole-space heat data are $u_0\in L^2(\mathbb R^n)$; multipliers define $u\in C([0,\infty);L^2)$, with $u(t)\in H^m$ for every integer $m\ge0$ at $t>0$. For the wave formula take $u_0\in H^1$, $u_1\in L^2$ and interpret $\sin(2\pi|\xi|t)/(2\pi|\xi|)$ as $t$ at zero. The solution belongs to $C(\mathbb R;H^1)\cap C^1(\mathbb R;L^2)$ and satisfies the equation distributionally. For spectral statements assume a densely defined nonnegative self-adjoint operator $L$ on a Hilbert space with compact resolvent and a complete orthonormal eigenbasis. A concrete example is the Dirichlet Laplacian on a bounded Lipschitz domain. Data $u_0$ lie in that Hilbert space. These spectral assumptions are standard theorem inputs, not consequences of boundedness of the domain alone.

\begin{example}[Heat equation diagonalizes frequency by frequency]\label{ex:iii28-expand-01}
For \(u_t=\Delta u\) on \(\mathbb R^n\), Fourier transformation gives
\[
\partial_t\widehat u(t,\xi)
=-4\pi^2|\xi|^2\widehat u(t,\xi).
\]
Solving this scalar ODE yields
\[
\widehat u(t,\xi)
=e^{-4\pi^2t|\xi|^2}\widehat u_0(\xi).
\]
High frequencies decay faster, which simultaneously explains smoothing,
energy decay, and the Gaussian heat kernel after inverse transformation.
\end{example}

\begin{example}[Wave modes oscillate instead of decay]\label{ex:iii28-expand-02}
For \(u_{tt}=\Delta u\), Fourier transformation gives
\[
\partial_{tt}\widehat u+4\pi^2|\xi|^2\widehat u=0.
\]
Each frequency is a harmonic oscillator:
\[
\widehat u(t,\xi)
=\widehat u_0(\xi)\cos(2\pi|\xi|t)
+\frac{\widehat u_1(\xi)}{2\pi|\xi|}
\sin(2\pi|\xi|t),
\]
with the sine quotient interpreted as t at zero frequency. The cosine factor is bounded by one; the velocity multiplier is bounded by |t|. These oscillate without heat-type high-frequency damping.
\end{example}

\begin{example}[Eigenfunction expansion solves the bounded-domain heat problem]\label{ex:iii28-expand-03}
If \(L\phi_k=\lambda_k\phi_k\) with an orthonormal eigenbasis and
\[
u_0=\sum_k c_k\phi_k,
\]
then the heat-type problem \(u_t+Lu=0\) has formal solution
\[
u(t)=\sum_k e^{-t\lambda_k}c_k\phi_k.
\]
Each PDE mode reduces to the scalar ODE
\(c_k'(t)+\lambda_kc_k(t)=0\). Boundary conditions are already encoded in the
eigenfunctions, making spectral expansion the bounded-domain analogue of the
Fourier transform.
\end{example}

\begin{remark}[Spectral truncation, eigen-residuals, and near-spectrum sensitivity]
The spectral formulas in this chapter are exact once the self-adjoint operator,
its domain, and its eigenpairs are fixed.  Numerical PDE work usually replaces
them by a discretized operator \(L_h\), finitely many computed eigenpairs, and
a spectral cutoff.

For a computed normalized discrete eigenpair
\((\widetilde\lambda,\widetilde\phi)\), the quantity
\[
\|L_h\widetilde\phi-\widetilde\lambda\widetilde\phi\|
\]
checks the discrete eigen-equation, while inner products between computed
eigenvectors check loss of orthogonality.  Small values verify the discrete
spectral calculation; they do not by themselves bound the error between
\(L_h\) and the continuum operator \(L\).

Resolvent calculations also inherit conditioning directly from spectral
distance.  For self-adjoint \(L\) and \(\lambda\notin\sigma(L)\),
\[
\|(L-\lambda)^{-1}\|
=
\frac{1}{\operatorname{dist}(\lambda,\sigma(L))}.
\]
Therefore dividing by a symbol or spectral factor near zero can strongly
amplify data, discretization, or solver error.  A small residual is meaningful
only together with the distance-to-spectrum or another stability estimate.

Finally, truncating an eigenfunction expansion introduces a tail that should be
reported in the natural Hilbert/energy norm.  For the heat equation, high
modes may be strongly damped at \(t>0\), while the same cutoff can be much less
accurate near \(t=0\).  The truncation index, residual tolerance, spatial
discretization, and spectral-resolution criterion are therefore separate
pieces of the numerical specification.
\end{remark}

\section{Core structural results}

\begin{theorem}[Heat transform solution]\label{thm:iii28-01}
For $u_0\in L^2(\mathbb R^n)$, $\widehat u(t,\xi)=e^{-4\pi^2t|\xi|^2}\widehat u_0(\xi)$ defines the unique mild L2 heat solution, continuous at $t=0$ in L2 and smooth for $t>0$.
\begin{proof}
On Schwartz data, transformation and differentiation give the scalar ODE and its displayed solution. For L2 data define the solution by this bounded multiplier. DCT and Plancherel give continuity at zero; for $t>0$ every power of $|\xi|$ times the multiplier is bounded, yielding all Sobolev derivatives and differentiability in time away from zero. Extension by density gives the unique contraction semigroup (mild) solution.
\end{proof}
\end{theorem}

\begin{theorem}[Spectral heat expansion]\label{thm:iii28-02}
Under the nonnegative self-adjoint compact-resolvent assumptions above, $u(t)=\sum_ke^{-t\lambda_k}\langle u_0,\phi_k\rangle\phi_k$ converges in the Hilbert space, tends to $u_0$ there as $t\downarrow0$, and solves $u_t+Lu=0$ strongly for $t>0$.
\begin{proof}
Completeness gives square-summable initial coefficients. Since $\lambda_k\ge0$, the exponentials are bounded by one, so dominated convergence for their square sums gives convergence and the initial limit. For $t>0$, $\sup_{s\ge0}s^me^{-ts}<\infty$ for every integer $m$, justifying termwise differentiation in the Hilbert space and membership in $D(L^m)$. Scalar coefficient uniqueness gives uniqueness in this evolution class.
\end{proof}
\end{theorem}

\begin{theorem}[Resolvent diagonalization]\label{thm:iii28-03}
For $\lambda$ off the spectrum, $(L-\lambda)^{-1}$ multiplies each eigenfunction coefficient by $(\lambda_k-\lambda)^{-1}$.
\begin{proof}
Apply $L-\lambda$ to the eigenfunction expansion and solve the scalar coefficient equations independently.
\end{proof}
\end{theorem}

(L-\lambda)u=f
\]
and has residual
\[
r=f-(L-\lambda)\widetilde u,
\qquad
\|r\|\le10^{-6}.
\]

\begin{enumerate}
\item For \(\lambda=3.99\), bound \(\|u-\widetilde u\|\).
\item Repeat for \(\lambda=0\).
\item Explain why the same residual norm can imply very different solution
accuracy.
\end{enumerate}
\end{exercise}
\begin{hint}
Use
\[
u-\widetilde u=(L-\lambda)^{-1}r
\]
and
\[
\|(L-\lambda)^{-1}\|
=
1/\operatorname{dist}(\lambda,\sigma(L)).
\]
\end{hint}
\begin{solution}
For \(\lambda=3.99\), the distance to the spectrum is \(0.01\), hence
\[
\|u-\widetilde u\|
\le
100\,\|r\|
\le
10^{-4}.
\]
For \(\lambda=0\), the spectral distance is \(1\), so
\[
\|u-\widetilde u\|
\le
10^{-6}.
\]
Thus the residual alone does not determine forward solution accuracy.
Near the spectrum the inverse problem is more sensitive, and the same
backward residual can be amplified by a much larger resolvent norm.
\end{solution}
