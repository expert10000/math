\chapter{Sobolev Spaces}
\label{ch:iii21}

Sobolev spaces measure both a function and its weak derivatives in $L^p$. They provide the natural energy spaces for variational PDEs, compactness arguments, and weak boundary conditions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Sobolev norms and determine membership in \(W^{k,p}\) or \(H^k\) for explicit functions;
\item use weak derivatives to formulate Sobolev-space regularity;
\item apply basic embedding or inequality results only after checking dimension, exponent, and domain hypotheses;
\item interpret traces or boundary values in settings where the chapter establishes them;
\item use completeness and Hilbert-space structure of \(H^k\) in variational arguments;
\item distinguish Sobolev regularity from pointwise smoothness by examples and counterexamples;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
Let $\Omega\subset\mathbb R^n$ be open, $k$ a nonnegative integer and $1\le p\le\infty$. The zeroth weak derivative is $u$ itself. For $p<\infty$ use $\|u\|_{W^{k,p}}=(\sum_{|\alpha|\le k}\|D^\alpha u\|_p^p)^{1/p}$, and for $p=\infty$ use the maximum of the component norms. $W_0^{1,p}$ is closure in this norm and $H_0^1=W_0^{1,2}$. On bounded Lipschitz domains and $1\le p<\infty$, the standard trace theorem identifies $W_0^{1,p}$ with the kernel of the continuous trace map into $L^p(\partial\Omega)$; this identification is not claimed for $p=\infty$. Cusp power tests below assume $\alpha>0$ and finite $p$; for $p=\infty$ the cusp is in $W^{1,\infty}$ exactly when $\alpha\ge1$.

\section{Definition of $W^{k,p}$}
A function lies in $W^{k,p}(\Omega)$ when all weak derivatives through order $k$ belong to $L^p$.

\section{Sobolev norms}
Use the p-sum of derivative norms specified above; for p=2 its square is the sum of squared L2 component norms and is induced by an inner product. The plain sum of component norms is equivalent but is not in general a Hilbert norm.


% BEGIN VOL03-EXPANSION III21-example-01
\begin{example}[Testing Sobolev membership of a cusp]\label{ex:iii21-expand-01}
On \((-1,1)\), let \(u(x)=|x|^\alpha\) with \(\alpha>0\). Away from zero,
\[
u'(x)=\alpha\,\operatorname{sgn}(x)|x|^{\alpha-1}.
\]
Thus \(u\in W^{1,p}\) near zero exactly when both
\(|x|^{\alpha p}\) and \(|x|^{(\alpha-1)p}\) are integrable there. The
derivative condition is the restrictive one:
\[
(\alpha-1)p>-1.
\]
This gives a sharp regularity threshold from a one-line power test.
\end{example}
% END VOL03-EXPANSION III21-example-01
\section{$H^k$ notation}
$H^k=W^{k,2}$ is especially important because inner-product methods apply.

\section{Completeness}
Sobolev spaces are Banach spaces, and $H^k$ spaces are Hilbert spaces.

\section{Zero-boundary spaces}
$W_0^{1,p}$ is the closure of compactly supported smooth functions and encodes homogeneous Dirichlet boundary data.

\section{Poincare inequality}
On suitable bounded domains, the $L^p$ norm of a zero-boundary function is controlled by its gradient.


% BEGIN VOL03-EXPANSION III21-example-02
\begin{example}[Poincare turns gradient into a full norm]\label{ex:iii21-expand-02}
For \(u\in H_0^1(\Omega)\) on a bounded Lipschitz domain,
\[
\|u\|_2\le C\|\nabla u\|_2.
\]
Therefore
\[
\|u\|_{H^1}^2=\|u\|_2^2+\|\nabla u\|_2^2
\le (C^2+1)\|\nabla u\|_2^2.
\]
On the zero-boundary space, gradient energy alone controls the full Sobolev
norm. This is the coercivity mechanism behind the Dirichlet weak problem.
\end{example}
% END VOL03-EXPANSION III21-example-02
\section{Embeddings}
Standard Sobolev embedding theorem (full proof deferred): on a bounded Lipschitz domain, $W^{1,p}\hookrightarrow L^{np/(n-p)}$ for $1\le p<n$, and $W^{1,p}\hookrightarrow C^{0,1-n/p}(\overline\Omega)$ for $n<p<\infty$. At $p=n>1$ one obtains every finite $L^q$, not continuity in general. The one-dimensional $W^{1,1}$ case has absolutely continuous representatives.


% BEGIN VOL03-EXPANSION III21-example-03
\begin{example}[Sobolev embedding requires the exponent-dimension check]\label{ex:iii21-expand-03}
In one dimension, \(H^1(0,1)\) functions have continuous representatives and
obey a bound obtained from the fundamental theorem plus Cauchy--Schwarz. In
higher dimensions the same \(H^1\) regularity need not imply continuity.
Therefore an embedding statement must always be checked against dimension,
derivative order, and integrability exponent rather than transferred by
analogy.
\end{example}
% END VOL03-EXPANSION III21-example-03
\section{Compactness}
Standard Rellich theorem (full proof deferred): if $\Omega$ is bounded Lipschitz, $W^{1,p}(\Omega)\hookrightarrow L^q(\Omega)$ is compact for $1\le p<n$ and $1\le q<np/(n-p)$, and for $n\le p<\infty$ and finite $q\ge1$. Bounded sequences then have strongly convergent subsequences in these target spaces.

\section{Core structural results}

\begin{theorem}[Completeness of $W^{k,p}$]\label{thm:iii21-01}
For $1\le p\le\infty$, $W^{k,p}$ is complete.
\begin{proof}
A Cauchy sequence is Cauchy in $L^p$ for every derivative component; pass to limits in each component and use distributional convergence to verify that the limiting components are the weak derivatives of the limiting function.
\end{proof}
\end{theorem}

\begin{theorem}[Poincare inequality]\label{thm:iii21-02}
On a bounded Lipschitz domain, $\|u\|_{L^p}\le C\|\nabla u\|_{L^p}$ for $u\in W_0^{1,p}$.
\begin{proof}
Place $\Omega$ inside a cube of side $\ell$ and extend $u\in C_c^\infty(\Omega)$ by zero. On each line parallel to the first coordinate, the fundamental theorem and Holder give $\int|u|^p\le\ell^p\int|D_1u|^p$ for finite $p$. Integrate the remaining coordinates. For $p=\infty$, $|u|\le\ell\|D_1u\|_\infty$. Pass to the defining closure of $W_0^{1,p}$.
\end{proof}
\end{theorem}

\begin{theorem}[Hilbert structure of $H^k$]\label{thm:iii21-03}
$H^k$ is a Hilbert space with inner product obtained by summing $L^2$ inner products of weak derivatives through order $k$.
\begin{proof}
Completeness follows from Sobolev completeness and the norm is induced by the stated positive-definite inner product.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[A one-dimensional \(H^1\) membership test]\label{ex:iii21-worked-01}
On \((-1,1)\), \(u(x)=|x|\) belongs to \(H^1\): \(u\in L^2\) and its weak
derivative \(\operatorname{sgn}(x)\) is also in \(L^2\). But its second
distributional derivative is \(2\delta_0\), so \(u\notin H^2(-1,1)\).
\end{example}

\begin{example}[One-dimensional Poincaré inequality]\label{ex:iii21-worked-02}
For \(u\in H_0^1(0,L)\), the absolutely continuous representative satisfies
\[
u(x)=\int_0^x u'(t)\,dt.
\]
Cauchy--Schwarz gives
\[
|u(x)|^2\le x\int_0^L|u'|^2,
\]
and integration in \(x\) yields a concrete bound
\(\|u\|_2\le C_L\|u'\|_2\).
\end{example}

\begin{example}[Sobolev norm records derivatives]\label{ex:iii21-worked-03}
For \(W^{1,p}(\Omega)\),
\[
\|u\|_{W^{1,p}}
=\left(\|u\|_p^p+\sum_j\|\partial_ju\|_p^p\right)^{1/p}
\]
for finite \(p\). The space is not merely an \(L^p\) space: membership requires
weak derivatives in the same integrability class.
\end{example}

\begin{example}[A power singularity in \(W^{1,p}\)]\label{ex:iii21-worked-04}
On \((0,1)\), let \(u(x)=x^\alpha\). Then
\(u'\sim x^{\alpha-1}\). Thus \(u\in W^{1,p}(0,1)\) precisely when both
\(\alpha p>-1\) and \((\alpha-1)p>-1\). For \(\alpha\ge0\), the derivative
condition is the restrictive one.
\end{example}

\section{Solved dossiers}
Each dossier combines several ideas from the chapter in a longer argument. The aim is to make hypotheses, calculations, and proof strategy explicit.

\begin{problem}[A one-dimensional \(H^1\) membership test]\label{prob:iii21-dossier-01}
Determine whether \(|x|\) belongs to \(H^1(-1,1)\) and \(H^2(-1,1)\).
\end{problem}
\begin{solution}
It belongs to \(H^1\) because both \(u\) and \(u'\) are in \(L^2\), but not
to \(H^2\) because the second derivative is the Dirac mass \(2\delta_0\).
\end{solution}

\begin{problem}[One-dimensional Poincaré inequality]\label{prob:iii21-dossier-02}
Derive a one-dimensional Poincaré estimate for \(H_0^1(0,L)\).
\end{problem}
\begin{solution}
Use the fundamental theorem for the absolutely continuous representative,
then Cauchy--Schwarz and integrate the resulting pointwise estimate.
\end{solution}

\begin{problem}[Sobolev norm records derivatives]\label{prob:iii21-dossier-03}
State the membership test for \(W^{1,p}(\Omega)\).
\end{problem}
\begin{solution}
One needs \(u\in L^p(\Omega)\) and every first weak derivative
\(\partial_ju\in L^p(\Omega)\).
\end{solution}

\begin{problem}[A power singularity in \(W^{1,p}\)]\label{prob:iii21-dossier-04}
Determine the \(W^{1,p}(0,1)\) condition for \(u(x)=x^\alpha\).
\end{problem}
\begin{solution}
Require integrability of \(x^{\alpha p}\) and
\(x^{(\alpha-1)p}\) at zero; equivalently
\(\alpha p>-1\) and \((\alpha-1)p>-1\).
\end{solution}

\begin{problem}[Zero-boundary space]\label{prob:iii21-dossier-05}
Give the intrinsic definition of \(H_0^1(\Omega)\).
\end{problem}
\begin{solution}
It is the closure of \(C_c^\infty(\Omega)\) in the \(H^1\) norm. Identifying
this with zero trace requires additional domain/trace hypotheses.
\end{solution}

\begin{problem}[Completeness]\label{prob:iii21-dossier-06}
Why is \(W^{1,p}\) complete for \(1\le p\le\infty\)?
\end{problem}
\begin{solution}
A Cauchy sequence is Cauchy in \(L^p\) together with each weak derivative;
the limits satisfy the weak derivative identities after passing to the limit.
\end{solution}

\begin{problem}[Sobolev versus classical differentiability]\label{prob:iii21-dossier-07}
Can a Sobolev function fail to be classically differentiable everywhere?
\end{problem}
\begin{solution}
Yes. Weak derivatives are \(L^p\) objects defined a.e.; \(|x|\in H^1\) is
the elementary example with a corner.
\end{solution}

\begin{problem}[Poincaré role]\label{prob:iii21-dossier-08}
Why is Poincaré inequality important on \(H_0^1\)?
\end{problem}
\begin{solution}
It controls the \(L^2\) part of the norm by the gradient, making
\(\|\nabla u\|_2\) an equivalent norm and supplying coercivity for elliptic
problems.
\end{solution}

\end{solution}
