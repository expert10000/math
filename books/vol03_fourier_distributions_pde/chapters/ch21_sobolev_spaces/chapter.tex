\chapter{Sobolev Spaces}
\label{ch:iii21}

Sobolev spaces measure both a function and its weak derivatives in $L^p$. They provide the natural energy spaces for variational PDEs, compactness arguments, and weak boundary conditions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Sobolev norms and determine membership in \(W^{k,p}\) or \(H^k\) for explicit functions;
\item use weak derivatives to formulate Sobolev-space regularity;
\item apply basic embedding or inequality results only after checking dimension, exponent, and domain hypotheses;
\item interpret traces or boundary values in settings where the chapter establishes them;
\item use completeness and Hilbert-space structure of \(H^k\) in variational arguments;
\item distinguish Sobolev regularity from pointwise smoothness by examples and counterexamples;
\end{itemize}

\section{Definition of $W^{k,p}$}
A function lies in $W^{k,p}(\Omega)$ when all weak derivatives through order $k$ belong to $L^p$.

\section{Sobolev norms}
Sum the $L^p$ norms of derivatives up to order $k$; for $p=2$ this gives a Hilbert-space norm.


% BEGIN VOL03-EXPANSION III21-example-01
\begin{example}[Testing Sobolev membership of a cusp]\label{ex:iii21-expand-01}
On \((-1,1)\), let \(u(x)=|x|^\alpha\) with \(\alpha>0\). Away from zero,
\[
u'(x)=\alpha\,\operatorname{sgn}(x)|x|^{\alpha-1}.
\]
Thus \(u\in W^{1,p}\) near zero exactly when both
\(|x|^{\alpha p}\) and \(|x|^{(\alpha-1)p}\) are integrable there. The
derivative condition is the restrictive one:
\[
(\alpha-1)p>-1.
\]
This gives a sharp regularity threshold from a one-line power test.
\end{example}
% END VOL03-EXPANSION III21-example-01
\section{$H^k$ notation}
$H^k=W^{k,2}$ is especially important because inner-product methods apply.

\section{Completeness}
Sobolev spaces are Banach spaces, and $H^k$ spaces are Hilbert spaces.

\section{Zero-boundary spaces}
$W_0^{1,p}$ is the closure of compactly supported smooth functions and encodes homogeneous Dirichlet boundary data.

\section{Poincare inequality}
On suitable bounded domains, the $L^p$ norm of a zero-boundary function is controlled by its gradient.


% BEGIN VOL03-EXPANSION III21-example-02
\begin{example}[Poincare turns gradient into a full norm]\label{ex:iii21-expand-02}
For \(u\in H_0^1(\Omega)\) on a bounded Lipschitz domain,
\[
\|u\|_2\le C\|\nabla u\|_2.
\]
Therefore
\[
\|u\|_{H^1}^2=\|u\|_2^2+\|\nabla u\|_2^2
\le (C^2+1)\|\nabla u\|_2^2.
\]
On the zero-boundary space, gradient energy alone controls the full Sobolev
norm. This is the coercivity mechanism behind the Dirichlet weak problem.
\end{example}
% END VOL03-EXPANSION III21-example-02
\section{Embeddings}
Sufficient Sobolev regularity implies improved integrability or continuity.


% BEGIN VOL03-EXPANSION III21-example-03
\begin{example}[Sobolev embedding requires the exponent-dimension check]\label{ex:iii21-expand-03}
In one dimension, \(H^1(0,1)\) functions have continuous representatives and
obey a bound obtained from the fundamental theorem plus Cauchy--Schwarz. In
higher dimensions the same \(H^1\) regularity need not imply continuity.
Therefore an embedding statement must always be checked against dimension,
derivative order, and integrability exponent rather than transferred by
analogy.
\end{example}
% END VOL03-EXPANSION III21-example-03

\section{Core structural results}

\begin{theorem}[Completeness of $W^{k,p}$]\label{thm:iii21-01}
For $1\le p\le\infty$, $W^{k,p}$ is complete.
\begin{proof}
A Cauchy sequence is Cauchy in $L^p$ for every derivative component; pass to limits in each component and use distributional convergence to verify that the limiting components are the weak derivatives of the limiting function.
\end{proof}
\end{theorem}

\begin{theorem}[Poincare inequality]\label{thm:iii21-02}
On a bounded Lipschitz domain, $\|u\|_{L^p}\le C\|\nabla u\|_{L^p}$ for $u\in W_0^{1,p}$.
\begin{proof}
Prove first for smooth compactly supported functions by integrating along suitable line segments or using contradiction and compactness, then extend by density.
\end{proof}
\end{theorem}

\begin{theorem}[Hilbert structure of $H^k$]\label{thm:iii21-03}
$H^k$ is a Hilbert space with inner product obtained by summing $L^2$ inner products of weak derivatives through order $k$.
\begin{proof}
Completeness follows from Sobolev completeness and the norm is induced by the stated positive-definite inner product.
\end{proof}
\end{theorem}

\end{solution}
