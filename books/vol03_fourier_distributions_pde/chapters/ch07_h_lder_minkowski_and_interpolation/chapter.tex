\chapter{Hölder, Minkowski and Interpolation}
\label{ch:iii07}

Hölder and Minkowski are the fundamental inequalities of $L^p$ geometry. Interpolation then explains how information at two exponents controls intermediate exponents.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item apply Hölder's inequality with correctly chosen conjugate exponents;
\item derive Minkowski's inequality from Hölder and use it to prove the triangle inequality in \(L^p\);
\item compute interpolation exponents from endpoint data in elementary interpolation inequalities;
\item identify equality or near-equality cases in Hölder-type estimates;
\item use Hölder and Minkowski to control products, sums, and integral operators;
\item construct examples showing why exponent restrictions such as \(1/p+1/q=1\) are structural rather than cosmetic;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

For $1<p<\infty$, the conjugate exponent $q$ satisfies $1/p+1/q=1$.

\section{Young inequality}
The scalar bound $ab\le a^p/p+b^q/q$ encodes the convex duality behind Hölder.

\section{Hölder inequality}
Products of $L^p$ and $L^q$ functions lie in $L^1$ with sharp norm control.

\begin{example}[Holder with equality]\label{ex:iii07-expand-01}
For \(p=q=2\), Holder is Cauchy--Schwarz. If \(g=cf\), then
\[
\int|fg|=|c|\|f\|_2^2=\|f\|_2\|g\|_2,
\]
so equality holds. This agrees with the general equality condition that
\(|f|^p\) and \(|g|^q\) are proportional almost everywhere.
\end{example}

\begin{example}[Equality structure]\label{ex:iii07-expand-02}
After normalizing nonzero \(f\) and \(g\), equality in Young's inequality
occurs precisely when the normalized powers agree. Hence Holder equality is
governed by
\[
\frac{|f|^p}{\|f\|_p^p}
=
\frac{|g|^q}{\|g\|_q^q}
\quad\text{a.e.}
\]
This identifies extremizers, not just the sharp constant.
\end{example}

\section{Interpolation of norms}
Intermediate $L^r$ norms are logarithmically convex between endpoint norms.

\begin{example}[Interpolating L3 between L2 and L6]\label{ex:iii07-expand-03}
Solve
\[
\frac13=\frac{\theta}{2}+\frac{1-\theta}{6},
\]
so \(\theta=1/2\). The log-convex interpolation inequality gives
\[
\|f\|_3\le\|f\|_2^{1/2}\|f\|_6^{1/2}.
\]
The weights act on reciprocal exponents, which is the key bookkeeping rule.
\end{example}

\section{Core structural results}

\begin{theorem}[Hölder inequality]\label{thm:iii07-01}
If $1<p,q<\infty$ are conjugate, then $\int|fg|\le\|f\|_p\|g\|_q$.
\begin{proof}
Normalize the two norms to one, apply Young's inequality pointwise to the normalized product, integrate, and rescale.
\end{proof}
\end{theorem}

\begin{theorem}[Minkowski inequality]\label{thm:iii07-02}
For $1\le p<\infty$, $\|f+g\|_p\le\|f\|_p+\|g\|_p$.
\begin{proof}
For $p>1$, estimate $\int|f+g|^p$ by pairing $|f+g|^{p-1}$ with $|f|+|g|$ and apply Hölder; the case $p=1$ is immediate.
\end{proof}
\end{theorem}

\begin{theorem}[Log-convex interpolation]\label{thm:iii07-03}
If $1/r=\theta/p+(1-\theta)/q$, then $\|f\|_r\le\|f\|_p^\theta\|f\|_q^{1-\theta}$.
\begin{proof}
Factor $|f|^r$ into suitable powers of $|f|^p$ and $|f|^q$ and apply Hölder with reciprocal weights.
\end{proof}
\end{theorem}

\end{exercise}
\begin{hint}

\end{hint}

\end{solution}

\end{exercise}

\end{solution}

\end{solution}

The output is bounded by the Lq norm of the kernel section times the Lp norm of f.
\end{solution}
