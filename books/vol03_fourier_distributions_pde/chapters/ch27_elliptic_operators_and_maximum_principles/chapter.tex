\chapter{Elliptic Operators and Maximum Principles}
\label{ch:iii27}

Maximum principles turn the sign of an elliptic operator into strong qualitative control. They yield uniqueness, comparison, boundary estimates, and positivity of Green functions without explicitly solving the PDE.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item classify elementary second-order operators as elliptic by inspecting the principal symbol or coefficient matrix;
\item apply weak or classical maximum principles only after checking ellipticity and sign hypotheses;
\item use maximum principles to prove uniqueness of boundary-value problems;
\item derive comparison principles from the maximum principle;
\item construct examples showing how sign changes or degeneracy can invalidate the conclusion;
\item connect elliptic estimates with qualitative control of interior extrema and boundary data;
\end{itemize}

\section{Core structural results}

\begin{theorem}[Weak maximum principle]\label{thm:iii27-01}
If $Lu\ge0$ for a uniformly elliptic operator with appropriate lower-order signs, then the maximum of $u$ is controlled by the boundary maximum.
\begin{proof}
At a hypothetical strict interior maximum, the gradient vanishes and the Hessian is negative semidefinite; ellipticity makes the principal second-order contribution have the wrong sign, contradicting the differential inequality after the standard perturbation argument.
\end{proof}
\end{theorem}

\begin{theorem}[Comparison principle]\label{thm:iii27-02}
If $Lu\ge Lv$ in $\Omega$ and $u\le v$ on the boundary, then $u\le v$ in $\Omega$ under the maximum-principle hypotheses.
\begin{proof}
Apply the weak maximum principle to $u-v$.
\end{proof}
\end{theorem}

\begin{theorem}[Dirichlet uniqueness]\label{thm:iii27-03}
A homogeneous Dirichlet solution of $Lu=0$ is identically zero under the maximum-principle hypotheses.
\begin{proof}
Apply the maximum principle to $u$ and to $-u$ to show both maximum and minimum are zero.
\end{proof}
\end{theorem}
