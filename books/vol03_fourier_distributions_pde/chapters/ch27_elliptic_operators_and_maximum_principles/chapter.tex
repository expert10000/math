\chapter{Elliptic Operators and Maximum Principles}
\label{ch:iii27}

Maximum principles turn the sign of an elliptic operator into strong qualitative control. They yield uniqueness, comparison, boundary estimates, and positivity of Green functions without explicitly solving the PDE.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item classify elementary second-order operators as elliptic by inspecting the principal symbol or coefficient matrix;
\item apply weak or classical maximum principles only after checking ellipticity and sign hypotheses;
\item use maximum principles to prove uniqueness of boundary-value problems;
\item derive comparison principles from the maximum principle;
\item construct examples showing how sign changes or degeneracy can invalidate the conclusion;
\item connect elliptic estimates with qualitative control of interior extrema and boundary data;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{structure}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{limit or representation}\;\longrightarrow\;\text{application}.}
\]

\paragraph{Mathematical setting.}
Fix $Lu=\sum a_{ij}(x)D_{ij}u+\sum b_i(x)D_iu+c(x)u$ with positive principal sign, real bounded continuous coefficients, symmetric $A$, $\xi^TA(x)\xi\ge\theta|\xi|^2$ for $\theta>0$, and $c\le0$. Let $\Omega$ be bounded and $u\in C^2(\Omega)\cap C(\overline\Omega)$. Then $Lu\ge0$ implies $\max_{\overline\Omega}u\le\max(0,\max_{\partial\Omega}u)$; for $c=0$ the zero may be omitted. Under the same hypotheses comparison uses $Lu\ge Lv$, not the reverse inequality. When using $-\Delta$, reverse the operator inequality. The strong principle for $Lu\ge0$ requires a nonnegative attained maximum and connected $\Omega$; if $c=0$, any attained maximum suffices. Hopf at a strict nonnegative boundary maximum additionally requires an interior tangent ball and C1 regularity at that point; the outward normal derivative is strictly positive. Strong and Hopf principles are standard theorems with full proofs deferred.

\section{Elliptic operators}
Pointwise ellipticity means positive definiteness of the symmetric principal matrix; uniform ellipticity additionally requires a positive lower bound independent of the point. This chapter uses uniform ellipticity.

\begin{example}[Checking uniform ellipticity from a matrix]\label{ex:iii27-expand-01}
Consider
\[
Lu=3u_{xx}+2u_{xy}+2u_{yy}.
\]
The symmetric principal matrix is
\[
A=\begin{pmatrix}3&1\\1&2\end{pmatrix}.
\]
Its eigenvalues are positive because \(\operatorname{tr}A=5>0\) and
\(\det A=5>0\). Hence
\[
\xi^T A\xi\ge \lambda_{\min}|\xi|^2,
\]
so the operator is uniformly elliptic with ellipticity constant
\(\lambda_{\min}>0\).
\end{example}

\section{Weak maximum principle}
For suitable operators, a subsolution cannot achieve a strict positive interior maximum beyond its boundary maximum.

\section{Strong maximum principle}
Under the connected-domain hypotheses above, a subsolution with Lu nonnegative attaining a nonnegative interior maximum is constant. If c=0 the maximum need not be nonnegative. Apply this to -u for the corresponding minimum statement.

\section{Comparison principle}
If $Lu\ge Lv$ and $u\le v$ on the boundary, then $u\le v$ inside, under the chapter hypotheses.

\begin{example}[Comparison gives uniqueness in one line]\label{ex:iii27-expand-02}
Suppose \(Lu=Lv\) in \(\Omega\) and \(u=v\) on the boundary. Apply the
comparison principle first to \(u,v\) and then to \(v,u\). It gives
\(u\le v\) and \(v\le u\), hence \(u=v\). Equivalently the maximum principle
applied to \(w=u-v\) and \(-w\) shows both extrema of \(w\) are zero.
\end{example}

\begin{example}[A quadratic barrier converts geometry into a bound]\label{ex:iii27-expand-03}
In a ball \(B_R\), the function
\[
b(x)=R^2-|x|^2
\]
is nonnegative inside, zero on the boundary, and satisfies
\[
-\Delta b=2n.
\]
Multiples of \(b\) can therefore dominate solutions of Poisson inequalities
with bounded forcing. Comparing \(u\) to \(C b\) transfers a source bound into
an interior pointwise estimate without solving the PDE explicitly.
\end{example}

\section{Core structural results}

\begin{theorem}[Weak maximum principle]\label{thm:iii27-01}
Under the coefficient, domain and regularity hypotheses above, $Lu\ge0$ implies $\max_{\overline\Omega}u\le\max(0,\max_{\partial\Omega}u)$. If $c=0$, the right side is the boundary maximum.
\begin{proof}
Choose $\psi(x)=e^{\kappa x_1}$. Bounded coefficients and $a_{11}\ge\theta$ imply $L\psi=(\kappa^2a_{11}+\kappa b_1+c)\psi>0$ for large $\kappa$. Thus $L(u+\varepsilon\psi)>0$. A positive interior maximum of this perturbed function has zero gradient, negative semidefinite Hessian and $c(u+\varepsilon\psi)\le0$, contradicting the strict inequality. Its maximum is therefore at most the larger of zero and its boundary maximum. Let $\varepsilon\downarrow0$. If $c=0$, subtract the boundary maximum first.
\end{proof}
\end{theorem}

\begin{theorem}[Comparison principle]\label{thm:iii27-02}
If $Lu\ge Lv$ in $\Omega$ and $u\le v$ on the boundary, then $u\le v$ in $\Omega$ under the maximum-principle hypotheses.
\begin{proof}
Apply the weak maximum principle to $u-v$.
\end{proof}
\end{theorem}

\begin{theorem}[Dirichlet uniqueness]\label{thm:iii27-03}
A homogeneous Dirichlet solution of $Lu=0$ is identically zero under the maximum-principle hypotheses.
\begin{proof}
Apply the maximum principle to $u$ and to $-u$ to show both maximum and minimum are zero.
\end{proof}
\end{theorem}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Wrong lower-order sign]\label{exr:iii27-expand-10}
Why can a bad zeroth-order sign invalidate a maximum principle?
\end{exercise}
\begin{hint}
At an interior maximum, the lower-order term may reinforce rather than contradict the maximum.
\end{hint}
\begin{solution}
On $(0,\pi)$, $u(x)=\sin x$ satisfies $u''+u=0$, has zero boundary values and a positive interior maximum. Here $c=1>0$, so the stated maximum and uniqueness conclusions fail.
\end{solution}

\begin{exercise}[Degenerate ellipticity]\label{exr:iii27-expand-11}

\begin{solution}
The interior Hessian sign may not force a contradiction.
\end{solution}

\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Positivity]\label{exr:iii27-expand-13}
Why do nonnegative forcing/boundary data often imply a sign for elliptic solutions?
\end{exercise}
\begin{hint}
Apply comparison with zero or a suitable barrier.
\end{hint}
\begin{solution}
For $-\Delta u=f\ge0$ and nonnegative boundary values, apply the positive-principal-sign maximum principle to $-u$: $\Delta(-u)=f\ge0$ and its boundary values are nonpositive. Hence $-u\le0$ and $u\ge0$.
\end{solution}

\end{hint}
\begin{solution}
A discrete solution violating known bounds can signal instability or implementation error.
\end{solution}
