\chapter{Branches of the Logarithm and Roots}
\label{ch:iv14}

Branches of logarithms and roots exist exactly when the topology of the domain allows one to choose arguments continuously. The obstruction is winding around the origin.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct branches of the logarithm on simply connected domains avoiding the origin;
\item define branches of powers and roots from a chosen logarithm;
\item track argument ranges and branch cuts explicitly;
\item determine how analytic continuation around the origin changes logarithm and root branches;
\item distinguish local branches from globally single-valued functions;
\item choose branch cuts adapted to contour-integral and mapping problems;
\end{itemize}

\section{Multivalued logarithm}
Solving $e^w=z$ gives values differing by integer multiples of $2\pi i$.

% BEGIN VOL04-EXPANSION IV14-example-01
\begin{example}[Principal logarithm]\label{ex:iv14-expand-01}
On \(\mathbb C\setminus(-\infty,0]\), define
\[
\operatorname{Log}z=\log|z|+i\operatorname{Arg}z,\qquad -\pi<\operatorname{Arg}z<\pi.
\]
This is holomorphic and satisfies \((\operatorname{Log}z)'=1/z\). The removed ray prevents a loop around zero from forcing the argument to jump by \(2\pi\).
\end{example}
% END VOL04-EXPANSION IV14-example-01
\section{Branches}
A branch of logarithm on a domain is a holomorphic function $L$ with $e^{L(z)}=z$.

\section{Argument functions}
Writing $L(z)=\log|z|+i\operatorname{Arg}(z)$ links branches of logarithm to continuous arguments.

\section{Slit domains}
Removing a ray from the plane yields standard simply connected domains supporting logarithm branches.

% BEGIN VOL04-EXPANSION IV14-example-02
\begin{example}[A square-root branch]\label{ex:iv14-expand-02}
On the same slit plane, define
\[
\sqrt z=\exp\!\left(\tfrac12\operatorname{Log}z\right).
\]
Then \((\sqrt z)^2=z\), and the chosen branch has positive real value on the positive real axis. The second branch is its negative.
\end{example}
% END VOL04-EXPANSION IV14-example-02
\section{Nth roots}
A branch of $z^{1/n}$ is obtained from a logarithm branch by exponentiation.

\section{Powers}
For complex exponent $\alpha$, define $z^\alpha=e^{\alpha L(z)}$ relative to a chosen logarithm branch.

\section{Monodromy}
Analytic continuation around the origin changes the logarithm by $2\pi i$ and roots by roots of unity.

% BEGIN VOL04-EXPANSION IV14-example-03
\begin{example}[Why the punctured plane has no global logarithm]\label{ex:iv14-expand-03}
If a holomorphic logarithm \(L\) existed on \(\mathbb C\setminus\{0\}\), then \(L'=1/z\). But
\[
\int_{|z|=1}\frac{dz}{z}=2\pi i\ne0,
\]
whereas the integral of a derivative around a closed curve is zero. Thus no single valued holomorphic logarithm exists on the punctured plane.
\end{example}
% END VOL04-EXPANSION IV14-example-03
\section{Integral criterion}
A logarithm branch exists when the integral of $1/z$ around every closed contour in the domain vanishes.

\section{Topological obstruction}
Nonzero winding around zero prevents a globally single-valued logarithm.

\section{Core structural results}

\begin{theorem}[Logarithm branch criterion]\label{thm:iv14-01}
On a domain $D\subset\mathbb C\setminus\{0\}$, a holomorphic logarithm exists if $\int_\gamma dz/z=0$ for every closed contour $\gamma$ in $D$.
\begin{proof}
The vanishing integral condition gives a primitive $L$ of $1/z$. Then $e^{-L(z)}z$ has derivative zero, so after adding a constant to $L$ one obtains $e^{L(z)}=z$.
\end{proof}
\end{theorem}

\begin{theorem}[Simply connected logarithm theorem]\label{thm:iv14-02}
Every simply connected domain avoiding zero admits a holomorphic logarithm.
\begin{proof}
The function $1/z$ is holomorphic on the domain, and Cauchy's theorem makes all its closed contour integrals vanish. Apply the logarithm branch criterion.
\end{proof}
\end{theorem}

\begin{theorem}[Root from logarithm]\label{thm:iv14-03}
If $L$ is a logarithm branch on $D$, then $R(z)=e^{L(z)/n}$ is a holomorphic branch of the $n$th root.
\begin{proof}
Directly, $R(z)^n=e^{L(z)}=z$, and holomorphicity follows from composition.
\end{proof}
\end{theorem}

\begin{exercise}[Root branches]\label{exr:iv14-expand-05}
How many holomorphic branches of the \(n\)-th root differ by constant factors on a connected branch domain?
\end{exercise}
\begin{hint}
Multiply one branch by the \(n\)-th roots of unity.
\end{hint}
\begin{solution}
There are \(n\) choices, obtained from one branch by factors \(e^{2\pi i k/n}\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Derivative of a logarithm branch]\label{exr:iv14-expand-06}
Prove that any holomorphic branch \(L\) with \(e^L=z\) satisfies \(L'=1/z\).
\end{exercise}
\begin{hint}
Differentiate the identity \(e^{L(z)}=z\).
\end{hint}
\begin{solution}
The chain rule gives \(e^{L(z)}L'(z)=1\). Since \(e^{L(z)}=z\), one gets \(L'(z)=1/z\).
\end{solution}

\begin{exercise}[Branches differ by constants]\label{exr:iv14-expand-07}
Show that two logarithm branches on a connected domain differ by an integer multiple of \(2\pi i\).
\end{exercise}
\begin{hint}
Exponentiate their difference.
\end{hint}
\begin{solution}
If \(L_1,L_2\) are branches, \(e^{L_1-L_2}=1\). The difference is continuous and takes values in the discrete set \(2\pi i\mathbb Z\), hence is constant on the connected domain.
\end{solution}

\begin{exercise}[Root from logarithm]\label{exr:iv14-expand-08}
Prove that \(\exp(L/n)\) is an \(n\)-th root branch when \(L\) is a logarithm branch.
\end{exercise}
\begin{hint}
Raise the expression to the \(n\)-th power.
\end{hint}
\begin{solution}
Its \(n\)-th power is \(e^L=z\), and holomorphicity follows by composition.
\end{solution}

\begin{exercise}[No logarithm on a winding domain]\label{exr:iv14-expand-09}
Show that if a domain contains a closed curve with nonzero winding about zero, it cannot admit a holomorphic logarithm of \(z\).
\end{exercise}
\begin{hint}
A logarithm would have derivative \(1/z\).
\end{hint}
\begin{solution}
The integral of its derivative around the curve would be zero, but \(\int dz/z=2\pi i\) times the nonzero winding number, a contradiction.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Branch crossing]\label{exr:iv14-expand-10}
Why is the principal logarithm discontinuous if one tries to include both sides of its cut with the same argument convention?
\end{exercise}
\begin{hint}
Approach a negative real point from above and below.
\end{hint}
\begin{solution}
The principal arguments approach \(\pi\) and \(-\pi\), differing by \(2\pi\), so no continuous single valued extension crosses the cut.
\end{solution}

\begin{exercise}[Punctured annulus]\label{exr:iv14-expand-11}
Does every annulus around zero admit a single valued logarithm?
\end{exercise}
\begin{hint}
An annulus contains loops winding once around zero.
\end{hint}
\begin{solution}
No. The integral of \(1/z\) on such a loop is nonzero, obstructing a logarithm.
\end{solution}

\begin{exercise}[Branch independence of powers]\label{exr:iv14-expand-12}
Is \(z^\alpha\) independent of logarithm branch for arbitrary complex \(\alpha\)?
\end{exercise}
\begin{hint}
Changing a logarithm by \(2\pi i k\) changes the exponential by a factor.
\end{hint}
\begin{solution}
In general no. The factor is \(e^{2\pi i k\alpha}\), which equals one for every integer \(k\) only in special cases such as integer \(\alpha\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Fractional power domain]\label{exr:iv14-expand-13}
Why is a slit plane a natural domain for \(z^{1/3}\)?
\end{exercise}
\begin{hint}
It supports a single valued logarithm.
\end{hint}
\begin{solution}
Once a logarithm is fixed, \(z^{1/3}=e^{L/3}\) is holomorphic there, avoiding the monodromy caused by circling zero.
\end{solution}

\begin{exercise}[Primitive of reciprocal]\label{exr:iv14-expand-14}
On what kind of region can \(1/z\) have a holomorphic primitive?
\end{exercise}
\begin{hint}
A primitive is a logarithm branch.
\end{hint}
\begin{solution}
Exactly on regions where a logarithm branch exists; in particular, simply connected regions avoiding zero support such a primitive.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Branch on a sector]\label{exr:iv14-expand-15}
Construct a logarithm on the sector \(\alpha<\arg z<\beta\) when \(\beta-\alpha<2\pi\).
\end{exercise}
\begin{hint}
Choose the argument continuously in that interval.
\end{hint}
\begin{solution}
Define \(L(z)=\log|z|+i\arg z\) with \(\arg z\in(\alpha,\beta)\). The sector cannot wrap fully around zero, so the argument is single valued.
\end{solution}

\begin{exercise}[Monodromy of the square root]\label{exr:iv14-expand-16}
What happens to a locally chosen square root after analytic continuation once around zero?
\end{exercise}
\begin{hint}
The logarithm changes by \(2\pi i\).
\end{hint}
\begin{solution}
Half the logarithm changes by \(\pi i\), so the exponential is multiplied by \(-1\). A second circuit returns to the original value.
\end{solution}
