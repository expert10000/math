\chapter{Elliptic Functions}
\label{ch:iv28}

Elliptic functions are meromorphic functions periodic with respect to a rank-two lattice. Compactness of the associated torus forces strong balance laws for zeros, poles, and residues.

\section{Double periodicity}
An elliptic function satisfies $f(z+\omega)=f(z)$ for every lattice period $\omega\in\Lambda$.

% BEGIN VOL04-EXPANSION IV28-example-01
\begin{example}[Residues in one period cell]\label{ex:iv28-expand-01}
Let \(f\) be elliptic for \(\Lambda\), with no poles on the boundary of a fundamental parallelogram \(P\). Opposite edge integrals cancel by periodicity, so
\[
\int_{\partial P}f(z)\,dz=0.
\]
The residue theorem gives
\[
\sum_{a\in P}\operatorname{Res}(f,a)=0.
\]
Therefore an elliptic function cannot have exactly one simple pole modulo the lattice with nonzero residue.
\end{example}

\begin{example}[Zero and pole count from the logarithmic derivative]\label{ex:iv28-expand-02}
For a nonzero elliptic function \(f\), the logarithmic derivative \(f'/f\) is also elliptic. Opposite edges cancel, hence
\[
\frac1{2\pi i}\int_{\partial P}\frac{f'}{f}\,dz=0.
\]
By the argument principle, the left side equals the number of zeros minus the number of poles in \(P\), counted with multiplicity. Thus every period cell contains equally many zeros and poles.
\end{example}

\section{Core structural results}

\begin{theorem}[Residue sum theorem]\label{thm:iv28-01}
The sum of residues of an elliptic function in a fundamental parallelogram containing no poles on its boundary is zero.
\begin{proof}
Integrate around the boundary. Opposite edges cancel because the function is periodic and the orientations are opposite. The residue theorem then forces the interior residue sum to vanish.
\end{proof}
\end{theorem}

\begin{theorem}[Equal zero and pole count]\label{thm:iv28-02}
A nonzero elliptic function has equally many zeros and poles in a fundamental parallelogram, counted with multiplicity.
\begin{proof}
Apply the argument principle to $f'/f$. This quotient is elliptic, so opposite-edge contour integrals cancel, leaving zero; hence zeros minus poles equals zero.
\end{proof}
\end{theorem}

\begin{theorem}[No single simple pole]\label{thm:iv28-03}
There is no elliptic function with exactly one simple pole modulo the lattice.
\begin{proof}
Such a function would have one nonzero residue in a fundamental parallelogram, contradicting the residue-sum theorem.
\end{proof}
\end{theorem}

\(r=-3\).

\end{exercise}
\begin{hint}
Use zero-pole balance.
\end{hint}
\begin{solution}
It has five zeros counted with multiplicity.
\end{solution}

\begin{exercise}[Single pole]\label{exr:iv28-expand-03}
Can an elliptic function have one simple pole modulo the lattice?
\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
No, because its nonzero residue could not be canceled.
\end{solution}

\begin{exercise}[Derivative periodicity]\label{exr:iv28-expand-04}
If \(f\) is elliptic, is \(f'\) elliptic?
\end{exercise}
\begin{hint}
Differentiate the period identity.
\end{hint}
\begin{solution}
Yes. Differentiating \(f(z+\omega)=f(z)\) gives \(f'(z+\omega)=f'(z)\).
\end{solution}

\begin{exercise}[Log derivative]\label{exr:iv28-expand-05}
Why is \(f'/f\) elliptic where defined?
\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
The quotient inherits every period of \(f\), with meromorphic singularities at zeros and poles.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Residue theorem]\label{exr:iv28-expand-06}
Prove residue balance for an elliptic function.
\end{exercise}
\begin{hint}
Integrate around a fundamental parallelogram.
\end{hint}
\begin{solution}
Each edge integral cancels with the opposite translated edge. The boundary integral is zero, so the residue theorem forces the interior residue sum to vanish.
\end{solution}

\begin{exercise}[Zero-pole theorem]\label{exr:iv28-expand-07}
Prove that zeros and poles balance in a period cell.
\end{exercise}
\begin{hint}
Apply the argument principle to \(f'/f\).
\end{hint}
\begin{solution}
The boundary integral cancels by periodicity and equals \(2\pi i\) times zeros minus poles, so the difference is zero.
\end{solution}

\begin{exercise}[Nonconstant implies pole]\label{exr:iv28-expand-08}
Prove that a nonconstant elliptic function must have a pole.
\end{exercise}
\begin{hint}
Descend to the compact torus.
\end{hint}
\begin{solution}
Without poles it would be holomorphic on the compact torus and therefore constant, contradiction.
\end{solution}

\begin{exercise}[Position constraint idea]\label{exr:iv28-expand-09}
Explain why integrating \(z f'(z)/f(z)\) around a period cell leads to a lattice-valued constraint on sums of zeros and poles.
\end{exercise}
\begin{hint}
Opposite-edge cancellation now leaves period multiples.
\end{hint}
\begin{solution}
Residues give the weighted sum of zero locations minus pole locations. Boundary terms differ by lattice periods, so the resulting difference is determined modulo the lattice.
\end{solution}
