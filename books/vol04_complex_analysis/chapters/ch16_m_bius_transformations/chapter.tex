\chapter{Möbius Transformations}
\label{ch:iv16}

Möbius transformations are the automorphisms of the Riemann sphere generated by fractional linear maps. They preserve generalized circles and provide the elementary coordinate changes of global complex analysis.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Möbius transformations from images of three points;
\item determine images of circles and lines under fractional-linear transformations;
\item use cross ratios to verify invariance and solve mapping constraints;
\item classify fixed points and normal forms of simple Möbius maps;
\item construct automorphisms of the disk and upper half-plane;
\item track the point at infinity correctly in calculations on the Riemann sphere;
\end{itemize}

\section{Fractional linear maps}
A Möbius map has form $T(z)=(az+b)/(cz+d)$ with $ad-bc\ne0$, extended naturally to infinity.

\section{Matrix representation}
Scalar multiples of the matrix $\begin{pmatrix}a&b\\c&d\end{pmatrix}$ define the same Möbius transformation.

\section{Composition}
Composition corresponds to matrix multiplication modulo nonzero scalars.

\section{Three-point normalization}
A unique Möbius transformation sends any ordered triple of distinct sphere points to any other such triple.

\section{Cross-ratio}
The cross-ratio is invariant under Möbius transformations.

\section{Generalized circles}
Lines and circles on the plane are circles on the sphere, and Möbius transformations map generalized circles to generalized circles.

\section{Elementary generators}
Translations, dilations/rotations, and inversion generate all Möbius transformations.

\section{Fixed points}
Fixed points solve a quadratic equation and organize the elementary dynamics of the transformation.

\section{Disk and half-plane maps}
Specific Möbius maps give conformal equivalences between the unit disk and half-plane.

\section{Core structural results}

\begin{theorem}[Three-point theorem]\label{thm:iv16-01}
Given distinct $z_1,z_2,z_3$ and distinct $w_1,w_2,w_3$ on the Riemann sphere, there is a unique Möbius transformation with $T(z_j)=w_j$.
\begin{proof}
Construct maps sending each triple to $(0,1,\infty)$ using cross-ratio formulas, then compose one with the inverse of the other. Uniqueness follows because a Möbius map fixing three distinct points is the identity.
\end{proof}
\end{theorem}

\begin{theorem}[Cross-ratio invariance]\label{thm:iv16-02}
Möbius transformations preserve the cross-ratio of four points.
\begin{proof}
It suffices to check translations, nonzero scalings, and inversion, which generate the Möbius group; direct algebra verifies invariance for each generator.
\end{proof}
\end{theorem}

\begin{theorem}[Generalized-circle preservation]\label{thm:iv16-03}
Every Möbius transformation maps circles and lines to circles or lines.
\begin{proof}
Translations and similarities preserve both classes; inversion maps circles/lines to generalized circles by an explicit equation calculation. Use the generator decomposition.
\end{proof}
\end{theorem}
