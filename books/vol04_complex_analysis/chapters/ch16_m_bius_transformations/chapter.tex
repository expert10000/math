\chapter{Möbius Transformations}
\label{ch:iv16}

Möbius transformations are the automorphisms of the Riemann sphere generated by fractional linear maps. They preserve generalized circles and provide the elementary coordinate changes of global complex analysis.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Möbius transformations from images of three points;
\item determine images of circles and lines under fractional-linear transformations;
\item use cross ratios to verify invariance and solve mapping constraints;
\item classify fixed points and normal forms of simple Möbius maps;
\item construct automorphisms of the disk and upper half-plane;
\item track the point at infinity correctly in calculations on the Riemann sphere;
\end{itemize}

\section{Fractional linear maps}
A Möbius map has form $T(z)=(az+b)/(cz+d)$ with $ad-bc\ne0$, extended naturally to infinity.

% BEGIN VOL04-EXPANSION IV16-example-01
\begin{example}[Upper half-plane to unit disk]\label{ex:iv16-expand-01}
The Möbius map
\[
T(z)=\frac{z-i}{z+i}
\]
sends \(i\) to \(0\). If \(x\in\mathbb R\), then \(|x-i|=|x+i|\), so \(|T(x)|=1\). A point such as \(2i\) maps to \(1/3\), showing that the upper half-plane maps to the unit disk.
\end{example}
% END VOL04-EXPANSION IV16-example-01
\section{Matrix representation}
Scalar multiples of the matrix $\begin{pmatrix}a&b\\c&d\end{pmatrix}$ define the same Möbius transformation.

\section{Composition}
Composition corresponds to matrix multiplication modulo nonzero scalars.

\section{Three-point normalization}
A unique Möbius transformation sends any ordered triple of distinct sphere points to any other such triple.

% BEGIN VOL04-EXPANSION IV16-example-02
\begin{example}[Three-point normalization]\label{ex:iv16-expand-02}
The cross-ratio map
\[
T(z)=\frac{(z-z_1)(z_2-z_3)}{(z-z_3)(z_2-z_1)}
\]
sends \(z_1\mapsto0\), \(z_2\mapsto1\), and \(z_3\mapsto\infty\). Thus any three distinct points on the Riemann sphere can be normalized to \(0,1,\infty\), after which many geometric questions become simpler.
\end{example}
% END VOL04-EXPANSION IV16-example-02
\section{Cross-ratio}
The cross-ratio is invariant under Möbius transformations.

\section{Generalized circles}
Lines and circles on the plane are circles on the sphere, and Möbius transformations map generalized circles to generalized circles.

\section{Elementary generators}
Translations, dilations/rotations, and inversion generate all Möbius transformations.

% BEGIN VOL04-EXPANSION IV16-example-03
\begin{example}[A line becomes a circle]\label{ex:iv16-expand-03}
For \(T(z)=1/z\), the vertical line \(\operatorname{Re}z=1\) can be written \(z=1+iy\). If \(w=1/z=u+iv\), then solving \(z=1/w\) and imposing real part one gives
\[
\left(u-\tfrac12\right)^2+v^2=\tfrac14.
\]
Thus a line not through the pole maps to a circle through the origin.
\end{example}

\section{Core structural results}

\begin{theorem}[Three-point theorem]\label{thm:iv16-01}
Given distinct $z_1,z_2,z_3$ and distinct $w_1,w_2,w_3$ on the Riemann sphere, there is a unique Möbius transformation with $T(z_j)=w_j$.
\begin{proof}
Construct maps sending each triple to $(0,1,\infty)$ using cross-ratio formulas, then compose one with the inverse of the other. Uniqueness follows because a Möbius map fixing three distinct points is the identity.
\end{proof}
\end{theorem}

\begin{theorem}[Cross-ratio invariance]\label{thm:iv16-02}
Möbius transformations preserve the cross-ratio of four points.
\begin{proof}
It suffices to check translations, nonzero scalings, and inversion, which generate the Möbius group; direct algebra verifies invariance for each generator.
\end{proof}
\end{theorem}

\begin{theorem}[Generalized-circle preservation]\label{thm:iv16-03}
Every Möbius transformation maps circles and lines to circles or lines.
\begin{proof}
Translations and similarities preserve both classes; inversion maps circles/lines to generalized circles by an explicit equation calculation. Use the generator decomposition.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Evaluate a map]\label{exr:iv16-expand-01}
For \(T(z)=(z-1)/(z+1)\), compute \(T(1)\), \(T(0)\), and \(T(\infty)\).
\end{exercise}
\begin{hint}
Use the leading coefficient ratio at infinity.
\end{hint}
\begin{solution}
The values are \(0\), \(-1\), and \(1\).
\end{solution}

\begin{exercise}[Inverse map]\label{exr:iv16-expand-02}
Find the inverse of \(w=(z-i)/(z+i)\).
\end{exercise}
\begin{hint}
Solve algebraically for \(z\).
\end{hint}
\begin{solution}
One obtains \(z=i(1+w)/(1-w)\).
\end{solution}

\begin{exercise}[Pole and infinity]\label{exr:iv16-expand-03}
For \(T(z)=(2z+3)/(z-4)\), where does \(4\) map, and where does infinity map?
\end{exercise}
\begin{hint}
Use the sphere convention.
\end{hint}
\begin{solution}
The point \(4\) maps to infinity, and infinity maps to the leading coefficient ratio \(2\).
\end{solution}

\begin{exercise}[Determinant]\label{exr:iv16-expand-04}
Is \((2z+2)/(z+1)\) a genuine Möbius transformation?
\end{exercise}
\begin{hint}
Check \(ad-bc\).
\end{hint}
\begin{solution}
No. The determinant is zero and the expression simplifies to the constant two away from the cancelled point.
\end{solution}

\begin{exercise}[Fixed points]\label{exr:iv16-expand-05}
Find the fixed points of \(T(z)=1/z\) on the sphere.
\end{exercise}
\begin{hint}
Solve \(z=1/z\).
\end{hint}
\begin{solution}
The finite fixed points are \(z=1\) and \(z=-1\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Inverse is Möbius]\label{exr:iv16-expand-06}
Prove that a Möbius transformation with nonzero determinant has a Möbius inverse.
\end{exercise}
\begin{hint}
Solve \(w=(az+b)/(cz+d)\) for \(z\).
\end{hint}
\begin{solution}
Rearrangement gives \(z=(dw-b)/(-cw+a)\), whose determinant equals the original nonzero determinant up to sign.
\end{solution}

\begin{exercise}[Composition closure]\label{exr:iv16-expand-07}
Show that the composition of two Möbius transformations is Möbius.
\end{exercise}
\begin{hint}
Represent each transformation by a nonsingular two by two matrix up to scalar.
\end{hint}
\begin{solution}
Composition corresponds to matrix multiplication. The product remains nonsingular, so the resulting fractional linear map is Möbius.
\end{solution}

\begin{exercise}[Generalized circles]\label{exr:iv16-expand-08}
Explain why Möbius transformations send circles and lines to circles or lines.
\end{exercise}
\begin{hint}
It suffices to check translations, nonzero scalings and rotations, and inversion.
\end{hint}
\begin{solution}
Affine maps preserve circles and lines. Inversion maps generalized circles to generalized circles, and every Möbius transformation decomposes into these elementary maps.
\end{solution}

\begin{exercise}[Three-point uniqueness]\label{exr:iv16-expand-09}
Prove that a Möbius transformation is uniquely determined by the images of three distinct points.
\end{exercise}
\begin{hint}
Compose two candidates with the inverse of one and study a map fixing three points.
\end{hint}
\begin{solution}
A nonidentity Möbius map has at most two fixed points because its fixed-point equation is quadratic. Hence a map fixing three distinct sphere points is the identity, proving uniqueness.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Zero determinant]\label{exr:iv16-expand-10}
What fails when \(ad-bc=0\)?
\end{exercise}
\begin{hint}
The numerator and denominator become proportional.
\end{hint}
\begin{solution}
The fractional expression is constant where defined and has no Möbius inverse, so it is not an automorphism of the sphere.
\end{solution}

\begin{exercise}[Forgetting infinity]\label{exr:iv16-expand-11}
Why must infinity be included when discussing global Möbius maps?
\end{exercise}
\begin{hint}
Denominator zeros map naturally to infinity.
\end{hint}
\begin{solution}
Adding infinity turns every nonsingular fractional linear expression into a bijective holomorphic map of the Riemann sphere.
\end{solution}

\begin{exercise}[Interior versus boundary]\label{exr:iv16-expand-12}
If a Möbius map sends the real axis to the unit circle, is checking boundary images alone enough to know which half-plane maps inside?
\end{exercise}
\begin{hint}
Test one interior point.
\end{hint}
\begin{solution}
No. The two complementary regions could be interchanged. Evaluating one point in the chosen half-plane determines the side.
\end{solution}

\end{solution}
