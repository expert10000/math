\chapter{Complex Integration}
\label{ch:iv04}

Complex integration integrates a complex-valued function along an oriented curve. Parameterization, estimates, primitives, and deformation ideas prepare the way for Cauchy's theorem.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item parameterize piecewise smooth contours and compute complex line integrals directly;
\item apply contour reversal, concatenation, and reparameterization rules correctly;
\item estimate contour integrals using the ML inequality;
\item identify primitives and use them to evaluate path-independent integrals;
\item distinguish path independence from mere equality along one family of curves;
\item choose contours adapted to the geometry of the integrand and domain;
\end{itemize}

\section{Parametrized curves}
A piecewise $C^1$ curve $\gamma:[a,b]\to\mathbb C$ carries orientation and tangent data.

\section{Contour integral}
Define $\int_\gamma f(z)dz=\int_a^bf(\gamma(t))\gamma'(t)dt$.

\section{Reparameterization}
Orientation-preserving reparameterizations leave contour integrals unchanged; reversing orientation changes the sign.

\section{ML estimate}
$|\int_\gamma fdz|\le L(\gamma)\sup_\gamma|f|$ gives the basic quantitative bound.

\section{Primitives}
If $F'=f$, then contour integrals of $f$ depend only on endpoints.

\section{Closed curves}
Functions with primitives integrate to zero around every closed curve.

\section{Polygonal and circular contours}
Line segments, circles, and piecewise polygonal paths provide standard computational contours.

\section{Deformation preview}
When holomorphicity holds in the intervening region, later theorems will make integrals invariant under suitable contour deformations.

\section{Parameter singularities}
Care is required when curves pass through poles, branch cuts, or points where the integrand is not defined.

\section{Core structural results}

\begin{theorem}[Reparameterization invariance]\label{thm:iv04-01}
An orientation-preserving $C^1$ change of parameter leaves the contour integral unchanged.
\begin{proof}
Substitute the new parameter into the real integral and use the chain rule and ordinary change-of-variables formula.
\end{proof}
\end{theorem}

\begin{theorem}[ML inequality]\label{thm:iv04-02}
If $f$ is continuous on a rectifiable curve $\gamma$, then $|\int_\gamma fdz|\le L(\gamma)\max_\gamma|f|$.
\begin{proof}
Take absolute values in the parametrized integral and bound $|f(\gamma(t))|$ by its maximum; the remaining integral of $|\gamma'|$ is the curve length.
\end{proof}
\end{theorem}

\begin{theorem}[Fundamental theorem for contour integrals]\label{thm:iv04-03}
If $F'=f$ on a domain containing $\gamma$, then $\int_\gamma fdz=F(\gamma(b))-F(\gamma(a))$.
\begin{proof}
Differentiate $F(\gamma(t))$ by the chain rule and integrate the resulting ordinary derivative over the parameter interval.
\end{proof}
\end{theorem}

\begin{theorem}[Primitive criterion on connected domains]\label{thm:iv04-04}
If every closed contour integral of continuous $f$ vanishes in a connected open set, then $f$ has a primitive.
\begin{proof}
Fix a base point and define $F(z)$ by integrating $f$ along any path from the base point to $z$; closed-integral vanishing gives path independence, and a short terminal segment proves $F'=f$.
\end{proof}
\end{theorem}
