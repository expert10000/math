\chapter{Complex Integration}
\label{ch:iv04}

Complex integration integrates a complex-valued function along an oriented curve. Parameterization, estimates, primitives, and deformation ideas prepare the way for Cauchy's theorem.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item parameterize piecewise smooth contours and compute complex line integrals directly;
\item apply contour reversal, concatenation, and reparameterization rules correctly;
\item estimate contour integrals using the ML inequality;
\item identify primitives and use them to evaluate path-independent integrals;
\item distinguish path independence from mere equality along one family of curves;
\item choose contours adapted to the geometry of the integrand and domain;
\end{itemize}

\begin{example}[Parametrizing a line segment]\label{ex:iv04-expand-01}
Let \(\gamma(t)=t(1+i)\) for \(0\le t\le1\). Then \(\gamma'(t)=1+i\), and
\[
\int_\gamma z\,dz=\int_0^1 t(1+i)(1+i)\,dt
=\frac{(1+i)^2}{2}=i.
\]
The calculation makes the definition operational: substitute the parametrization and multiply by its complex velocity.
\end{example}

\begin{example}[A circle integral with orientation]\label{ex:iv04-expand-02}
For the positively oriented circle \(\gamma(t)=Re^{it}\), \(0\le t\le2\pi\),
\[
\int_\gamma \overline z\,dz
=\int_0^{2\pi}Re^{-it}\,iRe^{it}\,dt
=2\pi iR^2.
\]
Reversing the orientation changes the sign. The example also shows that closed contour integrals need not vanish for nonholomorphic integrands.
\end{example}

\begin{example}[A primitive makes the path irrelevant]\label{ex:iv04-expand-03}
For \(f(z)=3z^2\), a primitive is \(F(z)=z^3\). Therefore every piecewise smooth path from \(z=-1\) to \(z=2+i\) has
\[
\int_\gamma 3z^2\,dz=F(2+i)-F(-1)=(2+i)^3+1.
\]
No parametrization of the chosen path is needed once a primitive is known.
\end{example}

\section{Parameter singularities}
Care is required when curves pass through poles, branch cuts, or points where the integrand is not defined.

% VOL04-NUMERICAL-CONTOUR-IV04
\begin{remark}[Computed contour integrals and singularity clearance]
The contour integral
\[
\int_\gamma f(z)\,dz
=
\int_a^b f(\gamma(t))\gamma'(t)\,dt
\]
is an exact object and is invariant under orientation-preserving
reparameterization.  A finite quadrature rule applied to the parameter
integral is a different object: changing the parameterization changes the
locations and density of the sampled points and can therefore change the
numerical error even when the exact contour integral is unchanged.

For a computed value \(Q_N\), a refinement quantity such as
\[
\eta_N=|Q_{2N}-Q_N|
\]
is a useful convergence indicator when the same contour, parameterization, and
quadrature family are used.  It is not automatically a rigorous bound on
\[
\left|\int_\gamma f(z)\,dz-Q_{2N}\right|.
\]
Such a bound requires an error theorem or another verified estimate appropriate
to the integrand and rule.

Distance from singularities is also part of the numerical specification.  If
the integrand contains a factor \((z-a)^{-1}\), then along the contour
\[
\left|\frac{1}{z-a}\right|
\le
\frac{1}{\delta},
\qquad
\delta=\min_{z\in\gamma}|z-a|.
\]
A small \(\delta\) can make the parameter integrand sharply varying and amplify
evaluation or quadrature error even though the pole remains strictly off the
contour.  Numerical contour calculations should therefore report the contour,
parameterization, quadrature/refinement rule, tolerance, and any relevant
singularity clearance.
\end{remark}

\section{Core structural results}

\begin{theorem}[Reparameterization invariance]\label{thm:iv04-01}
An orientation-preserving $C^1$ change of parameter leaves the contour integral unchanged.
\begin{proof}
Substitute the new parameter into the real integral and use the chain rule and ordinary change-of-variables formula.
\end{proof}
\end{theorem}

\begin{theorem}[ML inequality]\label{thm:iv04-02}
If $f$ is continuous on a rectifiable curve $\gamma$, then $|\int_\gamma fdz|\le L(\gamma)\max_\gamma|f|$.
\begin{proof}
Take absolute values in the parametrized integral and bound $|f(\gamma(t))|$ by its maximum; the remaining integral of $|\gamma'|$ is the curve length.
\end{proof}
\end{theorem}

\begin{theorem}[Fundamental theorem for contour integrals]\label{thm:iv04-03}
If $F'=f$ on a domain containing $\gamma$, then $\int_\gamma fdz=F(\gamma(b))-F(\gamma(a))$.
\begin{proof}
Differentiate $F(\gamma(t))$ by the chain rule and integrate the resulting ordinary derivative over the parameter interval.
\end{proof}
\end{theorem}

\begin{theorem}[Primitive criterion on connected domains]\label{thm:iv04-04}
If every closed contour integral of continuous $f$ vanishes in a connected open set, then $f$ has a primitive.
\begin{proof}
Fix a base point and define $F(z)$ by integrating $f$ along any path from the base point to $z$; closed-integral vanishing gives path independence, and a short terminal segment proves $F'=f$.
\end{proof}
\end{theorem}

\begin{exercise}[Contour-quadrature refinement indicator]\label{exr:iv04-numerical-contour-01}
Two computations of the same parameterized contour integral give
\[
Q_N=1+2i,
\qquad
Q_{2N}=1.0003+2.0004i.
\]
Compute
\[
\eta_N=|Q_{2N}-Q_N|.
\]
What does this number test, and why is it not by itself a rigorous error bound
for the exact contour integral?
\end{exercise}
\begin{hint}
The difference is \(0.0003+0.0004i\).  Its modulus forms a \(3\)-\(4\)-\(5\)
triangle after scaling by \(10^{-4}\).
\end{hint}
\begin{solution}
We have
\[
\eta_N
=
|0.0003+0.0004i|
=
5\times10^{-4}.
\]
This shows that the two chosen resolutions agree to about that scale under the
same numerical procedure.  It is a refinement indicator, not automatically a
bound on the error of \(Q_{2N}\).  A rigorous error conclusion additionally
requires assumptions and an estimate linking refinement differences to the
true quadrature remainder; nearby singularities, poor sampling, or correlated
errors can make two approximations close for the wrong reason.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Reparametrization]\label{exr:iv04-expand-06}
Prove invariance of a contour integral under an orientation preserving smooth reparametrization.
\end{exercise}
\begin{hint}
Make a one variable substitution in the parameter integral.
\end{hint}
\begin{solution}
If \(t=\phi(s)\) with \(\phi'>0\), then \((\gamma\circ\phi)'=\gamma'(\phi)\phi'\). The ordinary substitution formula gives the same integral.
\end{solution}

\begin{exercise}[Concatenation]\label{exr:iv04-expand-07}
Prove that the integral over a concatenated path is the sum of the two integrals.
\end{exercise}
\begin{hint}
Split the parameter interval at the joining time.
\end{hint}
\begin{solution}
After a standard reparametrization, the parameter integral splits into two ordinary integrals, one for each component path.
\end{solution}

\begin{exercise}[ML inequality]\label{exr:iv04-expand-08}
Prove \(|\int_\gamma f\,dz|\le ML\) when \(|f|\le M\) and the contour length is \(L\).
\end{exercise}
\begin{hint}
Take absolute values inside the parameter integral.
\end{hint}
\begin{solution}
The triangle inequality gives at most \(\int M|\gamma'(t)|dt=M L\).
\end{solution}

\begin{exercise}[Primitive theorem]\label{exr:iv04-expand-09}
If \(F'=f\), prove \(\int_\gamma f\,dz=F(\gamma(b))-F(\gamma(a))\).
\end{exercise}
\begin{hint}
Differentiate \(F(\gamma(t))\).
\end{hint}
\begin{solution}
The chain rule gives \((F\circ\gamma)'=f(\gamma(t))\gamma'(t)\). The real fundamental theorem of calculus then gives the endpoint difference.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{hint}
Use conjugation.
\end{hint}

\end{solution}

\begin{exercise}[Endpoint data alone]\label{exr:iv04-expand-11}
Can endpoint data determine \(\int_\gamma f\,dz\) for an arbitrary continuous \(f\)?
\end{exercise}
\begin{hint}
Path independence requires additional structure such as a primitive.
\end{hint}
\begin{solution}
No. Without a primitive or a relevant holomorphic theorem, different paths with the same endpoints may give different integrals.
\end{solution}

\begin{exercise}[Orientation forgotten]\label{exr:iv04-expand-12}
What error results from reversing a contour but keeping the same integral value?
\end{exercise}
\begin{hint}
Track the sign of the derivative of the reversed parametrization.
\end{hint}
\begin{solution}
Reversal multiplies the integral by \(-1\). Forgetting orientation therefore changes the answer by a sign.
\end{solution}

\begin{exercise}[Arc contribution estimate]\label{exr:iv04-expand-13}

\end{hint}

\end{solution}

\end{hint}

The integral becomes a sum of ordinary one dimensional integrals on simple intervals, making both quadrature and error control transparent.
\end{solution}
