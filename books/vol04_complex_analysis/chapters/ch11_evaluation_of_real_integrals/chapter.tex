\chapter{Evaluation of Real Integrals}
\label{ch:iv11}

Real integrals can often be evaluated by embedding the integrand into a meromorphic complex function and choosing contours whose auxiliary pieces vanish or are explicitly computable.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item select contour geometries that convert real integrals into residue calculations;
\item estimate large-arc and small-arc contributions and justify their vanishing limits;
\item handle even rational integrands with semicircular contours;
\item use indentation or principal-value contours when poles lie on the real axis;
\item track orientation and branch choices when extracting the desired real integral;
\item distinguish ordinary improper integrals from Cauchy principal values;
\end{itemize}

% BEGIN VOL04-EXPANSION IV11-example-01
\begin{example}[The basic rational integral]\label{ex:iv11-expand-01}
For \(a>0\), integrate \(1/(z^2+a^2)\) over the upper semicircle. The only enclosed pole is \(ia\), with residue \(1/(2ia)\). The arc contribution tends to zero because the integrand is \(O(R^{-2})\). Hence
\[
\int_{-\infty}^{\infty}\frac{dx}{x^2+a^2}=2\pi i\frac1{2ia}=\frac{\pi}{a}.
\]
\end{example}

\begin{example}[A quartic denominator]\label{ex:iv11-expand-03}
For \(1/(z^4+1)\), the upper half-plane poles are \(e^{i\pi/4}\) and \(e^{3i\pi/4}\). Their residues are \(1/(4z_k^3)\). Summing them and multiplying by \(2\pi i\), while the large arc vanishes, gives
\[
\int_{-\infty}^{\infty}\frac{dx}{x^4+1}=\frac{\pi}{\sqrt2}.
\]
Evenness then yields \(\int_0^\infty dx/(x^4+1)=\pi/(2\sqrt2)\).
\end{example}

\section{Core structural results}

\begin{theorem}[Upper-half-plane residue method]\label{thm:iv11-01}
Suppose a rational function has no real poles and decays fast enough that the upper semicircle contribution vanishes. Then its real-line integral equals $2\pi i$ times the residues of poles in the upper half-plane.
\begin{proof}
Apply the residue theorem to the closed contour consisting of the real segment and upper semicircle, then send the radius to infinity using the decay estimate.
\end{proof}
\end{theorem}

\begin{theorem}[Unit-circle substitution]\label{thm:iv11-02}
For a $2\pi$-periodic rational expression in $\sin t$ and $\cos t$, the substitution $z=e^{it}$ converts the integral into a contour integral on $|z|=1$.
\begin{proof}
Use $dt=dz/(iz)$, $\cos t=(z+z^{-1})/2$, and $\sin t=(z-z^{-1})/(2i)$; simplify to a rational function of $z$ and apply residues inside the unit circle.
\end{proof}
\end{theorem}

\begin{theorem}[Principal-value indentation rule]\label{thm:iv11-03}
A simple pole on a smooth contour contributes half of the usual residue term under a symmetric indentation, with sign determined by the indentation orientation.
\begin{proof}
Replace a small neighborhood of the pole by a semicircular detour, approximate the integrand by its residue divided by $z-a$, and compute the limiting half-circle integral.
\end{proof}
\end{theorem}

\end{exercise}
\begin{hint}
Use the upper half-plane exponential contour.
\end{hint}
\begin{solution}
The value is \(\pi e^{-b}\).
\end{solution}

\begin{exercise}[Half line]\label{exr:iv11-expand-03}
Evaluate \(\int_0^\infty dx/(x^2+1)\).
\end{exercise}
\begin{hint}
Use evenness.
\end{hint}
\begin{solution}
It is half of \(\pi\), so the value is \(\pi/2\).
\end{solution}

\begin{exercise}[Arc decay]\label{exr:iv11-expand-04}
If a rational integrand is \(O(R^{-3})\) on a semicircle of radius \(R\), what order bound holds for the arc integral?
\end{exercise}
\begin{hint}
Multiply the supremum by arc length.
\end{hint}
\begin{solution}
The arc length is \(O(R)\), so the integral is \(O(R^{-2})\) and tends to zero.
\end{solution}

\begin{exercise}[Quartic half line]\label{exr:iv11-expand-05}
Use the full line value to compute \(\int_0^\infty dx/(x^4+1)\).
\end{exercise}

The integrand is even.
\end{hint}

The value is \(\pi/(2\sqrt2)\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Semicircle arc lemma for rational decay]\label{exr:iv11-expand-06}
Prove that if \(|f(z)|\le C/R^{1+\epsilon}\) on the upper semicircle of radius \(R\), then the arc integral tends to zero.
\end{exercise}
\begin{hint}
Use the ML inequality and arc length \(\pi R\).
\end{hint}
\begin{solution}
The bound is \(\pi R C/R^{1+\epsilon}=\pi C/R^\epsilon\to0\).
\end{solution}

\begin{exercise}[Cosine from exponential integral]\label{exr:iv11-expand-07}
Justify extracting the cosine integral as the real part of an exponential contour integral.
\end{exercise}
\begin{hint}
Use absolute convergence or a suitable limiting argument.
\end{hint}
\begin{solution}
Since \(e^{ibx}=\cos(bx)+i\sin(bx)\), linearity separates real and imaginary parts. For the rational denominator used here, both limits are controlled by the contour estimates.
\end{solution}

\begin{exercise}[Even reduction]\label{exr:iv11-expand-08}
Prove that an integrable even function satisfies \(\int_{-\infty}^{\infty}f=2\int_0^\infty f\).
\end{exercise}
\begin{hint}
Substitute \(x=-t\) on the negative half line.
\end{hint}
\begin{solution}
The negative half line integral becomes \(\int_0^\infty f(-t)dt=\int_0^\infty f(t)dt\), and adding the positive half gives twice that value.
\end{solution}

\begin{exercise}[Choosing the half-plane]\label{exr:iv11-expand-09}
For \(b<0\), explain why the lower half-plane is the natural choice for \(e^{ibz}\).
\end{exercise}
\begin{hint}
Inspect \(|e^{ib(x+iy)}|\).
\end{hint}
\begin{solution}
The modulus is \(e^{-by}\). When \(b<0\), this decays for \(y<0\), so the lower semicircle provides damping rather than growth.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Insufficient decay]\label{exr:iv11-expand-10}
Why does an \(O(1/R)\) bound on a semicircle not by itself force the arc integral to vanish?
\end{exercise}
\begin{hint}
Multiply by the arc length.
\end{hint}
\begin{solution}
The ML bound is only \(O(1)\), so it need not tend to zero. Stronger decay or oscillatory cancellation is required.
\end{solution}

\begin{exercise}[Real pole]\label{exr:iv11-expand-11}
What obstructs applying the ordinary upper semicircle argument when the integrand has a pole on the real axis?
\end{exercise}
\begin{hint}
The contour passes through the singularity.
\end{hint}
\begin{solution}
The real integral and contour integral are not ordinary integrals under the standard theorem. One needs indentation and often a principal value interpretation.
\end{solution}

\begin{exercise}[Wrong exponential half-plane]\label{exr:iv11-expand-12}
What happens if \(b>0\) but the contour for \(e^{ibz}\) is closed in the lower half-plane?
\end{exercise}
\begin{hint}
Check the exponential modulus.
\end{hint}
\begin{solution}
There \(y<0\), so \(e^{-by}\) grows exponentially. The large arc estimate generally fails.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Cauchy density normalization]\label{exr:iv11-expand-13}
Use the rational integral to verify that \(p(x)=a/(\pi(x^2+a^2))\) integrates to one for \(a>0\).
\end{exercise}
\begin{hint}
Factor the constant outside the known integral.
\end{hint}
\begin{solution}
The integral is \(a/\pi\cdot\pi/a=1\), confirming normalization of the Cauchy density.
\end{solution}

\begin{exercise}[Transform pair]\label{exr:iv11-expand-14}

\end{hint}

\end{solution}
