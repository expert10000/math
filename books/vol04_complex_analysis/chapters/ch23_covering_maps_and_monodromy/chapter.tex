\chapter{Covering Maps and Monodromy}
\label{ch:iv23}

Covering maps formalize the idea of many sheets lying over one base domain. Path lifting and monodromy translate topology of loops into permutations of analytic branches.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify the local lifting property for elementary covering maps;
\item lift paths and homotopies once a starting point is fixed;
\item compute monodromy permutations from continuation around loops;
\item relate deck transformations to symmetries of a covering;
\item distinguish simply connected base domains from domains with nontrivial monodromy;
\item use covering-space language to organize branches of logarithms and roots;
\end{itemize}

\section{Covering definition}
A covering map is locally a disjoint union of homeomorphic sheets over evenly covered neighborhoods.

% BEGIN VOL04-EXPANSION IV23-example-01
\begin{example}[Lifting a loop under the exponential map]\label{ex:iv23-expand-01}
Consider \(p(w)=e^w\) and the unit-circle loop \(\gamma(t)=e^{2\pi it}\). Starting at \(w_0=0\), the lift
\[
\widetilde\gamma(t)=2\pi it
\]
satisfies \(e^{\widetilde\gamma(t)}=\gamma(t)\). Its endpoint is \(2\pi i\), not \(0\). Thus a closed base loop can lift to a path joining different points in the same fiber, and the endpoint records monodromy.
\end{example}
% END VOL04-EXPANSION IV23-example-01
\section{Sheets}
Each component over an evenly covered neighborhood projects homeomorphically to the neighborhood.

\section{Path lifting}
A path in the base has a unique lift once its starting point in the fiber is chosen.

\section{Loop lifting}
A loop may lift to a path whose endpoint is a different point of the same fiber.

% BEGIN VOL04-EXPANSION IV23-example-02
\begin{example}[Cyclic monodromy of a power map]\label{ex:iv23-expand-02}
For \(p(w)=w^n\) on \(\mathbb C^*\), the fiber over \(1\) consists of \(e^{2\pi ik/n}\). A positive loop once around zero lifts from the root \(e^{2\pi ik/n}\) to \(e^{2\pi i(k+1)/n}\). Hence the monodromy permutation is the \(n\)-cycle
\[
(0\ 1\ \cdots\ n-1).
\]
After \(n\) turns the lifted path closes.
\end{example}
% END VOL04-EXPANSION IV23-example-02
\section{Monodromy action}
Homotopy classes of loops act by permutations on the fiber.

\section{Deck transformations}
Automorphisms of the covering commute with the projection and permute sheets globally.

\section{Exponential covering}
$z\mapsto e^z$ covers $\mathbb C^*$ with deck translations by $2\pi i\mathbb Z$.

% BEGIN VOL04-EXPANSION IV23-example-03
\begin{example}[Deck transformations of the exponential covering]\label{ex:iv23-expand-03}
Every translation
\[
T_k(w)=w+2\pi i k
\]
satisfies \(e^{T_k(w)}=e^w\), so it is a deck transformation. Conversely, a deck transformation must send \(0\) to another point of the fiber over \(1\), hence to \(2\pi i k\); uniqueness of lifts then forces it to equal \(T_k\) everywhere. The deck group is therefore isomorphic to \(\mathbb Z\).
\end{example}
% END VOL04-EXPANSION IV23-example-03
\section{Power maps}
$z\mapsto z^n$ covers $\mathbb C^*$ by itself with cyclic monodromy.

\section{Analytic continuation}
Branches of inverse functions move along lifted paths, making covering theory a natural language for continuation without branch points.

\section{Core structural results}

\begin{theorem}[Unique path lifting]\label{thm:iv23-01}
For a covering $p:X\to Y$, any path $\gamma$ in $Y$ and any $x_0\in p^{-1}(\gamma(0))$ have a unique lift starting at $x_0$.
\begin{proof}
Cover the compact parameter interval by subintervals mapping into evenly covered neighborhoods and construct the lift step by step. Local uniqueness on each sheet forces global uniqueness.
\end{proof}
\end{theorem}

\begin{theorem}[Homotopy invariance of monodromy]\label{thm:iv23-02}
Homotopic loops with fixed base point induce the same permutation of the fiber.
\begin{proof}
Use the homotopy lifting property for coverings. The lifted endpoints vary continuously in a discrete fiber, hence remain constant through the homotopy.
\end{proof}
\end{theorem}

\begin{theorem}[Exponential covering]\label{thm:iv23-03}
The exponential map $\exp:\mathbb C\to\mathbb C^*$ is a covering map with deck group $z\mapsto z+2\pi i k$.
\begin{proof}
Small disks in $\mathbb C^*$ avoiding zero admit logarithm branches; their inverse images are disjoint translates of one logarithm branch, and exponential is biholomorphic on each translate.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Exponential lift]\label{exr:iv23-expand-01}
Lift \(\gamma(t)=e^{4\pi it}\) under \(e^w\) starting at zero.

Take a continuous logarithm along the path.
\end{hint}

The lift is \(\widetilde\gamma(t)=4\pi it\), ending at \(4\pi i\).

The fiber is \(\{1,i,-1,-i\}\).
\end{solution}

It ends at \(i\).

\subsection*{Proofs}

\begin{exercise}[Unique path lifting]\label{exr:iv23-expand-06}
Outline a proof of unique path lifting for coverings.
\end{exercise}
\begin{hint}
Cover the compact path image by evenly covered neighborhoods.
\end{hint}
\begin{solution}
Subdivide the interval so each segment lies in one evenly covered neighborhood. The starting sheet determines a unique local lift, and the pieces concatenate uniquely.
\end{solution}

\begin{exercise}[Homotopy invariance]\label{exr:iv23-expand-07}
Why do homotopic based loops induce the same monodromy permutation?
\end{exercise}
\begin{hint}
Lift the homotopy and inspect endpoints.
\end{hint}
\begin{solution}
Lifted endpoints depend continuously on the homotopy parameter but lie in a discrete fiber, so they remain constant.
\end{solution}

\begin{exercise}[Exponential covering]\label{exr:iv23-expand-08}
Prove that \(e^w:\mathbb C\to\mathbb C^*\) is a covering map.
\end{exercise}
\begin{hint}
Use local logarithm branches.
\end{hint}
\begin{solution}
A small disk avoiding zero admits a logarithm. Its inverse image is a disjoint union of translates of that logarithm branch by \(2\pi i k\), each mapped biholomorphically onto the disk.
\end{solution}

\begin{exercise}[Deck group]\label{exr:iv23-expand-09}
Prove that every deck transformation of the exponential covering is a translation by \(2\pi i k\).
\end{exercise}
\begin{hint}
Determine the image of one point, then use uniqueness of lifts.
\end{hint}
\begin{solution}
The image of zero lies in its fiber, so it is \(2\pi i k\). The deck map and that translation are lifts of the same exponential map with the same starting value, hence coincide.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Branch point covering]\label{exr:iv23-expand-10}
Is \(w\mapsto w^2\) a covering map from all of \(\mathbb C\) to all of \(\mathbb C\)?
\end{exercise}
\begin{hint}
Inspect a neighborhood of zero.
\end{hint}
\begin{solution}
No. The map is not locally a disjoint union of homeomorphic sheets over zero because the two sheets merge there.
\end{solution}

\begin{exercise}[Endpoint depends only on loop]\label{exr:iv23-expand-11}
Does the endpoint of a lifted loop depend only on the base loop and the chosen starting point?
\end{exercise}
\begin{hint}
Use uniqueness of lifts.
\end{hint}
\begin{solution}
Yes for the specified parametrized loop. For homotopy classes, homotopy lifting shows it depends only on the based homotopy class.
\end{solution}

\begin{exercise}[Trivial monodromy]\label{exr:iv23-expand-12}
Must every nontrivial covering have nontrivial monodromy for every loop?
\end{exercise}
\begin{hint}
Consider a contractible loop.
\end{hint}
\begin{solution}
No. Contractible loops act trivially even in a nontrivial covering.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Analytic continuation]\label{exr:iv23-expand-13}
How does path lifting model analytic continuation of inverse branches?
\end{exercise}
\begin{hint}
A local inverse chooses one point in a fiber.
\end{hint}
\begin{solution}
As the base point moves, the chosen inverse value follows the unique lifted path. Loop endpoints encode the permutation of branches.
\end{solution}

\begin{exercise}[Root tracking]\label{exr:iv23-expand-14}
Why is monodromy useful when numerically continuing roots of a polynomial depending on a parameter?
\end{exercise}
\begin{hint}
Follow roots along parameter loops.
\end{hint}
\begin{solution}
Continuation transports each root continuously and the endpoint permutation reveals how branches exchange around discriminant values.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Monodromy representation]\label{exr:iv23-expand-15}
Explain why monodromy gives a group homomorphism from the fundamental group into a permutation group.
\end{exercise}
\begin{hint}
Compare lifting a concatenation with successive lifts.
\end{hint}
\begin{solution}
Lifting concatenated loops applies the first endpoint permutation and then the second, so loop multiplication corresponds to composition of permutations.
\end{solution}

\begin{exercise}[Regular covering]\label{exr:iv23-expand-16}
What special feature of the exponential covering makes its deck group act transitively on each fiber?
\end{exercise}
\begin{hint}
Any two fiber points differ by a period.
\end{hint}
\begin{solution}
Fiber points differ by \(2\pi i k\), and the corresponding deck translation sends one to the other. This is the hallmark of a regular covering.
\end{solution}
