\chapter{Covering Maps and Monodromy}
\label{ch:iv23}

Covering maps formalize the idea of many sheets lying over one base domain. Path lifting and monodromy translate topology of loops into permutations of analytic branches.

\section*{Learning goals}
The reader should be able to state the definitions precisely, prove the core results, choose valid contours or local expansions, and diagnose which domain and regularity hypotheses are essential.

\section*{Conceptual roadmap}
\[
\boxed{\text{local holomorphic structure}\;\longrightarrow\;\text{integral or series representation}\;\longrightarrow\;\text{global consequence}.}

\section{Covering definition}
A covering map is locally a disjoint union of homeomorphic sheets over evenly covered neighborhoods.

\section{Sheets}
Each component over an evenly covered neighborhood projects homeomorphically to the neighborhood.

\section{Path lifting}
A path in the base has a unique lift once its starting point in the fiber is chosen.

\section{Loop lifting}
A loop may lift to a path whose endpoint is a different point of the same fiber.

\section{Monodromy action}
Homotopy classes of loops act by permutations on the fiber.

\section{Deck transformations}
Automorphisms of the covering commute with the projection and permute sheets globally.

\section{Exponential covering}
$z\mapsto e^z$ covers $\mathbb C^*$ with deck translations by $2\pi i\mathbb Z$.

\section{Power maps}
$z\mapsto z^n$ covers $\mathbb C^*$ by itself with cyclic monodromy.

\section{Analytic continuation}
Branches of inverse functions move along lifted paths, making covering theory a natural language for continuation without branch points.

\section{Core structural results}

\begin{theorem}[Unique path lifting]\label{thm:iv23-01}
For a covering $p:X\to Y$, any path $\gamma$ in $Y$ and any $x_0\in p^{-1}(\gamma(0))$ have a unique lift starting at $x_0$.
\begin{proof}
Cover the compact parameter interval by subintervals mapping into evenly covered neighborhoods and construct the lift step by step. Local uniqueness on each sheet forces global uniqueness.
\end{proof}
\end{theorem}

\begin{theorem}[Homotopy invariance of monodromy]\label{thm:iv23-02}
Homotopic loops with fixed base point induce the same permutation of the fiber.
\begin{proof}
Use the homotopy lifting property for coverings. The lifted endpoints vary continuously in a discrete fiber, hence remain constant through the homotopy.
\end{proof}
\end{theorem}

\begin{theorem}[Exponential covering]\label{thm:iv23-03}
The exponential map $\exp:\mathbb C\to\mathbb C^*$ is a covering map with deck group $z\mapsto z+2\pi i k$.
\begin{proof}
Small disks in $\mathbb C^*$ avoiding zero admit logarithm branches; their inverse images are disjoint translates of one logarithm branch, and exponential is biholomorphic on each translate.
\end{proof}
\end{theorem}
