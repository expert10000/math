\chapter{From Analytic Continuation to Riemann Surfaces}
\label{ch:iv22}

Analytic continuation naturally leads beyond planar single-valued functions. A Riemann surface is the geometric object on which a locally holomorphic multivalued expression becomes an ordinary single-valued holomorphic function.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item explain how multivalued analytic continuation motivates passing from the plane to a Riemann surface;
\item construct local coordinate charts that make a multivalued relation single-valued;
\item identify branch points and sheets in elementary algebraic examples;
\item translate pathwise analytic continuation into geometric movement on a surface;
\item distinguish a branched covering from an ordinary covering away from branch points;
\item relate local analytic data to the global surface obtained by continuation;
\end{itemize}

\boxed{\text{local holomorphic structure}\;\longrightarrow\;\text{integral or series representation}\;\longrightarrow\;\text{global consequence}.}

\section{Local branches}
Square roots, logarithms, and inverse functions have locally single-valued holomorphic branches.

% BEGIN VOL04-EXPANSION IV22-example-01
\begin{example}[Two local square-root branches]\label{ex:iv22-expand-01}
On a disk avoiding the origin, choose a holomorphic logarithm and set
\[
\sqrt z=\exp\!\left(\frac12\operatorname{Log}z\right).
\]
The second branch is its negative. Continuing the first branch once around the origin changes \(\operatorname{Log}z\) by \(2\pi i\), hence multiplies the square root by \(e^{\pi i}=-1\). The endpoint has the same base coordinate but the opposite germ, which is precisely why the surface of germs contains two points over a generic base point.
\end{example}

\begin{example}[The logarithm surface as infinitely many sheets]\label{ex:iv22-expand-02}
A germ of \(\log z\) near \(1\) has values differing by \(2\pi i k\). Continuing once counterclockwise around zero sends the value \(0\) at \(1\) to \(2\pi i\). After \(m\) turns it becomes \(2\pi i m\). Thus the maximal continuation has infinitely many sheets, naturally indexed by the integer winding number.
\end{example}

\begin{example}[A coordinate near a branch point]\label{ex:iv22-expand-03}
For the algebraic relation \(w^2=z\), the projection to the \(z\)-plane is not a local coordinate at \((0,0)\). Instead use \(w\) itself as the surface coordinate. Then the projection is
\[
z=w^2.
\]
The surface is perfectly smooth at the point even though the projection ramifies there. This separates intrinsic smoothness of the Riemann surface from singular behavior of a chosen projection.
\end{example}

\section{Core structural results}

\begin{theorem}[Local single-valuedness]\label{thm:iv22-01}
Every point of a Riemann surface has a neighborhood on which a holomorphic projection that is locally biholomorphic can be used as a complex coordinate.
\begin{proof}
This is built into the chart structure: a local coordinate is a homeomorphism to a plane domain with holomorphic transition maps.
\end{proof}
\end{theorem}

\begin{theorem}[Germ-surface principle]\label{thm:iv22-02}
Compatible analytic continuations of a germ can be organized into a set of germs with a natural projection to the base point and a local complex coordinate given by that projection.
\begin{proof}
Around each germ choose a representative on a small disk. Nearby points correspond to the germs of the same representative; these neighborhoods provide charts and compatible transition maps.
\end{proof}
\end{theorem}

\begin{theorem}[Single-valued lift]\label{thm:iv22-03}
A multivalued analytic expression becomes single-valued when evaluated on its surface of analytic germs.
\begin{proof}
A point of the surface records not just the base coordinate but the chosen local germ, so evaluation of that germ is unambiguous.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Square-root monodromy]\label{exr:iv22-expand-01}
What happens to a chosen branch of \(\sqrt z\) after one loop around zero?
\end{exercise}
\begin{hint}
Track the logarithm change by \(2\pi i\).
\end{hint}
\begin{solution}
The square root changes sign.
\end{solution}

\begin{exercise}[Logarithm monodromy]\label{exr:iv22-expand-02}
What happens to \(\log z\) after one positive loop around zero?
\end{exercise}
\begin{hint}
Use the argument increment.
\end{hint}
\begin{solution}
Its value increases by \(2\pi i\).
\end{solution}

\begin{exercise}[Fiber size]\label{exr:iv22-expand-03}
How many germs of \(\sqrt z\) lie over a generic nonzero point?
\end{exercise}
\begin{hint}
Solve \(w^2=z\).
\end{hint}
\begin{solution}
There are two, corresponding to the two square roots.
\end{solution}

\begin{exercise}[Local coordinate]\label{exr:iv22-expand-04}
For \(w^2=z\) near the branch point, which variable is a good surface coordinate?
\end{exercise}
\begin{hint}
Choose the variable in which the relation has no branching.
\end{hint}
\begin{solution}
The coordinate \(w\) is good; the projection is then \(w\mapsto w^2\).
\end{solution}

\begin{exercise}[Single-valued lift]\label{exr:iv22-expand-05}
Why is the square root single valued on its two-sheeted surface?
\end{exercise}
\begin{hint}
A surface point remembers the branch germ.
\end{hint}
\begin{solution}
Evaluating the recorded germ gives one unambiguous value at each surface point.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Germ charts]\label{exr:iv22-expand-06}
Explain how representatives of analytic germs define local charts on the surface of germs.
\end{exercise}
\begin{hint}
Use one representative on a small disk.
\end{hint}
\begin{solution}
Nearby points are assigned the germs of the same holomorphic representative. Projection to the disk is one to one on that neighborhood and gives the chart.
\end{solution}

\begin{exercise}[Holomorphic transitions]\label{exr:iv22-expand-07}
Prove that transition maps between overlapping germ charts are holomorphic.
\end{exercise}
\begin{hint}
Both charts use the same base coordinate where the representatives agree.
\end{hint}
\begin{solution}
On overlap the projection coordinates agree, so the transition is the identity on an open plane set and hence holomorphic.
\end{solution}

\begin{exercise}[Maximal continuation connectedness]\label{exr:iv22-expand-08}
Why is the surface generated from one germ by analytic continuation naturally connected?
\end{exercise}
\begin{hint}
Every point is reached along a continuation path from the initial germ.
\end{hint}
\begin{solution}
The lifted continuation path lies in the germ surface and connects the initial point to the target point, so all generated points lie in one connected component.
\end{solution}

\begin{exercise}[Intrinsic holomorphy]\label{exr:iv22-expand-09}
Show that the definition of a holomorphic function on a Riemann surface is independent of the chosen chart.
\end{exercise}
\begin{hint}
Insert a holomorphic transition map.
\end{hint}
\begin{solution}
If the coordinate expression is holomorphic in one chart, composing with a biholomorphic transition map gives a holomorphic expression in every compatible chart.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Plane only]\label{exr:iv22-expand-10}
Can a globally single-valued square root exist on \(\mathbb C^*\)?
\end{exercise}
\begin{hint}
Continue around the unit circle.
\end{hint}
\begin{solution}
No. One loop changes the sign, contradicting single valuedness.
\end{solution}

\begin{exercise}[Branch point as surface singularity]\label{exr:iv22-expand-11}
Is the point over \(z=0\) on \(w^2=z\) singular as a Riemann-surface point?
\end{exercise}
\begin{hint}
Use \(w\) as a coordinate.
\end{hint}
\begin{solution}
No. The surface is smooth there; only the projection to the \(z\)-plane is ramified.
\end{solution}

\begin{exercise}[One sheet for logarithm]\label{exr:iv22-expand-12}
Can the maximal logarithm surface have finitely many sheets?
\end{exercise}
\begin{hint}
Count values after repeated loops.
\end{hint}
\begin{solution}
No. Each winding adds a distinct multiple of \(2\pi i\), giving infinitely many sheets.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Inverse functions]\label{exr:iv22-expand-13}
Why do Riemann surfaces help with multivalued inverse functions?
\end{exercise}
\begin{hint}
Each local inverse branch is a germ.
\end{hint}
\begin{solution}
The surface separates different inverse germs lying over the same base point, turning the multivalued inverse into an ordinary holomorphic function upstairs.
\end{solution}

\begin{exercise}[Algebraic functions]\label{exr:iv22-expand-14}
How does the surface viewpoint resolve the ambiguity of an algebraic relation such as \(w^3=z(z-1)\)?
\end{exercise}
\begin{hint}
Treat each compatible local solution as a sheet.
\end{hint}
\begin{solution}
Local roots glue into a surface on which \(w\) is single valued; branching is encoded in the projection to the \(z\)-sphere.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Continuation obstruction]\label{exr:iv22-expand-15}
What topological feature of the base detects whether a local logarithm can be continued single valued globally?
\end{exercise}
\begin{hint}
Look at loops around the missing origin.
\end{hint}
\begin{solution}
Nontrivial winding around zero creates nonzero logarithmic monodromy. A simply connected domain avoiding zero removes that obstruction.
\end{solution}

\begin{exercise}[Surface versus branch cut]\label{exr:iv22-expand-16}
Compare using a branch cut with using the full Riemann surface for \(\log z\).
\end{exercise}
\begin{hint}
A branch cut discards paths, while the surface retains them.
\end{hint}
\begin{solution}
A cut selects one sheet on a simply connected domain and sacrifices continuity across the cut. The Riemann surface keeps all continuations simultaneously and records their sheet changes geometrically.
\end{solution}
