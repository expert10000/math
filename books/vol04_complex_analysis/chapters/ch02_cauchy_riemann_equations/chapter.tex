\chapter{Cauchy–Riemann Equations}
\label{ch:iv02}

The Cauchy–Riemann equations translate complex differentiability into real partial-derivative relations. They reveal the geometric link between holomorphic maps, conformality, harmonic functions, and complex linearity.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item derive the Cauchy--Riemann equations from complex differentiability;
\item check the Cauchy--Riemann equations for explicit functions and combine them with regularity hypotheses correctly;
\item reconstruct harmonic conjugates from compatible partial-derivative data;
\item distinguish necessary Cauchy--Riemann conditions from sufficient conditions for holomorphicity;
\item express the derivative in terms of real partial derivatives once holomorphicity is established;
\item construct examples where the Cauchy--Riemann equations hold at a point but complex differentiability still fails;
\end{itemize}

\section{Cartesian form}
For $f=u+iv$, complex differentiability suggests $u_x=v_y$ and $u_y=-v_x$.

% BEGIN VOL04-EXPANSION IV02-example-01
\begin{example}[Cauchy--Riemann check for a cubic]\label{ex:iv02-expand-01}
Write \(z=x+iy\). For \(f(z)=z^3\),
\[
u=x^3-3xy^2,\qquad v=3x^2y-y^3.
\]
Then \(u_x=3x^2-3y^2=v_y\) and \(u_y=-6xy=-v_x\). The partial derivatives are polynomials and therefore continuous, so the sufficiency theorem applies everywhere. Moreover \(f'(z)=u_x+iv_x=3z^2\), agreeing with direct complex differentiation.
\end{example}

\section{Harmonic components}

\begin{example}[Recovering a harmonic conjugate]\label{ex:iv02-expand-02}
Let \(u(x,y)=x^2-y^2\). A conjugate \(v\) must satisfy
\[
v_y=u_x=2x,\qquad v_x=-u_y=2y.
\]
Integrating the first equation gives \(v=2xy+g(x)\). Differentiating in \(x\) and comparing with the second equation yields \(g'(x)=0\). Hence \(v=2xy+C\), and \(u+iv=z^2+iC\).
\end{example}
% END VOL04-EXPANSION IV02-example-02
\section{Harmonic conjugates}
Locally on simply connected regions, harmonic functions admit harmonic conjugates under the appropriate integrability condition.

\section{Jacobian geometry}
The real Jacobian of a holomorphic map with nonzero derivative is a rotation-dilation matrix.

\section{Failure examples}
Functions such as $\bar z$ and $|z|^2$ expose why partial derivatives alone do not guarantee holomorphicity.

\begin{example}[Pointwise equations without differentiability]\label{ex:iv02-expand-03}
Define \(f(0)=0\) and, for \(z=x+iy\ne0\),
\[
f(z)=\frac{x^3}{x^2+y^2}+i\frac{y^3}{x^2+y^2}.
\]
The coordinate partial derivatives at the origin exist and give \(u_x(0)=v_y(0)=1\) and \(u_y(0)=-v_x(0)=0\), so the Cauchy--Riemann equations hold at that point. Along \(y=x\), however, \(f(x+ix)/(x+ix)=1/2\), whereas along the real axis the quotient is \(1\). Thus the equations at one point, without a suitable differentiability hypothesis, are not sufficient.
\end{example}

\section{Core structural results}

\begin{theorem}[Cauchy--Riemann necessity]\label{thm:iv02-01}
If $f=u+iv$ is complex differentiable at $z_0$, then $u_x=v_y$ and $u_y=-v_x$ at $z_0$.
\begin{proof}
Evaluate the difference quotient along real increments and along purely imaginary increments and equate the two complex limits.
\end{proof}
\end{theorem}

\begin{theorem}[Sufficiency under $C^1$ hypotheses]\label{thm:iv02-02}
If $u,v$ have continuous first partials near $z_0$ satisfying the Cauchy--Riemann equations, then $f=u+iv$ is complex differentiable there.
\begin{proof}
Use the real differentiability expansion and the equations to identify the real derivative matrix with complex multiplication by $u_x+iv_x$.
\end{proof}
\end{theorem}

\begin{theorem}[Harmonicity]\label{thm:iv02-03}
If $f=u+iv$ is holomorphic with continuous second partials, then $\Delta u=\Delta v=0$.
\begin{proof}
Differentiate the Cauchy--Riemann equations and use equality of mixed partials to obtain cancellation in each Laplacian.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Quadratic CR check]\label{exr:iv02-expand-01}
Verify the Cauchy--Riemann equations for \(f(z)=z^2\).
\end{exercise}
\begin{hint}
Expand \((x+iy)^2\).
\end{hint}
\begin{solution}
Here \(u=x^2-y^2\) and \(v=2xy\). Thus \(u_x=2x=v_y\) and \(u_y=-2y=-v_x\).
\end{solution}

\begin{exercise}[Derivative from partials]\label{exr:iv02-expand-02}
For \(f(z)=z^3\), compute \(f'\) from \(u_x+iv_x\).
\end{exercise}
\begin{hint}
Use the real and imaginary parts of the cubic.
\end{hint}
\begin{solution}
The expression is \(3x^2-3y^2+i6xy=3(x+iy)^2=3z^2\).
\end{solution}

\begin{exercise}[Harmonic test]\label{exr:iv02-expand-03}
Check that \(u(x,y)=e^x\cos y\) is harmonic.
\end{exercise}
\begin{hint}
Compute the two second partial derivatives.
\end{hint}
\begin{solution}
One finds \(u_{xx}=e^x\cos y\) and \(u_{yy}=-e^x\cos y\), so \(\Delta u=0\).
\end{solution}

\begin{exercise}[Find a conjugate]\label{exr:iv02-expand-04}
Find a harmonic conjugate of \(u=e^x\cos y\).
\end{exercise}
\begin{hint}
Use \(v_y=u_x\) and then check \(v_x=-u_y\).
\end{hint}
\begin{solution}
Integrating \(v_y=e^x\cos y\) gives \(v=e^x\sin y+C\), which satisfies the other equation.
\end{solution}

\begin{exercise}[Jacobian determinant]\label{exr:iv02-expand-05}
If \(f'(z_0)=a+ib\), compute the determinant of the real Jacobian at \(z_0\).
\end{exercise}
\begin{hint}
The matrix is \(\begin{pmatrix}a&-b\\b&a\end{pmatrix}\).
\end{hint}
\begin{solution}
Its determinant is \(a^2+b^2=|f'(z_0)|^2\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Necessity of CR]\label{exr:iv02-expand-06}
Derive the Cauchy--Riemann equations from the complex difference quotient.
\end{exercise}
\begin{hint}
Use real increments and then imaginary increments.
\end{hint}
\begin{solution}
The real increment limit is \(u_x+iv_x\). The imaginary increment limit is \(v_y-iu_y\). Equality yields \(u_x=v_y\) and \(u_y=-v_x\).
\end{solution}

\begin{exercise}[Harmonic real part]\label{exr:iv02-expand-07}
Assuming continuous second partials, prove that the real part of a holomorphic function is harmonic.
\end{exercise}
\begin{hint}
Differentiate the two CR equations once more.
\end{hint}
\begin{solution}
From \(u_x=v_y\) obtain \(u_{xx}=v_{yx}\), and from \(u_y=-v_x\) obtain \(u_{yy}=-v_{xy}\). Equality of mixed partials gives \(u_{xx}+u_{yy}=0\).
\end{solution}

\begin{exercise}[Harmonic imaginary part]\label{exr:iv02-expand-08}
Prove the analogous harmonicity statement for the imaginary part.
\end{exercise}
\begin{hint}
Differentiate in the opposite order.
\end{hint}
\begin{solution}
The differentiated CR equations give \(v_{xx}=-u_{yx}\) and \(v_{yy}=u_{xy}\), whose sum is zero.
\end{solution}

\begin{exercise}[Local angle scaling]\label{exr:iv02-expand-09}
Show that when \(f'(z_0)\ne0\), the real derivative preserves oriented angles.
\end{exercise}
\begin{hint}
Write the Jacobian as multiplication by the complex number \(f'(z_0)\).
\end{hint}
\begin{solution}
Multiplication by a nonzero complex number is a positive dilation followed by a rotation. It therefore changes every tangent direction by the same angle and preserves oriented angle differences.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Conjugation]\label{exr:iv02-expand-10}
Test the CR equations for \(f(z)=\overline z\).
\end{exercise}
\begin{hint}
Use \(u=x\) and \(v=-y\).
\end{hint}
\begin{solution}
One has \(u_x=1\) and \(v_y=-1\), so the first equation fails everywhere.
\end{solution}

\begin{exercise}[Real valued holomorphic function]\label{exr:iv02-expand-11}
Can a nonconstant real valued function be holomorphic on a connected open set?
\end{exercise}
\begin{hint}
Set the imaginary part equal to zero in the CR equations.
\end{hint}
\begin{solution}
No. The equations force both first partial derivatives of the real part to vanish, so the function is locally constant and hence constant on each connected component.
\end{solution}

\begin{exercise}[Pointwise CR]\label{exr:iv02-expand-12}
Why is satisfying the CR equations at one point not by itself sufficient for complex differentiability?
\end{exercise}
\begin{hint}
The equations concern only coordinate partial derivatives.
\end{hint}
\begin{solution}
Complex differentiability requires a two dimensional linear approximation. Without real differentiability or suitable continuity of partials, coordinate partial information may miss oblique approaches.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Potential and stream function]\label{exr:iv02-expand-13}
If \(u\) is a harmonic potential and \(v\) is its harmonic conjugate, what geometric relation holds between their gradients?
\end{exercise}
\begin{hint}
Use the CR equations.
\end{hint}
\begin{solution}
The equations give \(\nabla v=(-u_y,u_x)\), a quarter turn of \(\nabla u\). Their level curves are therefore orthogonal where the gradients are nonzero.
\end{solution}

\begin{exercise}[Local area scaling]\label{exr:iv02-expand-14}
How does a holomorphic map with derivative \(f'(z_0)\ne0\) scale infinitesimal area?
\end{exercise}
\begin{hint}
Use the Jacobian determinant.
\end{hint}
\begin{solution}
The local area factor is \(|f'(z_0)|^2\), the determinant of the real derivative matrix.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Polar CR equations]\label{exr:iv02-expand-15}
Derive the polar Cauchy--Riemann equations for \(f=u+iv\) away from the origin.
\end{exercise}
\begin{hint}
Express \(u_x,u_y,v_x,v_y\) through radial and angular derivatives.
\end{hint}
\begin{solution}
Substituting the polar chain rule into the Cartesian equations yields \(u_r=v_\theta/r\) and \(v_r=-u_\theta/r\).
\end{solution}

\begin{exercise}[Conjugate obstruction]\label{exr:iv02-expand-16}
Explain why a harmonic function on a punctured plane can fail to have a single valued harmonic conjugate.
\end{exercise}
\begin{hint}
Consider \(u(z)=\log|z|\).
\end{hint}
\begin{solution}
Locally its conjugate is an argument function. Going once around the origin changes the argument by \(2\pi\), so no globally single valued conjugate exists on the punctured plane.
\end{solution}
