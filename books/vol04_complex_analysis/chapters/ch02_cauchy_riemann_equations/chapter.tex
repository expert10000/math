\chapter{Cauchy–Riemann Equations}
\label{ch:iv02}

The Cauchy–Riemann equations translate complex differentiability into real partial-derivative relations. They reveal the geometric link between holomorphic maps, conformality, harmonic functions, and complex linearity.

\section{Sufficiency with regularity}
If first partials exist continuously near a point and satisfy Cauchy–Riemann there, then the function is holomorphic nearby.

\section{Harmonic components}
Twice differentiable holomorphic functions have harmonic real and imaginary parts.

\section{Harmonic conjugates}
Locally on simply connected regions, harmonic functions admit harmonic conjugates under the appropriate integrability condition.

\section{Jacobian geometry}
The real Jacobian of a holomorphic map with nonzero derivative is a rotation-dilation matrix.

\section{Failure examples}
Functions such as $\bar z$ and $|z|^2$ expose why partial derivatives alone do not guarantee holomorphicity.

\section{Core structural results}

\begin{theorem}[Cauchy--Riemann necessity]\label{thm:iv02-01}
If $f=u+iv$ is complex differentiable at $z_0$, then $u_x=v_y$ and $u_y=-v_x$ at $z_0$.
\begin{proof}
Evaluate the difference quotient along real increments and along purely imaginary increments and equate the two complex limits.
\end{proof}
\end{theorem}

\begin{theorem}[Sufficiency under $C^1$ hypotheses]\label{thm:iv02-02}
If $u,v$ have continuous first partials near $z_0$ satisfying the Cauchy--Riemann equations, then $f=u+iv$ is complex differentiable there.
\begin{proof}
Use the real differentiability expansion and the equations to identify the real derivative matrix with complex multiplication by $u_x+iv_x$.
\end{proof}
\end{theorem}

\begin{theorem}[Harmonicity]\label{thm:iv02-03}
If $f=u+iv$ is holomorphic with continuous second partials, then $\Delta u=\Delta v=0$.
\begin{proof}
Differentiate the Cauchy--Riemann equations and use equality of mixed partials to obtain cancellation in each Laplacian.
\end{proof}
\end{theorem}
