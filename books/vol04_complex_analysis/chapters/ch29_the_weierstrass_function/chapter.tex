\chapter{The Weierstrass $\wp$-Function}
\label{ch:iv29}

The Weierstrass $\wp$-function is the canonical elliptic function attached to a lattice. Its carefully regularized lattice sum gives a doubly periodic even function with one double pole per period cell and an algebraic differential equation.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct the Weierstrass \(\wp\)-function from its lattice series and justify convergence on compacta away from the lattice;
\item derive evenness, double periodicity, and pole structure of \(\wp\);
\item compute the differential equation satisfied by \(\wp\) and identify the invariants \(g_2,g_3\);
\item use \(\wp'\) and local expansions to analyze zeros and ramification;
\item relate lattice symmetries to special values of the Weierstrass function;
\item distinguish the analytic definition of \(\wp\) from the algebraic cubic relation it later parametrizes;
\end{itemize}

\section{Regularized lattice sum}
Define $\wp(z)=z^{-2}+\sum_{\omega\in\Lambda\setminus\{0\}}[(z-\omega)^{-2}-\omega^{-2}]$.

% BEGIN VOL04-EXPANSION IV29-example-01
\begin{example}[Why the Weierstrass sum converges]\label{ex:iv29-expand-01}
For bounded \(z\) and large lattice points \(\omega\),
\[
\frac1{(z-\omega)^2}-\frac1{\omega^2}
=\frac{2z\omega-z^2}{\omega^2(z-\omega)^2}
=O(|\omega|^{-3}).
\]
The lattice sum \(\sum_{\omega\ne0}|\omega|^{-3}\) converges. Therefore the regularized series for \(\wp(z)\) converges normally on compact sets avoiding \(\Lambda\). The subtraction term is exactly what improves the decay from quadratic to cubic order.
\end{example}
% END VOL04-EXPANSION IV29-example-01
\section{Normal convergence}
The subtraction of $\omega^{-2}$ improves convergence so the series converges normally on compact sets avoiding the lattice.

\section{Evenness}
Pairing lattice points $\omega$ and $-\omega$ shows $\wp(-z)=\wp(z)$.

\section{Double periodicity}
The derivative $\wp'$ is manifestly lattice periodic, and comparison constants show $\wp$ itself is elliptic.

% BEGIN VOL04-EXPANSION IV29-example-02
\begin{example}[A half-period critical point]\label{ex:iv29-expand-02}
Let \(a\) be a nonzero half-period, so \(2a\in\Lambda\). Periodicity and oddness of \(\wp'\) give
\[
\wp'(a)=\wp'(a-2a)=\wp'(-a)=-\wp'(a).
\]
Hence \(\wp'(a)=0\). The three nonzero half-period classes therefore map to the three finite branch values of the degree-two map \(\wp:\mathbb C/\Lambda\to\widehat{\mathbb C}\).
\end{example}
% END VOL04-EXPANSION IV29-example-02
\section{Pole structure}
Modulo the lattice, $\wp$ has one double pole at the origin with zero residue.

\section{Derivative}
The derivative $\wp'$ is odd and has a triple pole at the lattice points.

\section{Half-period values}
At nonzero half-periods, $\wp'$ vanishes; their $\wp$ values are the roots $e_1,e_2,e_3$ of the cubic relation.

% BEGIN VOL04-EXPANSION IV29-example-03
\begin{example}[Recovering the cubic differential equation]\label{ex:iv29-expand-03}
Near zero,
\[
\wp(z)=z^{-2}+O(z^2),\qquad \wp'(z)=-2z^{-3}+O(z).
\]
Choose lattice invariants \(g_2,g_3\) so that the principal part of
\[
F=(\wp')^2-4\wp^3+g_2\wp+g_3
\]
cancels. Then \(F\) is elliptic and has no poles, hence is constant on the compact torus. Matching the constant term gives \(F=0\), so
\[
(\wp')^2=4\wp^3-g_2\wp-g_3.
\]
\end{example}
% END VOL04-EXPANSION IV29-example-03
\section{Differential equation}
There are lattice invariants $g_2,g_3$ with $(\wp')^2=4\wp^3-g_2\wp-g_3$.

\section{Second derivative relation}
Differentiating the cubic gives $\wp''=6\wp^2-g_2/2$.

\section{Uniformization preview}
The map $z\mapsto(\wp(z),\wp'(z))$ parametrizes a nonsingular cubic curve when the discriminant is nonzero.

\section{Core structural results}

\begin{theorem}[Normal convergence of the Weierstrass series]\label{thm:iv29-01}
The defining series for $\wp$ converges normally on compact subsets of $\mathbb C\setminus\Lambda$.
\begin{proof}
For bounded $z$ and large $|\omega|$, expand $(z-\omega)^{-2}-\omega^{-2}=O(|z|/|\omega|^3)$. The lattice sum $\sum_{\omega\ne0}|\omega|^{-3}$ converges, so the Weierstrass M-test applies away from lattice points.
\end{proof}
\end{theorem}

\begin{theorem}[Elliptic differential equation]\label{thm:iv29-02}
There exist constants $g_2=60\sum_{\omega\ne0}\omega^{-4}$ and $g_3=140\sum_{\omega\ne0}\omega^{-6}$ such that $(\wp')^2=4\wp^3-g_2\wp-g_3$.
\begin{proof}
Compare Laurent expansions at zero. The difference $(\wp')^2-4\wp^3+g_2\wp+g_3$ is elliptic and has its principal part cancelled, hence is entire on the torus and therefore constant. The chosen coefficients make the constant zero.
\end{proof}
\end{theorem}

\begin{theorem}[Half-period critical points]\label{thm:iv29-03}
If $2a\in\Lambda$ but $a\notin\Lambda$, then $\wp'(a)=0$.
\begin{proof}
Because $\wp'$ is odd and periodic, $\wp'(a)=\wp'(a-2a)=\wp'(-a)=-\wp'(a)$.
\end{proof}
\end{theorem}

\begin{theorem}[Degree-two map to the sphere]\label{thm:iv29-04}
The function $\wp$ induces a degree-two meromorphic map $\mathbb C/\Lambda\to\widehat{\mathbb C}$, branched at the three nonzero half-periods and the origin.
\begin{proof}
Generically $\wp(z)=\wp(-z)$ gives two points in a fiber. Fixed points of the involution $z\mapsto-z$ modulo the lattice are exactly the four half-period classes, where the local multiplicity rises to two.
\end{proof}
\end{theorem}

\subsection*{Proofs}

\begin{exercise}[Normal convergence]\label{exr:iv29-expand-06}
Prove normal convergence of the regularized lattice series away from the lattice.
\end{exercise}
\begin{hint}
Use the \(O(|\omega|^{-3})\) estimate.
\end{hint}
\begin{solution}
On a compact set, the estimate is uniform for large \(\omega\), and the lattice sum of inverse cubes converges, so the Weierstrass M-test applies.
\end{solution}

\begin{exercise}[Evenness]\label{exr:iv29-expand-07}
Prove \(\wp(-z)=\wp(z)\).
\end{exercise}
\begin{hint}
Pair lattice points \(\omega\) and \(-\omega\).
\end{hint}
\begin{solution}
Replacing \(z\) by \(-z\) and reindexing the lattice by \(\omega\mapsto-\omega\) leaves the regularized sum unchanged.
\end{solution}

\begin{exercise}[Half-period zeros]\label{exr:iv29-expand-08}
Prove that \(\wp'\) vanishes at all three nonzero half-period classes.
\end{exercise}
\begin{hint}
Use \(a\equiv-a\pmod\Lambda\).
\end{hint}
\begin{solution}
Periodicity gives \(\wp'(a)=\wp'(-a)\), while oddness gives \(\wp'(-a)=-\wp'(a)\), hence the value is zero.
\end{solution}

\begin{exercise}[Differential equation]\label{exr:iv29-expand-09}
Explain why canceling the principal part in the cubic expression forces a global identity.
\end{exercise}
\begin{hint}
An elliptic function with no poles is constant.
\end{hint}
\begin{solution}
After cancellation the difference descends to a holomorphic function on the compact torus, so it is constant. Matching one Laurent coefficient determines that constant as zero.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Unregularized sum]\label{exr:iv29-expand-10}
Why is \(\sum_{\omega\in\Lambda}(z-\omega)^{-2}\) not used naively as the definition of \(\wp\)?
\end{exercise}
\begin{hint}
Absolute convergence in two lattice dimensions is borderline.
\end{hint}
\begin{solution}
The inverse-square lattice sum is not absolutely convergent in the required manner. Regularization removes the problematic leading tail and yields normal convergence.
\end{solution}

\begin{exercise}[Simple pole]\label{exr:iv29-expand-11}
Could \(\wp\) have a simple pole at zero?
\end{exercise}
\begin{hint}
Use evenness.
\end{hint}
\begin{solution}
No. An even Laurent series has no odd-power term such as \(z^{-1}\); the leading pole is double.
\end{solution}

\begin{exercise}[Discriminant zero]\label{exr:iv29-expand-12}
Does the cubic \(y^2=4x^3-g_2x-g_3\) remain a smooth elliptic curve when its discriminant vanishes?
\end{exercise}
\begin{hint}
A repeated root creates a singular point.
\end{hint}
\begin{solution}
No. Vanishing discriminant makes the cubic singular, so the smooth torus uniformization requires nonzero discriminant.
\end{solution}
