\chapter{The Weierstrass $\wp$-Function}
\label{ch:iv29}

The Weierstrass $\wp$-function is the canonical elliptic function attached to a lattice. Its carefully regularized lattice sum gives a doubly periodic even function with one double pole per period cell and an algebraic differential equation.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct the Weierstrass \(\wp\)-function from its lattice series and justify convergence on compacta away from the lattice;
\item derive evenness, double periodicity, and pole structure of \(\wp\);
\item compute the differential equation satisfied by \(\wp\) and identify the invariants \(g_2,g_3\);
\item use \(\wp'\) and local expansions to analyze zeros and ramification;
\item relate lattice symmetries to special values of the Weierstrass function;
\item distinguish the analytic definition of \(\wp\) from the algebraic cubic relation it later parametrizes;
\end{itemize}

\section{Regularized lattice sum}
Define $\wp(z)=z^{-2}+\sum_{\omega\in\Lambda\setminus\{0\}}[(z-\omega)^{-2}-\omega^{-2}]$.

\section{Normal convergence}
The subtraction of $\omega^{-2}$ improves convergence so the series converges normally on compact sets avoiding the lattice.

\section{Evenness}
Pairing lattice points $\omega$ and $-\omega$ shows $\wp(-z)=\wp(z)$.

\section{Double periodicity}
The derivative $\wp'$ is manifestly lattice periodic, and comparison constants show $\wp$ itself is elliptic.

\section{Pole structure}
Modulo the lattice, $\wp$ has one double pole at the origin with zero residue.

\section{Derivative}
The derivative $\wp'$ is odd and has a triple pole at the lattice points.

\section{Half-period values}
At nonzero half-periods, $\wp'$ vanishes; their $\wp$ values are the roots $e_1,e_2,e_3$ of the cubic relation.

\section{Differential equation}
There are lattice invariants $g_2,g_3$ with $(\wp')^2=4\wp^3-g_2\wp-g_3$.

\section{Second derivative relation}
Differentiating the cubic gives $\wp''=6\wp^2-g_2/2$.

\section{Uniformization preview}
The map $z\mapsto(\wp(z),\wp'(z))$ parametrizes a nonsingular cubic curve when the discriminant is nonzero.

\section{Core structural results}

\begin{theorem}[Normal convergence of the Weierstrass series]\label{thm:iv29-01}
The defining series for $\wp$ converges normally on compact subsets of $\mathbb C\setminus\Lambda$.
\begin{proof}
For bounded $z$ and large $|\omega|$, expand $(z-\omega)^{-2}-\omega^{-2}=O(|z|/|\omega|^3)$. The lattice sum $\sum_{\omega\ne0}|\omega|^{-3}$ converges, so the Weierstrass M-test applies away from lattice points.
\end{proof}
\end{theorem}

\begin{theorem}[Elliptic differential equation]\label{thm:iv29-02}
There exist constants $g_2=60\sum_{\omega\ne0}\omega^{-4}$ and $g_3=140\sum_{\omega\ne0}\omega^{-6}$ such that $(\wp')^2=4\wp^3-g_2\wp-g_3$.
\begin{proof}
Compare Laurent expansions at zero. The difference $(\wp')^2-4\wp^3+g_2\wp+g_3$ is elliptic and has its principal part cancelled, hence is entire on the torus and therefore constant. The chosen coefficients make the constant zero.
\end{proof}
\end{theorem}

\begin{theorem}[Half-period critical points]\label{thm:iv29-03}
If $2a\in\Lambda$ but $a\notin\Lambda$, then $\wp'(a)=0$.
\begin{proof}
Because $\wp'$ is odd and periodic, $\wp'(a)=\wp'(a-2a)=\wp'(-a)=-\wp'(a)$.
\end{proof}
\end{theorem}

\begin{theorem}[Degree-two map to the sphere]\label{thm:iv29-04}
The function $\wp$ induces a degree-two meromorphic map $\mathbb C/\Lambda\to\widehat{\mathbb C}$, branched at the three nonzero half-periods and the origin.
\begin{proof}
Generically $\wp(z)=\wp(-z)$ gives two points in a fiber. Fixed points of the involution $z\mapsto-z$ modulo the lattice are exactly the four half-period classes, where the local multiplicity rises to two.
\end{proof}
\end{theorem}
