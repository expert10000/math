\chapter{Laurent Series}
\label{ch:iv08}

Laurent series extend power-series expansions to annuli and encode singular behavior through negative powers. The principal part distinguishes removable singularities, poles, and essential singularities.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct Laurent expansions on specified annuli and identify the correct convergence region;
\item compute principal and analytic parts of Laurent series;
\item derive Laurent coefficients from contour integrals;
\item choose among geometric-series expansions according to the relative sizes of \(|z-a|\);
\item use Laurent series to classify singular behavior and compute residues;
\item distinguish expansions of the same function on different annuli;
\end{itemize}

\section{Laurent expansions}
A function holomorphic on an annulus has a unique expansion $\sum_{n=-\infty}^{\infty}a_n(z-z_0)^n$.

\section{Annuli of convergence}
Different singularities determine concentric annuli on which different Laurent expansions may be valid.

\section{Coefficient formula}
Laurent coefficients are contour integrals $a_n=(2\pi i)^{-1}\int f(\zeta)(\zeta-z_0)^{-n-1}d\zeta$.

\section{Principal part}
Negative-power terms constitute the principal part and measure the singular component.

\section{Positive part}
Nonnegative powers form an ordinary Taylor series.

\section{Geometric expansions}
Rational functions are expanded by rewriting factors into geometric-series form suited to the chosen annulus.

\section{Uniqueness}
The Laurent coefficients are uniquely determined by contour integrals.

\section{Residue preview}
The coefficient $a_{-1}$ is the residue and controls contour integrals around the singularity.

\section{Annular deformation}
Coefficient integrals are independent of the chosen circle within the annulus.

\section{Core structural results}

\begin{theorem}[Laurent theorem]\label{thm:iv08-01}
A function holomorphic on an annulus $r<|z-z_0|<R$ has a unique Laurent series converging normally on compact subannuli.
\begin{proof}
Apply Cauchy's formula to a region bounded by two circles, expand the outer and inner Cauchy kernels geometrically in the appropriate ratios, and justify termwise integration by uniform convergence on compact subannuli.
\end{proof}
\end{theorem}

\begin{theorem}[Laurent coefficient formula]\label{thm:iv08-02}
For any circle $|\zeta-z_0|=\rho$ inside the annulus, $a_n=\frac1{2\pi i}\int\frac{f(\zeta)}{(\zeta-z_0)^{n+1}}d\zeta$.
\begin{proof}
Insert the normally convergent Laurent series into the integral and use orthogonality of powers around the circle; only the $n$th coefficient survives.
\end{proof}
\end{theorem}

\begin{theorem}[Uniqueness]\label{thm:iv08-03}
A Laurent expansion on a fixed annulus is unique.
\begin{proof}
The coefficient integral recovers every coefficient directly from the function, so two expansions must have identical coefficients.
\end{proof}
\end{theorem}
