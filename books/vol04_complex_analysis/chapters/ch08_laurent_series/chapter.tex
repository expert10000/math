\chapter{Laurent Series}
\label{ch:iv08}

Laurent series extend power-series expansions to annuli and encode singular behavior through negative powers. The principal part distinguishes removable singularities, poles, and essential singularities.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct Laurent expansions on specified annuli and identify the correct convergence region;
\item compute principal and analytic parts of Laurent series;
\item derive Laurent coefficients from contour integrals;
\item choose among geometric-series expansions according to the relative sizes of \(|z-a|\);
\item use Laurent series to classify singular behavior and compute residues;
\item distinguish expansions of the same function on different annuli;
\end{itemize}

\section{Laurent expansions}
A function holomorphic on an annulus has a unique expansion $\sum_{n=-\infty}^{\infty}a_n(z-z_0)^n$.

% BEGIN VOL04-EXPANSION IV08-example-01
\begin{example}[One function, two Laurent annuli]\label{ex:iv08-expand-01}
For \(f(z)=1/(z(z-1))\), partial fractions give \(-1/z+1/(z-1)\). On \(0<|z|<1\),
\[
\frac1{z-1}=-\frac1{1-z}=-\sum_{n\ge0}z^n,
\]
so \(f(z)=-z^{-1}-\sum_{n\ge0}z^n\). For \(|z|>1\), instead
\[
\frac1{z-1}=\frac1z\frac1{1-1/z}=\sum_{n\ge1}z^{-n},
\]
and cancellation of the first term gives \(f(z)=\sum_{n\ge2}z^{-n}\). The annulus is part of the answer.
\end{example}
% END VOL04-EXPANSION IV08-example-01
\section{Annuli of convergence}
Different singularities determine concentric annuli on which different Laurent expansions may be valid.

\section{Coefficient formula}
Laurent coefficients are contour integrals $a_n=(2\pi i)^{-1}\int f(\zeta)(\zeta-z_0)^{-n-1}d\zeta$.

\section{Principal part}
Negative-power terms constitute the principal part and measure the singular component.

% BEGIN VOL04-EXPANSION IV08-example-02
\begin{example}[Essential singularity encoded by infinitely many negative powers]\label{ex:iv08-expand-02}
Substituting \(1/z\) into the exponential series gives
\[
e^{1/z}=1+\frac1z+\frac1{2!z^2}+\frac1{3!z^3}+\cdots.
\]
The Laurent series has infinitely many negative powers, so the singularity at zero is essential. The coefficient of \(z^{-1}\) is \(1\), anticipating the residue.
\end{example}
% END VOL04-EXPANSION IV08-example-02
\section{Positive part}
Nonnegative powers form an ordinary Taylor series.

\section{Geometric expansions}
Rational functions are expanded by rewriting factors into geometric-series form suited to the chosen annulus.

\section{Uniqueness}
The Laurent coefficients are uniquely determined by contour integrals.

% BEGIN VOL04-EXPANSION IV08-example-03
\begin{example}[Expansion about a shifted point]\label{ex:iv08-expand-03}
Expand \(1/(z+1)\) about \(z_0=1\). With \(w=z-1\),
\[
\frac1{z+1}=\frac1{2+w}=\frac12\sum_{n\ge0}(-w/2)^n,
\]
valid for \(|w|<2\). For \(|w|>2\), rewrite instead as \(w^{-1}(1+2/w)^{-1}\), producing a Laurent series in negative powers. The same singularity at \(z=-1\) sets the annular boundary.
\end{example}
% END VOL04-EXPANSION IV08-example-03
\section{Residue preview}
The coefficient $a_{-1}$ is the residue and controls contour integrals around the singularity.

\section{Annular deformation}
Coefficient integrals are independent of the chosen circle within the annulus.

\section{Core structural results}

\begin{theorem}[Laurent theorem]\label{thm:iv08-01}
A function holomorphic on an annulus $r<|z-z_0|<R$ has a unique Laurent series converging normally on compact subannuli.
\begin{proof}
Apply Cauchy's formula to a region bounded by two circles, expand the outer and inner Cauchy kernels geometrically in the appropriate ratios, and justify termwise integration by uniform convergence on compact subannuli.
\end{proof}
\end{theorem}

\begin{theorem}[Laurent coefficient formula]\label{thm:iv08-02}
For any circle $|\zeta-z_0|=\rho$ inside the annulus, $a_n=\frac1{2\pi i}\int\frac{f(\zeta)}{(\zeta-z_0)^{n+1}}d\zeta$.
\begin{proof}
Insert the normally convergent Laurent series into the integral and use orthogonality of powers around the circle; only the $n$th coefficient survives.
\end{proof}
\end{theorem}

\begin{theorem}[Uniqueness]\label{thm:iv08-03}
A Laurent expansion on a fixed annulus is unique.
\begin{proof}
The coefficient integral recovers every coefficient directly from the function, so two expansions must have identical coefficients.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Inner expansion]\label{exr:iv08-expand-01}
Find the Laurent series of \(1/(z(1-z))\) for \(0<|z|<1\).
\end{exercise}
\begin{hint}
Use the geometric series for \(1/(1-z)\).
\end{hint}
\begin{solution}
The series is \(z^{-1}+1+z+z^2+\cdots\).
\end{solution}

\begin{exercise}[Outer expansion]\label{exr:iv08-expand-02}
Expand \(1/(z-2)\) for \(|z|>2\).
\end{exercise}
\begin{hint}
Factor out \(z\).
\end{hint}
\begin{solution}
One obtains \(z^{-1}(1-2/z)^{-1}=\sum_{n\ge0}2^n z^{-n-1}\).
\end{solution}

\begin{exercise}[Coefficient extraction]\label{exr:iv08-expand-03}
What is the coefficient of \(z^{-1}\) in \(e^{1/z}\)?
\end{exercise}
\begin{hint}
Read it from the exponential series.
\end{hint}
\begin{solution}
The coefficient is \(1\).
\end{solution}

\begin{exercise}[Principal part]\label{exr:iv08-expand-04}
Find the principal part of \(3/z^4-2/z+7+z\) at zero.
\end{exercise}
\begin{hint}
Keep only negative powers.
\end{hint}
\begin{solution}
The principal part is \(3z^{-4}-2z^{-1}\).
\end{solution}

\begin{exercise}[Annulus boundaries]\label{exr:iv08-expand-05}
For a Laurent expansion about zero of \(1/((z-1)(z-3))\), name the natural radial regions.
\end{exercise}
\begin{hint}
Use distances from the center to the singularities.
\end{hint}
\begin{solution}
The regions are \(|z|<1\), \(1<|z|<3\), and \(|z|>3\), excluding the singular circles themselves.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Laurent coefficient formula]\label{exr:iv08-expand-06}
Derive the contour formula for a Laurent coefficient \(a_n\) on an annulus.
\end{exercise}
\begin{hint}
Multiply by \((z-z_0)^{-n-1}\) and integrate around a circle in the annulus.
\end{hint}
\begin{solution}
Termwise integration kills every power except the one with exponent \(-1\), yielding \(a_n=(2\pi i)^{-1}\int f(z)/(z-z_0)^{n+1}dz\).
\end{solution}

\begin{exercise}[Uniqueness]\label{exr:iv08-expand-07}
Prove uniqueness of the Laurent expansion on an annulus.
\end{exercise}
\begin{hint}
Use the coefficient formula.
\end{hint}
\begin{solution}
Every coefficient is determined by the contour integral formula, so two Laurent expansions of the same function must have identical coefficients.
\end{solution}

\begin{exercise}[Independence of radius]\label{exr:iv08-expand-08}
Show that Laurent coefficients do not depend on which circle inside the annulus is used.
\end{exercise}
\begin{hint}
Apply Cauchy's theorem to the region between two circles.
\end{hint}
\begin{solution}
The relevant coefficient integrand is holomorphic throughout the intervening annulus, so its integrals over the two positively oriented circles are equal.
\end{solution}

\begin{exercise}[Taylor as a special Laurent series]\label{exr:iv08-expand-09}
Explain why a Laurent series with no negative powers is a Taylor series.
\end{exercise}
\begin{hint}
The nonnegative coefficients define an ordinary power series.
\end{hint}
\begin{solution}
With all negative coefficients zero, the representation extends analytically across the center and is precisely the Taylor expansion there.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Wrong geometric expansion]\label{exr:iv08-expand-10}
Why is \(1/(1-z)=\sum z^n\) invalid for \(|z|>1\)?
\end{exercise}
\begin{hint}
Check the geometric ratio.
\end{hint}
\begin{solution}
The terms \(z^n\) do not even tend to zero when \(|z|>1\). One must expand in powers of \(1/z\) instead.
\end{solution}

\begin{exercise}[Annulus omitted]\label{exr:iv08-expand-11}
Why is a Laurent formula incomplete without its annulus of validity?
\end{exercise}
\begin{hint}
Different geometric rewritings can produce different series.
\end{hint}
\begin{solution}
The same function can have distinct Laurent expansions in different annuli centered at the same point. The convergence region determines which series represents the function.
\end{solution}

\begin{exercise}[Finite principal part]\label{exr:iv08-expand-12}
Does a finite principal part necessarily indicate an essential singularity?
\end{exercise}
\begin{hint}
Compare the singularity classification by negative terms.
\end{hint}
\begin{solution}
No. A finite nonempty principal part corresponds to a pole; infinitely many negative terms characterize an essential singularity.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Residue preview]\label{exr:iv08-expand-13}
Why is the \((z-z_0)^{-1}\) Laurent coefficient especially important for contour integration?
\end{exercise}
\begin{hint}
Integrate each Laurent monomial around a circle.
\end{hint}
\begin{solution}
Every monomial integrates to zero except exponent \(-1\), whose integral is \(2\pi i\). Thus that coefficient alone controls the local contour contribution.
\end{solution}

\begin{exercise}[Local singularity diagnosis]\label{exr:iv08-expand-14}
How does a computed Laurent series distinguish removable, pole, and essential behavior?
\end{exercise}
\begin{hint}
Inspect the principal part.
\end{hint}
\begin{solution}
No negative powers means removable; finitely many negative powers means a pole; infinitely many negative powers means an essential singularity.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Three radial regions]\label{exr:iv08-expand-15}
Find the form of the Laurent expansion of \(1/(z(z-1)(z-2))\) on each of \(0<|z|<1\), \(1<|z|<2\), and \(|z|>2\).
\end{exercise}
\begin{hint}
Use partial fractions and choose a geometric expansion for each factor appropriate to the region.
\end{hint}
\begin{solution}
Partial fractions reduce the problem to terms at \(0,1,2\). In each annulus, expand factors with singularities outside in nonnegative powers of \(z\) and factors inside in negative powers of \(z\); the resulting series are different but unique on their regions.
\end{solution}

\begin{exercise}[Center one]\label{exr:iv08-expand-16}
Describe the natural annuli for the Laurent expansion of \(1/(z(z-2))\) about \(z_0=1\).
\end{exercise}
\begin{hint}
Measure the two singularity distances from the new center.
\end{hint}
\begin{solution}
Both singularities are at distance one from the center. Thus there are only two radial regions: \(|z-1|<1\) and \(|z-1|>1\), with the singular circle separating them.
\end{solution}
