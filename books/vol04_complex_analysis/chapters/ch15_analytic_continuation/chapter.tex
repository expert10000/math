\chapter{Analytic Continuation}
\label{ch:iv15}

Analytic continuation extends local holomorphic data across overlapping domains. The identity theorem guarantees uniqueness, while singularities and monodromy determine how far continuation can proceed.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item continue analytic germs along paths by overlapping local expansions;
\item use the identity theorem to prove uniqueness of continuation on overlaps;
\item identify natural barriers to analytic continuation from singularities or monodromy;
\item compare continuations along different paths in simply connected and non-simply-connected domains;
\item construct examples of multivalued continuation such as logarithms or roots;
\item state precisely when analytic continuation yields a single-valued global function;
\end{itemize}

\section{Germs}
A germ records the local behavior of a holomorphic function near one point, modulo agreement on smaller neighborhoods.

\begin{example}[Continuing the geometric series]\label{ex:iv15-expand-01}
The series \(\sum_{n\ge0}z^n\) defines \(1/(1-z)\) only for \(|z|<1\). The rational formula, however, is holomorphic on \(\mathbb C\setminus\{1\}\) and agrees with the series on the disk. It therefore supplies an analytic continuation far beyond the original circle of convergence, except across the genuine singularity at \(1\).
\end{example}

\begin{example}[Uniqueness on overlaps]\label{ex:iv15-expand-03}
Suppose two analytic continuations of the same germ are defined on overlapping connected regions. On a neighborhood where both agree with the original germ, their difference vanishes. The identity theorem then forces them to agree throughout the connected overlap. This is the local uniqueness mechanism behind analytic continuation.
\end{example}

\section{Core structural results}

\begin{theorem}[Uniqueness of analytic continuation]\label{thm:iv15-01}
If two holomorphic continuations agree on one nonempty open subset of a connected overlap, then they agree on the entire overlap.
\begin{proof}
Their difference is holomorphic and vanishes on an open set, hence by the identity theorem vanishes everywhere on the connected overlap.
\end{proof}
\end{theorem}

\begin{theorem}[Continuation along a fixed chain is unique]\label{thm:iv15-02}
Given a chain of connected overlapping domains and an initial germ, at most one compatible holomorphic element can occupy each successive domain.
\begin{proof}
Apply uniqueness of analytic continuation at each overlap inductively.
\end{proof}
\end{theorem}

\begin{theorem}[Pathwise local uniqueness]\label{thm:iv15-03}
Two continuations of the same initial germ along sufficiently close subdivisions of the same path agree wherever both are defined.
\begin{proof}
Refine the subdivisions to common overlapping neighborhoods and repeatedly apply the identity theorem.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Rational continuation]\label{exr:iv15-expand-01}
Give an analytic continuation of \(\sum_{n\ge0}z^n\) beyond \(|z|<1\).
\end{exercise}
\begin{hint}
Use its closed form.
\end{hint}
\begin{solution}
The continuation is \(1/(1-z)\) on \(\mathbb C\setminus\{1\}\).
\end{solution}

\begin{exercise}[Continuation obstruction]\label{exr:iv15-expand-02}
What point blocks continuation of \(1/(1-z)\) as a holomorphic function?
\end{exercise}
\begin{hint}
Locate the pole.
\end{hint}
\begin{solution}
The point \(z=1\) is a genuine pole and cannot be crossed by a holomorphic continuation of that function.
\end{solution}

\begin{exercise}[Overlap rule]\label{exr:iv15-expand-03}
Two analytic elements agree on a nonempty open subset of a connected overlap. What follows?
\end{exercise}
\begin{hint}
Use the identity theorem.
\end{hint}
\begin{solution}
They agree on the whole connected overlap.
\end{solution}

\begin{exercise}[Loop effect]\label{exr:iv15-expand-04}
After one loop around zero, what happens to a continued square root?
\end{exercise}
\begin{hint}
Track the logarithm increment.
\end{hint}
\begin{solution}
It changes sign.
\end{solution}

\begin{exercise}[Two loops]\label{exr:iv15-expand-05}
What happens to that square root after two loops?
\end{exercise}
\begin{hint}
Apply the sign change twice.
\end{hint}
\begin{solution}
It returns to the original branch value.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Uniqueness of continuation]\label{exr:iv15-expand-06}
Prove uniqueness of analytic continuation along a fixed chain of overlapping connected neighborhoods.
\end{exercise}
\begin{hint}
Induct across the overlaps using the identity theorem.
\end{hint}
\begin{solution}
Agreement on the initial germ forces agreement on the first overlap, then the next, and so on. Thus every step of the continuation chain is uniquely determined.
\end{solution}

\begin{exercise}[Continuation preserves algebraic identities]\label{exr:iv15-expand-07}
If two analytic functions satisfy \(F^2=g\) on the initial germ and both sides continue, show the identity persists.
\end{exercise}
\begin{hint}
Continue the difference \(F^2-g\).
\end{hint}
\begin{solution}
The difference is analytic and initially zero. Uniqueness and the identity theorem force it to remain zero wherever the continuation is defined.
\end{solution}

\begin{exercise}[Path reversal]\label{exr:iv15-expand-08}
Show that continuation along a path followed by its reverse returns the original germ when all continuations exist uniquely along the path.
\end{exercise}
\begin{hint}
Use local uniqueness step by step in reverse order.
\end{hint}
\begin{solution}
Each overlap transition is uniquely invertible by the same identity theorem argument, so retracing the chain recovers the starting analytic element.
\end{solution}

\begin{exercise}[Simply connected monodromy principle]\label{exr:iv15-expand-09}
Explain why path independent continuation is expected on a simply connected region when continuation is possible along every path.
\end{exercise}
\begin{hint}
Homotope any two paths with common endpoints while avoiding singularities.
\end{hint}
\begin{solution}
Local uniqueness makes the terminal germ stable under small path deformations. A homotopy between the paths then forces their terminal germs to agree.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Disconnected overlap]\label{exr:iv15-expand-10}
Why does agreement on one component of a disconnected overlap not force agreement on another component?
\end{exercise}
\begin{hint}
The identity theorem propagates through connectedness.
\end{hint}
\begin{solution}
The functions may differ on another component because there is no connected analytic path of equality joining the components.
\end{solution}

\begin{exercise}[Circle of convergence as barrier]\label{exr:iv15-expand-11}
Is the boundary of a Taylor disk always a natural barrier to analytic continuation?
\end{exercise}
\begin{hint}
Use the geometric series as a counterexample.
\end{hint}
\begin{solution}
No. Its Taylor series stops at \(|z|=1\), but the function continues through most boundary points. The radius records the nearest singularity relative to the chosen center, not an automatic global barrier.
\end{solution}

\begin{exercise}[Path dependence]\label{exr:iv15-expand-12}
Can analytic continuation around a loop return a different germ?
\end{exercise}
\begin{hint}
Use the square root around zero.
\end{hint}
\begin{solution}
Yes. Multivalued analytic phenomena exhibit monodromy; the terminal germ can differ even at the same base point when the domain has nontrivial loops around branch points.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Extending local formulas]\label{exr:iv15-expand-13}
Why is analytic continuation useful when a power series converges only locally?
\end{exercise}
\begin{hint}
The locally defined analytic function may have compatible representations on overlapping regions.
\end{hint}
\begin{solution}
Continuation transports the same analytic object beyond one convergence disk while preserving identities and derivatives on overlaps.
\end{solution}

\begin{exercise}[Detecting branch structure]\label{exr:iv15-expand-14}
How can continuation around loops reveal a hidden Riemann surface?
\end{exercise}
\begin{hint}
Record how branch values permute after closed loops.
\end{hint}
\begin{solution}
Nontrivial permutations show that one planar domain cannot support a single valued version. The collection of branches and their continuation rules points toward a covering or branched surface.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Logarithm monodromy]\label{exr:iv15-expand-15}
Describe the effect of one positive circuit around zero on an analytically continued logarithm.
\end{exercise}
\begin{hint}
Track the argument change.
\end{hint}
\begin{solution}
The logarithm increases by \(2\pi i\). Repeated circuits add integer multiples of \(2\pi i\).
\end{solution}

\begin{exercise}[Algebraic branch permutation]\label{exr:iv15-expand-16}
For \(w^3=z\), describe the monodromy after one positive loop around zero.
\end{exercise}
\begin{hint}
A logarithm increment of \(2\pi i\) is divided by three.
\end{hint}
\begin{solution}
Each cube root is multiplied by \(e^{2\pi i/3}\), cyclically permuting the three branches. Three loops return each branch to itself.
\end{solution}
