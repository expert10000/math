\chapter{Compactification and Genus}
\label{ch:iv26}

Compactification adds missing points in a way compatible with the complex structure, while genus records the basic topological type of a compact connected Riemann surface.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compactify basic noncompact Riemann surfaces by adding points with valid local coordinates;
\item compute or identify genus from polygonal, covering, or Euler-characteristic data;
\item distinguish genus zero, one, and higher-genus compact surfaces in standard examples;
\item relate punctures, compactification points, and meromorphic extension;
\item use Riemann--Hurwitz-type data where available to constrain genus;
\item check that the added points produce a compact Riemann surface rather than only a topological compactification;
\end{itemize}

\section{Adding infinity}
The complex plane compactifies to the Riemann sphere by adding one point with coordinate $w=1/z$.

\begin{example}[Compactifying the complex plane]\label{ex:iv26-expand-01}
Add one point \(\infty\) to \(\mathbb C\). Near infinity use the coordinate
\[
w=\frac1z.
\]
A neighborhood of \(\infty\) corresponds to \(|z|>R\) together with infinity, which maps to \(|w|<1/R\). The reciprocal transition is holomorphic away from zero, giving the one-point compactification its standard Riemann-surface structure.
\end{example}

\section{Genus}
For a compact orientable surface, genus counts handles and determines Euler characteristic $\chi=2-2g$.

% BEGIN VOL04-EXPANSION IV26-example-02
\begin{example}[Genus from Euler characteristic]\label{ex:iv26-expand-02}
For a compact orientable surface,
\[
\chi=2-2g.
\]
The sphere has \(\chi=2\), hence \(g=0\). A torus has a cell decomposition with one vertex, two edges and one face, so \(\chi=1-2+1=0\), hence \(g=1\). This topological invariant matches the analytic distinction between the sphere and complex tori.
\end{example}
% END VOL04-EXPANSION IV26-example-02

\section{Branched-cover viewpoint}
Maps to the sphere reveal genus through their degree and ramification.

\begin{example}[Riemann--Hurwitz for a torus double cover]\label{ex:iv26-expand-03}
Let \(X\to\widehat{\mathbb C}\) be a double cover branched simply at four points. Then
\[
2-2g(X)=2\cdot2-4=0,
\]
so \(g(X)=1\). Thus any such compact double cover has torus topology. The calculation is the bridge from branch data to global genus.
\end{example}

For finite branched covers, Euler characteristics differ by a total ramification correction.

\section{Compactness consequences}
Holomorphic functions, divisors, and differential forms become globally constrained on compact surfaces.

\section{Core structural results}

\begin{theorem}[Holomorphic functions on compact surfaces]\label{thm:iv26-01}
Every holomorphic function from a compact connected Riemann surface to $\mathbb C$ is constant.
\begin{proof}
The absolute value attains a maximum by compactness. In a local coordinate the maximum-modulus principle forces local constancy, and connectedness propagates it globally.
\end{proof}
\end{theorem}

\begin{theorem}[Riemann--Hurwitz formula]\label{thm:iv26-02}
For a degree-$d$ holomorphic map $f:X\to Y$ of compact Riemann surfaces, $2g_X-2=d(2g_Y-2)+\sum_{p\in X}(e_p-1)$.
\begin{proof}
Triangulate the target with branch values among vertices and lift the triangulation. Faces and edges lift $d$-fold, while vertices lose exactly the total ramification multiplicity. Comparing Euler characteristics yields the formula.
\end{proof}
\end{theorem}

\begin{theorem}[Genus of a double cover]\label{thm:iv26-03}
A degree-two cover of the sphere branched simply at $r$ points has genus $g=(r-2)/2$.
\begin{proof}
Apply Riemann--Hurwitz with $d=2$, target genus zero, and total ramification $r$: $2g-2=-4+r$.
\end{proof}
\end{theorem}

\begin{exercise}[Compact maximum principle]\label{exr:iv26-expand-06}
Prove that every holomorphic function on a compact connected Riemann surface is constant.
\end{exercise}
\begin{hint}
The modulus attains a maximum.
\end{hint}

\begin{exercise}[Sphere compactification]\label{exr:iv26-expand-07}
Show that the reciprocal coordinate produces a compatible chart at infinity.
\end{exercise}
\begin{hint}
Check the overlap transition.
\end{hint}
\begin{solution}
On the overlap with the finite chart, \(w=1/z\) is biholomorphic. Therefore adjoining the point \(w=0\) defines a valid complex chart around infinity.
\end{solution}

\begin{exercise}[Torus Euler characteristic]\label{exr:iv26-expand-08}
Prove from a square cell decomposition that a torus has Euler characteristic zero.
\end{exercise}
\begin{hint}
Count vertices, edges and faces after identifications.
\end{hint}
\begin{solution}
All four vertices identify to one, opposite edge pairs give two edges, and there is one face, so \(\chi=1-2+1=0\).
\end{solution}

\begin{exercise}[Riemann--Hurwitz genus]\label{exr:iv26-expand-09}
Derive \(g=(r-2)/2\) for a double cover of the sphere with \(r\) simple branch points.
\end{exercise}
\begin{hint}
Insert degree two and total ramification \(r\).
\end{hint}
\begin{solution}
The formula gives \(2-2g=4-r\), hence \(g=(r-2)/2\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Remove compactness]\label{exr:iv26-expand-10}
Does the theorem that every holomorphic function is constant remain true on a noncompact Riemann surface?
\end{exercise}
\begin{hint}
Use the complex plane.
\end{hint}
\begin{solution}
No. The identity function on \(\mathbb C\) is a nonconstant holomorphic example.
\end{solution}

\begin{exercise}[Every puncture fills]\label{exr:iv26-expand-11}
Can every puncture of a Riemann surface be filled holomorphically?
\end{exercise}
\begin{hint}
Consider essential singular behavior.
\end{hint}
\begin{solution}
No. Filling requires the complex structure and relevant functions or maps to extend appropriately; essential or topological behavior can obstruct a chosen extension.
\end{solution}

\begin{exercise}[Genus from local charts]\label{exr:iv26-expand-12}
Can genus be read from one local coordinate chart?
\end{exercise}
\begin{hint}
Genus is global topology.
\end{hint}
\begin{solution}
No. Every surface is locally disk-like; genus measures global handle structure and requires global information.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Projective completion]\label{exr:iv26-expand-13}
Why does adding points at infinity help when studying affine algebraic curves?
\end{exercise}
\begin{hint}
Compactness gives global constraints.
\end{hint}
\begin{solution}
The projective completion packages behavior at infinity into finitely many surface points and makes tools such as divisors, meromorphic functions and Riemann--Hurwitz global.
\end{solution}

\begin{exercise}[Meromorphic maps]\label{exr:iv26-expand-14}
Why are meromorphic functions on a compact Riemann surface naturally viewed as holomorphic maps to the sphere?
\end{exercise}
\begin{hint}
Treat poles as values equal to infinity.
\end{hint}
\begin{solution}
Near a pole, the reciprocal is holomorphic and vanishes, so adjoining the sphere point at infinity turns the meromorphic function into a holomorphic map.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Degree and ramification]\label{exr:iv26-expand-15}
A degree-three map from a genus-two surface to the sphere has total ramification \(R\). Find \(R\).
\end{exercise}
\begin{hint}
Apply Riemann--Hurwitz.
\end{hint}
\begin{solution}
\(2-2\cdot2=3\cdot2-R\), so \(-2=6-R\) and \(R=8\).
\end{solution}

\begin{exercise}[Why genus is invariant]\label{exr:iv26-expand-16}
Explain why biholomorphic compact Riemann surfaces must have the same genus.
\end{exercise}
\begin{hint}
A biholomorphism is in particular a homeomorphism.
\end{hint}
\begin{solution}
Genus is a topological invariant of compact orientable surfaces, and a biholomorphism preserves the underlying topology.
\end{solution}
