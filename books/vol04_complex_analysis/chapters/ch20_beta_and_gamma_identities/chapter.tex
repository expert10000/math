\chapter{Beta and Gamma Identities}
\label{ch:iv20}

The Beta function packages a two-endpoint integral and connects directly to Gamma through a change of variables. Their identities provide a compact calculus for factorials, binomial coefficients, trigonometric integrals, and parameter differentiation.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item derive the Beta--Gamma relation by a justified change of variables in a double integral;
\item compute Beta integrals and reduce them to Gamma values;
\item apply reflection and duplication formulas to simplify special-function expressions;
\item track convergence conditions on parameters before manipulating integrals;
\item use analytic continuation to extend identities beyond their initial integral domains;
\item check special cases against factorial or trigonometric values as consistency tests;
\end{itemize}

\section{Beta integral}
For $\Re x,\Re y>0$, $B(x,y)=\int_0^1 t^{x-1}(1-t)^{y-1}dt$.

% BEGIN VOL04-EXPANSION IV20-example-01
\begin{example}[Beta at integer parameters]\label{ex:iv20-expand-01}
For positive integers \(m,n\),
\[
B(m,n)=\frac{\Gamma(m)\Gamma(n)}{\Gamma(m+n)}
=\frac{(m-1)!(n-1)!}{(m+n-1)!}.
\]
Thus \(B(3,4)=2!3!/6!=1/60\). Directly, this says
\[
\int_0^1 t^2(1-t)^3\,dt=\frac1{60}.
\]
The Gamma quotient turns a two-endpoint integral into a factorial identity.
\end{example}
% END VOL04-EXPANSION IV20-example-01
\section{Symmetry}
The substitution $t\mapsto1-t$ gives $B(x,y)=B(y,x)$.

\section{Gamma relation}
A two-dimensional change of variables yields $B(x,y)=\Gamma(x)\Gamma(y)/\Gamma(x+y)$.

\section{Recurrences}
Integration by parts and the Gamma relation produce rational recurrences in the parameters.

% BEGIN VOL04-EXPANSION IV20-example-02
\begin{example}[A sine power integral]\label{ex:iv20-expand-02}
Using \(t=\sin^2\theta\),
\[
\int_0^{\pi/2}\sin^{a-1}\theta\cos^{b-1}\theta\,d\theta
=\frac12B\!\left(\frac a2,\frac b2\right).
\]
With \(a=6\) and \(b=2\),
\[
\int_0^{\pi/2}\sin^5\theta\cos\theta\,d\theta
=\frac12 B(3,1)=\frac16.
\]
This agrees with the elementary substitution \(u=\sin\theta\).
\end{example}

\section{Parameter derivatives}

\begin{example}[Differentiating a parameter]\label{ex:iv20-expand-03}
For \(\Re x,\Re y>0\), differentiate the Beta integral with respect to \(x\):
\[
\frac{\partial B}{\partial x}(x,y)
=\int_0^1 t^{x-1}(1-t)^{y-1}\log t\,dt.
\]
From the Gamma quotient,
\[
\frac{\partial B}{\partial x}(x,y)
=B(x,y)\bigl(\psi(x)-\psi(x+y)\bigr).
\]
At \((1,1)\), this gives \(\int_0^1\log t\,dt=-1\).
\end{example}
% END VOL04-EXPANSION IV20-example-03

\section{Core structural results}

\begin{theorem}[Beta--Gamma identity]\label{thm:iv20-01}
For $\Re x,\Re y>0$, $B(x,y)=\Gamma(x)\Gamma(y)/\Gamma(x+y)$.
\begin{proof}
Write the product of Gamma integrals over the first quadrant and substitute $u=r t$, $v=r(1-t)$ with $r>0$, $0<t<1$. The Jacobian is $r$, and the integral factors into $\Gamma(x+y)B(x,y)$.
\end{proof}
\end{theorem}

\begin{theorem}[Trigonometric Beta formula]\label{thm:iv20-02}
For $a,b>0$, $\int_0^{\pi/2}\sin^{a-1}\theta\cos^{b-1}\theta\,d\theta=\frac12B(a/2,b/2)$.
\begin{proof}
Set $t=\sin^2\theta$, so $dt=2\sin\theta\cos\theta\,d\theta$ and collect the resulting powers.
\end{proof}
\end{theorem}

\begin{theorem}[Beta recurrence]\label{thm:iv20-03}
$B(x+1,y)=\frac{x}{x+y}B(x,y)$ in the common domain of definition.
\begin{proof}
Use the Beta--Gamma identity and the Gamma functional equation in numerator and denominator.
\end{proof}
\end{theorem}

\begin{exercise}[Simple Beta value]\label{exr:iv20-expand-01}

\end{hint}

\begin{exercise}[Symmetry]\label{exr:iv20-expand-02}

Use \(B(x,y)=B(y,x)\).
\end{hint}

The two values are equal, and both are \(1!4!/6!=1/30\).

\subsection*{Proofs}

\begin{exercise}[Symmetry proof]\label{exr:iv20-expand-06}
Prove \(B(x,y)=B(y,x)\) directly from the integral.
\end{exercise}
\begin{hint}
Use \(u=1-t\).
\end{hint}
\begin{solution}
The substitution exchanges the factors \(t^{x-1}\) and \((1-t)^{y-1}\) and preserves the interval.
\end{solution}

\begin{exercise}[Beta Gamma identity]\label{exr:iv20-expand-07}
Prove the Beta Gamma identity from the product of two Gamma integrals.
\end{exercise}
\begin{hint}
Use \(u=rt\) and \(v=r(1-t)\).
\end{hint}
\begin{solution}
The first quadrant transforms to \(r>0\), \(0<t<1\) with Jacobian \(r\). The product factors as \(\Gamma(x+y)B(x,y)\).
\end{solution}

\begin{exercise}[Trigonometric form]\label{exr:iv20-expand-08}
Derive the trigonometric Beta formula.
\end{exercise}
\begin{hint}
Set \(t=\sin^2\theta\).
\end{hint}
\begin{solution}
The differential contributes \(2\sin\theta\cos\theta\,d\theta\), and the remaining powers combine to give the stated half Beta integral.
\end{solution}

\begin{exercise}[Recurrence proof]\label{exr:iv20-expand-09}
Prove the recurrence for \(B(x+1,y)\).
\end{exercise}
\begin{hint}
Apply the Gamma quotient to both sides.
\end{hint}
\begin{solution}
Use \(\Gamma(x+1)=x\Gamma(x)\) and \(\Gamma(x+y+1)=(x+y)\Gamma(x+y)\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Endpoint failure]\label{exr:iv20-expand-10}
What happens to the defining Beta integral if \(\Re x\le0\)?
\end{exercise}
\begin{hint}
Inspect the endpoint \(t=0\).
\end{hint}
\begin{solution}
The factor \(t^{x-1}\) is generally not integrable there, so the defining integral can fail even though meromorphic continuation may still exist.
\end{solution}

\begin{exercise}[Symmetry domain]\label{exr:iv20-expand-11}
Does the symmetry identity disappear after meromorphic continuation?
\end{exercise}
\begin{hint}
Compare the Gamma quotient under swapping variables.
\end{hint}
\begin{solution}
No. The quotient is symmetric wherever both sides are defined, and continuation preserves the identity meromorphically.
\end{solution}

\begin{exercise}[Differentiate without control]\label{exr:iv20-expand-12}
May one always differentiate the Beta integral with respect to a parameter at the boundary of its convergence region?
\end{exercise}
\begin{hint}
Check for a common integrable majorant.
\end{hint}
\begin{solution}
No. Differentiation under the integral requires uniform integrability control. Boundary parameters can introduce nonintegrable logarithmic singularities.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Probability normalization]\label{exr:iv20-expand-13}
Why does \(1/B(a,b)\) normalize the density \(t^{a-1}(1-t)^{b-1}\) on \((0,1)\)?
\end{exercise}
\begin{hint}
Integrate the density kernel.
\end{hint}
\begin{solution}
Its integral is exactly \(B(a,b)\), so division by this value makes the total mass equal to one.
\end{solution}

\begin{exercise}[Central binomial coefficient]\label{exr:iv20-expand-14}
Explain why half-integer Gamma values naturally produce central binomial coefficients.
\end{exercise}
\begin{hint}
Insert \(\Gamma(n+1/2)\) into a Beta quotient.
\end{hint}
\begin{solution}
The half-integer formula contains \((2n)!/(4^n n!)\), the factorial ratio underlying \(\binom{2n}{n}\).
\end{solution}
