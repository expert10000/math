\chapter{Addition Formulas}
\label{ch:iv30}

Addition formulas express the group structure of the complex torus directly in the values of $\wp$ and $\wp'$. They are the analytic form of the chord-and-tangent law on the associated cubic curve.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item derive addition formulas for \(\wp\) and related elliptic functions from analytic identities;
\item specialize the addition formula to obtain duplication formulas;
\item track poles and symmetry to verify candidate elliptic identities;
\item use the identity theorem to prove equality of elliptic functions once principal parts and enough values agree;
\item translate addition on the torus into algebraic formulas for \(\wp\) and \(\wp'\);
\item apply addition identities to compute special values or group-law coordinates;
\end{itemize}

\section{Translation on the torus}
Addition of complex arguments modulo the lattice is the group law on $\mathbb C/\Lambda$.

% BEGIN VOL04-EXPANSION IV30-example-01
\begin{example}[Using the addition formula]\label{ex:iv30-expand-01}
For points \(u,v\) with \(\wp(u)\ne\wp(v)\), set
\[
m=\frac{\wp'(u)-\wp'(v)}{\wp(u)-\wp(v)}.
\]
Then
\[
\wp(u+v)=-\wp(u)-\wp(v)+\frac{m^2}{4}.
\]
Thus the sum coordinate is determined by the two points and the chord slope. The same expression is exactly the \(x\)-coordinate computation in the chord law on \(y^2=4x^3-g_2x-g_3\).
\end{example}
% END VOL04-EXPANSION IV30-example-01
\section{Need for an addition law}
Since $\wp$ is even, $\wp(u+v)$ cannot be expressed from $\wp(u),\wp(v)$ alone without derivative data.

\section{Weierstrass addition formula}
A rational expression in $\wp(u),\wp(v),\wp'(u),\wp'(v)$ gives $\wp(u+v)$.

\section{Duplication formula}
Taking $v\to u$ yields the tangent or duplication formula.

% BEGIN VOL04-EXPANSION IV30-example-02
\begin{example}[Duplication as a tangent limit]\label{ex:iv30-expand-02}
Let \(v\to u\). The chord quotient satisfies
\[
\frac{\wp'(u)-\wp'(v)}{\wp(u)-\wp(v)}\longrightarrow
\frac{\wp''(u)}{\wp'(u)}
\]
when \(\wp'(u)\ne0\). Therefore
\[
\wp(2u)=-2\wp(u)+\frac14\left(\frac{\wp''(u)}{\wp'(u)}\right)^2.
\]
The chord has become the tangent, matching the geometric duplication law on the cubic.
\end{example}
% END VOL04-EXPANSION IV30-example-02
\section{Subtraction formula}
Replacing $v$ by $-v$ gives a companion formula for $\wp(u-v)$.

\section{Pole comparison proof}
Differences of candidate expressions can be shown constant by comparing elliptic principal parts.

\section{Cubic geometry}
The slope term in the addition formula is the slope of the chord joining points on the Weierstrass cubic.

% BEGIN VOL04-EXPANSION IV30-example-03
\begin{example}[Adding a half-period]\label{ex:iv30-expand-03}
Let \(a\) be a nonzero half-period, so \(\wp'(a)=0\). In the addition formula with \(v=a\), the slope becomes
\[
\frac{\wp'(u)}{\wp(u)-\wp(a)}.
\]
Hence translation by a two-torsion point is represented by a rational expression in \(\wp(u)\) and \(\wp'(u)\). The simplification reflects the fact that half-periods are points of order two on the torus.
\end{example}
% END VOL04-EXPANSION IV30-example-03
\section{Special values}
Half-periods and torsion points simplify the formulas because $\wp'$ may vanish.

\section{Division phenomena}
Iterating addition generates rational functions describing multiplication-by-$n$ on the elliptic curve.

\section{Core structural results}

\begin{theorem}[Weierstrass addition formula]\label{thm:iv30-01}
For $u,v,u\pm v\notin\Lambda$ and $\wp(u)\ne\wp(v)$, $\wp(u+v)=-\wp(u)-\wp(v)+\frac14\left(\frac{\wp'(u)-\wp'(v)}{\wp(u)-\wp(v)}\right)^2$.
\begin{proof}
As a function of one variable, compare both sides after shifting by the other argument. Their poles and principal parts agree; the difference is an elliptic function without poles, hence constant. Evaluate at a convenient limiting point to determine the constant.
\end{proof}
\end{theorem}

\begin{theorem}[Duplication formula]\label{thm:iv30-02}
The limit $v\to u$ in the addition formula gives $\wp(2u)=-2\wp(u)+\frac14(\wp''(u)/\wp'(u))^2$ whenever the displayed quotient is defined.
\begin{proof}
Apply l'Hôpital's rule to the chord-slope quotient as $v\to u$.
\end{proof}
\end{theorem}

\begin{theorem}[Chord interpretation]\label{thm:iv30-03}
Under $z\mapsto(\wp(z),\wp'(z))$, the analytic addition formula agrees with the chord-and-tangent law on $y^2=4x^3-g_2x-g_3$.
\begin{proof}
The squared chord slope determines the third intersection's $x$-coordinate by Vieta's formula; reflection in the $x$-axis changes the sign of $y$, matching the torus inverse $z\mapsto-z$.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Chord slope]\label{exr:iv30-expand-01}
Write the slope quantity appearing in the Weierstrass addition formula.
\end{exercise}
\begin{hint}
Use the difference of derivative values divided by the difference of function values.
\end{hint}
\begin{solution}
It is \((\wp'(u)-\wp'(v))/(\wp(u)-\wp(v))\).
\end{solution}

\begin{exercise}[Inverse point]\label{exr:iv30-expand-02}
What happens to \((\wp(z),\wp'(z))\) when \(z\) is replaced by \(-z\)?
\end{exercise}
\begin{hint}
Use parity.
\end{hint}
\begin{solution}
The first coordinate is unchanged and the second changes sign.
\end{solution}

\begin{exercise}[Two-torsion]\label{exr:iv30-expand-03}
Why is a half-period a point of order two on the torus?
\end{exercise}
\begin{hint}
Double it modulo the lattice.
\end{hint}
\begin{solution}
Because \(2a\in\Lambda\), so \(2[a]=[0]\).
\end{solution}

\begin{exercise}[Duplication slope]\label{exr:iv30-expand-04}
What replaces the chord slope in the limit \(v\to u\)?
\end{exercise}
\begin{hint}
Apply l'Hopital to numerator and denominator.
\end{hint}
\begin{solution}
It becomes \(\wp''(u)/\wp'(u)\).
\end{solution}

\begin{exercise}[Subtraction]\label{exr:iv30-expand-05}
How is a formula for \(\wp(u-v)\) obtained from the addition formula?
\end{exercise}
\begin{hint}
Replace \(v\) by \(-v\).
\end{hint}
\begin{solution}
Use \(\wp(-v)=\wp(v)\) and \(\wp'(-v)=-\wp'(v)\) in the same addition identity.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Pole comparison]\label{exr:iv30-expand-06}
Explain the standard elliptic-function proof strategy for the addition formula.
\end{exercise}
\begin{hint}
Treat one side minus the other as a function of one variable.
\end{hint}
\begin{solution}
Show the difference is elliptic and that all poles and principal parts cancel. It is then holomorphic on the compact torus and hence constant; evaluate a convenient limit to determine the constant.
\end{solution}

\begin{exercise}[Duplication formula]\label{exr:iv30-expand-07}
Derive the duplication formula from the addition formula.
\end{exercise}
\begin{hint}
Let \(v\to u\) and use l'Hopital.
\end{hint}
\begin{solution}
The chord quotient tends to \(\wp''(u)/\wp'(u)\), yielding the tangent formula for \(\wp(2u)\).
\end{solution}

\begin{exercise}[Chord geometry]\label{exr:iv30-expand-08}
Why does the squared slope determine the third intersection x-coordinate on the Weierstrass cubic?
\end{exercise}
\begin{hint}
Substitute the line equation into the cubic and use Vieta.
\end{hint}
\begin{solution}
The resulting cubic in \(x\) has the two known intersection abscissas and a third one. Comparing the quadratic coefficient expresses the third abscissa in terms of the slope squared and the first two abscissas.
\end{solution}

\begin{exercise}[Compatibility with inverse]\label{exr:iv30-expand-09}
Prove that torus inversion corresponds to reflection across the x-axis on the cubic.
\end{exercise}
\begin{hint}
Use parity of \(\wp\) and \(\wp'\).
\end{hint}
\begin{solution}
Under \(z\mapsto-z\), \(x=\wp(z)\) stays fixed while \(y=\wp'(z)\) changes sign, exactly the cubic group inverse.
\end{solution}

\end{hint}

\end{solution}

\begin{exercise}[Only wp values]\label{exr:iv30-expand-12}

\end{solution}
