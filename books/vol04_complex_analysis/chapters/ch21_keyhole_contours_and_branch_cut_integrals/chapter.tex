\chapter{Keyhole Contours and Branch-Cut Integrals}
\label{ch:iv21}

Keyhole contours turn branch discontinuities into computable factors. They are the standard mechanism for integrals involving fractional powers, logarithms, and branch cuts.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item choose a branch of the logarithm appropriate for a keyhole contour;
\item compute the jump of a branch-dependent integrand across the two sides of a cut;
\item estimate inner- and outer-circle contributions in keyhole contours;
\item derive real integrals from residue sums and branch discontinuities;
\item track argument conventions and orientation signs explicitly;
\item distinguish branch-cut contributions from pole residues;
\end{itemize}

\begin{example}[Keyhole evaluation of a rational power integral]\label{ex:iv21-expand-01}
Fix \(0<a<1\) and use the branch \(0<\arg z<2\pi\) for \(z^{a-1}\). For
\[
f(z)=\frac{z^{a-1}}{1+z},
\]
the upper bank contributes \(I=\int_0^\infty x^{a-1}/(1+x)\,dx\), while the lower bank contributes \(-e^{2\pi i(a-1)}I\). The only enclosed pole is at \(-1\), whose residue is \(e^{i\pi(a-1)}\). Hence
\[
(1-e^{2\pi ia})I=2\pi i e^{i\pi(a-1)},
\]
which simplifies to \(I=\pi/\sin(\pi a)\).
\end{example}

\begin{example}[Why both circular arcs vanish]\label{ex:iv21-expand-02}
For the same integrand, on the outer circle \(|z|=R\),
\[
|f(z)|=O(R^{a-2}),
\]
so arc length \(O(R)\) gives contribution \(O(R^{a-1})\to0\) because \(a<1\). On the inner circle \(|z|=\varepsilon\),
\[
|f(z)|=O(\varepsilon^{a-1}),
\]
so the contribution is \(O(\varepsilon^a)\to0\) because \(a>0\). These two inequalities explain the exact parameter window \(0<a<1\).
\end{example}

\begin{example}[Producing a logarithm by parameter differentiation]\label{ex:iv21-expand-03}
Differentiate
\[
I(a)=\int_0^\infty \frac{x^{a-1}}{1+x}\,dx=\frac{\pi}{\sin(\pi a)}
\]
for \(0<a<1\). Then
\[
\int_0^\infty \frac{x^{a-1}\log x}{1+x}\,dx
=-\pi^2\frac{\cos(\pi a)}{\sin^2(\pi a)}.
\]
At \(a=1/2\), the right side vanishes, reflecting the symmetry \(x\mapsto1/x\).
\end{example}

\section{Core structural results}

\begin{theorem}[Keyhole jump identity]\label{thm:iv21-01}
For a branch cut along the positive real axis with argument in $(0,2\pi)$, the upper and lower boundary values of $z^\alpha$ differ by the factor $e^{2\pi i\alpha}$.
\begin{proof}
On the upper bank the argument tends to $0$, while on the lower bank it tends to $2\pi$. Substitution into $e^{\alpha(\log r+i\arg z)}$ gives the factor.
\end{proof}
\end{theorem}

\begin{theorem}[Keyhole reduction principle]\label{thm:iv21-02}
If the circular arcs vanish, the keyhole contour integral reduces to a scalar jump factor times the corresponding real integral along the cut.
\begin{proof}
Write the contour as upper bank plus outer circle plus lower bank plus inner circle. Let the radii tend to their limits; the circle contributions vanish and the two bank integrals combine using their branch-value relation and opposite orientations.
\end{proof}
\end{theorem}

\begin{theorem}[Differentiation in exponent]\label{thm:iv21-03}
When a keyhole identity is locally uniformly convergent in a parameter $\alpha$, differentiating it with respect to $\alpha$ introduces factors of $\operatorname{Log} z$ and yields logarithmic integral identities.
\begin{proof}
Differentiate $z^\alpha=e^{\alpha\operatorname{Log} z}$ under the integral sign, justified by a common integrable majorant on a compact parameter subregion.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Jump factor]\label{exr:iv21-expand-01}
For the branch \(0<\arg z<2\pi\), find the ratio of lower-bank to upper-bank values of \(z^\alpha\) on the positive axis.
\end{exercise}
\begin{hint}
The arguments are \(2\pi\) and \(0\).
\end{hint}
\begin{solution}
The ratio is \(e^{2\pi i\alpha}\).
\end{solution}

\begin{exercise}[Outer arc order]\label{exr:iv21-expand-02}
If \(f(z)=z^{a-1}/(1+z)\), what is the order of the outer-circle integral?
\end{exercise}
\begin{hint}
Multiply the magnitude estimate by arc length.
\end{hint}
\begin{solution}
It is \(O(R^{a-1})\), so it vanishes when \(a<1\).
\end{solution}

\begin{exercise}[Inner arc order]\label{exr:iv21-expand-03}
For the same integrand, what is the inner-circle order?
\end{exercise}
\begin{hint}
Near zero the denominator is bounded away from zero.
\end{hint}
\begin{solution}
The contribution is \(O(\varepsilon^a)\), so it vanishes when \(a>0\).
\end{solution}

\begin{exercise}[Pole residue]\label{exr:iv21-expand-04}
Find the residue of \(z^{a-1}/(1+z)\) at \(z=-1\) for \(0<\arg z<2\pi\).
\end{exercise}
\begin{hint}
Use \(\arg(-1)=\pi\).
\end{hint}
\begin{solution}
The residue is \((-1)^{a-1}=e^{i\pi(a-1)}\) on the chosen branch.
\end{solution}

\begin{exercise}[Logarithmic derivative]\label{exr:iv21-expand-05}
Differentiate \(x^{a-1}\) with respect to \(a\).
\end{exercise}
\begin{hint}
Write the power as an exponential.
\end{hint}
\begin{solution}
The derivative is \(x^{a-1}\log x\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Keyhole reduction]\label{exr:iv21-expand-06}
Prove that the two banks combine into a scalar jump factor times one real integral.
\end{exercise}
\begin{hint}
Track the opposite orientation on the lower bank.
\end{hint}
\begin{solution}
The lower bank runs from large radius to small radius and carries the branch phase. Reversing its limits produces a minus sign, so the sum is the upper integral times one minus the phase factor.
\end{solution}

\begin{exercise}[Parameter window]\label{exr:iv21-expand-07}
Prove that \(0<a<1\) is exactly the range making both circular arcs vanish for \(z^{a-1}/(1+z)\).
\end{exercise}
\begin{hint}
Estimate separately at zero and infinity.
\end{hint}
\begin{solution}
The inner contribution behaves like \(\varepsilon^a\), requiring \(a>0\); the outer contribution behaves like \(R^{a-1}\), requiring \(a<1\).
\end{solution}

\begin{exercise}[Orientation sign]\label{exr:iv21-expand-08}
Explain rigorously why the lower-bank integral has the opposite sign from the upper-bank integral after parametrization.
\end{exercise}
\begin{hint}
Write the lower bank as \(z=x e^{2\pi i}\) with decreasing \(x\).
\end{hint}
\begin{solution}
Its parameter runs from \(R\) down to \(\varepsilon\), so reversing the limits contributes the negative sign.
\end{solution}

\begin{exercise}[Differentiate under the integral]\label{exr:iv21-expand-09}
Give a justification for differentiating the keyhole identity with respect to \(a\) on a compact subinterval of \((0,1)\).
\end{exercise}
\begin{hint}
Find one integrable bound for the differentiated real integrand.
\end{hint}
\begin{solution}
On a compact parameter interval \([\delta,1-\delta]\), the derivative is bounded by a constant times \((x^{\delta-1}+x^{-\delta})|\log x|/(1+x)\), which is integrable at zero and infinity.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Wrong branch]\label{exr:iv21-expand-10}
What is wrong with using a keyhole contour before specifying a branch of \(z^\alpha\)?
\end{exercise}
\begin{hint}
The boundary values on the two banks are branch dependent.
\end{hint}
\begin{solution}
Without a branch, the integrand is not single valued on the contour domain, so neither the jump factor nor the residue values are determined.
\end{solution}

\begin{exercise}[Pole on the cut]\label{exr:iv21-expand-11}
Can the standard residue argument be used unchanged if a pole lies on the branch cut?
\end{exercise}
\begin{hint}
The contour would pass through a singularity.
\end{hint}
\begin{solution}
No. One must indent around the pole, use a principal value prescription, or choose a different cut or contour.
\end{solution}

\begin{exercise}[Arc estimate failure]\label{exr:iv21-expand-12}
What fails when \(a\ge1\) in the standard keyhole example?
\end{exercise}
\begin{hint}
Inspect the outer-circle bound.
\end{hint}
\begin{solution}
The estimate \(O(R^{a-1})\) no longer tends to zero, so the contour does not reduce to just the bank integrals and residues.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Mellin transform]\label{exr:iv21-expand-13}
Explain why keyhole contours naturally evaluate Mellin-type integrals.
\end{exercise}
\begin{hint}
Mellin kernels are complex powers.
\end{hint}
\begin{solution}
A Mellin kernel \(x^{s-1}\) acquires a controlled phase across a branch cut, so the jump converts a contour integral into the desired real Mellin integral.
\end{solution}

\begin{exercise}[Branch-cut logarithms]\label{exr:iv21-expand-14}
How can one create integrals containing \((\log x)^2\)?
\end{exercise}
\begin{hint}
Differentiate twice with respect to the exponent.
\end{hint}
\begin{solution}
Two parameter derivatives of \(x^{a-1}\) produce \(x^{a-1}(\log x)^2\), while the differentiated closed form provides the evaluation.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Hankel contour]\label{exr:iv21-expand-15}
Why is a Hankel contour around the negative real axis useful for reciprocal Gamma representations?
\end{exercise}
\begin{hint}
Think about the jump of a complex power and exponential decay.
\end{hint}
\begin{solution}
The power has a simple phase jump across the negative axis while \(e^z\) decays along the two rays, producing a contour representation closely adapted to \(1/\Gamma(s)\).
\end{solution}

\begin{exercise}[Wedge variant]\label{exr:iv21-expand-16}
How would the jump factor change if the branch cut were bounded by arguments \(\theta_0\) and \(\theta_0+2\pi\)?
\end{exercise}
\begin{hint}
Evaluate \(z^\alpha\) on the two boundary arguments.
\end{hint}
\begin{solution}
Both values gain a common factor \(e^{i\alpha\theta_0}\), while their ratio remains \(e^{2\pi i\alpha}\). The essential jump is therefore unchanged.
\end{solution}
