\chapter{The Gamma Function}
\label{ch:iv19}

The Gamma function extends factorials to the complex plane through an integral, functional equation, and analytic continuation. It is the prototype of a special function whose poles and growth encode arithmetic structure.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item derive the functional equation \(\Gamma(z+1)=z\Gamma(z)\) from the defining integral;
\item extend the Gamma function meromorphically using recurrence;
\item locate poles and compute residues at nonpositive integers;
\item apply reflection or multiplication identities under the correct hypotheses;
\item estimate Gamma asymptotics in elementary regimes developed in the chapter;
\item translate factorial identities into Gamma-function statements and back again;
\end{itemize}

\section{Euler integral}
For $\Re s>0$, define $\Gamma(s)=\int_0^\infty t^{s-1}e^{-t}dt$.

% BEGIN VOL04-EXPANSION IV19-example-01
\begin{example}[Gamma at half integers]\label{ex:iv19-expand-01}
Starting from
\[
\Gamma\!\left(\frac12\right)=\int_0^\infty t^{-1/2}e^{-t}\,dt=\sqrt{\pi},
\]
the functional equation gives
\[
\Gamma\!\left(\frac32\right)=\frac12\sqrt{\pi},\qquad
\Gamma\!\left(\frac52\right)=\frac34\sqrt{\pi}.
\]
More generally,
\[
\Gamma\!\left(n+\frac12\right)=\frac{(2n)!}{4^n n!}\sqrt{\pi}.
\]
This converts many Gaussian and trigonometric integrals into factorial calculations.
\end{example}

\begin{example}[A pole and its residue]\label{ex:iv19-expand-02}
Use \(\Gamma(s+1)=s\Gamma(s)\). Since \(\Gamma(s+1)\to1\) as \(s\to0\),
\[
\Gamma(s)=\frac{\Gamma(s+1)}{s}=\frac1s+O(1).
\]
Hence \(s=0\) is a simple pole with residue \(1\). Repeating the functional equation yields
\[
\operatorname{Res}_{s=-n}\Gamma(s)=\frac{(-1)^n}{n!}.
\]
The local computation shows exactly how meromorphic continuation records the factorial recursion at negative integers.
\end{example}

\begin{example}[A Gamma integral with scaling]\label{ex:iv19-expand-03}
For \(a>0\) and \(\Re s>0\), substitute \(u=at\) in
\[
I=\int_0^\infty t^{s-1}e^{-at}\,dt.
\]
Then
\[
I=a^{-s}\int_0^\infty u^{s-1}e^{-u}\,du=a^{-s}\Gamma(s).
\]
For example, \(\int_0^\infty t^{3/2}e^{-2t}\,dt=2^{-5/2}\Gamma(5/2)=3\sqrt{\pi}/(2^{9/2})\).
\end{example}

\section{Core structural results}

\begin{theorem}[Gamma functional equation]\label{thm:iv19-01}
For $\Re s>0$, $\Gamma(s+1)=s\Gamma(s)$.
\begin{proof}
Integrate $\int_0^\infty t^s e^{-t}dt$ by parts. Boundary terms vanish because $t^s e^{-t}$ decays at infinity and tends to zero at the origin under the real-part condition.
\end{proof}
\end{theorem}

\begin{theorem}[Meromorphic continuation]\label{thm:iv19-02}
The functional equation extends $\Gamma$ meromorphically to $\mathbb C$ with simple poles at $0,-1,-2,\dots$.
\begin{proof}
For any $N$, define $\Gamma(s)=\Gamma(s+N)/(s(s+1)\cdots(s+N-1))$ where the numerator integral is holomorphic for $\Re(s+N)>0$. These local formulas agree on overlaps by the functional equation.
\end{proof}
\end{theorem}

\begin{theorem}[Residue formula]\label{thm:iv19-03}
At $s=-n$, $n\ge0$, $\operatorname{Res}(\Gamma,-n)=(-1)^n/n!$.
\begin{proof}
Start from $\operatorname{Res}(\Gamma,0)=1$ using $\Gamma(s+1)=s\Gamma(s)$ and propagate residues recursively through the functional equation.
\end{proof}
\end{theorem}

\subsection*{Proofs}

\begin{exercise}[Functional equation]\label{exr:iv19-expand-06}
Prove \(\Gamma(s+1)=s\Gamma(s)\) for \(\Re s>0\).
\end{exercise}
\begin{hint}
Integrate by parts and check both endpoint terms.
\end{hint}
\begin{solution}
Take \(u=t^s\) and \(dv=e^{-t}dt\). The boundary terms vanish, leaving \(s\int_0^\infty t^{s-1}e^{-t}dt\).
\end{solution}

\begin{exercise}[Holomorphy]\label{exr:iv19-expand-07}
Explain why the Euler integral defines a holomorphic function on \(\Re s>0\).
\end{exercise}
\begin{hint}
Work on a compact vertical strip inside the half plane.
\end{hint}
\begin{solution}
On compact subsets, \(t^{s-1}e^{-t}\) and its parameter derivatives are dominated by integrable functions near zero and infinity, so differentiation under the integral sign is valid.
\end{solution}

\begin{exercise}[Pole locations]\label{exr:iv19-expand-08}
Prove that the functional equation gives only simple poles at the nonpositive integers.
\end{exercise}
\begin{hint}
Shift the argument into the right half plane.
\end{hint}
\begin{solution}
For sufficiently large \(N\), write \(\Gamma(s)=\Gamma(s+N)/(s(s+1)\cdots(s+N-1))\). The numerator is holomorphic and nonzero at the relevant points, while each denominator factor occurs once.
\end{solution}

\begin{exercise}[Residue recursion]\label{exr:iv19-expand-09}
Prove \(\operatorname{Res}_{s=-n}\Gamma(s)=(-1)^n/n!\).
\end{exercise}
\begin{hint}
Compare residues in \(\Gamma(s+1)=s\Gamma(s)\).
\end{hint}
\begin{solution}
Starting with residue \(1\) at zero, each step to the left divides by the value of \(s\) at the new pole, producing alternating signs and factorial denominators.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Convergence at zero]\label{exr:iv19-expand-10}
Does the Euler integral converge for \(\Re s=0\)?
\end{exercise}
\begin{hint}
Inspect the magnitude of \(t^{s-1}\) near zero.
\end{hint}
\begin{solution}
In general no. Its magnitude behaves like \(t^{-1}\), whose integral diverges logarithmically near zero.
\end{solution}

\begin{exercise}[Entire or meromorphic]\label{exr:iv19-expand-11}
Is \(\Gamma\) entire after continuation?
\end{exercise}
\begin{hint}
Recall the values forced by the functional equation at nonpositive integers.
\end{hint}
\begin{solution}
No. It is meromorphic with simple poles at \(0,-1,-2,\ldots\).
\end{solution}

\begin{exercise}[Zeros of Gamma]\label{exr:iv19-expand-12}
Can the functional equation alone permit a zero of \(\Gamma\) in the right half plane?
\end{exercise}
\begin{hint}
Use the Euler integral only for positive real arguments, then think about the reciprocal Gamma function.
\end{hint}
\begin{solution}
The integral is positive on the positive real axis, but excluding complex zeros requires additional theory. One should not infer zero freeness on the whole half plane from positivity on one line.
\end{solution}
