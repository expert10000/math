\chapter{The Gamma Function}
\label{ch:iv19}

The Gamma function extends factorials to the complex plane through an integral, functional equation, and analytic continuation. It is the prototype of a special function whose poles and growth encode arithmetic structure.

\section{Core structural results}

\begin{theorem}[Gamma functional equation]\label{thm:iv19-01}
For $\Re s>0$, $\Gamma(s+1)=s\Gamma(s)$.
\begin{proof}
Integrate $\int_0^\infty t^s e^{-t}dt$ by parts. Boundary terms vanish because $t^s e^{-t}$ decays at infinity and tends to zero at the origin under the real-part condition.
\end{proof}
\end{theorem}

\begin{theorem}[Meromorphic continuation]\label{thm:iv19-02}
The functional equation extends $\Gamma$ meromorphically to $\mathbb C$ with simple poles at $0,-1,-2,\dots$.
\begin{proof}
For any $N$, define $\Gamma(s)=\Gamma(s+N)/(s(s+1)\cdots(s+N-1))$ where the numerator integral is holomorphic for $\Re(s+N)>0$. These local formulas agree on overlaps by the functional equation.
\end{proof}
\end{theorem}

\begin{theorem}[Residue formula]\label{thm:iv19-03}
At $s=-n$, $n\ge0$, $\operatorname{Res}(\Gamma,-n)=(-1)^n/n!$.
\begin{proof}
Start from $\operatorname{Res}(\Gamma,0)=1$ using $\Gamma(s+1)=s\Gamma(s)$ and propagate residues recursively through the functional equation.
\end{proof}
\end{theorem}
