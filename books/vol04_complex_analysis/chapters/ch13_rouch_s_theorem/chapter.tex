\chapter{Rouché's Theorem}
\label{ch:iv13}

Rouché's theorem compares two holomorphic functions on a boundary and concludes that they have the same number of interior zeros. It is one of the most efficient tools for root counting and perturbative existence.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify Rouché inequalities on a contour and choose the dominant term strategically;
\item use Rouché's theorem to count zeros in disks, annuli, or half-planes after suitable changes of variable;
\item compare candidate functions on the boundary rather than throughout the interior;
\item track multiplicities in the zero count;
\item combine Rouché with the argument principle when direct estimates are awkward;
\item construct examples where a non-strict boundary inequality is insufficient for the theorem;
\end{itemize}

\section{Core structural results}

\begin{theorem}[Rouché theorem]\label{thm:iv13-01}
If $f,g$ are holomorphic inside and on a simple closed contour $\Gamma$ and $|g|<|f|$ on $\Gamma$, then $f$ and $f+g$ have the same number of zeros inside, counted with multiplicity.
\begin{proof}
For $0\le t\le1$, $f+tg$ is nonzero on $\Gamma$ because $|tg|<|f|$. Hence the winding number of $(f+tg)(\Gamma)$ about zero is constant in $t$. Apply the argument principle at $t=0$ and $t=1$.
\end{proof}
\end{theorem}

\begin{theorem}[Polynomial root localization]\label{thm:iv13-02}
If on $|z|=R$ one term of a polynomial strictly dominates the sum of all others, then the polynomial has the same number of zeros in $|z|<R$ as that dominant term.
\begin{proof}
Apply Rouché with the dominant term as $f$ and the sum of the remaining terms as $g$.
\end{proof}
\end{theorem}

\begin{theorem}[Perturbative persistence]\label{thm:iv13-03}
If $f$ has no zeros on $\Gamma$ and $\|g\|_{\Gamma}<\min_{\Gamma}|f|$, then $f+g$ has the same interior zero count as $f$.
\begin{proof}
The displayed inequality is precisely the strict Rouché hypothesis.
\end{proof}
\end{theorem}
