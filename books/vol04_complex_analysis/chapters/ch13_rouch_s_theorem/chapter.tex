\chapter{Rouché's Theorem}
\label{ch:iv13}

Rouché's theorem compares two holomorphic functions on a boundary and concludes that they have the same number of interior zeros. It is one of the most efficient tools for root counting and perturbative existence.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify Rouché inequalities on a contour and choose the dominant term strategically;
\item use Rouché's theorem to count zeros in disks, annuli, or half-planes after suitable changes of variable;
\item compare candidate functions on the boundary rather than throughout the interior;
\item track multiplicities in the zero count;
\item combine Rouché with the argument principle when direct estimates are awkward;
\item construct examples where a non-strict boundary inequality is insufficient for the theorem;
\end{itemize}

The hypothesis $|g|<|f|$ on a contour ensures that $f$ and $f+g$ never cross zero along the boundary.

\end{example}

\section{Core structural results}

\begin{theorem}[Rouché theorem]\label{thm:iv13-01}
If $f,g$ are holomorphic inside and on a simple closed contour $\Gamma$ and $|g|<|f|$ on $\Gamma$, then $f$ and $f+g$ have the same number of zeros inside, counted with multiplicity.
\begin{proof}
For $0\le t\le1$, $f+tg$ is nonzero on $\Gamma$ because $|tg|<|f|$. Hence the winding number of $(f+tg)(\Gamma)$ about zero is constant in $t$. Apply the argument principle at $t=0$ and $t=1$.
\end{proof}
\end{theorem}

\begin{theorem}[Polynomial root localization]\label{thm:iv13-02}
If on $|z|=R$ one term of a polynomial strictly dominates the sum of all others, then the polynomial has the same number of zeros in $|z|<R$ as that dominant term.
\begin{proof}
Apply Rouché with the dominant term as $f$ and the sum of the remaining terms as $g$.
\end{proof}
\end{theorem}

\begin{theorem}[Perturbative persistence]\label{thm:iv13-03}
If $f$ has no zeros on $\Gamma$ and $\|g\|_{\Gamma}<\min_{\Gamma}|f|$, then $f+g$ has the same interior zero count as $f$.
\begin{proof}
The displayed inequality is precisely the strict Rouché hypothesis.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Dominant monomial]\label{exr:iv13-expand-01}
How many zeros does \(z^7+2\) have in \(|z|<2\)?
\end{exercise}
\begin{hint}
Compare with \(z^7\) on \(|z|=2\).
\end{hint}
\begin{solution}
Since \(2<2^7\), Rouché gives seven zeros.
\end{solution}

\begin{exercise}[Small disk]\label{exr:iv13-expand-02}
How many zeros does \(z^4+5z+1\) have in \(|z|<1\)?
\end{exercise}
\begin{hint}
Compare with \(5z\).
\end{hint}
\begin{solution}
On the unit circle, \(|z^4+1|\le2<5\), so there is one zero.
\end{solution}

\begin{exercise}[Constant dominance]\label{exr:iv13-expand-03}
Show that \(2+z+z^2\) has no zeros in \(|z|<1/2\).
\end{exercise}
\begin{hint}
Compare with the constant two.
\end{hint}
\begin{solution}
On \(|z|=1/2\), \(|z+z^2|\le3/4<2\). Hence the polynomial has the same zero count as the nonzero constant, namely zero.
\end{solution}

\begin{exercise}[Annular subtraction]\label{exr:iv13-expand-04}
A polynomial has two zeros in \(|z|<1\) and six in \(|z|<3\). How many lie in \(1<|z|<3\)?
\end{exercise}
\begin{hint}
Subtract the counts.
\end{hint}
\begin{solution}
There are four zeros in the annulus.
\end{solution}

\begin{exercise}[Multiplicity]\label{exr:iv13-expand-05}
If Rouché compares a function to \((z-a)^4\), how many zeros are certified inside the contour?
\end{exercise}
\begin{hint}
Zeros are counted with multiplicity.
\end{hint}
\begin{solution}
Exactly four zeros, counted with multiplicity.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Rouché from argument principle]\label{exr:iv13-expand-06}
Prove Rouché's theorem using the family \(f_t=f+t g\) when \(|g|<|f|\) on the contour.
\end{exercise}
\begin{hint}
Show no \(f_t\) vanishes on the boundary.
\end{hint}
\begin{solution}
The inequality gives \(|f_t|\ge|f|-t|g|>0\). The boundary winding number of \(f_t\) about zero is integer valued and continuous in \(t\), hence constant. The argument principle gives equal zero counts at \(t=0\) and \(t=1\).
\end{solution}

\begin{exercise}[Symmetric version]\label{exr:iv13-expand-07}
Show that if \(|f-g|<|f|\) on a contour, then \(f\) and \(g\) have the same number of zeros inside.
\end{exercise}
\begin{hint}
Apply Rouché to \(f\) and perturbation \(g-f\).
\end{hint}
\begin{solution}
The hypothesis is precisely \(|g-f|<|f|\), so \(f+(g-f)=g\) has the same zero count as \(f\).
\end{solution}

\begin{exercise}[Local stability]\label{exr:iv13-expand-08}
Prove that a sufficiently small uniform perturbation preserves the number of zeros in a disk whose boundary contains no zeros.
\end{exercise}
\begin{hint}
The minimum of \(|f|\) on the compact boundary is positive.
\end{hint}
\begin{solution}
Let \(m=\min_C|f|>0\). Any perturbation \(h\) with \(\max_C|h|<m\) satisfies Rouché, so \(f+h\) has the same number of zeros.
\end{solution}

\begin{exercise}[FTA by Rouché]\label{exr:iv13-expand-09}
Use Rouché to prove that a degree \(n\) polynomial has \(n\) zeros counted with multiplicity.
\end{exercise}
\begin{hint}
On a sufficiently large circle, the leading term dominates all lower terms.
\end{hint}
\begin{solution}
For large \(R\), \(|a_n z^n|>\sum_{k<n}|a_k|R^k\) on \(|z|=R\). Thus the polynomial has the same zero count as \(a_n z^n\), namely \(n\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Non-strict inequality]\label{exr:iv13-expand-10}
Why is \(|g|\le|f|\) not the standard sufficient hypothesis in Rouché's theorem?
\end{exercise}
\begin{hint}
Equality can allow the homotopy to pass through zero on the boundary.
\end{hint}
\begin{solution}
The strict inequality guarantees boundary nonvanishing for the entire deformation. Without it, additional analysis is needed and the conclusion can fail.
\end{solution}

\begin{exercise}[Zero on boundary]\label{exr:iv13-expand-11}
Why must the comparison be arranged so the relevant functions have no boundary zeros?
\end{exercise}
\begin{hint}
The zero count can jump when a zero crosses the contour.
\end{hint}
\begin{solution}
A boundary zero makes the argument-principle winding undefined and destroys the homotopy certificate used in the proof.
\end{solution}

\begin{exercise}[Poor comparison]\label{exr:iv13-expand-12}
If neither term dominates on the chosen circle, does Rouché imply that the zero counts differ?
\end{exercise}
\begin{hint}
Failure of a sufficient condition is not a negative conclusion.
\end{hint}
\begin{solution}
No. It only means that this comparison and contour do not certify the count. A different comparison or contour may work.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Root localization]\label{exr:iv13-expand-13}
How can two Rouché counts localize polynomial roots to an annulus?
\end{exercise}
\begin{hint}
Count inside an inner and an outer circle.
\end{hint}
\begin{solution}
Subtracting the certified counts gives exactly how many roots lie between the circles, without solving the polynomial.
\end{solution}

\begin{exercise}[Perturbed characteristic equation]\label{exr:iv13-expand-14}
A characteristic polynomial \(p\) has a known zero count in a region. How can Rouché certify that modeling error \(h\) does not change it?
\end{exercise}
\begin{hint}
Bound \(|h|\) by the minimum boundary magnitude of \(p\).
\end{hint}
\begin{solution}
If \(\max_C|h|<\min_C|p|\), then \(p+h\) and \(p\) have identical zero counts inside the boundary.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Two-scale count]\label{exr:iv13-expand-15}
For \(z^6+4z^2+1\), find useful circles on which different terms dominate and explain how to infer an annular root count.
\end{exercise}
\begin{hint}
Try a small circle where the constant dominates and a larger circle where \(z^6\) dominates.
\end{hint}
\begin{solution}
Choose radii satisfying \(4R^2+R^6<1\) for the inner disk and \(4R^2+1<R^6\) for the outer disk. Rouché then gives zero inner roots and six outer roots, so all six lie in the intervening annulus.
\end{solution}

\begin{exercise}[Comparison with a factor]\label{exr:iv13-expand-16}
Suppose \(f(z)=p(z)+h(z)\) and \(p\) has a multiple zero. Does Rouché preserve the multiplicity count under a small boundary perturbation?
\end{exercise}
\begin{hint}
Rouché counts zeros with multiplicity rather than tracking individual labels.
\end{hint}
\begin{solution}
Yes. It preserves the total number counted with multiplicity inside the contour, even though a multiple zero may split into several nearby simple zeros.
\end{solution}
