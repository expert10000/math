\chapter{Branched Coverings}
\label{ch:iv24}

Branched coverings relax ordinary covering behavior at isolated ramification points. Locally they are modeled by power maps, and the ramification indices encode how sheets merge.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item identify branch points and ramification indices of elementary holomorphic maps;
\item write local models in the form \(w=z^e\) after suitable coordinates;
\item compute branching data for polynomial and rational maps;
\item distinguish ramification points upstairs from branch values downstairs;
\item apply degree counting with multiplicities in branched coverings;
\item relate local ramification behavior to global topology in examples;
\end{itemize}

\section{Branched covering}
Away from a discrete branch set, the map is an ordinary covering; over branch points sheets merge.

\begin{example}[Local ramification index]\label{ex:iv24-expand-01}
Consider \(p(w)=w^3\) near \(w=0\). In the local coordinate \(w\), the map already has the normal form
\[
z=w^3.
\]
The ramification index is \(e=3\). A small nonzero value of \(z\) has three nearby preimages, while the fiber over zero contains one point with multiplicity three. The derivative \(3w^2\) vanishes at the ramification point.
\end{example}

\section{Local power model}
For a nonconstant holomorphic map between Riemann surfaces, suitable local coordinates put the map in the form $w=z^e$ at every point; ramification means $e>1$.

\section{Ramification index}
The integer $e$ measures local multiplicity; $e>1$ means ramification.

\section{Branch values}
Images of ramification points are branch values in the base.

% BEGIN VOL04-EXPANSION IV24-example-02
\begin{example}[Branching of a quadratic polynomial]\label{ex:iv24-expand-02}
View \(p(w)=w^2-1\) as a map of Riemann spheres. The finite critical point is \(w=0\), with branch value \(-1\). Infinity is also ramified because in coordinates \(u=1/w\) and \(v=1/z\),
\[
v=\frac{u^2}{1-u^2}=u^2+O(u^4).
\]
Thus the degree-two map has two ramification points, each of index two.
\end{example}
% END VOL04-EXPANSION IV24-example-02
\section{Square-root surface}
The surface $w^2=z$ is a two-sheeted cover of the sphere branched at zero and infinity.

\section{Polynomial maps}
A polynomial viewed on the sphere gives a branched covering whose critical points are ramification points.

\section{Monodromy around branch values}
Loops around branch values permute sheets nontrivially.

% BEGIN VOL04-EXPANSION IV24-example-03
\begin{example}[Genus from Riemann--Hurwitz]\label{ex:iv24-expand-03}
For a degree-two map \(X\to\widehat{\mathbb C}\) with \(r\) simple branch points, Riemann--Hurwitz gives
\[
2-2g(X)=2\cdot2-r.
\]
Hence
\[
g(X)=\frac{r-2}{2}.
\]
A double cover branched at four points has genus one, while branching at six points gives genus two.
\end{example}

\section{Degree}
Every fiber of a nonconstant holomorphic map between connected compact Riemann surfaces has the same total multiplicity. Away from branch values a degree-$d$ map has $d$ distinct points in the fiber; over branch values some points merge, but their local multiplicities still sum to $d$.

\section{Riemann--Hurwitz preview}
Total ramification contributes to the genus relation between compact surfaces.

\section{Core structural results}

\begin{theorem}[Local ramification model]\label{thm:iv24-01}
A nonconstant holomorphic map between Riemann surfaces has local coordinates in which it is $z\mapsto z^e$ for a unique integer $e\ge1$.
\begin{proof}
Expand the map in a local coordinate around the source point, factor out the first nonzero term of order $e$, and absorb the nonvanishing holomorphic factor into a new local coordinate using a holomorphic $e$th root.
\end{proof}
\end{theorem}

\begin{theorem}[Critical-point criterion]\label{thm:iv24-02}
A point is ramified iff the derivative of a local coordinate representation vanishes there.
\begin{proof}
In the local model $z^e$, the derivative at zero is nonzero exactly when $e=1$.
\end{proof}
\end{theorem}

\begin{theorem}[Constant degree with multiplicity]\label{thm:iv24-03}
Let $f:X\to Y$ be a nonconstant holomorphic map between connected compact Riemann surfaces. Then there is an integer $d\ge1$ such that for every $y\in Y$,
\[
\sum_{p\in f^{-1}(y)} e_p=d,
\]
where $e_p$ is the local ramification index of $f$ at $p$. In particular, away from branch values the fiber consists of exactly $d$ distinct points.
\begin{proof}
Compactness makes $f$ proper, so every fiber is finite. Around each $p\in f^{-1}(y)$, the local model is $z\mapsto z^{e_p}$; hence every sufficiently close regular value has exactly $e_p$ preimages in that neighborhood. Summing over the fiber shows that the total multiplicity is locally constant as a function of $y$. Since $Y$ is connected, this integer is constant; call it $d$.
\end{proof}
\end{theorem}

\begin{problem}[Constant degree with multiplicity]\label{prob:iv24-problem-11}
Let $f:X\to Y$ be a nonconstant holomorphic map between connected compact Riemann surfaces. Prove that the sum of local ramification indices over every fiber is one constant integer $d$, and explain what happens over a branch value.
\end{problem}
\begin{solution}
Compactness makes $f$ proper, hence fibers are finite. The local power model $z\mapsto z^{e_p}$ shows that near a fiber point $p$, nearby regular values have $e_p$ preimages. Summing these local counts proves local constancy of the total fiber multiplicity, and connectedness of $Y$ makes it globally constant. Over a branch value there may be fewer distinct points, but their ramification indices still sum to the same degree $d$.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Critical value confusion]\label{exr:iv24-expand-10}
Is every point over a branch value necessarily ramified?
\end{exercise}
\begin{hint}
A branch value can have several preimages.
\end{hint}
\begin{solution}
No. At least one preimage is ramified, but other points in the same fiber can be unramified.
\end{solution}

\begin{exercise}[Derivative zero means singular surface]\label{exr:iv24-expand-11}
Does a vanishing derivative of the projection imply the source Riemann surface is singular?
\end{exercise}
\begin{hint}
Distinguish the map from the surface.
\end{hint}
\begin{solution}
No. It means the map is ramified. The source can be a perfectly smooth Riemann surface.
\end{solution}

\begin{exercise}[Degree from distinct points only]\label{exr:iv24-expand-12}
Can degree be computed by counting only distinct points in a branch fiber?
\end{exercise}
\begin{hint}
Remember multiplicities.
\end{hint}
\begin{solution}
No. At branch fibers several sheets merge, so multiplicities must be counted to recover the degree.
\end{solution}
