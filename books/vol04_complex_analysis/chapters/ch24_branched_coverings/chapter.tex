\chapter{Branched Coverings}
\label{ch:iv24}

Branched coverings relax ordinary covering behavior at isolated ramification points. Locally they are modeled by power maps, and the ramification indices encode how sheets merge.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item identify branch points and ramification indices of elementary holomorphic maps;
\item write local models in the form \(w=z^e\) after suitable coordinates;
\item compute branching data for polynomial and rational maps;
\item distinguish ramification points upstairs from branch values downstairs;
\item apply degree counting with multiplicities in branched coverings;
\item relate local ramification behavior to global topology in examples;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{local holomorphic structure}\;\longrightarrow\;\text{integral or series representation}\;\longrightarrow\;\text{global consequence}.}
\]

\section{Branched covering}
Away from a discrete branch set, the map is an ordinary covering; over branch points sheets merge.

\begin{example}[Local ramification index]\label{ex:iv24-expand-01}
Consider \(p(w)=w^3\) near \(w=0\). In the local coordinate \(w\), the map already has the normal form
\[
z=w^3.
\]
The ramification index is \(e=3\). A small nonzero value of \(z\) has three nearby preimages, while the fiber over zero contains one point with multiplicity three. The derivative \(3w^2\) vanishes at the ramification point.
\end{example}

Near a ramification point, suitable coordinates often put the map in form $w=z^e$.

\section{Ramification index}
The integer $e$ measures local multiplicity; $e>1$ means ramification.

\section{Branch values}
Images of ramification points are branch values in the base.

% BEGIN VOL04-EXPANSION IV24-example-02
\begin{example}[Branching of a quadratic polynomial]\label{ex:iv24-expand-02}
View \(p(w)=w^2-1\) as a map of Riemann spheres. The finite critical point is \(w=0\), with branch value \(-1\). Infinity is also ramified because in coordinates \(u=1/w\) and \(v=1/z\),
\[
v=\frac{u^2}{1-u^2}=u^2+O(u^4).
\]
Thus the degree-two map has two ramification points, each of index two.
\end{example}
% END VOL04-EXPANSION IV24-example-02
\section{Square-root surface}
The surface $w^2=z$ is a two-sheeted cover of the sphere branched at zero and infinity.

\section{Polynomial maps}
A polynomial viewed on the sphere gives a branched covering whose critical points are ramification points.

\section{Monodromy around branch values}
Loops around branch values permute sheets nontrivially.

% BEGIN VOL04-EXPANSION IV24-example-03
\begin{example}[Genus from Riemann--Hurwitz]\label{ex:iv24-expand-03}
For a degree-two map \(X\to\widehat{\mathbb C}\) with \(r\) simple branch points, Riemann--Hurwitz gives
\[
2-2g(X)=2\cdot2-r.
\]
Hence
\[
g(X)=\frac{r-2}{2}.
\]
A double cover branched at four points has genus one, while branching at six points gives genus two.
\end{example}
% END VOL04-EXPANSION IV24-example-03
\section{Degree}
Away from branch values, every fiber has the same number of points counted with multiplicity.

\section{Riemann--Hurwitz preview}
Total ramification contributes to the genus relation between compact surfaces.

\section{Core structural results}

\begin{theorem}[Local ramification model]\label{thm:iv24-01}
A nonconstant holomorphic map between Riemann surfaces has local coordinates in which it is $z\mapsto z^e$ for a unique integer $e\ge1$.
\begin{proof}
Expand the map in a local coordinate around the source point, factor out the first nonzero term of order $e$, and absorb the nonvanishing holomorphic factor into a new local coordinate using a holomorphic $e$th root.
\end{proof}
\end{theorem}

\begin{theorem}[Critical-point criterion]\label{thm:iv24-02}
A point is ramified iff the derivative of a local coordinate representation vanishes there.
\begin{proof}
In the local model $z^e$, the derivative at zero is nonzero exactly when $e=1$.
\end{proof}
\end{theorem}

\begin{theorem}[Constant generic degree]\label{thm:iv24-03}
For a proper finite branched covering, the sum of ramification multiplicities in a fiber is constant away from degenerations and equals the degree.
\begin{proof}
Locally, zeros of the equation $p(x)=y$ are counted with multiplicity; the argument principle makes this count stable as $y$ moves without crossing branch values.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Ramification index]\label{exr:iv24-expand-01}
Find the ramification index of \(w\mapsto w^5\) at zero.
\end{exercise}
\begin{hint}
Read the exponent in the local power model.
\end{hint}
\begin{solution}
The index is \(5\).
\end{solution}

\begin{exercise}[Derivative criterion]\label{exr:iv24-expand-02}
Is \(w=0\) ramified for \(p(w)=w^3+w\)?
\end{exercise}
\begin{hint}
Compute \(p'(0)\).
\end{hint}
\begin{solution}
No. Since \(p'(0)=1\), the local index is one.
\end{solution}

\begin{exercise}[Critical points]\label{exr:iv24-expand-03}
Find the finite critical points of \(p(w)=w^3-3w\).
\end{exercise}
\begin{hint}
Solve \(p'(w)=0\).
\end{hint}
\begin{solution}
The critical points are \(w=\pm1\).
\end{solution}

\begin{exercise}[Branch values]\label{exr:iv24-expand-04}
Find the corresponding finite branch values for \(w^3-3w\).
\end{exercise}
\begin{hint}
Evaluate the polynomial at the critical points.
\end{hint}
\begin{solution}
The branch values are \(-2\) at \(w=1\) and \(2\) at \(w=-1\).
\end{solution}

\begin{exercise}[Double-cover genus]\label{exr:iv24-expand-05}
Find the genus of a double cover of the sphere with eight simple branch points.
\end{exercise}
\begin{hint}
Use \(2-2g=4-r\).
\end{hint}
\begin{solution}
With \(r=8\), one gets \(g=3\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Local power model]\label{exr:iv24-expand-06}
Outline why a nonconstant holomorphic map has local form \(z\mapsto z^e\).
\end{exercise}
\begin{hint}
Factor the first nonzero Taylor term and absorb the unit.
\end{hint}
\begin{solution}
Write the map as \(z^e h(z)\) with \(h(0)\ne0\). A local holomorphic \(e\)-th root of \(h\) produces a new coordinate in which the map is exactly a power.
\end{solution}

\begin{exercise}[Derivative criterion]\label{exr:iv24-expand-07}
Prove that ramification is equivalent to vanishing derivative in local coordinates.
\end{exercise}
\begin{hint}
Differentiate the power normal form.
\end{hint}
\begin{solution}
The derivative of \(z^e\) at zero is nonzero exactly for \(e=1\).
\end{solution}

\begin{exercise}[Generic degree]\label{exr:iv24-expand-08}
Explain why the number of preimages counted with multiplicity is locally constant away from branch values.
\end{exercise}
\begin{hint}
Use the argument principle on the equation \(p(x)-y=0\).
\end{hint}
\begin{solution}
As \(y\) varies without crossing a critical value, the boundary has no zeros and the argument-principle count cannot jump.
\end{solution}

\begin{exercise}[Even number of simple branch points]\label{exr:iv24-expand-09}
Use Riemann--Hurwitz to show that a double cover of the sphere has an even number of simple branch points.
\end{exercise}
\begin{hint}
Solve the formula for the branch count.
\end{hint}
\begin{solution}
The formula gives \(r=2g+2\), which is even.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Critical value confusion]\label{exr:iv24-expand-10}
Is every point over a branch value necessarily ramified?
\end{exercise}
\begin{hint}
A branch value can have several preimages.
\end{hint}
\begin{solution}
No. At least one preimage is ramified, but other points in the same fiber can be unramified.
\end{solution}

\begin{exercise}[Derivative zero means singular surface]\label{exr:iv24-expand-11}
Does a vanishing derivative of the projection imply the source Riemann surface is singular?
\end{exercise}
\begin{hint}
Distinguish the map from the surface.
\end{hint}
\begin{solution}
No. It means the map is ramified. The source can be a perfectly smooth Riemann surface.
\end{solution}

\begin{exercise}[Degree from distinct points only]\label{exr:iv24-expand-12}
Can degree be computed by counting only distinct points in a branch fiber?
\end{exercise}
\begin{hint}
Remember multiplicities.
\end{hint}
\begin{solution}
No. At branch fibers several sheets merge, so multiplicities must be counted to recover the degree.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Hyperelliptic curve]\label{exr:iv24-expand-13}
For \(y^2=\prod_{j=1}^{2g+2}(x-a_j)\) with distinct roots, explain the natural map to the sphere.
\end{exercise}
\begin{hint}
Project to the \(x\)-coordinate.
\end{hint}
\begin{solution}
The projection is a degree-two branched cover, with simple branching over the roots and genus \(g\) by Riemann--Hurwitz.
\end{solution}

\begin{exercise}[Polynomial dynamics]\label{exr:iv24-expand-14}
Why are critical values central when studying a polynomial as a branched cover?
\end{exercise}
\begin{hint}
They are exactly where local inverse branches fail to remain separate.
\end{hint}
\begin{solution}
Away from critical values the map is a genuine covering, while crossing or looping around critical values changes the organization and monodromy of inverse branches.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Infinity ramification]\label{exr:iv24-expand-15}
Determine the ramification index of a degree \(d\) polynomial at infinity as a map of spheres.
\end{exercise}
\begin{hint}
Use coordinates \(u=1/w\) and \(v=1/p(w)\).
\end{hint}
\begin{solution}
The leading term gives \(v\sim c^{-1}u^d\), so infinity has ramification index \(d\).
\end{solution}

\begin{exercise}[Cubic sphere map]\label{exr:iv24-expand-16}
Check Riemann--Hurwitz for a generic cubic polynomial map of the sphere.
\end{exercise}
\begin{hint}
There are two finite simple critical points and ramification at infinity.
\end{hint}
\begin{solution}
The two finite points contribute one each and infinity contributes two, for total ramification four. Then \(2=3\cdot2-4\), as required for a sphere source.
\end{solution}
