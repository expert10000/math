\chapter{Cauchy's Integral Formula}
\label{ch:iv06}

Cauchy's integral formula reconstructs a holomorphic function from boundary values. Its differentiated forms yield analyticity, derivative estimates, Liouville's theorem, and the mean-value property.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item apply Cauchy's integral formula to evaluate values and derivatives of holomorphic functions;
\item derive Cauchy estimates from contour bounds;
\item use Cauchy's formula to prove infinite differentiability and analyticity;
\item compute contour integrals containing powers of \((z-a)^{-1}\) by matching the correct derivative formula;
\item choose a circle contained in the domain to optimize an estimate;
\item use Cauchy estimates to prove Liouville-type or derivative-vanishing consequences;
\end{itemize}

\end{example}

\section{Core structural results}

\begin{theorem}[Cauchy integral formula]\label{thm:iv06-01}
If $f$ is holomorphic on and inside a positively oriented simple closed contour $\Gamma$ and $z$ lies inside, then $f(z)=\frac1{2\pi i}\int_\Gamma\frac{f(\zeta)}{\zeta-z}d\zeta$.
\begin{proof}
Subtract $f(z)$ from the numerator so the apparent singularity becomes removable, apply Cauchy's theorem to the regular part, and compute the integral of $(\zeta-z)^{-1}$ around the contour as $2\pi i$.
\end{proof}
\end{theorem}

\begin{theorem}[Cauchy derivative formula]\label{thm:iv06-02}
$f^{(n)}(z)=\frac{n!}{2\pi i}\int_\Gamma\frac{f(\zeta)}{(\zeta-z)^{n+1}}d\zeta$.
\begin{proof}
Differentiate the Cauchy formula with respect to $z$; uniform separation of $z$ from the contour justifies differentiation under the integral.
\end{proof}
\end{theorem}

\begin{theorem}[Cauchy estimate]\label{thm:iv06-03}
If $|f|\le M$ on $|\zeta-z_0|=R$, then $|f^{(n)}(z_0)|\le n!M/R^n$.
\begin{proof}
Apply the derivative formula on the circle and use the ML inequality with length $2\pi R$ and denominator magnitude $R^{n+1}$.
\end{proof}
\end{theorem}

\begin{theorem}[Liouville theorem]\label{thm:iv06-04}
Every bounded entire function is constant.
\begin{proof}
Apply the Cauchy estimate for the first derivative on circles of radius $R$ and let $R\to\infty$, forcing $f'=0$.
\end{proof}
\end{theorem}

\end{hint}

\subsection*{Proofs}

\begin{exercise}[Higher derivative formula]\label{exr:iv06-expand-06}
Derive the higher derivative Cauchy formula from the basic formula.
\end{exercise}
\begin{hint}
Differentiate the kernel with respect to the interior point.
\end{hint}
\begin{solution}
Uniform control on a smaller disk justifies differentiation under the contour integral. Repeating gives \(f^{(n)}(a)=n!/(2\pi i)\int f(z)/(z-a)^{n+1}dz\).
\end{solution}

\begin{exercise}[Cauchy estimates]\label{exr:iv06-expand-07}
Prove the standard Cauchy derivative estimate.
\end{exercise}
\begin{hint}
Take moduli in the higher derivative formula on a circle of radius \(R\).
\end{hint}
\begin{solution}
The kernel contributes \(R^{-(n+1)}\), the contour length contributes \(2\pi R\), and the prefactor gives \(|f^{(n)}(a)|\le n!M/R^n\).
\end{solution}

\begin{exercise}[Liouville theorem]\label{exr:iv06-expand-08}
Use Cauchy estimates to prove that every bounded entire function is constant.
\end{exercise}
\begin{hint}
Let the radius tend to infinity in the first derivative estimate.
\end{hint}
\begin{solution}
If \(|f|\le M\) globally, then \(|f'(a)|\le M/R\) for every \(R\). Letting \(R\to\infty\) gives \(f'(a)=0\) for every \(a\), so \(f\) is constant.
\end{solution}

\begin{exercise}[Mean value identity]\label{exr:iv06-expand-09}
Derive the mean value formula for a holomorphic function on a circle.
\end{exercise}
\begin{hint}
Parametrize the Cauchy formula on \(z=a+Re^{it}\).
\end{hint}
\begin{solution}
The factors \(Re^{it}\) cancel, leaving \(f(a)=\frac1{2\pi}\int_0^{2\pi}f(a+Re^{it})dt\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Singularity inside]\label{exr:iv06-expand-10}
Can Cauchy's formula be applied with \(f(z)=1/z\) and center \(a=1\) on a circle enclosing zero?
\end{exercise}
\begin{hint}
The numerator function in the formula must be holomorphic throughout the interior.
\end{hint}
\begin{solution}
Not directly if the chosen circle encloses the pole at zero. One must isolate singularities or use later residue methods.
\end{solution}

\begin{exercise}[Point on contour]\label{exr:iv06-expand-11}
What fails if the evaluation point \(a\) lies on the contour in Cauchy's formula?
\end{exercise}
\begin{hint}
The kernel has a singularity on the path.
\end{hint}
\begin{solution}
The ordinary contour integral is not covered by the formula and may be undefined. The point must lie strictly inside the contour.
\end{solution}

\begin{exercise}[Wrong orientation]\label{exr:iv06-expand-12}
How does clockwise orientation change the Cauchy formula value?
\end{exercise}
\begin{hint}
Reverse the contour orientation.
\end{hint}
\begin{solution}
The integral changes sign, so the usual \(2\pi i f(a)\) becomes \(-2\pi i f(a)\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Coefficient recovery]\label{exr:iv06-expand-13}
Explain how Cauchy's formula recovers Taylor coefficients from boundary data.
\end{exercise}
\begin{hint}
Combine the higher derivative formula with \(a_n=f^{(n)}(a)/n!\).
\end{hint}
\begin{solution}
One gets \(a_n=(2\pi i)^{-1}\int f(z)/(z-a)^{n+1}dz\). Thus each local coefficient is encoded by the function on any surrounding contour inside the analytic domain.
\end{solution}

\begin{exercise}[Maximum growth constraint]\label{exr:iv06-expand-14}
Why do Cauchy estimates make very slow growth restrictive for entire functions?
\end{exercise}
\begin{hint}
Choose a large circle and compare growth with the power of its radius in the denominator.
\end{hint}
\begin{solution}
If the maximum modulus grows more slowly than \(R^n\), the estimate can force the \(n\)-th derivative to vanish. Repeating such bounds yields polynomial or constant rigidity results.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Fundamental theorem of algebra]\label{exr:iv06-expand-15}
Use Liouville's theorem to outline a proof that a nonconstant complex polynomial has a zero.
\end{exercise}
\begin{hint}
Assume the polynomial has no zeros and examine its reciprocal.
\end{hint}
\begin{solution}
If \(p\) has no zeros, \(1/p\) is entire. It tends to zero at infinity and is bounded on a large disk, so it is bounded everywhere. Liouville makes it constant, contradicting nonconstancy of \(p\).
\end{solution}

\begin{exercise}[All derivatives from one contour]\label{exr:iv06-expand-16}
Explain why knowing a holomorphic function on one circle determines every derivative at its center.
\end{exercise}
\begin{hint}
Use the family of higher derivative formulas.
\end{hint}
\begin{solution}
For every \(n\), the same boundary values enter \(f^{(n)}(a)=n!/(2\pi i)\int f(z)/(z-a)^{n+1}dz\). Thus one contour determines the full Taylor germ at the center.
\end{solution}
