\chapter{Residues and the Residue Theorem}
\label{ch:iv10}

Residues compress the local singular data relevant to contour integration into one Laurent coefficient. The residue theorem converts global contour integrals into finite sums of local contributions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute residues at simple and higher-order poles using the most efficient available formula;
\item apply the residue theorem after identifying all enclosed singularities and contour orientations;
\item compute residues at infinity and use global residue balance;
\item use logarithmic derivatives to count zeros and poles with multiplicity;
\item choose between Laurent expansion, limit, derivative, and partial-fraction methods for residue computation;
\item audit winding numbers and contour geometry before summing residue contributions;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{local holomorphic structure}\;\longrightarrow\;\text{integral or series representation}\;\longrightarrow\;\text{global consequence}.}
\]

\section{Residue definition}
The residue at an isolated singularity is the Laurent coefficient of $(z-z_0)^{-1}$.

% BEGIN VOL04-EXPANSION IV10-example-01
\begin{example}[Residue at a simple pole]\label{ex:iv10-expand-01}
For \(f(z)=e^z/(z-1)\),
\[
\operatorname{Res}_{z=1}f=\lim_{z\to1}(z-1)f(z)=e.
\]
Therefore a positively oriented contour enclosing \(1\) and no other singularities has integral \(2\pi i e\).
\end{example}
% END VOL04-EXPANSION IV10-example-01
\section{Simple poles}
For a simple pole of $g/h$ with $h(z_0)=0$ and $h'(z_0)\ne0$, the residue is $g(z_0)/h'(z_0)$.

\section{Higher-order poles}
Derivative formulas compute residues at poles of finite order.

\section{Residue theorem}
A contour integral equals $2\pi i$ times the sum of enclosed residues, weighted by winding number in the general form.

% BEGIN VOL04-EXPANSION IV10-example-02
\begin{example}[A second order pole]\label{ex:iv10-expand-02}
Consider \(f(z)=1/(z^2(z-1))\) at \(z=0\). Since the pole has order two,
\[
\operatorname{Res}_{z=0}f
=\left.\frac{d}{dz}\frac1{z-1}\right|_{z=0}=-1.
\]
The derivative formula avoids computing the full Laurent series.
\end{example}
% END VOL04-EXPANSION IV10-example-02
\section{Local-to-global principle}
Only the $-1$ Laurent terms contribute to closed contour integrals.

\section{Residue at infinity}
For meromorphic functions on the sphere, the residue at infinity balances the finite residues.

\section{Logarithmic derivative}
Residues of $f'/f$ record zero and pole multiplicities.

% BEGIN VOL04-EXPANSION IV10-example-03
\begin{example}[Summing local contributions]\label{ex:iv10-expand-03}
For
\[
f(z)=\frac{1}{(z-1)(z+1)},
\]
the residues at \(1\) and \(-1\) are \(1/2\) and \(-1/2\). A contour enclosing both has zero integral, while a contour enclosing only \(1\) has integral \(\pi i\). The residue theorem converts a global contour integral into a finite sum of local coefficients.
\end{example}

\section{Core structural results}

\begin{theorem}[Residue theorem]\label{thm:iv10-01}
If $f$ is holomorphic inside and on a positively oriented contour except for finitely many isolated singularities $a_k$, then $\int_\Gamma fdz=2\pi i\sum_k\operatorname{Res}(f,a_k)$ in the simple winding-one setting.
\begin{proof}
Remove small disks around the singularities, apply Cauchy's theorem to the remaining multiply connected region, and let the small circles shrink; each circle integral tends to $2\pi i$ times the Laurent $-1$ coefficient.
\end{proof}
\end{theorem}

\begin{theorem}[Simple-pole formula]\label{thm:iv10-02}
If $f=g/h$ with $g,h$ holomorphic, $h(a)=0$, and $h'(a)\ne0$, then $\operatorname{Res}(f,a)=g(a)/h'(a)$.
\begin{proof}
Factor $h(z)=(z-a)k(z)$ with $k(a)=h'(a)$; then $(z-a)f(z)=g(z)/k(z)$ and evaluate at $a$.
\end{proof}
\end{theorem}

\begin{theorem}[Higher-pole residue formula]\label{thm:iv10-03}
If $f$ has a pole of order $m$ at $a$, then $\operatorname{Res}(f,a)=\frac1{(m-1)!}\left[\frac{d^{m-1}}{dz^{m-1}}((z-a)^mf(z))\right]_{z=a}$.
\begin{proof}
The bracketed factor is holomorphic; its Taylor coefficient of degree $m-1$ is exactly the Laurent coefficient multiplying $(z-a)^{-1}$.
\end{proof}
\end{theorem}

\begin{theorem}[Global residue balance]\label{thm:iv10-04}
For a meromorphic function on the Riemann sphere, the sum of all residues including infinity is zero.
\begin{proof}
Integrate around a sufficiently large circle and compare the exterior coordinate $w=1/z$ with the finite residues inside; the residue at infinity is defined to cancel the large-circle contribution.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Simple rational residue]\label{exr:iv10-expand-01}
Find \(\operatorname{Res}_{z=2}1/(z(z-2))\).
\end{exercise}
\begin{hint}
Multiply by \(z-2\) and take the limit.
\end{hint}
\begin{solution}
The residue is \(1/2\).
\end{solution}

\begin{exercise}[Exponential residue]\label{exr:iv10-expand-02}
Find the residue of \(e^z/(z-i)\) at \(i\).
\end{exercise}
\begin{hint}
It is a simple pole.
\end{hint}
\begin{solution}
The residue is \(e^i\).
\end{solution}

\begin{exercise}[Second order residue]\label{exr:iv10-expand-03}
Find \(\operatorname{Res}_{z=0}e^z/z^2\).
\end{exercise}
\begin{hint}
Read the coefficient of \(z^{-1}\) or differentiate the numerator.
\end{hint}
\begin{solution}
Since \(e^z=1+z+\cdots\), the residue is \(1\).
\end{solution}

\begin{exercise}[Contour sum]\label{exr:iv10-expand-04}
A contour encloses simple poles with residues \(2\) and \(-3i\). Find the integral.
\end{exercise}
\begin{hint}
Multiply the residue sum by \(2\pi i\).
\end{hint}
\begin{solution}
The integral is \(2\pi i(2-3i)=6\pi+4\pi i\).
\end{solution}

\begin{exercise}[Residue from Laurent series]\label{exr:iv10-expand-05}
What is the residue at zero of \(4z^{-3}-7z^{-1}+2+z\)?
\end{exercise}
\begin{hint}
Select the coefficient of \(z^{-1}\).
\end{hint}
\begin{solution}
The residue is \(-7\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Residue theorem from Laurent series]\label{exr:iv10-expand-06}
Prove the residue theorem for finitely many isolated singularities inside a contour.
\end{exercise}
\begin{hint}
Excise small circles around the singularities and apply Cauchy's theorem to the remaining region.
\end{hint}
\begin{solution}
The outer integral equals the sum of the small circle integrals. Each small circle integral is \(2\pi i\) times the local \(z^{-1}\) Laurent coefficient, yielding the residue sum.
\end{solution}

\begin{exercise}[Simple pole formula]\label{exr:iv10-expand-07}
Prove \(\operatorname{Res}_a(g/h)=g(a)/h'(a)\) when \(h(a)=0\), \(h'(a)\ne0\), and \(g(a)\ne0\).
\end{exercise}
\begin{hint}
Factor the simple zero of \(h\).
\end{hint}
\begin{solution}
Write \(h(z)=(z-a)k(z)\) with \(k(a)=h'(a)\). Then \((z-a)g/h=g/k\), whose limit at \(a\) is \(g(a)/h'(a)\).
\end{solution}

\begin{exercise}[Higher pole formula]\label{exr:iv10-expand-08}
Derive the residue formula for a pole of order \(m\).
\end{exercise}
\begin{hint}
Multiply by \((z-a)^m\) and take the Taylor coefficient of order \(m-1\).
\end{hint}
\begin{solution}
If \(g=(z-a)^m f\) is holomorphic, the \((z-a)^{-1}\) term of \(f\) comes from the \((m-1)\)-st Taylor coefficient of \(g\), giving \(g^{(m-1)}(a)/(m-1)!\).
\end{solution}

\begin{exercise}[Residue of a derivative]\label{exr:iv10-expand-09}
Show that the residue of \(f'\) at an isolated singularity is zero.
\end{exercise}
\begin{hint}
Differentiate the Laurent series termwise.
\end{hint}
\begin{solution}
A derivative term has exponent \(n-1\) with coefficient \(na_n\). Exponent \(-1\) would require \(n=0\), whose coefficient is zero. Hence there is no \((z-a)^{-1}\) term.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Pole on contour]\label{exr:iv10-expand-10}
Why is the ordinary residue theorem not directly applicable when a pole lies on the contour?
\end{exercise}
\begin{hint}
The integrand is not defined continuously along the path.
\end{hint}
\begin{solution}
The contour integral is not an ordinary integral under the theorem's hypotheses. Indentation or principal value methods require separate arguments.
\end{solution}

\begin{exercise}[Orientation]\label{exr:iv10-expand-11}
What happens to the residue theorem for clockwise orientation?
\end{exercise}
\begin{hint}
Reverse the contour.
\end{hint}
\begin{solution}
The integral becomes \(-2\pi i\) times the enclosed residue sum.
\end{solution}

\begin{exercise}[Outside residues]\label{exr:iv10-expand-12}
Should poles outside a contour be included in the residue sum?
\end{exercise}
\begin{hint}
Only singularities in the interior contribute.
\end{hint}
\begin{solution}
No. Exterior poles do not affect the contour integral as long as the integrand is holomorphic on a neighborhood of the contour itself.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Logarithmic derivative]\label{exr:iv10-expand-13}
If \(f\) has a zero of order \(m\) at \(a\), find the residue of \(f'/f\) there.
\end{exercise}
\begin{hint}
Factor \(f=(z-a)^m g\).
\end{hint}
\begin{solution}
Then \(f'/f=m/(z-a)+g'/g\), and the second term is holomorphic at \(a\). The residue is \(m\).
\end{solution}

\begin{exercise}[Residue at infinity]\label{exr:iv10-expand-14}
For a rational function decaying as \(1/z^2\) or faster, what relation holds among all finite residues?
\end{exercise}
\begin{hint}
Integrate on a sufficiently large circle and let its radius grow.
\end{hint}
\begin{solution}
The large circle integral tends to zero, so the sum of all finite residues is zero. Equivalently the residue at infinity vanishes.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Counting zeros and poles]\label{exr:iv10-expand-15}
Explain why residues of \(f'/f\) naturally count zeros positively and poles negatively.
\end{exercise}
\begin{hint}
Use local factorizations at zeros and poles.
\end{hint}
\begin{solution}
A zero of order \(m\) contributes residue \(m\); a pole of order \(n\) contributes \(-n\). Integrating \(f'/f\) therefore counts the divisor inside the contour.
\end{solution}

\begin{exercise}[Residue at infinity formula]\label{exr:iv10-expand-16}
Show formally that \(\operatorname{Res}_\infty f=-\sum \operatorname{Res}_{a_k}f\) for a meromorphic function on the sphere.
\end{exercise}
\begin{hint}
Take a large positively oriented circle enclosing all finite poles.
\end{hint}
\begin{solution}
The finite residue theorem gives the large circle integral as \(2\pi i\) times the finite sum. Under \(w=1/z\), the same contour is a negatively oriented small circle around infinity, producing the stated negative relation.
\end{solution}
