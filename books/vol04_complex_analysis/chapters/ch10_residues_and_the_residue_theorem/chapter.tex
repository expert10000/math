\chapter{Residues and the Residue Theorem}
\label{ch:iv10}

Residues compress the local singular data relevant to contour integration into one Laurent coefficient. The residue theorem converts global contour integrals into finite sums of local contributions.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute residues at simple and higher-order poles using the most efficient available formula;
\item apply the residue theorem after identifying all enclosed singularities and contour orientations;
\item compute residues at infinity and use global residue balance;
\item use logarithmic derivatives to count zeros and poles with multiplicity;
\item choose between Laurent expansion, limit, derivative, and partial-fraction methods for residue computation;
\item audit winding numbers and contour geometry before summing residue contributions;
\end{itemize}

\section{Residue definition}
The residue at an isolated singularity is the Laurent coefficient of $(z-z_0)^{-1}$.

\section{Simple poles}
For a simple pole of $g/h$ with $h(z_0)=0$ and $h'(z_0)\ne0$, the residue is $g(z_0)/h'(z_0)$.

\section{Higher-order poles}
Derivative formulas compute residues at poles of finite order.

\section{Residue theorem}
A contour integral equals $2\pi i$ times the sum of enclosed residues, weighted by winding number in the general form.

\section{Local-to-global principle}
Only the $-1$ Laurent terms contribute to closed contour integrals.

\section{Residue at infinity}
For meromorphic functions on the sphere, the residue at infinity balances the finite residues.

\section{Logarithmic derivative}
Residues of $f'/f$ record zero and pole multiplicities.

\section{Computational strategy}
Factor denominators, classify enclosed poles, choose the simplest residue formula, and audit contour orientation.

\section{Parameter dependence}
Residue calculations often reduce integral identities to algebraic sums depending analytically on parameters.

\section{Core structural results}

\begin{theorem}[Residue theorem]\label{thm:iv10-01}
If $f$ is holomorphic inside and on a positively oriented contour except for finitely many isolated singularities $a_k$, then $\int_\Gamma fdz=2\pi i\sum_k\operatorname{Res}(f,a_k)$ in the simple winding-one setting.
\begin{proof}
Remove small disks around the singularities, apply Cauchy's theorem to the remaining multiply connected region, and let the small circles shrink; each circle integral tends to $2\pi i$ times the Laurent $-1$ coefficient.
\end{proof}
\end{theorem}

\begin{theorem}[Simple-pole formula]\label{thm:iv10-02}
If $f=g/h$ with $g,h$ holomorphic, $h(a)=0$, and $h'(a)\ne0$, then $\operatorname{Res}(f,a)=g(a)/h'(a)$.
\begin{proof}
Factor $h(z)=(z-a)k(z)$ with $k(a)=h'(a)$; then $(z-a)f(z)=g(z)/k(z)$ and evaluate at $a$.
\end{proof}
\end{theorem}

\begin{theorem}[Higher-pole residue formula]\label{thm:iv10-03}
If $f$ has a pole of order $m$ at $a$, then $\operatorname{Res}(f,a)=\frac1{(m-1)!}\left[\frac{d^{m-1}}{dz^{m-1}}((z-a)^mf(z))\right]_{z=a}$.
\begin{proof}
The bracketed factor is holomorphic; its Taylor coefficient of degree $m-1$ is exactly the Laurent coefficient multiplying $(z-a)^{-1}$.
\end{proof}
\end{theorem}

\begin{theorem}[Global residue balance]\label{thm:iv10-04}
For a meromorphic function on the Riemann sphere, the sum of all residues including infinity is zero.
\begin{proof}
Integrate around a sufficiently large circle and compare the exterior coordinate $w=1/z$ with the finite residues inside; the residue at infinity is defined to cancel the large-circle contribution.
\end{proof}
\end{theorem}
