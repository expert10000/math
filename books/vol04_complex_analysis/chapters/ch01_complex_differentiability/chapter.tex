\chapter{Complex Differentiability}
\label{ch:iv01}

Complex differentiability is much more rigid than real differentiability. A single complex derivative must approximate increments from every direction while respecting multiplication by $i$, and this rigidity quickly leads to analyticity.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item test complex differentiability at a point by comparing directional difference quotients or the real derivative;
\item translate between the complex derivative and the corresponding real-linear map;
\item verify holomorphicity of elementary functions and identify exceptional points where differentiability fails;
\item apply product, quotient, and chain rules in complex-variable calculations;
\item construct first-order complex linearizations with a controlled little-o remainder;
\item produce examples showing why real differentiability alone does not imply complex differentiability;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{local holomorphic structure}\;\longrightarrow\;\text{integral or series representation}\;\longrightarrow\;\text{global consequence}.}
\]

\section{Complex derivative}
A function is complex differentiable at $z_0$ if the difference quotient has one limit as $z\to z_0$ through arbitrary complex directions.

% BEGIN VOL04-EXPANSION IV01-example-01
\begin{example}[Derivative of a quadratic]\label{ex:iv01-expand-01}
Let \(f(z)=z^2\) and fix \(z_0\). Then
\[
\frac{f(z_0+h)-f(z_0)}{h}
=\frac{2z_0h+h^2}{h}=2z_0+h.
\]
The limit is \(2z_0\) independently of the direction in which \(h\to0\). Hence \(f\) is complex differentiable everywhere and \(f'(z_0)=2z_0\). The calculation exhibits the decisive requirement: one complex limit must work for every approach direction.
\end{example}
% END VOL04-EXPANSION IV01-example-01
\section{Real-linear viewpoint}
Writing $f=u+iv$ identifies the real derivative with a $2\times2$ matrix; complex differentiability means that matrix is multiplication by one complex number.

\section{Directional consistency}
Approaching along real and imaginary directions gives necessary compatibility conditions.

\section{Holomorphic functions}
A function holomorphic on an open set is complex differentiable at every point there.

% BEGIN VOL04-EXPANSION IV01-example-02
\begin{example}[Conjugation fails the direction test]\label{ex:iv01-expand-02}
For \(f(z)=\overline z\) at the origin, real increments give
\[
\frac{f(h)-f(0)}{h}=1,
\]
while purely imaginary increments \(h=it\) give
\[
\frac{f(it)-f(0)}{it}=\frac{-it}{it}=-1.
\]
The directional limits disagree, so conjugation is not complex differentiable at \(0\). Translating the same calculation shows that it is nowhere complex differentiable.
\end{example}
% END VOL04-EXPANSION IV01-example-02
\section{Elementary examples}
Polynomials and rational functions away from poles are holomorphic; conjugation and modulus fail complex differentiability except at special points.

\section{Algebra rules}
Sums, products, quotients, and compositions obey familiar derivative rules where defined.

\section{Local linearization}
$f(z_0+h)=f(z_0)+f'(z_0)h+o(|h|)$ is the coordinate-free local approximation.

% BEGIN VOL04-EXPANSION IV01-example-03
\begin{example}[A function differentiable only at one point]\label{ex:iv01-expand-03}
Consider \(f(z)=|z|^2=z\overline z\). At \(0\),
\[
\frac{f(h)-f(0)}{h}=\frac{|h|^2}{h}=\overline h\longrightarrow0,
\]
so \(f'(0)=0\). At a nonzero point \(z_0\), increments along the real and imaginary directions produce different linear terms because of the \(\overline h\) contribution. Thus a function may be complex differentiable at an isolated point without being holomorphic on any neighborhood of that point.
\end{example}

\section{Core structural results}

\begin{theorem}[Complex-linear characterization]\label{thm:iv01-01}
A real-differentiable map $f:\mathbb C\to\mathbb C$ is complex differentiable at $z_0$ iff its real derivative commutes with multiplication by $i$.
\begin{proof}
If $Df$ is multiplication by $f'(z_0)$ it clearly commutes with $i$. Conversely any real-linear map commuting with $i$ is determined by its value at $1$ and equals complex multiplication by that value.
\end{proof}
\end{theorem}

\begin{theorem}[Product rule]\label{thm:iv01-02}
If $f,g$ are complex differentiable at $z_0$, then $(fg)'=f'g+fg'$.
\begin{proof}
Use the increment identity $f(z)g(z)-f_0g_0=(f-f_0)g+f_0(g-g_0)$ and divide by $z-z_0$, using continuity implied by differentiability.
\end{proof}
\end{theorem}

\begin{theorem}[Chain rule]\label{thm:iv01-03}
If $f$ is differentiable at $z_0$ and $g$ at $f(z_0)$, then $(g\circ f)'(z_0)=g'(f(z_0))f'(z_0)$.
\begin{proof}
Insert the complex-linear first-order expansions for $f$ and $g$ and combine their little-o remainders.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Polynomial derivative]\label{exr:iv01-expand-01}
Compute the complex derivative of \(f(z)=3z^4-2z+7\).
\end{exercise}
\begin{hint}
Use the algebra rules for powers and sums.
\end{hint}
\begin{solution}
The power rule gives \(f'(z)=12z^3-2\).
\end{solution}

\begin{exercise}[Reciprocal derivative]\label{exr:iv01-expand-02}
Differentiate \(f(z)=1/(z-2i)\) on its domain.
\end{exercise}
\begin{hint}
Apply the reciprocal or quotient rule away from the pole.
\end{hint}
\begin{solution}
For \(z\ne2i\), \(f'(z)=-(z-2i)^{-2}\).
\end{solution}

\begin{exercise}[Chain rule]\label{exr:iv01-expand-03}
Find the derivative of \(f(z)=(z^2+1)^5\).
\end{exercise}
\begin{hint}
Differentiate the outer fifth power and then the inner quadratic.
\end{hint}
\begin{solution}
The chain rule gives \(f'(z)=10z(z^2+1)^4\).
\end{solution}

\begin{exercise}[Directional diagnostic]\label{exr:iv01-expand-04}
Test complex differentiability of \(f(z)=\operatorname{Re}z\) at \(0\).
\end{exercise}
\begin{hint}
Compare real and imaginary increments.
\end{hint}
\begin{solution}
Along real increments the quotient is \(1\), while along imaginary increments it is \(0\). Therefore the derivative does not exist.
\end{solution}

\begin{exercise}[Linearization]\label{exr:iv01-expand-05}
Write the first order complex linearization of \(e^z\) at \(z_0\).
\end{exercise}
\begin{hint}
Use the derivative \(e^{z_0}\).
\end{hint}
\begin{solution}
The linearization is \(e^{z_0}+e^{z_0}(z-z_0)\), with an error \(o(|z-z_0|)\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Differentiability implies continuity]\label{exr:iv01-expand-06}
Prove that complex differentiability at \(z_0\) implies continuity there.
\end{exercise}
\begin{hint}
Factor \(f(z)-f(z_0)\) by \(z-z_0\).
\end{hint}
\begin{solution}
Write \(f(z)-f(z_0)=(z-z_0)Q(z)\), where \(Q(z)\to f'(z_0)\). The factor \(Q\) is bounded near \(z_0\), so the product tends to zero.
\end{solution}

\begin{exercise}[Product rule]\label{exr:iv01-expand-07}
Prove the complex product rule directly from difference quotients.
\end{exercise}
\begin{hint}
Add and subtract \(f(z)g(z_0)\).
\end{hint}
\begin{solution}
Split the increment into \(f(z)(g(z)-g(z_0))+g(z_0)(f(z)-f(z_0))\), divide by \(z-z_0\), and use continuity of \(f\).
\end{solution}

\begin{exercise}[Reciprocal rule]\label{exr:iv01-expand-08}
If \(f(z_0)\ne0\), prove that \(1/f\) is differentiable at \(z_0\) and find its derivative.
\end{exercise}
\begin{hint}
Use a difference of reciprocals and continuity of \(f\).
\end{hint}
\begin{solution}
The quotient becomes \(-(f(z)-f(z_0))/((z-z_0)f(z)f(z_0))\). Passing to the limit gives \(-(f'(z_0))/f(z_0)^2\).
\end{solution}

\begin{exercise}[Complex linear real derivative]\label{exr:iv01-expand-09}
Prove that a real linear map \(T:\mathbb C\to\mathbb C\) is complex linear exactly when \(T(i)=iT(1)\).
\end{exercise}
\begin{hint}
Every complex number is \(x+iy\).
\end{hint}
\begin{solution}
If the condition holds, real linearity gives \(T(x+iy)=xT(1)+yT(i)=(x+iy)T(1)\). The converse follows by evaluating a complex linear map at \(i\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Absolute value]\label{exr:iv01-expand-10}
Is \(f(z)=|z|\) complex differentiable at \(0\)? Justify the answer.
\end{exercise}
\begin{hint}
Compare positive and negative real increments.
\end{hint}
\begin{solution}
For positive real \(h\), the quotient is \(1\); for negative real \(h\), it is \(-1\). Hence no complex derivative exists.
\end{solution}

\begin{exercise}[A single derivative is not holomorphicity]\label{exr:iv01-expand-11}
Does complex differentiability of a function at one point imply holomorphicity near that point?
\end{exercise}
\begin{hint}
Use \(|z|^2\).
\end{hint}
\begin{solution}
No. The function \(|z|^2\) is complex differentiable at \(0\) with derivative zero, but it is not complex differentiable at any nonzero point.
\end{solution}

\begin{exercise}[Quotient rule hypothesis]\label{exr:iv01-expand-12}
What fails if the quotient rule is applied to \(f/g\) at a point where \(g=0\)?
\end{exercise}
\begin{hint}
First ask whether the quotient is even defined near the point.
\end{hint}
\begin{solution}
The quotient need not be defined at the point and its reciprocal factor can be singular. The nonvanishing denominator hypothesis is essential.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Error estimate from linearization]\label{exr:iv01-expand-13}
Suppose \(f'(z_0)=a\). Explain how the derivative predicts \(f(z_0+h)\) for small \(h\).
\end{exercise}
\begin{hint}
Use the definition of the little o remainder.
\end{hint}
\begin{solution}
One has \(f(z_0+h)=f(z_0)+ah+r(h)\) with \(|r(h)|/|h|\to0\). Thus multiplication by \(a\) gives the first order change in both magnitude and direction.
\end{solution}

\begin{exercise}[Real linear classifier]\label{exr:iv01-expand-14}
For \(T(z)=az+b\overline z\), determine when \(T\) is complex linear.
\end{exercise}
\begin{hint}
Compare \(T(iz)\) with \(iT(z)\).
\end{hint}
\begin{solution}
The conjugate term changes sign under multiplication by \(i\). Equality for every \(z\) forces \(b=0\); then \(T(z)=az\) is complex linear.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Two direction agreement is not enough]\label{exr:iv01-expand-15}
Explain why agreement of the real and imaginary directional quotients alone does not prove complex differentiability without additional regularity.
\end{exercise}
\begin{hint}
Directional tests provide necessary conditions, not uniform control over all directions.
\end{hint}
\begin{solution}
The derivative requires one limit as \(h\to0\) through arbitrary complex directions. A function can be engineered so the quotients along the coordinate axes agree while quotients along a curved or oblique path do not.
\end{solution}

\begin{exercise}[Derivative of inversion as geometry]\label{exr:iv01-expand-16}
Show that \(f(z)=1/z\) has derivative \(-1/z^2\) and interpret the local map as a rotation-dilation.
\end{exercise}
\begin{hint}
Compute the difference quotient and then write the nonzero derivative in polar form.
\end{hint}
\begin{solution}
The quotient simplifies to \(-1/(z(z+h))\), giving \(-1/z^2\). Any nonzero complex derivative is multiplication by a complex number, hence locally a dilation by its modulus followed by a rotation by its argument.
\end{solution}
