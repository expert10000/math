\chapter{Lattices and Complex Tori}
\label{ch:iv27}

A lattice in the complex plane is a discrete rank-two subgroup, and quotienting by it produces a compact complex torus. This is the analytic model of every genus-one Riemann surface after choosing a base point.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct a complex torus as \(\mathbb C/\Lambda\) from a lattice;
\item choose fundamental parallelograms and identify equivalent boundary points;
\item verify that holomorphic structure descends through the lattice quotient;
\item classify lattice changes induced by basis transformations and scaling;
\item compute basic topological and complex-analytic invariants of the torus;
\item relate periodic functions on \(\mathbb C\) to meromorphic functions on the torus;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{local holomorphic structure}\;\longrightarrow\;\text{integral or series representation}\;\longrightarrow\;\text{global consequence}.}
\]

\section{Lattices}
A lattice has form $\Lambda=\mathbb Z\omega_1+\mathbb Z\omega_2$ with $\omega_1/\omega_2\notin\mathbb R$.

% BEGIN VOL04-EXPANSION IV27-example-01
\begin{example}[Reducing a point to a fundamental parallelogram]\label{ex:iv27-expand-01}
For the square lattice \(\Lambda=\mathbb Z+i\mathbb Z\), take \(z=3.4-2.7i\). Subtract the lattice element \(3-3i\):
\[
z-(3-3i)=0.4+0.3i.
\]
Thus \(z\) and \(0.4+0.3i\) represent the same torus point. Reduction modulo the two periods is the concrete quotient operation behind \(\mathbb C/\Lambda\).
\end{example}
% END VOL04-EXPANSION IV27-example-01
\section{Fundamental parallelogram}
Every complex number is congruent modulo the lattice to a point in a chosen fundamental parallelogram, with boundary identifications.

\section{Complex torus}
The quotient $\mathbb C/\Lambda$ inherits a compact Riemann-surface structure from translations.

\section{Quotient projection}
The map $\pi:\mathbb C\to\mathbb C/\Lambda$ is a holomorphic covering map.

% BEGIN VOL04-EXPANSION IV27-example-02
\begin{example}[Changing lattice basis]\label{ex:iv27-expand-02}
Let \(\Lambda=\mathbb Z\omega_1+\mathbb Z\omega_2\) and define
\[
\omega_1'=\omega_1+\omega_2,\qquad \omega_2'=\omega_2.
\]
The change matrix has determinant one, so the new pair generates the same lattice. Conversely,
\[
\omega_1=\omega_1'-\omega_2',\qquad \omega_2=\omega_2'.
\]
The torus depends on the lattice subgroup, not on a particular basis.
\end{example}
% END VOL04-EXPANSION IV27-example-02
\section{Change of lattice basis}
Replacing a lattice basis by an $SL_2(\mathbb Z)$ basis does not change the lattice.

\section{Normalized modulus}
After scaling, a lattice can be represented by $\mathbb Z+\mathbb Z\tau$ with $\Im\tau>0$.

\section{Modular equivalence}
Different $\tau$ values related by fractional linear $SL_2(\mathbb Z)$ transformations describe isomorphic normalized tori.

% BEGIN VOL04-EXPANSION IV27-example-03
\begin{example}[Normalizing a lattice]\label{ex:iv27-expand-03}
Given noncollinear periods \(\omega_1,\omega_2\), scale by \(1/\omega_1\). The lattice becomes
\[
\mathbb Z+\mathbb Z\tau,\qquad \tau=\frac{\omega_2}{\omega_1}.
\]
After reversing a basis vector if necessary, choose \(\Im\tau>0\). This isolates the complex shape parameter while removing the irrelevant overall scale.
\end{example}
% END VOL04-EXPANSION IV27-example-03
\section{Holomorphic differentials}
The form $dz$ descends to a nowhere-vanishing holomorphic differential on the torus.

\section{Genus one}
The torus has Euler characteristic zero and genus one.

\section{Core structural results}

\begin{theorem}[Complex structure on the quotient]\label{thm:iv27-01}
For a lattice $\Lambda$, the quotient $\mathbb C/\Lambda$ has a unique Riemann-surface structure making the projection locally biholomorphic.
\begin{proof}
Choose disks small enough that distinct lattice translates are disjoint. Projection restricts to a homeomorphism on each such disk, and transition maps between lifted charts are translations, hence holomorphic.
\end{proof}
\end{theorem}

\begin{theorem}[Compactness of the torus]\label{thm:iv27-02}
$\mathbb C/\Lambda$ is compact.
\begin{proof}
The closure of a fundamental parallelogram is compact and its projection surjects onto the quotient. A continuous image of a compact set is compact.
\end{proof}
\end{theorem}

\begin{theorem}[Basis-change invariance]\label{thm:iv27-03}
If $(\omega_1',\omega_2')=(a\omega_1+b\omega_2,c\omega_1+d\omega_2)$ with an integral matrix of determinant $\pm1$, then the generated lattice is unchanged.
\begin{proof}
The new generators are integral combinations of the old, and invertibility over $\mathbb Z$ expresses the old generators as integral combinations of the new.
\end{proof}
\end{theorem}

\subsection*{Standard computations}

\begin{exercise}[Square lattice reduction]\label{exr:iv27-expand-01}
Reduce \(2.3+4.8i\) modulo \(\mathbb Z+i\mathbb Z\) to the unit square.
\end{exercise}
\begin{hint}
Subtract integer real and imaginary parts.
\end{hint}
\begin{solution}
One representative is \(0.3+0.8i\).
\end{solution}

\begin{exercise}[Basis determinant]\label{exr:iv27-expand-02}
Does the pair \(\omega_1'=2\omega_1+\omega_2\), \(\omega_2'=\omega_1+\omega_2\) generate the same lattice?
\end{exercise}
\begin{hint}
Compute the determinant of the integer matrix.
\end{hint}
\begin{solution}
Yes. The determinant is \(2\cdot1-1\cdot1=1\).
\end{solution}

\begin{exercise}[Normalized modulus]\label{exr:iv27-expand-03}
Normalize the lattice generated by \(2\) and \(1+3i\).
\end{exercise}
\begin{hint}
Divide both periods by \(2\).
\end{hint}
\begin{solution}
It becomes \(\mathbb Z+\mathbb Z((1+3i)/2)\).
\end{solution}

\begin{exercise}[Torus covering]\label{exr:iv27-expand-04}
What is the deck group of \(\mathbb C\to\mathbb C/\Lambda\)?
\end{exercise}
\begin{hint}
Deck maps are period translations.
\end{hint}
\begin{solution}
It is the translation group \(z\mapsto z+\lambda\) for \(\lambda\in\Lambda\).
\end{solution}

\begin{exercise}[Euler characteristic]\label{exr:iv27-expand-05}
What is the genus of \(\mathbb C/\Lambda\)?
\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
It has genus one.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Quotient complex structure]\label{exr:iv27-expand-06}
Prove that \(\mathbb C/\Lambda\) has a Riemann-surface structure.
\end{exercise}
\begin{hint}
Use sufficiently small disks with disjoint lattice translates.
\end{hint}
\begin{solution}
Projection is injective on each small disk and transitions between different lifts are translations, hence holomorphic.
\end{solution}

\begin{exercise}[Compactness]\label{exr:iv27-expand-07}
Prove that the complex torus is compact.
\end{exercise}
\begin{hint}
Project a closed fundamental parallelogram.
\end{hint}
\begin{solution}
The closed parallelogram is compact and its projection is continuous and surjective, so the quotient is compact.
\end{solution}

\begin{exercise}[Basis invariance]\label{exr:iv27-expand-08}
Prove that an integral basis change with determinant \(\pm1\) leaves the lattice unchanged.
\end{exercise}
\begin{hint}
Use the integer inverse matrix.
\end{hint}
\begin{solution}
New generators are integer combinations of old ones, and the inverse matrix with integer entries expresses the old generators as combinations of the new ones.
\end{solution}

\begin{exercise}[Descending differential]\label{exr:iv27-expand-09}
Explain why \(dz\) descends to the torus.
\end{exercise}
\begin{hint}
Check invariance under deck translations.
\end{hint}
\begin{solution}
Translations have differential one, so \(d(z+\lambda)=dz\). Thus the form is invariant and defines a holomorphic one-form downstairs.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Collinear generators]\label{exr:iv27-expand-10}
Do \(1\) and \(2\) generate a complex lattice of rank two?
\end{exercise}
\begin{hint}
Check real linear independence.
\end{hint}
\begin{solution}
No. Their ratio is real, so the subgroup is one-dimensional rather than a discrete rank-two lattice spanning the plane.
\end{solution}

\begin{exercise}[Determinant two]\label{exr:iv27-expand-11}
Does the basis change matrix \(\begin{pmatrix}2&0\\0&1\end{pmatrix}\) preserve the lattice?
\end{exercise}
\begin{hint}
Its determinant is not \(\pm1\).
\end{hint}
\begin{solution}
Not in general. It generates a sublattice of index two.
\end{solution}

\begin{exercise}[Noncompact quotient]\label{exr:iv27-expand-12}
Would quotienting by only one nonzero period give a compact Riemann surface?
\end{exercise}
\begin{hint}
The remaining transverse direction is unbounded.
\end{hint}
\begin{solution}
No. The quotient is a cylinder and remains noncompact.
\end{solution}

\begin{exercise}[Periodic functions]\label{exr:iv27-expand-13}
Why does a doubly periodic meromorphic function descend to \(\mathbb C/\Lambda\)?
\end{exercise}
\begin{hint}
It is constant on lattice equivalence classes.
\end{hint}
\begin{solution}
Periodicity makes the function well defined on quotient points, and local quotient charts preserve meromorphicity.
\end{solution}

\begin{exercise}[Genus-one moduli]\label{exr:iv27-expand-14}
What information does the normalized parameter \(\tau\) retain?
\end{exercise}
\begin{hint}
Overall scaling was removed.
\end{hint}
\begin{solution}
It records the complex shape of the lattice, modulo changes of lattice basis by modular transformations.
\end{solution}

\begin{exercise}[Modular basis move]\label{exr:iv27-expand-15}
Show that swapping a normalized basis sends \(\tau\) to \(-1/\tau\) after rescaling.
\end{exercise}
\begin{hint}
Use the new basis \((\tau,-1)\) and normalize the first generator.
\end{hint}
\begin{solution}
Dividing by \(\tau\) gives generators \(1\) and \(-1/\tau\), so the normalized parameter transforms accordingly.
\end{solution}

\begin{exercise}[No holomorphic coordinate globally]\label{exr:iv27-expand-16}
Why can the torus not admit one global chart into an open subset of \(\mathbb C\)?
\end{exercise}
\begin{hint}
A chart is a homeomorphism onto an open plane set.
\end{hint}
\begin{solution}
The torus is compact, while a nonempty open subset of \(\mathbb C\) is not compact. Hence no single global planar chart exists.
\end{solution}
