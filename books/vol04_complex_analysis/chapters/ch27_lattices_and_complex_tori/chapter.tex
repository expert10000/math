\chapter{Lattices and Complex Tori}
\label{ch:iv27}

A lattice in the complex plane is a discrete rank-two subgroup, and quotienting by it produces a compact complex torus. This is the analytic model of every genus-one Riemann surface after choosing a base point.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct a complex torus as \(\mathbb C/\Lambda\) from a lattice;
\item choose fundamental parallelograms and identify equivalent boundary points;
\item verify that holomorphic structure descends through the lattice quotient;
\item classify lattice changes induced by basis transformations and scaling;
\item compute basic topological and complex-analytic invariants of the torus;
\item relate periodic functions on \(\mathbb C\) to meromorphic functions on the torus;
\end{itemize}

\section{Lattices}
A lattice has form $\Lambda=\mathbb Z\omega_1+\mathbb Z\omega_2$ with $\omega_1/\omega_2\notin\mathbb R$.

\section{Fundamental parallelogram}
Every complex number is congruent modulo the lattice to a point in a chosen fundamental parallelogram, with boundary identifications.

\section{Complex torus}
The quotient $\mathbb C/\Lambda$ inherits a compact Riemann-surface structure from translations.

\section{Quotient projection}
The map $\pi:\mathbb C\to\mathbb C/\Lambda$ is a holomorphic covering map.

\section{Change of lattice basis}
Replacing a lattice basis by an $SL_2(\mathbb Z)$ basis does not change the lattice.

\section{Normalized modulus}
After scaling, a lattice can be represented by $\mathbb Z+\mathbb Z\tau$ with $\Im\tau>0$.

\section{Modular equivalence}
Different $\tau$ values related by fractional linear $SL_2(\mathbb Z)$ transformations describe isomorphic normalized tori.

\section{Holomorphic differentials}
The form $dz$ descends to a nowhere-vanishing holomorphic differential on the torus.

\section{Genus one}
The torus has Euler characteristic zero and genus one.

\section{Core structural results}

\begin{theorem}[Complex structure on the quotient]\label{thm:iv27-01}
For a lattice $\Lambda$, the quotient $\mathbb C/\Lambda$ has a unique Riemann-surface structure making the projection locally biholomorphic.
\begin{proof}
Choose disks small enough that distinct lattice translates are disjoint. Projection restricts to a homeomorphism on each such disk, and transition maps between lifted charts are translations, hence holomorphic.
\end{proof}
\end{theorem}

\begin{theorem}[Compactness of the torus]\label{thm:iv27-02}
$\mathbb C/\Lambda$ is compact.
\begin{proof}
The closure of a fundamental parallelogram is compact and its projection surjects onto the quotient. A continuous image of a compact set is compact.
\end{proof}
\end{theorem}

\begin{theorem}[Basis-change invariance]\label{thm:iv27-03}
If $(\omega_1',\omega_2')=(a\omega_1+b\omega_2,c\omega_1+d\omega_2)$ with an integral matrix of determinant $\pm1$, then the generated lattice is unchanged.
\begin{proof}
The new generators are integral combinations of the old, and invertibility over $\mathbb Z$ expresses the old generators as integral combinations of the new.
\end{proof}
\end{theorem}
