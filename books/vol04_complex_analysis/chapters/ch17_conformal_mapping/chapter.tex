\chapter{Conformal Mapping}
\label{ch:iv17}

Conformal maps preserve angles locally and reorganize domains without destroying holomorphic structure. Holomorphic maps with nonzero derivative are conformal, and the Riemann mapping theorem gives a universal model for simply connected proper planar domains.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify conformality from holomorphicity and nonvanishing derivative;
\item compute local angle and scale changes from the derivative;
\item construct conformal equivalences by composing standard maps;
\item use disk, half-plane, strip, and sector mappings in explicit domain problems;
\item apply Schwarz-type principles or normalization to constrain conformal maps;
\item distinguish conformal equivalence from mere homeomorphism;
\end{itemize}

\section{Riemann mapping theorem}
\begin{theorem}[Riemann mapping theorem]\label{thm:iv17-riemann-mapping}
Every nonempty simply connected proper domain $D\subsetneq\mathbb C$ is biholomorphic to the unit disk $\mathbb D$.
\end{theorem}
\paragraph{Proof scope.}
The existence theorem is used here as a standard external theorem. A complete proof requires normal-family/Montel machinery not developed in this volume. The normalization and uniqueness statement that follows is proved from Schwarz's lemma and the disk-automorphism classification developed locally.

\section{Normalization}
Fixing one point and the argument of the derivative gives uniqueness of the Riemann map.

\section{Boundary caution}
Interior conformal equivalence does not automatically provide regular extension to arbitrary boundaries.

\section{Core structural results}

\begin{theorem}[Holomorphic maps are conformal off critical points]\label{thm:iv17-01}
If $f$ is holomorphic and $f'(z_0)\ne0$, then $f$ preserves oriented angles at $z_0$.
\begin{proof}
The first-order expansion is $f(z_0+h)-f(z_0)=f'(z_0)h+o(|h|)$. Multiplication by the nonzero complex number $f'(z_0)$ rotates and scales every tangent direction by the same amount.
\end{proof}
\end{theorem}

\begin{theorem}[Schwarz lemma]\label{thm:iv17-02}
If $f:\mathbb D\to\mathbb D$ is holomorphic and $f(0)=0$, then $|f(z)|\le|z|$ and $|f'(0)|\le1$, with equality at a nonzero point or in derivative only for rotations.
\begin{proof}
Apply the maximum principle to $g(z)=f(z)/z$, with removable value $g(0)=f'(0)$. For every $r<1$, maximum modulus on $|z|=r$ gives $|g(z)|\le1/r$ inside; let $r\to1$. Equality invokes the maximum-modulus rigidity.
\end{proof}
\end{theorem}

\begin{theorem}[Automorphisms of the disk]\label{thm:iv17-03}
Every automorphism of $\mathbb D$ has form $e^{i\theta}(z-a)/(1-\bar a z)$ for some $a\in\mathbb D$.
\begin{proof}
Move the image of zero back to zero with a standard disk Möbius map. Schwarz lemma applied to the resulting automorphism and its inverse forces it to be a rotation.
\end{proof}
\end{theorem}

\begin{theorem}[Normalized Riemann-map uniqueness]\label{thm:iv17-04}
If two biholomorphisms $f,g:D\to\mathbb D$ send $z_0$ to zero and have positive real derivative at $z_0$, then $f=g$.
\begin{proof}
The disk automorphism $g\circ f^{-1}$ fixes zero. By the disk automorphism classification it is a rotation, and the derivative normalization forces that rotation to be the identity.
\end{proof}
\end{theorem}

\subsection*{Proofs}

\begin{exercise}[Nonzero derivative preserves angles]\label{exr:iv17-expand-06}
Prove local conformality of a holomorphic function where its derivative is nonzero.
\end{exercise}
\begin{hint}
Use the first order expansion.
\end{hint}
\begin{solution}
Near \(z_0\), \(f(z_0+h)-f(z_0)=f'(z_0)h+o(|h|)\). The leading multiplication by a nonzero complex number is a rotation-dilation, so tangent angles are preserved in the limit.
\end{solution}

\begin{exercise}[Inverse conformality]\label{exr:iv17-expand-07}
If a holomorphic injective map has nonzero derivative, show its local inverse is holomorphic and conformal.
\end{exercise}
\begin{hint}
Use the complex inverse function theorem.
\end{hint}
\begin{solution}
The inverse derivative is \(1/f'(z)\), which is nonzero. Hence the inverse is holomorphic and angle preserving locally.
\end{solution}

\begin{exercise}[Composition]\label{exr:iv17-expand-08}
Show that a composition of conformal maps is conformal where both are defined.
\end{exercise}
\begin{hint}
Use the chain rule.
\end{hint}
\begin{solution}
The derivative of the composition is the product of two nonzero complex derivatives, hence is nonzero and represents the composition of two rotation-dilations.
\end{solution}

\begin{exercise}[Harmonic pullback]\label{exr:iv17-expand-09}
Explain why composing a harmonic function with a conformal coordinate change preserves harmonicity in two dimensions.
\end{exercise}
\begin{hint}
Use the Laplacian transformation under a holomorphic map.
\end{hint}
\begin{solution}
The Laplacian picks up the positive factor \(|f'|^2\). Thus if the original Laplacian is zero, the pulled back one is also zero.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{solution}

\end{solution}

\begin{exercise}[Holomorphic but not injective globally]\label{exr:iv17-expand-11}

\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
No. The exponential has nonzero derivative everywhere but is periodic, so it is not globally injective on the whole plane.
\end{solution}

\end{exercise}
\begin{hint}

\end{hint}
\begin{solution}
No. Interior conformality alone does not guarantee a well behaved boundary extension.
\end{solution}
