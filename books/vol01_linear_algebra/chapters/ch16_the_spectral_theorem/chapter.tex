\chapter{The Spectral Theorem}
\label{ch:i16-the-spectral-theorem}

The spectral theorem is the central structure theorem for self-adjoint, symmetric and normal operators. It says that the right compatibility with the inner product upgrades triangularization to orthonormal diagonalization. Functional calculus and positive operators then become simple operations on eigenvalues.

\section{Adjoints}
The adjoint $T^*$ is characterized by $\langle Tv,w\rangle=\langle v,T^*w\rangle$.

\section{Self-adjoint and normal operators}
$T$ is self-adjoint when $T=T^*$ and normal when $TT^*=T^*T$.

\section{Reality of the spectrum}
Eigenvalues of self-adjoint operators are real, and eigenspaces for distinct eigenvalues are orthogonal.

\section{Spectral theorem}
A finite-dimensional complex operator is normal if and only if it is unitarily diagonalizable. Over $\mathbb R$, symmetric operators are orthogonally diagonalizable.

\section{Functional calculus}
If $T=UDU^*$, define $f(T)=Uf(D)U^*$ for functions defined on the spectrum.

\section{Positive operators}
A self-adjoint operator is positive semidefinite when $\langle Tv,v\rangle\ge0$; spectrally this means all eigenvalues are nonnegative.

\section{Core structural results}

\begin{theorem}[Orthogonality of self-adjoint eigenspaces]\label{thm:i16-01}
Eigenvectors of a self-adjoint operator associated with distinct eigenvalues are orthogonal.
\begin{proof}
If $Tv=\lambda v$ and $Tw=\mu w$, then $\lambda\langle v,w\rangle=\langle Tv,w\rangle=\langle v,Tw\rangle=\mu\langle v,w\rangle$.
\end{proof}
\end{theorem}

\begin{theorem}[Finite-dimensional spectral theorem]\label{thm:i16-02}
Every self-adjoint operator on a finite-dimensional inner-product space has an orthonormal basis of eigenvectors.
\begin{proof}
Choose an eigenvector, take its orthogonal complement, show that complement is invariant under self-adjointness, and induct on dimension.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Real symmetric matrix]\label{ex:i16-worked-01}
$\begin{pmatrix}2&1\\1&2\end{pmatrix}$ has orthogonal eigenvectors $(1,1)$ and $(1,-1)$ with eigenvalues $3$ and $1$.
\end{example}

\begin{example}[Positive square root]\label{ex:i16-worked-02}
For positive $A=UDU^*$, set $A^{1/2}=UD^{1/2}U^*$; then $A^{1/2}$ is positive and squares to $A$.
\end{example}

\begin{example}[Normal but not self-adjoint]\label{ex:i16-worked-03}
A nontrivial complex unitary rotation is normal but may not be self-adjoint.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Adjoint matrix]\label{prob:i16-dossier-01}
In the standard complex inner product, what matrix represents the adjoint of $A$?
\end{problem}
\begin{solution}
The adjoint is represented by the conjugate transpose $A^*=\overline A^T$, because $\langle Ax,y\rangle=x^*A^*y=\langle x,A^*y\rangle$.
\end{solution}

\begin{problem}[Real eigenvalues]\label{prob:i16-dossier-02}
Prove an eigenvalue of a self-adjoint operator is real.
\end{problem}
\begin{solution}
If $Tv=\lambda v$, then $\lambda\|v\|^2=\langle Tv,v\rangle=\langle v,Tv\rangle=\overline\lambda\|v\|^2$. Since $v\ne0$, $\lambda=\overline\lambda$.
\end{solution}

\begin{problem}[Orthogonal eigenspaces]\label{prob:i16-dossier-03}
Why are eigenspaces of a self-adjoint operator for distinct eigenvalues orthogonal?
\end{problem}
\begin{solution}
If $Tv=\lambda v$ and $Tw=\mu w$, self-adjointness gives $\lambda\langle v,w\rangle=\langle Tv,w\rangle=\langle v,Tw\rangle=\mu\langle v,w\rangle$. Distinctness forces the inner product to vanish.
\end{solution}

\begin{problem}[Diagonalizing a symmetric matrix]\label{prob:i16-dossier-04}
Diagonalize $A=\begin{pmatrix}2&1\\1&2\end{pmatrix}$ orthogonally.
\end{problem}
\begin{solution}
Eigenvalues are $3$ and $1$ with eigenvectors $(1,1)$ and $(1,-1)$. Normalizing them gives an orthogonal matrix $Q$ with $Q^TAQ=\operatorname{diag}(3,1)$.
\end{solution}

\begin{problem}[Positive square root]\label{prob:i16-dossier-05}
How does the spectral theorem construct $A^{1/2}$ for positive semidefinite $A$?
\end{problem}
\begin{solution}
Write $A=U\operatorname{diag}(\lambda_i)U^*$ with $\lambda_i\ge0$. Define $A^{1/2}=U\operatorname{diag}(\sqrt{\lambda_i})U^*$. It is positive and squares to $A$.
\end{solution}

\begin{problem}[Normal operator]\label{prob:i16-dossier-06}
Why is every unitary operator normal?
\end{problem}
\begin{solution}
For unitary $U$, $U^*U=UU^*=I$, so $UU^*=U^*U$.
\end{solution}

\begin{problem}[Polynomial functional calculus]\label{prob:i16-dossier-07}
If $T=UDU^*$ is normal, compute $p(T)$.
\end{problem}
\begin{solution}
$p(T)=Up(D)U^*$, where $p(D)$ is obtained by applying $p$ to each diagonal eigenvalue. This follows because powers of $T$ preserve the same diagonalizing basis.
\end{solution}

\begin{problem}[Positivity and spectrum]\label{prob:i16-dossier-08}
Why is a self-adjoint operator positive semidefinite iff all eigenvalues are nonnegative?
\end{problem}
\begin{solution}
In an orthonormal eigenbasis, $\langle Tv,v\rangle=\sum_i\lambda_i|v_i|^2$. This is nonnegative for all $v$ exactly when each $\lambda_i\ge0$.
\end{solution}

\begin{problem}[Invariant orthogonal complement]\label{prob:i16-dossier-09}
If $U$ is invariant under self-adjoint $T$, show $U^\perp$ is invariant.
\end{problem}
\begin{solution}
For $w\in U^\perp$ and $u\in U$, $\langle Tw,u\rangle=\langle w,Tu\rangle=0$ because $Tu\in U$. Hence $Tw\in U^\perp$.
\end{solution}

\begin{problem}[Spectral norm]\label{prob:i16-dossier-10}
For self-adjoint $T$, what is $\|T\|_2$ in terms of eigenvalues?
\end{problem}
\begin{solution}
Because the operator is unitarily diagonalizable, its singular values are $|\lambda_i|$. Therefore the operator norm is $\max_i|\lambda_i|$.
\end{solution}

\begin{problem}[Spectral projector]\label{prob:i16-dossier-11}
How can the projection onto an eigenspace be expressed when eigenvalues are distinct?
\end{problem}
\begin{solution}
Use Lagrange interpolation: $P_i=\prod_{j\ne i}(T-\lambda_jI)/(\lambda_i-\lambda_j)$. It acts as $1$ on the $\lambda_i$ eigenspace and $0$ on the others.
\end{solution}

\begin{problem}[Spectral synthesis]\label{prob:i16-dossier-12}
Why is the spectral theorem stronger than ordinary diagonalization?
\end{problem}
\begin{solution}
It provides an orthonormal eigenbasis, not merely an arbitrary eigenbasis. Consequently the change-of-basis matrix is unitary, preserving the inner product and making geometric and numerical conclusions immediately available.
\end{solution}

\begin{exercise}\label{exr:i16-06}
Why is the orthogonal complement of an invariant eigenspace also invariant for a self-adjoint operator?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
If $w$ is orthogonal to the eigenspace and $v$ is in it, then $\langle Tw,v\rangle=\langle w,Tv\rangle=\lambda\langle w,v\rangle=0$.
\end{solution}

\begin{exercise}\label{exr:i16-07}
What extra conclusion does normality give beyond triangularization?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
A normal operator can be diagonalized by a unitary basis change, so the basis can be chosen orthonormal.
\end{solution}

\begin{exercise}\label{exr:i16-08}
Relate spectral theorem to quadratic forms.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
A real symmetric matrix can be orthogonally diagonalized, reducing its quadratic form to a sum of scalar multiples of squares.
\end{solution}

\section*{Chapter summary}
The spectral theorem is the central structure theorem for self-adjoint, symmetric and normal operators. It says that the right compatibility with the inner product upgrades triangularization to orthonormal diagonalization. Functional calculus and positive operators then become simple operations on eigenvalues.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
