\chapter{Canonical Forms}
\label{ch:i12-canonical-forms}

When diagonalization fails, canonical forms measure the failure rather than hiding it. Jordan form over an algebraically closed field records nilpotent chains attached to each eigenvalue, while rational canonical form works over arbitrary fields using invariant factors. Both arise from viewing a vector space as a module over the polynomial ring via $p\cdot v=p(T)v$.

\section*{Learning goals}
By the end of the chapter, the reader should be able to:
\begin{itemize}
\item explain and use generalized eigenvectors in basis-free and coordinate form;
\item explain and use nilpotent operators in basis-free and coordinate form;
\item explain and use jordan blocks in basis-free and coordinate form;
\item explain and use jordan canonical form in basis-free and coordinate form;
\item explain and use primary decomposition in basis-free and coordinate form;
\item explain and use rational canonical form in basis-free and coordinate form;
\end{itemize}

\section*{Conceptual roadmap}
The chapter is organized around a repeated pattern:
\[
\boxed{\text{definition}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{coordinate calculation}\;\longrightarrow\;\text{application}.}
\]
Whenever possible, we separate the invariant mathematical object from a chosen coordinate representation.

\section{Generalized eigenvectors}
A vector lies in the generalized eigenspace for $\lambda$ if $(T-\lambda I)^k v=0$ for some $k$.

\section{Nilpotent operators}
Nilpotent operators organize into chains $v,Nv,\ldots$. Their Jordan blocks encode chain lengths.

\section{Jordan blocks}
A Jordan block $J_r(\lambda)$ has $\lambda$ on the diagonal and ones on the superdiagonal. Its minimal polynomial is $(t-\lambda)^r$.

\section{Jordan canonical form}
Over an algebraically closed field, every operator is similar to a direct sum of Jordan blocks, unique up to block order.

\section{Primary decomposition}
If the minimal polynomial factors into pairwise coprime powers, the space decomposes into invariant primary components.

\section{Rational canonical form}
Over any field, invariant factors determine a block diagonal companion-matrix form. Jordan form is the split refinement of this module-theoretic structure.

\section{Core structural results}

\begin{theorem}[Generalized eigenspace decomposition]\label{thm:i12-01}
If $m_T(t)=\prod_i(t-\lambda_i)^{r_i}$ with distinct $\lambda_i$, then $V=\bigoplus_i\ker(T-\lambda_iI)^{r_i}$.
\begin{proof}
The factors are pairwise coprime. Bézout identities in $\mathbb F[t]$ produce polynomial projectors whose images are the primary components.
\end{proof}
\end{theorem}

\begin{theorem}[Jordan block invariants]\label{thm:i12-02}
For a nilpotent operator, the numbers $\dim\ker N^k$ determine the Jordan block sizes.
\begin{proof}
The increment $\dim\ker N^k-\dim\ker N^{k-1}$ counts blocks of size at least $k$; successive differences recover the exact sizes.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Single Jordan block]\label{ex:i12-worked-01}
$J_3(0)$ is nilpotent of index $3$; kernel dimensions are $1,2,3$.
\end{example}

\begin{example}[Two blocks]\label{ex:i12-worked-02}
$J_2(0)\oplus J_1(0)$ has kernel dimensions $2,3$, revealing two blocks, one of which has length at least $2$.
\end{example}

\begin{example}[Companion matrix]\label{ex:i12-worked-03}
The companion matrix of a monic polynomial has that polynomial as both characteristic and minimal polynomial when it represents a cyclic module.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Nilpotent chain]\label{prob:i12-dossier-01}
For $N=J_3(0)$ compute $\dim\ker N$, $\dim\ker N^2$, and $\dim\ker N^3$.
\end{problem}
\begin{solution}
The standard Jordan chain has lengths three. The kernels have dimensions $1,2,3$ respectively. The increments $1,1,1$ indicate one block of size at least $1,2,3$.
\end{solution}

\begin{problem}[Two nilpotent blocks]\label{prob:i12-dossier-02}
If a nilpotent operator has Jordan blocks of sizes $3$ and $1$, compute $\dim\ker N^k$ for $k=1,2,3$.
\end{problem}
\begin{solution}
At $k=1$, each block contributes one kernel dimension, giving $2$. At $k=2$, the size-3 block contributes $2$ and the size-1 block $1$, giving $3$. At $k=3$ the total is $4$.
\end{solution}

\begin{problem}[Geometric multiplicity in Jordan form]\label{prob:i12-dossier-03}
How is the geometric multiplicity of $\lambda$ read from Jordan form?
\end{problem}
\begin{solution}
Each Jordan block for $\lambda$ contributes exactly one eigenvector direction, so the geometric multiplicity is the number of Jordan blocks carrying $\lambda$.
\end{solution}

\begin{problem}[Minimal polynomial from blocks]\label{prob:i12-dossier-04}
If the largest Jordan block for eigenvalue $2$ has size $4$ and for eigenvalue $-1$ has size $2$, find the minimal polynomial.
\end{problem}
\begin{solution}
The exponents are the largest block sizes, so $m_T(t)=(t-2)^4(t+1)^2$.
\end{solution}

\begin{problem}[Diagonalizability from blocks]\label{prob:i12-dossier-05}
When is a Jordan form diagonal?
\end{problem}
\begin{solution}
Exactly when every Jordan block has size $1$. Equivalently every eigenvalue's generalized eigenspace is already its ordinary eigenspace.
\end{solution}

\begin{problem}[Generalized eigenspace]\label{prob:i12-dossier-06}
Show $\ker(T-\lambda I)^k$ is $T$-invariant.
\end{problem}
\begin{solution}
$T$ commutes with every polynomial in $T$, so $(T-\lambda I)^kT(v)=T(T-\lambda I)^kv$. If the latter is zero, $Tv$ lies in the same kernel.
\end{solution}

\begin{problem}[Primary decomposition idea]\label{prob:i12-dossier-07}
Why do coprime factors of the minimal polynomial lead to a direct-sum decomposition?
\end{problem}
\begin{solution}
Bézout identities between the coprime polynomial factors produce polynomial projectors in $T$ whose images are the corresponding primary kernels and whose sum is the identity.
\end{solution}

\begin{problem}[Real versus complex canonical form]\label{prob:i12-dossier-08}
Why may Jordan form fail over $\mathbb R$?
\end{problem}
\begin{solution}
Jordan form requires the minimal polynomial to split into linear factors. A real operator can have irreducible quadratic factors, such as a planar rotation. Rational canonical form does not require splitting.
\end{solution}

\begin{problem}[Companion matrix]\label{prob:i12-dossier-09}
What is the conceptual role of a companion matrix?
\end{problem}
\begin{solution}
It represents multiplication by $t$ on the cyclic module $\mathbb F[t]/(p)$. Its characteristic and minimal polynomial are both the defining monic polynomial $p$ when the module is cyclic.
\end{solution}

\begin{problem}[Kernel increments]\label{prob:i12-dossier-10}
How do the increments $d_k=\dim\ker N^k-\dim\ker N^{k-1}$ encode block sizes?
\end{problem}
\begin{solution}
$d_k$ counts the number of Jordan blocks of size at least $k$. Therefore $d_k-d_{k+1}$ counts blocks of size exactly $k$.
\end{solution}

\begin{problem}[Similarity invariants]\label{prob:i12-dossier-11}
Name data preserved by similarity that are visible in Jordan form.
\end{problem}
\begin{solution}
Eigenvalues, algebraic and geometric multiplicities, minimal polynomial, and the multiset of Jordan block sizes are all similarity invariants. Jordan form packages them completely over an algebraically closed field.
\end{solution}

\begin{problem}[Canonical-form synthesis]\label{prob:i12-dossier-12}
Why is the module viewpoint $p\cdot v=p(T)v$ natural?
\end{problem}
\begin{solution}
It turns invariant subspaces into submodules and polynomial identities into annihilators. The structure theorem for finitely generated modules over $\mathbb F[t]$ then yields rational canonical form, with Jordan form as the split primary refinement.
\end{solution}

\section*{Chapter summary}
When diagonalization fails, canonical forms measure the failure rather than hiding it. Jordan form over an algebraically closed field records nilpotent chains attached to each eigenvalue, while rational canonical form works over arbitrary fields using invariant factors.
