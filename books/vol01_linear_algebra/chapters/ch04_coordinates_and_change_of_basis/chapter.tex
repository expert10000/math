\chapter{Coordinates and Change of Basis}
\label{ch:i04-coordinates-and-change-of-basis}

Coordinates turn an abstract vector space into a copy of $\mathbb F^n$, but the copy depends on a chosen basis. Change-of-basis matrices explain precisely how two coordinate descriptions of the same vector are related. This is the first systematic appearance of conjugation and covariance.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute the coordinate vector of an abstract vector relative to a specified ordered basis;
\item construct a change-of-basis matrix from one ordered basis to another;
\item convert coordinate vectors between two bases using the change-of-basis matrix and its inverse;
\item derive the similarity relation between matrices of the same operator in different bases;
\item choose a basis adapted to a subspace or decomposition in order to simplify coordinates;
\item detect and correct an invalid change-of-basis computation by checking what the columns represent;
\end{itemize}
\section*{Conceptual roadmap}
The chapter is organized around a repeated pattern:
\[
\boxed{\text{definition}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{coordinate calculation}\;\longrightarrow\;\text{application}.}
\]
Whenever possible, we separate the invariant mathematical object from a chosen coordinate representation.

\section{Coordinate isomorphisms}
For an ordered basis $B=(v_1,\ldots,v_n)$, the coordinate map $\kappa_B:V\to\mathbb F^n$, $v\mapsto[v]_B$, is a linear isomorphism.


% BEGIN VOL01-EXPANSION I04-example-01
\begin{example}[A coordinate isomorphism in practice]\label{ex:i04-expand-01}
For \(B=((1,1),(1,-1))\), the coordinate map sends
\[
(x,y)\longmapsto [ (x,y) ]_B
=\begin{pmatrix}(x+y)/2\\(x-y)/2\end{pmatrix}.
\]
It is linear, and reconstructing gives
\[
\frac{x+y}{2}(1,1)+\frac{x-y}{2}(1,-1)=(x,y).
\]
Thus the coordinate map is visibly invertible.

This example makes the abstraction concrete: choosing a basis is choosing a
linear isomorphism from the vector space to a coordinate space.
\end{example}
% END VOL01-EXPANSION I04-example-01
\section{Transition matrices}
Given bases $B,C$, the matrix $P_{C\leftarrow B}$ is defined by $[v]_C=P_{C\leftarrow B}[v]_B$. Its columns are the $C$-coordinates of the vectors of $B$.


% BEGIN VOL01-EXPANSION I04-example-02
\begin{example}[Building and checking a transition matrix]\label{ex:i04-expand-02}
Let \(B=((1,1),(1,-1))\) and \(E\) be the standard basis. Since the columns of
\(P_{E\leftarrow B}\) are the \(E\)-coordinates of the \(B\)-vectors,
\[
P_{E\leftarrow B}=\begin{pmatrix}1&1\\1&-1\end{pmatrix}.
\]
For \(v=3b_1+2b_2\),
\[
[v]_E=P_{E\leftarrow B}\begin{pmatrix}3\\2\end{pmatrix}
=\begin{pmatrix}5\\1\end{pmatrix}.
\]
Direct reconstruction gives the same vector, which checks both the columns and
the direction of the arrow in the notation.
\end{example}
% END VOL01-EXPANSION I04-example-02
\section{Inverse transitions}
Changing from $B$ to $C$ and back gives $P_{B\leftarrow C}=P_{C\leftarrow B}^{-1}$.

\section{Coordinate covariants}
Vectors and linear maps do not change when the basis changes; their coordinate arrays do. Good notation separates the object from its representation.

\section{Dual coordinates}
Linear functionals transform by the inverse transpose of the vector change-of-basis matrix. This anticipates the distinction between vectors and covectors.

\section{Conditioning of bases}
A mathematically valid basis can be numerically poor if its vectors are nearly dependent. Orthogonal bases mitigate this instability.


% BEGIN VOL01-EXPANSION I04-example-03
\begin{example}[Near dependence and numerical instability]\label{ex:i04-expand-03}
Take \(B_\varepsilon=((1,0),(1,\varepsilon))\). Its basis matrix is
\[
P_\varepsilon=\begin{pmatrix}1&1\\0&\varepsilon\end{pmatrix},
\qquad
P_\varepsilon^{-1}
=\begin{pmatrix}1&-1/\varepsilon\\0&1/\varepsilon\end{pmatrix}.
\]
For every nonzero \(\varepsilon\) this is a valid basis, but when
\(|\varepsilon|\) is small, the inverse has very large entries.

So algebraic invertibility and numerical reliability are different questions.
The determinant confirms validity; the size of the inverse warns about
sensitivity.
\end{example}
% END VOL01-EXPANSION I04-example-03
\section{Core structural results}

\begin{theorem}[Transition matrix rule]\label{thm:i04-01}
The columns of $P_{C\leftarrow B}$ are $[v_1]_C,\ldots,[v_n]_C$ when $B=(v_1,\ldots,v_n)$.
\begin{proof}
Apply the transition identity to each basis vector. Its $B$-coordinate is the corresponding standard vector, so the resulting $C$-coordinate is the corresponding column.
\end{proof}
\end{theorem}

\begin{theorem}[Cocycle identity]\label{thm:i04-02}
For bases $B,C,D$, $P_{D\leftarrow B}=P_{D\leftarrow C}P_{C\leftarrow B}$.
\begin{proof}
Apply the two coordinate changes successively to an arbitrary coordinate vector and use uniqueness of the representing matrix.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Two planar bases]\label{ex:i04-worked-01}
For $B=((1,0),(1,1))$ and standard $E$, $P_{E\leftarrow B}=\begin{pmatrix}1&1\\0&1\end{pmatrix}$.
\end{example}

\begin{example}[Polynomial bases]\label{ex:i04-worked-02}
The monomial basis and shifted basis $1,(x-1),(x-1)^2$ of $\mathcal P_2$ are related by expanding powers of $x-1$.
\end{example}

\begin{example}[Scaling a basis]\label{ex:i04-worked-03}
Replacing $v_i$ by $c_i v_i$ multiplies the corresponding transition column by $c_i$.
\end{example}

\section{Diagnostic checklist}
Before moving on, it is useful to distinguish four levels of understanding:
\begin{enumerate}
\item recognize the definitions in unfamiliar examples;
\item prove the core structural statements without coordinates when possible;
\item translate the same statement into a matrix or coordinate computation;
\item know which hypotheses are essential and produce a counterexample when one is removed.
\end{enumerate}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Transition matrix from columns]\label{prob:i04-dossier-01}
For $B=((1,1),(1,-1))$ and the standard basis $E$, compute $P_{E\leftarrow B}$ and its inverse.
\end{problem}
\begin{solution}
The columns are the $E$-coordinates of the $B$-vectors, so $P_{E\leftarrow B}=\begin{pmatrix}1&1\\1&-1\end{pmatrix}$. Its inverse is $\frac12\begin{pmatrix}1&1\\1&-1\end{pmatrix}$, which is $P_{B\leftarrow E}$.
\end{solution}

\begin{problem}[Coordinates of a polynomial]\label{prob:i04-dossier-02}
Find the coordinates of $x^2+1$ in $B=(1,x-1,(x-1)^2)$.
\end{problem}
\begin{solution}
Write $x=(x-1)+1$. Then $x^2+1=(x-1)^2+2(x-1)+2$, so the coordinate vector is $(2,2,1)^T$.
\end{solution}

\begin{problem}[Cocycle identity]\label{prob:i04-dossier-03}
Given bases $B,C,D$, prove $P_{D\leftarrow B}=P_{D\leftarrow C}P_{C\leftarrow B}$.
\end{problem}
\begin{solution}
For every vector $v$, first change $B$-coordinates to $C$-coordinates and then to $D$-coordinates. Thus $[v]_D=P_{D\leftarrow C}P_{C\leftarrow B}[v]_B$. Uniqueness of the matrix realizing this coordinate transformation gives the identity.
\end{solution}

\begin{problem}[A badly conditioned basis]\label{prob:i04-dossier-04}
Why is $B_\varepsilon=((1,0),(1,\varepsilon))$ a basis for $\varepsilon\ne0$ but numerically poor when $|\varepsilon|$ is small?
\end{problem}
\begin{solution}
The determinant of the basis matrix is $\varepsilon$, so it is invertible for $\varepsilon\ne0$. Its inverse contains entries of size $1/\varepsilon$, so small perturbations in coordinates can be greatly amplified as $\varepsilon\to0$.
\end{solution}

\begin{problem}[Permutation of basis vectors]\label{prob:i04-dossier-05}
If $C=(v_2,v_1,v_3)$ is obtained from $B=(v_1,v_2,v_3)$ by swapping the first two vectors, describe $P_{C\leftarrow B}$.
\end{problem}
\begin{solution}
It is the permutation matrix that swaps the first two coordinates: $\begin{pmatrix}0&1&0\\1&0&0\\0&0&1\end{pmatrix}$. The same matrix is its inverse.
\end{solution}

\begin{problem}[Scaling a basis]\label{prob:i04-dossier-06}
Let $C=(2v_1,3v_2)$ when $B=(v_1,v_2)$. Find $P_{C\leftarrow B}$.
\end{problem}
\begin{solution}
If $v=a v_1+b v_2=c(2v_1)+d(3v_2)$, then $c=a/2$ and $d=b/3$. Thus $P_{C\leftarrow B}=\operatorname{diag}(1/2,1/3)$.
\end{solution}

\begin{problem}[Coordinates versus objects]\label{prob:i04-dossier-07}
Two coordinate columns for the same vector are different. Explain why this does not contradict uniqueness of coordinates.
\end{problem}
\begin{solution}
Coordinates are unique only after the basis is fixed. Different bases define different coordinate isomorphisms $V\to\mathbb F^n$, so the same abstract vector can have different coordinate columns without ambiguity.
\end{solution}

\begin{problem}[Dual transformation]\label{prob:i04-dossier-08}
Suppose vector coordinates change by $x_C=Px_B$. How must a row covector $\ell$ change so that the scalar $\ell(x)$ is invariant?
\end{problem}
\begin{solution}
We require $\ell_C x_C=\ell_B x_B$ for all $x_B$. Since $x_C=Px_B$, this gives $\ell_C P=\ell_B$, hence $\ell_C=\ell_B P^{-1}$. In column-covector convention this becomes the inverse transpose.
\end{solution}

\begin{problem}[Coordinate map isomorphism]\label{prob:i04-dossier-09}
Prove the coordinate map $\kappa_B:V\to\mathbb F^n$ is linear and bijective.
\end{problem}
\begin{solution}
Linearity follows by combining coefficients in the basis expansion. Every coordinate vector gives a linear combination of the basis vectors, so the map is onto. Uniqueness of basis coordinates gives injectivity.
\end{solution}

\begin{problem}[Three-basis computation]\label{prob:i04-dossier-10}
If $P_{C\leftarrow B}=A$ and $P_{D\leftarrow C}=B$, what is $P_{B\leftarrow D}$?
\end{problem}
\begin{solution}
First $P_{D\leftarrow B}=BA$. Therefore the reverse transition is $(BA)^{-1}=A^{-1}B^{-1}$.
\end{solution}

\begin{problem}[Basis from columns]\label{prob:i04-dossier-11}
Why does every invertible matrix $P$ determine a new ordered basis relative to a fixed basis $E$?
\end{problem}
\begin{solution}
Take as new basis vectors the vectors whose $E$-coordinate columns are the columns of $P$. Invertibility means those columns are independent and span the whole space, so they form a basis.
\end{solution}

\begin{problem}[Coordinate synthesis]\label{prob:i04-dossier-12}
Explain why changing basis is not the same operation as applying a linear transformation to vectors.
\end{problem}
\begin{solution}
A basis change changes only the description of the same vector: the geometric object is fixed and coordinates are transformed by a transition matrix. Applying an operator changes the vector itself. Confusing these roles is the source of many incorrect conjugation formulas.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:i04-01}
Compute $P_{E\leftarrow B}$ for $B=((1,1),(1,-1))$.
\end{exercise}
\begin{hint}
Write each new basis vector in the old coordinates and use those coordinate columns to build the conversion matrix. Solve \(v=a_1b_1+\cdots+a_nb_n\); the coefficients you obtain are the coordinate vector.
\end{hint}
\begin{solution}
Its columns are the basis vectors in standard coordinates: $\begin{pmatrix}1&1\\1&-1\end{pmatrix}$.
\end{solution}

\begin{exercise}\label{exr:i04-02}
Find $P_{B\leftarrow E}$ for the preceding basis.
\end{exercise}
\begin{hint}
Build the change-of-basis matrix column by column from the new basis vectors written in the old coordinates.
\end{hint}
\begin{solution}
Invert the matrix to get $\frac12\begin{pmatrix}1&1\\1&-1\end{pmatrix}$.
\end{solution}

\begin{exercise}\label{exr:i04-03}
Verify the cocycle identity for three arbitrary bases.
\end{exercise}
\begin{hint}
Check the direction of the change-of-basis matrix by applying it to a coordinate vector whose answer you know.
\end{hint}
\begin{solution}
For every $v$, substitute $[v]_C=P_{C\leftarrow B}[v]_B$ into $[v]_D=P_{D\leftarrow C}[v]_C$; equality for all coordinate vectors gives the matrix identity.
\end{solution}

\begin{exercise}\label{exr:i04-04}
Why must every transition matrix be invertible?
\end{exercise}
\begin{hint}
The inverse change-of-basis matrix reverses the coordinate conversion; verify by composing the two conversions.
\end{hint}
\begin{solution}
It represents the composition of two coordinate isomorphisms, hence an isomorphism.
\end{solution}

\begin{exercise}\label{exr:i04-05}
Compute coordinates of $p(x)=x^2$ in $B=(1,x-1,(x-1)^2)$.
\end{exercise}
\begin{hint}
For an operator matrix, insert identity changes of coordinates on the domain and codomain before simplifying.
\end{hint}
\begin{solution}
Since $x=(x-1)+1$, $x^2=(x-1)^2+2(x-1)+1$, so the coordinates are $(1,2,1)^T$.
\end{solution}

\begin{exercise}\label{exr:i04-06}
How do coordinates change if the basis order is permuted?
\end{exercise}
\begin{hint}
Write each new basis vector in the old coordinates and use those coordinate columns to build the conversion matrix. Similarity should appear as \(P^{-1}AP\); determine which side corresponds to changing input coordinates first.
\end{hint}
\begin{solution}
The coordinate column is permuted inversely to the basis order; the transition is represented by a permutation matrix.
\end{solution}

\begin{exercise}\label{exr:i04-07}
Explain why the vector itself is basis-independent.
\end{exercise}
\begin{hint}
Write each new basis vector in the old coordinates and use those coordinate columns to build the conversion matrix. An adapted basis should begin with a basis of the distinguished subspace and then be extended to the whole space.
\end{hint}
\begin{solution}
A basis selects a coordinate isomorphism but does not alter the element $v\in V$; only its numerical representative changes.
\end{solution}

\begin{exercise}\label{exr:i04-08}
Why are nearly parallel planar basis vectors numerically problematic?
\end{exercise}
\begin{hint}
Test a proposed formula on each basis vector; a matrix representation is determined by those columns.
\end{hint}
\begin{solution}
The transition matrix is close to singular, so its inverse can have large entries and amplify small data errors.
\end{solution}

\section{Graded supplementary exercises}
The following exercises extend the original exercise set without replacing it. They are grouped by mathematical role and increase in synthesis.

\subsection*{Standard computations}

\begin{exercise}[Transition columns]\label{exr:i04-expand-01}
For \(B=((1,1),(2,1))\), compute \(P_{E\leftarrow B}\).
\end{exercise}
\begin{hint}
Use the basis vectors as columns in standard coordinates.
\end{hint}
\begin{solution}
\(\begin{pmatrix}1&2\\1&1\end{pmatrix}\).
\end{solution}

\begin{exercise}[Reverse transition]\label{exr:i04-expand-02}
Find \(P_{B\leftarrow E}\) for the preceding basis.
\end{exercise}
\begin{hint}
Invert \(P_{E\leftarrow B}\).
\end{hint}
\begin{solution}
The determinant is \(-1\), so \(P_{B\leftarrow E}=\begin{pmatrix}-1&2\\1&-1\end{pmatrix}\).
\end{solution}

\begin{exercise}[Coordinate conversion]\label{exr:i04-expand-03}
If \([v]_B=(3,-1)^T\) and \(P_{E\leftarrow B}=\begin{pmatrix}1&2\\1&1\end{pmatrix}\), find \([v]_E\).
\end{exercise}
\begin{hint}
Multiply the transition matrix by the coordinate column.
\end{hint}
\begin{solution}
\([v]_E=(1,2)^T\).
\end{solution}

\begin{exercise}[Shifted polynomial basis]\label{exr:i04-expand-04}
Find \([x^2]_{(1,x-1,(x-1)^2)}\).
\end{exercise}
\begin{hint}
Write \(x=(x-1)+1\) and expand.
\end{hint}
\begin{solution}
\(x^2=1+2(x-1)+(x-1)^2\), so coordinates are \((1,2,1)^T\).
\end{solution}

\begin{exercise}[Cocycle]\label{exr:i04-expand-05}
If \(P_{C\leftarrow B}=A\), \(P_{D\leftarrow C}=B_0\), express \(P_{D\leftarrow B}\).
\end{exercise}
\begin{hint}
Apply the two coordinate changes in succession.
\end{hint}
\begin{solution}
\(P_{D\leftarrow B}=B_0A\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Transition invertibility]\label{exr:i04-expand-06}
Prove every transition matrix is invertible.
\end{exercise}
\begin{hint}
The reverse coordinate change provides an inverse map.
\end{hint}
\begin{solution}
\(P_{B\leftarrow C}P_{C\leftarrow B}=I\), so the matrices are inverses.
\end{solution}

\begin{exercise}[Cocycle identity]\label{exr:i04-expand-07}
Prove \(P_{D\leftarrow B}=P_{D\leftarrow C}P_{C\leftarrow B}\).
\end{exercise}
\begin{hint}
Apply both sides to an arbitrary \(B\)-coordinate vector.
\end{hint}
\begin{solution}
Both products produce the unique \(D\)-coordinate column of the same vector.
\end{solution}

\begin{exercise}[Coordinate map linearity]\label{exr:i04-expand-08}
Prove \(v\mapsto[v]_B\) is linear.
\end{exercise}
\begin{hint}
Compare basis coefficients of \(av+bw\).
\end{hint}
\begin{solution}
Coefficients combine componentwise, so \([av+bw]_B=a[v]_B+b[w]_B\).
\end{solution}

\begin{exercise}[Similarity derivation]\label{exr:i04-expand-09}
Derive \([T]_C=P^{-1}[T]_B P\) for \(P=P_{B\leftarrow C}\).
\end{exercise}
\begin{hint}
Follow a \(C\)-coordinate vector into \(B\), apply \(T\), then return.
\end{hint}
\begin{solution}
The three steps are \(P\), \([T]_B\), \(P^{-1}\), giving the conjugation formula.
\end{solution}

\subsection*{Counterexamples}

\begin{exercise}[Wrong direction]\label{exr:i04-expand-10}
Explain why using \(P_{C\leftarrow B}\) where \(P_{B\leftarrow C}\) is required can give a plausible but wrong result.
\end{exercise}
\begin{hint}
Test the matrix on a basis vector whose converted coordinates are known.
\end{hint}
\begin{solution}
The two matrices are inverses, generally unequal; a basis-vector test exposes the reversed direction.
\end{solution}

\begin{exercise}[Singular columns]\label{exr:i04-expand-11}
Why can a singular matrix not be a transition matrix?
\end{exercise}
\begin{hint}
A transition matrix represents an isomorphism between coordinate spaces.
\end{hint}
\begin{solution}
A singular matrix is not invertible, so it cannot encode reversible basis conversion.
\end{solution}

\begin{exercise}[Coordinates are not vectors]\label{exr:i04-expand-12}
Give an example where the same vector has different coordinate columns in two bases.
\end{exercise}
\begin{hint}
Use \(v=(1,1)\), standard basis, and \(B=((1,1),(1,-1))\).
\end{hint}
\begin{solution}
\([v]_E=(1,1)^T\) but \([v]_B=(1,0)^T\).
\end{solution}

\subsection*{Applications}

\begin{exercise}[Adapted basis]\label{exr:i04-expand-13}
For \(U=\operatorname{span}(1,1)\subset\mathbb R^2\), choose a basis adapted to \(U\).
\end{exercise}
\begin{hint}
Begin with a basis of \(U\) and add any vector outside it.
\end{hint}
\begin{solution}
\(((1,1),(1,-1))\) is adapted; \(U\) is the first coordinate axis in these coordinates.
\end{solution}

\begin{exercise}[Numerical conditioning]\label{exr:i04-expand-14}
Why is \(B_\varepsilon=((1,0),(1,\varepsilon))\) unstable for small \(\varepsilon\)?
\end{exercise}
\begin{hint}
Inspect the inverse of its basis matrix.
\end{hint}
\begin{solution}
The inverse contains \(1/\varepsilon\), so small input perturbations can cause large coordinate changes.
\end{solution}

\subsection*{Challenges}

\begin{exercise}[Dual coordinate law]\label{exr:i04-expand-15}
If vector columns transform by \(x_C=Px_B\), derive the row-covector law preserving \(\ell(x)\).
\end{exercise}
\begin{hint}
Require \(\ell_Cx_C=\ell_Bx_B\) for all \(x_B\).
\end{hint}
\begin{solution}
\(\ell_CP=\ell_B\), so \(\ell_C=\ell_BP^{-1}\); in column convention this is inverse transpose.
\end{solution}

\begin{exercise}[Three basis reversal]\label{exr:i04-expand-16}
If \(P_{C\leftarrow B}=A\), \(P_{D\leftarrow C}=B_0\), find \(P_{B\leftarrow D}\).
\end{exercise}
\begin{hint}
First compose forward, then invert.
\end{hint}
\begin{solution}
\(P_{D\leftarrow B}=B_0A\), hence \(P_{B\leftarrow D}=A^{-1}B_0^{-1}\).
\end{solution}

\section*{Chapter summary}
Coordinates turn an abstract vector space into a copy of $\mathbb F^n$, but the copy depends on a chosen basis. Change-of-basis matrices explain precisely how two coordinate descriptions of the same vector are related. This is the first systematic appearance of conjugation and covariance.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
