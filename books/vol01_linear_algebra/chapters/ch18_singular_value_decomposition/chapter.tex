\chapter{Singular-Value Decomposition}
\label{ch:i18-singular-value-decomposition}

The singular-value decomposition is the universal orthogonal factorization of a linear map between finite-dimensional inner-product spaces. Unlike eigenvalue decomposition, it applies to every rectangular matrix. Singular values quantify stretching, rank, conditioning and optimal low-rank approximation.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute singular values from the eigenvalues of \(A^*A\);
\item construct left and right singular vectors and assemble a singular-value decomposition;
\item read rank, operator norm, and null-space information from the singular values;
\item compute the Moore--Penrose pseudoinverse from an SVD;
\item solve a least-squares problem with the pseudoinverse, including rank-deficient cases;
\item form and interpret the best low-rank approximation obtained by truncating the SVD;
\end{itemize}

\section{Singular values}
The singular values of $A$ are the nonnegative square roots of the eigenvalues of $A^*A$.


% BEGIN VOL01-EXPANSION I18-example-01
\begin{example}[Constructing an SVD from \(A^TA\)]\label{ex:i18-expand-01}
Let
\[
A=\begin{pmatrix}3&0\\0&1\\0&0\end{pmatrix}.
\]
Then
\[
A^TA=\begin{pmatrix}9&0\\0&1\end{pmatrix},
\]
so the singular values are \(3\) and \(1\), with right singular vectors
\(v_1=e_1,v_2=e_2\). The left singular vectors are
\[
u_1=\frac{Av_1}{3}=e_1,\qquad
u_2=\frac{Av_2}{1}=e_2.
\]
Extending by \(u_3=e_3\) gives \(A=U\Sigma V^T\). Multiplying the factors back
recovers \(A\), providing the final construction check.
\end{example}
% END VOL01-EXPANSION I18-example-01
\section{SVD theorem}
Every $m\times n$ matrix has a factorization $A=U\Sigma V^*$ with $U,V$ unitary (orthogonal over $\mathbb R$) and $\Sigma$ diagonal rectangular with nonnegative entries.

\section{Construction from the spectral theorem}
Diagonalize $A^*A$ to obtain right singular vectors; normalize their images under $A$ to obtain left singular vectors.

\section{Rank and norms}
The rank is the number of positive singular values. The operator norm is the largest singular value, and the Frobenius norm squared is the sum of squared singular values.

\section{Pseudoinverse}
Replacing each nonzero singular value by its reciprocal gives the Moore–Penrose pseudoinverse.


% BEGIN VOL01-EXPANSION I18-example-02
\begin{example}[Pseudoinverse and a rank-deficient least-squares problem]\label{ex:i18-expand-02}
For
\[
A=\begin{pmatrix}1&1\\1&1\end{pmatrix},
\]
the rank is one. Write
\[
A=2uu^T,\qquad u=\frac1{\sqrt2}(1,1).
\]
Hence
\[
A^+=\frac12uu^T
=\frac14\begin{pmatrix}1&1\\1&1\end{pmatrix}.
\]
For \(b=(2,0)^T\),
\[
x_*=A^+b=(1/2,1/2)^T.
\]
The fitted vector \(Ax_*=(1,1)^T\) is the projection of \(b\) onto
\(\operatorname{im}A\), and among all coefficient vectors producing that fit,
\(x_*\) has minimum norm.
\end{example}
% END VOL01-EXPANSION I18-example-02
\section{Low-rank approximation}
Truncating the SVD after the largest $r$ singular values gives a best rank-$r$ approximation in operator and Frobenius norms.


% BEGIN VOL01-EXPANSION I18-example-03
\begin{example}[Low-rank approximation and error read directly from singular values]\label{ex:i18-expand-03}
Suppose a matrix has singular values
\[
10,\quad 3,\quad 1,\quad 0.2.
\]
The best rank-two approximation keeps only the first two SVD terms. The optimal
spectral-norm error is
\[
\sigma_3=1,
\]
while the Frobenius error is
\[
\sqrt{1^2+0.2^2}=\sqrt{1.04}.
\]
Thus the singular-value tail quantifies exactly what is lost by compression.
There is no need to search over all rank-two matrices: the truncated SVD is
already optimal.
\end{example}
% END VOL01-EXPANSION I18-example-03
\section{Core structural results}

\begin{theorem}[Singular-value decomposition]\label{thm:i18-01}
For every matrix $A\in\mathbb F^{m\times n}$, there exist unitary $U,V$ and a rectangular diagonal $\Sigma$ with nonnegative entries such that $A=U\Sigma V^*$.
\begin{proof}
Apply the spectral theorem to the positive semidefinite matrix $A^*A$. For each positive eigenvalue $\sigma_i^2$, set $u_i=Av_i/\sigma_i$; extend the resulting orthonormal lists to bases.
\end{proof}
\end{theorem}

\begin{theorem}[Rank from singular values]\label{thm:i18-02}
The number of nonzero singular values equals $\operatorname{rank}A$.
\begin{proof}
The kernels of $A$ and $A^*A$ coincide, so they have the same nullity. Rank–nullity then shows $A^*A$ has as many positive eigenvalues as the rank of $A$.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Diagonal matrix]\label{ex:i18-worked-01}
For a real diagonal matrix, singular values are the absolute values of its diagonal entries.
\end{example}

\begin{example}[Rank-one matrix]\label{ex:i18-worked-02}
If $A=uv^*$, there is a single nonzero singular value $\|u\|\|v\|$.
\end{example}

\begin{example}[Condition number]\label{ex:i18-worked-03}
For invertible $A$, the Euclidean condition number is $\kappa_2(A)=\sigma_{\max}/\sigma_{\min}$.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Singular values of a diagonal matrix]\label{prob:i18-dossier-01}
Find the singular values of $\operatorname{diag}(3,-2,0)$.
\end{problem}
\begin{solution}
$A^TA=\operatorname{diag}(9,4,0)$, so the singular values are the nonnegative square roots $3,2,0$.
\end{solution}

\begin{problem}[Kernel identity]\label{prob:i18-dossier-02}
Prove $\ker(A^*A)=\ker A$.
\end{problem}
\begin{solution}
If $Av=0$, then $A^*Av=0$. Conversely, if $A^*Av=0$, then $0=\langle A^*Av,v\rangle=\langle Av,Av\rangle=\|Av\|^2$, so $Av=0$.
\end{solution}

\begin{problem}[Rank from singular values]\label{prob:i18-dossier-03}
Why does the number of nonzero singular values equal $\operatorname{rank}A$?
\end{problem}
\begin{solution}
$A^*A$ has the same kernel as $A$, hence the same nullity. Rank-nullity then implies the number of positive eigenvalues of $A^*A$, i.e. nonzero singular values, equals the rank of $A$.
\end{solution}

\begin{problem}[Rank-one SVD]\label{prob:i18-dossier-04}
If $A=uv^*$ with nonzero $u,v$, what is its nonzero singular value?
\end{problem}
\begin{solution}
Write $A=(u/\|u\|)(\|u\|\|v\|)(v/\|v\|)^*$. This is already a rank-one SVD, so the sole nonzero singular value is $\|u\|\|v\|$.
\end{solution}

\begin{problem}[Operator norm]\label{prob:i18-dossier-05}
Why is $\|A\|_2$ the largest singular value?
\end{problem}
\begin{solution}
In an SVD $A=U\Sigma V^*$, unitary factors preserve norm. Thus maximizing $\|Ax\|$ over unit $x$ reduces to maximizing $\|\Sigma y\|$, whose largest stretch is the largest diagonal singular value.
\end{solution}

\begin{problem}[Frobenius norm]\label{prob:i18-dossier-06}
Express $\|A\|_F^2$ in terms of singular values.
\end{problem}
\begin{solution}
Unitary invariance of the Frobenius norm gives $\|A\|_F^2=\|\Sigma\|_F^2=\sum_i\sigma_i^2$.
\end{solution}

\begin{problem}[Pseudoinverse]\label{prob:i18-dossier-07}
How is the Moore–Penrose pseudoinverse constructed from an SVD?
\end{problem}
\begin{solution}
If $A=U\Sigma V^*$, define $A^+=V\Sigma^+U^*$, where each nonzero singular value $\sigma_i$ is replaced by $1/\sigma_i$ and zeros remain zero.
\end{solution}

\begin{problem}[Condition number]\label{prob:i18-dossier-08}
For invertible $A$, why is $\kappa_2(A)=\sigma_{\max}/\sigma_{\min}$?
\end{problem}
\begin{solution}
$\|A\|_2=\sigma_{\max}$ and $\|A^{-1}\|_2=1/\sigma_{\min}$. Their product is the stated ratio.
\end{solution}

\begin{problem}[Rectangular geometry]\label{prob:i18-dossier-09}
Why does the SVD use two different orthonormal bases?
\end{problem}
\begin{solution}
A rectangular map has different domain and codomain spaces. Right singular vectors form an orthonormal basis of input directions; left singular vectors form orthonormal output directions. The diagonal $\Sigma$ records the stretch between matched directions.
\end{solution}

\begin{problem}[Low-rank truncation]\label{prob:i18-dossier-10}
What is the rank of the matrix obtained by keeping only the first $r$ nonzero SVD terms?
\end{problem}
\begin{solution}
Each term $\sigma_i u_iv_i^*$ has rank one and the corresponding left/right vectors are orthogonal. Keeping $r$ positive terms produces rank exactly $r$.
\end{solution}

\begin{problem}[Best approximation principle]\label{prob:i18-dossier-11}
State the Eckart–Young conclusion for truncated SVD.
\end{problem}
\begin{solution}
The matrix obtained by keeping the largest $r$ singular values is a best rank-$r$ approximation in both spectral and Frobenius norms. The spectral error is $\sigma_{r+1}$ and the squared Frobenius error is $\sum_{i>r}\sigma_i^2$.
\end{solution}

\begin{problem}[SVD synthesis]\label{prob:i18-dossier-12}
How does SVD unify rank, conditioning, least squares, and data compression?
\end{problem}
\begin{solution}
Nonzero singular values determine rank; their ratio controls conditioning; the pseudoinverse solves least-squares/minimum-norm problems; and decay of singular values quantifies how accurately the map can be compressed to low rank. One factorization exposes all four phenomena.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:i18-01}
Find the singular values of $\operatorname{diag}(3,-2)$.
\end{exercise}
\begin{hint}
First determine \(\lambda\) from \(\det(A-\lambda I)=0\), then solve \((A-\lambda I)v=0\) for the eigenspace. Compute \(A^*A\); the singular values are the square roots of its nonnegative eigenvalues.
\end{hint}
\begin{solution}
They are $3$ and $2$, the square roots of eigenvalues $9$ and $4$ of $A^TA$.
\end{solution}

\begin{exercise}\label{exr:i18-02}
Why is $A^*A$ positive semidefinite?
\end{exercise}
\begin{hint}
Choose orthonormal eigenvectors of \(A^*A\) as right singular vectors, then set \(u_i=Av_i/\sigma_i\) for \(\sigma_i>0\).
\end{hint}
\begin{solution}
$\langle A^*Av,v\rangle=\langle Av,Av\rangle=\|Av\|^2\ge0$.
\end{solution}

\begin{exercise}\label{exr:i18-03}
Show $\ker(A^*A)=\ker A$.
\end{exercise}
\begin{hint}
Count the nonzero singular values; that number is the rank.
\end{hint}
\begin{solution}
If $Av=0$ then clearly $A^*Av=0$. Conversely, $A^*Av=0$ implies $0=\langle A^*Av,v\rangle=\|Av\|^2$.
\end{solution}

\begin{exercise}\label{exr:i18-04}
What is the rank of a matrix with singular values $5,2,0,0$?
\end{exercise}
\begin{hint}
Start from \(A^*A\): its orthonormal eigenvectors are right singular vectors and its eigenvalues are the squared singular values. The operator norm induced by the Euclidean norm is the largest singular value.
\end{hint}
\begin{solution}
The rank is $2$, the number of nonzero singular values.
\end{solution}

\begin{exercise}\label{exr:i18-05}
Describe the pseudoinverse of a diagonal singular-value matrix.
\end{exercise}
\begin{hint}
Start from \(A^*A\): its orthonormal eigenvectors are right singular vectors and its eigenvalues are the squared singular values. For the pseudoinverse, reciprocate only the nonzero singular values and transpose the SVD factors.
\end{hint}
\begin{solution}
Transpose the rectangular shape and replace each positive $\sigma_i$ by $1/\sigma_i$, leaving zeros unchanged.
\end{solution}

\begin{exercise}\label{exr:i18-06}
Why is SVD available for rectangular matrices while diagonalization is not?
\end{exercise}
\begin{hint}
Find the eigenspaces first and use orthogonality between distinct eigenspaces before assembling an orthonormal eigenbasis. In a rank-deficient least-squares problem, the pseudoinverse selects the minimum-norm coefficient vector.
\end{hint}
\begin{solution}
SVD uses the square self-adjoint matrices $A^*A$ and $AA^*$ and two possibly different orthonormal bases; ordinary diagonalization requires an endomorphism.
\end{solution}

\begin{exercise}\label{exr:i18-07}
What does the largest singular value measure?
\end{exercise}
\begin{hint}
Start from \(A^*A\): its orthonormal eigenvectors are right singular vectors and its eigenvalues are the squared singular values. For best rank-\(k\) approximation, keep the \(k\) largest singular values and discard the rest.
\end{hint}
\begin{solution}
It is $\max_{\|x\|=1}\|Ax\|$, the maximum stretching factor.
\end{solution}

\begin{exercise}\label{exr:i18-08}
Explain truncated SVD conceptually.
\end{exercise}
\begin{hint}
Start from \(A^*A\): its orthonormal eigenvectors are right singular vectors and its eigenvalues are the squared singular values. Verify the SVD by multiplying \(U\Sigma V^*\) and by checking that the columns of \(U\) and \(V\) are orthonormal.
\end{hint}
\begin{solution}
Keep only the dominant orthogonal input/output modes and their singular values; the discarded modes contribute the smallest possible approximation error for the prescribed rank.
\end{solution}

\subsection*{Standard computations}

\begin{exercise}[Diagonal SVD]\label{exr:i18-expand-01}
Find the singular values of \(\operatorname{diag}(-4,2,0)\).
\end{exercise}
\begin{hint}
Take square roots of eigenvalues of \(A^TA\).
\end{hint}
\begin{solution}
The singular values are \(4,2,0\).
\end{solution}

\begin{exercise}[Rank from singular values]\label{exr:i18-expand-02}
A matrix has singular values \(7,2,0,0\). Find its rank.
\end{exercise}
\begin{hint}
Count positive singular values.
\end{hint}
\begin{solution}
The rank is \(2\).
\end{solution}

\begin{exercise}[Operator norm]\label{exr:i18-expand-03}
For singular values \(5,3,1\), compute \(\|A\|_2\).
\end{exercise}
\begin{hint}
The Euclidean operator norm is the largest singular value.
\end{hint}
\begin{solution}
\(\|A\|_2=5\).
\end{solution}

\begin{exercise}[Pseudoinverse diagonal]\label{exr:i18-expand-04}
Find the pseudoinverse of \(\Sigma=\operatorname{diag}(4,0,2)\).
\end{exercise}
\begin{hint}
Reciprocate nonzero singular values only.
\end{hint}
\begin{solution}
\(\Sigma^+=\operatorname{diag}(1/4,0,1/2)\).
\end{solution}

\begin{exercise}[Condition number]\label{exr:i18-expand-05}
An invertible matrix has singular values \(12,4,3\). Find \(\kappa_2(A)\).
\end{exercise}
\begin{hint}
Use \(\sigma_{\max}/\sigma_{\min}\).
\end{hint}
\begin{solution}
\(\kappa_2(A)=12/3=4\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Kernel identity]\label{exr:i18-expand-06}
Prove \(\ker(A^*A)=\ker A\).
\end{exercise}
\begin{hint}
Use \(\langle A^*Av,v\rangle=\|Av\|^2\).
\end{hint}
\begin{solution}
One inclusion is immediate; the reverse follows because \(A^*Av=0\) implies \(\|Av\|^2=0\).
\end{solution}

\begin{exercise}[Rank from SVD]\label{exr:i18-expand-07}
Prove the number of positive singular values equals \(\operatorname{rank}A\).
\end{exercise}
\begin{hint}
Use the kernel identity or the diagonal form \(U\Sigma V^*\).
\end{hint}
\begin{solution}
Unitary factors preserve rank, and the rank of \(\Sigma\) is exactly the number of its nonzero diagonal entries.
\end{solution}

\begin{exercise}[Operator norm]\label{exr:i18-expand-08}
Prove \(\|A\|_2=\sigma_{\max}\).
\end{exercise}
\begin{hint}
Move to SVD coordinates, where unitary factors preserve norms.
\end{hint}
\begin{solution}
For \(A=U\Sigma V^*\), maximizing \(\|Ax\|\) over unit \(x\) reduces to maximizing \(\|\Sigma y\|\), achieved in the largest-singular-value coordinate.
\end{solution}

\begin{exercise}[Minimum-norm pseudoinverse solution]\label{exr:i18-expand-09}
Why does \(A^+b\) give the minimum-norm solution when \(Ax=b\) is consistent?
\end{exercise}
\begin{hint}
Decompose coefficient space into \((\ker A)^\perp\oplus\ker A\).
\end{hint}
\begin{solution}
The pseudoinverse solution lies in \((\ker A)^\perp\); every other solution adds a kernel vector orthogonal to it, increasing norm by Pythagoras.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Eigenvalues versus singular values]\label{exr:i18-expand-10}
Show singular values are not generally absolute values of eigenvalues.
\end{exercise}
\begin{hint}
Use a nonnormal nilpotent matrix.
\end{hint}
\begin{solution}
For \(A=\begin{pmatrix}0&1\\0&0\end{pmatrix}\), both eigenvalues are \(0\), but singular values are \(1,0\).
\end{solution}

\begin{exercise}[Normal equations sensitivity]\label{exr:i18-expand-11}
Why can \(A^TA\) hide numerical accuracy even though its eigenvalues give singular values squared?
\end{exercise}
\begin{hint}
Compare condition numbers.
\end{hint}
\begin{solution}
\(\kappa_2(A^TA)=\kappa_2(A)^2\), so relative sensitivity is squared.
\end{solution}

\begin{exercise}[Truncating arbitrary modes]\label{exr:i18-expand-12}
Why is deleting arbitrary columns or coordinates not equivalent to optimal low-rank approximation?
\end{exercise}
\begin{hint}
SVD modes are selected by singular value and orthogonal input/output directions.
\end{hint}
\begin{solution}
Arbitrary coordinate truncation depends on basis and need not minimize either spectral or Frobenius error.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Data compression]\label{exr:i18-expand-13}
How does singular-value decay justify low-rank compression?
\end{exercise}
\begin{hint}
Compare the energy/error in the discarded singular-value tail.
\end{hint}
\begin{solution}
If the tail singular values are small, truncated SVD stores only dominant modes while guaranteeing small optimal approximation error.
\end{solution}

\begin{exercise}[Ill-conditioning]\label{exr:i18-expand-14}
What does a very small nonzero \(\sigma_{\min}\) mean for solving \(Ax=b\)?
\end{exercise}
\begin{hint}
Relate \(\|A^{-1}\|_2\) to the smallest singular value.
\end{hint}
\begin{solution}
\(\|A^{-1}\|_2=1/\sigma_{\min}\) is large, so perturbations in data can be strongly amplified.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Eckart--Young error]\label{exr:i18-expand-15}
A matrix has singular values \(9,4,2,1\). State the best rank-two errors in spectral and Frobenius norms.
\end{exercise}
\begin{hint}
Discard the singular values after the second.
\end{hint}
\begin{solution}
The spectral error is \(2\); the Frobenius error is \(\sqrt{2^2+1^2}=\sqrt5\).
\end{solution}

\begin{exercise}[SVD and polar decomposition]\label{exr:i18-expand-16}
For invertible \(A=U\Sigma V^*\), derive a polar decomposition \(A=QH\) with \(Q\) unitary and \(H\) positive.
\end{exercise}
\begin{hint}
Group \(UV^*\) and \(V\Sigma V^*\).
\end{hint}
\begin{solution}
Take \(Q=UV^*\) and \(H=V\Sigma V^*=(A^*A)^{1/2}\); then \(QH=A\).
\end{solution}

\section*{Chapter summary}
The singular-value decomposition is the universal orthogonal factorization of a linear map between finite-dimensional inner-product spaces. Unlike eigenvalue decomposition, it applies to every rectangular matrix. Singular values quantify stretching, rank, conditioning and optimal low-rank approximation.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
