\chapter{Kernels, Images and Isomorphisms}
\label{ch:i06-kernels-images-and-isomorphisms}

Kernel and image measure exactly how a linear transformation fails to be injective or surjective. The rank–nullity theorem balances those two defects and turns structural questions into dimension counts. Quotient spaces then package the first isomorphism theorem in its natural form.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute bases for the kernel and image of a linear transformation;
\item use rank-nullity to determine missing dimensions and to test consistency of computed bases;
\item decide injectivity from the kernel and surjectivity from the image;
\item prove that a finite-dimensional map is an isomorphism from equivalent injective, surjective, or rank conditions;
\item construct an explicit isomorphism between two vector spaces of the same finite dimension;
\item relate solutions of \(Tx=b\) to one particular solution plus the kernel of \(T\);
\end{itemize}
\section*{Conceptual roadmap}
The chapter is organized around a repeated pattern:
\[
\boxed{\text{definition}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{coordinate calculation}\;\longrightarrow\;\text{application}.}
\]
Whenever possible, we separate the invariant mathematical object from a chosen coordinate representation.

\section{Kernel}
$\ker T=\{v:T(v)=0\}$ is a subspace of the domain. A linear map is injective exactly when its kernel is zero.

\section{Image and rank}
$\operatorname{im}T=\{T(v):v\in V\}$ is a subspace of the codomain. Its dimension is the rank.

\section{Rank–nullity}
For finite-dimensional $V$, $\dim V=\dim\ker T+\dim\operatorname{im}T$.

\section{Injective versus surjective}
For a linear endomorphism of a finite-dimensional space, injective, surjective and bijective are equivalent.

\section{Quotients}
For a subspace $U\subseteq V$, the quotient $V/U$ consists of cosets $v+U$. It has dimension $\dim V-\dim U$ in finite dimensions.

\section{First isomorphism theorem}
The map $V/\ker T\to\operatorname{im}T$, $v+\ker T\mapsto T(v)$, is a well-defined linear isomorphism.

\section{Core structural results}

\begin{theorem}[Rank–nullity theorem]\label{thm:i06-01}
If $T:V\to W$ and $V$ is finite-dimensional, then $\dim V=\dim\ker T+\dim\operatorname{im}T$.
\begin{proof}
Choose a basis of the kernel and extend it to a basis of $V$. The images of the added basis vectors form a basis of the image.
\end{proof}
\end{theorem}

\begin{theorem}[First isomorphism theorem]\label{thm:i06-02}
Every linear map induces an isomorphism $V/\ker T\cong\operatorname{im}T$.
\begin{proof}
The induced map is well-defined because changing a representative by a kernel vector does not change its image; injectivity and surjectivity are immediate.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Differentiation]\label{ex:i06-worked-01}
On $\mathcal P_n$, differentiation has kernel the constants and image $\mathcal P_{n-1}$, so nullity $1$ plus rank $n$ equals $n+1$.
\end{example}

\begin{example}[Projection]\label{ex:i06-worked-02}
The projection $\mathbb R^3\to\mathbb R^2$, $(x,y,z)\mapsto(x,y)$, has one-dimensional kernel and rank $2$.
\end{example}

\begin{example}[Trace functional]\label{ex:i06-worked-03}
$\operatorname{tr}:M_n\to\mathbb F$ is surjective; its kernel has dimension $n^2-1$.
\end{example}

\section{Diagnostic checklist}
Before moving on, it is useful to distinguish four levels of understanding:
\begin{enumerate}
\item recognize the definitions in unfamiliar examples;
\item prove the core structural statements without coordinates when possible;
\item translate the same statement into a matrix or coordinate computation;
\item know which hypotheses are essential and produce a counterexample when one is removed.
\end{enumerate}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Rank-nullity from equations]\label{prob:i06-dossier-01}
For $T:\mathbb R^4\to\mathbb R^2$, $T(x_1,x_2,x_3,x_4)=(x_1+x_3,x_2+x_4)$, find rank and nullity.
\end{problem}
\begin{solution}
The map is onto because $(a,b)=T(a,b,0,0)$, so rank is $2$. The kernel is given by $x_3=-x_1$, $x_4=-x_2$, hence has dimension $2$. Rank plus nullity is $4$.
\end{solution}

\begin{problem}[Kernel detects injectivity]\label{prob:i06-dossier-02}
Prove that a linear map is injective iff its kernel is $\{0\}$.
\end{problem}
\begin{solution}
If $T(v)=T(w)$, then $T(v-w)=0$; trivial kernel forces $v=w$. Conversely, if $T$ is injective and $T(v)=0=T(0)$, then $v=0$.
\end{solution}

\begin{problem}[First isomorphism theorem]\label{prob:i06-dossier-03}
For $T(x,y,z)=(x,y)$, identify $\mathbb R^3/\ker T$ with $\mathbb R^2$.
\end{problem}
\begin{solution}
The kernel is the $z$-axis. Map the coset $(x,y,z)+\ker T$ to $(x,y)$. This is well-defined, linear, injective, and onto; vertical displacement is exactly what the quotient forgets.
\end{solution}

\begin{problem}[Trace kernel]\label{prob:i06-dossier-04}
Compute the dimension of $\{A\in M_3(\mathbb R):\operatorname{tr}A=0\}$.
\end{problem}
\begin{solution}
Trace is a nonzero linear functional from a 9-dimensional space to $\mathbb R$, hence has rank $1$. Rank-nullity gives kernel dimension $8$.
\end{solution}

\begin{problem}[Image of differentiation]\label{prob:i06-dossier-05}
For $D:\mathcal P_4\to\mathcal P_3$, determine rank and nullity.
\end{problem}
\begin{solution}
The kernel consists of constants, dimension $1$. Every cubic has an antiderivative of degree at most $4$, so $D$ is onto $\mathcal P_3$, which has dimension $4$. Thus rank $4$, nullity $1$.
\end{solution}

\begin{problem}[Injective endomorphism]\label{prob:i06-dossier-06}
Why is an injective linear endomorphism of a finite-dimensional space automatically surjective?
\end{problem}
\begin{solution}
If $\dim V=n$ and the kernel is zero, rank-nullity gives rank $n$. The image is an $n$-dimensional subspace of the $n$-dimensional codomain $V$, so it equals $V$.
\end{solution}

\begin{problem}[A quotient dimension]\label{prob:i06-dossier-07}
If $\dim V=10$ and $\dim U=4$, compute $\dim(V/U)$.
\end{problem}
\begin{solution}
Choose a basis of $U$ and extend it to a basis of $V$. The six cosets of the added basis vectors form a basis of $V/U$, so the quotient dimension is $10-4=6$.
\end{solution}

\begin{problem}[Rank under composition]\label{prob:i06-dossier-08}
Show $\operatorname{rank}(ST)\le\min(\operatorname{rank}S,\operatorname{rank}T)$.
\end{problem}
\begin{solution}
The image of $ST$ is $S(\operatorname{im}T)$, so its dimension cannot exceed $\dim\operatorname{im}T$. It is also a subspace of $\operatorname{im}S$, so it cannot exceed $\operatorname{rank}S$.
\end{solution}

\begin{problem}[Same kernel, different maps]\label{prob:i06-dossier-09}
Give two nonzero linear maps $\mathbb R^2\to\mathbb R$ with the same kernel.
\end{problem}
\begin{solution}
Take $T(x,y)=x$ and $S(x,y)=2x$. Both have kernel the $y$-axis, but they are different maps. The kernel alone does not determine a linear map.
\end{solution}

\begin{problem}[Surjective map splitting]\label{prob:i06-dossier-10}
Let $T:V\to W$ be surjective with finite dimensions. Show there is a subspace $U\subset V$ such that $T|_U:U\to W$ is an isomorphism.
\end{problem}
\begin{solution}
Choose a basis of $\ker T$ and extend it to a basis of $V$. Let $U$ be the span of the added vectors. Rank-nullity shows their number is $\dim W$, and their images form a basis of $W$, so the restriction is bijective.
\end{solution}

\begin{problem}[Kernel of a matrix operator]\label{prob:i06-dossier-11}
Let $L_A:M_2(\mathbb R)\to M_2(\mathbb R)$ be $L_A(X)=AX$ with invertible $A$. Find its kernel and rank.
\end{problem}
\begin{solution}
If $AX=0$, multiply by $A^{-1}$ to get $X=0$, so the kernel is trivial. The domain has dimension $4$, hence rank is $4$; in fact $L_A$ is an isomorphism.
\end{solution}

\begin{problem}[Structural synthesis]\label{prob:i06-dossier-12}
Explain why rank-nullity is a quantitative form of the first isomorphism theorem.
\end{problem}
\begin{solution}
The first isomorphism theorem says $V/\ker T\cong\operatorname{im}T$. Taking finite dimensions gives $\dim V-\dim\ker T=\dim\operatorname{im}T$, which rearranges exactly to rank-nullity.
\end{solution}
