\chapter{Orthogonal and Unitary Operators}
\label{ch:i15-orthogonal-and-unitary-operators}

Orthogonal and unitary operators are the linear symmetries of inner-product geometry. They preserve lengths, angles and orthogonality, and their matrices satisfy $Q^*Q=I$. Reflections, rotations and Householder transformations show how geometric symmetries become structured matrices.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item test whether a real or complex matrix is orthogonal or unitary using \(A^*A=I\);
\item prove that orthogonal and unitary operators preserve inner products, norms, and angles;
\item compute the inverse of an orthogonal or unitary matrix from its adjoint;
\item classify simple planar orthogonal maps as rotations or reflections;
\item show that every eigenvalue of a unitary operator has modulus one;
\item construct examples distinguishing orthogonal/unitary operators from merely diagonalizable ones;
\end{itemize}
\section*{Conceptual roadmap}
The chapter is organized around a repeated pattern:
\[
\boxed{\text{definition}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{coordinate calculation}\;\longrightarrow\;\text{application}.}
\]
Whenever possible, we separate the invariant mathematical object from a chosen coordinate representation.

\section{Isometries}
A linear map is an isometry when it preserves norms, equivalently inner products.

\section{Orthogonal and unitary matrices}
A real matrix is orthogonal when $Q^TQ=I$; a complex matrix is unitary when $Q^*Q=I$.


% BEGIN VOL01-EXPANSION I15-example-01
\begin{example}[Testing a matrix by orthonormal columns]\label{ex:i15-expand-01}
Consider
\[
Q=\frac15\begin{pmatrix}3&-4\\4&3\end{pmatrix}.
\]
Its columns have norm \(1\) and inner product \(0\), hence
\(Q^TQ=I\). Therefore \(Q\) is orthogonal and
\[
Q^{-1}=Q^T.
\]
Its determinant is \(1\), so it is orientation preserving. The entries match
\(\cos\theta=3/5\), \(\sin\theta=4/5\), identifying the map as a rotation.
The norm and determinant checks agree with the geometric interpretation.
\end{example}
% END VOL01-EXPANSION I15-example-01
\section{Equivalent characterizations}
For a square matrix, orthonormal columns, orthonormal rows and inverse equal to adjoint are equivalent.

\section{Determinant and orientation}
A real orthogonal matrix has determinant $\pm1$. Determinant $+1$ gives orientation-preserving transformations.

\section{Reflections and Householder maps}
For unit $u$, $H=I-2uu^*$ reflects across the hyperplane $u^\perp$.


% BEGIN VOL01-EXPANSION I15-example-02
\begin{example}[A Householder reflection constructed from a normal]\label{ex:i15-expand-02}
Let \(u=(1,1)/\sqrt2\). Then
\[
H=I-2uu^T
=\begin{pmatrix}0&-1\\-1&0\end{pmatrix}.
\]
We have
\[
Hu=-u,
\]
while every vector orthogonal to \(u\), such as \((1,-1)\), is fixed.
Also \(H^T H=H^2=I\). Thus the algebraic formula and the geometry agree:
\(H\) reflects across the line \(u^\perp=\operatorname{span}(1,-1)\).
\end{example}
% END VOL01-EXPANSION I15-example-02
\section{Rotations and planar blocks}
Real orthogonal operators decompose into invariant one- and two-dimensional pieces, with planar rotation blocks on the latter.


% BEGIN VOL01-EXPANSION I15-example-03
\begin{example}[Classifying a planar orthogonal map]\label{ex:i15-expand-03}
Let
\[
Q=\begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
Its columns are orthonormal, so \(Q\in O(2)\), but
\[
\det Q=-1.
\]
Hence it cannot be a rotation. The vector \((1,1)\) is fixed and
\((1,-1)\) changes sign, so \(Q\) is reflection across the line \(y=x\).

This illustrates a reliable planar classification: first verify orthogonality,
then use determinant and fixed directions to distinguish rotations from
reflections.
\end{example}
% END VOL01-EXPANSION I15-example-03
\section{Core structural results}

\begin{theorem}[Inner-product preservation criterion]\label{thm:i15-01}
A linear map preserves inner products if and only if it preserves norms.
\begin{proof}
One direction is immediate. For the converse, recover the inner product from the norm using the polarization identity.
\end{proof}
\end{theorem}

\begin{theorem}[Adjoint inverse criterion]\label{thm:i15-02}
A matrix $Q$ is unitary exactly when $Q^{-1}=Q^*$.
\begin{proof}
This is simply a rearrangement of $Q^*Q=I$, with invertibility automatic.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Planar rotation]\label{ex:i15-worked-01}
The standard rotation matrix is orthogonal with determinant $1$.
\end{example}

\begin{example}[Reflection]\label{ex:i15-worked-02}
$\operatorname{diag}(1,-1)$ reflects across the $x$-axis and has determinant $-1$.
\end{example}

\begin{example}[Permutation matrix]\label{ex:i15-worked-03}
Every permutation matrix is orthogonal because its columns are a rearrangement of the standard orthonormal basis.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Orthogonal inverse]\label{prob:i15-dossier-01}
If $Q^TQ=I$, show $Q^{-1}=Q^T$.
\end{problem}
\begin{solution}
The equation already says $Q^T$ is a left inverse. For square matrices a left inverse is also a right inverse, so $Q$ is invertible and $Q^{-1}=Q^T$.
\end{solution}

\begin{problem}[Norm preservation]\label{prob:i15-dossier-02}
Show an orthogonal matrix preserves Euclidean norm.
\end{problem}
\begin{solution}
$\|Qx\|^2=(Qx)^T(Qx)=x^TQ^TQx=x^Tx=\|x\|^2$.
\end{solution}

\begin{problem}[Determinant sign]\label{prob:i15-dossier-03}
Why does a real orthogonal matrix have determinant $\pm1$?
\end{problem}
\begin{solution}
Take determinants of $Q^TQ=I$: $(\det Q)^2=1$. Hence $\det Q=\pm1$.
\end{solution}

\begin{problem}[Householder reflection]\label{prob:i15-dossier-04}
For a unit vector $u$, show $H=I-2uu^T$ is orthogonal.
\end{problem}
\begin{solution}
Let $P=uu^T$, so $P^2=P$ and $P^T=P$. Then $H^TH=H^2=(I-2P)^2=I-4P+4P^2=I$.
\end{solution}

\begin{problem}[Reflection geometry]\label{prob:i15-dossier-05}
What does $H=I-2uu^T$ do to $u$ and to vectors orthogonal to $u$?
\end{problem}
\begin{solution}
$Hu=u-2u=-u$. If $v\perp u$, then $u^Tv=0$ and $Hv=v$. Thus $H$ reflects across the hyperplane $u^\perp$.
\end{solution}

\begin{problem}[Unitary eigenvalues]\label{prob:i15-dossier-06}
Why must every eigenvalue of a unitary matrix have modulus $1$?
\end{problem}
\begin{solution}
If $Uv=\lambda v$ and $v\ne0$, norm preservation gives $\|v\|=\|Uv\|=|\lambda|\|v\|$, so $|\lambda|=1$.
\end{solution}

\begin{problem}[Permutation matrices]\label{prob:i15-dossier-07}
Why is every permutation matrix orthogonal?
\end{problem}
\begin{solution}
Its columns are a reordering of the standard orthonormal basis, so they remain orthonormal. Hence $P^TP=I$.
\end{solution}

\begin{problem}[Orientation]\label{prob:i15-dossier-08}
What distinguishes rotations from reflections inside $O(n)$?
\end{problem}
\begin{solution}
Both preserve inner products, but orientation-preserving orthogonal maps have determinant $+1$ and form $SO(n)$. Reflections have determinant $-1$.
\end{solution}

\begin{problem}[Product of orthogonal matrices]\label{prob:i15-dossier-09}
Show the product of two orthogonal matrices is orthogonal.
\end{problem}
\begin{solution}
If $Q^TQ=I$ and $R^TR=I$, then $(QR)^T(QR)=R^TQ^TQR=R^TR=I$.
\end{solution}

\begin{problem}[Columns and rows]\label{prob:i15-dossier-10}
Why do orthonormal columns of a square matrix imply orthonormal rows?
\end{problem}
\begin{solution}
Orthonormal columns give $Q^*Q=I$, so $Q$ is invertible with inverse $Q^*$. Hence also $QQ^*=I$, which says the rows are orthonormal.
\end{solution}

\begin{problem}[Condition number]\label{prob:i15-dossier-11}
What is the Euclidean condition number of an orthogonal matrix?
\end{problem}
\begin{solution}
All singular values equal $1$, so $\kappa_2(Q)=1$. Orthogonal maps neither stretch nor compress, making them optimally conditioned.
\end{solution}

\begin{problem}[Symmetry synthesis]\label{prob:i15-dossier-12}
Why are orthogonal/unitary transformations the natural coordinate changes for inner-product problems?
\end{problem}
\begin{solution}
They preserve the inner product, so lengths, angles, orthogonality, numerical norms, and spectral geometry are unchanged. This lets one simplify coordinates without distorting the geometric structure under study.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:i15-01}
Show an orthogonal matrix preserves Euclidean norm.
\end{exercise}
\begin{hint}
Compute \(A^*A\) and compare it with \(I\); this single identity packages preservation of the inner product. Compute \(A^*A\); orthogonal or unitary means this product is exactly the identity.
\end{hint}
\begin{solution}
$\|Qx\|^2=x^TQ^TQx=x^Tx$.
\end{solution}

\begin{exercise}\label{exr:i15-02}
Why do orthogonal matrices have determinant $\pm1$?
\end{exercise}
\begin{hint}
Reduce to triangular form while recording how each row operation changes the determinant. Use \(\langle Ux,Uy\rangle=\langle x,y\rangle\) to deduce norm and angle preservation.
\end{hint}
\begin{solution}
Taking determinants in $Q^TQ=I$ gives $(\det Q)^2=1$.
\end{solution}

\begin{exercise}\label{exr:i15-03}
Verify a Householder matrix is orthogonal.
\end{exercise}
\begin{hint}
After the first projection, the vector already lies in the target subspace; project that vector one more time. For a unitary matrix, the inverse is \(A^*\); verify both products equal \(I\).
\end{hint}
\begin{solution}
With $P=uu^T$ and $P^2=P$, $(I-2P)^2=I$. Since it is symmetric, its transpose is its inverse.
\end{solution}

\begin{exercise}\label{exr:i15-04}
What does a unitary operator do to an orthonormal basis?
\end{exercise}
\begin{hint}
Compute \(A^*A\) and compare it with \(I\); this single identity packages preservation of the inner product. In the plane, use determinant \(\pm1\) together with the trace or fixed directions to distinguish rotations from reflections.
\end{hint}
\begin{solution}
It sends it to another orthonormal basis because all pairwise inner products are preserved.
\end{solution}

\begin{exercise}\label{exr:i15-05}
Show the inverse of a unitary matrix is unitary.
\end{exercise}
\begin{hint}
Compute \(A^*A\) and compare it with \(I\); this single identity packages preservation of the inner product. If \(Uv=\lambda v\), compare \(\|Uv\|\) with \(\|v\|\) to constrain \(|\lambda|\).
\end{hint}
\begin{solution}
If $Q^{-1}=Q^*$, then $(Q^{-1})^*=Q$ and $(Q^{-1})^*Q^{-1}=QQ^*=I$.
\end{solution}

\begin{exercise}\label{exr:i15-06}
Give an orthogonal matrix with determinant $-1$ in $\mathbb R^3$.
\end{exercise}
\begin{hint}
Compute \(A^*A\) and compare it with \(I\); this single identity packages preservation of the inner product. Columns of a unitary matrix form an orthonormal basis; test them with inner products.
\end{hint}
\begin{solution}
For example $\operatorname{diag}(1,1,-1)$.
\end{solution}

\begin{exercise}\label{exr:i15-07}
Why is every eigenvalue of a complex unitary matrix of modulus one?
\end{exercise}
\begin{hint}
First determine \(\lambda\) from \(\det(A-\lambda I)=0\), then solve \((A-\lambda I)v=0\) for the eigenspace. Products of unitary operators remain unitary because the adjoint reverses multiplication order.
\end{hint}
\begin{solution}
If $Qv=\lambda v$, norm preservation gives $\|v\|=\|Qv\|=|\lambda|\|v\|$, so $|\lambda|=1$.
\end{solution}

\begin{exercise}\label{exr:i15-08}
Explain why orthogonal change of coordinates is numerically attractive.
\end{exercise}
\begin{hint}
Write the relevant inner products explicitly and use bilinearity or sesquilinearity before simplifying norms or angles. To disprove unitarity, it is enough to find one vector whose norm is not preserved.
\end{hint}
\begin{solution}
Its inverse is the transpose and its condition number in the Euclidean norm is $1$, so it does not amplify relative errors.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Orthogonality test]\label{exr:i15-expand-01}
Determine whether \(Q=\frac1{\sqrt2}\begin{pmatrix}1&-1\\1&1\end{pmatrix}\) is orthogonal.
\end{exercise}
\begin{hint}
Compute \(Q^TQ\).
\end{hint}
\begin{solution}
\(Q^TQ=I\), so \(Q\) is orthogonal.
\end{solution}

\begin{exercise}[Inverse]\label{exr:i15-expand-02}
Find the inverse of an orthogonal matrix \(Q\).
\end{exercise}
\begin{hint}
Use the defining equation \(Q^TQ=I\).
\end{hint}
\begin{solution}
\(Q^{-1}=Q^T\).
\end{solution}

\begin{exercise}[Householder matrix]\label{exr:i15-expand-03}
Construct the Householder reflection for \(u=e_1\in\mathbb R^3\).
\end{exercise}
\begin{hint}
Use \(H=I-2uu^T\).
\end{hint}
\begin{solution}
\(H=\operatorname{diag}(-1,1,1)\).
\end{solution}

\begin{exercise}[Rotation angle]\label{exr:i15-expand-04}
For \(Q=\begin{pmatrix}0&-1\\1&0\end{pmatrix}\), identify the rotation angle.
\end{exercise}
\begin{hint}
Compare with \(\begin{pmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{pmatrix}\).
\end{hint}
\begin{solution}
\(\theta=\pi/2\) modulo \(2\pi\).
\end{solution}

\begin{exercise}[Complex unitary]\label{exr:i15-expand-05}
Check that \(U=\operatorname{diag}(i,-i)\) is unitary.
\end{exercise}
\begin{hint}
Compute \(U^*U\).
\end{hint}
\begin{solution}
\(U^*=\operatorname{diag}(-i,i)\), and \(U^*U=I\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Norm preservation]\label{exr:i15-expand-06}
Prove \(Q^*Q=I\) implies \(\|Qx\|=\|x\|\).
\end{exercise}
\begin{hint}
Compute the squared norm using the inner product.
\end{hint}
\begin{solution}
\(\|Qx\|^2=x^*Q^*Qx=x^*x\).
\end{solution}

\begin{exercise}[Determinant modulus]\label{exr:i15-expand-07}
Prove a real orthogonal matrix has determinant \(\pm1\).
\end{exercise}
\begin{hint}
Take determinants of \(Q^TQ=I\).
\end{hint}
\begin{solution}
\((\det Q)^2=1\), hence \(\det Q=\pm1\).
\end{solution}

\begin{exercise}[Unitary eigenvalue modulus]\label{exr:i15-expand-08}
Prove every eigenvalue of a unitary operator has modulus one.
\end{exercise}
\begin{hint}
Apply norm preservation to an eigenvector.
\end{hint}
\begin{solution}
If \(Uv=\lambda v\), then \(\|v\|=\|Uv\|=|\lambda|\|v\|\); since \(v\ne0\), \(|\lambda|=1\).
\end{solution}

\begin{exercise}[Product closure]\label{exr:i15-expand-09}
Prove the product of unitary matrices is unitary.
\end{exercise}
\begin{hint}
Remember \((UV)^*=V^*U^*\).
\end{hint}
\begin{solution}
\((UV)^*(UV)=V^*U^*UV=V^*V=I\).
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Diagonalizable not orthogonal]\label{exr:i15-expand-10}
Give a diagonalizable real matrix that is not orthogonal.
\end{exercise}
\begin{hint}
Use a non-unit scaling diagonal matrix.
\end{hint}
\begin{solution}
\(\operatorname{diag}(2,1)\) is diagonal but does not preserve norm.
\end{solution}

\begin{exercise}[Determinant one is insufficient]\label{exr:i15-expand-11}
Give a real matrix with determinant \(1\) that is not orthogonal.
\end{exercise}
\begin{hint}
Use a shear.
\end{hint}
\begin{solution}
\(\begin{pmatrix}1&1\\0&1\end{pmatrix}\) has determinant \(1\) but its columns are not orthonormal.
\end{solution}

\begin{exercise}[Norm of one vector]\label{exr:i15-expand-12}
Explain why preserving the norm of one nonzero vector does not prove a map is orthogonal.
\end{exercise}
\begin{hint}
Construct a map fixing one direction but stretching another.
\end{hint}
\begin{solution}
\(\operatorname{diag}(1,2)\) preserves \(\|e_1\|\) but doubles \(e_2\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Rigid coordinate change]\label{exr:i15-expand-13}
Why are orthogonal changes of coordinates natural in Euclidean geometry?
\end{exercise}
\begin{hint}
List the geometric quantities preserved by the inner product.
\end{hint}
\begin{solution}
They preserve lengths, distances, angles and orthogonality, so coordinate simplification does not distort geometry.
\end{solution}

\begin{exercise}[Householder QR]\label{exr:i15-expand-14}
Why are Householder reflections effective for QR factorization?
\end{exercise}
\begin{hint}
Use their ability to map a vector to a coordinate axis while preserving norm.
\end{hint}
\begin{solution}
A Householder reflector can zero all entries below a chosen pivot in one stable orthogonal step.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Planar classification]\label{exr:i15-expand-15}
Prove every real \(2\times2\) orthogonal matrix with determinant \(1\) has rotation form.
\end{exercise}
\begin{hint}
Let the first unit column be \((c,s)\) and determine the second using orientation.
\end{hint}
\begin{solution}
Orthogonality forces the second column to be \((-s,c)\), with \(c^2+s^2=1\), so \(c=\cos\theta,s=\sin\theta\).
\end{solution}

\begin{exercise}[Reflection eigenstructure]\label{exr:i15-expand-16}
For a Householder reflection \(H=I-2uu^T\), determine all eigenvalues and eigenspaces.
\end{exercise}
\begin{hint}
Split space into \(\operatorname{span}(u)\oplus u^\perp\).
\end{hint}
\begin{solution}
\(-1\) occurs on \(\operatorname{span}(u)\); \(+1\) occurs on \(u^\perp\).
\end{solution}

\section*{Chapter summary}
Orthogonal and unitary operators are the linear symmetries of inner-product geometry. They preserve lengths, angles and orthogonality, and their matrices satisfy $Q^*Q=I$. Reflections, rotations and Householder transformations show how geometric symmetries become structured matrices.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
