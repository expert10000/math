\chapter{Quadratic Forms}
\label{ch:i17-quadratic-forms}

Quadratic forms encode second-order geometry. Symmetric bilinear forms and symmetric matrices represent the same finite-dimensional data once a basis is chosen. Congruence, not similarity, is the natural basis-change relation, and Sylvester's law of inertia identifies the basis-independent signature.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item associate a symmetric or Hermitian matrix with a quadratic form and recover the form from the matrix;
\item diagonalize a quadratic form by an orthogonal or unitary change of coordinates when appropriate;
\item classify a real quadratic form as positive definite, negative definite, semidefinite, or indefinite;
\item apply eigenvalues or Sylvester-type criteria to decide definiteness;
\item compute the signature and identify its invariance under congruence;
\item transform a quadratic expression to principal-axis form and interpret the resulting geometry;
\end{itemize}
\section*{Conceptual roadmap}
The chapter is organized around a repeated pattern:
\[
\boxed{\text{definition}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{coordinate calculation}\;\longrightarrow\;\text{application}.}
\]
Whenever possible, we separate the invariant mathematical object from a chosen coordinate representation.

\section{Quadratic and bilinear forms}
A quadratic form on a real vector space has the form $q(v)=B(v,v)$ for a symmetric bilinear form $B$.

\section{Matrix representation}
In coordinates, $q(x)=x^TAx$ with $A$ symmetric. Cross terms appear twice the corresponding off-diagonal entry.


% BEGIN VOL01-EXPANSION I17-example-01
\begin{example}[Extracting the symmetric matrix correctly]\label{ex:i17-expand-01}
For
\[
q(x,y)=3x^2+4xy+2y^2,
\]
the symmetric matrix is
\[
A=\begin{pmatrix}3&2\\2&2\end{pmatrix},
\]
because
\[
[x\ y]A\binom{x}{y}=3x^2+(2+2)xy+2y^2.
\]
The factor of two in the cross term is the main bookkeeping point. Multiplying
out \(x^TAx\) is the quickest check that the matrix has been extracted
correctly.
\end{example}
% END VOL01-EXPANSION I17-example-01
\section{Polarization}
Over characteristic not $2$, the symmetric bilinear form is recovered from $q$ by $B(u,v)=\tfrac12(q(u+v)-q(u)-q(v))$.

\section{Congruence}
Under coordinate change $x=Py$, the matrix transforms as $A\mapsto P^TAP$.


% BEGIN VOL01-EXPANSION I17-example-02
\begin{example}[Congruence and completing the square]\label{ex:i17-expand-02}
Consider
\[
q(x,y)=x^2+4xy+y^2.
\]
Set \(u=x+2y\), so \(x=u-2y\). Then
\[
q=(u-2y)^2+4(u-2y)y+y^2=u^2-3y^2.
\]
Thus an invertible change of variables converts the form to one positive and
one negative square. The inertia is therefore \((1,1,0)\).

This is congruence geometry, not similarity: both copies of the coordinate
vector change in \(x^TAx\).
\end{example}
% END VOL01-EXPANSION I17-example-02
\section{Sylvester law of inertia}
Every real symmetric quadratic form is congruent to a diagonal form with entries $1,-1,0$, and the counts of positive, negative and zero entries are invariant.

\section{Definiteness}
A symmetric form is positive definite exactly when all eigenvalues are positive; equivalent criteria include positivity of leading principal minors.


% BEGIN VOL01-EXPANSION I17-example-03
\begin{example}[Definiteness by two independent tests]\label{ex:i17-expand-03}
For
\[
A=\begin{pmatrix}2&1\\1&2\end{pmatrix},
\]
the leading principal minors are \(2>0\) and \(3>0\), so Sylvester's criterion
says the form is positive definite. Independently, the eigenvalues are \(3\)
and \(1\), both positive.

The two tests agree. The eigenvalue method explains the geometry after
orthogonal diagonalization; the principal-minor method can be faster when the
matrix is already explicit.
\end{example}
% END VOL01-EXPANSION I17-example-03
\section{Core structural results}

\begin{theorem}[Polarization identity]\label{thm:i17-01}
For a real quadratic form $q(v)=B(v,v)$ with symmetric bilinear $B$, $B(u,v)=\frac12(q(u+v)-q(u)-q(v))$.
\begin{proof}
Expand $q(u+v)=B(u,u)+2B(u,v)+B(v,v)$ and rearrange.
\end{proof}
\end{theorem}

\begin{theorem}[Sylvester law of inertia]\label{thm:i17-02}
The numbers of positive, negative and zero squares in a real diagonalization by congruence are basis-independent.
\begin{proof}
A congruence change is an invertible change of variables; maximal dimensions of subspaces on which the form is positive or negative definite are intrinsic and determine the counts.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Conic form]\label{ex:i17-worked-01}
$x^2+2xy+y^2=(x+y)^2$ has rank one and is positive semidefinite.
\end{example}

\begin{example}[Indefinite form]\label{ex:i17-worked-02}
$x^2-y^2$ has inertia $(1,1,0)$ and vanishes on the two lines $x=\pm y$.
\end{example}

\begin{example}[Hessian viewpoint]\label{ex:i17-worked-03}
At a critical point of a smooth function, the Hessian defines a quadratic form whose signature describes local second-order behavior.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Matrix of a quadratic form]\label{prob:i17-dossier-01}
Find the symmetric matrix for $q(x,y)=3x^2+4xy+2y^2$.
\end{problem}
\begin{solution}
$q=[x\ y]\begin{pmatrix}3&2\\2&2\end{pmatrix}\binom{x}{y}$ because the cross term is $2a_{12}xy=4xy$.
\end{solution}

\begin{problem}[Polarization]\label{prob:i17-dossier-02}
Recover the symmetric bilinear form $B$ from a real quadratic form $q$.
\end{problem}
\begin{solution}
$B(u,v)=\frac12(q(u+v)-q(u)-q(v))$. Expanding $q(u+v)=B(u,u)+2B(u,v)+B(v,v)$ proves the formula.
\end{solution}

\begin{problem}[Congruence versus similarity]\label{prob:i17-dossier-03}
Why does a basis change send a quadratic-form matrix to $P^TAP$?
\end{problem}
\begin{solution}
If old coordinates are $x=Py$, then $q=x^TAx=y^TP^TAPy$. Both copies of the coordinate vector change, producing congruence rather than similarity.
\end{solution}

\begin{problem}[Definiteness by eigenvalues]\label{prob:i17-dossier-04}
How can the spectral theorem test positive definiteness of a real symmetric matrix?
\end{problem}
\begin{solution}
Orthogonally diagonalize $A=QDQ^T$. Then $x^TAx=y^TDy=\sum_i\lambda_i y_i^2$. This is positive for all nonzero $x$ exactly when every eigenvalue is positive.
\end{solution}

\begin{problem}[Indefinite form]\label{prob:i17-dossier-05}
Classify $q(x,y)=x^2-y^2$.
\end{problem}
\begin{solution}
It takes positive values, such as at $(1,0)$, and negative values, such as at $(0,1)$, so it is indefinite. Its inertia is $(1,1,0)$.
\end{solution}

\begin{problem}[Rank-one semidefinite form]\label{prob:i17-dossier-06}
Classify $q(x,y)=(x+y)^2$.
\end{problem}
\begin{solution}
The form is nonnegative and vanishes on the nonzero line $x+y=0$, so it is positive semidefinite but not definite. Its matrix has rank $1$.
\end{solution}

\begin{problem}[Diagonalizing a cross term]\label{prob:i17-dossier-07}
Diagonalize $2xy$ by an orthogonal change of variables.
\end{problem}
\begin{solution}
Set $u=(x+y)/\sqrt2$, $v=(x-y)/\sqrt2$. Then $x=(u+v)/\sqrt2$, $y=(u-v)/\sqrt2$, so $2xy=u^2-v^2$.
\end{solution}

\begin{problem}[Inertia invariance]\label{prob:i17-dossier-08}
What intrinsic meaning do the positive and negative inertia indices have?
\end{problem}
\begin{solution}
They are the maximal dimensions of subspaces on which the quadratic form is positive definite or negative definite. Because those maximal dimensions are basis-independent, the inertia is invariant under congruence.
\end{solution}

\begin{problem}[Hessian interpretation]\label{prob:i17-dossier-09}
At a critical point, how does the Hessian quadratic form classify a nondegenerate point?
\end{problem}
\begin{solution}
Positive definite Hessian gives a strict local minimum, negative definite gives a strict local maximum, and an indefinite Hessian gives a saddle point. The signature captures the second-order geometry.
\end{solution}

\begin{problem}[Principal-minor criterion]\label{prob:i17-dossier-10}
For $A=\begin{pmatrix}a&b\\b&c\end{pmatrix}$, state the positive-definite criterion using leading principal minors.
\end{problem}
\begin{solution}
Sylvester's criterion gives $a>0$ and $ac-b^2>0$. These conditions are equivalent to both eigenvalues being positive.
\end{solution}

\begin{problem}[Null directions]\label{prob:i17-dossier-11}
What does a zero eigenvalue of a symmetric quadratic-form matrix mean geometrically?
\end{problem}
\begin{solution}
It gives a nonzero direction $v$ with $Av=0$, hence $q(v)=0$ and no quadratic curvature in that direction. The form is degenerate.
\end{solution}

\begin{problem}[Quadratic-form synthesis]\label{prob:i17-dossier-12}
Why is signature more appropriate than individual matrix entries as geometric data?
\end{problem}
\begin{solution}
Entries depend strongly on coordinates, while signature is invariant under invertible changes of variables. It records the intrinsic counts of positive, negative, and null directions.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:i17-01}
Find the symmetric matrix for $q(x,y)=2x^2+6xy+5y^2$.
\end{exercise}
\begin{hint}
Write the quadratic form as \(x^TAx\) with \(A\) symmetric by averaging the mixed-term coefficients.
\end{hint}
\begin{solution}
$A=\begin{pmatrix}2&3\\3&5\end{pmatrix}$ because the cross term is $2a_{12}xy$.
\end{solution}

\begin{exercise}\label{exr:i17-02}
Recover the bilinear form from a quadratic form by polarization.
\end{exercise}
\begin{hint}
Pass to the symmetric or Hermitian matrix of the form and inspect its eigenvalues or a valid definiteness criterion. Diagonalize the symmetric matrix orthogonally; the diagonal entries are the principal quadratic coefficients.
\end{hint}
\begin{solution}
Use $B(u,v)=\tfrac12(q(u+v)-q(u)-q(v))$.
\end{solution}

\begin{exercise}\label{exr:i17-03}
Why does a basis change act by congruence rather than similarity?
\end{exercise}
\begin{hint}
Write each new basis vector in the old coordinates and use those coordinate columns to build the conversion matrix. For definiteness, inspect the signs of all eigenvalues rather than individual matrix entries.
\end{hint}
\begin{solution}
Both copies of the coordinate vector change in $x^TAx$, producing $y^TP^TAPy$.
\end{solution}

\begin{exercise}\label{exr:i17-04}
Determine the definiteness of $x^2-y^2$.
\end{exercise}
\begin{hint}
Use leading principal minors only when the hypotheses of Sylvester's criterion apply.
\end{hint}
\begin{solution}
It is indefinite because it takes both positive and negative values.
\end{solution}

\begin{exercise}\label{exr:i17-05}
What is the inertia of $x^2+y^2$?
\end{exercise}
\begin{hint}
Under a congruence transformation \(A\mapsto P^TAP\), count positive and negative squares rather than eigenvalues themselves.
\end{hint}
\begin{solution}
$(2,0,0)$: two positive directions, no negative or null directions.
\end{solution}

\begin{exercise}\label{exr:i17-06}
Show positive definite implies nonsingular.
\end{exercise}
\begin{hint}
Pass to the symmetric or Hermitian matrix of the form and inspect its eigenvalues or a valid definiteness criterion. Complete the square one variable at a time if an orthogonal diagonalization is unnecessary.
\end{hint}
\begin{solution}
A nonzero kernel vector would satisfy $q(v)=v^TAv=0$, contradicting positive definiteness.
\end{solution}

\begin{exercise}\label{exr:i17-07}
Diagonalize $2xy$ by a linear change of variables.
\end{exercise}
\begin{hint}
The signature is the number of positive squares minus the number of negative squares after diagonalization.
\end{hint}
\begin{solution}
Set $u=(x+y)/\sqrt2$, $v=(x-y)/\sqrt2$; then $2xy=u^2-v^2$.
\end{solution}

\begin{exercise}\label{exr:i17-08}
Explain the role of the spectral theorem for symmetric quadratic forms.
\end{exercise}
\begin{hint}
Find the eigenspaces first and use orthogonality between distinct eigenspaces before assembling an orthonormal eigenbasis. Test the form on strategically chosen vectors to refute positive or negative definiteness quickly.
\end{hint}
\begin{solution}
Orthogonal diagonalization of the symmetric matrix expresses the form as a weighted sum of coordinate squares.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Matrix extraction]\label{exr:i17-expand-01}
Find the symmetric matrix of \(q(x,y)=5x^2-6xy+2y^2\).
\end{exercise}
\begin{hint}
Remember the off-diagonal entry contributes twice.
\end{hint}
\begin{solution}
\(A=\begin{pmatrix}5&-3\\-3&2\end{pmatrix}\).
\end{solution}

\begin{exercise}[Evaluate a form]\label{exr:i17-expand-02}
For \(A=\begin{pmatrix}2&1\\1&3\end{pmatrix}\), compute \(q(1,-1)\).
\end{exercise}
\begin{hint}
Evaluate \(x^TAx\).
\end{hint}
\begin{solution}
\(q(1,-1)=3\).
\end{solution}

\begin{exercise}[Complete the square]\label{exr:i17-expand-03}
Diagonalize \(x^2+2xy+2y^2\) by completing the square.
\end{exercise}
\begin{hint}
Group \(x^2+2xy=(x+y)^2-y^2\).
\end{hint}
\begin{solution}
\(q=(x+y)^2+y^2\), so the form is positive definite.
\end{solution}

\begin{exercise}[Sylvester criterion]\label{exr:i17-expand-04}
Classify \(A=\begin{pmatrix}3&1\\1&1\end{pmatrix}\).
\end{exercise}
\begin{hint}
Check leading principal minors.
\end{hint}
\begin{solution}
\(3>0\) and \(\det A=2>0\), so \(A\) is positive definite.
\end{solution}

\begin{exercise}[Inertia]\label{exr:i17-expand-05}
Find the inertia of \(q=x_1^2+x_2^2-4x_3^2\).
\end{exercise}
\begin{hint}
Count positive, negative and zero diagonal coefficients.
\end{hint}
\begin{solution}
The inertia is \((2,1,0)\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Polarization]\label{exr:i17-expand-06}
Prove \(B(u,v)=\frac12(q(u+v)-q(u)-q(v))\) for a real symmetric bilinear form.
\end{exercise}
\begin{hint}
Expand \(q(u+v)=B(u+v,u+v)\).
\end{hint}
\begin{solution}
Symmetry makes the two cross terms equal, giving \(q(u+v)=q(u)+2B(u,v)+q(v)\).
\end{solution}

\begin{exercise}[Congruence law]\label{exr:i17-expand-07}
Derive \(A\mapsto P^TAP\) under \(x=Py\).
\end{exercise}
\begin{hint}
Substitute the coordinate change into \(x^TAx\).
\end{hint}
\begin{solution}
\(q=y^TP^TAPy\).
\end{solution}

\begin{exercise}[Positive definite nonsingular]\label{exr:i17-expand-08}
Prove a positive-definite symmetric matrix is invertible.
\end{exercise}
\begin{hint}
A nonzero kernel vector would make the quadratic form vanish.
\end{hint}
\begin{solution}
If \(Av=0\) with \(v\ne0\), then \(v^TAv=0\), contradicting positive definiteness.
\end{solution}

\begin{exercise}[Eigenvalue criterion]\label{exr:i17-expand-09}
Prove a real symmetric matrix is positive definite iff all eigenvalues are positive.
\end{exercise}
\begin{hint}
Use orthogonal diagonalization.
\end{hint}
\begin{solution}
In eigen-coordinates \(q=\sum\lambda_i y_i^2\), which is positive for all nonzero \(y\) exactly when all \(\lambda_i>0\).
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Positive diagonal entries]\label{exr:i17-expand-10}
Show positive diagonal entries do not guarantee positive definiteness.
\end{exercise}
\begin{hint}
Choose a symmetric matrix with a large off-diagonal term.
\end{hint}
\begin{solution}
\(\begin{pmatrix}1&2\\2&1\end{pmatrix}\) has positive diagonal entries but eigenvalues \(3,-1\), so it is indefinite.
\end{solution}

\begin{exercise}[Similarity is wrong relation]\label{exr:i17-expand-11}
Explain why similarity is not the natural coordinate change for quadratic forms.
\end{exercise}
\begin{hint}
Track both appearances of the coordinate vector in \(x^TAx\).
\end{hint}
\begin{solution}
Substitution \(x=Py\) gives \(P^TAP\), not \(P^{-1}AP\).
\end{solution}

\begin{exercise}[Semidefinite not definite]\label{exr:i17-expand-12}
Give a nonzero positive-semidefinite quadratic form that is not positive definite.
\end{exercise}
\begin{hint}
Use a square of one linear form.
\end{hint}
\begin{solution}
\((x+y)^2\) is nonnegative but vanishes on the nonzero line \(x+y=0\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Hessian classification]\label{exr:i17-expand-13}
At a nondegenerate critical point, what do positive, negative and mixed Hessian eigenvalues indicate?
\end{exercise}
\begin{hint}
Interpret the quadratic Taylor term.
\end{hint}
\begin{solution}
All positive gives a local minimum, all negative a local maximum, and mixed signs a saddle.
\end{solution}

\begin{exercise}[Principal axes]\label{exr:i17-expand-14}
Why does orthogonal diagonalization reveal principal axes of a conic or quadric?
\end{exercise}
\begin{hint}
Eigenvectors remove cross terms without distorting Euclidean angles.
\end{hint}
\begin{solution}
In orthonormal eigen-coordinates the form is a weighted sum of squares, so coordinate axes align with intrinsic principal directions.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Signature invariance]\label{exr:i17-expand-15}
Explain why the numbers of positive and negative squares cannot change under invertible coordinate changes.
\end{exercise}
\begin{hint}
Interpret them as maximal dimensions of positive- and negative-definite subspaces.
\end{hint}
\begin{solution}
Those maximal dimensions are intrinsic properties of the form, so congruence cannot alter them; this is Sylvester's law of inertia.
\end{solution}

\begin{exercise}[Degenerate form]\label{exr:i17-expand-16}
For \(q(x,y,z)=(x+y)^2+(y+z)^2\), find the null space and rank.
\end{exercise}
\begin{hint}
Set both squares to zero.
\end{hint}
\begin{solution}
\(x=-y\) and \(z=-y\), so the null space is \(\operatorname{span}(-1,1,-1)\); in dimension \(3\), rank is \(2\).
\end{solution}

\section*{Chapter summary}
Quadratic forms encode second-order geometry. Symmetric bilinear forms and symmetric matrices represent the same finite-dimensional data once a basis is chosen. Congruence, not similarity, is the natural basis-change relation, and Sylvester's law of inertia identifies the basis-independent signature.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
