\chapter{Quadratic Forms}
\label{ch:i17-quadratic-forms}

Quadratic forms encode second-order geometry. Symmetric bilinear forms and symmetric matrices represent the same finite-dimensional data once a basis is chosen. Congruence, not similarity, is the natural basis-change relation, and Sylvester's law of inertia identifies the basis-independent signature.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item associate a symmetric or Hermitian matrix with a quadratic form and recover the form from the matrix;
\item diagonalize a quadratic form by an orthogonal or unitary change of coordinates when appropriate;
\item classify a real quadratic form as positive definite, negative definite, semidefinite, or indefinite;
\item apply eigenvalues or Sylvester-type criteria to decide definiteness;
\item compute the signature and identify its invariance under congruence;
\item transform a quadratic expression to principal-axis form and interpret the resulting geometry;
\end{itemize}
\section*{Conceptual roadmap}
The chapter is organized around a repeated pattern:
\[
\boxed{\text{definition}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{coordinate calculation}\;\longrightarrow\;\text{application}.}
\]
Whenever possible, we separate the invariant mathematical object from a chosen coordinate representation.

\section{Quadratic and bilinear forms}
A quadratic form on a real vector space has the form $q(v)=B(v,v)$ for a symmetric bilinear form $B$.

\section{Matrix representation}
In coordinates, $q(x)=x^TAx$ with $A$ symmetric. Cross terms appear twice the corresponding off-diagonal entry.

\section{Polarization}
Over characteristic not $2$, the symmetric bilinear form is recovered from $q$ by $B(u,v)=\tfrac12(q(u+v)-q(u)-q(v))$.

\section{Congruence}
Under coordinate change $x=Py$, the matrix transforms as $A\mapsto P^TAP$.

\section{Sylvester law of inertia}
Every real symmetric quadratic form is congruent to a diagonal form with entries $1,-1,0$, and the counts of positive, negative and zero entries are invariant.

\section{Definiteness}
A symmetric form is positive definite exactly when all eigenvalues are positive; equivalent criteria include positivity of leading principal minors.

\section{Core structural results}

\begin{theorem}[Polarization identity]\label{thm:i17-01}
For a real quadratic form $q(v)=B(v,v)$ with symmetric bilinear $B$, $B(u,v)=\frac12(q(u+v)-q(u)-q(v))$.
\begin{proof}
Expand $q(u+v)=B(u,u)+2B(u,v)+B(v,v)$ and rearrange.
\end{proof}
\end{theorem}

\begin{theorem}[Sylvester law of inertia]\label{thm:i17-02}
The numbers of positive, negative and zero squares in a real diagonalization by congruence are basis-independent.
\begin{proof}
A congruence change is an invertible change of variables; maximal dimensions of subspaces on which the form is positive or negative definite are intrinsic and determine the counts.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Conic form]\label{ex:i17-worked-01}
$x^2+2xy+y^2=(x+y)^2$ has rank one and is positive semidefinite.
\end{example}

\begin{example}[Indefinite form]\label{ex:i17-worked-02}
$x^2-y^2$ has inertia $(1,1,0)$ and vanishes on the two lines $x=\pm y$.
\end{example}

\begin{example}[Hessian viewpoint]\label{ex:i17-worked-03}
At a critical point of a smooth function, the Hessian defines a quadratic form whose signature describes local second-order behavior.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Matrix of a quadratic form]\label{prob:i17-dossier-01}
Find the symmetric matrix for $q(x,y)=3x^2+4xy+2y^2$.
\end{problem}
\begin{solution}
$q=[x\ y]\begin{pmatrix}3&2\\2&2\end{pmatrix}\binom{x}{y}$ because the cross term is $2a_{12}xy=4xy$.
\end{solution}

\begin{problem}[Polarization]\label{prob:i17-dossier-02}
Recover the symmetric bilinear form $B$ from a real quadratic form $q$.
\end{problem}
\begin{solution}
$B(u,v)=\frac12(q(u+v)-q(u)-q(v))$. Expanding $q(u+v)=B(u,u)+2B(u,v)+B(v,v)$ proves the formula.
\end{solution}

\begin{problem}[Congruence versus similarity]\label{prob:i17-dossier-03}
Why does a basis change send a quadratic-form matrix to $P^TAP$?
\end{problem}
\begin{solution}
If old coordinates are $x=Py$, then $q=x^TAx=y^TP^TAPy$. Both copies of the coordinate vector change, producing congruence rather than similarity.
\end{solution}

\begin{problem}[Definiteness by eigenvalues]\label{prob:i17-dossier-04}
How can the spectral theorem test positive definiteness of a real symmetric matrix?
\end{problem}
\begin{solution}
Orthogonally diagonalize $A=QDQ^T$. Then $x^TAx=y^TDy=\sum_i\lambda_i y_i^2$. This is positive for all nonzero $x$ exactly when every eigenvalue is positive.
\end{solution}

\begin{problem}[Indefinite form]\label{prob:i17-dossier-05}
Classify $q(x,y)=x^2-y^2$.
\end{problem}
\begin{solution}
It takes positive values, such as at $(1,0)$, and negative values, such as at $(0,1)$, so it is indefinite. Its inertia is $(1,1,0)$.
\end{solution}

\begin{problem}[Rank-one semidefinite form]\label{prob:i17-dossier-06}
Classify $q(x,y)=(x+y)^2$.
\end{problem}
\begin{solution}
The form is nonnegative and vanishes on the nonzero line $x+y=0$, so it is positive semidefinite but not definite. Its matrix has rank $1$.
\end{solution}

\begin{problem}[Diagonalizing a cross term]\label{prob:i17-dossier-07}
Diagonalize $2xy$ by an orthogonal change of variables.
\end{problem}
\begin{solution}
Set $u=(x+y)/\sqrt2$, $v=(x-y)/\sqrt2$. Then $x=(u+v)/\sqrt2$, $y=(u-v)/\sqrt2$, so $2xy=u^2-v^2$.
\end{solution}

\begin{problem}[Inertia invariance]\label{prob:i17-dossier-08}
What intrinsic meaning do the positive and negative inertia indices have?
\end{problem}
\begin{solution}
They are the maximal dimensions of subspaces on which the quadratic form is positive definite or negative definite. Because those maximal dimensions are basis-independent, the inertia is invariant under congruence.
\end{solution}

\begin{problem}[Hessian interpretation]\label{prob:i17-dossier-09}
At a critical point, how does the Hessian quadratic form classify a nondegenerate point?
\end{problem}
\begin{solution}
Positive definite Hessian gives a strict local minimum, negative definite gives a strict local maximum, and an indefinite Hessian gives a saddle point. The signature captures the second-order geometry.
\end{solution}

\begin{problem}[Principal-minor criterion]\label{prob:i17-dossier-10}
For $A=\begin{pmatrix}a&b\\b&c\end{pmatrix}$, state the positive-definite criterion using leading principal minors.
\end{problem}
\begin{solution}
Sylvester's criterion gives $a>0$ and $ac-b^2>0$. These conditions are equivalent to both eigenvalues being positive.
\end{solution}

\begin{problem}[Null directions]\label{prob:i17-dossier-11}
What does a zero eigenvalue of a symmetric quadratic-form matrix mean geometrically?
\end{problem}
\begin{solution}
It gives a nonzero direction $v$ with $Av=0$, hence $q(v)=0$ and no quadratic curvature in that direction. The form is degenerate.
\end{solution}

\begin{problem}[Quadratic-form synthesis]\label{prob:i17-dossier-12}
Why is signature more appropriate than individual matrix entries as geometric data?
\end{problem}
\begin{solution}
Entries depend strongly on coordinates, while signature is invariant under invertible changes of variables. It records the intrinsic counts of positive, negative, and null directions.
\end{solution}

\section*{Chapter summary}
Quadratic forms encode second-order geometry. Symmetric bilinear forms and symmetric matrices represent the same finite-dimensional data once a basis is chosen. Congruence, not similarity, is the natural basis-change relation, and Sylvester's law of inertia identifies the basis-independent signature.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
