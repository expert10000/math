\chapter{Diagonalization and Minimal Polynomials}
\label{ch:i11-diagonalization-and-minimal-polynomials}

Diagonalization asks whether an operator can be decomposed into independent one-dimensional modes. The minimal polynomial gives the sharp algebraic criterion: an operator is diagonalizable exactly when its minimal polynomial splits into distinct linear factors. This viewpoint unifies eigenvalue counting with polynomial identities satisfied by the operator.

\section{Diagonalizable operators}
An operator is diagonalizable when it has a basis of eigenvectors, equivalently when some matrix representation is diagonal.

\section{Direct sums of eigenspaces}
If the characteristic polynomial splits, diagonalizability is equivalent to $V$ being the direct sum of its eigenspaces.

\section{Minimal polynomial}
The minimal polynomial $m_T$ is the monic polynomial of least positive degree with $m_T(T)=0$.

\section{Divisibility property}
Every polynomial $p$ with $p(T)=0$ is divisible by $m_T$. In particular, $m_T$ divides the characteristic polynomial.

\section{Square-free criterion}
If $m_T$ splits, then $T$ is diagonalizable exactly when $m_T$ has no repeated root.

\section{Spectral projectors by interpolation}
For distinct eigenvalues, Lagrange interpolation polynomials in $T$ produce projections onto eigenspaces.

\section{Core structural results}

\begin{theorem}[Dimension criterion]\label{thm:i11-01}
If $\chi_T$ splits, then $T$ is diagonalizable if and only if the sum of eigenspace dimensions equals $\dim V$.
\begin{proof}
Distinct eigenspaces form a direct sum. Their union of bases is an eigenbasis exactly when the direct sum has full dimension.
\end{proof}
\end{theorem}

\begin{theorem}[Minimal-polynomial criterion]\label{thm:i11-02}
An operator is diagonalizable if and only if its minimal polynomial is a product of distinct linear factors.
\begin{proof}
For a diagonal operator, a polynomial annihilates it exactly when it vanishes at every diagonal eigenvalue. Conversely, use Lagrange polynomials to construct complementary eigenspace projectors.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Projection]\label{ex:i11-worked-01}
A projection $P^2=P$ has minimal polynomial dividing $t(t-1)$, which has distinct roots, so every projection is diagonalizable.
\end{example}

\begin{example}[Involution]\label{ex:i11-worked-02}
If $T^2=I$ over characteristic not $2$, then $m_T$ divides $(t-1)(t+1)$, so $T$ is diagonalizable.
\end{example}

\begin{example}[Nontrivial shear]\label{ex:i11-worked-03}
The shear has minimal polynomial $(t-1)^2$, so it is not diagonalizable.
\end{example}

\section{Solved dossiers}
These dossiers are longer worked problems. They complement the shorter exercise--hint--solution triads by combining several ideas in one argument.

\begin{problem}[Projection minimal polynomial]\label{prob:i11-dossier-01}
If $P^2=P$ and $P\ne0,I$, determine the minimal polynomial.
\end{problem}
\begin{solution}
$P$ satisfies $t(t-1)$, and neither $t$ nor $t-1$ alone annihilates it because $P$ is neither zero nor identity. Thus $m_P(t)=t(t-1)$, which has distinct roots, so $P$ is diagonalizable.
\end{solution}

\begin{problem}[Involution]\label{prob:i11-dossier-02}
If $T^2=I$ over a field of characteristic not $2$, prove $T$ is diagonalizable.
\end{problem}
\begin{solution}
The minimal polynomial divides $(t-1)(t+1)$, a product of distinct linear factors. Therefore the minimal polynomial is square-free and split, which is the diagonalizability criterion.
\end{solution}

\begin{problem}[Shear obstruction]\label{prob:i11-dossier-03}
Why is a nontrivial shear not diagonalizable?
\end{problem}
\begin{solution}
For $A=I+N$ with $N\ne0$ and $N^2=0$, the minimal polynomial is $(t-1)^2$. The repeated factor violates the square-free minimal-polynomial criterion.
\end{solution}

\begin{problem}[Eigenspace dimension test]\label{prob:i11-dossier-04}
An operator on a 5-dimensional space has eigenspaces of dimensions $2,1,2$ for three distinct eigenvalues. Is it diagonalizable?
\end{problem}
\begin{solution}
Distinct eigenspaces form a direct sum, and their dimensions sum to $5$. Their combined bases therefore form a basis of eigenvectors, so the operator is diagonalizable.
\end{solution}

\begin{problem}[Minimal polynomial from diagonal data]\label{prob:i11-dossier-05}
For $D=\operatorname{diag}(1,1,3,5,5)$ find $m_D$.
\end{problem}
\begin{solution}
The minimal polynomial must vanish at each distinct diagonal value and needs no repeated factors: $m_D=(t-1)(t-3)(t-5)$.
\end{solution}

\begin{problem}[Annihilating polynomial divisibility]\label{prob:i11-dossier-06}
Why does the minimal polynomial divide every polynomial $p$ with $p(T)=0$?
\end{problem}
\begin{solution}
Divide $p=qm_T+r$ with $\deg r<\deg m_T$. Evaluating at $T$ gives $0=r(T)$. Minimality of $m_T$ forces $r=0$, so $m_T\mid p$.
\end{solution}

\begin{problem}[Spectral projectors]\label{prob:i11-dossier-07}
For distinct eigenvalues $\lambda_1,\lambda_2$, write the projector onto $E_{\lambda_1}$ as a polynomial in $T$.
\end{problem}
\begin{solution}
$P_1=(T-\lambda_2I)/(\lambda_1-\lambda_2)$. It acts as $1$ on $E_{\lambda_1}$ and $0$ on $E_{\lambda_2}$.
\end{solution}

\begin{problem}[Repeated characteristic root]\label{prob:i11-dossier-08}
Why does a repeated root of the characteristic polynomial not imply non-diagonalizability?
\end{problem}
\begin{solution}
The identity has characteristic polynomial $(t-1)^n$ but minimal polynomial $t-1$. Diagonalizability is controlled by repeated roots in the minimal polynomial, not merely in the characteristic polynomial.
\end{solution}

\begin{problem}[Direct eigenspace sum]\label{prob:i11-dossier-09}
Prove eigenspaces for distinct eigenvalues have zero intersection.
\end{problem}
\begin{solution}
If $v$ lies in both $E_\lambda$ and $E_\mu$ with $\lambda\ne\mu$, then $\lambda v=Tv=\mu v$, so $(\lambda-\mu)v=0$ and hence $v=0$.
\end{solution}

\begin{problem}[Diagonalization by eigenbasis]\label{prob:i11-dossier-10}
Given $n$ independent eigenvectors in an $n$-dimensional space, how do you diagonalize the operator?
\end{problem}
\begin{solution}
Use the eigenvectors as an ordered basis. The image of each basis vector is its eigenvalue times itself, so the representing matrix is diagonal with those eigenvalues.
\end{solution}

\begin{problem}[Polynomial functional test]\label{prob:i11-dossier-11}
If $T$ is diagonalizable, why is every polynomial $p(T)$ diagonalizable in the same basis?
\end{problem}
\begin{solution}
In an eigenbasis, $T$ is diagonal. Applying a polynomial entrywise to the diagonal gives another diagonal matrix, so $p(T)$ is diagonal in the same basis.
\end{solution}

\begin{problem}[Diagonalization synthesis]\label{prob:i11-dossier-12}
Compare the roles of characteristic and minimal polynomials in diagonalization.
\end{problem}
\begin{solution}
The characteristic polynomial tells which eigenvalues are available and their algebraic multiplicities. The minimal polynomial records the largest Jordan-type obstruction. Splitting of the characteristic polynomial is necessary, while square-freeness of the split minimal polynomial is the decisive criterion.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:i11-01}
Why is every operator with distinct eigenvalues equal in number to the dimension diagonalizable?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Eigenvectors for distinct eigenvalues are independent, yielding a basis.
\end{solution}

\begin{exercise}\label{exr:i11-02}
Find the minimal polynomial of a diagonal matrix with diagonal entries $1,1,3$.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
It is $(t-1)(t-3)$; repeated diagonal occurrences do not create repeated factors in the minimal polynomial.
\end{solution}

\begin{exercise}\label{exr:i11-03}
Find the minimal polynomial of a nontrivial $2\times2$ shear.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
It is $(t-1)^2$: $A-I$ is nonzero but $(A-I)^2=0$.
\end{solution}

\begin{exercise}\label{exr:i11-04}
Show every projection is diagonalizable without computing a determinant.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Decompose $V=\operatorname{im}P\oplus\ker P$. The operator acts as $1$ on the image and $0$ on the kernel.
\end{solution}

\begin{exercise}\label{exr:i11-05}
If $T^2=I$, what are the possible eigenvalues?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Any eigenvalue $\lambda$ satisfies $\lambda^2=1$, so $\lambda=\pm1$.
\end{solution}

\begin{exercise}\label{exr:i11-06}
Why must $m_T$ divide every annihilating polynomial?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Divide $p$ by $m_T$; the remainder has smaller degree and also annihilates $T$, so minimality forces the remainder to vanish.
\end{solution}

\begin{exercise}\label{exr:i11-07}
Construct the spectral projector onto $E_{\lambda_i}$ for distinct eigenvalues.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Use $P_i=\prod_{j\ne i}(T-\lambda_jI)/(\lambda_i-\lambda_j)$; it equals $1$ on $E_{\lambda_i}$ and $0$ on the other eigenspaces.
\end{solution}

\begin{exercise}\label{exr:i11-08}
Explain why repeated roots in the characteristic polynomial do not by themselves prevent diagonalization.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
The identity matrix has characteristic polynomial $(t-1)^n$ but is diagonal; what matters is the minimal polynomial, which is $t-1$.
\end{solution}

\section*{Chapter summary}
Diagonalization asks whether an operator can be decomposed into independent one-dimensional modes. The minimal polynomial gives the sharp algebraic criterion: an operator is diagonalizable exactly when its minimal polynomial splits into distinct linear factors. This viewpoint unifies eigenvalue counting with polynomial identities satisfied by the operator.
The important habit is to move fluently between invariant statements and concrete coordinates while remembering which conclusions depend on finite dimensionality, the scalar field, or the inner product.
