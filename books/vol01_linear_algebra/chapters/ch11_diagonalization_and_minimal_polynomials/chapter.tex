\chapter{Diagonalization and Minimal Polynomials}
\label{ch:i11-diagonalization-and-minimal-polynomials}

Diagonalization asks whether an operator can be decomposed into independent one-dimensional modes. The minimal polynomial gives the sharp algebraic criterion: an operator is diagonalizable exactly when its minimal polynomial splits into distinct linear factors. This viewpoint unifies eigenvalue counting with polynomial identities satisfied by the operator.

\section{Diagonalizable operators}
An operator is diagonalizable when it has a basis of eigenvectors, equivalently when some matrix representation is diagonal.

\section{Direct sums of eigenspaces}
If the characteristic polynomial splits, diagonalizability is equivalent to $V$ being the direct sum of its eigenspaces.

\section{Minimal polynomial}
The minimal polynomial $m_T$ is the monic polynomial of least positive degree with $m_T(T)=0$.

\section{Divisibility property}
Every polynomial $p$ with $p(T)=0$ is divisible by $m_T$. In particular, $m_T$ divides the characteristic polynomial.

\section{Square-free criterion}
If $m_T$ splits, then $T$ is diagonalizable exactly when $m_T$ has no repeated root.

\section{Spectral projectors by interpolation}
For distinct eigenvalues, Lagrange interpolation polynomials in $T$ produce projections onto eigenspaces.

\section{Core structural results}

\begin{theorem}[Dimension criterion]\label{thm:i11-01}
If $\chi_T$ splits, then $T$ is diagonalizable if and only if the sum of eigenspace dimensions equals $\dim V$.
\begin{proof}
Distinct eigenspaces form a direct sum. Their union of bases is an eigenbasis exactly when the direct sum has full dimension.
\end{proof}
\end{theorem}

\begin{theorem}[Minimal-polynomial criterion]\label{thm:i11-02}
An operator is diagonalizable if and only if its minimal polynomial is a product of distinct linear factors.
\begin{proof}
For a diagonal operator, a polynomial annihilates it exactly when it vanishes at every diagonal eigenvalue. Conversely, use Lagrange polynomials to construct complementary eigenspace projectors.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Projection]\label{ex:i11-worked-01}
A projection $P^2=P$ has minimal polynomial dividing $t(t-1)$, which has distinct roots, so every projection is diagonalizable.
\end{example}

\begin{example}[Involution]\label{ex:i11-worked-02}
If $T^2=I$ over characteristic not $2$, then $m_T$ divides $(t-1)(t+1)$, so $T$ is diagonalizable.
\end{example}

\begin{example}[Nontrivial shear]\label{ex:i11-worked-03}
The shear has minimal polynomial $(t-1)^2$, so it is not diagonalizable.
\end{example}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:i11-01}
Why is every operator with distinct eigenvalues equal in number to the dimension diagonalizable?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Eigenvectors for distinct eigenvalues are independent, yielding a basis.
\end{solution}

\begin{exercise}\label{exr:i11-02}
Find the minimal polynomial of a diagonal matrix with diagonal entries $1,1,3$.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
It is $(t-1)(t-3)$; repeated diagonal occurrences do not create repeated factors in the minimal polynomial.
\end{solution}

\begin{exercise}\label{exr:i11-03}
Find the minimal polynomial of a nontrivial $2\times2$ shear.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
It is $(t-1)^2$: $A-I$ is nonzero but $(A-I)^2=0$.
\end{solution}

\begin{exercise}\label{exr:i11-04}
Show every projection is diagonalizable without computing a determinant.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Decompose $V=\operatorname{im}P\oplus\ker P$. The operator acts as $1$ on the image and $0$ on the kernel.
\end{solution}

\begin{exercise}\label{exr:i11-05}
If $T^2=I$, what are the possible eigenvalues?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Any eigenvalue $\lambda$ satisfies $\lambda^2=1$, so $\lambda=\pm1$.
\end{solution}

\begin{exercise}\label{exr:i11-06}
Why must $m_T$ divide every annihilating polynomial?
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Divide $p$ by $m_T$; the remainder has smaller degree and also annihilates $T$, so minimality forces the remainder to vanish.
\end{solution}

\begin{exercise}\label{exr:i11-07}
Construct the spectral projector onto $E_{\lambda_i}$ for distinct eigenvalues.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
Use $P_i=\prod_{j\ne i}(T-\lambda_jI)/(\lambda_i-\lambda_j)$; it equals $1$ on $E_{\lambda_i}$ and $0$ on the other eigenspaces.
\end{solution}

\begin{exercise}\label{exr:i11-08}
Explain why repeated roots in the characteristic polynomial do not by themselves prevent diagonalization.
\end{exercise}
\begin{hint}
Identify the definition or structural theorem in this chapter that directly controls the question, then reduce the calculation to that statement.
\end{hint}
\begin{solution}
The identity matrix has characteristic polynomial $(t-1)^n$ but is diagonal; what matters is the minimal polynomial, which is $t-1$.
\end{solution}
