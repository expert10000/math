\section{Supplementary worked examples}
\label{sec:vii05-pedagogy}

\begin{example}[Differentials form covectors]\label{ex:vii05-ped-01}
For a smooth function \(f\), the differential \(df_p\) is the covector \(v\mapsto v(f)\).  In coordinates,
\[
df_p=\sum_i \frac{\partial f}{\partial x^i}(p)\,dx^i.
\]
Thus gradients are metric-dependent vectors, while differentials exist before any metric has been chosen.
\end{example}

\begin{example}[Conormal space of a level set]\label{ex:vii05-ped-02}
If \(M=F^{-1}(c)\subset\mathbb R^n\) is a regular hypersurface, then \(dF_p\) annihilates \(T_pM\).  The one-dimensional conormal space is therefore spanned by \(dF_p\).  For the unit sphere this covector is \(2\sum p_i\,dx^i\).
\end{example}

\begin{example}[Pullback acts contravariantly]\label{ex:vii05-ped-03}
For \(f:M\to N\) and \(\alpha\in T^*_{f(p)}N\),
\[
(f^*\alpha)_p(v)=\alpha(df_pv).
\]
Vectors move forward by \(df\), while covectors move backward by \(f^*\).  This dual direction is the prototype for the pullback of differential forms in VII/09.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Compute a differential]\label{exr:vii05-ped-01}
For \(f(x,y)=x^2e^y\), compute \(df_{(1,0)}\).
\end{exercise}
\begin{hint}
Compute the two partial derivatives.
\end{hint}
\begin{solution}
\(df=2xe^y\,dx+x^2e^y\,dy\), so \(df_{(1,0)}=2\,dx+dy\).
\end{solution}

\begin{exercise}[Dual basis]\label{exr:vii05-ped-02}
If \(e_1,e_2\) is a basis, state the defining equations for its dual basis \(e^1,e^2\).
\end{exercise}
\begin{hint}
Use Kronecker delta notation.
\end{hint}
\begin{solution}
The dual basis satisfies \(e^i(e_j)=\delta^i_j\).
\end{solution}

\begin{exercise}[Pullback of a covector]\label{exr:vii05-ped-03}
For \(f(x,y)=(x+y,x-y)\), compute \(f^*(du)\) and \(f^*(dv)\).
\end{exercise}
\begin{hint}
Differentiate the component functions \(u=x+y\) and \(v=x-y\).
\end{hint}
\begin{solution}
\(f^*du=dx+dy\) and \(f^*dv=dx-dy\).
\end{solution}

\begin{exercise}[Sphere conormal]\label{exr:vii05-ped-04}
Find a covector spanning the conormal line to \(S^2\) at \(p=(0,0,1)\).
\end{exercise}
\begin{hint}
Use \(F=x^2+y^2+z^2\).
\end{hint}
\begin{solution}
\(dF_p=2\,dz\), so \(dz\) spans the conormal line.
\end{solution}

\begin{exercise}[Coordinate covectors]\label{exr:vii05-ped-05}
Under \(u=x+y,\ v=x-y\), express \(dx,dy\) in terms of \(du,dv\).
\end{exercise}
\begin{hint}
Solve the two linear equations.
\end{hint}
\begin{solution}
\(dx=\tfrac12(du+dv)\) and \(dy=\tfrac12(du-dv)\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Differential is linear]\label{exr:vii05-ped-06}
Prove \(df_p:T_pM\to\mathbb R\) is linear.
\end{exercise}
\begin{hint}
Use linearity of derivations.
\end{hint}
\begin{solution}
For tangent derivations \(v,w\) and scalars \(a,b\), \(df_p(av+bw)=(av+bw)(f)=a\,v(f)+b\,w(f)\).
\end{solution}

\begin{exercise}[Pullback composition]\label{exr:vii05-ped-07}
Prove \((g\circ f)^*=f^*\circ g^*\) on covectors.
\end{exercise}
\begin{hint}
Evaluate both sides on a tangent vector.
\end{hint}
\begin{solution}
Both sides send \(\alpha\) and \(v\) to \(\alpha(dg\,df\,v)\), using the chain rule.
\end{solution}

\begin{exercise}[Conormal annihilator]\label{exr:vii05-ped-08}
Prove that \(dF_p\) annihilates \(T_pF^{-1}(c)\).
\end{exercise}
\begin{hint}
Differentiate \(F\circ\gamma=c\) along tangent curves.
\end{hint}
\begin{solution}
For any tangent velocity \(v\), \(dF_p(v)=0\); equivalently \(T_pF^{-1}(c)\subset\ker dF_p\), and regularity gives equality.
\end{solution}

\begin{exercise}[Cotangent dimension]\label{exr:vii05-ped-09}
Prove \(\dim T_p^*M=\dim T_pM\).
\end{exercise}
\begin{hint}
Use finite-dimensional linear algebra.
\end{hint}
\begin{solution}
The cotangent space is the dual of the finite-dimensional tangent space, and a vector-space dual has the same dimension.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Differential equals gradient]\label{exr:vii05-ped-10}
Test the statement before a Riemannian metric is chosen.
\end{exercise}
\begin{hint}
Ask which object is naturally a covector.
\end{hint}
\begin{solution}
False as stated.  \(df\) is canonical; the gradient is the vector obtained from \(df\) after using a metric to identify vectors and covectors.
\end{solution}

\begin{exercise}[Covectors push forward naturally]\label{exr:vii05-ped-11}
Test whether a general smooth map has a canonical pushforward on covectors.
\end{exercise}
\begin{hint}
Dualize \(df_p:T_pM\to T_{f(p)}N\).
\end{hint}
\begin{solution}
The natural dual map goes in the reverse direction \(T^*_{f(p)}N\to T_p^*M\); there is no general covector pushforward without extra hypotheses or structure.
\end{solution}

\begin{exercise}[Nonzero \(df\) vanishes on no tangent vector]\label{exr:vii05-ped-12}
Test this in dimension at least two.
\end{exercise}
\begin{hint}
A nonzero linear functional has a kernel.
\end{hint}
\begin{solution}
False.  On an \(n\)-dimensional space with \(n>1\), a nonzero covector has an \((n-1)\)-dimensional kernel.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Constraint forces]\label{exr:vii05-ped-13}
Interpret \(dF\) for a hypersurface constraint \(F=c\).
\end{exercise}
\begin{hint}
Compare its kernel with tangent velocities.
\end{hint}
\begin{solution}
\(dF\) annihilates admissible tangent motions, so it spans the conormal direction in which a constraint force covector naturally lives.
\end{solution}

\begin{exercise}[Coordinate-free first derivatives]\label{exr:vii05-ped-14}
Explain why \(df_p\) is a coordinate-independent first derivative.
\end{exercise}
\begin{hint}
Compare its coordinate formulas under a chart change.
\end{hint}
\begin{solution}
Its component row transforms by the inverse Jacobian exactly as a covector, so the underlying linear functional is independent of coordinates.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Annihilator dimension]\label{exr:vii05-ped-15}
If \(W\subset V\) has dimension \(k\) in an \(n\)-dimensional space, prove \(\dim W^\circ=n-k\).
\end{exercise}
\begin{hint}
Extend a basis of \(W\) to a basis of \(V\) and use the dual basis.
\end{hint}
\begin{solution}
The dual covectors corresponding to the added \(n-k\) basis vectors vanish on \(W\) and form a basis of the annihilator.
\end{solution}

\begin{exercise}[Cotangent map of a diffeomorphism]\label{exr:vii05-ped-16}
Show that a diffeomorphism induces a cotangent-space isomorphism by pullback.
\end{exercise}
\begin{hint}
Use the pullback of the inverse map.
\end{hint}
\begin{solution}
If \(f\) is a diffeomorphism, \(f^*:T^*_{f(p)}N\to T_p^*M\) has inverse \((f^{-1})^*\) by functoriality.
\end{solution}
