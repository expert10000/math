\chapter{Kleinian Groups and Boundary Geometry}
\label{ch:vii37-kleinian}

\section{Discrete groups in three-dimensional hyperbolic geometry}

\begin{definition}[Kleinian group]\label{def:vii37-kleinian}
A \emph{Kleinian group} is a discrete subgroup
\[
\Gamma<\operatorname{PSL}(2,\mathbb C).
\]
It acts by orientation-preserving isometries on \(\mathbb H^3\) and by Möbius transformations on the ideal boundary \(\widehat{\mathbb C}\).
\end{definition}

Interior discreteness implies proper discontinuity on \(\mathbb H^3\), but the boundary action is typically not properly discontinuous everywhere.  The boundary therefore splits into two dynamically different pieces.


\section{Limit set and domain of discontinuity}

\begin{definition}[Limit set]\label{def:vii37-limitset}
For an infinite Kleinian group \(\Gamma\), the limit set \(\Lambda(\Gamma)\subset\widehat{\mathbb C}\) is the set of accumulation points of an orbit \(\Gamma p\), where \(p\in\mathbb H^3\).
\end{definition}

\begin{definition}[Domain of discontinuity]\label{def:vii37-domain}
The domain of discontinuity is
\[
\Omega(\Gamma)=\widehat{\mathbb C}\setminus\Lambda(\Gamma).
\]
\end{definition}

\begin{theorem}[Boundary proper discontinuity]\label{thm:vii37-omega-proper}
The action of \(\Gamma\) on \(\Omega(\Gamma)\) is properly discontinuous.
\end{theorem}

The limit set is closed, \(\Gamma\)-invariant, and independent of the chosen interior base point.


\section{Elementary and non-elementary groups}

\begin{definition}[Elementary Kleinian group]\label{def:vii37-elementary}
A Kleinian group is elementary if its limit set has at most two points.  Otherwise it is non-elementary.
\end{definition}

A cyclic parabolic group has one-point limit set.  A cyclic loxodromic group has a two-point limit set consisting of the fixed points of its generator.  Non-elementary groups have much richer dynamics; their limit sets are infinite and perfect.


\section{Loxodromic fixed points and density}

\begin{theorem}[Density of loxodromic fixed points]\label{thm:vii37-lox-density}
For a non-elementary Kleinian group, the fixed points of loxodromic elements are dense in \(\Lambda(\Gamma)\).
\end{theorem}

This theorem makes the one-element north--south dynamics from VII/36 into a global tool for understanding the group action on the limit set.


\section{Fuchsian groups as special Kleinian groups}

A Fuchsian group, viewed inside \(\operatorname{PSL}(2,\mathbb C)\), preserves a generalized circle \(C\subset\widehat{\mathbb C}\) and the corresponding totally geodesic plane in \(\mathbb H^3\).  Its limit set lies in \(C\).

\begin{example}[First kind]\label{ex:vii37-first-kind}
For a cofinite Fuchsian group such as \(\operatorname{PSL}(2,\mathbb Z)\), the limit set is the entire invariant circle.
\end{example}


\section{Convex hulls and convex cores}

Let \(\operatorname{Hull}(\Lambda)\subset\mathbb H^3\) denote the smallest closed hyperbolically convex set whose ideal boundary contains \(\Lambda(\Gamma)\).

\begin{definition}[Convex core]\label{def:vii37-convex-core}
The convex core of the quotient \(M=\Gamma\backslash\mathbb H^3\) is
\[
\operatorname{Core}(M)=\Gamma\backslash\operatorname{Hull}(\Lambda(\Gamma)).
\]
\end{definition}

The convex core packages the part of the quotient directly controlled by recurrent group dynamics.  Outside it, geometry heads toward conformal boundary components when \(\Omega(\Gamma)\ne\varnothing\).


\section{Quotient three-manifolds and orbifolds}

\begin{theorem}[Hyperbolic quotient]\label{thm:vii37-quotient}
If \(\Gamma\) is torsion-free and Kleinian, then
\[
M=\Gamma\backslash\mathbb H^3
\]
is an orientable hyperbolic three-manifold.  With torsion, the natural quotient is a hyperbolic orbifold.
\end{theorem}

Parabolic subgroups generate cusp ends, while loxodromic elements generate closed geodesics, now possibly accompanied by rotational holonomy around their axes.


\section{Conformal boundary}

Because the action on \(\Omega(\Gamma)\) is properly discontinuous, one may form the quotient
\[
\partial_c M=\Gamma\backslash\Omega(\Gamma).
\]
When nonempty, this is the \emph{conformal boundary} of the hyperbolic three-manifold or orbifold.  It is naturally a Riemann surface or Riemann orbifold because Möbius transformations are conformal.

\begin{warning}
The conformal boundary is not the same object as the topological ideal boundary \(\widehat{\mathbb C}\), and it may be empty when \(\Lambda(\Gamma)=\widehat{\mathbb C}\).
\end{warning}


\section{Schottky groups as a geometric model}

A classical Schottky group is generated by loxodromic transformations pairing disjoint Jordan curves (classically, circles) on the Riemann sphere.  Its domain of discontinuity is nonempty and its limit set is a Cantor-type fractal.

\begin{remark}
Schottky groups provide a clean model in which algebraic freeness, hyperbolic quotient geometry, conformal boundary, and fractal limit sets can all be seen at once.
\end{remark}


\section{Why the boundary matters}

The limit set records asymptotic orbit accumulation, while the domain of discontinuity records where the boundary action remains geometrically tame.  Together they convert a discrete subgroup of matrices into a global dynamical object.

For non-elementary groups, \(\Lambda(\Gamma)\) is the smallest nonempty closed \(\Gamma\)-invariant subset of \(\widehat{\mathbb C}\).  This minimality is one of the basic organizing principles of Kleinian dynamics.


\section{Bridge to discrete geodesic problems}

Chapters VII/31--VII/37 built a continuous geometric arc from hyperbolic models through discrete isometry groups and quotient geometry.  VII/38 changes viewpoint: geodesics are no longer sought on smooth model spaces alone, but on triangulated and computational representations.  The next part studies discrete geodesic problems and the algorithms used to approximate or compute them.

\section{Solved problem dossiers}

\begin{problem}[Cyclic loxodromic limit set]\label{prob:vii37-01}
Let \(g(z)=2z\).  Determine \(\Lambda(\langle g\rangle)\) and \(\Omega(\langle g\rangle)\).
\end{problem}

\begin{solution}
For every \(z\ne0,\infty\), forward iterates tend to \(\infty\) and backward iterates tend to \(0\).  Thus \(\Lambda=\{0,\infty\}\) and \(\Omega=\widehat{\mathbb C}\setminus\{0,\infty\}=\mathbb C^*\).
\end{solution}

\begin{problem}[Cyclic parabolic limit set]\label{prob:vii37-02}
Let \(g(z)=z+1\).  Determine its limit set.
\end{problem}

\begin{solution}
The unique fixed point on the sphere is \(\infty\).  For every finite \(z\), \(g^n(z)=z+n\to\infty\) as \(|n|\to\infty\) in the spherical topology.  Hence \(\Lambda=\{\infty\}\).
\end{solution}

\begin{problem}[Why the limit set is invariant]\label{prob:vii37-03}
Prove that \(\Lambda(\Gamma)\) is \(\Gamma\)-invariant.
\end{problem}

\begin{solution}
If \(x=\lim \gamma_n p\) and \(h\in\Gamma\), then by continuity on the compactification, \(hx=\lim h\gamma_n p\).  Since \(h\gamma_n\in\Gamma\), the point \(hx\) is again a limit point.  Applying the same argument to \(h^{-1}\) gives equality.
\end{solution}

\begin{problem}[Base-point independence]\label{prob:vii37-04}
Explain why the limit set does not depend on the chosen interior point \(p\).
\end{problem}

\begin{solution}
For \(p,q\in\mathbb H^3\), isometry invariance gives \(d(\gamma p,\gamma q)=d(p,q)\) for all \(\gamma\).  Two sequences staying at bounded hyperbolic distance converge to the same ideal point, so the orbit accumulation sets coincide.
\end{solution}

\begin{problem}[Fuchsian limit set inside a circle]\label{prob:vii37-05}
Let \(\Gamma<\operatorname{PSL}(2,\mathbb R)\) be viewed as Kleinian.  Show \(\Lambda(\Gamma)\subset\mathbb R\cup\{\infty\}\).
\end{problem}

\begin{solution}
The group preserves the totally geodesic plane \(P\cong\mathbb H^2\) whose ideal boundary is \(\mathbb R\cup\{\infty\}\).  Choosing the orbit base point inside \(P\), every orbit point lies in \(P\), so all ideal accumulation points lie in \(\partial_\infty P\).
\end{solution}

\begin{problem}[Closed geodesic from a loxodromic]\label{prob:vii37-06}
Let \(\Gamma\) be torsion-free and contain a primitive loxodromic \(g\).  Explain the resulting closed geodesic in \(M=\Gamma\backslash\mathbb H^3\).
\end{problem}

\begin{solution}
The loxodromic preserves its unique axis and translates along it by \(\ell>0\).  Quotienting the axis by \(\langle g\rangle\) gives a circle of length \(\ell\), which maps locally isometrically to a closed geodesic in \(M\).
\end{solution}

\begin{problem}[Conformal boundary of a cyclic loxodromic group]\label{prob:vii37-07}
For \(g(z)=qz\), \(|q|>1\), describe \(\Omega/\langle g\rangle\).
\end{problem}

\begin{solution}
Here \(\Omega=\mathbb C^*\).  Passing to logarithmic coordinate \(u=\log z\), multiplication by \(q\) becomes translation by \(\log q\), while \(u\sim u+2\pi i\).  The quotient is therefore a complex torus when the two periods are independent over \(\mathbb R\).
\end{solution}

\begin{problem}[When conformal boundary is empty]\label{prob:vii37-08}
What happens if \(\Lambda(\Gamma)=\widehat{\mathbb C}\)?
\end{problem}

\begin{solution}
Then \(\Omega(\Gamma)=\varnothing\), so there is no conformal-boundary quotient.  The boundary action is dynamically recurrent everywhere.
\end{solution}

\begin{problem}[Elementary versus non-elementary dynamics]\label{prob:vii37-09}
Contrast a cyclic loxodromic group with a non-elementary Kleinian group.
\end{problem}

\begin{solution}
The cyclic loxodromic group has exactly two limit points and completely organized north--south dynamics.  A non-elementary group has an infinite perfect limit set, with loxodromic fixed points dense and many competing attracting--repelling pairs.
\end{solution}

\begin{problem}[Convex hull for two points]\label{prob:vii37-10}
What is the convex hull in \(\mathbb H^3\) of a two-point limit set?
\end{problem}

\begin{solution}
The smallest closed convex set with those two ideal endpoints is the unique geodesic joining them.  Thus the convex core of a cyclic loxodromic quotient is the closed geodesic obtained from its axis.
\end{solution}

\begin{problem}[Why Schottky groups are non-elementary]\label{prob:vii37-11}
A rank-two classical Schottky group has a Cantor limit set.  Explain why it is non-elementary.
\end{problem}

\begin{solution}
A Cantor set is infinite, so the limit set has more than two points.  By definition the group is therefore non-elementary.
\end{solution}

\begin{problem}[Minimality of the limit set]\label{prob:vii37-12}
Suppose \(\Gamma\) is non-elementary and \(C\subset\widehat{\mathbb C}\) is nonempty, closed, and \(\Gamma\)-invariant.  Explain why one expects \(\Lambda(\Gamma)\subset C\).
\end{problem}

\begin{solution}
The attracting and repelling fixed points of loxodromic elements lie in every nonempty closed invariant set once one iterates suitable points.  Since those fixed points are dense in \(\Lambda(\Gamma)\), closedness forces \(\Lambda(\Gamma)\subset C\).
\end{solution}

\section{Challenge problems}

\begin{problem}[A fractal versus a circle]\label{prob:vii37-ch01}
Compare the qualitative boundary geometry of a cofinite Fuchsian group and a classical Schottky group.
\end{problem}

\begin{solution}
A cofinite Fuchsian group has limit set equal to its invariant circle, leaving two disk components in the sphere when regarded in three dimensions.  A classical Schottky group has a Cantor-type limit set, so its domain of discontinuity is connected in the standard classical situation and has a highly nontrivial quotient.  The contrast shows how discreteness alone permits radically different boundary geometry.
\end{solution}

\begin{problem}[Convex-core viewpoint]\label{prob:vii37-ch02}
Why is the convex core often a better finite geometric object than the whole quotient \(\Gamma\backslash\mathbb H^3\)?
\end{problem}

\begin{solution}
The whole quotient may contain infinite ends extending toward components of the conformal boundary.  The convex core removes those geometrically redundant outer regions and retains the quotient of the convex hull of the recurrent limit-set dynamics.  For geometrically finite groups it is controlled by finite-type data up to cusp structure.
\end{solution}

\begin{problem}[Parabolic degeneration intuition]\label{prob:vii37-ch03}
Explain qualitatively how a loxodromic element can degenerate toward a parabolic one in a family of Kleinian groups.
\end{problem}

\begin{solution}
As translation length tends to zero, the two loxodromic fixed points may coalesce into one ideal fixed point.  In the limit the invariant axis disappears as a two-ended object and the isometry becomes parabolic.  This mechanism underlies cusp formation in deformation theory.
\end{solution}

\begin{problem}[Boundary quotient as complex geometry]\label{prob:vii37-ch04}
Why does Kleinian theory naturally connect hyperbolic three-geometry to Riemann surfaces?
\end{problem}

\begin{solution}
The interior quotient is hyperbolic three-dimensional, while the tame boundary region \(\Omega\) carries the standard complex structure of the Riemann sphere.  Since the group acts there by Möbius transformations, the quotient \(\Omega/\Gamma\) inherits a Riemann-surface or orbifold structure.  Thus one group controls both three-dimensional hyperbolic and two-dimensional conformal geometry.
\end{solution}

\begin{problem}[Transition to computation]\label{prob:vii37-ch05}
Give a precise conceptual reason that discrete geodesic algorithms form a natural next part after Kleinian groups.
\end{problem}

\begin{solution}
The preceding chapters show how global geometry can be encoded by discrete algebraic data acting on smooth spaces.  Computational geometry makes a complementary discretization: the space itself is represented by a mesh or graph, and geodesic questions become finite optimization or PDE problems.  The transition preserves the central theme of recovering continuous geometric structure from discrete information.
\end{solution}
