\section{Supplementary worked examples}
\label{sec:vii37-pedagogy}

\begin{example}[Cyclic loxodromic group]\label{ex:vii37-ped-01}
For \(\Gamma=\langle z\mapsto\lambda z\rangle\) with \(|\lambda|\ne1\), the limit set is the pair \(\{0,\infty\}\).  This is an elementary Kleinian group with north--south boundary dynamics.
\end{example}

\begin{example}[Cyclic parabolic group]\label{ex:vii37-ped-02}
For \(\Gamma=\langle z\mapsto z+1\rangle\), the unique limit point is \(\infty\).  The domain of discontinuity is the sphere minus that point, and the group is again elementary.
\end{example}

\begin{example}[Schottky picture]\label{ex:vii37-ped-03}
A classical Schottky group pairs disjoint boundary circles by loxodromic generators.  Its limit set is Cantor-like, while the complement supports properly discontinuous boundary action and a nontrivial conformal quotient.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Limit set cyclic lox]\label{exr:vii37-ped-01}
What is the limit set of \(\langle z\mapsto2z\rangle\)?
\end{exercise}

\begin{hint}
Find the two fixed points.
\end{hint}

\begin{solution}
\(\{0,\infty\}\).
\end{solution}

\begin{exercise}[Limit set parabolic]\label{exr:vii37-ped-02}
What is the limit set of \(\langle z\mapsto z+1\rangle\)?
\end{exercise}

\begin{hint}
Iterates accumulate only at one ideal point.
\end{hint}

\begin{solution}
\(\{\infty\}\).
\end{solution}

\begin{exercise}[Domain]\label{exr:vii37-ped-03}
How is \(\Omega(\Gamma)\) defined?
\end{exercise}

\begin{hint}
Take a complement in the Riemann sphere.
\end{hint}

\begin{solution}
\(\widehat{\mathbb C}\setminus\Lambda(\Gamma)\).
\end{solution}

\begin{exercise}[Elementary]\label{exr:vii37-ped-04}
When is a Kleinian group elementary?
\end{exercise}

\begin{hint}
Count limit-set points.
\end{hint}

\begin{solution}
When its limit set has at most two points.
\end{solution}

\begin{exercise}[Convex core]\label{exr:vii37-ped-05}
Define the convex core of \(\Gamma\backslash\mathbb H^3\).
\end{exercise}

\begin{hint}
Quotient the convex hull of the limit set.
\end{hint}

\begin{solution}
\(\Gamma\backslash\operatorname{Hull}(\Lambda(\Gamma))\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Base-point independence]\label{exr:vii37-ped-06}
Why is the limit set independent of the interior orbit base point?
\end{exercise}

\begin{hint}
Orbits of two base points stay within bounded hyperbolic distance under the same group elements.
\end{hint}

\begin{solution}
Bounded-distance sequences approach the same ideal accumulation points, so the boundary accumulation set is unchanged.
\end{solution}

\begin{exercise}[Invariant closed]\label{exr:vii37-ped-07}
Why is \(\Lambda(\Gamma)\) closed and \(\Gamma\)-invariant?
\end{exercise}

\begin{hint}
Use accumulation points and continuity of the action.
\end{hint}

\begin{solution}
Limits of orbit accumulation points remain accumulation points, and group elements map orbit sequences to orbit sequences.
\end{solution}

\begin{exercise}[Quotient manifold]\label{exr:vii37-ped-08}
Why does torsion-free discreteness give a hyperbolic three-manifold quotient?
\end{exercise}

\begin{hint}
The interior action is free and properly discontinuous.
\end{hint}

\begin{solution}
Local neighborhoods descend isometrically with no orbifold stabilizers.
\end{solution}

\begin{exercise}[Conformal boundary]\label{exr:vii37-ped-09}
Why is \(\Gamma\backslash\Omega(\Gamma)\) naturally conformal?
\end{exercise}

\begin{hint}
Boundary transformations are Mobius maps.
\end{hint}

\begin{solution}
Mobius maps are conformal, so the quotient inherits a Riemann-surface or orbifold conformal structure.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Boundary action proper everywhere]\label{exr:vii37-ped-10}
Test the claim for a non-elementary Kleinian group.
\end{exercise}

\begin{hint}
The limit set is precisely where dynamics accumulate.
\end{hint}

\begin{solution}
False.  Proper discontinuity holds on \(\Omega\), not generally on \(\Lambda\).
\end{solution}

\begin{exercise}[Limit set always finite]\label{exr:vii37-ped-11}
Test the claim.
\end{exercise}

\begin{hint}
Non-elementary groups have rich limit sets.
\end{hint}

\begin{solution}
False.  Non-elementary limit sets are infinite and perfect.
\end{solution}

\begin{exercise}[Conformal boundary always nonempty]\label{exr:vii37-ped-12}
Test the claim.
\end{exercise}

\begin{hint}
It is empty when the limit set is the whole sphere.
\end{hint}

\begin{solution}
False.  Some Kleinian groups have \(\Omega=\varnothing\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Fractal visualization]\label{exr:vii37-ped-13}
Why are limit sets natural objects for computational visualization?
\end{exercise}

\begin{hint}
Iterate group generators on boundary points.
\end{hint}

\begin{solution}
They encode asymptotic dynamics and often form self-similar or fractal sets that can be approximated by orbit sampling.
\end{solution}

\begin{exercise}[Convex core role]\label{exr:vii37-ped-14}
What geometry does the convex core isolate?
\end{exercise}

\begin{hint}
Focus on the hull of recurrent boundary dynamics.
\end{hint}

\begin{solution}
It captures the smallest convex part of the quotient carrying the essential hyperbolic dynamics associated with the limit set.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Fuchsian special case]\label{exr:vii37-ped-15}
Where does the limit set of a Fuchsian group lie when viewed as Kleinian?
\end{exercise}

\begin{hint}
The group preserves a generalized circle.
\end{hint}

\begin{solution}
It lies in the invariant boundary circle of the preserved totally geodesic plane.
\end{solution}

\begin{exercise}[Schottky challenge]\label{exr:vii37-ped-16}
Why can a free Schottky group have a Cantor-type limit set?
\end{exercise}

\begin{hint}
Repeated circle pairings create nested allowed boundary pieces.
\end{hint}

\begin{solution}
Iterating loxodromic pairings removes complementary domains in a nested way, leaving a compact perfect totally disconnected invariant set.
\end{solution}
