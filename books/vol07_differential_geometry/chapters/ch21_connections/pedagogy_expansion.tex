\section{Supplementary worked examples}
\label{sec:vii21-pedagogy}

\begin{example}[Flat connection in a rotating frame]\label{ex:vii21-ped-01}
The standard connection on \(\mathbb R^2\) has zero coefficients in a Cartesian frame.  In a rotating orthonormal frame \(e'=eA(\theta)\), its connection form becomes \(\omega'=A^{-1}dA\), making the inhomogeneous frame-change term visible even though the underlying connection is flat.
\end{example}

\begin{example}[A bundle connection on a trivial line bundle]\label{ex:vii21-ped-02}
On the trivial line bundle over \(\mathbb R\), \(\nabla s=ds+a(x)s\,dx\) is a connection for every smooth function \(a\).  A parallel section solves \(s'+a(x)s=0\), so bundle geometry immediately becomes a linear ODE.
\end{example}

\begin{example}[Difference of connections is tensorial]\label{ex:vii21-ped-03}
If \(\widetilde\nabla=\nabla+A\) with \(A\in\Omega^1(M;\operatorname{End}TM)\), then all non-tensorial derivative terms cancel in \(\widetilde\nabla_XY-\nabla_XY=A(X)Y\).  Connections form an affine space rather than a vector space.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Slot linearity]\label{exr:vii21-ped-01}
Which slot of \(\nabla_XY\) is \(C^\infty\)-linear?
\end{exercise}

\begin{hint}
Recall the defining axioms.
\end{hint}

\begin{solution}
The first slot \(X\) is \(C^\infty\)-linear; the second obeys a Leibniz rule.
\end{solution}

\begin{exercise}[Coordinate derivative]\label{exr:vii21-ped-02}
If all \(\Gamma^k_{ij}=0\), compute \(\nabla_XY\) in coordinates.
\end{exercise}

\begin{hint}
Use the coordinate formula.
\end{hint}

\begin{solution}
\(\nabla_XY=X^i(\partial_iY^k)\partial_k\).
\end{solution}

\begin{exercise}[Line-bundle parallel section]\label{exr:vii21-ped-03}
Solve \(s'+2s=0\) with \(s(0)=1\).
\end{exercise}

\begin{hint}
Use the scalar linear ODE.
\end{hint}

\begin{solution}
\(s(x)=e^{-2x}\).
\end{solution}

\begin{exercise}[Connection form]\label{exr:vii21-ped-04}
For a rank-one bundle with \(\nabla e=\alpha\otimes e\), compute \(\nabla(fe)\).
\end{exercise}

\begin{hint}
Apply Leibniz.
\end{hint}

\begin{solution}
\(\nabla(fe)=(df+f\alpha)\otimes e\).
\end{solution}

\begin{exercise}[Parallel components]\label{exr:vii21-ped-05}
Write the component ODE for a vector field along \(\gamma\).
\end{exercise}

\begin{hint}
Insert \(X=\dot\gamma\) in the coordinate formula.
\end{hint}

\begin{solution}
\(\dot V^k+\Gamma^k_{ij}(\gamma)\dot x^iV^j=0\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Difference tensor]\label{exr:vii21-ped-06}
Prove the difference of two connections is \(C^\infty\)-linear in both arguments.
\end{exercise}

\begin{hint}
Subtract the two Leibniz rules.
\end{hint}

\begin{solution}
The derivative terms cancel in each slot, leaving a tensorial bilinear map.
\end{solution}

\begin{exercise}[Affine family]\label{exr:vii21-ped-07}
Prove \(\nabla+A\) is a connection when \(A\in\Omega^1(M;\operatorname{End}E)\).
\end{exercise}

\begin{hint}
Check the Leibniz rule.
\end{hint}

\begin{solution}
Tensoriality of \(A\) adds no derivatives of scalar functions, so the connection axioms remain valid.
\end{solution}

\begin{exercise}[Gauge law]\label{exr:vii21-ped-08}
Derive \(\omega'=A^{-1}\omega A+A^{-1}dA\) from \(e'=eA\).
\end{exercise}

\begin{hint}
Differentiate the new frame and rewrite in it.
\end{hint}

\begin{solution}
Using \(\nabla(eA)=(\nabla e)A+e\,dA\) and \(e=e'A^{-1}\) gives the stated transformation law.
\end{solution}

\begin{exercise}[Parallel uniqueness]\label{exr:vii21-ped-09}
Why is parallel transport along a fixed curve uniquely determined by initial data?
\end{exercise}

\begin{hint}
Recognize a linear first-order ODE.
\end{hint}

\begin{solution}
Standard uniqueness for linear ODE systems determines the component vector uniquely.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Christoffel tensor]\label{exr:vii21-ped-10}
Test the claim: \(\Gamma^k_{ij}\) are tensor components.
\end{exercise}

\begin{hint}
Change from Cartesian to polar coordinates in the Euclidean plane.
\end{hint}

\begin{solution}
False.  They vanish in Cartesian coordinates and not in polar coordinates for the same connection.
\end{solution}

\begin{exercise}[Zero coefficients globally]\label{exr:vii21-ped-11}
Test the claim: every connection admits coordinates with all coefficients zero on an open set.
\end{exercise}

\begin{hint}
Think about curvature.
\end{hint}

\begin{solution}
False in general.  Nonzero curvature obstructs a frame or coordinate system with identically vanishing coefficients on an open set.
\end{solution}

\begin{exercise}[Connection equals derivative]\label{exr:vii21-ped-12}
Test the claim: a connection is an ordinary derivative of vector-valued functions.
\end{exercise}

\begin{hint}
Vectors at different points live in different fibers.
\end{hint}

\begin{solution}
False intrinsically; a connection supplies the extra comparison rule required to differentiate sections.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Gauge potential]\label{exr:vii21-ped-13}
Interpret a connection one-form as a gauge potential on a trivial bundle.
\end{exercise}

\begin{hint}
Use the transformation law.
\end{hint}

\begin{solution}
Its inhomogeneous transformation law is exactly the characteristic behavior of a gauge potential under change of local frame.
\end{solution}

\begin{exercise}[Numerical transport]\label{exr:vii21-ped-14}
What ODE would a numerical code integrate to transport tangent vectors along a known path?
\end{exercise}

\begin{hint}
Use the parallel component equation.
\end{hint}

\begin{solution}
It integrates \(\dot V^k=-\Gamma^k_{ij}(\gamma)\dot x^iV^j\) together with the given path data.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Normal frame at a point]\label{exr:vii21-ped-15}
Explain why one can often choose a frame with connection coefficients zero at one point but not on a whole neighborhood.
\end{exercise}

\begin{hint}
Compare pointwise normalization with curvature.
\end{hint}

\begin{solution}
A frame can absorb the connection value at one point, while derivatives of the connection encode curvature and cannot generally be removed on an open set.
\end{solution}

\begin{exercise}[Two connections]\label{exr:vii21-ped-16}
Let \(\widetilde\nabla_XY=\nabla_XY+\alpha(X)Y\).  Show this is a connection and identify the difference tensor.
\end{exercise}

\begin{hint}
Here \(\alpha\) is a one-form.
\end{hint}

\begin{solution}
It is a connection; the difference tensor is \(A(X)Y=\alpha(X)Y\).
\end{solution}
