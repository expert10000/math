\section{Supplementary worked examples}
\label{sec:vii22-pedagogy}

\begin{example}[Polar Christoffel symbols]\label{ex:vii22-ped-01}
For \(g=dr^2+r^2d\theta^2\), the Levi--Civita formula gives \(\Gamma^r_{\theta\theta}=-r\) and \(\Gamma^\theta_{r\theta}=\Gamma^\theta_{\theta r}=1/r\).  Nonzero symbols here describe curvilinear coordinates on flat space, not intrinsic curvature.
\end{example}

\begin{example}[Conformal Euclidean metric]\label{ex:vii22-ped-02}
For \(g=e^{2\varphi}(dx^2+dy^2)\), the Christoffel symbols are linear in first derivatives of \(\varphi\).  Thus a conformal factor changes the connection even though it preserves angles.
\end{example}

\begin{example}[Sphere by tangential projection]\label{ex:vii22-ped-03}
For the round sphere embedded in \(\mathbb R^3\), the Levi--Civita derivative of a tangent field is obtained by differentiating in the ambient space and orthogonally projecting back to the tangent plane.  The discarded normal component is the second fundamental form.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Torsion symmetry]\label{exr:vii22-ped-01}
What symmetry do Christoffel symbols satisfy in a coordinate frame for a torsion-free connection?
\end{exercise}

\begin{hint}
Coordinate fields commute.
\end{hint}

\begin{solution}
\(\Gamma^k_{ij}=\Gamma^k_{ji}\).
\end{solution}

\begin{exercise}[Cartesian symbols]\label{exr:vii22-ped-02}
Compute the Levi--Civita Christoffel symbols for the Euclidean metric in Cartesian coordinates.
\end{exercise}

\begin{hint}
All metric coefficients are constant.
\end{hint}

\begin{solution}
All Christoffel symbols vanish.
\end{solution}

\begin{exercise}[Polar radial]\label{exr:vii22-ped-03}
For the polar metric, compute \(\nabla_{\partial_\theta}\partial_\theta\).
\end{exercise}

\begin{hint}
Use \(\Gamma^r_{\theta\theta}=-r\).
\end{hint}

\begin{solution}
\(\nabla_{\partial_\theta}\partial_\theta=-r\partial_r\).
\end{solution}

\begin{exercise}[Polar mixed]\label{exr:vii22-ped-04}
Compute \(\nabla_{\partial_r}\partial_\theta\).
\end{exercise}

\begin{hint}
Use the mixed Christoffel symbol.
\end{hint}

\begin{solution}
\(\nabla_{\partial_r}\partial_\theta=(1/r)\partial_\theta\).
\end{solution}

\begin{exercise}[Constant diagonal metric]\label{exr:vii22-ped-05}
For constant \(g=4dx^2+9dy^2\), compute all Christoffel symbols.
\end{exercise}

\begin{hint}
Differentiate the coefficients.
\end{hint}

\begin{solution}
All vanish.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Uniqueness]\label{exr:vii22-ped-06}
Explain why metric compatibility plus zero torsion determine at most one connection.
\end{exercise}

\begin{hint}
Use the Koszul formula.
\end{hint}

\begin{solution}
The Koszul formula determines \(g(\nabla_XY,Z)\) for every \(Z\), and nondegeneracy determines \(\nabla_XY\).
\end{solution}

\begin{exercise}[Parallel norm]\label{exr:vii22-ped-07}
Prove a parallel field has constant norm for the Levi--Civita connection.
\end{exercise}

\begin{hint}
Differentiate \(g(V,V)\).
\end{hint}

\begin{solution}
Metric compatibility gives \(d g(V,V)/dt=2g(DV/dt,V)=0\).
\end{solution}

\begin{exercise}[Projection connection]\label{exr:vii22-ped-08}
Why is tangential projection of the ambient derivative metric compatible on an embedded Euclidean submanifold?
\end{exercise}

\begin{hint}
Differentiate the ambient inner product.
\end{hint}

\begin{solution}
The ambient derivative satisfies the product rule and normal components are orthogonal to tangent fields, yielding metric compatibility.
\end{solution}

\begin{exercise}[Koszul in commuting frame]\label{exr:vii22-ped-09}
Simplify the Koszul formula for coordinate vector fields.
\end{exercise}

\begin{hint}
All Lie brackets vanish.
\end{hint}

\begin{solution}
Only derivatives of metric coefficients remain, leading directly to the Christoffel formula.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Zero symbols imply flat]\label{exr:vii22-ped-10}
Test the claim: if Christoffel symbols vanish at one point, curvature vanishes there.
\end{exercise}

\begin{hint}
Use normal coordinates.
\end{hint}

\begin{solution}
False.  Normal coordinates make symbols vanish at a point on any Riemannian manifold; curvature survives in their derivatives.
\end{solution}

\begin{exercise}[Nonzero symbols imply curved]\label{exr:vii22-ped-11}
Test the claim: nonzero Christoffel symbols imply nonzero curvature.
\end{exercise}

\begin{hint}
Use Euclidean polar coordinates.
\end{hint}

\begin{solution}
False.  The Euclidean plane has nonzero polar Christoffel symbols but zero curvature.
\end{solution}

\begin{exercise}[Metric compatibility alone]\label{exr:vii22-ped-12}
Test the claim: metric compatibility alone determines the Levi--Civita connection.
\end{exercise}

\begin{hint}
Connections with torsion may still preserve the metric.
\end{hint}

\begin{solution}
False.  The torsion-free condition is also essential for uniqueness.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Robot on a sphere]\label{exr:vii22-ped-13}
Why is the Levi--Civita connection the natural rule for transporting an orientation frame along a spherical surface?
\end{exercise}

\begin{hint}
It preserves tangent inner products.
\end{hint}

\begin{solution}
It keeps transported frames tangent and orthonormal while introducing no torsion, so it respects the sphere's intrinsic metric.
\end{solution}

\begin{exercise}[Conformal optics]\label{exr:vii22-ped-14}
What changes under a conformal metric: angles or the Levi--Civita connection?
\end{exercise}

\begin{hint}
Separate angle preservation from metric derivatives.
\end{hint}

\begin{solution}
Angles are preserved, but the Levi--Civita connection generally changes because derivatives of the conformal factor enter its Christoffel symbols.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[One-dimensional connection]\label{exr:vii22-ped-15}
Show every one-dimensional metric \(g=E(x)dx^2\) has \(\Gamma^1_{11}=E'/(2E)\).
\end{exercise}

\begin{hint}
Apply the Christoffel formula.
\end{hint}

\begin{solution}
\(\Gamma^1_{11}=\tfrac12 E^{-1}(2E'-E')=E'/(2E)\).
\end{solution}

\begin{exercise}[Geodesic-ready data]\label{exr:vii22-ped-16}
Given a metric matrix numerically, what derivatives are needed to assemble the Levi--Civita connection?
\end{exercise}

\begin{hint}
Inspect the Christoffel formula.
\end{hint}

\begin{solution}
First spatial derivatives of the metric coefficients and the inverse metric are sufficient.
\end{solution}
