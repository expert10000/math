\chapter{Riemannian Metrics}
\label{ch:vii20-riemannian-metrics}

The first fundamental form of a surface is an inner product that varies smoothly
from tangent plane to tangent plane.  A Riemannian metric takes this as the
definition on an arbitrary smooth manifold:
\[
\boxed{g_p:T_pM\times T_pM\to\mathbb R.}
\]

\section{Definition and local coefficients}

\begin{definition}[Riemannian metric]\label{def:vii20-metric}
A Riemannian metric on a smooth manifold \(M\) is a smooth symmetric positive
definite \((0,2)\)-tensor field \(g\). Thus each \(g_p\) is an inner product on
\(T_pM\).
\end{definition}

In coordinates \(x^1,\dots,x^n\),
\[
g=g_{ij}\,dx^i\otimes dx^j,
\qquad
g_{ij}=g\!\left(\frac{\partial}{\partial x^i},
\frac{\partial}{\partial x^j}\right).
\]

\begin{proposition}[Positive-definite matrix]\label{prop:vii20-positive}
The matrix \((g_{ij})\) is symmetric and positive definite at every point, hence
invertible.  Its inverse is denoted \((g^{ij})\).
\end{proposition}

\section{Lengths and angles}

\begin{definition}[Norm and angle]\label{def:vii20-norm-angle}
For \(v,w\in T_pM\),
\[
|v|_g=\sqrt{g(v,v)}.
\]
For nonzero \(v,w\),
\[
\cos\theta=\frac{g(v,w)}{|v|_g|w|_g}.
\]
\end{definition}

\begin{definition}[Length of a curve]\label{def:vii20-curve-length}
For a piecewise smooth curve \(\gamma:[a,b]\to M\),
\[
L_g(\gamma)=\int_a^b |\dot\gamma(t)|_g\,dt.
\]
\end{definition}

\begin{proposition}[Reparametrization invariance]\label{prop:vii20-length-invariance}
Curve length is invariant under orientation-preserving \(C^1\) reparametrization.
\end{proposition}

\section{Riemannian distance}

\begin{definition}[Riemannian distance]\label{def:vii20-distance}
For points in the same connected component,
\[
d_g(p,q)=\inf_\gamma L_g(\gamma),
\]
where the infimum ranges over piecewise smooth curves from \(p\) to \(q\).
\end{definition}

\begin{theorem}[Metric-space property]\label{thm:vii20-distance-metric}
On each connected component, \(d_g\) is a genuine metric and induces the original
manifold topology.
\end{theorem}

\begin{remark}
The existence and local minimizing behavior of geodesics will be developed after
connections, in VII/23.  The distance can be defined before solving the geodesic
equation.
\end{remark}

\section{Pullback and induced metrics}

\begin{definition}[Pullback metric]\label{def:vii20-pullback}
If \(F:M\to N\) is smooth and \(h\) is a Riemannian metric on \(N\), define
\[
(F^*h)_p(u,v)=h_{F(p)}(dF_pu,dF_pv).
\]
If \(F\) is an immersion, \(F^*h\) is a Riemannian metric on \(M\).
\end{definition}

\begin{example}[Induced metric on a surface]\label{ex:vii20-induced-surface}
For an immersion \(X:U\to\mathbb R^3\), the pullback of the Euclidean metric is
\[
X^*\langle\cdot,\cdot\rangle
=
E\,du^2+2F\,du\,dv+G\,dv^2,
\]
the first fundamental form from VII/15.
\end{example}

\begin{definition}[Riemannian isometry]\label{def:vii20-isometry}
A diffeomorphism \(F:(M,g)\to(N,h)\) is an isometry if
\[
F^*h=g.
\]
\end{definition}

\begin{proposition}[Isometries preserve length]\label{prop:vii20-isometry-length}
Riemannian isometries preserve tangent-vector norms, curve lengths, angles, and
Riemannian distance.
\end{proposition}

\section{Product metrics}

\begin{definition}[Product metric]\label{def:vii20-product}
For Riemannian manifolds \((M,g)\) and \((N,h)\), the product metric is
\[
(g\oplus h)_{(p,q)}((u_1,v_1),(u_2,v_2))
=
g_p(u_1,u_2)+h_q(v_1,v_2).
\]
\end{definition}

\section{Musical isomorphisms}

\begin{definition}[Flat and sharp]\label{def:vii20-musical}
A metric identifies tangent and cotangent spaces:
\[
\flat:T_pM\to T_p^*M,\qquad
v^\flat=g(v,\cdot),
\]
and its inverse is
\[
\sharp:T_p^*M\to T_pM.
\]
\end{definition}

In coordinates,
\[
(v^\flat)_i=g_{ij}v^j,
\qquad
(\alpha^\sharp)^i=g^{ij}\alpha_j.
\]

\begin{definition}[Gradient]\label{def:vii20-gradient}
For \(f\in C^\infty(M)\),
\[
\operatorname{grad}f=(df)^\sharp,
\]
equivalently
\[
g(\operatorname{grad}f,X)=df(X)
\]
for every vector field \(X\).
\end{definition}

\begin{proposition}[Coordinate gradient]\label{prop:vii20-gradient-coord}
\[
\operatorname{grad}f
=
g^{ij}\frac{\partial f}{\partial x^j}\frac{\partial}{\partial x^i}.
\]
\end{proposition}

\section{Riemannian volume density}

\begin{definition}[Riemannian density]\label{def:vii20-density}
In a coordinate chart, the canonical positive density is
\[
dV_g=\sqrt{\det(g_{ij})}\,|dx^1\cdots dx^n|.
\]
On an oriented Riemannian manifold this corresponds to the Riemannian volume form.
\end{definition}

\begin{proposition}[Coordinate invariance]\label{prop:vii20-density-invariance}
The factor \(\sqrt{\det g}\) transforms with the inverse absolute Jacobian, so
\(dV_g\) is globally well defined.
\end{proposition}

\section{Conformal metrics}

\begin{definition}[Conformal change]\label{def:vii20-conformal}
If \(\varphi\in C^\infty(M)\), then
\[
\widetilde g=e^{2\varphi}g
\]
is a conformally equivalent metric.
\end{definition}

\begin{proposition}[Effect on lengths and volume]\label{prop:vii20-conformal-scaling}
Under \(\widetilde g=e^{2\varphi}g\),
\[
|v|_{\widetilde g}=e^\varphi|v|_g,
\qquad
dV_{\widetilde g}=e^{n\varphi}dV_g.
\]
Angles are unchanged.
\end{proposition}

\section{Existence of metrics}

\begin{theorem}[Existence]\label{thm:vii20-existence}
Every smooth paracompact manifold admits a Riemannian metric.
\end{theorem}

\begin{proof}
Choose a locally finite atlas, put the Euclidean inner product on each chart, and
combine the resulting local metrics using a smooth partition of unity. Positive
weighted sums of positive-definite bilinear forms remain positive definite.
\end{proof}

\section{Examples}

\begin{example}[Euclidean metric]\label{ex:vii20-euclidean}
On \(\mathbb R^n\),
\[
g=\sum_i dx^i\otimes dx^i.
\]
\end{example}

\begin{example}[Round sphere]\label{ex:vii20-sphere}
The round metric on \(S^n\) is the metric induced by the Euclidean metric of
\(\mathbb R^{n+1}\).
\end{example}

\begin{example}[Polar coordinates]\label{ex:vii20-polar}
On the Euclidean plane away from the origin,
\[
g=dr^2+r^2d\theta^2.
\]
The nonconstant coefficients reflect coordinates, not intrinsic curvature.
\end{example}

\section{Solved problem dossiers}

\begin{problem}[Legacy metric problem]\label{prob:vii20-01}
Show a smooth positive-definite matrix field \(g_{ij}\) defines a Riemannian metric in a coordinate chart.
\end{problem}
\begin{solution}
Set \(g(v,w)=g_{ij}v^iw^j\). Symmetry and positive definiteness are pointwise, while smoothness follows from the coefficient functions.
\end{solution}

\begin{problem}[Coordinate-change problem]\label{prob:vii20-02}
Derive the transformation law for metric coefficients.
\end{problem}
\begin{solution}
If \(x=x(y)\), substitute \(\partial/\partial y^a=(\partial x^i/\partial y^a)\partial_i\). Then \(\widetilde g_{ab}=g_{ij}(\partial x^i/\partial y^a)(\partial x^j/\partial y^b)\).
\end{solution}

\begin{problem}[Polar-coordinate problem]\label{prob:vii20-03}
Derive \(dr^2+r^2d\theta^2\) from the Cartesian Euclidean metric.
\end{problem}
\begin{solution}
Substitute \(dx=\cos\theta\,dr-r\sin\theta\,d\theta\) and \(dy=\sin\theta\,dr+r\cos\theta\,d\theta\), then simplify.
\end{solution}

\begin{problem}[Induced-metric problem]\label{prob:vii20-04}
Show an immersion pulls an ambient metric back to a Riemannian metric.
\end{problem}
\begin{solution}
Smoothness and symmetry are immediate. Positive definiteness follows from injectivity of the differential.
\end{solution}

\begin{problem}[Length-invariance problem]\label{prob:vii20-05}
Prove curve length is parametrization invariant.
\end{problem}
\begin{solution}
Under \(t=\phi(s)\), \((\gamma\circ\phi)'=\dot\gamma(\phi)\phi'\); integrate the resulting speed and substitute variables.
\end{solution}

\begin{problem}[Distance problem]\label{prob:vii20-06}
Explain why Riemannian distance satisfies the triangle inequality.
\end{problem}
\begin{solution}
Concatenate a nearly minimizing curve from \(p\) to \(q\) with one from \(q\) to \(r\), then take infima.
\end{solution}

\begin{problem}[Isometry problem]\label{prob:vii20-07}
Show Riemannian isometries preserve distance.
\end{problem}
\begin{solution}
They preserve every curve length, hence preserve the infimum of lengths over joining curves.
\end{solution}

\begin{problem}[Musical-isomorphism problem]\label{prob:vii20-08}
Derive the coordinate formulas for \(\flat\) and \(\sharp\).
\end{problem}
\begin{solution}
Lowering an index gives \(\alpha_i=g_{ij}v^j\); multiplying by \(g^{ki}\) recovers \(v^k=g^{ki}\alpha_i\).
\end{solution}

\begin{problem}[Gradient problem]\label{prob:vii20-09}
Derive the coordinate expression for \(\operatorname{grad}f\).
\end{problem}
\begin{solution}
Write \(g_{ij}(\operatorname{grad}f)^i=\partial_jf\) and multiply by \(g^{kj}\).
\end{solution}

\begin{problem}[Volume-density problem]\label{prob:vii20-10}
Explain why \(\sqrt{\det g}\,|dx^1\cdots dx^n|\) is coordinate invariant.
\end{problem}
\begin{solution}
The metric matrix transforms by \(J^TgJ\), so its determinant gains \((\det J)^2\); the coordinate density contributes the inverse absolute Jacobian.
\end{solution}

\begin{problem}[Conformal problem]\label{prob:vii20-11}
Compute the effect of \(\widetilde g=e^{2\varphi}g\) on volume.
\end{problem}
\begin{solution}
In dimension \(n\), the determinant scales by \(e^{2n\varphi}\), so its square root scales by \(e^{n\varphi}\).
\end{solution}

\begin{problem}[Existence problem]\label{prob:vii20-12}
Construct a global Riemannian metric from local Euclidean metrics.
\end{problem}
\begin{solution}
Choose a locally finite partition of unity subordinate to an atlas and form the positive weighted sum of the local metrics.
\end{solution}

\section{Challenge problems}

\begin{problem}[Topology from distance]\label{prob:vii20-ch01}
Why does \(d_g\) induce the original manifold topology?
\end{problem}
\begin{solution}
On a relatively compact coordinate ball, the eigenvalues of \(g_{ij}\) are bounded above and below by positive constants. Hence Riemannian length is locally comparable to Euclidean length, making the two local topologies identical.
\end{solution}

\begin{problem}[Metric completeness preview]\label{prob:vii20-ch02}
Distinguish metric completeness from compactness and explain why completeness will matter for geodesics.
\end{problem}
\begin{solution}
Metric completeness means every \(d_g\)-Cauchy sequence converges. Compact Riemannian manifolds are complete, but noncompact ones may also be complete. Later Hopf--Rinow theory relates metric completeness to extendability of geodesics.
\end{solution}

\begin{problem}[Conformal inverse metric]\label{prob:vii20-ch03}
If \(\widetilde g=e^{2\varphi}g\), compute \(\widetilde g^{ij}\) and \(\operatorname{grad}_{\widetilde g}f\).
\end{problem}
\begin{solution}
The inverse metric scales by \(e^{-2\varphi}\), hence \(\operatorname{grad}_{\widetilde g}f=e^{-2\varphi}\operatorname{grad}_g f\).
\end{solution}

\begin{problem}[Product distance bounds]\label{prob:vii20-ch04}
For the product metric, compare \(d_{g\oplus h}\) with the factor distances.
\end{problem}
\begin{solution}
Any joining curve has length at least each projected length, so the product distance dominates both factor distances. A product of nearly minimizing constant-speed curves gives the sharper Euclidean-combination behavior.
\end{solution}

\begin{problem}[Metrics as bundle sections]\label{prob:vii20-ch05}
Reinterpret a Riemannian metric in the language of VII/07.
\end{problem}
\begin{solution}
A metric is a smooth section of \(\operatorname{Sym}^2T^*M\) whose value lies in the open cone of positive-definite bilinear forms at every point.
\end{solution}
