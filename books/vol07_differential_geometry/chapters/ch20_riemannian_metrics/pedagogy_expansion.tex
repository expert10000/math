\section{Supplementary worked examples}
\label{sec:vii20-pedagogy}

\begin{example}[Gradient in polar coordinates]\label{ex:vii20-ped-01}
For the Euclidean plane in polar coordinates, \(g=dr^2+r^2d\theta^2\).  If \(f(r,\theta)=r^2\), then \(df=2r\,dr\) and \(\operatorname{grad}f=2r\,\partial_r\).  The computation shows how the inverse metric, not the coordinate appearance alone, raises the index of \(df\).
\end{example}

\begin{example}[Conformal scaling]\label{ex:vii20-ped-02}
If \(\widetilde g=e^{2\varphi}g\) on an \(n\)-manifold, then tangent lengths scale by \(e^\varphi\) and volume by \(e^{n\varphi}\).  In dimension two, area therefore scales by \(e^{2\varphi}\) while angles remain unchanged.
\end{example}

\begin{example}[Product cylinder]\label{ex:vii20-ped-03}
On \(S^1\times\mathbb R\) with product metric \(d\theta^2+dz^2\), the coordinate fields are orthonormal.  A curve \((\theta(t),z(t))\) has speed \(\sqrt{\dot\theta^2+\dot z^2}\), so the intrinsic geometry is locally Euclidean even though the usual embedding as a cylinder is curved extrinsically.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Metric matrix]\label{exr:vii20-ped-01}
For \(g=4\,dx^2+dy^2\), compute the norm of \(v=\partial_x+2\partial_y\).
\end{exercise}

\begin{hint}
Evaluate \(g(v,v)\).
\end{hint}

\begin{solution}
\(g(v,v)=4+4=8\), so \(|v|_g=2\sqrt2\).
\end{solution}

\begin{exercise}[Angle]\label{exr:vii20-ped-02}
For the same metric, compute the angle between \(\partial_x\) and \(\partial_y\).
\end{exercise}

\begin{hint}
Compute their metric inner product.
\end{hint}

\begin{solution}
They are orthogonal, so the angle is \(\pi/2\).
\end{solution}

\begin{exercise}[Curve length]\label{exr:vii20-ped-03}
Compute the length of \(\gamma(t)=(t,0)\), \(0\le t\le1\), for \(g=4dx^2+dy^2\).
\end{exercise}

\begin{hint}
Compute the constant speed.
\end{hint}

\begin{solution}
The speed is \(2\), hence the length is \(2\).
\end{solution}

\begin{exercise}[Inverse metric]\label{exr:vii20-ped-04}
Find \((g^{ij})\) for \(g=4dx^2+dy^2\).
\end{exercise}

\begin{hint}
Invert the diagonal matrix.
\end{hint}

\begin{solution}
\((g^{ij})=\operatorname{diag}(1/4,1)\).
\end{solution}

\begin{exercise}[Volume density]\label{exr:vii20-ped-05}
Find \(dV_g\) for \(g=4dx^2+dy^2\).
\end{exercise}

\begin{hint}
Use \(\sqrt{\det g}\).
\end{hint}

\begin{solution}
\(dV_g=2\,|dx\,dy|\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Isometries preserve distance]\label{exr:vii20-ped-06}
Prove a Riemannian isometry preserves the distance function.
\end{exercise}

\begin{hint}
First prove it preserves the length of every curve.
\end{hint}

\begin{solution}
Pullback equality preserves speeds and hence curve lengths; taking infima over joining curves gives equality of distances.
\end{solution}

\begin{exercise}[Product positivity]\label{exr:vii20-ped-07}
Prove \(g\oplus h\) is positive definite.
\end{exercise}

\begin{hint}
Evaluate on a nonzero pair \((u,v)\).
\end{hint}

\begin{solution}
\((g\oplus h)((u,v),(u,v))=g(u,u)+h(v,v)>0\) unless both vectors vanish.
\end{solution}

\begin{exercise}[Gradient characterization]\label{exr:vii20-ped-08}
Prove the gradient is uniquely determined by \(g(\operatorname{grad}f,X)=df(X)\).
\end{exercise}

\begin{hint}
Use nondegeneracy of the metric.
\end{hint}

\begin{solution}
If two vectors satisfy the identity, their difference pairs to zero with every tangent vector and is therefore zero.
\end{solution}

\begin{exercise}[Conformal angles]\label{exr:vii20-ped-09}
Prove \(e^{2\varphi}g\) has the same angles as \(g\).
\end{exercise}

\begin{hint}
Track the common scale factor in numerator and norms.
\end{hint}

\begin{solution}
The numerator gains \(e^{2\varphi}\) and the two norms together gain the same factor, so the cosine is unchanged.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Coordinate coefficients imply curvature]\label{exr:vii20-ped-10}
Test the claim: nonconstant metric coefficients imply nonzero intrinsic curvature.
\end{exercise}

\begin{hint}
Use Euclidean polar coordinates.
\end{hint}

\begin{solution}
False.  \(dr^2+r^2d\theta^2\) has nonconstant coefficients but is the flat Euclidean metric.
\end{solution}

\begin{exercise}[Immersion hypothesis]\label{exr:vii20-ped-11}
Test the claim: the pullback of a Riemannian metric by any smooth map is Riemannian.
\end{exercise}

\begin{hint}
Consider a constant map.
\end{hint}

\begin{solution}
False.  A constant map has zero differential, so the pullback form is degenerate.
\end{solution}

\begin{exercise}[Distance before geodesics]\label{exr:vii20-ped-12}
Test the claim: Riemannian distance cannot be defined until geodesics are known.
\end{exercise}

\begin{hint}
Use the infimum-of-lengths definition.
\end{hint}

\begin{solution}
False.  Distance is defined from lengths of all piecewise smooth joining curves; geodesics are then used to analyze minimizers.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Anisotropic travel]\label{exr:vii20-ped-13}
Interpret the metric \(g=9dx^2+dy^2\) as a cost model for motion.
\end{exercise}

\begin{hint}
Compare unit coordinate displacements.
\end{hint}

\begin{solution}
A unit displacement in the \(x\)-direction has metric length \(3\), while one in the \(y\)-direction has length \(1\); the metric encodes direction-dependent cost.
\end{solution}

\begin{exercise}[Conformal area]\label{exr:vii20-ped-14}
On a surface, if \(\widetilde g=4g\), by what factor does area scale?
\end{exercise}

\begin{hint}
Write \(4=e^{2\varphi}\).
\end{hint}

\begin{solution}
Lengths scale by \(2\), so areas scale by \(4\).
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Metric reconstruction]\label{exr:vii20-ped-15}
Suppose a coordinate basis satisfies \(|\partial_x|=2\), \(|\partial_y|=3\), and the angle between them is \(\pi/3\).  Write the metric matrix.
\end{exercise}

\begin{hint}
Use \(g_{12}=|e_1||e_2|\cos\theta\).
\end{hint}

\begin{solution}
The matrix is \(\begin{pmatrix}4&3\\3&9\end{pmatrix}\), which is positive definite since its determinant is \(27\).
\end{solution}

\begin{exercise}[Conformal gradient]\label{exr:vii20-ped-16}
If \(\widetilde g=e^{2\varphi}g\), express \(\operatorname{grad}_{\widetilde g}f\) in terms of \(\operatorname{grad}_g f\).
\end{exercise}

\begin{hint}
Use the defining pairing for the gradient.
\end{hint}

\begin{solution}
\(\operatorname{grad}_{\widetilde g}f=e^{-2\varphi}\operatorname{grad}_g f\).
\end{solution}
