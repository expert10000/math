\chapter{Ricci and Scalar Curvature}
\label{ch:vii27-ricci-scalar}

The Riemann tensor contains all local curvature information, but it has many
components.  Two canonical contractions summarize important averages of sectional
curvature:
\[
\boxed{\operatorname{Ric}\qquad\text{and}\qquad \operatorname{Scal}.}
\]

\section{Ricci curvature}

\begin{definition}[Ricci tensor]\label{def:vii27-ricci}
For \(X,Y\in T_pM\), define
\[
\operatorname{Ric}(X,Y)
=
\operatorname{tr}\bigl(Z\mapsto R(Z,X)Y\bigr).
\]
If \(e_1,\dots,e_n\) is an orthonormal basis,
\[
\operatorname{Ric}(X,Y)
=
\sum_{i=1}^n g(R(e_i,X)Y,e_i).
\]
\end{definition}

\begin{proposition}[Symmetry]\label{prop:vii27-ricci-symmetric}
For the Levi--Civita connection,
\[
\operatorname{Ric}(X,Y)=\operatorname{Ric}(Y,X).
\]
\end{proposition}

\begin{proof}
Use the pair symmetry and first Bianchi identity of the Riemann tensor, then sum
over an orthonormal basis.
\end{proof}

\section{Ricci curvature as an average}

Let \(v\) be a unit tangent vector and complete it to an orthonormal basis
\[
v,e_2,\dots,e_n.
\]

\begin{proposition}[Sum of sectional curvatures]\label{prop:vii27-ricci-sectional}
\[
\operatorname{Ric}(v,v)
=
\sum_{i=2}^n K(\operatorname{span}\{v,e_i\}).
\]
\end{proposition}

Thus Ricci curvature in direction \(v\) is the sum of sectional curvatures of all
orthogonal two-planes containing \(v\).

\section{Scalar curvature}

\begin{definition}[Scalar curvature]\label{def:vii27-scalar}
The scalar curvature is the metric trace of the Ricci tensor:
\[
\operatorname{Scal}
=
\operatorname{tr}_g\operatorname{Ric}.
\]
For an orthonormal basis,
\[
\operatorname{Scal}
=
\sum_{i=1}^n \operatorname{Ric}(e_i,e_i).
\]
\end{definition}

\begin{proposition}[Sum over sectional curvatures]\label{prop:vii27-scalar-sectional}
For an orthonormal basis \(e_1,\dots,e_n\),
\[
\operatorname{Scal}
=
2\sum_{1\le i<j\le n}K(e_i,e_j).
\]
\end{proposition}

\section{Coordinate contractions}

With curvature components consistent with VII/26,
\[
R^\ell{}_{kij},
\]
the Ricci components are obtained by contraction:
\[
\operatorname{Ric}_{kj}
=
R^i{}_{kij}.
\]
The scalar curvature is
\[
\operatorname{Scal}
=
g^{kj}\operatorname{Ric}_{kj}.
\]

\begin{warning}[Component conventions]\label{warn:vii27-components}
If a source uses the opposite sign for the Riemann tensor, both Ricci and scalar
curvature change sign.  Geometric comparisons must therefore begin with the
underlying Riemann convention.
\end{warning}

\section{Constant sectional curvature}

\begin{theorem}[Space-form contractions]\label{thm:vii27-space-form}
If an \(n\)-dimensional manifold has constant sectional curvature \(c\), then
\[
\operatorname{Ric}
=
(n-1)c\,g,
\]
and
\[
\operatorname{Scal}
=
n(n-1)c.
\]
\end{theorem}

\begin{proof}
For a unit vector \(v\), there are \(n-1\) orthogonal two-planes
\(\operatorname{span}\{v,e_i\}\), each of curvature \(c\).  Hence
\(\operatorname{Ric}(v,v)=(n-1)c\). Polarization gives the Ricci formula, and one
more trace gives the scalar formula.
\end{proof}

\begin{example}[Round sphere]\label{ex:vii27-sphere}
For the round sphere \(S^n(a)\),
\[
\operatorname{Ric}
=
\frac{n-1}{a^2}g,
\qquad
\operatorname{Scal}
=
\frac{n(n-1)}{a^2}.
\]
\end{example}

\begin{example}[Euclidean space]\label{ex:vii27-euclidean}
For Euclidean space,
\[
\operatorname{Ric}=0,
\qquad
\operatorname{Scal}=0.
\]
\end{example}

\section{Einstein metrics}

\begin{definition}[Einstein metric]\label{def:vii27-einstein}
A Riemannian metric is Einstein if
\[
\operatorname{Ric}=\lambda g
\]
for some function \(\lambda\).
\end{definition}

\begin{proposition}[Constancy in dimension at least three]\label{prop:vii27-einstein-lambda}
On a connected Einstein manifold of dimension \(n\ge3\), the factor \(\lambda\) is
constant.
\end{proposition}

\begin{proof}
The contracted Bianchi identity gives
\[
\operatorname{div}\operatorname{Ric}
=
\frac12\,d\operatorname{Scal}.
\]
If \(\operatorname{Ric}=\lambda g\), then
\[
d\lambda
=
\frac n2\,d\lambda.
\]
For \(n\ne2\), this forces \(d\lambda=0\).
\end{proof}

\section{The Einstein tensor}

\begin{definition}[Einstein tensor]\label{def:vii27-einstein-tensor}
Define
\[
G
=
\operatorname{Ric}
-\frac12\operatorname{Scal}\,g.
\]
\end{definition}

\begin{theorem}[Divergence-free identity]\label{thm:vii27-divfree}
The contracted second Bianchi identity implies
\[
\operatorname{div}G=0.
\]
\end{theorem}

\begin{remark}
This identity is purely Riemannian and differential-geometric.  Its Lorentzian
role in gravitation belongs later, after indefinite metrics are introduced in
VII/29--VII/30.
\end{remark}

\section{Low dimensions}

\begin{proposition}[Dimension two]\label{prop:vii27-dim2}
On a Riemannian surface,
\[
\operatorname{Ric}=K\,g,
\qquad
\operatorname{Scal}=2K.
\]
\end{proposition}

\begin{proposition}[Dimension three preview]\label{prop:vii27-dim3}
In dimension three the Riemann tensor is completely determined by the Ricci tensor
and scalar curvature.  Equivalently, the Weyl tensor introduced in VII/28 vanishes
identically.
\end{proposition}

\section{Geometric meaning}

Sectional curvature resolves curvature plane by plane.  Ricci curvature averages
those planes through one direction.  Scalar curvature averages over all orthogonal
two-planes at a point.  These successive contractions trade directional detail for
compact geometric information.

\section{Solved problem dossiers}
\begin{problem}[Ricci-contraction problem]\label{prob:vii27-01}
Derive the orthonormal-basis expression for the Ricci tensor from the trace definition.
\end{problem}
\begin{solution}
The trace of \(Z\mapsto R(Z,X)Y\) is the sum of its diagonal matrix entries \(g(R(e_i,X)Y,e_i)\).
\end{solution}

\begin{problem}[Symmetry problem]\label{prob:vii27-02}
Prove the Ricci tensor is symmetric.
\end{problem}
\begin{solution}
Use the Riemann pair symmetry and first Bianchi identity to rewrite the two traces and show their difference vanishes.
\end{solution}

\begin{problem}[Sectional-average problem]\label{prob:vii27-03}
Prove \(\operatorname{Ric}(v,v)=\sum_{i=2}^nK(v,e_i)\) for unit \(v\).
\end{problem}
\begin{solution}
Choose an orthonormal basis beginning with \(v\). The \(i=1\) term vanishes by skew symmetry; the remaining terms are exactly sectional curvatures.
\end{solution}

\begin{problem}[Scalar-average problem]\label{prob:vii27-04}
Derive \(\operatorname{Scal}=2\sum_{i<j}K(e_i,e_j)\).
\end{problem}
\begin{solution}
Expand the Ricci trace as a double sum. Terms with \(i=j\) vanish, and every unordered pair occurs twice.
\end{solution}

\begin{problem}[Space-form problem]\label{prob:vii27-05}
Derive Ricci and scalar curvature for constant sectional curvature.
\end{problem}
\begin{solution}
Apply the sectional-average formula and then take the metric trace.
\end{solution}

\begin{problem}[Sphere problem]\label{prob:vii27-06}
Compute the contractions for \(S^n(a)\).
\end{problem}
\begin{solution}
Insert \(c=1/a^2\) into the space-form formulas.
\end{solution}

\begin{problem}[Surface problem]\label{prob:vii27-07}
Show \(\operatorname{Ric}=Kg\) and \(\operatorname{Scal}=2K\) in dimension two.
\end{problem}
\begin{solution}
For any unit \(v\), the single orthogonal plane has curvature \(K\), hence \(\operatorname{Ric}(v,v)=K\); polarization gives the tensor identity.
\end{solution}

\begin{problem}[Einstein problem]\label{prob:vii27-08}
Show a constant-sectional-curvature metric is Einstein.
\end{problem}
\begin{solution}
Space forms satisfy \(\operatorname{Ric}=(n-1)c\,g\), which is exactly the Einstein condition.
\end{solution}

\begin{problem}[Contracted-Bianchi problem]\label{prob:vii27-09}
State the contracted Bianchi identity and derive divergence freeness of the Einstein tensor.
\end{problem}
\begin{solution}
The identity is \(\operatorname{div}\operatorname{Ric}=\frac12d\operatorname{Scal}\). Since \(\nabla g=0\), subtracting \(\frac12\operatorname{Scal}g\) yields zero divergence.
\end{solution}

\begin{problem}[Einstein-factor problem]\label{prob:vii27-10}
Prove the proportionality factor of an Einstein metric is constant when \(n\ge3\).
\end{problem}
\begin{solution}
Substitute \(\operatorname{Ric}=\lambda g\) and \(\operatorname{Scal}=n\lambda\) into contracted Bianchi to obtain \((1-n/2)d\lambda=0\).
\end{solution}

\begin{problem}[Dimension-three problem]\label{prob:vii27-11}
Explain why Ricci contains all Riemann curvature information in dimension three.
\end{problem}
\begin{solution}
The trace-free Weyl component is identically zero in dimension three, leaving only the algebraic part constructed from Ricci, scalar curvature, and the metric.
\end{solution}

\begin{problem}[Hierarchy problem]\label{prob:vii27-12}
Compare sectional, Ricci, and scalar curvature.
\end{problem}
\begin{solution}
Sectional curvature is plane-resolved; Ricci sums planes containing a chosen direction; scalar curvature traces once more and averages all orthogonal two-planes.
\end{solution}

\section{Challenge problems}
\begin{problem}[Scaling law]\label{prob:vii27-ch01}
If \(\widetilde g=c^2g\) for constant \(c>0\), determine the scaling of sectional, Ricci-as-a-\((0,2)\)-tensor, and scalar curvature.
\end{problem}
\begin{solution}
The Levi--Civita connection is unchanged. Sectional curvature scales as \(c^{-2}\); the \((0,2)\) Ricci tensor is unchanged under constant scaling; scalar curvature scales as \(c^{-2}\) because the inverse metric contributes \(c^{-2}\).
\end{solution}

\begin{problem}[Product manifolds]\label{prob:vii27-ch02}
Describe Ricci curvature of a Riemannian product \(M\times N\).
\end{problem}
\begin{solution}
The Levi--Civita connection splits and mixed sectional curvatures vanish. Ricci is the direct sum \(\operatorname{Ric}_{M\times N}=\operatorname{Ric}_M\oplus\operatorname{Ric}_N\), and scalar curvatures add.
\end{solution}

\begin{problem}[Einstein product criterion]\label{prob:vii27-ch03}
When is a product of two Einstein manifolds Einstein under the unscaled product metric?
\end{problem}
\begin{solution}
If \(\operatorname{Ric}_M=\lambda_Mg_M\) and \(\operatorname{Ric}_N=\lambda_Ng_N\), the product is Einstein exactly when \(\lambda_M=\lambda_N\).
\end{solution}

\begin{problem}[Ricci versus sectional sign]\label{prob:vii27-ch04}
Does positive Ricci curvature force every sectional curvature to be positive?
\end{problem}
\begin{solution}
No in dimensions at least three. Ricci is only a sum of sectional curvatures through a direction, so some sectional curvatures can be negative while the total sum remains positive.
\end{solution}

\begin{problem}[Scalar curvature locality]\label{prob:vii27-ch05}
Explain why scalar curvature is a much coarser invariant than the full Riemann tensor.
\end{problem}
\begin{solution}
It is only a double contraction. Many distinct algebraic curvature tensors have the same scalar trace; Weyl curvature in particular is completely invisible to scalar curvature.
\end{solution}

\section*{Bridge to VII/28}
VII/28 subtracts precisely these Ricci and scalar traces from the Riemann tensor, leaving the trace-free Weyl curvature.
