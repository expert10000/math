\chapter{Weyl Curvature}
\label{ch:vii28-weyl}

Ricci and scalar curvature are traces of the Riemann tensor.  What remains after
all trace information is removed is the Weyl tensor.  In dimensions \(n\ge4\) it
captures the conformally invariant part of Riemannian curvature.

\section{Kulkarni--Nomizu product}

\begin{definition}[Kulkarni--Nomizu product]\label{def:vii28-kn}
For symmetric \((0,2)\)-tensors \(h,k\), define
\[
\begin{aligned}
(h\owedge k)(X,Y,Z,W)
={}&
h(X,Z)k(Y,W)+h(Y,W)k(X,Z)\\
&-h(X,W)k(Y,Z)-h(Y,Z)k(X,W).
\end{aligned}
\]
\end{definition}

This product produces a \((0,4)\)-tensor with the algebraic symmetries of a
curvature tensor.

\section{Schouten tensor}

\begin{definition}[Schouten tensor]\label{def:vii28-schouten}
For dimension \(n\ge3\), define
\[
P
=
\frac1{n-2}
\left(
\operatorname{Ric}
-
\frac{\operatorname{Scal}}{2(n-1)}g
\right).
\]
\end{definition}

\section{Weyl decomposition}

\begin{definition}[Weyl tensor]\label{def:vii28-weyl}
For \(n\ge3\), the Weyl tensor is
\[
W
=
\operatorname{Rm}-g\owedge P.
\]
\end{definition}

Equivalently,
\[
\boxed{
W
=
\operatorname{Rm}
-
\frac1{n-2}\,g\owedge\operatorname{Ric}
+
\frac{\operatorname{Scal}}{2(n-1)(n-2)}\,g\owedge g.
}
\]

\begin{theorem}[Trace-free property]\label{thm:vii28-tracefree}
Every metric contraction of the Weyl tensor vanishes.
\end{theorem}

\begin{remark}
Thus the Riemann tensor splits into a trace-free part \(W\) and a part determined
entirely by \(\operatorname{Ric}\), \(\operatorname{Scal}\), and \(g\).
\end{remark}

\section{Low dimensions}

\begin{theorem}[Dimensions two and three]\label{thm:vii28-lowdim}
In dimension two the Riemann tensor is determined by Gaussian curvature.  In
dimension three,
\[
W\equiv0,
\]
so the Riemann tensor is completely determined by Ricci curvature.
\end{theorem}

\begin{warning}[Weyl begins in dimension four]\label{warn:vii28-dim}
The Weyl tensor carries independent local curvature information only in dimensions
\(n\ge4\).
\end{warning}

\section{Constant curvature and Einstein metrics}

\begin{proposition}[Space forms are conformally flat]\label{prop:vii28-spaceform}
Every constant-sectional-curvature metric has
\[
W=0.
\]
\end{proposition}

\begin{proof}
For constant sectional curvature \(c\),
\[
\operatorname{Rm}
=
\frac c2\,g\owedge g
\]
with the Kulkarni--Nomizu convention above.  Substituting
\(\operatorname{Ric}=(n-1)cg\) and
\(\operatorname{Scal}=n(n-1)c\) into the Weyl decomposition cancels all terms.
\end{proof}

\begin{proposition}[Einstein simplification]\label{prop:vii28-einstein}
If
\[
\operatorname{Ric}=\lambda g,
\]
then
\[
\operatorname{Rm}
=
W
+
\frac{\lambda}{2(n-1)}\,g\owedge g.
\]
\end{proposition}

Thus on an Einstein manifold, Weyl curvature measures exactly the difference from
constant sectional curvature at the algebraic level.

\section{Conformal change}

Let
\[
\widetilde g=e^{2\varphi}g.
\]

\begin{theorem}[Conformal covariance]\label{thm:vii28-conformal}
For \(n\ge4\), the Weyl tensor is conformally invariant as a \((1,3)\)-tensor.
As a fully covariant \((0,4)\)-tensor,
\[
\widetilde W=e^{2\varphi}W.
\]
Consequently,
\[
W=0
\]
is a conformally invariant condition.
\end{theorem}

\begin{remark}
Riemann, Ricci, and scalar curvature all acquire derivative terms under a general
conformal change.  The specific trace subtraction in \(W\) cancels those
non-invariant pieces.
\end{remark}

\section{Local conformal flatness}

\begin{definition}[Locally conformally flat]\label{def:vii28-lcf}
A Riemannian metric is locally conformally flat if every point has a neighborhood
on which
\[
g=e^{2\varphi}g_{\mathrm{flat}}
\]
in suitable local coordinates.
\end{definition}

\begin{theorem}[Weyl criterion in dimension at least four]\label{thm:vii28-lcf-criterion}
For \(n\ge4\),
\[
W=0
\]
if and only if the metric is locally conformally flat.
\end{theorem}

\begin{warning}[Dimension three]\label{warn:vii28-cotton}
In dimension three \(W\) vanishes for every metric, so it cannot detect local
conformal flatness.  The relevant obstruction is the Cotton tensor.
\end{warning}

\section{Examples}

\begin{example}[Euclidean space]\label{ex:vii28-euclidean}
Euclidean space has \(\operatorname{Rm}=0\), hence \(W=0\).
\end{example}

\begin{example}[Round sphere]\label{ex:vii28-sphere}
The round sphere has constant sectional curvature, so \(W=0\).
Stereographic projection also exhibits its metric as conformal to the Euclidean
metric.
\end{example}

\begin{example}[Dimension three]\label{ex:vii28-three}
Every three-dimensional Riemannian metric has \(W=0\), even though it may have
nonzero Riemann, Ricci, and scalar curvature.
\end{example}

\section{Weyl curvature as the trace-free remainder}

At a point in dimension \(n\ge4\), the decomposition
\[
\operatorname{Rm}
=
W+g\owedge P
\]
separates curvature into:
\[
\text{trace-free conformal curvature}
\quad+\quad
\text{Ricci-controlled curvature}.
\]
This decomposition is the natural endpoint of the Riemannian curvature contraction
arc begun in VII/26.

\section{Transition beyond positive-definite metrics}

Everything so far has used positive-definite Riemannian metrics.  The algebra of
connections and curvature extends much further.  VII/29 replaces positive
definiteness by a nondegenerate bilinear form of arbitrary signature and develops
indefinite metrics.

\section{Solved problem dossiers}
\begin{problem}[Decomposition problem]\label{prob:vii28-01}
Derive the Weyl decomposition from the requirement that \(W\) have the Riemann symmetries and zero traces.
\end{problem}
\begin{solution}
Make the ansatz \(W=\operatorname{Rm}+a\,g\owedge\operatorname{Ric}+b\,\operatorname{Scal}\,g\owedge g\), contract, and solve for \(a=-1/(n-2)\) and \(b=1/[2(n-1)(n-2)]\).
\end{solution}

\begin{problem}[Trace-free problem]\label{prob:vii28-02}
Verify one Ricci-type contraction of the Weyl tensor vanishes.
\end{problem}
\begin{solution}
Contract the decomposition with \(g^{ik}\), use the contractions of \(g\owedge\operatorname{Ric}\) and \(g\owedge g\), and observe exact cancellation.
\end{solution}

\begin{problem}[Dimension-three problem]\label{prob:vii28-03}
Explain algebraically why \(W\equiv0\) in dimension three.
\end{problem}
\begin{solution}
The dimension of the space of algebraic Riemann tensors equals the dimension carried by the Ricci tensor in dimension three; the trace-free remainder has zero dimension.
\end{solution}

\begin{problem}[Space-form problem]\label{prob:vii28-04}
Show constant-sectional-curvature tensors have zero Weyl part.
\end{problem}
\begin{solution}
Insert \(\operatorname{Ric}=(n-1)cg\), \(\operatorname{Scal}=n(n-1)c\), and \(\operatorname{Rm}=(c/2)g\owedge g\) into the decomposition.
\end{solution}

\begin{problem}[Einstein problem]\label{prob:vii28-05}
Derive \(\operatorname{Rm}=W+\frac{\lambda}{2(n-1)}g\owedge g\) for an Einstein metric.
\end{problem}
\begin{solution}
Substitute \(\operatorname{Ric}=\lambda g\) and \(\operatorname{Scal}=n\lambda\) into the general decomposition and simplify.
\end{solution}

\begin{problem}[Conformal problem]\label{prob:vii28-06}
Explain why Weyl is the natural conformal curvature tensor.
\end{problem}
\begin{solution}
Under \(g\mapsto e^{2\varphi}g\), derivative terms entering Riemann curvature are all contained in the Schouten part \(g\owedge P\); the trace-free remainder is conformally invariant as a \((1,3)\)-tensor.
\end{solution}

\begin{problem}[Sphere problem]\label{prob:vii28-07}
Give two reasons the round sphere has vanishing Weyl tensor.
\end{problem}
\begin{solution}
It is a constant-curvature space form, and stereographic coordinates show it is locally conformal to Euclidean space; either property forces \(W=0\).
\end{solution}

\begin{problem}[Local-flatness problem]\label{prob:vii28-08}
Why does \(W=0\) mean local conformal flatness only for \(n\ge4\)?
\end{problem}
\begin{solution}
In dimension at least four Weyl is the complete local conformal obstruction. In dimension three it vanishes identically and the Cotton tensor supplies the missing differential obstruction.
\end{solution}

\begin{problem}[Ricci-information problem]\label{prob:vii28-09}
Explain what information Ricci loses that Weyl retains.
\end{problem}
\begin{solution}
Ricci is a contraction and records averages of sectional curvature; Weyl captures trace-free anisotropic curvature patterns invisible to every Ricci contraction.
\end{solution}

\begin{problem}[Scaling problem]\label{prob:vii28-10}
Under a constant rescaling \(\widetilde g=c^2g\), how does the fully covariant Weyl tensor scale?
\end{problem}
\begin{solution}
Constant rescaling is a conformal change with \(e^{2\varphi}=c^2\), so \(\widetilde W=c^2W\) as a \((0,4)\)-tensor.
\end{solution}

\begin{problem}[Low-dimensional hierarchy problem]\label{prob:vii28-11}
Summarize curvature data in dimensions two, three, and at least four.
\end{problem}
\begin{solution}
In dimension two one scalar \(K\) determines Riemann; in dimension three Ricci determines Riemann; from dimension four onward the independent Weyl tensor appears.
\end{solution}

\begin{problem}[Arc-synthesis problem]\label{prob:vii28-12}
Explain the curvature arc VII/26--VII/28.
\end{problem}
\begin{solution}
VII/26 defines full Riemann curvature, VII/27 extracts Ricci and scalar traces, and VII/28 isolates the remaining trace-free conformal curvature.
\end{solution}

\section{Challenge problems}
\begin{problem}[Dimension count for Weyl]\label{prob:vii28-ch01}
Derive the number of independent Weyl components in dimension \(n\ge4\).
\end{problem}
\begin{solution}
The algebraic Riemann tensor has \(n^2(n^2-1)/12\) components. The symmetric Ricci tensor has \(n(n+1)/2\) independent components and the decomposition is direct, so Weyl has \((n+2)(n+1)n(n-3)/12\) components.
\end{solution}

\begin{problem}[Einstein plus Weyl zero]\label{prob:vii28-ch02}
Show that an Einstein metric with \(W=0\) has constant sectional curvature when \(n\ge3\).
\end{problem}
\begin{solution}
The Einstein decomposition becomes \(\operatorname{Rm}=\frac{\lambda}{2(n-1)}g\owedge g\). Since \(\lambda\) is constant for \(n\ge3\), this is the space-form tensor with \(c=\lambda/(n-1)\).
\end{solution}

\begin{problem}[Conformal-flat sphere]\label{prob:vii28-ch03}
Use stereographic projection to explain why the round sphere is conformally flat.
\end{problem}
\begin{solution}
In stereographic coordinates the round metric has the form \(g=\Omega(x)^2g_{\mathrm{Eucl}}\) for an explicit positive factor \(\Omega\). Hence the sphere is locally conformal to Euclidean space and must have \(W=0\).
\end{solution}

\begin{problem}[Weyl versus scalar curvature]\label{prob:vii28-ch04}
Can \(W\neq0\) while scalar curvature is zero?
\end{problem}
\begin{solution}
Yes. Scalar curvature is only a double trace, while Weyl is trace free. A metric can have vanishing scalar curvature yet retain nonzero trace-free Riemann curvature.
\end{solution}

\begin{problem}[Curvature decomposition as representation theory preview]\label{prob:vii28-ch05}
Why is the decomposition into Weyl, trace-free Ricci, and scalar parts natural under orthogonal coordinate changes?
\end{problem}
\begin{solution}
Each piece is invariant under the orthogonal group action and corresponds to a distinct algebraic trace type. The decomposition separates irreducible-style components of the curvature tensor under the metric symmetry group.
\end{solution}
