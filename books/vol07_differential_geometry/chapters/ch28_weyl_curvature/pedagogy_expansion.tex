\section{Supplementary worked examples}
\label{sec:vii28-pedagogy}

\begin{example}[Three dimensions]\label{ex:vii28-ped-01}
In dimension three the Weyl tensor vanishes identically, so all Riemann curvature is determined algebraically by Ricci and scalar curvature.  There is no independent conformal curvature tensor.
\end{example}

\begin{example}[Space forms]\label{ex:vii28-ped-02}
Constant-sectional-curvature metrics have \(W=0\).  Hence Euclidean, spherical, and hyperbolic space forms are conformally flat at the local tensorial level.
\end{example}

\begin{example}[Einstein but not constant curvature]\label{ex:vii28-ped-03}
The product \(S^2(a)\times S^2(a)\) is Einstein because both factors have the same Ricci constant, but mixed tangent planes have sectional curvature zero while factor planes have curvature \(1/a^2\).  In dimension four this forces nonzero Weyl curvature.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Low dimension]\label{exr:vii28-ped-01}
What is \(W\) in dimension three?
\end{exercise}

\begin{hint}
Use the low-dimensional theorem.
\end{hint}

\begin{solution}
It vanishes identically.
\end{solution}

\begin{exercise}[Trace-free]\label{exr:vii28-ped-02}
What happens when any metric contraction is taken on \(W\)?
\end{exercise}

\begin{hint}
Recall its defining property.
\end{hint}

\begin{solution}
Every metric contraction is zero.
\end{solution}

\begin{exercise}[Space form]\label{exr:vii28-ped-03}
What is the Weyl tensor of a constant-curvature metric?
\end{exercise}

\begin{hint}
Substitute Ricci and scalar data into the decomposition.
\end{hint}

\begin{solution}
It is zero.
\end{solution}

\begin{exercise}[Einstein scalar]\label{exr:vii28-ped-04}
If \(\operatorname{Ric}=\lambda g\), what is scalar curvature?
\end{exercise}

\begin{hint}
Take the trace.
\end{hint}

\begin{solution}
\(\operatorname{Scal}=n\lambda\).
\end{solution}

\begin{exercise}[Conformal zero]\label{exr:vii28-ped-05}
If \(W=0\) for \(g\), what about \(e^{2\varphi}g\) in dimension at least four?
\end{exercise}

\begin{hint}
Use conformal covariance.
\end{hint}

\begin{solution}
Its Weyl tensor also vanishes.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Decomposition]\label{exr:vii28-ped-06}
Explain what information the Weyl tensor retains after subtracting Ricci and scalar traces.
\end{exercise}

\begin{hint}
Use the curvature decomposition.
\end{hint}

\begin{solution}
It is the totally trace-free part of the Riemann tensor, carrying curvature information invisible to all metric contractions.
\end{solution}

\begin{exercise}[Einstein simplification]\label{exr:vii28-ped-07}
Derive the Weyl decomposition when \(\operatorname{Ric}=\lambda g\).
\end{exercise}

\begin{hint}
Insert Ricci and scalar curvature into the general formula.
\end{hint}

\begin{solution}
One obtains \(\operatorname{Rm}=W+\frac{\lambda}{2(n-1)}g\owedge g\) with this convention.
\end{solution}

\begin{exercise}[Product nonzero Weyl]\label{exr:vii28-ped-08}
Why does \(S^2(a)\times S^2(a)\) have nonzero Weyl curvature?
\end{exercise}

\begin{hint}
Compare sectional curvatures with the consequence of \(W=0\) for an Einstein metric.
\end{hint}

\begin{solution}
An Einstein metric with \(W=0\) would have constant sectional curvature, but the product has factor planes of curvature \(1/a^2\) and mixed planes of curvature zero.
\end{solution}

\begin{exercise}[Conformal covariance]\label{exr:vii28-ped-09}
How does the fully covariant Weyl tensor scale under \(\widetilde g=e^{2\varphi}g\)?
\end{exercise}

\begin{hint}
Recall the weight.
\end{hint}

\begin{solution}
\(\widetilde W=e^{2\varphi}W\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Ricci-flat implies Weyl zero]\label{exr:vii28-ped-10}
Test the claim in dimension four.
\end{exercise}

\begin{hint}
Ricci-flat only removes the trace part.
\end{hint}

\begin{solution}
False.  Ricci-flat metrics can have nonzero Weyl curvature.
\end{solution}

\begin{exercise}[Weyl in dimension two]\label{exr:vii28-ped-11}
Test the claim: the Weyl decomposition supplies independent curvature in dimension two.
\end{exercise}

\begin{hint}
Dimension two has only Gaussian curvature.
\end{hint}

\begin{solution}
False.  The usual Weyl tensor carries no independent information there.
\end{solution}

\begin{exercise}[Conformal flat equals flat]\label{exr:vii28-ped-12}
Test the claim: \(W=0\) means the metric is flat.
\end{exercise}

\begin{hint}
Consider the round sphere.
\end{hint}

\begin{solution}
False.  Space forms with nonzero constant curvature are conformally flat but not flat.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Relativity]\label{exr:vii28-ped-13}
Why is Weyl curvature associated with tidal or radiative gravitational degrees of freedom in four dimensions?
\end{exercise}

\begin{hint}
Ricci is tied to contractions, Weyl is the trace-free remainder.
\end{hint}

\begin{solution}
Weyl encodes the part of spacetime curvature not fixed locally by Ricci contractions, including free tidal distortion in vacuum.
\end{solution}

\begin{exercise}[Conformal diagnostics]\label{exr:vii28-ped-14}
Why is \(W=0\) a useful conformal-flatness diagnostic in dimension at least four?
\end{exercise}

\begin{hint}
Use conformal invariance of the vanishing condition.
\end{hint}

\begin{solution}
Because vanishing is unchanged by conformal rescaling and locally conformally flat metrics must have zero Weyl tensor.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Product challenge]\label{exr:vii28-ped-15}
For \(S^2(a)\times S^2(a)\), identify two sectional curvatures that differ.
\end{exercise}

\begin{hint}
Use a plane tangent to one factor and a mixed plane.
\end{hint}

\begin{solution}
A factor plane has curvature \(1/a^2\), while a mixed plane has curvature \(0\).
\end{solution}

\begin{exercise}[Four-dimensional split]\label{exr:vii28-ped-16}
In dimension four, explain why knowing Ricci and scalar curvature is insufficient to reconstruct Riemann curvature.
\end{exercise}

\begin{hint}
Count the missing algebraic component conceptually.
\end{hint}

\begin{solution}
The Weyl tensor can vary independently while all Ricci contractions remain fixed, so the trace-free curvature must also be specified.
\end{solution}
