\chapter{Frenet Frames, Curvature and Torsion}
\label{ch:vii13-frenet-frames-curvature-and-torsion}

A regular curve carries more than a velocity.  Once arc length is chosen, successive derivatives separate the bending of the curve from the twisting of its osculating plane.  For a unit-speed space curve with nonzero curvature, the moving orthonormal frame
\[
(T,N,B)
\]
obeys the Frenet--Serret system
\[
\boxed{T'=\kappa N,\qquad N'=-\kappa T+\tau B,\qquad B'=-\tau N.}
\]
This chapter develops these invariants carefully, including formulas that do not require an arc-length parameter.

\section{Curvature of a unit-speed curve}
\begin{definition}[Curvature]\label{def:vii13-curvature}
Let \(\gamma:I\to\mathbb R^3\) be unit speed.  Its unit tangent is \(T=\gamma'\), and its curvature is
\[
\kappa(s)=\|T'(s)\|=\|\gamma''(s)\|.
\]
At a point where \(\kappa>0\), the principal normal is
\[
N=\frac{T'}{\kappa}.
\]
\end{definition}

\begin{proposition}[Acceleration is normal for unit speed]\label{prop:vii13-acc-normal}
For a unit-speed curve, \(T'\perp T\).
\end{proposition}
\begin{proof}
Differentiate \(\langle T,T\rangle=1\) to obtain \(2\langle T',T\rangle=0\).
\end{proof}

\begin{definition}[Binormal and Frenet frame]\label{def:vii13-binormal}
Where \(\kappa>0\), define
\[
B=T\times N.
\]
Then \((T,N,B)\) is a positively oriented orthonormal frame, called the Frenet frame.
\end{definition}

\begin{warning}[Straight points]\label{warn:vii13-straight-points}
If \(\kappa=0\), the principal normal and binormal are not determined by the curve at that point.  Curvature still makes sense, but the full Frenet frame does not.
\end{warning}

\section{Curvature in an arbitrary parameter}
\begin{theorem}[Parameter-independent curvature formula]\label{thm:vii13-curvature-formula}
For a regular curve \(\gamma(t)\subset\mathbb R^3\),
\[
\boxed{\kappa(t)=\frac{\|\gamma'(t)\times\gamma''(t)\|}{\|\gamma'(t)\|^3}.}
\]
For a planar curve \(\gamma=(x,y)\), this becomes
\[
\kappa=\frac{|x'y''-y'x''|}{(x'^2+y'^2)^{3/2}}.
\]
\end{theorem}
\begin{proof}
Write \(v=ds/dt=\|\gamma'\|\) and \(\gamma'=vT\).  Then \(\gamma''=v'T+v^2\kappa N\), so
\(\gamma'\times\gamma''=v^3\kappa B\).  Taking norms gives the formula.
\end{proof}

\begin{example}[Circle]\label{ex:vii13-circle}
For \(\gamma(t)=(R\cos t,R\sin t,0)\), one finds
\[
\kappa=\frac1R.
\]
Thus curvature is the reciprocal of the radius for a circle.
\end{example}

\section{Torsion and the Frenet--Serret equations}
\begin{definition}[Torsion]\label{def:vii13-torsion}
For a unit-speed curve with \(\kappa>0\), the torsion is
\[
\tau=-\langle B',N\rangle.
\]
\end{definition}

\begin{theorem}[Frenet--Serret equations]\label{thm:vii13-frenet-serret}
Where \(\kappa>0\),
\[
T'=\kappa N,\qquad
N'=-\kappa T+\tau B,\qquad
B'=-\tau N.
\]
\end{theorem}
\begin{proof}
The first equation is the definition of \(N\).  Differentiate the orthonormality relations among \(T,N,B\).  The derivative matrix of an orthonormal frame is skew-symmetric.  Since \(T'=\kappa N\) and \(B'\perp B,T\), the remaining coefficients are forced to be those displayed, with \(\tau=-\langle B',N\rangle\).
\end{proof}

\begin{theorem}[Torsion formula]\label{thm:vii13-torsion-formula}
If \(\gamma'\times\gamma''\neq0\), then for any regular parameter
\[
\boxed{\tau=
\frac{\det(\gamma',\gamma'',\gamma''')}{\|\gamma'\times\gamma''\|^2}.}
\]
\end{theorem}
\begin{proof}
The numerator is the scalar triple product measuring the component of \(\gamma'''\) in the binormal direction.  Expanding derivatives through \(\gamma'=vT\) and using the Frenet equations gives numerator \(v^6\kappa^2\tau\), while the denominator is \(v^6\kappa^2\).
\end{proof}

\section{Osculating geometry}
\begin{definition}[Osculating, normal and rectifying planes]\label{def:vii13-planes}
At a point where \(\kappa>0\), the osculating plane is \(\operatorname{span}(T,N)\), the normal plane is \(\operatorname{span}(N,B)\), and the rectifying plane is \(\operatorname{span}(T,B)\).
\end{definition}

\begin{proposition}[Osculating circle]\label{prop:vii13-osculating-circle}
At a point with \(\kappa>0\), the osculating circle has radius
\[
\rho=\frac1\kappa
\]
and center \(\gamma+\rho N\).  It matches position, tangent direction, and curvature to second order.
\end{proposition}

\section{Planar curves and torsion}
\begin{theorem}[Planarity criterion]\label{thm:vii13-planarity}
Let \(\gamma\) be connected, unit speed, and satisfy \(\kappa>0\).  Then \(\tau\equiv0\) if and only if the image of \(\gamma\) lies in an affine plane.
\end{theorem}
\begin{proof}
If \(\tau=0\), then \(B'=0\), so \(B\) is constant and \(\langle\gamma,B\rangle\) is constant because \(\gamma'=T\perp B\).  Conversely, for a curve in a fixed plane its binormal is constant, so \(B'=0\) and \(\tau=0\).
\end{proof}

\section{The circular helix}
\begin{example}[Helix]\label{ex:vii13-helix}
For
\[
\gamma(t)=(a\cos t,a\sin t,bt),\qquad a>0,
\]
the speed is \(\sqrt{a^2+b^2}\), and
\[
\kappa=\frac{a}{a^2+b^2},\qquad
\tau=\frac{b}{a^2+b^2}.
\]
Both invariants are constant.  The ratio \(\tau/\kappa=b/a\) records the slope of the helix.
\end{example}

\begin{theorem}[General helix criterion]\label{thm:vii13-general-helix}
A unit-speed curve with \(\kappa>0\) has tangent making a constant angle with a fixed direction if and only if \(\tau/\kappa\) is constant.
\end{theorem}

\section{Rigid motions and the fundamental theorem}
\begin{proposition}[Euclidean invariance]\label{prop:vii13-rigid-invariance}
Orientation-preserving rigid motions preserve \(\kappa\) and \(\tau\).  Orientation-reversing rigid motions preserve \(\kappa\) and reverse the sign convention for the oriented binormal, hence reverse the corresponding signed torsion convention.
\end{proposition}

\begin{theorem}[Fundamental theorem of space curves]\label{thm:vii13-fundamental-curves}
Let \(\kappa:I\to(0,\infty)\) and \(\tau:I\to\mathbb R\) be smooth.  There exists a unit-speed curve with these curvature and torsion functions, unique up to an orientation-preserving Euclidean rigid motion.
\end{theorem}
\begin{proof}[Proof idea]
Solve the Frenet--Serret linear ODE for an oriented orthonormal frame with prescribed initial frame, then integrate \(\gamma'=T\).  ODE uniqueness shows that two solutions with the same initial position and frame coincide; arbitrary initial data differ by a rigid motion.
\end{proof}

\section{Exercises}
\begin{exercise}\label{exr:vii13-01}
For a unit-speed curve prove \(\gamma''\perp\gamma'\).
\end{exercise}
\begin{hint}
Differentiate \(\|\gamma'\|^2=1\).
\end{hint}
\begin{solution}
The derivative gives \(2\langle\gamma'',\gamma'\rangle=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-02}
Compute the curvature of a straight line.
\end{exercise}
\begin{hint}
Its unit tangent is constant.
\end{hint}
\begin{solution}
For \(\gamma(s)=p+sT_0\), \(T'=0\), hence \(\kappa=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-03}
Compute the curvature of a circle of radius \(R\).
\end{exercise}
\begin{hint}
Use the arbitrary-parameter formula.
\end{hint}
\begin{solution}
\(\|\gamma'\times\gamma''\|=R^2\) and \(\|\gamma'\|^3=R^3\), so \(\kappa=1/R\).
\end{solution}

\begin{exercise}\label{exr:vii13-04}
Derive the planar determinant formula for curvature.
\end{exercise}
\begin{hint}
Embed the curve in the plane \(z=0\).
\end{hint}
\begin{solution}
The cross product has only third component \(x'y''-y'x''\), giving the stated formula.
\end{solution}

\begin{exercise}\label{exr:vii13-05}
Explain why \(N\) is undefined at a point with \(\kappa=0\).
\end{exercise}
\begin{hint}
Look at \(N=T'/\kappa\).
\end{hint}
\begin{solution}
Both the normalization and a distinguished bending direction fail when \(T'=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-06}
Prove \(B\perp T,N\) and \(\|B\|=1\).
\end{exercise}
\begin{hint}
Use properties of the cross product.
\end{hint}
\begin{solution}
Since \(T,N\) are orthonormal, \(T\times N\) is a unit vector perpendicular to both.
\end{solution}

\begin{exercise}\label{exr:vii13-07}
Show \(B'\perp B\).
\end{exercise}
\begin{hint}
Differentiate \(\langle B,B\rangle=1\).
\end{hint}
\begin{solution}
One obtains \(2\langle B',B\rangle=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-08}
Why is \(B'\perp T\)?
\end{exercise}
\begin{hint}
Differentiate \(\langle B,T\rangle=0\).
\end{hint}
\begin{solution}
\(\langle B',T\rangle=-\langle B,T'\rangle=-\kappa\langle B,N\rangle=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-09}
Derive \(B'=-\tau N\).
\end{exercise}
\begin{hint}
Use the previous two orthogonality relations.
\end{hint}
\begin{solution}
\(B'\) is parallel to \(N\), and the definition \(\tau=-\langle B',N\rangle\) gives the coefficient.
\end{solution}

\begin{exercise}\label{exr:vii13-10}
Derive the coefficient of \(T\) in \(N'\).
\end{exercise}
\begin{hint}
Differentiate \(\langle N,T\rangle=0\).
\end{hint}
\begin{solution}
\(\langle N',T\rangle=-\langle N,T'\rangle=-\kappa\).
\end{solution}

\begin{exercise}\label{exr:vii13-11}
Compute \(\kappa\) for \(\gamma(t)=(t,t^2,0)\).
\end{exercise}
\begin{hint}
Use the planar formula.
\end{hint}
\begin{solution}
\(\kappa(t)=2/(1+4t^2)^{3/2}\).
\end{solution}

\begin{exercise}\label{exr:vii13-12}
Show a planar curve with \(\kappa>0\) has zero torsion.
\end{exercise}
\begin{hint}
Its binormal is a fixed normal to the plane.
\end{hint}
\begin{solution}
The binormal is constant, hence \(B'=0\) and \(\tau=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-13}
Compute curvature and torsion of \((a\cos t,a\sin t,bt)\).
\end{exercise}
\begin{hint}
Use the cross-product and determinant formulas.
\end{hint}
\begin{solution}
The formulas give \(\kappa=a/(a^2+b^2)\) and \(\tau=b/(a^2+b^2)\).
\end{solution}

\begin{exercise}\label{exr:vii13-14}
What geometric data does the osculating plane contain?
\end{exercise}
\begin{hint}
Recall its spanning frame vectors.
\end{hint}
\begin{solution}
It contains the tangent and principal normal directions, so it captures the second-order bending of the curve.
\end{solution}

\begin{exercise}\label{exr:vii13-15}
Find the osculating-circle radius in terms of curvature.
\end{exercise}
\begin{hint}
Use the circle model.
\end{hint}
\begin{solution}
The radius is \(\rho=1/\kappa\).
\end{solution}

\begin{exercise}\label{exr:vii13-16}
Prove curvature is unchanged by translation.
\end{exercise}
\begin{hint}
Derivatives of a translated curve are unchanged.
\end{hint}
\begin{solution}
Both \(\gamma'\) and \(\gamma''\) are unchanged, hence so is the curvature formula.
\end{solution}

\begin{exercise}\label{exr:vii13-17}
Prove curvature is unchanged by an orthogonal linear map.
\end{exercise}
\begin{hint}
Orthogonal maps preserve norms and cross-product norms.
\end{hint}
\begin{solution}
The numerator and denominator in the curvature formula are preserved.
\end{solution}

\begin{exercise}\label{exr:vii13-18}
Explain why torsion is a third-order invariant.
\end{exercise}
\begin{hint}
Inspect its arbitrary-parameter formula.
\end{hint}
\begin{solution}
The determinant formula contains \(\gamma'''\), so third derivatives are required.
\end{solution}

\begin{exercise}\label{exr:vii13-19}
Show that \(\tau=0\) implies constant binormal when \(\kappa>0\).
\end{exercise}
\begin{hint}
Use the third Frenet equation.
\end{hint}
\begin{solution}
\(B'=-\tau N=0\), hence \(B\) is constant on the interval.
\end{solution}

\begin{exercise}\label{exr:vii13-20}
If \(\kappa\) and \(\tau\) are both constant with \(\kappa>0\), what model should one expect?
\end{exercise}
\begin{hint}
Compare with the circular helix.
\end{hint}
\begin{solution}
Up to a rigid motion the fundamental theorem identifies the curve with a circular helix; the case \(\tau=0\) is a circle.
\end{solution}

\begin{exercise}\label{exr:vii13-21}
Write the Frenet equations as a matrix ODE.
\end{exercise}
\begin{hint}
Use the ordered frame \((T,N,B)\).
\end{hint}
\begin{solution}
With frame columns, the coefficient matrix is skew-symmetric with entries \(\kappa\) and \(\tau\), expressing infinitesimal rotation.
\end{solution}

\begin{exercise}\label{exr:vii13-22}
Why does the Frenet coefficient matrix preserve orthonormality?
\end{exercise}
\begin{hint}
Check skew-symmetry.
\end{hint}
\begin{solution}
A solution \(F\) of \(F'=FA\) with \(A^T=-A\) satisfies \((F^TF)'=A^TF^TF+F^TFA=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-23}
What is the torsion of a circle?
\end{exercise}
\begin{hint}
A circle lies in a plane.
\end{hint}
\begin{solution}
Its binormal is constant, so \(\tau=0\).
\end{solution}

\begin{exercise}\label{exr:vii13-24}
Explain why \(\kappa>0\) is assumed in the fundamental theorem stated here.
\end{exercise}
\begin{hint}
The Frenet normal must be defined.
\end{hint}
\begin{solution}
Positive curvature guarantees a smooth distinguished normal and hence a nonsingular Frenet system.
\end{solution}

\section{Solved problem dossiers}
\begin{problem}[Legacy Frenet problem: construct the frame]\label{prob:vii13-01}
Given a unit-speed curve with \(\kappa>0\), construct \(T,N,B\) and prove it is an oriented orthonormal frame.
\end{problem}
\begin{solution}
Set \(T=\gamma'\), \(N=T'/\|T'\|\), and \(B=T\times N\). Unit speed gives \(T\perp T'\); the cross product then completes a positively oriented orthonormal frame.
\end{solution}

\begin{problem}[Legacy curvature problem]\label{prob:vii13-02}
Derive curvature in a non-unit parameter.
\end{problem}
\begin{solution}
Write \(\gamma'=vT\) and differentiate. The cross product kills the tangential term and leaves \(v^3\kappa B\), yielding \(\kappa=\|\gamma'\times\gamma''\|/\|\gamma'\|^3\).
\end{solution}

\begin{problem}[Legacy torsion problem]\label{prob:vii13-03}
Derive the determinant formula for torsion.
\end{problem}
\begin{solution}
Expand \(\gamma',\gamma'',\gamma'''\) in the Frenet frame. The scalar triple product equals \(v^6\kappa^2\tau\) and \(\|\gamma'\times\gamma''\|^2=v^6\kappa^2\).
\end{solution}

\begin{problem}[Frenet equations from orthonormality]\label{prob:vii13-04}
Recover the complete Frenet--Serret system from the derivatives of inner products.
\end{problem}
\begin{solution}
The derivative matrix of an orthonormal frame is skew-symmetric. The known entry \(T'=\kappa N\) and the definition of torsion determine all remaining entries.
\end{solution}

\begin{problem}[Planarity from torsion]\label{prob:vii13-05}
Prove the zero-torsion criterion for a connected curve with \(\kappa>0\).
\end{problem}
\begin{solution}
If \(\tau=0\), then \(B\) is constant and \(\langle\gamma,B\rangle\) is constant, placing the curve in a plane. The converse follows from constant plane normal.
\end{solution}

\begin{problem}[Circular helix dossier]\label{prob:vii13-06}
Compute the complete Frenet invariants of a circular helix.
\end{problem}
\begin{solution}
Direct differentiation gives constant speed, then the cross-product and triple-product formulas yield \(\kappa=a/(a^2+b^2)\), \(\tau=b/(a^2+b^2)\).
\end{solution}

\begin{problem}[Osculating circle dossier]\label{prob:vii13-07}
Explain why \(\gamma+(1/\kappa)N\) is the center of the second-order approximating circle.
\end{problem}
\begin{solution}
In arc length, Taylor expansion is \(\gamma(s+h)=\gamma+hT+\tfrac12h^2\kappa N+O(h^3)\), identical through second order to the circle with radius \(1/\kappa\) and that center.
\end{solution}

\begin{problem}[Rigid-motion invariance]\label{prob:vii13-08}
Prove curvature and signed torsion behavior under Euclidean rigid motions.
\end{problem}
\begin{solution}
Translations disappear under differentiation. Orthogonal maps preserve norms and scalar triple products up to determinant; hence curvature is invariant and signed torsion changes by the orientation sign.
\end{solution}

\begin{problem}[General helix criterion]\label{prob:vii13-09}
Prove that constant \(\tau/\kappa\) is equivalent to a fixed direction making constant angle with \(T\).
\end{problem}
\begin{solution}
If \(\tau/\kappa=c\), the vector \(cT+B\) has derivative \(c\kappa N-\tau N=0\). Conversely, differentiating a constant-angle relation and resolving in the Frenet frame yields a constant ratio.
\end{solution}

\begin{problem}[Frame ODE dossier]\label{prob:vii13-10}
Show the Frenet system evolves inside \(SO(3)\).
\end{problem}
\begin{solution}
Arrange \(T,N,B\) as an orthogonal matrix. Its logarithmic derivative is skew-symmetric, so the flow preserves determinant \(+1\) and orthonormality.
\end{solution}

\begin{problem}[Constant invariants]\label{prob:vii13-11}
Classify curves with positive constant curvature and constant torsion.
\end{problem}
\begin{solution}
The circular helix has the prescribed constants; uniqueness in the fundamental theorem implies every such curve is congruent to it.
\end{solution}

\begin{problem}[Straight-point pathology]\label{prob:vii13-12}
Explain what fails in Frenet theory at a zero-curvature point without claiming the curve itself is singular.
\end{problem}
\begin{solution}
Regularity concerns \(\gamma'\neq0\), so the curve remains regular. What fails is normalization of \(T'\), hence the distinguished principal normal and binormal.
\end{solution}

\section{Challenge problems}
\begin{problem}[Fundamental theorem by ODE]\label{prob:vii13-ch01}
Give a detailed existence-and-uniqueness argument for prescribed smooth \(\kappa>0\) and \(\tau\).
\end{problem}
\begin{solution}
Solve the skew-symmetric Frenet matrix ODE with an initial oriented orthonormal frame. Skew-symmetry preserves orthonormality. Integrate \(\gamma'=T\). Any two choices of initial position and frame are related by one orientation-preserving rigid motion, and ODE uniqueness propagates that relation.
\end{solution}

\begin{problem}[Spherical-curve test]\label{prob:vii13-ch02}
Assume \(\gamma\) is unit speed with \(\kappa>0\). Derive a condition involving \(\kappa\) and \(\tau\) that can be used to test whether the curve lies on a sphere.
\end{problem}
\begin{solution}
Write a candidate center in the normal plane as \(C=\gamma+aN+bB\), differentiate, and impose \(C'=0\). The resulting system begins with \(a=1/\kappa\) and relates \(a'\), \(b\), and \(\tau\); constancy of \(a^2+b^2\) then gives the radius.
\end{solution}

\begin{problem}[Curvature under scaling]\label{prob:vii13-ch03}
If \(\widetilde\gamma=c\gamma\) with \(c>0\), determine how curvature and torsion scale.
\end{problem}
\begin{solution}
Derivatives scale by \(c\). Thus \(\widetilde\kappa=c^2/c^3\,\kappa=\kappa/c\), and the torsion determinant scales by \(c^3\) while its denominator scales by \(c^4\), giving \(\widetilde\tau=\tau/c\).
\end{solution}

\begin{problem}[Signed planar curvature]\label{prob:vii13-ch04}
For an oriented regular planar curve define signed curvature and show it changes sign under orientation reversal.
\end{problem}
\begin{solution}
For unit speed choose the positively rotated normal \(n\) and write \(T'=k_s n\). Reversing the parameter reverses \(T\) but preserves the second derivative relation in a way that reverses the chosen oriented normal convention, hence \(k_s\) changes sign while \(|k_s|=\kappa\) does not.
\end{solution}

\begin{problem}[Frenet data and local congruence]\label{prob:vii13-ch05}
Two unit-speed curves have the same positive curvature and torsion on an interval. Prove they differ by one rigid motion on the entire interval.
\end{problem}
\begin{solution}
Choose the unique orientation-preserving rigid motion matching one point and the two Frenet frames there. The transformed first curve and the second solve the same Frenet ODE with identical initial data; uniqueness makes the frames, and then the integrated positions, agree everywhere.
\end{solution}

\section*{Bridge to VII/14}
Curves now have a complete local moving-frame theory.  VII/14 moves from one-dimensional images to regular surfaces in \(\mathbb R^3\), where two independent tangent directions replace the single tangent vector and local parametrizations become the basic coordinate device.
