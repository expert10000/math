\section{Supplementary worked examples}
\label{sec:vii10-pedagogy}

\begin{example}[Orientation from a volume form]\label{ex:vii10-ped-01}
A nowhere-vanishing \(n\)-form \(\Omega\) orients an \(n\)-manifold by declaring a basis \((v_1,\ldots,v_n)\) positive when \(\Omega(v_1,\ldots,v_n)>0\).  Multiplying \(\Omega\) by a positive function preserves the orientation; multiplying by a negative function reverses it.
\end{example}

\begin{example}[Orientation reversal changes an integral sign]\label{ex:vii10-ped-02}
On an oriented interval \([a,b]\),
\[
\int_{[b,a]} f(x)\,dx=-\int_{[a,b]} f(x)\,dx.
\]
The same principle holds in every dimension: reversing orientation changes the sign of integrals of top-degree forms.
\end{example}

\begin{example}[Area of the sphere from a parametrized patch]\label{ex:vii10-ped-03}
Away from one meridian, spherical coordinates give
\[
\Omega=\sin\phi\,d\phi\wedge d\theta
\]
for the standard oriented area form on the unit sphere.  Integrating over \(0<\phi<\pi\), \(0<\theta<2\pi\) yields \(4\pi\), with the orientation encoded by the order of the wedge factors.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Orientation of a basis]\label{exr:vii10-ped-01}
In \(\mathbb R^2\) with orientation \(dx\wedge dy\), determine the sign of the basis \(((1,1),(1,-1))\).
\end{exercise}
\begin{hint}
Compute the determinant of the basis matrix.
\end{hint}
\begin{solution}
The determinant is \(-2\), so the basis is negatively oriented.
\end{solution}

\begin{exercise}[Reverse coordinates]\label{exr:vii10-ped-02}
What happens to \(dx\wedge dy\) under the coordinate swap \((u,v)=(y,x)\)?
\end{exercise}
\begin{hint}
Compute \(du\wedge dv\).
\end{hint}
\begin{solution}
\(du\wedge dv=dy\wedge dx=-dx\wedge dy\), so the swap reverses orientation.
\end{solution}

\begin{exercise}[Circle orientation]\label{exr:vii10-ped-03}
Give the boundary orientation on the unit circle induced from the standard orientation of the unit disc.
\end{exercise}
\begin{hint}
Use outward-normal-first convention.
\end{hint}
\begin{solution}
The positive boundary direction is counterclockwise.
\end{solution}

\begin{exercise}[Simple top-form integral]\label{exr:vii10-ped-04}
Compute \(\int_{[0,1]^2} x\,dx\wedge dy\) with standard orientation.
\end{exercise}
\begin{hint}
Use an iterated integral.
\end{hint}
\begin{solution}
The value is \(\int_0^1\int_0^1x\,dx\,dy=1/2\).
\end{solution}

\begin{exercise}[Sphere area]\label{exr:vii10-ped-05}
Evaluate \(\int_0^{2\pi}\int_0^\pi \sin\phi\,d\phi\,d\theta\).
\end{exercise}
\begin{hint}
Integrate in \(\phi\) first.
\end{hint}
\begin{solution}
The value is \(2\pi\cdot2=4\pi\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Positive rescaling]\label{exr:vii10-ped-06}
Prove that \(\Omega\) and \(f\Omega\) determine the same orientation when \(f>0\).
\end{exercise}
\begin{hint}
Compare their values on a basis.
\end{hint}
\begin{solution}
\((f\Omega)(v_1,\ldots,v_n)=f(p)\Omega(v_1,\ldots,v_n)\), so positivity has the same sign.
\end{solution}

\begin{exercise}[Orientation atlas criterion]\label{exr:vii10-ped-07}
Prove that an atlas with positive transition Jacobians determines an orientation.
\end{exercise}
\begin{hint}
Use the standard orientation in each coordinate chart.
\end{hint}
\begin{solution}
Positive Jacobian determinants guarantee that the locally declared positive bases agree on overlaps, so they patch to a global orientation.
\end{solution}

\begin{exercise}[Integral sign reversal]\label{exr:vii10-ped-08}
Prove that reversing the orientation of \(M\) negates \(\int_M\omega\).
\end{exercise}
\begin{hint}
Use oriented coordinate charts in the definition of integration.
\end{hint}
\begin{solution}
Each oriented chart contributes the negative of its former coordinate integral after orientation reversal, so the whole partition-of-unity sum changes sign.
\end{solution}

\begin{exercise}[Orientation preserved by positive Jacobian]\label{exr:vii10-ped-09}
Show a local diffeomorphism preserves orientation exactly when its coordinate Jacobian determinant is positive.
\end{exercise}
\begin{hint}
Evaluate the pullback of a positive top form.
\end{hint}
\begin{solution}
The top form is multiplied by the determinant of the derivative; its sign is positive precisely when the map preserves orientation.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Every manifold is orientable]\label{exr:vii10-ped-10}
Test the claim with \(\mathbb{RP}^2\).
\end{exercise}
\begin{hint}
Recall the standard nonorientable surface.
\end{hint}
\begin{solution}
False.  The real projective plane is a connected smooth surface with no consistent global orientation.
\end{solution}

\begin{exercise}[A nowhere-zero top form is irrelevant]\label{exr:vii10-ped-11}
Test whether such a form carries orientation information.
\end{exercise}
\begin{hint}
Evaluate it on ordered bases.
\end{hint}
\begin{solution}
False.  A nowhere-zero top form determines an orientation directly by its sign on bases.
\end{solution}

\begin{exercise}[Integrals ignore orientation]\label{exr:vii10-ped-12}
Test by integrating \(dx\) over an interval with opposite orientations.
\end{exercise}
\begin{hint}
Reverse the parameter direction.
\end{hint}
\begin{solution}
False.  The two values differ by a minus sign.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Change of variables]\label{exr:vii10-ped-13}
Explain why orientation-preserving coordinate changes use the ordinary positive Jacobian in top-form integration.
\end{exercise}
\begin{hint}
Pull back the top-degree form.
\end{hint}
\begin{solution}
The determinant enters as the coefficient of the pulled-back top form; positivity ensures the chosen orientations agree.
\end{solution}

\begin{exercise}[Flux orientation]\label{exr:vii10-ped-14}
Explain why a choice of orientation matters for flux across a hypersurface.
\end{exercise}
\begin{hint}
Changing the normal orientation changes the induced top form.
\end{hint}
\begin{solution}
Reversing the hypersurface orientation reverses the sign of the flux integral, distinguishing inward from outward flux.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Orientability from a top form]\label{exr:vii10-ped-15}
Prove that a smooth \(n\)-manifold is orientable iff it admits a nowhere-vanishing smooth \(n\)-form.
\end{exercise}
\begin{hint}
One direction is immediate; for the other patch local positive forms with a partition of unity.
\end{hint}
\begin{solution}
A nonzero top form defines an orientation.  Conversely, choose positively oriented coordinate volume forms and combine them using a partition of unity; all local contributions have the same sign, so their sum never vanishes.
\end{solution}

\begin{exercise}[Degree-one sign test]\label{exr:vii10-ped-16}
Let \(f:S^1\to S^1\), \(f(e^{it})=e^{-it}\).  Determine whether it preserves orientation.
\end{exercise}
\begin{hint}
Use the angular coordinate derivative.
\end{hint}
\begin{solution}
The derivative is \(-1\), so \(f\) reverses orientation and pulls the positive angular form back to its negative.
\end{solution}
