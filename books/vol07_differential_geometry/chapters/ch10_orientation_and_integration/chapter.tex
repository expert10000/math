\chapter{Orientation and Integration}
\label{ch:vii10-orientation-and-integration}

Orientation records a coherent choice of handedness on tangent spaces. Once such a choice is made, compactly supported top-degree differential forms can be integrated intrinsically. The two central principles are
\[
\boxed{\text{orientation}=\text{coherent sign of frames},\qquad
\int_M\omega=\text{coordinate integrals glued by a partition of unity}.}
\]

\section{Orientation of vector spaces}
\begin{definition}[Orientation of a vector space]\label{def:vii10-vector-orientation}
Let $V$ be an $n$-dimensional real vector space. Two ordered bases are equivalent when the determinant of the change-of-basis matrix is positive. An \emph{orientation} of $V$ is one of the two equivalence classes.
\end{definition}

\begin{proposition}[Top exterior power criterion]\label{prop:vii10-top-wedge}
A nonzero element $\Omega\in\Lambda^nV^*$ determines an orientation by declaring a basis $(v_1,\dots,v_n)$ positive exactly when $\Omega(v_1,\dots,v_n)>0$.
\end{proposition}
\begin{proof}
Under a basis change $A$, an alternating $n$-form is multiplied by $\det A$. Hence the sign changes precisely under orientation-reversing basis changes.
\end{proof}

\section{Oriented manifolds and oriented atlases}
\begin{definition}[Oriented atlas]\label{def:vii10-oriented-atlas}
An atlas on an $n$-manifold $M$ is oriented if every transition map has positive Jacobian determinant wherever defined. An \emph{orientation} of $M$ is a maximal oriented atlas.
\end{definition}

\begin{theorem}[Tangent-space interpretation]\label{thm:vii10-tangent-orientation}
An orientation of $M$ is equivalent to a continuously varying choice of orientation of every tangent space $T_pM$.
\end{theorem}
\begin{proof}
An oriented chart transports the standard orientation of $\mathbb R^n$ to each tangent space. Positive transition Jacobians make the result chart-independent. Conversely, charts whose coordinate frames are positive form an oriented atlas.
\end{proof}

\begin{definition}[Orientation-preserving map]\label{def:vii10-orientation-map}
A local diffeomorphism $F:M\to N$ between oriented $n$-manifolds is orientation preserving at $p$ when $dF_p$ sends positive bases to positive bases; equivalently, its Jacobian in oriented charts has positive determinant.
\end{definition}

\begin{example}[Euclidean space]\label{ex:vii10-euclidean}
The standard ordered basis gives the standard orientation of $\mathbb R^n$. The coordinate form $dx^1\wedge\cdots\wedge dx^n$ is positive for this orientation.
\end{example}

\section{Orientability and orientation forms}
\begin{definition}[Orientation form]\label{def:vii10-orientation-form}
A nowhere-vanishing smooth $n$-form on an $n$-manifold is called an orientation form or volume form in the purely differential-topological sense.
\end{definition}

\begin{theorem}[Orientability criterion]\label{thm:vii10-orientation-form}
An $n$-manifold is orientable if and only if it admits a nowhere-vanishing smooth $n$-form.
\end{theorem}
\begin{proof}
A nowhere-zero top form chooses the positive bases pointwise. Conversely, choose positive coordinate top forms on an oriented atlas and glue them with a smooth partition of unity; positivity prevents cancellation to zero.
\end{proof}

\begin{proposition}[Connected manifolds have two orientations]\label{prop:vii10-two-orientations}
If $M$ is connected and orientable, then it has exactly two orientations.
\end{proposition}
\begin{proof}
Relative to one orientation, another assigns a sign $\pm1$ locally. This sign is locally constant and hence constant on a connected manifold.
\end{proof}

\begin{example}[Sphere]\label{ex:vii10-sphere}
The sphere $S^n\subset\mathbb R^{n+1}$ is orientable. If $N(p)=p$ is the outward normal and $\Omega$ is the standard $(n+1)$-form, then
\[
\omega_p(v_1,\ldots,v_n)=\Omega_p(N(p),v_1,\ldots,v_n)
\]
is a nowhere-vanishing $n$-form on $S^n$.
\end{example}

\begin{warning}[Orientability is global]\label{warning:vii10-global}
Every manifold is locally orientable. Nonorientability is the failure to make the local signs globally consistent; the M\"obius band is the basic example.
\end{warning}

\section{Compact support and partitions of unity}
\begin{definition}[Compact support]\label{def:vii10-compact-support}
For a differential form $\omega$, its support is the closure of the set on which it is nonzero. We write $\Omega_c^k(M)$ for compactly supported smooth $k$-forms.
\end{definition}

\begin{theorem}[Partition of unity]\label{thm:vii10-partition-unity}
Every smooth manifold admits smooth partitions of unity subordinate to any open cover.
\end{theorem}

\begin{remark}[Why partitions of unity enter integration]\label{rem:vii10-pou-role}
A compactly supported top form need not lie in one chart. A partition of unity decomposes it into finitely many chart-supported pieces, each of which can be integrated by ordinary multivariable calculus.
\end{remark}

\section{Integration in an oriented chart}
\begin{definition}[Integral of a chart-supported top form]\label{def:vii10-chart-integral}
Let $(U,x)$ be an oriented chart and let $\omega\in\Omega_c^n(M)$ have support contained in $U$. Write
\[
(x^{-1})^*\omega=f(u)\,du^1\wedge\cdots\wedge du^n.
\]
Define
\[
\int_M\omega:=\int_{x(U)}f(u)\,du.
\]
\end{definition}

\begin{theorem}[Chart independence]\label{thm:vii10-chart-independent}
The preceding integral is independent of the oriented chart containing the support.
\end{theorem}
\begin{proof}
For an orientation-preserving coordinate change $v=v(u)$, the coefficient of the pulled-back top form gains the factor $\det(Dv/Du)$, while the Euclidean change-of-variables theorem contributes exactly the same positive Jacobian factor.
\end{proof}

\section{Global integration}
\begin{definition}[Integral on an oriented manifold]\label{def:vii10-global-integral}
Choose a partition of unity $(\rho_i)$ subordinate to oriented coordinate neighborhoods. For $\omega\in\Omega_c^n(M)$ define
\[
\int_M\omega:=\sum_i\int_M\rho_i\omega.
\]
Only finitely many terms meet $\operatorname{supp}\omega$.
\end{definition}

\begin{theorem}[Well-definedness]\label{thm:vii10-well-defined}
The global integral is independent of the subordinate atlas and partition of unity.
\end{theorem}
\begin{proof}
Refine two decompositions simultaneously. On each overlap, chart independence applies; summing and using $\sum_i\rho_i=1$ recovers the same integral.
\end{proof}

\begin{proposition}[Linearity and positivity]\label{prop:vii10-linearity}
Integration is linear. If $\omega=f\Omega$ where $\Omega$ is a positive orientation form and $f\ge0$, then $\int_M\omega\ge0$ whenever the integral is defined.
\end{proposition}

\begin{theorem}[Change of variables on manifolds]\label{thm:vii10-change-variables}
If $F:M\to N$ is an orientation-preserving diffeomorphism of oriented $n$-manifolds and $\omega\in\Omega_c^n(N)$, then
\[
\int_M F^*\omega=\int_N\omega.
\]
If $F$ reverses orientation, the left-hand side equals $-\int_N\omega$.
\end{theorem}

\begin{example}[Polar coordinates as pullback]\label{ex:vii10-polar}
For $F(r,\theta)=(r\cos\theta,r\sin\theta)$ on $r>0$,
\[
F^*(dx\wedge dy)=r\,dr\wedge d\theta.
\]
Thus the familiar Jacobian $r$ is already encoded by pullback of the Euclidean area form.
\end{example}

\section{Integration on products}
\begin{proposition}[Product orientation]\label{prop:vii10-product-orientation}
If $M^m$ and $N^n$ are oriented, the ordered concatenation of positive bases of $T_pM$ and $T_qN$ defines the product orientation on $M\times N$.
\end{proposition}

\begin{proposition}[Separated product forms]\label{prop:vii10-product-integral}
For compactly supported top forms $\alpha\in\Omega_c^m(M)$ and $\beta\in\Omega_c^n(N)$,
\[
\int_{M\times N}\pi_M^*\alpha\wedge\pi_N^*\beta
=\left(\int_M\alpha\right)\left(\int_N\beta\right).
\]
\end{proposition}
\begin{proof}
Reduce by partitions of unity to product charts and apply Fubini's theorem.
\end{proof}

\section{Solved problem dossiers}

\begin{problem}[Legacy orientation problem: positive atlases]\label{prob:vii10-01}
Show that the positive-Jacobian condition on chart transitions is exactly what is needed to make coordinate-frame orientations consistent.
\end{problem}
\begin{solution}
Coordinate frames on an overlap are related by the transition derivative. Their orientations agree exactly when its determinant is positive. Thus an oriented atlas is precisely a compatible system of tangent-space orientations.
\end{solution}

\begin{problem}[Legacy orientation-form problem]\label{prob:vii10-02}
Prove the equivalence between orientability and the existence of a nowhere-vanishing top form.
\end{problem}
\begin{solution}
A nonzero top form selects positive tangent bases. Conversely, on an oriented atlas use the positive coordinate top forms and combine them with a partition of unity. Every summand is nonnegative relative to the chosen orientation and some summand is positive at each point, so the sum never vanishes.
\end{solution}

\begin{problem}[Legacy integration problem: chart independence]\label{prob:vii10-03}
Prove that the integral of a compactly supported top form contained in one oriented chart is independent of that chart.
\end{problem}
\begin{solution}
On an overlap write the form in both coordinate systems. The coefficients are related by the signed Jacobian determinant. Oriented transitions have positive determinant, so the Euclidean change-of-variables theorem shows the two integrals agree.
\end{solution}

\begin{problem}[Partition-of-unity globalization]\label{prob:vii10-04}
Derive a global integral from chart-supported integrals and prove independence of the partition.
\end{problem}
\begin{solution}
Decompose $\omega$ by a subordinate partition. Given two partitions, insert the common refinement $\rho_i\sigma_j$. Each refined term lies in a chart overlap, where chart independence holds. Summing first in either index gives the two original definitions.
\end{solution}

\begin{problem}[Change of variables]\label{prob:vii10-05}
Prove $\int_MF^*\omega=\int_N\omega$ for an orientation-preserving diffeomorphism.
\end{problem}
\begin{solution}
Choose a partition of unity subordinate to target charts. Its pullback is a partition on the source. Apply chartwise change of variables to each pulled-back summand and add.
\end{solution}

\begin{problem}[Outward orientation of the sphere]\label{prob:vii10-06}
Construct an explicit orientation form on $S^n$ from the ambient Euclidean form.
\end{problem}
\begin{solution}
Let $N(p)=p$ and define $\omega=\iota_N(dx^1\wedge\cdots\wedge dx^{n+1})|_{S^n}$. If $(v_1,\ldots,v_n)$ spans $T_pS^n$, then $(N,v_1,\ldots,v_n)$ is a basis of the ambient space, so $\omega$ never vanishes.
\end{solution}

\begin{problem}[The M\"obius obstruction]\label{prob:vii10-07}
Explain why the M\"obius band cannot carry a global orientation.
\end{problem}
\begin{solution}
Transport a local positive frame once around the central circle. The gluing reverses the transverse coordinate, returning the frame with opposite orientation. A globally consistent sign choice would therefore equal its own negative.
\end{solution}

\begin{problem}[Polar-coordinate Jacobian]\label{prob:vii10-08}
Recover the polar area factor from differential forms.
\end{problem}
\begin{solution}
Pull back $dx\wedge dy$ under $(r,\theta)\mapsto(r\cos\theta,r\sin\theta)$. The wedge computation gives $r\,dr\wedge d\theta$, so the Jacobian factor is intrinsic pullback data.
\end{solution}

\begin{problem}[Orientation of products]\label{prob:vii10-09}
Prove the product orientation is natural and compute the sign of the flip map.
\end{problem}
\begin{solution}
Concatenate positive bases of the factors. The flip moves $m$ vectors past $n$ vectors, hence changes orientation by $(-1)^{mn}$.
\end{solution}

\begin{problem}[Integration over a circle]\label{prob:vii10-10}
Using the standard counterclockwise orientation, compute $\int_{S^1}(-y\,dx+x\,dy)$.
\end{problem}
\begin{solution}
Parametrize by $\gamma(t)=(\cos t,\sin t)$, $0\le t\le2\pi$. Then $\gamma^*(-y\,dx+x\,dy)=dt$, so the integral is $2\pi$.
\end{solution}

\begin{problem}[Orientation-reversing symmetry]\label{prob:vii10-11}
Suppose an oriented compact manifold admits an orientation-reversing diffeomorphism $F$ and an invariant top form $F^*\omega=\omega$. Show $\int_M\omega=0$.
\end{problem}
\begin{solution}
Change of variables for an orientation-reversing diffeomorphism gives $\int_MF^*\omega=-\int_M\omega$. Invariance makes the left side equal to the original integral, so it must vanish.
\end{solution}

\begin{problem}[Top forms versus densities]\label{prob:vii10-12}
Compare the transformation law of an $n$-form and a density and explain the role of orientation.
\end{problem}
\begin{solution}
Top forms use $\det D\phi$ and therefore carry a sign. Densities use $|\det D\phi|$. Orientation chooses a coherent sign so top forms can be integrated globally as signed quantities.
\end{solution}


\section{Challenge problems}

\begin{problem}[Orientation double cover]\label{prob:vii10-ch01}
Show that every connected manifold $M$ has a canonical two-sheeted orientation cover whose points are pairs $(p,o_p)$ with $o_p$ an orientation of $T_pM$.
\end{problem}
\begin{solution}
Use oriented local charts to trivialize the two choices over each coordinate neighborhood. Transition functions either preserve or swap the two sheets according to the sign of their Jacobian. This defines a smooth two-sheeted cover. A global orientation is exactly a continuous section of this cover.
\end{solution}

\begin{problem}[Orientability of real projective space]\label{prob:vii10-ch02}
Determine for which $n$ the manifold $\mathbb{RP}^n$ is orientable.
\end{problem}
\begin{solution}
Use the antipodal quotient $S^n\to\mathbb{RP}^n$. The antipodal map on $S^n$ has degree $(-1)^{n+1}$ and preserves the sphere orientation exactly when $n$ is odd. Hence $\mathbb{RP}^n$ is orientable exactly for odd $n$.
\end{solution}

\begin{problem}[Integration and degree preview]\label{prob:vii10-ch03}
Let $F:M\to N$ be an orientation-preserving $d$-sheeted covering of compact connected oriented $n$-manifolds. Show $\int_MF^*\omega=d\int_N\omega$.
\end{problem}
\begin{solution}
Cover $N$ by evenly covered coordinate neighborhoods and choose a subordinate partition of unity. Each local piece has $d$ disjoint lifts, and change of variables gives the same integral on each lift. Summing yields the factor $d$.
\end{solution}

\begin{problem}[Compactly supported integral without global compactness]\label{prob:vii10-ch04}
Explain why $\int_M\omega$ is still well defined when $M$ is noncompact but $\omega$ has compact support.
\end{problem}
\begin{solution}
Only a compact neighborhood of $\operatorname{supp}\omega$ matters. A locally finite partition of unity intersects that compact set in finitely many terms, so the global definition reduces to a finite sum of ordinary chart integrals.
\end{solution}

\begin{problem}[Product orientation and Fubini sign]\label{prob:vii10-ch05}
Let $\alpha$ be an $m$-form on $M^m$ and $\beta$ an $n$-form on $N^n$. Compare the integral of $\pi_M^*\alpha\wedge\pi_N^*\beta$ with the corresponding form after swapping factors.
\end{problem}
\begin{solution}
The flip map exchanges the two factor blocks and changes orientation by $(-1)^{mn}$. Graded commutativity also gives $\beta\wedge\alpha=(-1)^{mn}\alpha\wedge\beta$. The two signs compensate exactly in the natural change-of-variables comparison.
\end{solution}

\section*{Bridge to VII/11}
Orientation makes signed integration possible. VII/11 identifies the precise compatibility between this integral and the exterior derivative: the integral of $d\omega$ over a manifold is the integral of $\omega$ over its oriented boundary.
