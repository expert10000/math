\chapter{Ruled and Developable Surfaces}
\label{ch:vii18-ruled-developable-surfaces}

A ruled surface is swept out by a one-parameter family of straight lines.  This simple construction produces planes, cylinders, cones, hyperboloids, helicoids, and many classical engineering surfaces.

\section{Ruled parametrizations}

\begin{definition}[Ruled surface]\label{def:vii18-ruled}
A parametrized surface is ruled if locally it can be written
\[
X(u,v)=\alpha(u)+v\,\beta(u),
\]
where \(\beta(u)\neq0\). For fixed \(u\), the curve \(v\mapsto X(u,v)\) is a straight line called a ruling.
\end{definition}

\begin{proposition}[Regularity criterion]\label{prop:vii18-regularity}
For
\[
X_u=\alpha'(u)+v\beta'(u),\qquad X_v=\beta(u),
\]
regularity is equivalent to
\[
\bigl(\alpha'(u)+v\beta'(u)\bigr)\times\beta(u)\neq0.
\]
\end{proposition}

\section{Standard examples}

\begin{example}[Plane]\label{ex:vii18-plane}
A plane is ruled in infinitely many ways.
\end{example}

\begin{example}[Generalized cylinder]\label{ex:vii18-cylinder}
If \(\beta\) is constant,
\[
X(u,v)=\alpha(u)+v b,
\]
the surface is a generalized cylinder.
\end{example}

\begin{example}[Cone]\label{ex:vii18-cone}
For a fixed vertex \(p\),
\[
X(u,v)=p+v\beta(u)
\]
is a cone away from singular points where the parametrization loses rank.
\end{example}

\begin{example}[Helicoid]\label{ex:vii18-helicoid}
\[
X(u,v)=(v\cos u,v\sin u,cu)
\]
is ruled: for each fixed \(u\), varying \(v\) traces a horizontal line.
\end{example}

\section{Developable surfaces}

\begin{definition}[Developable ruled surface]\label{def:vii18-developable}
A regular ruled surface is developable if its tangent plane is constant along each ruling.
\end{definition}

\begin{theorem}[Triple-product criterion]\label{thm:vii18-triple}
For
\[
X(u,v)=\alpha(u)+v\beta(u),
\]
with \(\beta\) normalized to unit length, the ruled surface is developable precisely when
\[
\det\bigl(\alpha'(u),\beta(u),\beta'(u)\bigr)=0.
\]
\end{theorem}

\begin{proof}
The tangent plane along a ruling is spanned by
\[
\beta,\qquad \alpha'+v\beta'.
\]
Its normal direction is
\[
(\alpha'+v\beta')\times\beta.
\]
Constancy of the tangent plane as \(v\) varies is equivalent to the vectors
\(\alpha',\beta,\beta'\) being linearly dependent, which is the vanishing triple product.
\end{proof}

\begin{theorem}[Developable implies zero Gaussian curvature]\label{thm:vii18-kzero}
At every regular point of a developable ruled surface,
\[
K=0.
\]
\end{theorem}

\begin{proof}
Along a ruling the tangent plane, hence the Gauss map, is constant. Therefore the ruling direction lies in \(\ker dN=\ker S\), so one principal curvature is zero and \(K=\det S=0\).
\end{proof}

\section{Gaussian curvature of a ruled surface}

\begin{proposition}[Sign restriction]\label{prop:vii18-k-nonpositive}
At regular nondevelopable points of a ruled surface in \(\mathbb R^3\),
\[
K<0.
\]
Thus a ruled surface has \(K\le0\) wherever regular.
\end{proposition}

\begin{remark}
The straight ruling has zero ambient curvature. Its normal curvature is therefore zero. If the surface is nondevelopable, this asymptotic direction is not a kernel direction of the whole shape operator, and the point is hyperbolic.
\end{remark}

\section{Striction and singular behavior}

\begin{definition}[Striction curve]\label{def:vii18-striction}
For a noncylindrical ruled surface with unit ruling field \(\beta\) and \(\beta'\neq0\), a distinguished base curve can be chosen so that its derivative is orthogonal to \(\beta'\). It is called a striction curve.
\end{definition}

\begin{remark}
The striction curve organizes where neighboring rulings come closest. For tangent developables it is also closely tied to the singular edge of regression.
\end{remark}

\section{Tangent developables}

\begin{definition}[Tangent developable]\label{def:vii18-tangent-developable}
For a regular space curve \(\alpha\),
\[
X(u,v)=\alpha(u)+v\alpha'(u)
\]
is its tangent developable.
\end{definition}

\begin{proposition}[Singular base curve]\label{prop:vii18-tangent-singular}
Along \(v=0\),
\[
X_u=\alpha',\qquad X_v=\alpha',
\]
so the parametrization is singular on the generating curve itself.
\end{proposition}

\section{Cones and cylinders as developables}

\begin{proposition}[Cylinders]\label{prop:vii18-cyl-developable}
Every generalized cylinder is developable.
\end{proposition}

\begin{proof}
Here \(\beta'=0\), so the triple product criterion vanishes identically.
\end{proof}

\begin{proposition}[Cones]\label{prop:vii18-cone-developable}
Every regular part of a cone is developable.
\end{proposition}

\begin{proof}
For \(X=p+v\beta(u)\), the vector \(\alpha'\) in the ruled representation may be taken as zero; equivalently the tangent plane along a generator is spanned by \(\beta,\beta'\), independent of \(v\).
\end{proof}

\section{The helicoid as a contrasting ruled surface}

\begin{proposition}[Helicoid is not developable]\label{prop:vii18-helicoid-not}
For \(c\neq0\), the helicoid is ruled but not developable; at its regular points its Gaussian curvature is strictly negative.
\end{proposition}

\section{Geometric meaning}

Developable surfaces are precisely the ruled surfaces whose tangent plane does not twist along a ruling. They are locally bendable from a plane without stretching in the classical smooth setting, and this is why cylinders and cones play a central role in fabrication, sheet materials, and geometric design.

\section{Solved problem dossiers}
\begin{problem}[Legacy ruled-surface problem]\label{prob:vii18-01}
Derive the basic tangent vectors and regularity condition for a ruled parametrization.
\end{problem}
\begin{solution}
Differentiate \(X=\alpha+v\beta\); regularity is exactly nonvanishing of \((\alpha'+v\beta')\times\beta\).
\end{solution}

\begin{problem}[Legacy developability problem]\label{prob:vii18-02}
Derive the scalar triple-product criterion.
\end{problem}
\begin{solution}
Require the tangent plane along each fixed ruling to be independent of \(v\). This is equivalent to linear dependence of \(\alpha',\beta,\beta'\).
\end{solution}

\begin{problem}[Legacy curvature problem]\label{prob:vii18-03}
Prove that a developable ruled surface has \(K=0\).
\end{problem}
\begin{solution}
The Gauss map is constant along each ruling, so the shape operator has a nontrivial kernel and determinant zero.
\end{solution}

\begin{problem}[Cylinder problem]\label{prob:vii18-04}
Verify directly that generalized cylinders are developable.
\end{problem}
\begin{solution}
The ruling field is constant, so \(\beta'=0\) and the triple product vanishes.
\end{solution}

\begin{problem}[Cone problem]\label{prob:vii18-05}
Verify directly that regular cone points are developable.
\end{problem}
\begin{solution}
Along a generator, the tangent plane is spanned by \(\beta,\beta'\) and is independent of the radial parameter.
\end{solution}

\begin{problem}[Tangent-developable problem]\label{prob:vii18-06}
Show why the tangent developable of a space curve is developable yet singular along its generating curve.
\end{problem}
\begin{solution}
The triple product vanishes because two vectors coincide, while at \(v=0\) the two parameter derivatives both equal \(\alpha'\).
\end{solution}

\begin{problem}[Helicoid problem]\label{prob:vii18-07}
Explain why the helicoid is ruled but nondevelopable.
\end{problem}
\begin{solution}
It is affine in the ruling parameter \(v\), but its tangent plane twists as \(v\) varies; equivalently the triple product criterion is nonzero.
\end{solution}

\begin{problem}[Asymptotic-ruling problem]\label{prob:vii18-08}
Show every ruling is an asymptotic direction.
\end{problem}
\begin{solution}
Each ruling is a straight curve, so its curvature vector vanishes and hence its normal curvature is zero.
\end{solution}

\begin{problem}[Sign-of-\(K\) problem]\label{prob:vii18-09}
Prove \(K\le0\) at regular points of a ruled surface.
\end{problem}
\begin{solution}
The existence of a zero normal-curvature direction excludes elliptic points. Nondevelopable points are hyperbolic, while developable points have \(K=0\).
\end{solution}

\begin{problem}[Gauss-map rank problem]\label{prob:vii18-10}
Describe the Gauss map on a developable surface.
\end{problem}
\begin{solution}
It is constant along each ruling, so its differential has a ruling-direction kernel and rank at most one.
\end{solution}

\begin{problem}[Striction problem]\label{prob:vii18-11}
Explain why changing the base curve by sliding along rulings can be used to impose a normalization.
\end{problem}
\begin{solution}
Replacing \(\alpha\) by \(\alpha+s(u)\beta\) changes the base curve without changing the surface. Choosing \(s\) appropriately can enforce the standard striction orthogonality condition.
\end{solution}

\begin{problem}[Fabrication interpretation]\label{prob:vii18-12}
Why are developables natural models for bending sheet material without stretching?
\end{problem}
\begin{solution}
Vanishing Gaussian curvature and constant tangent-plane behavior along rulings are compatible with local isometric bending from planar sheets, unlike doubly curved surfaces.
\end{solution}

\section{Challenge problems}
\begin{problem}[Derive the triple-product criterion carefully]\label{prob:vii18-ch01}
Starting from the normal \(X_u\times X_v\), show tangent-plane constancy along \(v\) is equivalent to \(\det(\alpha',\beta,\beta')=0\).
\end{problem}
\begin{solution}
The normal vector is \((\alpha'+v\beta')\times\beta=\alpha'\times\beta+v(\beta'\times\beta)\). Its direction is independent of \(v\) exactly when the two cross-product terms are linearly dependent, equivalent to \(\alpha',\beta,\beta'\) being linearly dependent.
\end{solution}

\begin{problem}[Rank of the shape operator]\label{prob:vii18-ch02}
Show that at a nonplanar regular point of a developable surface the shape operator has rank one.
\end{problem}
\begin{solution}
Developability gives a nonzero kernel direction. If the point is nonplanar then \(S\neq0\). On a two-dimensional tangent plane, a nonzero endomorphism with nontrivial kernel has rank one.
\end{solution}

\begin{problem}[Edge of regression preview]\label{prob:vii18-ch03}
For a tangent developable, explain geometrically why the base curve forms a singular edge.
\end{problem}
\begin{solution}
Neighboring tangent lines envelope the original curve. Along the curve itself the two parameter directions collapse to the same tangent vector, so the surface parametrization loses rank precisely along this envelope.
\end{solution}

\begin{problem}[Nondevelopable ruled points]\label{prob:vii18-ch04}
Show that a regular nondevelopable ruled point is hyperbolic.
\end{problem}
\begin{solution}
The ruling gives a zero normal-curvature direction. If \(K=0\), the self-adjoint shape operator would have a kernel and the Gauss map would fail to vary along that direction in the developable way. At a genuinely nondevelopable regular ruled point the two principal curvatures must have opposite signs, hence \(K<0\).
\end{solution}

\begin{problem}[Local flattening principle]\label{prob:vii18-ch05}
Explain, without proving the full theorem, why zero Gaussian curvature is the differential-geometric compatibility condition behind local flattening of a developable surface.
\end{problem}
\begin{solution}
Gaussian curvature is intrinsic to the first fundamental form. A plane has \(K=0\), so any surface locally isometric to a plane must also have \(K=0\). Classical developables satisfy this condition and admit local unfoldings away from singularities.
\end{solution}
