\chapter{First and Second Fundamental Forms}
\label{ch:vii15-first-and-second-fundamental-forms}

A surface patch supplies two kinds of local quadratic data.  The first fundamental form is the Euclidean inner product restricted to the tangent plane; it measures intrinsic lengths and angles.  The second fundamental form records the normal component of second derivatives and therefore measures how the surface bends inside \(\mathbb R^3\).  In coordinates,
\[
\boxed{I=E\,du^2+2F\,du\,dv+G\,dv^2,\qquad
II=e\,du^2+2f\,du\,dv+g\,dv^2.}
\]

\section{The first fundamental form}
\begin{definition}[First fundamental form]\label{def:vii15-first-form}
For a regular surface \(S\subset\mathbb R^3\), the first fundamental form at \(p\) is
\[
I_p(v,w)=\langle v,w\rangle,\qquad v,w\in T_pS.
\]
It is a positive-definite symmetric bilinear form.
\end{definition}

\begin{definition}[Metric coefficients]\label{def:vii15-EFG}
For a patch \(X(u,v)\), set
\[
E=\langle X_u,X_u\rangle,\qquad
F=\langle X_u,X_v\rangle,\qquad
G=\langle X_v,X_v\rangle.
\]
Then for \(w=aX_u+bX_v\),
\[
I(w,w)=Ea^2+2Fab+Gb^2.
\]
\end{definition}

\begin{proposition}[Positive definiteness]\label{prop:vii15-positive}
For a regular patch,
\[
E>0,\qquad EG-F^2>0.
\]
\end{proposition}
\begin{proof}
The matrix \(\begin{pmatrix}E&F\\F&G\end{pmatrix}\) is the Gram matrix of the independent vectors \(X_u,X_v\), hence positive definite.
\end{proof}

\section{Length, angle and area}
\begin{proposition}[Length of a surface curve]\label{prop:vii15-length}
If \(\alpha(t)=X(u(t),v(t))\), then
\[
\|\alpha'\|^2=E\,u'^2+2F\,u'v'+G\,v'^2,
\]
so
\[
L(\alpha)=\int\sqrt{E\,u'^2+2F\,u'v'+G\,v'^2}\,dt.
\]
\end{proposition}

\begin{proposition}[Angles]\label{prop:vii15-angle}
For tangent vectors \(a=a^iX_i\) and \(b=b^jX_j\),
\[
\cos\theta=\frac{I(a,b)}{\sqrt{I(a,a)I(b,b)}}.
\]
Coordinate curves are orthogonal exactly when \(F=0\).
\end{proposition}

\begin{theorem}[Area element]\label{thm:vii15-area-element}
The Euclidean area element of a patch is
\[
dA=\|X_u\times X_v\|\,du\,dv
=\sqrt{EG-F^2}\,du\,dv.
\]
\end{theorem}
\begin{proof}
The square of the cross-product norm is the Gram determinant
\(\|X_u\|^2\|X_v\|^2-\langle X_u,X_v\rangle^2=EG-F^2\).
\end{proof}

\section{The second fundamental form}
\begin{definition}[Second fundamental form coefficients]\label{def:vii15-efg}
Choose a local unit normal \(N\).  Define
\[
e=\langle X_{uu},N\rangle,\qquad
f=\langle X_{uv},N\rangle,\qquad
g=\langle X_{vv},N\rangle.
\]
The second fundamental form is
\[
II(aX_u+bX_v,aX_u+bX_v)=ea^2+2fab+gb^2.
\]
\end{definition}

\begin{proposition}[Normal acceleration]\label{prop:vii15-normal-acc}
For a surface curve \(\alpha(t)=X(u(t),v(t))\),
\[
\langle\alpha'',N\rangle
=e\,u'^2+2f\,u'v'+g\,v'^2.
\]
Thus \(II\) measures the normal component of acceleration determined by the tangent velocity.
\end{proposition}
\begin{proof}
Differentiate \(\alpha'=u'X_u+v'X_v\).  Terms involving \(u''X_u+v''X_v\) are tangent and vanish after pairing with \(N\); the remaining second-derivative terms give the displayed quadratic form.
\end{proof}

\begin{warning}[Dependence on the chosen normal]\label{warn:vii15-normal-sign}
Replacing \(N\) by \(-N\) leaves \(I\) unchanged but replaces \(II\) by \(-II\).  Extrinsic signed bending therefore depends on a choice of local orientation.
\end{warning}

\section{Normal curvature}
\begin{definition}[Normal curvature]\label{def:vii15-normal-curvature}
For a nonzero tangent direction \(v\in T_pS\), define
\[
\boxed{k_n(v)=\frac{II(v,v)}{I(v,v)}.}
\]
For a unit-speed surface curve with tangent \(v\), this equals the component of the curve curvature vector in the surface-normal direction.
\end{definition}

\begin{proposition}[Dependence only on direction]\label{prop:vii15-direction}
For any nonzero scalar \(c\),
\[
k_n(cv)=k_n(v).
\]
\end{proposition}
\begin{proof}
Both numerator and denominator are quadratic in \(v\), so the factor \(c^2\) cancels.
\end{proof}

\section{Coordinate change}
\begin{theorem}[Tensorial transformation]\label{thm:vii15-transform}
Under a parameter change \((u,v)=(u(\xi,\eta),v(\xi,\eta))\), the coefficient matrices of \(I\) and \(II\) transform by
\[
\widetilde G=J^T GJ,\qquad
\widetilde B=J^T BJ,
\]
provided the surface normal is transported consistently.  Thus both forms define bilinear forms independent of the chosen coordinates.
\end{theorem}

\section{Examples}
\begin{example}[Plane]\label{ex:vii15-plane}
For \(X(u,v)=p+ua+vb\) with constant independent vectors \(a,b\), all second derivatives vanish.  Hence
\[
II=0.
\]
The first form is the constant Gram form of \(a,b\).
\end{example}

\begin{example}[Sphere]\label{ex:vii15-sphere}
On the sphere of radius \(R\), choose the outward normal \(N=X/R\).  Differentiating \(\langle X_i,N\rangle=0\) gives
\[
\langle X_{ij},N\rangle=-\frac1R\langle X_i,X_j\rangle.
\]
With the sign convention used here,
\[
II=-\frac1R I.
\]
Choosing the inward normal reverses the sign.
\end{example}

\begin{example}[Cylinder]\label{ex:vii15-cylinder}
For \(X(u,v)=(R\cos u,R\sin u,v)\) and outward normal \(N=(\cos u,\sin u,0)\),
\[
E=R^2,\quad F=0,\quad G=1,
\]
and
\[
e=-R,\quad f=0,\quad g=0.
\]
Thus circumferential directions bend normally while axial directions have zero normal curvature.
\end{example}

\section{Local isometries and intrinsic data}
\begin{definition}[Local isometry of surfaces]\label{def:vii15-local-isometry}
A diffeomorphism \(\Phi:S\to\widetilde S\) is a local isometry if
\[
I_{\widetilde S}(d\Phi(v),d\Phi(w))=I_S(v,w)
\]
for tangent vectors in its domain.
\end{definition}

\begin{proposition}[Consequences of preserving \(I\)]\label{prop:vii15-isometry-consequences}
A local isometry preserves lengths of curves and angles between tangent vectors, and hence preserves the area density determined by \(I\).
\end{proposition}

\begin{example}[Plane and cylinder]\label{ex:vii15-plane-cylinder}
The map
\[
(u,v)\longmapsto(R\cos(u/R),R\sin(u/R),v)
\]
from a strip in the plane to the cylinder pulls back the cylinder's first form to
\[
du^2+dv^2.
\]
Thus the cylinder is locally isometric to the plane even though their second fundamental forms differ.
\end{example}

\begin{remark}[Intrinsic versus extrinsic]\label{rem:vii15-intrinsic-extrinsic}
The first fundamental form depends only on metric geometry within the surface.  The second fundamental form detects the embedding in \(\mathbb R^3\).  The plane--cylinder comparison is the basic example: same local first form, different bending.
\end{remark}

\section{Exercises}
\begin{exercise}\label{exr:vii15-01}
Compute \(E,F,G\) for a graph \(X(u,v)=(u,v,f(u,v))\).
\end{exercise}
\begin{hint}
Dot the two coordinate derivatives.
\end{hint}
\begin{solution}
\(E=1+f_u^2\), \(F=f_uf_v\), \(G=1+f_v^2\).
\end{solution}

\begin{exercise}\label{exr:vii15-02}
Prove \(EG-F^2>0\) for a regular patch.
\end{exercise}
\begin{hint}
Use the Gram determinant.
\end{hint}
\begin{solution}
\(EG-F^2=\|X_u\times X_v\|^2>0\).
\end{solution}

\begin{exercise}\label{exr:vii15-03}
When are coordinate curves orthogonal?
\end{exercise}
\begin{hint}
Inspect their tangent inner product.
\end{hint}
\begin{solution}
They are orthogonal exactly when \(F=\langle X_u,X_v\rangle=0\).
\end{solution}

\begin{exercise}\label{exr:vii15-04}
Derive the surface-curve speed formula.
\end{exercise}
\begin{hint}
Substitute \(\alpha'=u'X_u+v'X_v\).
\end{hint}
\begin{solution}
Expanding its squared norm gives \(Eu'^2+2Fu'v'+Gv'^2\).
\end{solution}

\begin{exercise}\label{exr:vii15-05}
Derive the area density from \(E,F,G\).
\end{exercise}
\begin{hint}
Use the cross-product identity.
\end{hint}
\begin{solution}
\(\|X_u\times X_v\|^2=EG-F^2\), hence \(dA=\sqrt{EG-F^2}\,du\,dv\).
\end{solution}

\begin{exercise}\label{exr:vii15-06}
Compute the first form of the cylinder.
\end{exercise}
\begin{hint}
Use the standard cylinder patch.
\end{hint}
\begin{solution}
\(E=R^2\), \(F=0\), \(G=1\), so \(I=R^2du^2+dv^2\).
\end{solution}

\begin{exercise}\label{exr:vii15-07}
Compute the first form of the plane patch \(X(u,v)=(u,v,0)\).
\end{exercise}
\begin{hint}
The coordinate tangent vectors are standard basis vectors.
\end{hint}
\begin{solution}
\(E=G=1\), \(F=0\), so \(I=du^2+dv^2\).
\end{solution}

\begin{exercise}\label{exr:vii15-08}
Why does reversing the surface normal reverse \(II\)?
\end{exercise}
\begin{hint}
Each coefficient is paired with \(N\).
\end{hint}
\begin{solution}
All of \(e,f,g\) change sign linearly, so the entire bilinear form changes sign.
\end{solution}

\begin{exercise}\label{exr:vii15-09}
Show a plane has zero second fundamental form.
\end{exercise}
\begin{hint}
Its patch is affine.
\end{hint}
\begin{solution}
All second derivatives of an affine patch vanish, hence \(e=f=g=0\).
\end{solution}

\begin{exercise}\label{exr:vii15-10}
Compute \(e,f,g\) for the standard cylinder with outward normal.
\end{exercise}
\begin{hint}
Differentiate twice.
\end{hint}
\begin{solution}
\(X_{uu}=(-R\cos u,-R\sin u,0)\), giving \(e=-R\); mixed and \(vv\) derivatives vanish, so \(f=g=0\).
\end{solution}

\begin{exercise}\label{exr:vii15-11}
Find the normal curvature of the axial direction on a cylinder.
\end{exercise}
\begin{hint}
Use \(v=X_v\).
\end{hint}
\begin{solution}
\(II(X_v,X_v)=g=0\), while \(I(X_v,X_v)=1\), so \(k_n=0\).
\end{solution}

\begin{exercise}\label{exr:vii15-12}
Find the normal curvature of the circumferential direction on the outward-oriented cylinder.
\end{exercise}
\begin{hint}
Use \(v=X_u\).
\end{hint}
\begin{solution}
\(k_n=e/E=-R/R^2=-1/R\) with the stated sign convention.
\end{solution}

\begin{exercise}\label{exr:vii15-13}
Prove \(k_n\) depends only on a tangent line, not its scaling.
\end{exercise}
\begin{hint}
Use quadratic homogeneity.
\end{hint}
\begin{solution}
Both forms scale by \(c^2\), so their ratio is unchanged.
\end{solution}

\begin{exercise}\label{exr:vii15-14}
For the sphere with outward normal, compute normal curvature in any direction.
\end{exercise}
\begin{hint}
Use \(II=-(1/R)I\).
\end{hint}
\begin{solution}
Every nonzero tangent direction has \(k_n=-1/R\).
\end{solution}

\begin{exercise}\label{exr:vii15-15}
What does \(F=0\) mean geometrically?
\end{exercise}
\begin{hint}
It is the mixed coefficient of \(I\).
\end{hint}
\begin{solution}
The coordinate tangent vectors, and hence the two coordinate curves at each point, meet orthogonally.
\end{solution}

\begin{exercise}\label{exr:vii15-16}
What does \(E=G\) and \(F=0\) mean for coordinates?
\end{exercise}
\begin{hint}
Compare the two coordinate basis lengths and angle.
\end{hint}
\begin{solution}
The coordinate directions are orthogonal and have equal scale; such coordinates are conformal.
\end{solution}

\begin{exercise}\label{exr:vii15-17}
Show local isometries preserve curve speed.
\end{exercise}
\begin{hint}
Apply the defining identity to \(\alpha'\).
\end{hint}
\begin{solution}
\(\|d\Phi(\alpha')\|^2=I(d\Phi\alpha',d\Phi\alpha')=I(\alpha',\alpha')\).
\end{solution}

\begin{exercise}\label{exr:vii15-18}
Show local isometries preserve curve length.
\end{exercise}
\begin{hint}
Integrate preserved speed.
\end{hint}
\begin{solution}
Length is the integral of speed, so pointwise speed preservation gives equal lengths.
\end{solution}

\begin{exercise}\label{exr:vii15-19}
Show local isometries preserve tangent angles.
\end{exercise}
\begin{hint}
Use the cosine formula in terms of \(I\).
\end{hint}
\begin{solution}
All numerator and denominator inner products are preserved, hence so is the angle.
\end{solution}

\begin{exercise}\label{exr:vii15-20}
Check the plane-to-cylinder map is locally isometric.
\end{exercise}
\begin{hint}
Differentiate with respect to \(u,v\).
\end{hint}
\begin{solution}
The images of the coordinate vectors are orthonormal: the \(u\)-derivative has unit length, the \(v\)-derivative has unit length, and they are orthogonal.
\end{solution}

\begin{exercise}\label{exr:vii15-21}
Why are the plane and cylinder not congruent despite local isometry?
\end{exercise}
\begin{hint}
Compare second fundamental forms.
\end{hint}
\begin{solution}
The plane has \(II=0\), whereas the cylinder has nonzero circumferential second-form coefficient.
\end{solution}

\begin{exercise}\label{exr:vii15-22}
Show the normal component of surface-curve acceleration depends only on its velocity.
\end{exercise}
\begin{hint}
Pair the second derivative expansion with \(N\).
\end{hint}
\begin{solution}
Tangential terms involving \(u'',v''\) vanish against \(N\), leaving only the quadratic expression in \(u',v'\).
\end{solution}

\begin{exercise}\label{exr:vii15-23}
How do \(I\) and \(II\) matrices transform under coordinate change?
\end{exercise}
\begin{hint}
Use bilinearity and the Jacobian matrix \(J\).
\end{hint}
\begin{solution}
Both transform by congruence: \(\widetilde G=J^TGJ\) and \(\widetilde B=J^TBJ\), with a consistent normal for \(II\).
\end{solution}

\begin{exercise}\label{exr:vii15-24}
Explain why \(I\) is positive definite while \(II\) need not be.
\end{exercise}
\begin{hint}
Compare their definitions.
\end{hint}
\begin{solution}
\(I(v,v)=\|v\|^2>0\) for \(v\neq0\). The second form is a signed normal-bending form and may be positive, negative, zero, or indefinite.
\end{solution}

\section{Solved problem dossiers}
\begin{problem}[Legacy first-form problem]\label{prob:vii15-01}
Derive the coefficients \(E,F,G\) and the quadratic line element.
\end{problem}
\begin{solution}
Expand the Euclidean inner product of \(dX=X_u\,du+X_v\,dv\) with itself to obtain \(E\,du^2+2F\,du\,dv+G\,dv^2\).
\end{solution}

\begin{problem}[Legacy area problem]\label{prob:vii15-02}
Prove \(dA=\sqrt{EG-F^2}\,du\,dv\).
\end{problem}
\begin{solution}
The area parallelogram is \(\|X_u\times X_v\|\), whose square is the Gram determinant \(EG-F^2\).
\end{solution}

\begin{problem}[Legacy second-form problem]\label{prob:vii15-03}
Derive \(e,f,g\) from normal acceleration.
\end{problem}
\begin{solution}
Differentiate a surface curve twice and pair with the normal. Terms multiplying \(u'',v''\) are tangent and disappear, leaving the quadratic expression with \(X_{uu},X_{uv},X_{vv}\).
\end{solution}

\begin{problem}[Graph fundamental forms]\label{prob:vii15-04}
Compute both forms for a graph \(z=f(u,v)\).
\end{problem}
\begin{solution}
The first coefficients are \(1+f_u^2,f_uf_v,1+f_v^2\). With \(W=\sqrt{1+f_u^2+f_v^2}\) and upward normal, the second coefficients are \(f_{uu}/W,f_{uv}/W,f_{vv}/W\).
\end{solution}

\begin{problem}[Sphere dossier]\label{prob:vii15-05}
Show the second form of the outward-oriented sphere is \(-(1/R)I\) in the convention of this chapter.
\end{problem}
\begin{solution}
Use \(N=X/R\) and differentiate \(\langle X_i,N\rangle=0\): \(\langle X_{ij},N\rangle=-\langle X_i,X_j\rangle/R\).
\end{solution}

\begin{problem}[Cylinder dossier]\label{prob:vii15-06}
Compute \(I\), \(II\), and normal curvature of the coordinate directions on the cylinder.
\end{problem}
\begin{solution}
\(I=R^2du^2+dv^2\), \(II=-Rdu^2\). Thus circumferential normal curvature is \(-1/R\) and axial normal curvature is zero.
\end{solution}

\begin{problem}[Normal-sign dossier]\label{prob:vii15-07}
Prove exactly which quantities in this chapter change when \(N\) is reversed.
\end{problem}
\begin{solution}
\(I,E,F,G\), lengths, angles, and area are unchanged. The coefficients \(e,f,g\), \(II\), and signed normal curvature all reverse sign.
\end{solution}

\begin{problem}[Coordinate-invariance dossier]\label{prob:vii15-08}
Prove the first and second forms are independent of the chosen patch.
\end{problem}
\begin{solution}
Under a parameter transition tangent components transform by the Jacobian. Bilinearity gives congruence transformation of both coefficient matrices, representing the same geometric bilinear forms.
\end{solution}

\begin{problem}[Plane--cylinder local isometry]\label{prob:vii15-09}
Prove a plane strip and cylinder patch have the same first fundamental form but different second forms.
\end{problem}
\begin{solution}
Using arc-length coordinate \(u=R\theta\), the cylinder first form becomes \(du^2+dv^2\), equal to the plane. The plane has \(II=0\), while the cylinder does not.
\end{solution}

\begin{problem}[Orthogonal coordinates dossier]\label{prob:vii15-10}
Assume \(F=0\). Simplify length, angle, and area formulas.
\end{problem}
\begin{solution}
Length density is \(Eu'^2+Gv'^2\), coordinate directions meet at right angles, and \(dA=\sqrt{EG}\,du\,dv\).
\end{solution}

\begin{problem}[Conformal coordinates dossier]\label{prob:vii15-11}
Assume \(E=G=\lambda^2\) and \(F=0\). Explain angle preservation relative to the parameter plane.
\end{problem}
\begin{solution}
The first form is \(\lambda^2(du^2+dv^2)\), a scalar multiple of the Euclidean metric, so all angles are unchanged although lengths are scaled by \(\lambda\).
\end{solution}

\begin{problem}[Normal curvature dossier]\label{prob:vii15-12}
Why does \(II(v,v)/I(v,v)\) equal the surface-normal component of curvature for a unit-speed surface curve?
\end{problem}
\begin{solution}
For unit speed, \(I(v,v)=1\) and the acceleration is the curvature vector. Its normal component is \(\langle\alpha'',N\rangle=II(v,v)\).
\end{solution}

\section{Challenge problems}
\begin{problem}[Tensor transformation from scratch]\label{prob:vii15-ch01}
Derive the coordinate-change formulas for both fundamental forms directly from \(\widetilde X=X\circ\phi\).
\end{problem}
\begin{solution}
First derivatives satisfy \(\widetilde X_a=X_iJ^i{}_a\), immediately giving \(\widetilde G=J^TGJ\). Second derivatives contain an additional term \(X_i\partial_a\partial_b\phi^i\), but this is tangent and vanishes when paired with \(N\); therefore \(\widetilde B=J^TBJ\).
\end{solution}

\begin{problem}[Developability preview]\label{prob:vii15-ch02}
Use the plane--cylinder example to explain how a surface can bend without changing its first fundamental form.
\end{problem}
\begin{solution}
Rolling the plane into a cylinder preserves the metric coefficients after using arc-length angular coordinate, so lengths and angles are unchanged. The second form becomes nonzero in one direction, showing extrinsic bending without intrinsic stretching.
\end{solution}

\begin{problem}[Normal section principle]\label{prob:vii15-ch03}
Let a plane through the surface normal and a tangent direction cut the surface in a regular curve. Show its signed curvature at the point equals the normal curvature of that tangent direction.
\end{problem}
\begin{solution}
The section curve lies in a plane containing \(N\), so its curvature vector at the point has no component orthogonal to that plane. The component along \(N\) is therefore its signed curvature, while the normal-acceleration formula identifies the same component with \(II(v,v)/I(v,v)\).
\end{solution}

\begin{problem}[Recover area from the metric]\label{prob:vii15-ch04}
Prove that the local area density is determined entirely by the first fundamental form, independent of the embedding coordinates.
\end{problem}
\begin{solution}
In any basis the area-square of the tangent parallelogram is the determinant of its Gram matrix. Under coordinate change the determinant acquires \((\det J)^2\), exactly balanced by the coordinate area transformation, so \(\sqrt{\det G}\,du\,dv\) is intrinsic.
\end{solution}

\begin{problem}[First versus second form data]\label{prob:vii15-ch05}
Give a precise argument that equal first fundamental forms do not determine the embedding up to rigid motion.
\end{problem}
\begin{solution}
The plane and cylinder provide explicit locally isometric patches with the same pulled-back first form. Their second forms differ, and a rigid motion would preserve both Euclidean first and second-order embedding data. Hence first-form equality alone does not imply congruence.
\end{solution}

\section*{Bridge to VII/16}
The second fundamental form records signed normal bending but has not yet been organized as a linear operator.  VII/16 introduces the Gauss map and shape operator, turning the derivative of the unit normal into the linear-algebraic mechanism behind the second fundamental form.
