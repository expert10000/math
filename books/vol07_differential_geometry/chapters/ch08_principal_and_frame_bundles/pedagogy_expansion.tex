\section{Supplementary worked examples}
\label{sec:vii08-pedagogy}

\begin{example}[Frames of a vector space]\label{ex:vii08-ped-01}
The frame bundle of the trivial rank-\(n\) bundle over \(M\) is \(M\times GL(n,\mathbb R)\).  A frame is an ordered basis, and changing a frame on the right by \(A\in GL(n)\) gives another basis.  This free transitive right action on each fiber is the prototype of a principal bundle.
\end{example}

\begin{example}[Oriented orthonormal frames]\label{ex:vii08-ped-02}
On an oriented Riemannian \(n\)-manifold, the oriented orthonormal frame bundle has structure group \(SO(n)\).  Passing from \(GL(n)\) to \(SO(n)\) records both metric orthonormality and orientation as a reduction of structure group.
\end{example}

\begin{example}[Associated vector bundle]\label{ex:vii08-ped-03}
Given a principal \(G\)-bundle \(P\to M\) and a representation \(\rho:G\to GL(V)\), the quotient
\[
P\times_G V=(P\times V)/((p,v)\sim(pg,\rho(g)^{-1}v))
\]
is a vector bundle.  Principal bundles therefore package the coordinate-change data from which many geometric bundles are built.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Right action on frames]\label{exr:vii08-ped-01}
If \(u:\mathbb R^n\to E_p\) is a frame and \(A\in GL(n)\), write the right action \(u\cdot A\).
\end{exercise}
\begin{hint}
Compose linear maps.
\end{hint}
\begin{solution}
\(u\cdot A=u\circ A\), which is again an isomorphism \(\mathbb R^n\to E_p\).
\end{solution}

\begin{exercise}[Frame bundle fiber]\label{exr:vii08-ped-02}
Identify the fiber of the frame bundle of a rank-\(n\) vector bundle.
\end{exercise}
\begin{hint}
List all ordered bases of one fiber.
\end{hint}
\begin{solution}
It is the set of linear isomorphisms \(\mathbb R^n\to E_p\), a \(GL(n,\mathbb R)\)-torsor.
\end{solution}

\begin{exercise}[Trivial principal bundle]\label{exr:vii08-ped-03}
Describe the principal \(G\)-bundle \(M\times G\to M\).
\end{exercise}
\begin{hint}
Use right multiplication in the second factor.
\end{hint}
\begin{solution}
The action is \((p,h)\cdot g=(p,hg)\); each fiber is freely and transitively acted on by \(G\).
\end{solution}

\begin{exercise}[Associated line bundle]\label{exr:vii08-ped-04}
For \(G=\{\pm1\}\) acting on \(\mathbb R\) by sign, describe the associated bundle to a double cover \(P\to M\).
\end{exercise}
\begin{hint}
Form \(P\times_G\mathbb R\).
\end{hint}
\begin{solution}
Points \((p,v)\) and \((p\cdot(-1),-v)\) are identified, producing the line bundle determined by the sign representation.
\end{solution}

\begin{exercise}[Gauge transformation locally]\label{exr:vii08-ped-05}
In a trivial principal \(G\)-bundle, what is the local form of a gauge transformation covering the identity?
\end{exercise}
\begin{hint}
It must preserve base points and commute with the right action.
\end{hint}
\begin{solution}
It has the form \((p,h)\mapsto(p,a(p)h)\) for a smooth map \(a:M\to G\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Frame action is free]\label{exr:vii08-ped-06}
Prove the right \(GL(n)\)-action on frames is free.
\end{exercise}
\begin{hint}
If \(u\circ A=u\), use injectivity of \(u\).
\end{hint}
\begin{solution}
Composing with \(u^{-1}\) gives \(A=I\), so only the identity fixes a frame.
\end{solution}

\begin{exercise}[Frame action is transitive]\label{exr:vii08-ped-07}
Prove the action is transitive on each frame-bundle fiber.
\end{exercise}
\begin{hint}
Given frames \(u,v\), solve \(u\circ A=v\).
\end{hint}
\begin{solution}
Take \(A=u^{-1}v\in GL(n)\), so \(u\cdot A=v\).
\end{solution}

\begin{exercise}[Associated-bundle well-defined fibers]\label{exr:vii08-ped-08}
Prove \(P\times_GV\) has fibers naturally modeled on \(V\).
\end{exercise}
\begin{hint}
Choose one point of the principal fiber as a frame.
\end{hint}
\begin{solution}
Fixing \(p\in P_x\), every class has a unique representative \([p,v]\), giving a linear identification with \(V\); changing \(p\) changes coordinates by \(\rho(g)\).
\end{solution}

\begin{exercise}[Section and triviality of a principal bundle]\label{exr:vii08-ped-09}
Show that a principal \(G\)-bundle with a global smooth section is trivial.
\end{exercise}
\begin{hint}
Use \((x,g)\mapsto s(x)g\).
\end{hint}
\begin{solution}
The map is smooth, \(G\)-equivariant, and fiberwise bijective; local trivializations show its inverse is smooth.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Principal fibers are vector spaces]\label{exr:vii08-ped-10}
Test this claim.
\end{exercise}
\begin{hint}
A principal fiber has a free transitive group action but no preferred origin.
\end{hint}
\begin{solution}
False.  A principal fiber is a \(G\)-torsor; without additional structure it is not a vector space.
\end{solution}

\begin{exercise}[Every frame bundle has a global section]\label{exr:vii08-ped-11}
Test the claim for \(TS^2\).
\end{exercise}
\begin{hint}
A global frame would give two everywhere independent tangent vector fields.
\end{hint}
\begin{solution}
False.  In particular a global frame would include a nowhere-zero tangent field, contradicting the hairy-ball theorem.
\end{solution}

\begin{exercise}[Left and right actions are interchangeable]\label{exr:vii08-ped-12}
Test this without changing formulas.
\end{exercise}
\begin{hint}
Check the cocycle and equivariance conventions.
\end{hint}
\begin{solution}
False as a literal statement: switching sides changes inverses and multiplication order.  The theories are equivalent only after systematically converting conventions.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Orthonormal frames]\label{exr:vii08-ped-13}
Explain how a Riemannian metric reduces the frame-bundle structure group from \(GL(n)\) to \(O(n)\).
\end{exercise}
\begin{hint}
Keep only frames orthonormal for the metric.
\end{hint}
\begin{solution}
Any two orthonormal frames differ by an orthogonal matrix, so the orthonormal-frame subbundle has group \(O(n)\).
\end{solution}

\begin{exercise}[Orientation reduction]\label{exr:vii08-ped-14}
Explain how an orientation further reduces \(O(n)\) to \(SO(n)\).
\end{exercise}
\begin{hint}
Keep only positively oriented orthonormal frames.
\end{hint}
\begin{solution}
Transition matrices between such frames are orthogonal with determinant \(+1\), hence lie in \(SO(n)\).
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Recover a vector bundle]\label{exr:vii08-ped-15}
Show that the vector bundle \(E\) is naturally isomorphic to the bundle associated to its frame bundle via the standard representation of \(GL(n)\).
\end{exercise}
\begin{hint}
Send \([u,v]\) to \(u(v)\).
\end{hint}
\begin{solution}
The map is well defined because \((uA,A^{-1}v)\) has the same image \(u(v)\); it is smooth and fiberwise linear bijective.
\end{solution}

\begin{exercise}[Gauge group of a trivial bundle]\label{exr:vii08-ped-16}
Identify the gauge group of \(M\times G\) with smooth maps \(M\to G\).
\end{exercise}
\begin{hint}
Use equivariance to determine a gauge transformation from its value at \((p,e)\).
\end{hint}
\begin{solution}
Every gauge map is \((p,h)\mapsto(p,a(p)h)\), and composition corresponds pointwise to multiplication of the smooth maps \(a\).
\end{solution}
