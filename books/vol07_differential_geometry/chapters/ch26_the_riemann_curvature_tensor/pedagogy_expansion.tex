\section{Supplementary worked examples}
\label{sec:vii26-pedagogy}

\begin{example}[Flat space]\label{ex:vii26-ped-01}
For the Euclidean connection all Cartesian Christoffel symbols and their derivatives vanish, hence \(R=0\).  Any nonzero coefficients in another coordinate system cancel in the full curvature expression.
\end{example}

\begin{example}[Constant sectional curvature]\label{ex:vii26-ped-02}
If every sectional curvature equals \(c\), then \(\operatorname{Rm}=\tfrac c2\,g\owedge g\) with this volume's Kulkarni--Nomizu convention.  The sphere has \(c>0\), Euclidean space \(c=0\), and hyperbolic space \(c<0\).
\end{example}

\begin{example}[Infinitesimal holonomy]\label{ex:vii26-ped-03}
For a tiny coordinate parallelogram spanned by \(X\) and \(Y\), transporting \(Z\) around its boundary changes it to first nontrivial order by an area-scaled multiple of \(R(X,Y)Z\).  Curvature is therefore the infinitesimal commutator of covariant derivatives.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Skew symmetry]\label{exr:vii26-ped-01}
What is \(R(Y,X)Z\) in terms of \(R(X,Y)Z\)?
\end{exercise}

\begin{hint}
Swap the first two entries.
\end{hint}

\begin{solution}
\(R(Y,X)Z=-R(X,Y)Z\).
\end{solution}

\begin{exercise}[Flat components]\label{exr:vii26-ped-02}
Compute \(R^\ell{}_{kij}\) when all Christoffel symbols vanish on a coordinate neighborhood.
\end{exercise}

\begin{hint}
Use the coordinate formula.
\end{hint}

\begin{solution}
All components vanish.
\end{solution}

\begin{exercise}[Sectional curvature]\label{exr:vii26-ped-03}
For orthonormal \(u,v\), how is \(K(u,v)\) read from \(\operatorname{Rm}\)?
\end{exercise}

\begin{hint}
Use the chosen sign convention.
\end{hint}

\begin{solution}
\(K(u,v)=\operatorname{Rm}(u,v,v,u)\) for the convention in this volume.
\end{solution}

\begin{exercise}[Sphere curvature]\label{exr:vii26-ped-04}
What is the sectional curvature of a round sphere of radius \(a\)?
\end{exercise}

\begin{hint}
Scale the unit-sphere result.
\end{hint}

\begin{solution}
\(K=1/a^2\).
\end{solution}

\begin{exercise}[Hyperbolic curvature]\label{exr:vii26-ped-05}
What is the sectional curvature of hyperbolic space with curvature radius \(a\)?
\end{exercise}

\begin{hint}
Use the standard scaling.
\end{hint}

\begin{solution}
\(K=-1/a^2\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Tensoriality]\label{exr:vii26-ped-06}
Why is the Lie-bracket term needed in \(R(X,Y)Z\)?
\end{exercise}

\begin{hint}
Track derivatives created by multiplying vector fields by functions.
\end{hint}

\begin{solution}
It cancels non-tensorial derivative terms so curvature is \(C^\infty\)-linear in its inputs.
\end{solution}

\begin{exercise}[Bianchi]\label{exr:vii26-ped-07}
State the first Bianchi identity.
\end{exercise}

\begin{hint}
Cyclically permute \(X,Y,Z\).
\end{hint}

\begin{solution}
\(R(X,Y)Z+R(Y,Z)X+R(Z,X)Y=0\).
\end{solution}

\begin{exercise}[Normal coordinates]\label{exr:vii26-ped-08}
Why can curvature be nonzero at a point where all Christoffel symbols vanish?
\end{exercise}

\begin{hint}
Look at derivatives of the symbols.
\end{hint}

\begin{solution}
The coordinate curvature formula still contains first derivatives of Christoffel symbols at that point.
\end{solution}

\begin{exercise}[Pair symmetry]\label{exr:vii26-ped-09}
State the pair symmetry of the fully covariant Riemann tensor.
\end{exercise}

\begin{hint}
Exchange the first pair with the second pair.
\end{hint}

\begin{solution}
\(\operatorname{Rm}(X,Y,Z,W)=\operatorname{Rm}(Z,W,X,Y)\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Curvature from symbols]\label{exr:vii26-ped-10}
Test the claim: curvature is just the derivative of Christoffel symbols.
\end{exercise}

\begin{hint}
Inspect the full formula.
\end{hint}

\begin{solution}
False.  Quadratic \(\Gamma\Gamma\) terms are also required away from normal coordinates.
\end{solution}

\begin{exercise}[One plane determines Riemann]\label{exr:vii26-ped-11}
Test the claim: one sectional curvature determines the full Riemann tensor in dimension at least three.
\end{exercise}

\begin{hint}
There are many two-planes.
\end{hint}

\begin{solution}
False.  The full tensor encodes curvature across all tangent two-planes, subject to algebraic symmetries.
\end{solution}

\begin{exercise}[Sign-independent formulas]\label{exr:vii26-ped-12}
Test the claim: curvature signs can be compared across books without checking conventions.
\end{exercise}

\begin{hint}
Different authors reverse the definition of \(R\).
\end{hint}

\begin{solution}
False.  Reversing the Riemann operator convention reverses sectional, Ricci, and scalar curvature signs.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Tidal interpretation]\label{exr:vii26-ped-13}
Why is curvature relevant to relative acceleration of nearby geodesics?
\end{exercise}

\begin{hint}
Recall the Jacobi equation qualitatively.
\end{hint}

\begin{solution}
The Riemann tensor couples the separation vector to the geodesic tangent, governing how initially nearby geodesics focus or diverge.
\end{solution}

\begin{exercise}[Small-loop computation]\label{exr:vii26-ped-14}
What data would a numerical small-loop holonomy experiment approximate?
\end{exercise}

\begin{hint}
Measure the linear map after one loop and divide by enclosed area.
\end{hint}

\begin{solution}
It approximates the curvature operator on the plane spanned by the loop directions.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Space-form tensor]\label{exr:vii26-ped-15}
For orthonormal \(u,v\) in constant curvature \(c\), verify \(\operatorname{Rm}(u,v,v,u)=c\).
\end{exercise}

\begin{hint}
Insert the constant-curvature formula.
\end{hint}

\begin{solution}
The Kulkarni--Nomizu expression evaluates to \(c\) on an orthonormal pair.
\end{solution}

\begin{exercise}[Zero sectional challenge]\label{exr:vii26-ped-16}
If every sectional curvature at a point is zero, what can you conclude about the Riemann tensor there?
\end{exercise}

\begin{hint}
Sectional curvatures determine an algebraic curvature tensor by polarization.
\end{hint}

\begin{solution}
The full Riemann tensor vanishes at that point.
\end{solution}
