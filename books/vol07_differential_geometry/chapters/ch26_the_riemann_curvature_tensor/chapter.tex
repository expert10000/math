\chapter{The Riemann Curvature Tensor}
\label{ch:vii26-riemann-curvature}

Holonomy records the finite failure of parallel transport to be path independent.
Curvature is the infinitesimal version of that failure.  For the Levi--Civita
connection we use the convention
\[
\boxed{
R(X,Y)Z
=
\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
}
\]
With this convention the round sphere has positive sectional curvature.

\section{Definition and tensoriality}

\begin{definition}[Curvature operator]\label{def:vii26-curvature-operator}
For vector fields \(X,Y,Z\),
\[
R(X,Y)Z
=
\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
\]
\end{definition}

\begin{theorem}[Tensoriality]\label{thm:vii26-tensoriality}
The map \(R(X,Y)Z\) is \(C^\infty(M)\)-linear in each of \(X,Y,Z\). Hence
\[
R\in\Gamma\!\left(\Lambda^2T^*M\otimes\operatorname{End}(TM)\right).
\]
\end{theorem}

\begin{proof}
Linearity in \(X\) and \(Y\) follows after the derivative terms created by
multiplication by a function cancel against the Lie-bracket term.  Linearity in
\(Z\) follows because the second-derivative terms in a function cancel between the
two commutators.  Therefore the value at a point depends only on the tangent
vectors at that point.
\end{proof}

\begin{proposition}[Skew symmetry in the first two entries]\label{prop:vii26-skew12}
\[
R(X,Y)=-R(Y,X).
\]
\end{proposition}

\section{The \((0,4)\) curvature tensor}

\begin{definition}[Riemann tensor]\label{def:vii26-rm4}
Define
\[
\operatorname{Rm}(X,Y,Z,W)
=
g(R(X,Y)Z,W).
\]
\end{definition}

For the Levi--Civita connection this tensor satisfies strong algebraic symmetries.

\begin{theorem}[Riemann symmetries]\label{thm:vii26-symmetries}
For all tangent vectors,
\[
\operatorname{Rm}(X,Y,Z,W)
=
-\operatorname{Rm}(Y,X,Z,W),
\]
\[
\operatorname{Rm}(X,Y,Z,W)
=
-\operatorname{Rm}(X,Y,W,Z),
\]
and
\[
\operatorname{Rm}(X,Y,Z,W)
=
\operatorname{Rm}(Z,W,X,Y).
\]
\end{theorem}

\begin{theorem}[First Bianchi identity]\label{thm:vii26-bianchi1}
\[
R(X,Y)Z+R(Y,Z)X+R(Z,X)Y=0.
\]
Equivalently,
\[
\operatorname{Rm}(X,Y,Z,W)
+
\operatorname{Rm}(Y,Z,X,W)
+
\operatorname{Rm}(Z,X,Y,W)=0.
\]
\end{theorem}

\begin{proof}
Expand the definition of \(R\), sum cyclically, and use torsion freeness
\(\nabla_XY-\nabla_YX=[X,Y]\) together with the Jacobi identity for Lie brackets.
\end{proof}

\section{Coordinate components}

Let
\[
R(\partial_i,\partial_j)\partial_k
=
R^\ell{}_{kij}\partial_\ell.
\]

\begin{theorem}[Coordinate formula]\label{thm:vii26-components}
In a coordinate chart,
\[
R^\ell{}_{kij}
=
\partial_i\Gamma^\ell_{jk}
-\partial_j\Gamma^\ell_{ik}
+\Gamma^m_{jk}\Gamma^\ell_{im}
-\Gamma^m_{ik}\Gamma^\ell_{jm}.
\]
\end{theorem}

Lowering the first index gives
\[
R_{\ell kij}
=
g_{\ell m}R^m{}_{kij}.
\]

\begin{warning}[Index conventions]\label{warn:vii26-index}
Different books permute the positions of \(i,j,k,\ell\) and may define \(R\) with
the opposite overall sign.  Always check the operator convention before comparing
component formulas or the sign of sectional curvature.
\end{warning}

\section{Normal coordinates}

At a point \(p\), choose normal coordinates:
\[
g_{ij}(p)=\delta_{ij},
\qquad
\Gamma^k_{ij}(p)=0.
\]

\begin{proposition}[Curvature survives normal coordinates]\label{prop:vii26-normal}
At the center of normal coordinates,
\[
R^\ell{}_{kij}(p)
=
\partial_i\Gamma^\ell_{jk}(p)
-
\partial_j\Gamma^\ell_{ik}(p).
\]
Thus one can kill the connection coefficients at a point, but not generally their
first derivatives.
\end{proposition}

\section{Sectional curvature}

\begin{definition}[Sectional curvature]\label{def:vii26-sectional}
Let \(\sigma\subset T_pM\) be a two-plane spanned by linearly independent
\(u,v\). Its sectional curvature is
\[
K(\sigma)
=
\frac{g(R(u,v)v,u)}
{g(u,u)g(v,v)-g(u,v)^2}.
\]
For an orthonormal pair \(u,v\),
\[
K(\sigma)=g(R(u,v)v,u).
\]
\end{definition}

\begin{proposition}[Well-definedness]\label{prop:vii26-sectional-well}
\(K(\sigma)\) depends only on the two-plane \(\sigma\), not on the chosen basis.
\end{proposition}

\begin{remark}
In dimension two there is only one tangent two-plane at each point.  Its sectional
curvature is exactly the Gaussian curvature of the Riemannian surface.
\end{remark}

\section{Constant sectional curvature}

\begin{definition}[Space form curvature]\label{def:vii26-space-form}
A Riemannian manifold has constant sectional curvature \(c\) if every tangent
two-plane has sectional curvature \(c\).
\end{definition}

\begin{theorem}[Curvature tensor of a space form]\label{thm:vii26-space-form-r}
With the convention of this chapter, constant sectional curvature \(c\) is
equivalent to
\[
R(X,Y)Z
=
c\bigl(g(Y,Z)X-g(X,Z)Y\bigr).
\]
Equivalently,
\[
\operatorname{Rm}(X,Y,Z,W)
=
c\bigl(g(Y,Z)g(X,W)-g(X,Z)g(Y,W)\bigr).
\]
\end{theorem}

\begin{example}[Euclidean space]\label{ex:vii26-euclidean}
For \(\mathbb R^n\) with its Euclidean metric,
\[
R=0,
\qquad
K=0.
\]
\end{example}

\begin{example}[Round sphere]\label{ex:vii26-sphere}
The round sphere of radius \(a\) has
\[
K=\frac1{a^2}.
\]
Hence
\[
R(X,Y)Z
=
\frac1{a^2}
\bigl(g(Y,Z)X-g(X,Z)Y\bigr).
\]
\end{example}

\begin{example}[Hyperbolic model preview]\label{ex:vii26-hyperbolic}
Hyperbolic space has constant negative sectional curvature.  Concrete models are
developed in VII/31--VII/35.
\end{example}

\section{Curvature and commutators of covariant derivatives}

The defining equation may be read as
\[
[\nabla_X,\nabla_Y]Z
=
\nabla_{[X,Y]}Z+R(X,Y)Z.
\]
Thus curvature is exactly the correction term preventing covariant derivatives
from commuting as though the connection were globally flat.

\section{Second Bianchi identity}

\begin{theorem}[Second Bianchi identity]\label{thm:vii26-bianchi2}
For the Levi--Civita connection,
\[
(\nabla_XR)(Y,Z)
+
(\nabla_YR)(Z,X)
+
(\nabla_ZR)(X,Y)
=
0.
\]
\end{theorem}

\begin{remark}
Contracting this differential identity produces the contracted Bianchi identity,
which becomes central after Ricci and scalar curvature are introduced in VII/27.
\end{remark}

\section{Curvature and holonomy}

For a small loop spanning tangent directions \(X,Y\), the leading departure of
parallel transport from the identity is governed by \(R(X,Y)\).  Thus VII/25 and
VII/26 are two scales of the same phenomenon:
\[
\text{finite loop transport}
\longleftrightarrow
\text{infinitesimal curvature}.
\]

\section{Solved problem dossiers}
\begin{problem}[Tensoriality problem]\label{prob:vii26-01}
Prove \(R(X,Y)Z\) is \(C^\infty\)-linear in all three vector fields.
\end{problem}
\begin{solution}
Expand after multiplying each entry by a function. Leibniz terms cancel between the covariant-derivative commutator and the Lie-bracket correction.
\end{solution}

\begin{problem}[First-Bianchi problem]\label{prob:vii26-02}
Derive the first Bianchi identity for a torsion-free connection.
\end{problem}
\begin{solution}
Expand the cyclic sum and use \(\nabla_XY-\nabla_YX=[X,Y]\) together with the Jacobi identity.
\end{solution}

\begin{problem}[Component problem]\label{prob:vii26-03}
Derive the coordinate formula for \(R^\ell{}_{kij}\).
\end{problem}
\begin{solution}
Apply the definition to coordinate vector fields, use \([\partial_i,\partial_j]=0\), and expand two covariant derivatives of \(\partial_k\).
\end{solution}

\begin{problem}[Normal-coordinate problem]\label{prob:vii26-04}
Explain what normal coordinates reveal about curvature.
\end{problem}
\begin{solution}
At the center \(\Gamma=0\), so curvature is built directly from first derivatives of connection coefficients, showing why curvature is second-order metric information.
\end{solution}

\begin{problem}[Sectional-curvature problem]\label{prob:vii26-05}
Show the sectional-curvature quotient is invariant under changing the basis of the same plane.
\end{problem}
\begin{solution}
Both numerator and denominator transform by the square of the determinant of the \(2\times2\) basis-change matrix; the quotient is unchanged.
\end{solution}

\begin{problem}[Surface problem]\label{prob:vii26-06}
Show sectional curvature specializes to Gaussian curvature in dimension two.
\end{problem}
\begin{solution}
The unique tangent two-plane has orthonormal basis \(e_1,e_2\), and \(K=g(R(e_1,e_2)e_2,e_1)\), the intrinsic Gaussian curvature.
\end{solution}

\begin{problem}[Space-form problem]\label{prob:vii26-07}
Verify the constant-curvature tensor has sectional curvature \(c\).
\end{problem}
\begin{solution}
For orthonormal \(u,v\), the formula gives \(R(u,v)v=c\,u\), so \(g(R(u,v)v,u)=c\).
\end{solution}

\begin{problem}[Sphere problem]\label{prob:vii26-08}
Determine the Riemann tensor of a round sphere of radius \(a\).
\end{problem}
\begin{solution}
The sphere is a space form with \(c=1/a^2\), so substitute this value in the space-form formula.
\end{solution}

\begin{problem}[Flatness problem]\label{prob:vii26-09}
Why does \(R=0\) imply local path independence on a simply connected sufficiently small domain?
\end{problem}
\begin{solution}
A flat connection satisfies the integrability condition for a local parallel frame. In such a frame the connection form vanishes and transport depends only on endpoints.
\end{solution}

\begin{problem}[Holonomy problem]\label{prob:vii26-10}
Relate the sign convention for small-loop transport to the sign convention for \(R\).
\end{problem}
\begin{solution}
The area-order term may be written \(I\pm\varepsilon^2R(X,Y)\) depending on loop orientation and curvature convention. Fixing both removes the ambiguity.
\end{solution}

\begin{problem}[Second-Bianchi problem]\label{prob:vii26-11}
Explain why the second Bianchi identity is a differential analogue of Jacobi.
\end{problem}
\begin{solution}
Curvature is a commutator of covariant derivatives corrected by Lie brackets; cyclically differentiating and applying commutator/Jacobi relations yields the identity.
\end{solution}

\begin{problem}[Dimension-count problem]\label{prob:vii26-12}
How many independent components can an algebraic Riemann tensor have in dimension \(n\)?
\end{problem}
\begin{solution}
Using the Riemann symmetries and Bianchi identity gives \(n^2(n^2-1)/12\) independent components.
\end{solution}

\section{Challenge problems}
\begin{problem}[Independent-component count]\label{prob:vii26-ch01}
Derive \(n^2(n^2-1)/12\) for the dimension of the space of algebraic curvature tensors.
\end{problem}
\begin{solution}
View \(\operatorname{Rm}\) as a symmetric bilinear form on \(\Lambda^2V\), then impose the first Bianchi identity. Starting with \(m(m+1)/2\), \(m=n(n-1)/2\), subtract the \(\binom n4\) Bianchi constraints and simplify.
\end{solution}

\begin{problem}[Curvature from sectional curvature]\label{prob:vii26-ch02}
Explain why knowledge of all sectional curvatures determines the full Riemann tensor.
\end{problem}
\begin{solution}
The values \(g(R(u,v)v,u)\) determine the curvature quadratic form on decomposable bivectors. Polarization, together with the Riemann symmetries and Bianchi identity, recovers the full algebraic tensor.
\end{solution}

\begin{problem}[Normal-coordinate expansion preview]\label{prob:vii26-ch03}
What kind of term should appear at second order in the Taylor expansion of \(g_{ij}\) in normal coordinates?
\end{problem}
\begin{solution}
Since \(g_{ij}(p)=\delta_{ij}\) and first derivatives vanish, the quadratic term must be a universal contraction of the Riemann tensor with \(x^kx^\ell\); its precise coefficient depends on the component convention.
\end{solution}

\begin{problem}[Jacobi-field preview]\label{prob:vii26-ch04}
Why should \(R(J,\dot\gamma)\dot\gamma\) appear in the equation for infinitesimal separation of nearby geodesics?
\end{problem}
\begin{solution}
Commuting covariant derivatives of a two-parameter family of geodesics produces curvature. The variation field therefore satisfies a second-order equation whose restoring/focusing term is the curvature operator.
\end{solution}

\begin{problem}[Curvature operator on bivectors]\label{prob:vii26-ch05}
Describe how \(\operatorname{Rm}\) can be viewed as a self-adjoint operator on \(\Lambda^2T_pM\).
\end{problem}
\begin{solution}
The pair symmetry makes the bilinear form on bivectors symmetric. Using the metric-induced inner product on \(\Lambda^2T_pM\), Riesz representation yields a self-adjoint curvature operator.
\end{solution}
