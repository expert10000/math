\section{Supplementary worked examples}
\label{sec:vii03-pedagogy}

\begin{example}[Smoothness of the height function on a sphere]\label{ex:vii03-ped-01}
For \(S^2\subset\mathbb R^3\), the height map \(h(x,y,z)=z\) is smooth because it is the restriction of a smooth Euclidean function.  In a local parametrization \(X(u,v)\), its coordinate representative is simply the third component of \(X\), which is smooth.
\end{example}

\begin{example}[A global diffeomorphism from a local formula]\label{ex:vii03-ped-02}
The map \(\log:(0,\infty)\to\mathbb R\) is smooth with smooth inverse \(\exp\).  This is the simplest example showing that a diffeomorphism requires both global bijectivity and smooth inverse, not merely a nonzero derivative.
\end{example}

\begin{example}[A smooth map with critical points]\label{ex:vii03-ped-03}
The projection \(h:S^2\to\mathbb R\), \(h(x,y,z)=z\), has differential of rank one away from the poles and rank zero at the poles.  Smoothness does not require constant rank; rank becomes relevant only for stronger local normal-form statements.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Chart representative]\label{exr:vii03-ped-01}
For \(f:S^1\to\mathbb R\), \(f(x,y)=x\), write its representative in the upper semicircle chart \(x\mapsto(x,\sqrt{1-x^2})\).
\end{exercise}
\begin{hint}
Compose \(f\) with the chart inverse.
\end{hint}
\begin{solution}
The representative is \(x\mapsto x\), hence smooth.
\end{solution}

\begin{exercise}[Exponential diffeomorphism]\label{exr:vii03-ped-02}
Verify directly that \(\exp:\mathbb R\to(0,\infty)\) is a diffeomorphism.
\end{exercise}
\begin{hint}
Identify its inverse.
\end{hint}
\begin{solution}
It is bijective, smooth, and has smooth inverse \(\log\).
\end{solution}

\begin{exercise}[Circle power map]\label{exr:vii03-ped-03}
For \(F:S^1\to S^1\), \(F(z)=z^k\), determine whether \(F\) is a diffeomorphism.
\end{exercise}
\begin{hint}
Check bijectivity as well as smoothness.
\end{hint}
\begin{solution}
It is smooth for every integer \(k\); it is a diffeomorphism exactly for \(k=\pm1\).
\end{solution}

\begin{exercise}[Smooth product map]\label{exr:vii03-ped-04}
If \(f:M\to P\) and \(g:N\to Q\) are smooth, write the product map and its chart form.
\end{exercise}
\begin{hint}
Use product charts.
\end{hint}
\begin{solution}
\((f\times g)(p,q)=(f(p),g(q))\), and in product charts its representative is the product of the two Euclidean representatives, hence smooth.
\end{solution}

\begin{exercise}[Restriction]\label{exr:vii03-ped-05}
Show that the restriction of a smooth map \(F:\mathbb R^2\to\mathbb R\) to \(S^1\) is smooth.
\end{exercise}
\begin{hint}
Use local parametrizations of \(S^1\).
\end{hint}
\begin{solution}
In any circle chart \(X\), the representative is \(F\circ X\), a composition of smooth Euclidean maps.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Composition theorem]\label{exr:vii03-ped-06}
Prove that the composition of smooth maps between smooth manifolds is smooth.
\end{exercise}
\begin{hint}
Write the coordinate representative of the composition.
\end{hint}
\begin{solution}
In compatible charts it is the composition of the two smooth Euclidean representatives, hence smooth.
\end{solution}

\begin{exercise}[Locality of smoothness]\label{exr:vii03-ped-07}
Prove that a map is smooth if it is smooth on an open cover of its domain.
\end{exercise}
\begin{hint}
Every point lies in one member of the cover.
\end{hint}
\begin{solution}
Smoothness is defined locally in charts, so a chart neighborhood can be chosen inside a covering set where the map is already smooth.
\end{solution}

\begin{exercise}[Diffeomorphisms preserve dimension]\label{exr:vii03-ped-08}
Prove that diffeomorphic connected manifolds have the same dimension.
\end{exercise}
\begin{hint}
Use chart representatives and the derivative of a local diffeomorphism.
\end{hint}
\begin{solution}
A diffeomorphism gives local diffeomorphisms between Euclidean opens; their derivatives are linear isomorphisms, forcing equal dimensions.
\end{solution}

\begin{exercise}[Smooth inverse criterion in one dimension]\label{exr:vii03-ped-09}
Let \(f:I\to J\) be a smooth bijection between intervals with \(f'\neq0\).  Prove that \(f^{-1}\) is smooth.
\end{exercise}
\begin{hint}
Apply the inverse function theorem at each point.
\end{hint}
\begin{solution}
The inverse function theorem gives a smooth local inverse everywhere; bijectivity makes these local inverses agree to form the global inverse.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Nonzero derivative implies global diffeomorphism]\label{exr:vii03-ped-10}
Test the claim for \(f:\mathbb R\to S^1\), \(f(t)=e^{it}\).
\end{exercise}
\begin{hint}
Check injectivity.
\end{hint}
\begin{solution}
False.  The map is a local diffeomorphism but is \(2\pi\)-periodic, so it is not globally injective.
\end{solution}

\begin{exercise}[Smooth bijection has smooth inverse]\label{exr:vii03-ped-11}
Test the claim using \(f(x)=x^3\) on \(\mathbb R\).
\end{exercise}
\begin{hint}
Inspect the inverse at the origin.
\end{hint}
\begin{solution}
False.  The inverse \(x^{1/3}\) is not differentiable at \(0\).
\end{solution}

\begin{exercise}[Rank is constant for smooth maps]\label{exr:vii03-ped-12}
Test this claim with the sphere height function.
\end{exercise}
\begin{hint}
Compare poles with nonpolar points.
\end{hint}
\begin{solution}
False.  Its rank is zero at the poles and one elsewhere.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Change of coordinates]\label{exr:vii03-ped-13}
Explain why the notion of a smooth map does not depend on the chosen compatible charts.
\end{exercise}
\begin{hint}
Relate two coordinate representatives by transition maps.
\end{hint}
\begin{solution}
Changing charts pre- and post-composes the representative with smooth transition diffeomorphisms, preserving smoothness.
\end{solution}

\begin{exercise}[Smooth symmetry]\label{exr:vii03-ped-14}
Show that any orthogonal matrix \(A\in O(n+1)\) restricts to a diffeomorphism of \(S^n\).
\end{exercise}
\begin{hint}
Use \(A^{-1}=A^T\).
\end{hint}
\begin{solution}
The restriction is smooth, bijective, and its inverse is the smooth restriction of \(A^T\).
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Local versus global inverse]\label{exr:vii03-ped-15}
Construct a smooth local diffeomorphism that is not a global diffeomorphism.
\end{exercise}
\begin{hint}
Use a covering map.
\end{hint}
\begin{solution}
The exponential \(t\mapsto e^{it}\) from \(\mathbb R\) to \(S^1\) has everywhere nonzero derivative in local coordinates but is not injective.
\end{solution}

\begin{exercise}[Conjugating smooth maps]\label{exr:vii03-ped-16}
If \(F:M\to M\) is smooth and \(h:M\to N\) is a diffeomorphism, prove \(hFh^{-1}:N\to N\) is smooth.
\end{exercise}
\begin{hint}
Use closure of smooth maps under composition.
\end{hint}
\begin{solution}
It is a composition of the smooth maps \(h\), \(F\), and \(h^{-1}\), hence smooth.
\end{solution}
