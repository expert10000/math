\section{Supplementary worked examples}
\label{sec:vii09-pedagogy}

\begin{example}[Polar area form]\label{ex:vii09-ped-01}
On the punctured plane, \(x=r\cos\theta\), \(y=r\sin\theta\) gives
\[
dx\wedge dy=r\,dr\wedge d\theta.
\]
The factor \(r\) is the Jacobian expressed intrinsically through the wedge product.  Differential forms therefore carry change-of-variables information without separately inserting a determinant.
\end{example}

\begin{example}[Pullback of a one-form]\label{ex:vii09-ped-02}
For \(f:\mathbb R\to\mathbb R^2\), \(f(t)=(t,t^2)\), and \(\alpha=y\,dx+x\,dy\),
\[
f^*\alpha=t^2\,dt+t(2t\,dt)=3t^2\,dt.
\]
The pullback substitutes the parametrization and differentiates its coordinate functions.
\end{example}

\begin{example}[Closed versus exact]\label{ex:vii09-ped-03}
On \(\mathbb R^2\setminus\{0\}\),
\[
\omega=\frac{-y\,dx+x\,dy}{x^2+y^2}
\]
is closed.  Its integral around the unit circle is \(2\pi\), so it cannot be exact globally.  This is the first concrete signal that differential forms detect topology.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[A wedge product]\label{exr:vii09-ped-01}
Compute \((dx+dy)\wedge(dx-dy)\).
\end{exercise}
\begin{hint}
Use bilinearity and \(dx\wedge dx=dy\wedge dy=0\).
\end{hint}
\begin{solution}
The result is \(-2\,dx\wedge dy\).
\end{solution}

\begin{exercise}[Exterior derivative]\label{exr:vii09-ped-02}
Compute \(d(x^2y\,dx+e^x\,dy)\).
\end{exercise}
\begin{hint}
Apply \(d(f\,dx)=df\wedge dx\) and similarly for \(dy\).
\end{hint}
\begin{solution}
\(d(x^2y\,dx)=-x^2\,dx\wedge dy\) and \(d(e^xdy)=e^x\,dx\wedge dy\), so \(d\alpha=(e^x-x^2)\,dx\wedge dy\).
\end{solution}

\begin{exercise}[Pullback along a curve]\label{exr:vii09-ped-03}
For \(f(t)=(\cos t,\sin t)\), compute \(f^*(x\,dy-y\,dx)\).
\end{exercise}
\begin{hint}
Substitute \(x,y,dx,dy\).
\end{hint}
\begin{solution}
It is \((\cos^2t+\sin^2t)\,dt=dt\).
\end{solution}

\begin{exercise}[Polar area]\label{exr:vii09-ped-04}
Derive \(dx\wedge dy=r\,dr\wedge d\theta\).
\end{exercise}
\begin{hint}
Differentiate \(x=r\cos\theta\) and \(y=r\sin\theta\).
\end{hint}
\begin{solution}
Expanding the wedge product cancels the \(dr\wedge dr\) and \(d\theta\wedge d\theta\) terms and leaves \(r\,dr\wedge d\theta\).
\end{solution}

\begin{exercise}[Closed one-form]\label{exr:vii09-ped-05}
Determine whether \(\alpha=2x\,dx+2y\,dy\) is exact.
\end{exercise}
\begin{hint}
Look for a potential.
\end{hint}
\begin{solution}
\(\alpha=d(x^2+y^2)\), so it is exact and therefore closed.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Graded commutativity]\label{exr:vii09-ped-06}
Prove \(\alpha\wedge\beta=(-1)^{pq}\beta\wedge\alpha\) for a \(p\)-form and \(q\)-form.
\end{exercise}
\begin{hint}
Reduce to decomposable forms and count transpositions.
\end{hint}
\begin{solution}
Moving \(p\) one-forms past \(q\) one-forms requires \(pq\) swaps, each contributing a minus sign.
\end{solution}

\begin{exercise}[Leibniz rule for \(d\)]\label{exr:vii09-ped-07}
State and verify the graded Leibniz rule for \(d(\alpha\wedge\beta)\).
\end{exercise}
\begin{hint}
Track the degree \(p=\deg\alpha\).
\end{hint}
\begin{solution}
The rule is \(d(\alpha\wedge\beta)=d\alpha\wedge\beta+(-1)^p\alpha\wedge d\beta\); it follows in coordinates from the ordinary product rule and antisymmetry.
\end{solution}

\begin{exercise}[Naturality of exterior derivative]\label{exr:vii09-ped-08}
Prove \(d(f^*\omega)=f^*(d\omega)\).
\end{exercise}
\begin{hint}
Check first on functions and coordinate one-forms, then use products and sums.
\end{hint}
\begin{solution}
Both operations commute on generators \(g\) because \(d(g\circ f)=f^*dg\); the graded Leibniz rule extends the identity to all forms.
\end{solution}

\begin{exercise}[Exact implies closed]\label{exr:vii09-ped-09}
Prove every exact form is closed.
\end{exercise}
\begin{hint}
Use \(d^2=0\).
\end{hint}
\begin{solution}
If \(\omega=d\eta\), then \(d\omega=d^2\eta=0\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Wedge product is commutative]\label{exr:vii09-ped-10}
Test the claim with \(dx\) and \(dy\).
\end{exercise}
\begin{hint}
Swap the factors.
\end{hint}
\begin{solution}
False: \(dx\wedge dy=-dy\wedge dx\).
\end{solution}

\begin{exercise}[Closed implies exact globally]\label{exr:vii09-ped-11}
Test the claim on \(\mathbb R^2\setminus\{0\}\).
\end{exercise}
\begin{hint}
Use the angular one-form.
\end{hint}
\begin{solution}
False.  The angular form is closed but has nonzero integral around the unit circle, so it is not exact.
\end{solution}

\begin{exercise}[Pullback preserves degree and direction]\label{exr:vii09-ped-12}
Test whether a map \(f:M\to N\) pulls a form on \(M\) to a form on \(N\).
\end{exercise}
\begin{hint}
Remember the contravariant direction.
\end{hint}
\begin{solution}
False.  Pullback sends forms on \(N\) to forms of the same degree on \(M\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Line integral]\label{exr:vii09-ped-13}
Explain why integrating a one-form along a parametrized curve uses pullback.
\end{exercise}
\begin{hint}
Reduce the integral to the parameter interval.
\end{hint}
\begin{solution}
\(\int_\gamma\alpha\) is defined as \(\int_I\gamma^*\alpha\), an ordinary integral of a one-form on the interval.
\end{solution}

\begin{exercise}[Jacobian through forms]\label{exr:vii09-ped-14}
Explain how a change of variables in a double integral appears from pullback of \(dx\wedge dy\).
\end{exercise}
\begin{hint}
Pull back the area form under the coordinate map.
\end{hint}
\begin{solution}
The coefficient of the pulled-back top form is the Jacobian determinant, including its orientation sign.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Angular form]\label{exr:vii09-ped-15}
Compute the pullback of \(\omega=(-y\,dx+x\,dy)/(x^2+y^2)\) to the unit circle.
\end{exercise}
\begin{hint}
Use \(x=\cos t,\ y=\sin t\).
\end{hint}
\begin{solution}
The pullback is \(dt\), so its integral over one positively oriented turn is \(2\pi\).
\end{solution}

\begin{exercise}[Poincare lemma on a star-shaped domain]\label{exr:vii09-ped-16}
State the local conclusion suggested by the Poincare lemma for closed forms on a star-shaped Euclidean domain.
\end{exercise}
\begin{hint}
Distinguish local topology from the punctured-plane example.
\end{hint}
\begin{solution}
Every closed positive-degree form on a star-shaped domain is exact; the obstruction in the punctured plane is therefore global rather than local.
\end{solution}
