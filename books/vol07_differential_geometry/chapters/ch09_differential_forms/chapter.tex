\chapter{Differential Forms}
\label{ch:vii09-differential-forms}

Differential forms are alternating covariant tensors that vary smoothly from point to point. They combine the cotangent bundle of VII/05 with the vector-bundle language of VII/07 and add two decisive operations: the wedge product and the exterior derivative.
\[
\boxed{\Omega^k(M)=\Gamma(\Lambda^kT^*M),\qquad
d:\Omega^k(M)\to\Omega^{k+1}(M),\qquad d^2=0.}
\]

\section{Alternating covariant tensors}
\begin{definition}[Alternating \(k\)-form at a point]\label{def:vii09-point-form}
A differential \(k\)-form at \(p\in M\) is an alternating multilinear map
\[
\omega_p:(T_pM)^k\to\mathbb R.
\]
The vector space of such maps is \(\Lambda^kT_p^*M\). By convention \(\Lambda^0T_p^*M=\mathbb R\).
\end{definition}

\begin{definition}[Smooth differential form]\label{def:vii09-smooth-form}
A smooth differential \(k\)-form is a smooth section of \(\Lambda^kT^*M\). The space of these forms is denoted \(\Omega^k(M)\), and \(\Omega^0(M)=C^\infty(M)\).
\end{definition}

\begin{proposition}[Dimension]\label{prop:vii09-dim}
If \(\dim M=n\), then
\[
\dim\Lambda^kT_p^*M=\binom nk,
\]
and \(\Lambda^kT_p^*M=0\) for \(k>n\).
\end{proposition}

\section{Coordinate expressions}
In local coordinates \(x^1,\ldots,x^n\), every \(k\)-form is uniquely
\[
\omega=\sum_{i_1<\cdots<i_k}\omega_{i_1\cdots i_k}
\,dx^{i_1}\wedge\cdots\wedge dx^{i_k},
\]
with smooth coefficient functions.

\begin{example}[One-forms]\label{ex:vii09-one-forms}
A one-form is locally \(\alpha=\sum_i a_i\,dx^i\). For a function \(f\), its differential from VII/05 is the one-form
\[
df=\sum_i\frac{\partial f}{\partial x^i}\,dx^i.
\]
\end{example}

\section{The wedge product}
\begin{definition}[Wedge product]\label{def:vii09-wedge}
For \(\alpha\in\Omega^k(M)\) and \(\beta\in\Omega^\ell(M)\), the wedge product \(\alpha\wedge\beta\in\Omega^{k+\ell}(M)\) is the alternating product of the two covariant tensors.
\end{definition}

\begin{theorem}[Graded algebra identities]\label{thm:vii09-wedge-laws}
For homogeneous forms,
\[
\alpha\wedge\beta=(-1)^{k\ell}\beta\wedge\alpha,
\qquad
(\alpha\wedge\beta)\wedge\gamma=\alpha\wedge(\beta\wedge\gamma).
\]
The wedge product is bilinear over \(\mathbb R\) and compatible with multiplication by smooth functions.
\end{theorem}

\begin{corollary}[Odd forms square to zero]\label{cor:vii09-odd-square}
If \(\alpha\) has odd degree, then \(\alpha\wedge\alpha=0\).
\end{corollary}

\section{Pullback of forms}
\begin{definition}[Pullback]\label{def:vii09-pullback}
For a smooth map \(F:M\to N\) and \(\omega\in\Omega^k(N)\), define
\[
(F^*\omega)_p(v_1,\ldots,v_k)
=\omega_{F(p)}(dF_pv_1,\ldots,dF_pv_k).
\]
\end{definition}

\begin{theorem}[Pullback identities]\label{thm:vii09-pullback-laws}
Pullback is linear, preserves degree, and satisfies
\[
F^*(\alpha\wedge\beta)=F^*\alpha\wedge F^*\beta,
\qquad
(G\circ F)^*=F^*\circ G^*,
\qquad
F^*(df)=d(f\circ F).
\]
\end{theorem}

\begin{example}[Polar coordinates]\label{ex:vii09-polar}
For \(F(r,\theta)=(r\cos\theta,r\sin\theta)\),
\[
F^*(dx)=\cos\theta\,dr-r\sin\theta\,d\theta,
\qquad
F^*(dy)=\sin\theta\,dr+r\cos\theta\,d\theta,
\]
so
\[
F^*(dx\wedge dy)=r\,dr\wedge d\theta.
\]
\end{example}

\section{Exterior derivative}
\begin{theorem}[Exterior derivative]\label{thm:vii09-exterior-derivative}
There is a unique family of linear maps
\[
d:\Omega^k(M)\to\Omega^{k+1}(M)
\]
such that:
\begin{enumerate}
\item for functions it is the ordinary differential;
\item \(d(\alpha\wedge\beta)=d\alpha\wedge\beta+(-1)^k\alpha\wedge d\beta\) for \(\alpha\in\Omega^k(M)\);
\item \(d^2=0\).
\end{enumerate}
In coordinates,
\[
d\!\left(\sum_I a_I\,dx^I\right)=\sum_I da_I\wedge dx^I.
\]
\end{theorem}

\begin{proof}
The coordinate formula defines \(d\). Equality of mixed partial derivatives gives \(d^2=0\), and a direct expansion proves the graded Leibniz rule. Naturality under coordinate changes shows the construction is intrinsic.
\end{proof}

\begin{example}[A one-form on \(\mathbb R^2\)]\label{ex:vii09-d-one-form}
For \(\alpha=P\,dx+Q\,dy\),
\[
d\alpha=(Q_x-P_y)\,dx\wedge dy.
\]
\end{example}

\begin{example}[A two-form on \(\mathbb R^3\)]\label{ex:vii09-d-two-form}
For
\[
\beta=P\,dy\wedge dz+Q\,dz\wedge dx+R\,dx\wedge dy,
\]
one obtains
\[
d\beta=(P_x+Q_y+R_z)\,dx\wedge dy\wedge dz.
\]
\end{example}

\section{Naturality of the exterior derivative}
\begin{theorem}[Pullback commutes with \(d\)]\label{thm:vii09-natural-d}
For every smooth \(F:M\to N\),
\[
d(F^*\omega)=F^*(d\omega).
\]
\end{theorem}
\begin{proof}
It holds first for functions by the chain rule, then for coordinate-generated forms by the graded Leibniz rule, and therefore locally for every form.
\end{proof}

\section{Closed and exact forms}
\begin{definition}[Closed and exact]\label{def:vii09-closed-exact}
A form \(\omega\) is closed if \(d\omega=0\). It is exact if \(\omega=d\eta\) for some form \(\eta\) of one lower degree.
\end{definition}

\begin{proposition}[Exact implies closed]\label{prop:vii09-exact-closed}
Every exact form is closed.
\end{proposition}
\begin{proof}
If \(\omega=d\eta\), then \(d\omega=d^2\eta=0\).
\end{proof}

\begin{theorem}[Poincaré lemma on star-shaped domains]\label{thm:vii09-poincare}
If \(U\subset\mathbb R^n\) is star-shaped, then every closed \(k\)-form on \(U\) with \(k\ge1\) is exact.
\end{theorem}
\begin{remark}
The theorem is local in spirit. Global closed forms need not be exact; the gap between the two notions becomes topological and leads later to de Rham cohomology.
\end{remark}

\section{Top-degree forms}
\begin{proposition}[Top-degree local form]\label{prop:vii09-top-form}
On an \(n\)-manifold, every \(n\)-form is locally
\[
\omega=f\,dx^1\wedge\cdots\wedge dx^n.
\]
Under a coordinate change \(y=y(x)\), the coordinate volume element transforms by the Jacobian determinant.
\end{proposition}

\begin{warning}[Integration is not yet defined]\label{warning:vii09-no-integration}
Top-degree forms are the correct objects to integrate on oriented manifolds, but orientation and integration are developed in VII/10. The present chapter uses only their algebraic and differential properties.
\end{warning}

\section{Exercises}

\begin{exercise}\label{exr:vii09-01}
Show that a \(0\)-form is exactly a smooth function.
\end{exercise}
\begin{hint}
Use \(\Lambda^0T_p^*M=\mathbb R\).
\end{hint}
\begin{solution}
A smooth section of the trivial rank-one bundle \(\Lambda^0T^*M\) is precisely a smooth real-valued function.
\end{solution}

\begin{exercise}\label{exr:vii09-02}
Compute \(\dim\Lambda^2(\mathbb R^5)^*\).
\end{exercise}
\begin{hint}
Use the binomial coefficient.
\end{hint}
\begin{solution}
The dimension is \(\binom52=10\).
\end{solution}

\begin{exercise}\label{exr:vii09-03}
Show that every \(k\)-form vanishes identically when \(k>\dim M\).
\end{exercise}
\begin{hint}
Alternating tensors vanish on linearly dependent arguments.
\end{hint}
\begin{solution}
Any \(k\) tangent vectors in an \(n\)-dimensional space with \(k>n\) are linearly dependent, so every alternating \(k\)-linear map is zero.
\end{solution}

\begin{exercise}\label{exr:vii09-04}
Show \(dx^i\wedge dx^i=0\).
\end{exercise}
\begin{hint}
Use graded commutativity for degree one.
\end{hint}
\begin{solution}
\(dx^i\wedge dx^i=-dx^i\wedge dx^i\), hence it vanishes over \(\mathbb R\).
\end{solution}

\begin{exercise}\label{exr:vii09-05}
Compute \(dy\wedge dx\) in terms of \(dx\wedge dy\).
\end{exercise}
\begin{hint}
Swap two one-forms.
\end{hint}
\begin{solution}
\(dy\wedge dx=-dx\wedge dy\).
\end{solution}

\begin{exercise}\label{exr:vii09-06}
Expand \((a\,dx+b\,dy)\wedge(c\,dx+d\,dy)\).
\end{exercise}
\begin{hint}
Discard repeated factors.
\end{hint}
\begin{solution}
The result is \((ad-bc)\,dx\wedge dy\).
\end{solution}

\begin{exercise}\label{exr:vii09-07}
Show the wedge product of a \(2\)-form and a \(1\)-form commutes.
\end{exercise}
\begin{hint}
Evaluate \((-1)^{2\cdot1}\).
\end{hint}
\begin{solution}
The sign is \(+1\), so \(\alpha\wedge\beta=\beta\wedge\alpha\).
\end{solution}

\begin{exercise}\label{exr:vii09-08}
Show pullback preserves wedge products.
\end{exercise}
\begin{hint}
Apply both sides to tangent vectors.
\end{hint}
\begin{solution}
Both sides evaluate by first applying \(dF\) to each tangent argument and then alternating the same tensor product, so they agree.
\end{solution}

\begin{exercise}\label{exr:vii09-09}
Compute the pullback of \(dx\) under \(F(t)=(t^2,t^3)\).
\end{exercise}
\begin{hint}
Differentiate the first coordinate.
\end{hint}
\begin{solution}
\(F^*(dx)=d(t^2)=2t\,dt\).
\end{solution}

\begin{exercise}\label{exr:vii09-10}
Compute the pullback of \(dy\) for the same map.
\end{exercise}
\begin{hint}
Differentiate the second coordinate.
\end{hint}
\begin{solution}
\(F^*(dy)=3t^2\,dt\).
\end{solution}

\begin{exercise}\label{exr:vii09-11}
Explain why the pullback of every \(2\)-form on \(\mathbb R^2\) to a curve is zero.
\end{exercise}
\begin{hint}
A one-dimensional manifold has no nonzero two-forms.
\end{hint}
\begin{solution}
The pullback is a two-form on the one-dimensional domain, hence vanishes.
\end{solution}

\begin{exercise}\label{exr:vii09-12}
Verify \(d(fg)=f\,dg+g\,df\).
\end{exercise}
\begin{hint}
Use the degree-zero Leibniz rule.
\end{hint}
\begin{solution}
The graded Leibniz rule has sign \(+1\) for degree zero and yields the ordinary product rule.
\end{solution}

\begin{exercise}\label{exr:vii09-13}
Compute \(d(P\,dx)\) on \(\mathbb R^2\).
\end{exercise}
\begin{hint}
Use \(dP\wedge dx\).
\end{hint}
\begin{solution}
Since \(dP=P_xdx+P_ydy\), \(d(Pdx)=-P_y\,dx\wedge dy\).
\end{solution}

\begin{exercise}\label{exr:vii09-14}
Compute \(d(Q\,dy)\) on \(\mathbb R^2\).
\end{exercise}
\begin{hint}
Use \(dQ\wedge dy\).
\end{hint}
\begin{solution}
Only \(Q_xdx\wedge dy\) survives, so \(d(Qdy)=Q_x\,dx\wedge dy\).
\end{solution}

\begin{exercise}\label{exr:vii09-15}
Verify directly that \(d(df)=0\) in coordinates.
\end{exercise}
\begin{hint}
Use equality of mixed partials.
\end{hint}
\begin{solution}
The coefficient of \(dx^i\wedge dx^j\) is \(f_{x^ix^j}-f_{x^jx^i}=0\).
\end{solution}

\begin{exercise}\label{exr:vii09-16}
Show every exact form is closed.
\end{exercise}
\begin{hint}
Apply \(d\) once more.
\end{hint}
\begin{solution}
If \(\omega=d\eta\), then \(d\omega=d^2\eta=0\).
\end{solution}

\begin{exercise}\label{exr:vii09-17}
Give a closed one-form on \(\mathbb R^2\) and exhibit a potential.
\end{exercise}
\begin{hint}
Take \(x\,dx+y\,dy\).
\end{hint}
\begin{solution}
It equals \(d\bigl(\tfrac12(x^2+y^2)\bigr)\), hence is exact and therefore closed.
\end{solution}

\begin{exercise}\label{exr:vii09-18}
Determine when \(Pdx+Qdy\) is closed.
\end{exercise}
\begin{hint}
Use the coordinate formula for \(d\alpha\).
\end{hint}
\begin{solution}
It is closed exactly when \(Q_x-P_y=0\).
\end{solution}

\begin{exercise}\label{exr:vii09-19}
Show that pullback sends closed forms to closed forms.
\end{exercise}
\begin{hint}
Use naturality of \(d\).
\end{hint}
\begin{solution}
If \(d\omega=0\), then \(d(F^*\omega)=F^*(d\omega)=0\).
\end{solution}

\begin{exercise}\label{exr:vii09-20}
Show that pullback sends exact forms to exact forms.
\end{exercise}
\begin{hint}
Pull back a primitive.
\end{hint}
\begin{solution}
If \(\omega=d\eta\), then \(F^*\omega=F^*d\eta=dF^*\eta\).
\end{solution}

\begin{exercise}\label{exr:vii09-21}
Compute \(F^*(dx\wedge dy)\) for polar coordinates.
\end{exercise}
\begin{hint}
Use the pulled-back one-forms and wedge them.
\end{hint}
\begin{solution}
Expanding gives \(r\,dr\wedge d\theta\).
\end{solution}

\begin{exercise}\label{exr:vii09-22}
Show that every top-degree form on an \(n\)-manifold is closed.
\end{exercise}
\begin{hint}
Its exterior derivative has degree \(n+1\).
\end{hint}
\begin{solution}
There are no nonzero \((n+1)\)-forms, so \(d\omega=0\).
\end{solution}

\begin{exercise}\label{exr:vii09-23}
Explain why top-degree closedness does not imply global exactness.
\end{exercise}
\begin{hint}
Recall the warning after the Poincaré lemma.
\end{hint}
\begin{solution}
Closedness is local/differential, while exactness may be obstructed globally; de Rham cohomology measures this difference.
\end{solution}

\begin{exercise}\label{exr:vii09-24}
State the next structural ingredient needed before integrating top-degree forms globally.
\end{exercise}
\begin{hint}
Look at VII/10.
\end{hint}
\begin{solution}
One needs orientation, together with a compatible definition of integration on manifolds.
\end{solution}


\section{Solved problem dossiers}

\begin{problem}[Legacy differential-form problem: coordinate basis]\label{prob:vii09-01}
Show that the wedge monomials \(dx^{i_1}\wedge\cdots\wedge dx^{i_k}\) with increasing indices form a local frame of \(\Lambda^kT^*M\).
\end{problem}
\begin{solution}
At each point they are the standard basis of the exterior power of the cotangent space. Their number is \(\binom nk\), and smooth dependence on the coordinate coframe makes them a local bundle frame.
\end{solution}

\begin{problem}[Legacy wedge-product problem]\label{prob:vii09-02}
Derive graded commutativity from alternation.
\end{problem}
\begin{solution}
Exchanging a block of \(k\) arguments with a block of \(\ell\) arguments requires \(k\ell\) transpositions, producing the sign \((-1)^{k\ell}\).
\end{solution}

\begin{problem}[Legacy pullback problem]\label{prob:vii09-03}
Prove pullback is functorial.
\end{problem}
\begin{solution}
For \(F:M\to N\) and \(G:N\to P\), the chain rule gives \(d(G\circ F)=dG\circ dF\). Substituting this into the pointwise definition of pullback yields \((G\circ F)^*=F^*G^*\).
\end{solution}

\begin{problem}[Legacy exterior-derivative problem]\label{prob:vii09-04}
Starting from the coordinate formula, prove the graded Leibniz rule.
\end{problem}
\begin{solution}
Differentiate the coefficient products and use that moving a one-form \(da\) past a degree-\(k\) factor contributes the sign \((-1)^k\). The two groups of terms are exactly \(d\alpha\wedge\beta\) and \((-1)^k\alpha\wedge d\beta\).
\end{solution}

\begin{problem}[Legacy \(d^2=0\) problem]\label{prob:vii09-05}
Prove \(d^2=0\) in coordinates.
\end{problem}
\begin{solution}
After applying \(d\) twice, coefficients involve symmetric second derivatives multiplied by antisymmetric wedge products. Pairing indices \(i,j\) cancels terms because mixed partials agree.
\end{solution}

\begin{problem}[Polar-coordinate dossier]\label{prob:vii09-06}
Compute the pullback of the Euclidean area form under polar coordinates.
\end{problem}
\begin{solution}
Pull back \(dx\) and \(dy\), wedge the results, and cancel the \(dr\wedge dr\) and \(d\theta\wedge d\theta\) terms. The surviving coefficient is \(r(\cos^2\theta+\sin^2\theta)=r\).
\end{solution}

\begin{problem}[Closed one-form dossier]\label{prob:vii09-07}
For \(\alpha=Pdx+Qdy\), derive the closedness condition and relate it locally to a potential.
\end{problem}
\begin{solution}
The derivative is \((Q_x-P_y)dx\wedge dy\), so closedness is \(Q_x=P_y\). On a star-shaped domain, the Poincaré lemma then gives \(\alpha=df\) for some potential \(f\).
\end{solution}

\begin{problem}[Three-dimensional two-form dossier]\label{prob:vii09-08}
Derive the divergence-type formula for \(d\beta\) when \(\beta=Pdy\wedge dz+Qdz\wedge dx+Rdx\wedge dy\).
\end{problem}
\begin{solution}
Differentiate each coefficient. Only the term whose differential supplies the missing coordinate one-form survives in each summand; after reordering signs, the coefficient is \(P_x+Q_y+R_z\).
\end{solution}

\begin{problem}[Naturality dossier]\label{prob:vii09-09}
Prove \(dF^*=F^*d\) for arbitrary forms from the function case.
\end{problem}
\begin{solution}
Locally every form is a sum of functions times wedges of coordinate differentials. Pullback respects wedges, and both operators obey the graded Leibniz rule. The function case follows from the chain rule, so the identity extends to all local forms.
\end{solution}

\begin{problem}[Exact-versus-closed dossier]\label{prob:vii09-10}
Explain the logical relation between exactness, closedness, and topology.
\end{problem}
\begin{solution}
The identity \(d^2=0\) makes exact forms closed on every manifold. The converse is true locally on contractible/star-shaped charts but may fail globally, so the quotient of closed forms by exact forms records global information.
\end{solution}

\begin{problem}[Top-form dossier]\label{prob:vii09-11}
Show how the coefficient of a top-degree form changes under coordinates.
\end{problem}
\begin{solution}
Wedge the transformed one-forms \(dy^i=\sum_j(\partial y^i/\partial x^j)dx^j\). Multilinearity and alternation give the determinant of the Jacobian matrix.
\end{solution}

\begin{problem}[Restriction dossier]\label{prob:vii09-12}
If \(i:S\hookrightarrow M\) is an embedded submanifold, explain how ambient forms restrict to \(S\).
\end{problem}
\begin{solution}
Restriction is simply pullback \(i^*\omega\). Pointwise it evaluates \(\omega\) only on tangent vectors lying in \(T_pS\subset T_pM\), and naturality gives \(d(i^*\omega)=i^*(d\omega)\).
\end{solution}


\section{Challenge problems}

\begin{problem}[Coordinate independence of \(d\)]\label{prob:vii09-ch01}
Prove that the coordinate formula for the exterior derivative defines an intrinsic operator.
\end{problem}
\begin{solution}
On overlaps, pullback by the coordinate-change diffeomorphism commutes with the coordinate-defined derivative because the identity holds on functions and extends via wedges and the graded Leibniz rule. Thus the two coordinate formulas agree on overlaps and glue to a global operator.
\end{solution}

\begin{problem}[Poincaré homotopy operator]\label{prob:vii09-ch02}
On a star-shaped domain, outline the homotopy operator proving that every closed positive-degree form is exact.
\end{problem}
\begin{solution}
Use the radial homotopy \(H_t(x)=tx\) and radial vector field \(R=\sum x^i\partial_{x^i}\). Define \(K\omega=\int_0^1 t^{k-1}\,\iota_R\omega(tx)\,dt\) for a \(k\)-form. The homotopy identity \(dK+Kd=\operatorname{id}-H_0^*\) implies \(\omega=dK\omega\) when \(k>0\) and \(d\omega=0\).
\end{solution}

\begin{problem}[Exterior algebra under a linear map]\label{prob:vii09-ch03}
For a linear map \(A:V\to W\), prove that pullback \(A^*:\Lambda^\bullet W^*\to\Lambda^\bullet V^*\) is a graded-algebra homomorphism.
\end{problem}
\begin{solution}
Define \(A^*\omega(v_1,\ldots,v_k)=\omega(Av_1,\ldots,Av_k)\). Multilinearity and alternation are preserved, and applying this definition to the alternation formula for wedges gives \(A^*(\alpha\wedge\beta)=A^*\alpha\wedge A^*\beta\).
\end{solution}

\begin{problem}[A global closed non-exact form preview]\label{prob:vii09-ch04}
On \(S^1\), explain why the angular one-form is locally \(d\theta\) but cannot be globally the differential of a single-valued smooth angle function.
\end{problem}
\begin{solution}
Local angle coordinates differ by additive constants, so their differentials glue to a global closed one-form. A global potential would provide a single-valued smooth angle whose change around the circle is zero, contradicting the nontrivial winding. A complete cohomological treatment comes later.
\end{solution}

\begin{problem}[Jacobian determinant and top forms]\label{prob:vii09-ch05}
Prove directly that \(dy^1\wedge\cdots\wedge dy^n=\det(\partial y/\partial x)\,dx^1\wedge\cdots\wedge dx^n\).
\end{problem}
\begin{solution}
Expand each \(dy^i\) in the \(dx^j\) basis. Terms with repeated indices vanish. The remaining terms are indexed by permutations, and their wedge signs reproduce exactly the Leibniz formula for the determinant.
\end{solution}
