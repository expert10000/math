\chapter{Fuchsian Groups}
\label{ch:vii35-fuchsian}

\section{Discrete subgroups of the hyperbolic isometry group}

\begin{definition}[Fuchsian group]\label{def:vii35-fuchsian}
A \emph{Fuchsian group} is a discrete subgroup
\[
\Gamma<\operatorname{PSL}(2,\mathbb R),
\]
acting on the upper half-plane \(\mathbb H\) by orientation-preserving hyperbolic isometries.
\end{definition}

Discreteness is a topological condition on the matrix group, not merely an assertion that the abstract group is countable.  It forces the identity to be isolated and prevents nontrivial group elements from accumulating at the identity.

\begin{proposition}[Isolation criterion]\label{prop:vii35-isolation}
A subgroup \(\Gamma<\operatorname{PSL}(2,\mathbb R)\) is discrete if and only if there is a neighborhood \(U\) of the identity such that \(U\cap\Gamma=\{1\}\).
\end{proposition}


\section{Proper discontinuity and stabilizers}

\begin{theorem}[Proper discontinuity]\label{thm:vii35-proper}
If \(\Gamma\) is Fuchsian, then its action on \(\mathbb H\) is properly discontinuous: for every compact \(K\subset\mathbb H\), only finitely many \(\gamma\in\Gamma\) satisfy
\[
\gamma K\cap K\ne\varnothing.
\]
\end{theorem}

\begin{corollary}[Finite stabilizers]\label{cor:vii35-stabilizers}
For every \(z\in\mathbb H\), the stabilizer \(\Gamma_z\) is finite.  If \(\Gamma\) is torsion-free, every stabilizer is trivial.
\end{corollary}

\begin{remark}
An interior fixed point of a nontrivial orientation-preserving hyperbolic isometry is elliptic.  Thus torsion-free Fuchsian groups contain no elliptic elements.
\end{remark}


\section{Quotient surfaces and orbifolds}

Proper discontinuity allows local hyperbolic geometry to descend to the quotient.

\begin{theorem}[Hyperbolic quotient]\label{thm:vii35-quotient}
If \(\Gamma\) is torsion-free and Fuchsian, then
\[
X=\Gamma\backslash\mathbb H
\]
is a smooth hyperbolic surface and the projection \(\mathbb H\to X\) is a covering map and a local isometry.
\end{theorem}

If elliptic elements are present, the quotient is naturally a hyperbolic orbifold; elliptic fixed points become cone points.


\section{Fundamental domains and side pairings}

\begin{definition}[Fundamental domain]\label{def:vii35-fundamental}
A region \(F\subset\mathbb H\) is a fundamental domain for \(\Gamma\) if its translates cover \(\mathbb H\), while distinct interiors \(\gamma F^\circ\) are disjoint.
\end{definition}

\begin{definition}[Dirichlet domain]\label{def:vii35-dirichlet}
For a point \(p\in\mathbb H\) with trivial stabilizer, the Dirichlet domain centered at \(p\) is
\[
D_p=\{z\in\mathbb H:d(z,p)\le d(z,\gamma p)\text{ for all }\gamma\in\Gamma\}.
\]
\end{definition}

Its sides lie on hyperbolic perpendicular bisectors.  Side-pairing transformations encode how boundary arcs are identified in the quotient.


\section{The modular group}

The basic arithmetic example is
\[
\operatorname{PSL}(2,\mathbb Z)<\operatorname{PSL}(2,\mathbb R).
\]
It is generated by
\[
T(z)=z+1,
\qquad
S(z)=-\frac1z.
\]
A standard fundamental domain is
\[
\mathcal F=\left\{z\in\mathbb H:|\operatorname{Re}z|\le\frac12,\ |z|\ge1\right\}.
\]

\begin{proposition}[Side pairings for the modular domain]\label{prop:vii35-modular-sides}
The vertical sides are paired by \(T\), while the two circular boundary arcs on \(|z|=1\) are paired by \(S\).
\end{proposition}

\begin{remark}[Orbifold points]\label{rem:vii35-modular-orbifold}
The modular quotient has elliptic orbifold points represented by \(i\) and \(e^{2\pi i/3}\), together with a cusp represented by \(\infty\).
\end{remark}


\section{Cusps and parabolic fixed points}

\begin{definition}[Cusp]\label{def:vii35-cusp}
A cusp of a Fuchsian quotient is an end associated with a \(\Gamma\)-orbit of parabolic fixed points on \(\partial\mathbb H=\mathbb R\cup\{\infty\}\).
\end{definition}

After conjugation, a primitive parabolic stabilizer is generated by a translation \(z\mapsto z+\omega\).  A sufficiently high horoball
\[
\{z:\operatorname{Im}z>Y\}
\]
then projects to a standard cusp neighborhood.


\section{Hyperbolic elements and closed geodesics}

\begin{theorem}[Axis to closed geodesic]\label{thm:vii35-axis-closed}
Let \(\gamma\in\Gamma\) be hyperbolic.  Its invariant axis \(A_\gamma\subset\mathbb H\) projects to a closed geodesic in \(\Gamma\backslash\mathbb H\).  For a primitive hyperbolic element, the length of that closed geodesic is the translation length \(\tau(\gamma)\).
\end{theorem}

Conjugate hyperbolic elements determine the same free-homotopy class of closed geodesic.  This is the first appearance of the deep dictionary between group conjugacy and quotient geometry.


\section{Cofinite and cocompact groups}

\begin{definition}[Cofinite and cocompact]\label{def:vii35-cofinite}
A Fuchsian group is \emph{cofinite} if \(\Gamma\backslash\mathbb H\) has finite hyperbolic area, and \emph{cocompact} if the quotient is compact.
\end{definition}

Cocompact groups have no cusps.  Cofinite noncocompact groups have finitely many cusp ends.

\begin{example}[Area of the modular surface]\label{ex:vii35-modular-area}
The standard modular fundamental domain has hyperbolic area
\[
\operatorname{Area}(\mathcal F)=\frac{\pi}{3}.
\]
\end{example}


\section{A first look at limit sets}

For \(z\in\mathbb H\), the orbit \(\Gamma z\) has no accumulation point inside \(\mathbb H\), but it may accumulate on the ideal boundary.

\begin{definition}[Limit set in dimension two]\label{def:vii35-limitset}
The limit set \(\Lambda(\Gamma)\subset\partial\mathbb H\) is the set of accumulation points of one, hence every, orbit \(\Gamma z\) in the ideal boundary.
\end{definition}

For the modular group the limit set is the entire boundary circle.  Boundary dynamics will become central after passing from \(\operatorname{PSL}(2,\mathbb R)\) to \(\operatorname{PSL}(2,\mathbb C)\).


\section{Bridge to hyperbolic three-space}

The group \(\operatorname{PSL}(2,\mathbb R)\) sits naturally inside \(\operatorname{PSL}(2,\mathbb C)\).  VII/36 enlarges the upper half-plane to hyperbolic three-space, identifies its ideal boundary with the Riemann sphere, and interprets complex Möbius transformations as all orientation-preserving isometries of \(\mathbb H^3\).

\section{Solved problem dossiers}

\begin{problem}[Discreteness of a cyclic hyperbolic group]\label{prob:vii35-01}
Let \(g(z)=e^\ell z\) with \(\ell>0\).  Show that \(\langle g\rangle\) is a discrete subgroup of \(\operatorname{PSL}(2,\mathbb R)\).
\end{problem}

\begin{solution}
The powers are \(g^n(z)=e^{n\ell}z\).  A sequence of distinct powers cannot converge to the identity because its translation lengths are \(|n|\ell\), bounded away from zero for \(n\ne0\).  Hence the identity is isolated and the cyclic subgroup is discrete.
\end{solution}

\begin{problem}[Torsion and freeness]\label{prob:vii35-02}
Suppose \(\Gamma\) is torsion-free and Fuchsian.  Prove that no nonidentity element fixes a point of \(\mathbb H\).
\end{problem}

\begin{solution}
An orientation-preserving hyperbolic isometry fixing an interior point is elliptic.  In a discrete subgroup an elliptic stabilizer is finite, so the element has finite order.  Torsion-freeness excludes this unless the element is the identity.
\end{solution}

\begin{problem}[Modular side pairing]\label{prob:vii35-03}
Verify that \(T(z)=z+1\) identifies the two vertical sides \(\operatorname{Re}z=\pm1/2\) of the standard modular domain.
\end{problem}

\begin{solution}
If \(z=-1/2+iy\), then \(Tz=1/2+iy\).  Thus \(T\) maps the left vertical boundary to the right one and preserves the hyperbolic metric.
\end{solution}

\begin{problem}[The inversion side pairing]\label{prob:vii35-04}
Show that \(S(z)=-1/z\) preserves the unit semicircle and exchanges its two halves.
\end{problem}

\begin{solution}
For \(|z|=1\), \(|S(z)|=1/|z|=1\).  Writing \(z=e^{i\theta}\) gives \(S(z)=-e^{-i\theta}=e^{i(\pi-\theta)}\), which exchanges the two arcs about the imaginary axis.
\end{solution}

\begin{problem}[Area of the modular domain]\label{prob:vii35-05}
Compute the hyperbolic area of \(\mathcal F=\{|x|\le1/2,\ x^2+y^2\ge1\}\).
\end{problem}

\begin{solution}
Using \(dA=dx\,dy/y^2\),
\[
\operatorname{Area}(\mathcal F)=\int_{-1/2}^{1/2}\int_{\sqrt{1-x^2}}^\infty \frac{dy\,dx}{y^2}
=\int_{-1/2}^{1/2}\frac{dx}{\sqrt{1-x^2}}
=2\arcsin(1/2)=\frac\pi3.
\]
\end{solution}

\begin{problem}[A standard cusp cylinder]\label{prob:vii35-06}
For \(T(z)=z+1\), describe the quotient of \(H_Y=\{y>Y\}\) by \(\langle T\rangle\).
\end{problem}

\begin{solution}
A fundamental strip is \(0\le x\le1\), \(y>Y\), with the two vertical sides identified.  Topologically the quotient is a cylinder, and metrically it is the standard cusp end whose horizontal circumference is \(1/y\), tending to zero as \(y\to\infty\).
\end{solution}

\begin{problem}[Axis and quotient loop]\label{prob:vii35-07}
Let \(g(z)=e^\ell z\).  Identify its axis and the length of the corresponding quotient geodesic.
\end{problem}

\begin{solution}
The imaginary axis is invariant and is the unique axis joining the fixed points \(0\) and \(\infty\).  Since \(d(i,e^\ell i)=\ell\), the projected primitive closed geodesic has length \(\ell\).
\end{solution}

\begin{problem}[Conjugacy and closed geodesics]\label{prob:vii35-08}
Explain why conjugate hyperbolic elements determine the same closed-geodesic geometry up to the quotient action.
\end{problem}

\begin{solution}
If \(hgh^{-1}\) is conjugate to \(g\), then its axis is \(h(A_g)\).  The two axes project to the same free-homotopy class after changing the lift by the deck transformation \(h\).  Translation length is conjugacy invariant.
\end{solution}

\begin{problem}[Finite stabilizer from discreteness]\label{prob:vii35-09}
Give a direct compactness argument that \(\Gamma_i\) is finite.
\end{problem}

\begin{solution}
The stabilizer of \(i\) in \(\operatorname{PSL}(2,\mathbb R)\) is a compact copy of \(\operatorname{SO}(2)\).  The intersection of a discrete subgroup with a compact set is finite; hence \(\Gamma_i\) is finite.
\end{solution}

\begin{problem}[Why a cusp prevents compactness]\label{prob:vii35-10}
Show geometrically why a quotient containing a standard cusp neighborhood cannot be compact.
\end{problem}

\begin{solution}
The cusp contains points represented by heights \(y\to\infty\).  Their injectivity radii tend to zero and they escape every compact subset of the quotient.  Thus the quotient has a noncompact end.
\end{solution}

\begin{problem}[Orbit accumulation]\label{prob:vii35-11}
Why can an orbit of a Fuchsian group have no accumulation point inside \(\mathbb H\)?
\end{problem}

\begin{solution}
Proper discontinuity implies that every compact neighborhood meets the orbit in only finitely many points.  Therefore an infinite orbit cannot accumulate in the interior; any accumulation in the compactification must occur on the ideal boundary.
\end{solution}

\begin{problem}[Modular limit set]\label{prob:vii35-12}
Explain why the rational points together with \(\infty\) already force the modular limit set to be the whole boundary.
\end{problem}

\begin{solution}
The orbit of \(\infty\) under \(\operatorname{PSL}(2,\mathbb Z)\) is \(\mathbb Q\cup\{\infty\}\).  Since the rationals are dense in the real boundary and the limit set is closed and invariant, its closure is all of \(\mathbb R\cup\{\infty\}\).
\end{solution}

\section{Challenge problems}

\begin{problem}[A presentation of the modular group]\label{prob:vii35-ch01}
Using the generators \(S(z)=-1/z\) and \(T(z)=z+1\), verify \(S^2=1\) in \(\operatorname{PSL}(2,\mathbb Z)\) and \((ST)^3=1\).  Interpret these relations geometrically.
\end{problem}

\begin{solution}
The matrix of \(S\) squares to \(-I\), hence to the identity in PSL.  A direct multiplication gives \((ST)^3=-I\), again the identity in PSL.  Geometrically these are rotations about the two elliptic orbifold points of orders two and three.  They lead to the presentation \(\operatorname{PSL}(2,\mathbb Z)\cong C_2*C_3\).
\end{solution}

\begin{problem}[Dirichlet domains are convex]\label{prob:vii35-ch02}
Show that a Dirichlet domain is an intersection of hyperbolic half-planes and hence geodesically convex.
\end{problem}

\begin{solution}
Each condition \(d(z,p)\le d(z,\gamma p)\) defines the half-plane bounded by the perpendicular bisector of \(p\) and \(\gamma p\).  Hyperbolic half-planes are geodesically convex, and arbitrary intersections of convex sets are convex.
\end{solution}

\begin{problem}[Primitive versus imprimitive geodesics]\label{prob:vii35-ch03}
If \(g=h^m\) with \(m>1\) and \(h\) primitive hyperbolic, compare the projected geodesics and their lengths.
\end{problem}

\begin{solution}
The elements have the same axis.  The quotient curve associated with \(g\) traverses the primitive closed geodesic \(m\) times, and \(\tau(g)=m\tau(h)\).
\end{solution}

\begin{problem}[From Fuchsian to Kleinian]\label{prob:vii35-ch04}
Embed a Fuchsian group into \(\operatorname{PSL}(2,\mathbb C)\).  Predict what geometric subset of \(\mathbb H^3\) it should preserve.
\end{problem}

\begin{solution}
Use the same real matrices, now viewed as complex matrices.  Their boundary action preserves the generalized circle \(\mathbb R\cup\{\infty\}\), so in \(\mathbb H^3\) they preserve the totally geodesic plane whose ideal boundary is that circle.  This plane is naturally a copy of \(\mathbb H^2\).
\end{solution}

\begin{problem}[Cusp width and conjugation]\label{prob:vii35-ch05}
Let a parabolic stabilizer be generated by \(z\mapsto z+\omega\) with \(\omega>0\).  Show that conjugating by a positive dilation can normalize the cusp width to one.
\end{problem}

\begin{solution}
Conjugate by \(h(z)=z/\omega\).  Then \(h(z+\omega)=z/\omega+1=T(h(z))\), so \(hgh^{-1}(z)=z+1\).  Thus the numerical cusp width depends on the chosen boundary coordinate, while the parabolic conjugacy type does not.
\end{solution}
