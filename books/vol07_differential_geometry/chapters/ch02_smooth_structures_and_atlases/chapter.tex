\chapter{Smooth Structures and Atlases}
\label{ch:vii02-smooth-structures-and-atlases}

VII/01 supplied topological manifolds. Smooth geometry begins by requiring changes of coordinates to be smooth:
\[
\boxed{\text{smooth structure}=\text{topological atlas}+\text{smooth transition maps}.}
\]

\section*{Standing conventions}
\(M\) is a Hausdorff, second-countable topological \(n\)-manifold. Euclidean smoothness means \(C^\infty\).

\section{Transition maps}
\begin{definition}[Transition map]\label{def:vii02-transition}
For charts \((U,\varphi)\), \((V,\psi)\), the transition is
\[
\psi\circ\varphi^{-1}:\varphi(U\cap V)\to\psi(U\cap V).
\]
\end{definition}
\begin{definition}[Smooth compatibility]\label{def:vii02-compatible}
Two charts are smoothly compatible when the overlap transition is a diffeomorphism of Euclidean open sets.
\end{definition}
\begin{proposition}[Restriction]\label{prop:vii02-restrictions}
Restrictions of compatible charts remain compatible.
\end{proposition}

\section{Atlases and smooth structures}
\begin{definition}[Smooth atlas]\label{def:vii02-smooth-atlas}
A smooth atlas is a covering atlas of pairwise smoothly compatible charts.
\end{definition}
\begin{definition}[Equivalent atlases]\label{def:vii02-equivalent-atlases}
Two smooth atlases are equivalent when every chart of one is compatible with every chart of the other.
\end{definition}
\begin{theorem}[Maximal atlas]\label{thm:vii02-maximal-atlas}
Every smooth atlas lies in a unique maximal smooth atlas consisting of all charts compatible with it.
\end{theorem}
\begin{definition}[Smooth structure]\label{def:vii02-smooth-structure}
A smooth structure is a maximal smooth atlas.
\end{definition}

\section{Smooth functions}
\begin{definition}[Smooth function]\label{def:vii02-smooth-function}
For open \(W\subset M\), \(f:W\to\mathbb R\) is smooth when \(f\circ\varphi^{-1}\) is smooth in every compatible chart.
\end{definition}
\begin{theorem}[Atlas equivalence from smooth functions]\label{thm:vii02-functions-equivalence}
Two smooth atlases are equivalent iff they determine the same smooth real-valued functions on every open set.
\end{theorem}
\begin{proof}
Equivalent atlases transfer coordinate expressions through smooth transition maps. Conversely, coordinate functions of charts detect the component functions of cross-transitions.
\end{proof}

\section{Standard structures}
\begin{example}[Euclidean structure]\label{ex:vii02-euclidean}
The identity chart generates the standard smooth structure on every Euclidean open set.
\end{example}
\begin{example}[Stereographic sphere atlas]\label{ex:vii02-sphere}
The north/south stereographic charts form a smooth atlas on \(S^n\); their transition is inversion away from the origin.
\end{example}

\section{Projective spaces}
\begin{definition}[Homogeneous charts]\label{def:vii02-projective-charts}
On \(\mathbb{RP}^n\), \(U_i=\{x_i\neq0\}\) is identified with \(\mathbb R^n\) by normalizing \(x_i=1\). The analogous complex chart identifies the standard open in \(\mathbb{CP}^n\) with \(\mathbb C^n\).
\end{definition}
\begin{theorem}[Real projective space]\label{thm:vii02-rp}
The homogeneous charts make \(\mathbb{RP}^n\) a smooth \(n\)-manifold.
\end{theorem}
\begin{theorem}[Complex projective space]\label{thm:vii02-cp}
The homogeneous charts make \(\mathbb{CP}^n\) a smooth real \(2n\)-manifold.
\end{theorem}
\begin{warning}[Pseudo-atlas warning]\label{warning:vii02-pseudo-atlas}
Compatible local formulas must still be accompanied by Hausdorffness and second countability.
\end{warning}

\section{Atlas versus structure}
\begin{remark}[Generators are not intrinsic]\label{rem:vii02-generators}
Restricting charts or adding compatible charts changes the atlas but not the maximal smooth structure.
\end{remark}
\section{Exercises}
\begin{exercise}\label{exr:vii02-01}
Let \((U,\varphi)\) and \((V,\psi)\) be charts on a topological \(n\)-manifold. Write the transition map and its domain precisely.
\end{exercise}
\begin{hint}
Restrict to \(U\cap V\) before composing.
\end{hint}
\begin{solution}
The transition map is \(\psi\circ\varphi^{-1}:\varphi(U\cap V)\to\psi(U\cap V)\), between open subsets of \(\mathbb R^n\).
\end{solution}

\begin{exercise}\label{exr:vii02-02}
Show that smooth compatibility of charts is symmetric.
\end{exercise}
\begin{hint}
Use the inverse transition map.
\end{hint}
\begin{solution}
If \(\psi\circ\varphi^{-1}\) is a diffeomorphism, its inverse is \(\varphi\circ\psi^{-1}\), hence smooth.
\end{solution}

\begin{exercise}\label{exr:vii02-03}
Show that restricting a smooth chart to a smaller open set gives a compatible chart.
\end{exercise}
\begin{hint}
Transitions restrict.
\end{hint}
\begin{solution}
Every new transition is a restriction of an old smooth transition, hence smooth.
\end{solution}

\begin{exercise}\label{exr:vii02-04}
Prove that every smooth atlas is contained in a unique maximal smooth atlas.
\end{exercise}
\begin{hint}
Take all compatible charts.
\end{hint}
\begin{solution}
Collect all charts compatible with the atlas. They are mutually compatible locally through generating charts; maximality and uniqueness follow immediately.
\end{solution}

\begin{exercise}\label{exr:vii02-05}
Show that two atlases define the same smooth structure iff their union is a smooth atlas.
\end{exercise}
\begin{hint}
Use cross-compatibility.
\end{hint}
\begin{solution}
Equal maximal atlases imply all cross-transitions are smooth. Conversely a smooth union generates one maximal atlas containing both.
\end{solution}

\begin{exercise}\label{exr:vii02-06}
Verify that the identity chart generates the usual smooth structure on an open \(U\subseteq\mathbb R^n\).
\end{exercise}
\begin{hint}
Compare all charts to Euclidean coordinates.
\end{hint}
\begin{solution}
The maximal extension consists precisely of charts whose coordinate changes with the identity chart are smooth diffeomorphisms.
\end{solution}

\begin{exercise}\label{exr:vii02-07}
Show affine coordinate changes \(x\mapsto Ax+b\), \(A\in GL(n,\mathbb R)\), are compatible with the standard structure.
\end{exercise}
\begin{hint}
Write the inverse.
\end{hint}
\begin{solution}
Both the affine map and \(y\mapsto A^{-1}(y-b)\) are smooth.
\end{solution}

\begin{exercise}\label{exr:vii02-08}
Prove the north/south stereographic charts form a smooth atlas on \(S^n\).
\end{exercise}
\begin{hint}
Use the overlap inversion.
\end{hint}
\begin{solution}
The overlap transition is \(x\mapsto x/\|x\|^2\), up to convention, smooth on \(\mathbb R^n\setminus\{0\}\).
\end{solution}

\begin{exercise}\label{exr:vii02-09}
Explain why the maximal atlas, rather than a chosen finite atlas, is the intrinsic smooth structure.
\end{exercise}
\begin{hint}
Refinements change generators but not compatibility.
\end{hint}
\begin{solution}
Restricting or adding compatible charts changes the written atlas while leaving all allowed coordinate changes and smooth functions unchanged.
\end{solution}

\begin{exercise}\label{exr:vii02-10}
Show equivalent atlases define the same smooth real-valued functions.
\end{exercise}
\begin{hint}
Insert a transition.
\end{hint}
\begin{solution}
Locally \(f\circ\psi^{-1}=(f\circ\varphi^{-1})\circ(\varphi\circ\psi^{-1})\), a composition of smooth maps.
\end{solution}

\begin{exercise}\label{exr:vii02-11}
Prove the converse using coordinate functions.
\end{exercise}
\begin{hint}
Each chart component is a smooth function.
\end{hint}
\begin{solution}
If both atlases give the same smooth functions, the components of every cross-transition are smooth; symmetry gives smooth inverse transitions.
\end{solution}

\begin{exercise}\label{exr:vii02-12}
Write the standard affine chart on \(\mathbb{RP}^n\) where \(x_i\neq0\).
\end{exercise}
\begin{hint}
Normalize \(x_i=1\).
\end{hint}
\begin{solution}
Send \([x]\) to all ratios \(x_j/x_i\), \(j\neq i\).
\end{solution}

\begin{exercise}\label{exr:vii02-13}
Compute the transition map between two standard \(\mathbb{RP}^n\) charts.
\end{exercise}
\begin{hint}
Divide the normalized coordinates by the new pivot coordinate.
\end{hint}
\begin{solution}
The components are rational functions with nonzero denominator on the overlap, so they are smooth.
\end{solution}

\begin{exercise}\label{exr:vii02-14}
Explain why \(\mathbb{CP}^n\) has real dimension \(2n\).
\end{exercise}
\begin{hint}
Each affine chart is \(\mathbb C^n\).
\end{hint}
\begin{solution}
Since \(\mathbb C^n\cong\mathbb R^{2n}\), the real manifold dimension is \(2n\).
\end{solution}

\begin{exercise}\label{exr:vii02-15}
Show the homogeneous projective charts cover projective space.
\end{exercise}
\begin{hint}
A nonzero vector has a nonzero coordinate.
\end{hint}
\begin{solution}
Every projective point lies in some chart \(U_i=\{x_i\neq0\}\).
\end{solution}

\begin{exercise}\label{exr:vii02-16}
Why must Hausdorffness and second countability be checked for a pseudo-atlas?
\end{exercise}
\begin{hint}
Local coordinate data do not imply global axioms.
\end{hint}
\begin{solution}
Locally Euclidean spaces can be non-Hausdorff or fail second countability, as VII/01 showed.
\end{solution}

\begin{exercise}\label{exr:vii02-17}
Show a finite atlas with Euclidean chart images gives second countability.
\end{exercise}
\begin{hint}
Pull back rational-ball bases.
\end{hint}
\begin{solution}
Each chart image has a countable basis; a finite union of their pullbacks is countable and forms a basis.
\end{solution}

\begin{exercise}\label{exr:vii02-18}
Verify Hausdorffness of \(\mathbb{RP}^n\) from \(S^n/\{\pm1\}\).
\end{exercise}
\begin{hint}
Use the closed antipodal relation.
\end{hint}
\begin{solution}
The finite group quotient of compact Hausdorff \(S^n\) by a closed equivalence relation is Hausdorff.
\end{solution}

\begin{exercise}\label{exr:vii02-19}
Show adding an already compatible chart does not change the smooth structure.
\end{exercise}
\begin{hint}
It already lies in the maximal atlas.
\end{hint}
\begin{solution}
The maximal extension before and after adding the chart is identical.
\end{solution}

\begin{exercise}\label{exr:vii02-20}
Show transition-map smoothness can be checked componentwise.
\end{exercise}
\begin{hint}
Use Euclidean vector-valued smoothness.
\end{hint}
\begin{solution}
A Euclidean map is smooth iff all coordinate components are smooth.
\end{solution}

\begin{exercise}\label{exr:vii02-21}
Give a smooth atlas on \(S^1\) using two stereographic charts.
\end{exercise}
\begin{hint}
Delete opposite points.
\end{hint}
\begin{solution}
Each deletion is homeomorphic to \(\mathbb R\), and the rational overlap transition is smooth.
\end{solution}

\begin{exercise}\label{exr:vii02-22}
Distinguish a topological atlas from a smooth atlas.
\end{exercise}
\begin{hint}
Compare overlap requirements.
\end{hint}
\begin{solution}
A topological atlas only requires homeomorphic charts; a smooth atlas also requires smooth transition diffeomorphisms.
\end{solution}

\begin{exercise}\label{exr:vii02-23}
Relate compatibility of two smooth structures to smoothness of the identity map.
\end{exercise}
\begin{hint}
Write its coordinate representative.
\end{hint}
\begin{solution}
The identity from one structure to the other is represented by cross-transition maps; it is a diffeomorphism exactly when the structures are equivalent.
\end{solution}

\begin{exercise}\label{exr:vii02-24}
State the conceptual boundary between VII/02 and VII/03.
\end{exercise}
\begin{hint}
Structure first, maps second.
\end{hint}
\begin{solution}
VII/02 decides which coordinates are smooth-compatible. VII/03 fixes those structures and studies smooth maps between manifolds.
\end{solution}

\section{Solved problem dossiers}
\begin{problem}[Legacy Problem 1(a): equivalent atlases and smooth functions]\label{prob:vii02-01}
Let \(\mathcal A,\mathcal B\) be smoothly equivalent atlases. Prove they determine the same smooth real-valued functions.
\end{problem}
\begin{solution}
Factor coordinate expressions through smooth cross-transition maps. The argument is symmetric.
\end{solution}

\begin{problem}[Legacy Problem 1(b): smooth functions determine atlas equivalence]\label{prob:vii02-02}
Assume two atlases define exactly the same smooth real-valued functions. Prove they are equivalent.
\end{problem}
\begin{solution}
Each coordinate function of a chart from one atlas is smooth for the other. Their coordinate expressions are the components of the cross-transition, hence smooth; reverse roles for the inverse.
\end{solution}

\begin{problem}[Legacy Problem 2(a): homogeneous charts on \(\mathbb{RP}^n\)]\label{prob:vii02-03}
Prove the homogeneous coordinate maps define a smooth atlas on real projective space.
\end{problem}
\begin{solution}
Normalize one nonzero coordinate. Overlap maps are rational with nonzero denominators, so smooth.
\end{solution}

\begin{problem}[Legacy Problem 2(b): homogeneous charts on \(\mathbb{CP}^n\)]\label{prob:vii02-04}
Prove the homogeneous coordinate maps define a smooth atlas on complex projective space.
\end{problem}
\begin{solution}
Each standard open is \(\mathbb C^n\); transitions are holomorphic rational maps, hence smooth as real maps.
\end{solution}

\begin{problem}[Projective-space global axioms]\label{prob:vii02-05}
Complete the pseudo-atlas proof by showing \(\mathbb{RP}^n,\mathbb{CP}^n\) are Hausdorff and second countable.
\end{problem}
\begin{solution}
Use compact group quotient models for Hausdorffness and the finite standard atlas for second countability.
\end{solution}

\begin{problem}[Maximal atlas theorem]\label{prob:vii02-06}
Construct the maximal atlas generated by a smooth atlas and prove uniqueness.
\end{problem}
\begin{solution}
Take all charts compatible with the atlas. Mutual compatibility follows locally through generating charts; maximality and uniqueness are immediate.
\end{solution}

\begin{problem}[Stereographic smooth structure]\label{prob:vii02-07}
Show the two stereographic charts generate a smooth structure on \(S^n\).
\end{problem}
\begin{solution}
The overlap inversion \(x/\|x\|^2\) is smooth away from zero.
\end{solution}

\begin{problem}[Atlas refinement invariance]\label{prob:vii02-08}
Prove an open refinement of a smooth atlas generates the same maximal atlas.
\end{problem}
\begin{solution}
Restricted charts are compatible with originals, and each original chart is locally covered by restrictions.
\end{solution}

\begin{problem}[Open-submanifold smooth structure]\label{prob:vii02-09}
Construct the canonical smooth structure on an open subset \(U\subset M\).
\end{problem}
\begin{solution}
Restrict the maximal atlas of \(M\) to domains intersected with \(U\).
\end{solution}

\begin{problem}[Unequal atlases, same structure]\label{prob:vii02-10}
Give two unequal atlases on \(\mathbb R\) that define the standard structure.
\end{problem}
\begin{solution}
Use the identity chart and a countable cover by restrictions of that chart.
\end{solution}

\begin{problem}[A topological chart outside the standard smooth structure]\label{prob:vii02-11}
Show \(x\mapsto x^3\) defines a topological chart on \(\mathbb R\) not compatible with the identity chart.
\end{problem}
\begin{solution}
The inverse transition \(x^{1/3}\) is not smooth at zero.
\end{solution}

\begin{problem}[Smooth structures from smooth functions]\label{prob:vii02-12}
Explain why the assignment \(U\mapsto C^\infty(U)\) determines the maximal smooth atlas.
\end{problem}
\begin{solution}
Coordinate functions characterize cross-transition smoothness, exactly as in the converse legacy atlas problem.
\end{solution}

\section{Challenges}
\begin{problem}[Challenge]\label{chal:vii02-01}
Construct two topological atlases on \(\mathbb R\) that are not smoothly equivalent.
\end{problem}
\begin{solution}
Use the identity and \(x\mapsto x^3\); the inverse transition \(x^{1/3}\) fails smoothness at zero.
\end{solution}

\begin{problem}[Challenge]\label{chal:vii02-02}
Prove \(\mathbb{RP}^n\) is a smooth quotient of \(S^n\) by the antipodal action using local slices.
\end{problem}
\begin{solution}
The free finite action gives neighborhoods disjoint from nontrivial translates; quotient them to charts and transfer the sphere smooth structure.
\end{solution}

\begin{problem}[Challenge]\label{chal:vii02-03}
Compare stereographic and graph atlases on \(S^2\) and prove equivalence.
\end{problem}
\begin{solution}
On common domains their transitions are explicit rational/square-root formulas with positive denominators, smooth both ways.
\end{solution}

\begin{problem}[Challenge]\label{chal:vii02-04}
Explain precisely what an exotic smooth structure would mean.
\end{problem}
\begin{solution}
It is a maximal smooth atlas on a fixed topological manifold not equivalent to another; the identity homeomorphism is not a diffeomorphism.
\end{solution}

\begin{problem}[Challenge]\label{chal:vii02-05}
Design a complete finite pseudo-atlas audit.
\end{problem}
\begin{solution}
Check cover, open chart domains/images, homeomorphism property, all overlap transitions, Hausdorffness, and second countability.
\end{solution}

\section*{Bridge to VII/03}
With smooth structures fixed, VII/03 defines smooth maps by smoothness of their coordinate representatives.
