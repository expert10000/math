\section{Supplementary worked examples}
\label{sec:vii14-pedagogy}

\begin{example}[A graph surface]\label{ex:vii14-ped-01}
For a smooth \(f:U\subset\mathbb R^2\to\mathbb R\),
\[
X(u,v)=(u,v,f(u,v))
\]
is a regular parametrization because \(X_u\times X_v=(-f_u,-f_v,1)\neq0\).  Every smooth graph is therefore a regular surface patch.
\end{example}

\begin{example}[Sphere patches]\label{ex:vii14-ped-02}
The unit sphere is covered by graph patches such as
\[
z=\sqrt{1-x^2-y^2}
\]
over the open unit disc and analogous patches for the other coordinate directions.  No one such graph covers the whole sphere, illustrating the genuinely local nature of surface coordinates.
\end{example}

\begin{example}[A torus parametrization]\label{ex:vii14-ped-03}
For \(R>r>0\),
\[
X(u,v)=((R+r\cos v)\cos u,(R+r\cos v)\sin u,r\sin v)
\]
has \(X_u\times X_v\neq0\).  The parameters trace the large circular direction and the small meridian direction, giving an explicit regular surface model with periodic coordinates.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Graph regularity]\label{exr:vii14-ped-01}
Show \(X(u,v)=(u,v,u^2+v^2)\) is regular everywhere.
\end{exercise}
\begin{hint}
Compute \(X_u\times X_v\).
\end{hint}
\begin{solution}
\(X_u=(1,0,2u)\), \(X_v=(0,1,2v)\), so \(X_u\times X_v=(-2u,-2v,1)\neq0\).
\end{solution}

\begin{exercise}[Plane tangent vectors]\label{exr:vii14-ped-02}
For \(X(u,v)=(u,v,au+bv)\), find \(X_u\) and \(X_v\).
\end{exercise}
\begin{hint}
Differentiate componentwise.
\end{hint}
\begin{solution}
\(X_u=(1,0,a)\) and \(X_v=(0,1,b)\).
\end{solution}

\begin{exercise}[Sphere tangent plane]\label{exr:vii14-ped-03}
Find the tangent plane to \(S^2\) at \(p=(0,0,1)\).
\end{exercise}
\begin{hint}
Use the normal \(p\).
\end{hint}
\begin{solution}
The tangent plane is \(z=1\) as an affine plane, with tangent vector space \(\{(a,b,0)\}\).
\end{solution}

\begin{exercise}[Cylinder parametrization]\label{exr:vii14-ped-04}
Check regularity of \(X(u,v)=(\cos u,\sin u,v)\).
\end{exercise}
\begin{hint}
Compute the cross product.
\end{hint}
\begin{solution}
\(X_u=(-\sin u,\cos u,0)\), \(X_v=(0,0,1)\), and their cross product \((\cos u,\sin u,0)\) never vanishes.
\end{solution}

\begin{exercise}[Surface of revolution]\label{exr:vii14-ped-05}
Write a parametrization of the surface obtained by rotating \((r(v),z(v))\) about the \(z\)-axis.
\end{exercise}
\begin{hint}
Use the angular variable \(u\).
\end{hint}
\begin{solution}
\(X(u,v)=(r(v)\cos u,r(v)\sin u,z(v))\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Regularity criterion]\label{exr:vii14-ped-06}
Prove \(X\) is a regular parametrization exactly when \(X_u\) and \(X_v\) are linearly independent.
\end{exercise}
\begin{hint}
Relate rank two of \(dX\) to its columns.
\end{hint}
\begin{solution}
The derivative matrix has columns \(X_u,X_v\), so rank two is precisely their linear independence; in \(\mathbb R^3\) this is equivalent to \(X_u\times X_v\neq0\).
\end{solution}

\begin{exercise}[Graph theorem]\label{exr:vii14-ped-07}
Prove every smooth graph in \(\mathbb R^3\) is a regular surface.
\end{exercise}
\begin{hint}
Use the third component of the cross product.
\end{hint}
\begin{solution}
For \(X=(u,v,f)\), \(X_u\times X_v=(-f_u,-f_v,1)\), whose last component is always \(1\).
\end{solution}

\begin{exercise}[Coordinate tangent span]\label{exr:vii14-ped-08}
Prove \(T_pS=\operatorname{span}\{X_u,X_v\}\) for a regular patch \(X\).
\end{exercise}
\begin{hint}
Use \(dX\) on the coordinate basis.
\end{hint}
\begin{solution}
The derivative sends the standard basis of \(\mathbb R^2\) to \(X_u,X_v\), and because \(dX\) has rank two its image is the tangent plane.
\end{solution}

\begin{exercise}[Reparametrization preserves regularity]\label{exr:vii14-ped-09}
Prove \(X\circ\phi\) is regular when \(X\) is regular and \(\phi\) is a local diffeomorphism of parameter domains.
\end{exercise}
\begin{hint}
Use the chain rule for derivatives.
\end{hint}
\begin{solution}
\(d(X\circ\phi)=dX\circ d\phi\); both maps have rank two, so their composition does as well.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Every immersion is an embedded surface]\label{exr:vii14-ped-10}
Test the claim with a self-intersecting parametrized surface.
\end{exercise}
\begin{hint}
Separate local rank from global injectivity/topology.
\end{hint}
\begin{solution}
False.  An immersion can be regular in parameters yet self-intersect, so its image need not be an embedded regular surface near the crossing.
\end{solution}

\begin{exercise}[One chart covers every regular surface]\label{exr:vii14-ped-11}
Test this on the sphere.
\end{exercise}
\begin{hint}
A chart image is open in \(\mathbb R^2\).
\end{hint}
\begin{solution}
False.  Compact surfaces such as \(S^2\) cannot be globally diffeomorphic to an open subset of \(\mathbb R^2\).
\end{solution}

\begin{exercise}[Cross product zero at one point is harmless]\label{exr:vii14-ped-12}
Test the regularity definition.
\end{exercise}
\begin{hint}
Rank must be two at every point of the patch.
\end{hint}
\begin{solution}
False.  A vanishing cross product means the parametrization fails to be regular at that point.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Tangent plane from data]\label{exr:vii14-ped-13}
Explain how a parametrized surface patch yields a tangent plane for local approximation.
\end{exercise}
\begin{hint}
Linearize \(X(u,v)\) at a parameter point.
\end{hint}
\begin{solution}
The first-order approximation is \(X(u_0,v_0)+aX_u+bX_v\), so the affine tangent plane is the best first-order planar model.
\end{solution}

\begin{exercise}[Surface meshing]\label{exr:vii14-ped-14}
Why are regular parameter patches useful for generating meshes?
\end{exercise}
\begin{hint}
Sample a parameter grid and map it by \(X\).
\end{hint}
\begin{solution}
Regularity prevents first-order collapse of the parameter directions, so local grid cells map to nondegenerate surface elements for sufficiently fine sampling.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Implicit versus parametric surface]\label{exr:vii14-ped-15}
Show a regular level surface \(F^{-1}(c)\subset\mathbb R^3\) can locally be written as a graph.
\end{exercise}
\begin{hint}
Use a nonzero component of \(\nabla F\).
\end{hint}
\begin{solution}
At a regular point some partial derivative is nonzero; the implicit function theorem solves for that coordinate as a smooth function of the other two.
\end{solution}

\begin{exercise}[Torus regularity]\label{exr:vii14-ped-16}
Verify the standard torus parametrization is regular when \(R>r>0\).
\end{exercise}
\begin{hint}
Compute the norm of \(X_u\times X_v\).
\end{hint}
\begin{solution}
Its norm is \(r(R+r\cos v)\), which is strictly positive because \(R-r>0\).
\end{solution}
