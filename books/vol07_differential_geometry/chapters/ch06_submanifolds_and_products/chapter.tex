\chapter{Submanifolds and Products}
\label{ch:vii06-submanifolds-and-products}

Submanifolds are manifolds sitting smoothly inside other manifolds; product manifolds combine independent geometric variables. Their tangent spaces are controlled by two basic linear patterns:
\[
\boxed{T_pF^{-1}(c)=\ker dF_p,\qquad
T_{(p,q)}(M\times N)\cong T_pM\oplus T_qN.}
\]

\section{Embedded submanifolds}
\begin{definition}[Embedded submanifold]\label{def:vii06-embedded}
A subset \(S\subset M^n\) is an embedded \(k\)-dimensional submanifold if every \(p\in S\) has a chart
\[
x:U\to x(U)\subset\mathbb R^n
\]
such that
\[
x(U\cap S)=x(U)\cap(\mathbb R^k\times\{0\}).
\]
Such a chart is called adapted to \(S\).
\end{definition}

\begin{proposition}[Induced smooth structure]\label{prop:vii06-induced-structure}
An embedded submanifold inherits a unique smooth \(k\)-manifold structure for which the inclusion \(i:S\hookrightarrow M\) is a smooth embedding.
\end{proposition}
\begin{proof}
Restrict the first \(k\) coordinate functions of adapted charts to \(S\). Their transition functions are restrictions of smooth ambient transitions.
\end{proof}

\begin{definition}[Embedding]\label{def:vii06-embedding}
A smooth map \(F:N\to M\) is an embedding if it is an immersion and a homeomorphism onto its image with the subspace topology.
\end{definition}

\begin{warning}[Immersed is not embedded]\label{warning:vii06-immersed-embedded}
An injective immersion need not be an embedding unless its topology agrees with the subspace topology. Likewise an immersion may have self-intersections and fail even to be injective.
\end{warning}

\section{Tangent spaces of embedded submanifolds}
\begin{theorem}[Tangent inclusion]\label{thm:vii06-tangent-inclusion}
For an embedded submanifold \(S\subset M\), the differential of inclusion
\[
di_p:T_pS\to T_pM
\]
is injective. We therefore identify \(T_pS\) with its image, a linear subspace of \(T_pM\).
\end{theorem}
\begin{proof}
In adapted coordinates the inclusion is \(u\mapsto(u,0)\), whose derivative is injective.
\end{proof}

\begin{theorem}[Curve characterization]\label{thm:vii06-curve-tangent}
Under this identification,
\[
T_pS=\{\dot\gamma(0):\gamma\text{ is a smooth curve in }S,\ \gamma(0)=p\}\subset T_pM.
\]
\end{theorem}

\section{Regular level sets}
\begin{definition}[Regular value]\label{def:vii06-regular-value}
For \(F:M^n\to N^m\), a point \(q\in N\) is a regular value if \(dF_p\) is surjective for every \(p\in F^{-1}(q)\).
\end{definition}

\begin{theorem}[Regular level-set theorem]\label{thm:vii06-regular-level}
If \(q\) is a regular value of \(F:M^n\to N^m\), then
\[
F^{-1}(q)
\]
is an embedded submanifold of dimension \(n-m\), and for each \(p\in F^{-1}(q)\),
\[
T_pF^{-1}(q)=\ker dF_p.
\]
\end{theorem}
\begin{proof}
Choose charts near \(p\) and \(q\). Surjectivity of the differential gives rank \(m\). The implicit-function/constant-rank normal form supplies local coordinates in which \(F\) is projection onto the last \(m\) coordinates. The level set is therefore a coordinate slice of dimension \(n-m\), and its tangent directions are precisely the kernel of the projection derivative.
\end{proof}

\begin{corollary}[Hypersurfaces]\label{cor:vii06-hypersurface}
If \(f:M^n\to\mathbb R\) and \(c\) is a regular value, then \(f^{-1}(c)\) is an embedded hypersurface and
\[
T_pf^{-1}(c)=\ker df_p.
\]
\end{corollary}

\begin{example}[Sphere]\label{ex:vii06-sphere}
For \(F(x)=\|x\|^2\) on \(\mathbb R^n\), every \(r^2>0\) is a regular value. Hence
\[
S_r^{n-1}=F^{-1}(r^2),
\qquad
T_pS_r^{n-1}=\{v:p\cdot v=0\}.
\]
\end{example}

\begin{example}[Special linear group]\label{ex:vii06-sl}
The determinant map \(\det:GL_n(\mathbb R)\to\mathbb R^\times\) has differential
\[
d(\det)_A(H)=\det(A)\operatorname{tr}(A^{-1}H).
\]
At \(A\in SL_n(\mathbb R)\), this is nonzero, so \(SL_n(\mathbb R)\) is a codimension-one embedded submanifold of \(GL_n(\mathbb R)\) with
\[
T_A SL_n=\{H:\operatorname{tr}(A^{-1}H)=0\}.
\]
\end{example}

\section{Graphs of smooth maps}
\begin{theorem}[Graph theorem]\label{thm:vii06-graph}
For smooth \(F:M\to N\), the graph
\[
\Gamma_F=\{(p,F(p)):p\in M\}
\]
is an embedded submanifold of \(M\times N\), diffeomorphic to \(M\).
\end{theorem}
\begin{proof}
The map \(p\mapsto(p,F(p))\) is an embedding; locally the product chart sends the graph to the graph of a Euclidean smooth function, which can be straightened by \((x,y)\mapsto(x,y-F(x))\).
\end{proof}

\section{Product manifolds}
\begin{theorem}[Product smooth structure]\label{thm:vii06-product-manifold}
If \(M^m\) and \(N^n\) are smooth manifolds, then \(M\times N\) has a canonical smooth structure of dimension \(m+n\), generated by product charts
\[
(U\times V,x\times y),\qquad
(x\times y)(p,q)=(x(p),y(q)).
\]
\end{theorem}
\begin{proof}
Product-chart transitions are products of smooth transition maps, hence smooth.
\end{proof}

\begin{proposition}[Product projections]\label{prop:vii06-projections}
The projections
\[
\pi_M:M\times N\to M,\qquad \pi_N:M\times N\to N
\]
are smooth submersions.
\end{proposition}

\begin{theorem}[Tangent splitting]\label{thm:vii06-tangent-product}
The map
\[
T_{(p,q)}(M\times N)\to T_pM\oplus T_qN,
\qquad
v\mapsto(d\pi_M(v),d\pi_N(v))
\]
is a canonical linear isomorphism.
\end{theorem}
\begin{proof}
In a product chart it is the standard identification
\(\mathbb R^{m+n}\cong\mathbb R^m\oplus\mathbb R^n\). Chart independence follows from the intrinsic projection formula.
\end{proof}

\begin{corollary}[Differential of a product map]\label{cor:vii06-product-map}
For smooth \(F:P\to M\) and \(G:P\to N\), the map
\[
(F,G):P\to M\times N
\]
satisfies
\[
d(F,G)_r(v)=\bigl(dF_r(v),dG_r(v)\bigr).
\]
\end{corollary}

\section{Diagonal and slices}
\begin{proposition}[Diagonal]\label{prop:vii06-diagonal}
The diagonal
\[
\Delta_M=\{(p,p):p\in M\}\subset M\times M
\]
is an embedded submanifold diffeomorphic to \(M\), and
\[
T_{(p,p)}\Delta_M=\{(v,v):v\in T_pM\}.
\]
\end{proposition}

\begin{proposition}[Product slices]\label{prop:vii06-slices}
For fixed \(q\in N\), the subset \(M\times\{q\}\) is an embedded submanifold with tangent space
\[
T_{(p,q)}(M\times\{q\})=T_pM\oplus\{0\}.
\]
\end{proposition}

\section{Transverse intersections}
\begin{definition}[Transverse submanifolds]\label{def:vii06-transverse}
Embedded submanifolds \(A,B\subset M\) meet transversely at \(p\in A\cap B\) if
\[
T_pA+T_pB=T_pM.
\]
\end{definition}

\begin{theorem}[Transverse intersection]\label{thm:vii06-transverse-intersection}
If \(A^a\) and \(B^b\) are embedded submanifolds of \(M^n\) and meet transversely, then \(A\cap B\) is locally an embedded submanifold of dimension \(a+b-n\), with
\[
T_p(A\cap B)=T_pA\cap T_pB.
\]
\end{theorem}
\begin{proof}
Locally describe one submanifold as a regular level set and restrict its defining submersion to the other. Transversality is exactly the surjectivity needed to apply the regular level-set theorem to that restriction.
\end{proof}

\section{Examples}
\begin{example}[Torus as a product]\label{ex:vii06-torus}
The torus \(T^2=S^1\times S^1\) is a smooth \(2\)-manifold, and
\[
T_{(p,q)}T^2\cong T_pS^1\oplus T_qS^1.
\]
\end{example}

\begin{example}[Cylinder]\label{ex:vii06-cylinder}
\(S^1\times\mathbb R\) is a \(2\)-manifold. Its tangent space splits into angular and axial directions.
\end{example}

\section{Exercises}
\begin{exercise}\label{exr:vii06-01}
Show an adapted chart induces a smooth chart on an embedded submanifold.
\end{exercise}
\begin{hint}
Restrict the first coordinates.
\end{hint}
\begin{solution}
The first \(k\) adapted coordinates give a homeomorphism of the submanifold slice onto an open subset of \(\mathbb R^k\), and transitions are smooth restrictions.
\end{solution}

\begin{exercise}\label{exr:vii06-02}
Prove the inclusion of an embedded submanifold is an immersion.
\end{exercise}
\begin{hint}
Use adapted coordinates.
\end{hint}
\begin{solution}
Locally the inclusion is \(u\mapsto(u,0)\), whose derivative is injective.
\end{solution}

\begin{exercise}\label{exr:vii06-03}
Identify \(T_pS\) for the linear subspace \(S=\mathbb R^k\times\{0\}\subset\mathbb R^n\).
\end{exercise}
\begin{hint}
Differentiate curves in the slice.
\end{hint}
\begin{solution}
It is \(\mathbb R^k\times\{0\}\).
\end{solution}

\begin{exercise}\label{exr:vii06-04}
Prove the curve characterization of tangent vectors to an embedded submanifold.
\end{exercise}
\begin{hint}
Use an adapted chart.
\end{hint}
\begin{solution}
Curves in the coordinate slice have zero normal velocity; conversely every vector with zero normal components is realized by a straight-line curve in the slice.
\end{solution}

\begin{exercise}\label{exr:vii06-05}
State the regular-value condition for \(F:M^n\to N^m\).
\end{exercise}
\begin{hint}
Use surjectivity.
\end{hint}
\begin{solution}
For every \(p\in F^{-1}(q)\), the map \(dF_p:T_pM\to T_qN\) must be surjective.
\end{solution}

\begin{exercise}\label{exr:vii06-06}
Compute the dimension of a regular level set.
\end{exercise}
\begin{hint}
Apply rank-nullity.
\end{hint}
\begin{solution}
It is \(n-m\).
\end{solution}

\begin{exercise}\label{exr:vii06-07}
Prove \(T_pF^{-1}(q)\subseteq\ker dF_p\).
\end{exercise}
\begin{hint}
Differentiate the constant curve \(F\circ\gamma=q\).
\end{hint}
\begin{solution}
If \(\dot\gamma(0)\in T_pF^{-1}(q)\), then \(dF_p\dot\gamma(0)=0\).
\end{solution}

\begin{exercise}\label{exr:vii06-08}
Explain why equality \(T_pF^{-1}(q)=\ker dF_p\) follows after regularity.
\end{exercise}
\begin{hint}
Compare dimensions.
\end{hint}
\begin{solution}
Both spaces have dimension \(n-m\), so the inclusion is equality.
\end{solution}

\begin{exercise}\label{exr:vii06-09}
Show \(S^{n-1}\subset\mathbb R^n\) is an embedded submanifold.
\end{exercise}
\begin{hint}
Use \(F(x)=\|x\|^2\).
\end{hint}
\begin{solution}
On \(F^{-1}(1)\), \(dF_x(v)=2x\cdot v\) is nonzero, hence surjective onto \(\mathbb R\).
\end{solution}

\begin{exercise}\label{exr:vii06-10}
Compute \(T_pS^{n-1}\).
\end{exercise}
\begin{hint}
Take the kernel of the constraint differential.
\end{hint}
\begin{solution}
\(T_pS^{n-1}=\{v:p\cdot v=0\}\).
\end{solution}

\begin{exercise}\label{exr:vii06-11}
Show the hyperboloid \(x^2+y^2-z^2=1\) is a smooth surface.
\end{exercise}
\begin{hint}
Check the gradient of the constraint on the level.
\end{hint}
\begin{solution}
The differential \((2x,2y,-2z)\) cannot vanish on the level \(1\), so the level is regular.
\end{solution}

\begin{exercise}\label{exr:vii06-12}
Show the graph of a smooth function between Euclidean spaces is an embedded submanifold.
\end{exercise}
\begin{hint}
Straighten \((x,y)\mapsto(x,y-F(x))\).
\end{hint}
\begin{solution}
This local diffeomorphism sends the graph to a coordinate slice.
\end{solution}

\begin{exercise}\label{exr:vii06-13}
Prove the graph map \(p\mapsto(p,F(p))\) is an immersion.
\end{exercise}
\begin{hint}
Differentiate it.
\end{hint}
\begin{solution}
Its differential is \(v\mapsto(v,dF_pv)\), which is injective.
\end{solution}

\begin{exercise}\label{exr:vii06-14}
Show product-chart transitions are smooth.
\end{exercise}
\begin{hint}
Write them componentwise.
\end{hint}
\begin{solution}
They are \((x'\circ x^{-1},y'\circ y^{-1})\), a product of smooth Euclidean maps.
\end{solution}

\begin{exercise}\label{exr:vii06-15}
Prove \(\dim(M\times N)=\dim M+\dim N\).
\end{exercise}
\begin{hint}
Use product charts.
\end{hint}
\begin{solution}
A product chart lands in \(\mathbb R^{m+n}\).
\end{solution}

\begin{exercise}\label{exr:vii06-16}
Show both product projections are submersions.
\end{exercise}
\begin{hint}
Use the tangent splitting.
\end{hint}
\begin{solution}
Their differentials are the canonical coordinate projections, hence surjective.
\end{solution}

\begin{exercise}\label{exr:vii06-17}
Prove the tangent splitting using product coordinates.
\end{exercise}
\begin{hint}
Write the coordinate basis.
\end{hint}
\begin{solution}
The first block of basis vectors comes from \(M\), the second from \(N\), yielding a direct sum.
\end{solution}

\begin{exercise}\label{exr:vii06-18}
Compute \(d(F,G)_p\).
\end{exercise}
\begin{hint}
Compose with the two projections.
\end{hint}
\begin{solution}
Since \(\pi_M\circ(F,G)=F\) and \(\pi_N\circ(F,G)=G\), the tangent splitting gives \(d(F,G)=(dF,dG)\).
\end{solution}

\begin{exercise}\label{exr:vii06-19}
Prove the diagonal is an embedded submanifold.
\end{exercise}
\begin{hint}
Use \(p\mapsto(p,p)\).
\end{hint}
\begin{solution}
The diagonal map is an embedding; in product coordinates a linear change \((x,y)\mapsto(x,y-x)\) straightens it.
\end{solution}

\begin{exercise}\label{exr:vii06-20}
Compute the tangent space of the diagonal.
\end{exercise}
\begin{hint}
Differentiate the diagonal map.
\end{hint}
\begin{solution}
It is \(\{(v,v):v\in T_pM\}\).
\end{solution}

\begin{exercise}\label{exr:vii06-21}
Show \(M\times\{q\}\) is embedded and compute its tangent space.
\end{exercise}
\begin{hint}
Use a product chart.
\end{hint}
\begin{solution}
It is a coordinate slice with tangent space \(T_pM\oplus\{0\}\).
\end{solution}

\begin{exercise}\label{exr:vii06-22}
For \(T^2=S^1\times S^1\), compute its tangent dimension and splitting.
\end{exercise}
\begin{hint}
Use the product theorem.

Check that two distinct coordinate axes in \(\mathbb R^2\) meet transversely at the origin.

\section{Solved problem dossiers}
\begin{problem}[Legacy submanifold problem: adapted coordinates]\label{prob:vii06-01}
Prove that the coordinate-slice definition gives an intrinsic smooth structure on an embedded submanifold.
\end{problem}
\begin{solution}
Restrict adapted charts to the slice. Ambient transition maps preserve the subset and restrict to smooth transitions; maximal completion gives the intrinsic structure.
\end{solution}

\begin{problem}[Legacy submanifold problem: tangent inclusion]\label{prob:vii06-02}
Show the inclusion of an embedded submanifold identifies its tangent space with a linear subspace of the ambient tangent space.
\end{problem}
\begin{solution}
In adapted coordinates \(di_p\) is the standard injection \(\mathbb R^k\hookrightarrow\mathbb R^n\).
\end{solution}

\begin{problem}[Legacy regular-level problem]\label{prob:vii06-03}
Prove that a regular level \(F^{-1}(q)\) is an embedded submanifold with tangent space \(\ker dF_p\).
\end{problem}
\begin{solution}
The implicit-function normal form makes \(F\) a coordinate projection. Its level is a coordinate slice and its tangent space is exactly the kernel of the projection derivative.
\end{solution}

\begin{problem}[Legacy product-manifold problem]\label{prob:vii06-04}
Construct the product smooth structure and prove its dimension.
\end{problem}
\begin{solution}
Use product charts; their transitions are products of smooth transitions, and their target dimension is \(m+n\).
\end{solution}

\begin{problem}[Legacy product tangent problem]\label{prob:vii06-05}
Prove the canonical tangent splitting for a product.
\end{problem}
\begin{solution}
Differentiate the two projections. In product coordinates the resulting map is the standard direct-sum identification.
\end{solution}

\begin{problem}[Legacy graph problem]\label{prob:vii06-06}
Show the graph of a smooth map is an embedded submanifold.
\end{problem}
\begin{solution}
The graph map is an embedding, or locally straighten the graph by \((x,y)\mapsto(x,y-F(x))\).
\end{solution}

\begin{problem}[Sphere as a regular level]\label{prob:vii06-07}
Recover both the manifold structure and tangent space of a sphere from one constraint.
\end{problem}
\begin{solution}
The value \(r^2>0\) of \(\|x\|^2\) is regular and the tangent kernel is \(p^\perp\).
\end{solution}

\begin{problem}[Matrix-group level set]\label{prob:vii06-08}
Show \(SL_n(\mathbb R)\) is a submanifold of \(GL_n(\mathbb R)\) and compute its tangent space.
\end{problem}
\begin{solution}
Use the regular level \(\det=1\) and \(d(\det)_A(H)=\det(A)\operatorname{tr}(A^{-1}H)\). The kernel gives the tangent space.
\end{solution}

\begin{problem}[Diagonal submanifold]\label{prob:vii06-09}
Show the diagonal is embedded and identify its tangent space.
\end{problem}
\begin{solution}
Differentiate \(p\mapsto(p,p)\): the image is the diagonal subspace \(\{(v,v)\}\).
\end{solution}

\begin{problem}[Product slice]\label{prob:vii06-10}
Show \(M\times\{q\}\) is canonically diffeomorphic to \(M\) and compute the inclusion differential.
\end{problem}
\begin{solution}
The map \(p\mapsto(p,q)\) is an embedding with differential \(v\mapsto(v,0)\).
\end{solution}

\begin{problem}[Transverse intersection]\label{prob:vii06-11}
Derive the tangent-space formula for a transverse intersection.
\end{problem}
\begin{solution}
Restrict a local defining submersion for one submanifold to the other; transversality gives surjectivity, and the regular-level theorem yields the intersection and tangent kernel.
\end{solution}

\begin{problem}[Embedding versus immersion]\label{prob:vii06-12}
Explain precisely why immersion alone does not guarantee an embedded submanifold image.
\end{problem}
\begin{solution}
Immersion controls only the differential. An embedding additionally requires global injectivity and the correct topology on the image; periodic immersed curves provide the simplest failure of injectivity.
\end{solution}

\section{Challenge problems}
\begin{problem}[Preimage theorem in full form]\label{prob:vii06-ch01}
If \(S\subset N\) is an embedded submanifold and \(F:M\to N\) is transverse to \(S\), show \(F^{-1}(S)\) is an embedded submanifold and compute its tangent space.
\end{problem}
\begin{solution}
Use adapted coordinates for \(S\), take the normal-coordinate projection as a local defining submersion, and compose with \(F\). Transversality makes the composite a submersion. Thus \(F^{-1}(S)\) is regular and \(T_pF^{-1}(S)=\{v:dF_pv\in T_{F(p)}S\}\).
\end{solution}

\begin{problem}[Fiber product preview]\label{prob:vii06-ch02}
For transverse maps \(F:M\to P\) and \(G:N\to P\), show \(M\times_PN=\{(m,n):F(m)=G(n)\}\) is locally a submanifold.
\end{problem}
\begin{solution}
Apply the preimage theorem to \(F\times G:M\times N\to P\times P\) and the diagonal \(\Delta_P\). Transversality to the diagonal is the required condition.
\end{solution}

\begin{problem}[Codimension and conormal space]\label{prob:vii06-ch03}
For a regular level \(S=F^{-1}(q)\) with \(F:M^n\to\mathbb R^k\), identify \((T_pS)^\circ\).
\end{problem}
\begin{solution}
Since \(T_pS=\ker dF_p\), VII/05 gives \((T_pS)^\circ=\operatorname{im}(dF_p^*)\), spanned by the differentials of the component constraints.
\end{solution}

\begin{problem}[Products preserve embeddings]\label{prob:vii06-ch04}
If \(i:A\hookrightarrow M\) and \(j:B\hookrightarrow N\) are embeddings, prove \(i\times j:A\times B\hookrightarrow M\times N\) is an embedding.
\end{problem}
\begin{solution}
The product map is an injective immersion because its differential is the direct sum of injective differentials, and the product of the two homeomorphisms onto their images is a homeomorphism onto the product image.
\end{solution}

\begin{problem}[Transversality dimension count]\label{prob:vii06-ch05}
Explain algebraically why \(\dim(T_pA\cap T_pB)=a+b-n\) under transversality.
\end{problem}
\begin{solution}
For subspaces \(U,V\subset E\), \(\dim(U+V)=\dim U+\dim V-\dim(U\cap V)\). Setting \(U+V=E\) gives the formula.
\end{solution}

\section*{Bridge to VII/07}
The spaces \(T_pM\) now vary point by point and interact cleanly with products and embedded submanifolds. VII/07 assembles these fibers into vector bundles, beginning with the tangent and cotangent bundles and then abstracting local triviality, bundle charts, sections, and bundle morphisms.
