\chapter{The Poincaré Metric}
\label{ch:vii32-poincare}

VII/31 introduced the upper half-plane and disk as coordinate models of the
hyperbolic plane.  The geometry becomes hyperbolic only after we equip these
domains with the Poincaré metric.

\section{Upper half-plane metric}

\begin{definition}[Poincaré metric on the half-plane]\label{def:vii32-half-metric}
On
\[
\mathbb H=\{z=x+iy:y>0\},
\]
define
\[
g_{\mathbb H}
=
\frac{dx^2+dy^2}{y^2}.
\]
Equivalently,
\[
ds_{\mathbb H}
=
\frac{|dz|}{\operatorname{Im}z}.
\]
\end{definition}

\begin{proposition}[Conformality]\label{prop:vii32-conformal}
The Poincaré metric is conformal to the Euclidean metric, with conformal factor
\(y^{-2}\).  Hence hyperbolic angles agree with Euclidean angles.
\end{proposition}

\section{Lengths and area}

For a smooth curve \(\gamma(t)=x(t)+iy(t)\),
\[
L_{\mathbb H}(\gamma)
=
\int
\frac{\sqrt{\dot x^2+\dot y^2}}{y}\,dt.
\]

\begin{definition}[Hyperbolic area]\label{def:vii32-area}
The Riemannian area form is
\[
dA_{\mathbb H}
=
\frac{dx\,dy}{y^2}.
\]
\end{definition}

\begin{remark}
The factor \(y^{-1}\) in length and \(y^{-2}\) in area makes the Euclidean
boundary \(y=0\) infinitely far away in the hyperbolic metric.
\end{remark}

\section{Vertical geodesics}

Consider the vertical curve
\[
\gamma(t)=x_0+i\,y(t).
\]

\begin{proposition}[Vertical distance]\label{prop:vii32-vertical}
For \(0<y_1<y_2\),
\[
d_{\mathbb H}(x_0+iy_1,x_0+iy_2)
=
\log\frac{y_2}{y_1}.
\]
\end{proposition}

\begin{proof}
The vertical segment has length
\[
\int_{y_1}^{y_2}\frac{dy}{y}
=
\log\frac{y_2}{y_1}.
\]
The geodesic equations or isometric normalization developed below show that this
segment is minimizing.
\end{proof}

\section{Christoffel symbols and geodesic equations}

For
\[
g_{xx}=g_{yy}=y^{-2},
\qquad
g_{xy}=0,
\]
the nonzero Christoffel symbols are
\[
\Gamma^x_{xy}=\Gamma^x_{yx}=-\frac1y,
\qquad
\Gamma^y_{xx}=\frac1y,
\qquad
\Gamma^y_{yy}=-\frac1y.
\]

\begin{proposition}[Geodesic equations]\label{prop:vii32-geodesic-equations}
An affinely parametrized geodesic satisfies
\[
\ddot x-\frac{2}{y}\dot x\dot y=0,
\]
\[
\ddot y+\frac1y(\dot x^2-\dot y^2)=0.
\]
\end{proposition}

\section{Geodesic shapes}

\begin{theorem}[Half-plane geodesics]\label{thm:vii32-half-geodesics}
The unparametrized geodesics of \((\mathbb H,g_{\mathbb H})\) are precisely
\[
\text{vertical lines}
\qquad\text{and}\qquad
\text{Euclidean semicircles orthogonal to }\mathbb R.
\]
\end{theorem}

\begin{remark}
This proves the model statement previewed in VII/31.  Orthogonality is Euclidean
because the metric is conformal.
\end{remark}

\section{Hyperbolic distance formula}

\begin{theorem}[Distance in the upper half-plane]\label{thm:vii32-distance}
For \(z,w\in\mathbb H\),
\[
\cosh d_{\mathbb H}(z,w)
=
1+\frac{|z-w|^2}{2\,\operatorname{Im}z\,\operatorname{Im}w}.
\]
Equivalently,
\[
\sinh\frac{d_{\mathbb H}(z,w)}2
=
\frac{|z-w|}
{2\sqrt{\operatorname{Im}z\,\operatorname{Im}w}}.
\]
\end{theorem}

\begin{example}[Points on one vertical line]\label{ex:vii32-vertical-distance}
For \(z=iy_1\) and \(w=iy_2\),
\[
d_{\mathbb H}(z,w)
=
\left|\log\frac{y_2}{y_1}\right|.
\]
\end{example}

\section{The disk metric}

Let
\[
\mathbb D=\{z:|z|<1\}.
\]

\begin{definition}[Poincaré disk metric]\label{def:vii32-disk-metric}
The disk metric of curvature \(-1\) is
\[
g_{\mathbb D}
=
\frac{4\,|dz|^2}{(1-|z|^2)^2}.
\]
Thus
\[
ds_{\mathbb D}
=
\frac{2|dz|}{1-|z|^2}.
\]
\end{definition}

\begin{proposition}[Cayley isometry]\label{prop:vii32-cayley-isometry}
The Cayley transform
\[
C(z)=i\frac{1+z}{1-z}
\]
satisfies
\[
C^*g_{\mathbb H}=g_{\mathbb D}.
\]
Hence it is an isometry between the two Poincaré models.
\end{proposition}

\section{Disk distance}

\begin{theorem}[Distance in the disk]\label{thm:vii32-disk-distance}
For \(z,w\in\mathbb D\),
\[
\tanh\frac{d_{\mathbb D}(z,w)}2
=
\left|
\frac{z-w}{1-\overline w z}
\right|.
\]
In particular,
\[
d_{\mathbb D}(0,r)
=
2\,\operatorname{artanh}r
=
\log\frac{1+r}{1-r}.
\]
\end{theorem}

\section{Infinite distance to the ideal boundary}

\begin{proposition}[Completeness at the boundary]\label{prop:vii32-boundary-infinite}
As \(r\to1^{-}\),
\[
d_{\mathbb D}(0,r)\to\infty.
\]
Likewise, approaching the real axis in the half-plane requires infinite
hyperbolic length.
\end{proposition}

\begin{remark}
The ideal boundary is therefore a genuine boundary at infinity, not a missing
finite-distance edge of the surface.
\end{remark}

\section{Gaussian curvature}

For a conformal metric
\[
g=e^{2u}(dx^2+dy^2)
\]
on a surface,
\[
K=-e^{-2u}\Delta u.
\]

\begin{theorem}[Constant curvature]\label{thm:vii32-curvature}
The Poincaré half-plane and disk metrics both have
\[
K=-1.
\]
\end{theorem}

\begin{proof}
For the half-plane,
\[
u=-\log y.
\]
Hence
\[
\Delta u=\frac1{y^2},
\qquad
e^{-2u}=y^2,
\]
so
\[
K=-y^2\frac1{y^2}=-1.
\]
The disk result follows from the Cayley isometry.
\end{proof}

\section{Hyperbolic circles}

\begin{proposition}[Euclidean appearance of hyperbolic circles]\label{prop:vii32-circles}
Metric circles in the Poincaré disk are Euclidean circles, although their
Euclidean center generally differs from their hyperbolic center.
\end{proposition}

\begin{example}[Circle centered at the origin]\label{ex:vii32-origin-circle}
The hyperbolic circle of radius \(R\) centered at \(0\) is the Euclidean circle
\[
|z|=\tanh\frac R2.
\]
\end{example}

\section{Area growth}

A hyperbolic disk of radius \(R\) has area
\[
A(R)=2\pi(\cosh R-1)
=
4\pi\sinh^2\frac R2.
\]

\begin{remark}
Exponential area growth is one of the clearest quantitative signatures of
negative curvature and sharply contrasts with the Euclidean formula \(\pi R^2\).
\end{remark}

\section{Solved problem dossiers}
\begin{problem}[Vertical-distance problem]\label{prob:vii32-01}
Derive the logarithmic distance formula along a vertical geodesic.
\end{problem}
\begin{solution}
Restrict the line element to \(dx=0\), obtaining \(ds=dy/y\), then integrate.
\end{solution}

\begin{problem}[Christoffel problem]\label{prob:vii32-02}
Derive the nonzero Christoffel symbols of \(g=y^{-2}(dx^2+dy^2)\).
\end{problem}
\begin{solution}
Use \(g^{ij}=y^2\delta^{ij}\) and the Levi--Civita formula; only \(y\)-derivatives are nonzero.
\end{solution}

\begin{problem}[Geodesic-equation problem]\label{prob:vii32-03}
Derive the coordinate geodesic equations.
\end{problem}
\begin{solution}
Insert the four nonzero Christoffel symbols into \(\ddot x^k+\Gamma^k_{ij}\dot x^i\dot x^j=0\).
\end{solution}

\begin{problem}[Curvature problem]\label{prob:vii32-04}
Compute \(K\) directly from the conformal factor.
\end{problem}
\begin{solution}
Write \(u=-\log y\), compute \(\Delta u=1/y^2\), and substitute in \(K=-e^{-2u}\Delta u\).
\end{solution}

\begin{problem}[Cayley-isometry problem]\label{prob:vii32-05}
Verify \(C^*g_{\mathbb H}=g_{\mathbb D}\).
\end{problem}
\begin{solution}
Use \(C'(z)=2i/(1-z)^2\) and \(\Im C(z)=(1-|z|^2)/|1-z|^2\); all \(|1-z|\) factors cancel.
\end{solution}

\begin{problem}[Disk-radial problem]\label{prob:vii32-06}
Derive \(d_{\mathbb D}(0,r)\).
\end{problem}
\begin{solution}
Along a radius, integrate \(2\,dr/(1-r^2)\) from \(0\) to \(r\).
\end{solution}

\begin{problem}[Boundary problem]\label{prob:vii32-07}
Prove radial approach to the unit circle has infinite length.
\end{problem}
\begin{solution}
The integral \(\int^{1}2\,dr/(1-r^2)\) diverges logarithmically.
\end{solution}

\begin{problem}[Distance-identity problem]\label{prob:vii32-08}
Show the two half-plane distance formulas are equivalent.
\end{problem}
\begin{solution}
Use \(\cosh d=1+2\sinh^2(d/2)\).
\end{solution}

\begin{problem}[Circle problem]\label{prob:vii32-09}
Find the Euclidean circle representing a hyperbolic circle centered at \(0\).
\end{problem}
\begin{solution}
Set \(d(0,z)=R\), so \(|z|=\tanh(R/2)\).
\end{solution}

\begin{problem}[Area problem]\label{prob:vii32-10}
Compute the area of a disk centered at the origin using disk coordinates.
\end{problem}
\begin{solution}
Integrate \(4r\,dr\,d\theta/(1-r^2)^2\) up to \(r=\tanh(R/2)\) and simplify.
\end{solution}

\begin{problem}[Completeness problem]\label{prob:vii32-11}
Explain why the Poincaré disk is complete.
\end{problem}
\begin{solution}
A curve escaping every compact set must approach \(|z|=1\), but the boundary is at infinite metric distance; Cauchy sequences therefore cannot escape through it.
\end{solution}

\begin{problem}[Conformal-angle problem]\label{prob:vii32-12}
Prove Euclidean orthogonality to the model boundary is meaningful hyperbolically.
\end{problem}
\begin{solution}
A positive conformal factor multiplies all inner products by the same scalar at a point, leaving angle cosines unchanged.
\end{solution}

\section{Challenge problems}
\begin{problem}[Distance from invariance]\label{prob:vii32-ch01}
Outline how one can derive the full half-plane distance formula by mapping one point to \(i\).
\end{problem}
\begin{solution}
Use a real Möbius isometry to send the geodesic through \(z,w\) to a vertical line, compute logarithmic distance there, and rewrite the result in an invariant algebraic form.
\end{solution}

\begin{problem}[Hyperbolic circumference]\label{prob:vii32-ch02}
Compute the circumference of a hyperbolic circle of radius \(R\).
\end{problem}
\begin{solution}
Differentiate the disk-area formula or integrate the metric along \(|z|=\tanh(R/2)\); the result is \(2\pi\sinh R\).
\end{solution}

\begin{problem}[Small-radius asymptotics]\label{prob:vii32-ch03}
Show hyperbolic disk area agrees with Euclidean area to leading order.
\end{problem}
\begin{solution}
Expand \(\cosh R=1+R^2/2+O(R^4)\), giving \(A(R)=\pi R^2+O(R^4)\).
\end{solution}

\begin{problem}[Curvature scaling]\label{prob:vii32-ch04}
If the metric is multiplied by \(a^2\), what is the new curvature?
\end{problem}
\begin{solution}
Constant metric scaling multiplies sectional curvature by \(a^{-2}\), so \(a^2g_{\mathbb H}\) has curvature \(-1/a^2\).
\end{solution}

\begin{problem}[Ideal-boundary metric effect]\label{prob:vii32-ch05}
Why can Euclidean points very near the ideal boundary be hyperbolically far apart even when their Euclidean separation is tiny?
\end{problem}
\begin{solution}
The line element magnifies Euclidean displacement by \(1/y\) in the half-plane or \(2/(1-|z|^2)\) in the disk, which diverges near the boundary.
\end{solution}

\section*{Bridge to VII/33}
VII/33 identifies the Möbius transformations whose derivative and imaginary-part factors cancel exactly in the Poincaré line element, producing the group \(PSL(2,\mathbb R)\) of orientation-preserving isometries.
