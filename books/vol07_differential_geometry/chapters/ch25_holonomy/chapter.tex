\chapter{Holonomy}
\label{ch:vii25-holonomy}

The previous four chapters form a single geometric mechanism:
\[
\boxed{
\text{connection}
\longrightarrow
\text{Levi--Civita connection}
\longrightarrow
\text{geodesics}
\longrightarrow
\text{parallel transport}
\longrightarrow
\text{holonomy}.
}
\]
A connection tells us how to differentiate vector fields.  The Levi--Civita
connection makes that rule canonical for a Riemannian metric.  Geodesics are curves
whose tangent vector transports parallel to itself.  Parallel transport compares
vectors along an arbitrary path.  Holonomy records what happens when the path
closes into a loop.

\section{Transport around loops}

Fix a Riemannian manifold \((M,g)\) and a base point \(p\in M\).

\begin{definition}[Holonomy transformation]\label{def:vii25-hol-transform}
If \(\gamma:[0,1]\to M\) is a piecewise smooth loop with
\(\gamma(0)=\gamma(1)=p\), its Levi--Civita parallel transport is an orthogonal
linear map
\[
P_\gamma:T_pM\to T_pM.
\]
This map is called the holonomy transformation of \(\gamma\).
\end{definition}

\begin{definition}[Holonomy group]\label{def:vii25-hol-group}
The holonomy group at \(p\) is
\[
\operatorname{Hol}_p(M,g)
=
\{P_\gamma:\gamma\text{ is a piecewise smooth loop based at }p\}.
\]
\end{definition}

\begin{theorem}[Group property]\label{thm:vii25-group}
\(\operatorname{Hol}_p(M,g)\) is a subgroup of the orthogonal group
\[
O(T_pM,g_p).
\]
\end{theorem}

\begin{proof}
The constant loop gives the identity.  Concatenation of loops corresponds to
composition of transport maps, and reversing a loop gives the inverse transport.
Levi--Civita transport preserves the inner product, so every element is orthogonal.
\end{proof}

\section{Why holonomy is genuinely geometric}

\begin{remark}[Path dependence becomes structure]\label{rem:vii25-path}
Parallel transport between fixed endpoints may depend on the path.  Holonomy does
not regard that dependence as a defect: it packages all loop-dependent
discrepancies into a group.
\end{remark}

\begin{example}[Euclidean space]\label{ex:vii25-euclidean}
In \(\mathbb R^n\) with its standard metric, Cartesian parallel transport keeps
components constant.  Hence every loop has
\[
P_\gamma=\operatorname{id},
\qquad
\operatorname{Hol}_p(\mathbb R^n)=\{I\}.
\]
\end{example}

\section{Base-point dependence}

Let \(p,q\) lie in the same connected component and let
\(\sigma\) be a path from \(p\) to \(q\).

\begin{theorem}[Base-point conjugacy]\label{thm:vii25-basepoint}
Parallel transport along \(\sigma\) conjugates the two holonomy groups:
\[
\operatorname{Hol}_q
=
P_\sigma\,
\operatorname{Hol}_p\,
P_\sigma^{-1}.
\]
\end{theorem}

\begin{proof}
Given a loop \(\gamma\) based at \(p\), form the loop at \(q\)
\[
\sigma^{-1}*\gamma*\sigma
\]
with the convention that paths are traversed from right to left in the displayed
composition.  The concatenation law for parallel transport gives the conjugation
formula.
\end{proof}

\begin{remark}
Thus the holonomy group is canonically attached to a base point only up to
conjugacy.  Its abstract isomorphism type is constant on a connected component.
\end{remark}

\section{Restricted holonomy}

\begin{definition}[Restricted holonomy]\label{def:vii25-restricted}
The restricted holonomy group
\[
\operatorname{Hol}^0_p(M,g)
\]
is generated by holonomy transformations of loops based at \(p\) that are
homotopic to the constant loop through based loops.
\end{definition}

\begin{proposition}[Normal subgroup]\label{prop:vii25-normal}
\(\operatorname{Hol}^0_p\) is a normal subgroup of
\(\operatorname{Hol}_p\).
\end{proposition}

\begin{proof}
Conjugating a contractible loop by any based loop again gives a contractible loop
after transporting the contraction along the conjugating path.
\end{proof}

\begin{remark}[Local versus global information]\label{rem:vii25-local-global}
Restricted holonomy is controlled by local differential geometry.  The quotient
\[
\operatorname{Hol}_p/\operatorname{Hol}^0_p
\]
can also contain information arising from the global topology of the manifold.
\end{remark}

\section{Flat connections and topology}

\begin{proposition}[Parallel frame criterion]\label{prop:vii25-parallel-frame}
If a connected open set \(U\) admits a global orthonormal frame
\(E_1,\dots,E_n\) satisfying
\[
\nabla E_i=0,
\]
then parallel transport in \(U\) is path independent and every loop in \(U\) has
trivial holonomy.
\end{proposition}

\begin{proof}
Write any vector as \(V=v^iE_i\).  Along a curve,
\[
\frac{DV}{dt}=\dot v^iE_i.
\]
Thus parallel fields are exactly those with constant coefficients in the parallel
frame.
\end{proof}

\begin{warning}[Flat does not automatically mean globally trivial]\label{warn:vii25-flat-global}
Vanishing curvature is a local statement.  On a nonsimply connected manifold a
flat connection may still have nontrivial full holonomy because loops can carry
global monodromy.
\end{warning}

\section{A two-dimensional picture}

On an oriented Riemannian surface, every tangent plane is an oriented Euclidean
two-plane.  Hence an orientation-preserving holonomy transformation is a rotation.

\begin{proposition}[Surface holonomy angle]\label{prop:vii25-surface-angle}
For an oriented Riemannian surface, the restricted holonomy group is contained in
\[
SO(2).
\]
Thus sufficiently small contractible loops act by rotations of the tangent plane.
\end{proposition}

\section{Sphere: transport detects curvature}

\begin{example}[Spherical triangle]\label{ex:vii25-sphere-triangle}
On the unit sphere, parallel transport around a positively oriented geodesic
triangle returns a tangent vector rotated by the spherical excess
\[
\alpha+\beta+\gamma-\pi.
\]
For the unit sphere this equals the area of the triangle.
\end{example}

\begin{remark}[Gauss--Bonnet preview]\label{rem:vii25-gauss-bonnet}
The equality between holonomy angle and enclosed curvature is a first glimpse of
Gauss--Bonnet.  The present chapter uses it as geometric motivation; the systematic
curvature tensor comes first in VII/26.
\end{remark}

\section{Small loops and infinitesimal holonomy}

Holonomy around a very small loop is close to the identity.  Its leading
nontrivial deviation is governed by curvature.

\begin{theorem}[Infinitesimal holonomy preview]\label{thm:vii25-infinitesimal}
Let \(X,Y\in T_pM\), and consider a sufficiently small coordinate parallelogram
with oriented area \(\varepsilon^2\) tangent to \(X,Y\).  Parallel transport around
its boundary has the schematic expansion
\[
P_{\partial\Box_\varepsilon}
=
I+\varepsilon^2 R(X,Y)+O(\varepsilon^3),
\]
up to the sign convention used in defining the curvature operator.
\end{theorem}

\begin{remark}
VII/26 will define \(R(X,Y)\) precisely and derive the algebraic identities of the
Riemann curvature tensor.  The present statement explains conceptually why
curvature is the infinitesimal generator of holonomy.
\end{remark}

\section{Holonomy of a connection on a vector bundle}

The construction is not limited to the tangent bundle.

\begin{definition}[Bundle holonomy]\label{def:vii25-bundle-holonomy}
Let \(E\to M\) be a vector bundle with connection.  For a loop \(\gamma\) based at
\(p\), parallel transport gives
\[
P_\gamma:E_p\to E_p.
\]
The subgroup of \(GL(E_p)\) generated by all such maps is the holonomy group of the
connection on \(E\).
\end{definition}

If the connection preserves a fiber metric, its holonomy lies in the corresponding
orthogonal or unitary group.

\section{Principal-bundle holonomy}

Let \(P\to M\) be a principal \(G\)-bundle with connection.

\begin{definition}[Principal holonomy]\label{def:vii25-principal}
Choose \(u\in P_p\).  Horizontally lift a based loop \(\gamma\) beginning at \(u\).
Its endpoint lies in the same fiber and therefore has the form
\[
u\cdot h_\gamma
\]
for a unique \(h_\gamma\in G\).  The elements \(h_\gamma\) form the principal
holonomy group based at \(u\).
\end{definition}

\begin{proposition}[Change of point in the fiber]\label{prop:vii25-fiber-conjugacy}
Replacing \(u\) by \(u\cdot g\) conjugates the principal holonomy subgroup by
\(g^{-1}(\cdot)g\).
\end{proposition}

\section{Holonomy and invariant tensors}

\begin{theorem}[Parallel tensors give holonomy invariants]\label{thm:vii25-parallel-invariant}
If a tensor field \(T\) is parallel,
\[
\nabla T=0,
\]
then its value \(T_p\) is fixed by the natural action of
\(\operatorname{Hol}_p\).
\end{theorem}

\begin{proof}
Parallel transport carries \(T_p\) to \(T_q\) along every path.  For a loop based
at \(p\), the endpoint is again \(p\), so the induced holonomy action must fix
\(T_p\).
\end{proof}

\begin{remark}[Reverse direction]\label{rem:vii25-reverse}
On suitable connected domains, holonomy-invariant fiber data can be extended by
parallel transport to parallel tensor fields.  This principle is one of the main
reasons holonomy is so powerful.
\end{remark}

\section{Riemannian holonomy and special geometry}

For an \(n\)-dimensional Riemannian manifold,
\[
\operatorname{Hol}_p\subseteq O(n),
\]
and on an oriented manifold the restricted holonomy lies in \(SO(n)\).  A proper
reduction of holonomy means that parallel transport preserves additional algebraic
structure beyond the metric.

\begin{example}[Parallel complex structure preview]\label{ex:vii25-complex}
If a Riemannian manifold carries a parallel orthogonal complex structure \(J\),
then holonomy preserves \(J\), so the holonomy group is contained in a unitary
group.
\end{example}

\begin{remark}
This observation leads toward Kähler, Calabi--Yau, \(G_2\), and
\(\operatorname{Spin}(7)\) geometry.  Those theories lie beyond the present volume,
but holonomy is the organizing language behind all of them.
\end{remark}

\section{The five-chapter arc}

The conceptual chain can now be read backwards as well as forwards:
\[
\begin{array}{c}
\text{holonomy measures accumulated parallel transport},\\[2mm]
\text{parallel transport integrates a connection along curves},\\[2mm]
\text{geodesics parallel-transport their own tangent vectors},\\[2mm]
\text{Levi--Civita makes the connection canonical from the metric},\\[2mm]
\text{general connections provide the differentiation framework}.
\end{array}
\]
The next chapter identifies the infinitesimal obstruction responsible for
nontrivial local holonomy: the Riemann curvature tensor.

\section{Solved problem dossiers}

\begin{problem}[Holonomy-group problem]\label{prob:vii25-01}
Prove the set of Levi--Civita loop transports at \(p\) forms a subgroup of \(O(T_pM)\).
\end{problem}
\begin{solution}
The constant loop gives identity, concatenation gives composition, reversal gives inverses, and metric compatibility makes every transport orthogonal.
\end{solution}


\begin{problem}[Base-point problem]\label{prob:vii25-02}
Derive the conjugacy relation between holonomy groups at two points.
\end{problem}
\begin{solution}
Join the base points by a path \(\sigma\), conjugate loops using \(\sigma\), and apply the parallel-transport concatenation rule.
\end{solution}


\begin{problem}[Restricted-holonomy problem]\label{prob:vii25-03}
Show contractible-loop transports form a normal subgroup after generation.
\end{problem}
\begin{solution}
Conjugating a contractible loop by any based loop yields another contractible loop, so the generated subgroup is normal.
\end{solution}


\begin{problem}[Flat-frame problem]\label{prob:vii25-04}
Show a global parallel frame implies path-independent transport.
\end{problem}
\begin{solution}
Every vector has constant coefficients in the parallel frame along a parallel extension; the endpoint depends only on those coefficients, not on the path.
\end{solution}


\begin{problem}[Sphere-triangle problem]\label{prob:vii25-05}
Explain geometrically why transport around a spherical triangle produces a rotation.
\end{problem}
\begin{solution}
Transport preserves the oriented metric, so the result lies in \(SO(2)\). Following the vector through the three geodesic edges yields a net turning equal to the spherical excess.
\end{solution}


\begin{problem}[Small-loop problem]\label{prob:vii25-06}
Why is holonomy around a shrinking loop close to the identity?
\end{problem}
\begin{solution}
The parallel ODE integrates a smooth connection over a path whose size goes to zero. First-order edge effects cancel around a closed infinitesimal parallelogram, leaving curvature at second order.
\end{solution}


\begin{problem}[Bundle-holonomy problem]\label{prob:vii25-07}
Generalize tangent-bundle holonomy to a connection on \(E\to M\).
\end{problem}
\begin{solution}
Loop transport is a linear automorphism of \(E_p\); collect all such automorphisms into a subgroup of \(GL(E_p)\).
\end{solution}


\begin{problem}[Principal-holonomy problem]\label{prob:vii25-08}
Relate horizontal lifts in a principal bundle to holonomy elements of the structure group.
\end{problem}
\begin{solution}
The horizontal lift of a based loop ends in the same fiber as it starts, so the endpoint differs by a unique right action \(h_\gamma\in G\).
\end{solution}


\begin{problem}[Invariant-tensor problem]\label{prob:vii25-09}
Prove a parallel tensor at \(p\) is fixed by holonomy.
\end{problem}
\begin{solution}
Parallel transport carries the tensor consistently along every path. A loop returns to the same fiber, so its induced action must leave the initial tensor unchanged.
\end{solution}


\begin{problem}[Full-versus-restricted problem]\label{prob:vii25-10}
Explain conceptually why restricted holonomy is local while full holonomy can be global.
\end{problem}
\begin{solution}
Contractible loops can be shrunk into local coordinate neighborhoods and probe curvature. Noncontractible loops can also detect topological monodromy.
\end{solution}


\begin{problem}[Euclidean-versus-sphere problem]\label{prob:vii25-11}
Contrast loop transport in \(\mathbb R^2\) and on \(S^2\).
\end{problem}
\begin{solution}
Euclidean transport is path independent and all loop holonomy is trivial. On the sphere, contractible loops can rotate vectors because curvature is nonzero.
\end{solution}


\begin{problem}[Arc-synthesis problem]\label{prob:vii25-12}
Explain the logical chain VII/21 through VII/25 in one paragraph.
\end{problem}
\begin{solution}
A connection defines covariant differentiation; a Riemannian metric selects the Levi--Civita connection; its autoparallel curves are geodesics; integrating the connection along curves gives parallel transport; closing those curves produces holonomy.
\end{solution}

\section{Challenge problems}

\begin{problem}[Topology without local curvature]\label{prob:vii25-ch01}
Describe how a flat connection can have trivial restricted holonomy but nontrivial full holonomy.
\end{problem}
\begin{solution}
Local flatness makes transport around contractible loops trivial, so restricted holonomy vanishes. On a nonsimply connected base, a representation of the fundamental group can still assign nontrivial transport to noncontractible loops, producing global monodromy.
\end{solution}


\begin{problem}[Holonomy representation preview]\label{prob:vii25-ch02}
When a flat connection has path-independent transport up to homotopy, what algebraic object does loop transport define?
\end{problem}
\begin{solution}
It defines a group homomorphism from the fundamental group \(\pi_1(M,p)\) to \(GL(E_p)\) (or the relevant structure group), called the monodromy or holonomy representation.
\end{solution}


\begin{problem}[Infinitesimal rectangle]\label{prob:vii25-ch03}
Explain why the first nontrivial term in a small rectangular loop is second order in its side lengths.
\end{problem}
\begin{solution}
Transport along one edge is first order in the connection. Opposite edges cancel those first-order contributions when the loop closes. The surviving commutator/derivative contribution is proportional to enclosed area and is encoded by curvature.
\end{solution}


\begin{problem}[Holonomy reduction]\label{prob:vii25-ch04}
Suppose a nonzero vector field \(V\) is globally parallel on a connected Riemannian manifold. What does this imply about holonomy?
\end{problem}
\begin{solution}
For every loop based at \(p\), parallel transport must return \(V_p\) to itself. Hence \(\operatorname{Hol}_p\) lies in the stabilizer of \(V_p\), a proper subgroup of \(O(T_pM)\) when \(V_p\neq0\).
\end{solution}


\begin{problem}[From holonomy to curvature]\label{prob:vii25-ch05}
Why is VII/26 the inevitable next chapter after holonomy?
\end{problem}
\begin{solution}
Holonomy records finite accumulated failure of path-independent transport. Shrinking loops converts that finite discrepancy into an infinitesimal bilinear operator in the spanning directions; that operator is the Riemann curvature tensor.
\end{solution}

\section*{Bridge to VII/26}
Holonomy detects the accumulated effect of curvature around loops.  VII/26 now
defines the Riemann curvature tensor directly from covariant derivatives and shows
how its algebraic identities control the infinitesimal geometry behind this
transport phenomenon.
