\section{Supplementary worked examples}
\label{sec:vii33-pedagogy}

\begin{example}[Three generators]\label{ex:vii33-ped-01}
Translations \(z\mapsto z+b\), positive dilations \(z\mapsto\lambda z\), and inversion \(z\mapsto-1/z\) are represented projectively by real determinant-one matrices and preserve the Poincare metric.  They generate a large part of the geometric intuition for \(PSL(2,\mathbb R)\).
\end{example}

\begin{example}[Sending \(i\) to a point]\label{ex:vii33-ped-02}
For \(z=x+iy\in\mathbb H\), the affine map \(w\mapsto yw+x\) sends \(i\) to \(z\).  After projective rescaling it is represented by an element of \(PSL(2,\mathbb R)\), proving transitivity constructively.
\end{example}

\begin{example}[Stabilizer of \(i\)]\label{ex:vii33-ped-03}
Matrices \(\begin{pmatrix}\cos\theta&\sin\theta\\-\sin\theta&\cos\theta\end{pmatrix}\) fix \(i\).  Their action realizes the isotropy subgroup as rotations of the tangent plane, matching the homogeneous-space picture \(\mathbb H\cong PSL(2,\mathbb R)/SO(2)\).
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Translation matrix]\label{exr:vii33-ped-01}
Give a determinant-one matrix for \(z\mapsto z+b\).
\end{exercise}

\begin{hint}
Use an upper triangular matrix.
\end{hint}

\begin{solution}
\(\begin{pmatrix}1&b\\0&1\end{pmatrix}\).
\end{solution}

\begin{exercise}[Inversion matrix]\label{exr:vii33-ped-02}
Give a matrix for \(z\mapsto-1/z\).
\end{exercise}

\begin{hint}
Swap numerator and denominator with a sign.
\end{hint}

\begin{solution}
\(\begin{pmatrix}0&-1\\1&0\end{pmatrix}\).
\end{solution}

\begin{exercise}[Imaginary part]\label{exr:vii33-ped-03}
For \(A\in SL(2,\mathbb R)\), state \(\operatorname{Im}(Az)\).
\end{exercise}

\begin{hint}
Recall the denominator factor.
\end{hint}

\begin{solution}
\(\operatorname{Im}(Az)=\operatorname{Im}z/|cz+d|^2\).
\end{solution}

\begin{exercise}[Derivative]\label{exr:vii33-ped-04}
State the derivative of \((az+b)/(cz+d)\) when \(ad-bc=1\).
\end{exercise}

\begin{hint}
Differentiate the quotient.
\end{hint}

\begin{solution}
\(1/(cz+d)^2\).
\end{solution}

\begin{exercise}[Fixed point]\label{exr:vii33-ped-05}
Verify \(z\mapsto-1/z\) fixes \(i\).
\end{exercise}

\begin{hint}
Compute \(-1/i\).
\end{hint}

\begin{solution}
It equals \(i\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Metric invariance]\label{exr:vii33-ped-06}
Prove a real determinant-one Mobius transformation preserves \(|dz|/\operatorname{Im}z\).
\end{exercise}

\begin{hint}
Combine derivative and imaginary-part formulas.
\end{hint}

\begin{solution}
Both numerator and denominator acquire the same factor \(|cz+d|^{-2}\).
\end{solution}

\begin{exercise}[Projectivization]\label{exr:vii33-ped-07}
Why do \(A\) and \(-A\) define the same Mobius transformation?
\end{exercise}

\begin{hint}
Both numerator and denominator change sign.
\end{hint}

\begin{solution}
The common factor cancels in the quotient.
\end{solution}

\begin{exercise}[Transitivity]\label{exr:vii33-ped-08}
Construct an isometry sending \(i\) to \(x+iy\).
\end{exercise}

\begin{hint}
Use dilation by \(y\) then translation by \(x\).
\end{hint}

\begin{solution}
\(w\mapsto yw+x\) does so.
\end{solution}

\begin{exercise}[Boundary action]\label{exr:vii33-ped-09}
Why does a real Mobius transformation send \(\mathbb R\cup\{\infty\}\) to itself?
\end{exercise}

\begin{hint}
Real inputs produce real outputs when defined.
\end{hint}

\begin{solution}
Real coefficients preserve the extended real line under fractional-linear action.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Determinant irrelevant]\label{exr:vii33-ped-10}
Test the claim: any real Mobius matrix with nonzero determinant acts isometrically on \(\mathbb H\) without rescaling.
\end{exercise}

\begin{hint}
Projectively normalize determinant sign and value.
\end{hint}

\begin{solution}
A positive determinant can be rescaled to determinant one; negative determinant reverses the half-plane/orientation behavior and is not an element of \(PSL(2,\mathbb R)\).
\end{solution}

\begin{exercise}[Matrix unique]\label{exr:vii33-ped-11}
Test the claim: each Mobius transformation has a unique representing matrix.
\end{exercise}

\begin{hint}
Recall projective scaling.
\end{hint}

\begin{solution}
False.  Any nonzero scalar multiple represents the same fractional-linear map.
\end{solution}

\begin{exercise}[All complex Mobius preserve half-plane]\label{exr:vii33-ped-12}
Test the claim.
\end{exercise}

\begin{hint}
Real coefficients are essential for this subgroup.
\end{hint}

\begin{solution}
False.  General complex Mobius maps act on the Riemann sphere, not necessarily preserving \(\mathbb H\).
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Normalization]\label{exr:vii33-ped-13}
Why is transitivity useful in hyperbolic calculations?
\end{exercise}

\begin{hint}
Move one point to \(i\).
\end{hint}

\begin{solution}
An isometry can normalize a configuration, reducing distance or angle calculations to a simpler representative without changing intrinsic quantities.
\end{solution}

\begin{exercise}[Boundary dynamics]\label{exr:vii33-ped-14}
Why is the extended real line important for classifying isometries?
\end{exercise}

\begin{hint}
Fixed points may lie at infinity.
\end{hint}

\begin{solution}
Elliptic, parabolic, and hyperbolic elements are distinguished by the number and location of fixed points in the interior and ideal boundary.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Stabilizer equation]\label{exr:vii33-ped-15}
Show the displayed rotation matrix fixes \(i\).
\end{exercise}

\begin{hint}
Substitute \(z=i\) and simplify.
\end{hint}

\begin{solution}
The numerator equals \(i\) times the denominator, so the quotient is \(i\).
\end{solution}

\begin{exercise}[Distance invariance]\label{exr:vii33-ped-16}
If \(A\in PSL(2,\mathbb R)\), what is \(d(Az,Aw)\)?
\end{exercise}

\begin{hint}
Metric isometries preserve distance.
\end{hint}

\begin{solution}
\(d(Az,Aw)=d(z,w)\).
\end{solution}
