\chapter{Möbius Transformations and PSL(2,R)}
\label{ch:vii33-mobius-psl2r}

The Poincaré metric is not merely a formula; it has a large symmetry group.
Real Möbius transformations act transitively on the upper half-plane and preserve
its hyperbolic metric.

\section{Fractional linear transformations}

\begin{definition}[Möbius transformation]\label{def:vii33-mobius}
For
\[
A=
\begin{pmatrix}
a&b\\ c&d
\end{pmatrix},
\qquad
ad-bc\ne0,
\]
define
\[
A\cdot z
=
\frac{az+b}{cz+d}
\]
on the Riemann sphere where the expression is defined.
\end{definition}

Multiplying \(A\) by a nonzero scalar does not change the transformation.

\section{The real determinant-one group}

\begin{definition}[Special linear group]\label{def:vii33-sl2}
\[
SL(2,\mathbb R)
=
\left\{
A\in M_2(\mathbb R):\det A=1
\right\}.
\]
\end{definition}

The matrices \(A\) and \(-A\) define the same Möbius map.

\begin{definition}[Projective special linear group]\label{def:vii33-psl2}
\[
PSL(2,\mathbb R)
=
SL(2,\mathbb R)/\{\pm I\}.
\]
\end{definition}

\section{Preservation of the upper half-plane}

\begin{theorem}[Imaginary-part transformation]\label{thm:vii33-imag}
For
\[
A=
\begin{pmatrix}a&b\\c&d\end{pmatrix}
\in SL(2,\mathbb R),
\]
\[
\operatorname{Im}(A\cdot z)
=
\frac{\operatorname{Im}z}{|cz+d|^2}.
\]
Therefore
\[
A(\mathbb H)=\mathbb H.
\]
\end{theorem}

\begin{proof}
Subtract the complex conjugate:
\[
\frac{az+b}{cz+d}
-
\frac{a\overline z+b}{c\overline z+d}
=
\frac{(ad-bc)(z-\overline z)}
{|cz+d|^2}.
\]
Since \(ad-bc=1\), taking imaginary parts gives the formula.
\end{proof}

\section{Derivative}

\begin{proposition}[Möbius derivative]\label{prop:vii33-derivative}
For \(A\in SL(2,\mathbb R)\),
\[
(A\cdot z)'
=
\frac1{(cz+d)^2}.
\]
Hence
\[
|d(A\cdot z)|
=
\frac{|dz|}{|cz+d|^2}.
\]
\end{proposition}

\section{Poincaré invariance}

\begin{theorem}[Metric invariance]\label{thm:vii33-metric-invariance}
Every element of \(PSL(2,\mathbb R)\) is an orientation-preserving isometry of
\((\mathbb H,g_{\mathbb H})\):
\[
A^*g_{\mathbb H}=g_{\mathbb H}.
\]
\end{theorem}

\begin{proof}
Using the derivative and imaginary-part formulas,
\[
\frac{|d(Az)|}{\operatorname{Im}(Az)}
=
\frac{|dz|/|cz+d|^2}
{\operatorname{Im}z/|cz+d|^2}
=
\frac{|dz|}{\operatorname{Im}z}.
\]
\end{proof}

\section{Generators}

Several elementary matrices produce familiar transformations:
\[
T_b(z)=z+b,
\qquad
D_\lambda(z)=\lambda z,
\qquad
S(z)=-\frac1z,
\]
where \(b\in\mathbb R\) and \(\lambda>0\).

\begin{proposition}[Elementary isometries]\label{prop:vii33-generators}
Translations \(T_b\), positive dilations \(D_\lambda\), and inversion \(S\) are
hyperbolic isometries of \(\mathbb H\).
\end{proposition}

\section{Transitivity}

\begin{theorem}[Transitive action]\label{thm:vii33-transitive}
The action of \(PSL(2,\mathbb R)\) on \(\mathbb H\) is transitive.
\end{theorem}

\begin{proof}
For \(z=x+iy\), the map
\[
w\longmapsto yw+x
\]
is represented projectively by an element of \(PSL(2,\mathbb R)\) and sends
\(i\) to \(z\).
\end{proof}

\section{Stabilizer of \(i\)}

\begin{theorem}[Isotropy subgroup]\label{thm:vii33-stabilizer}
The stabilizer of \(i\) consists of matrices
\[
\begin{pmatrix}
\cos\theta&\sin\theta\\
-\sin\theta&\cos\theta
\end{pmatrix}
\]
modulo \(\pm I\).  It is isomorphic to \(SO(2)\).
\end{theorem}

Thus
\[
\mathbb H
\cong
PSL(2,\mathbb R)/SO(2)
\]
as a homogeneous Riemannian space.

\section{Boundary action}

The same fractional linear formula acts on
\[
\partial_\infty\mathbb H
=
\mathbb R\cup\{\infty\}.
\]

\begin{proposition}[Boundary extension]\label{prop:vii33-boundary}
Every \(A\in PSL(2,\mathbb R)\) extends continuously to a homeomorphism of the
extended real line.
\end{proposition}

The boundary action will be central for classifying isometries in VII/34.

\section{Generalized circles}

\begin{theorem}[Circle preservation]\label{thm:vii33-circles}
Möbius transformations send generalized circles---Euclidean circles or straight
lines on the Riemann sphere---to generalized circles.
\end{theorem}

Since real Möbius maps preserve the real axis and conformal angles, they send
half-plane geodesics to half-plane geodesics.

\section{Cross ratios}

\begin{definition}[Cross ratio]\label{def:vii33-cross-ratio}
For four distinct points,
\[
[z_1,z_2;z_3,z_4]
=
\frac{(z_1-z_3)(z_2-z_4)}
{(z_1-z_4)(z_2-z_3)}.
\]
\end{definition}

\begin{theorem}[Cross-ratio invariance]\label{thm:vii33-cross-invariance}
For every Möbius transformation \(M\),
\[
[Mz_1,Mz_2;Mz_3,Mz_4]
=
[z_1,z_2;z_3,z_4].
\]
\end{theorem}

Cross ratios provide an efficient bridge between boundary geometry and
hyperbolic distance.

\section{Disk form of the group}

Conjugating \(PSL(2,\mathbb R)\) by the Cayley transform produces the
orientation-preserving isometry group of the Poincaré disk.

\begin{proposition}[Disk automorphism form]\label{prop:vii33-disk-auto}
Every orientation-preserving disk isometry has the form
\[
z\longmapsto
e^{i\theta}
\frac{z-a}{1-\overline a z},
\qquad
|a|<1.
\]
\end{proposition}

\section{Orientation reversal}

Complex conjugation
\[
z\mapsto-\overline z
\]
preserves the upper half-plane and the Poincaré metric but reverses orientation.

\begin{remark}
Thus \(PSL(2,\mathbb R)\) is the orientation-preserving isometry group.  The full
isometry group is obtained by adjoining one orientation-reversing isometry.
\end{remark}

\section{Solved problem dossiers}
\begin{problem}[Imaginary-part problem]\label{prob:vii33-01}
Derive \(\Im(Az)=\Im z/|cz+d|^2\).
\end{problem}
\begin{solution}
Subtract the conjugate fractional linear expressions and use \(ad-bc=1\).
\end{solution}

\begin{problem}[Derivative problem]\label{prob:vii33-02}
Derive \(A'(z)=1/(cz+d)^2\).
\end{problem}
\begin{solution}
Differentiate the quotient: the numerator simplifies to \(ad-bc=1\).
\end{solution}

\begin{problem}[Metric-invariance problem]\label{prob:vii33-03}
Prove the half-plane metric is invariant under \(PSL(2,\mathbb R)\).
\end{problem}
\begin{solution}
Substitute the derivative and imaginary-part formulas into \(|dw|/\Im w\); the same denominator cancels.
\end{solution}

\begin{problem}[Transitivity problem]\label{prob:vii33-04}
Construct an element sending \(i\) to \(x+iy\).
\end{problem}
\begin{solution}
The affine map \(z\mapsto yz+x\) does so. Represent it by a determinant-one matrix after multiplying numerator and denominator by a suitable scalar.
\end{solution}

\begin{problem}[Stabilizer problem]\label{prob:vii33-05}
Solve \(A(i)=i\) for \(A\in SL(2,\mathbb R)\).
\end{problem}
\begin{solution}
Equating real and imaginary parts gives \(d=a\), \(c=-b\), and \(a^2+b^2=1\), producing the rotation matrices.
\end{solution}

\begin{problem}[Geodesic problem]\label{prob:vii33-06}
Show real Möbius transformations send half-plane geodesics to geodesics.
\end{problem}
\begin{solution}
They preserve the real boundary, generalized circles, and angles; hence boundary-orthogonal generalized circles remain boundary-orthogonal.
\end{solution}

\begin{problem}[Cross-ratio problem]\label{prob:vii33-07}
Prove cross-ratio invariance.
\end{problem}
\begin{solution}
Use \(M(z_i)-M(z_j)=(ad-bc)(z_i-z_j)/((cz_i+d)(cz_j+d))\); all matrix and denominator factors cancel.
\end{solution}

\begin{problem}[Disk-action problem]\label{prob:vii33-08}
Derive the disk automorphism form by conjugating a half-plane isometry.
\end{problem}
\begin{solution}
Conjugate \(PSL(2,\mathbb R)\) with the Cayley map. Normalizing the image of \(a\) to \(0\) yields \((z-a)/(1-\bar a z)\), followed by a unit rotation.
\end{solution}

\begin{problem}[Boundary problem]\label{prob:vii33-09}
Explain why a real Möbius map extends to the ideal boundary.
\end{problem}
\begin{solution}
The fractional linear map acts continuously on the real projective line, with \(cz+d=0\) mapped to \(\infty\) and \(\infty\) mapped according to \(a/c\).
\end{solution}

\begin{problem}[Orientation problem]\label{prob:vii33-10}
Why are holomorphic Möbius maps orientation preserving?
\end{problem}
\begin{solution}
A nonzero complex derivative acts locally as a rotation-dilation, which has positive real Jacobian determinant.
\end{solution}

\begin{problem}[Full-group problem]\label{prob:vii33-11}
Why does adjoining one reflection generate all isometries once the orientation-preserving group is known?
\end{problem}
\begin{solution}
Any orientation-reversing isometry composed with a fixed reflection becomes orientation preserving; therefore every reversing isometry is a product of that reflection with an element of \(PSL(2,\mathbb R)\).
\end{solution}

\begin{problem}[Homogeneous-space problem]\label{prob:vii33-12}
Explain \(\mathbb H\cong PSL(2,\mathbb R)/SO(2)\).
\end{problem}
\begin{solution}
Transitivity identifies points with left cosets of the stabilizer of \(i\), and that stabilizer is \(SO(2)\).
\end{solution}

\section{Challenge problems}
\begin{problem}[Three-point rigidity]\label{prob:vii33-ch01}
Why is a Möbius transformation determined by its values on three distinct points?
\end{problem}
\begin{solution}
Cross-ratio invariance or direct linear algebra shows that fixing three distinct points forces the identity; composing two transformations with the same three images therefore gives the identity.
\end{solution}

\begin{problem}[Boundary triple]\label{prob:vii33-ch02}
Show \(PSL(2,\mathbb R)\) acts triply transitively on cyclically ordered boundary triples.
\end{problem}
\begin{solution}
A unique real Möbius transformation sends any ordered triple of distinct real-projective points to \((0,1,\infty)\); orientation preservation imposes the cyclic-order condition.
\end{solution}

\begin{problem}[Distance via cross ratio]\label{prob:vii33-ch03}
How can hyperbolic distance along a geodesic be expressed using its ideal endpoints?
\end{problem}
\begin{solution}
If \(a,b\) are the ideal endpoints and \(z,w\) lie on the geodesic, distance is the logarithm of an appropriate positive cross ratio; Möbius invariance reduces the proof to a vertical line.
\end{solution}

\begin{problem}[Lie-group dimension]\label{prob:vii33-ch04}
Why does \(PSL(2,\mathbb R)\) have real dimension three?
\end{problem}
\begin{solution}
A real \(2\times2\) matrix has four parameters and the equation \(ad-bc=1\) removes one; quotienting by the discrete subgroup \(\{\pm I\}\) does not change dimension.
\end{solution}

\begin{problem}[Lorentz-group connection]\label{prob:vii33-ch05}
Why should \(PSL(2,\mathbb R)\) be closely related to the identity component of \(SO(2,1)\)?
\end{problem}
\begin{solution}
Both act transitively and isometrically on the hyperbolic plane with the same one-dimensional compact stabilizer; more concretely, the adjoint action of \(SL(2,\mathbb R)\) preserves a quadratic form of signature \((2,1)\) and descends through \(\pm I\).
\end{solution}

\section*{Bridge to VII/34}
VII/34 studies individual elements of \(PSL(2,\mathbb R)\), classifying them as elliptic, parabolic, or hyperbolic by fixed points, traces, axes, and translation lengths.
