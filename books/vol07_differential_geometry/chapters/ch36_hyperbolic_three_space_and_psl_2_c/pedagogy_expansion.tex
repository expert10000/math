\section{Supplementary worked examples}
\label{sec:vii36-pedagogy}

\begin{example}[Vertical distance in \(\mathbb H^3\)]\label{ex:vii36-ped-01}
For points \((0,t_1)\) and \((0,t_2)\) on a vertical geodesic, the upper-half-space metric gives distance \(|\log(t_2/t_1)|\), exactly as in the two-dimensional half-plane.
\end{example}

\begin{example}[Horosphere metric]\label{ex:vii36-ped-02}
On the horizontal horosphere \(t=t_0\), the induced metric is \((dx^2+dy^2)/t_0^2\).  Each horosphere is intrinsically Euclidean, but its scale changes with height.
\end{example}

\begin{example}[Screw motion]\label{ex:vii36-ped-03}
After conjugation, a loxodromic isometry can combine translation by \(\ell\) along an axis with rotation by \(\theta\).  Its complex length \(L=\ell+i\theta\) and eigenvalues \(e^{\pm L/2}\) package both motions into one invariant.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Vertical distance]\label{exr:vii36-ped-01}
Compute the distance from \((0,1)\) to \((0,e^3)\).
\end{exercise}

\begin{hint}
Use the vertical logarithm.
\end{hint}

\begin{solution}
\(3\).
\end{solution}

\begin{exercise}[Volume density]\label{exr:vii36-ped-02}
State the hyperbolic volume form in upper half-space coordinates.
\end{exercise}

\begin{hint}
The conformal factor is \(t^{-2}\) in three dimensions.
\end{hint}

\begin{solution}
\(dx\,dy\,dt/t^3\).
\end{solution}

\begin{exercise}[Horosphere scale]\label{exr:vii36-ped-03}
At height \(t_0=2\), by what factor are Euclidean lengths scaled on the horosphere?
\end{exercise}

\begin{hint}
Use \(ds=|dz|/t_0\).
\end{hint}

\begin{solution}
By \(1/2\).
\end{solution}

\begin{exercise}[Boundary]\label{exr:vii36-ped-04}
What is \(\partial_\infty\mathbb H^3\)?
\end{exercise}

\begin{hint}
Compactify the complex plane by infinity.
\end{hint}

\begin{solution}
The Riemann sphere \(\widehat{\mathbb C}\).
\end{solution}

\begin{exercise}[Complex length]\label{exr:vii36-ped-05}
If \(L=2+i\pi/3\), what are translation length and rotation angle?
\end{exercise}

\begin{hint}
Read real and imaginary parts.
\end{hint}

\begin{solution}
Translation length \(2\), rotation angle \(\pi/3\) modulo \(2\pi\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Geodesic shape]\label{exr:vii36-ped-06}
Why are upper-half-space geodesics vertical lines or semicircles orthogonal to the boundary?
\end{exercise}

\begin{hint}
Reduce to a totally geodesic copy of \(\mathbb H^2\).
\end{hint}

\begin{solution}
Any two points and their ideal endpoints lie in a vertical hyperbolic plane, where the two-dimensional geodesic classification applies.
\end{solution}

\begin{exercise}[Extension uniqueness]\label{exr:vii36-ped-07}
Why is a boundary Mobius transformation's hyperbolic extension unique?
\end{exercise}

\begin{hint}
Orientation-preserving isometries are determined by their ideal-boundary action.
\end{hint}

\begin{solution}
The boundary action determines how all geodesics with ideal endpoints move, hence determines the interior isometry.
\end{solution}

\begin{exercise}[Fuchsian inclusion]\label{exr:vii36-ped-08}
Why does \(PSL(2,\mathbb R)\) preserve a totally geodesic plane inside \(\mathbb H^3\)?
\end{exercise}

\begin{hint}
It preserves the extended real circle in the boundary.
\end{hint}

\begin{solution}
The unique hyperbolic plane with that ideal boundary is therefore invariant.
\end{solution}

\begin{exercise}[Trace complex]\label{exr:vii36-ped-09}
Why is \(|\operatorname{tr}A|\) alone insufficient in \(PSL(2,\mathbb C)\)?
\end{exercise}

\begin{hint}
Rotation contributes a complex phase.
\end{hint}

\begin{solution}
Complex trace data encode both translation and rotational parts; taking only absolute value discards phase information.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[All loxodromics pure translations]\label{exr:vii36-ped-10}
Test the claim in three dimensions.
\end{exercise}

\begin{hint}
Allow a rotational part around the axis.
\end{hint}

\begin{solution}
False.  Generic loxodromics are screw motions with nonzero rotation angle.
\end{solution}

\begin{exercise}[Boundary is complex plane only]\label{exr:vii36-ped-11}
Test the claim.
\end{exercise}

\begin{hint}
Do not forget the point at infinity.
\end{hint}

\begin{solution}
False.  The ideal boundary is the Riemann sphere \(\mathbb C\cup\{\infty\}\).
\end{solution}

\begin{exercise}[Horosphere hyperbolic]\label{exr:vii36-ped-12}
Test the claim: the intrinsic metric of a horosphere has negative curvature.
\end{exercise}

\begin{hint}
Its induced metric is a constant multiple of Euclidean metric.
\end{hint}

\begin{solution}
False.  Horospheres are intrinsically flat.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Cusp cross-section]\label{exr:vii36-ped-13}
Why do horospheres model cusp cross-sections in hyperbolic three-manifolds?
\end{exercise}

\begin{hint}
Parabolic subgroups preserve horospheres based at their fixed point.
\end{hint}

\begin{solution}
Quotienting an invariant horosphere by the parabolic subgroup produces the Euclidean cross-section of the cusp.
\end{solution}

\begin{exercise}[3D visualization]\label{exr:vii36-ped-14}
What extra motion appears in \(\mathbb H^3\) compared with hyperbolic-plane translations along an axis?
\end{exercise}

\begin{hint}
There is room to rotate around the axis.
\end{hint}

\begin{solution}
A loxodromic isometry may rotate as well as translate, producing complex length.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Eigenvalue relation]\label{exr:vii36-ped-15}
If a loxodromic lift has eigenvalues \(e^{L/2},e^{-L/2}\), what is its trace?
\end{exercise}

\begin{hint}
Add them.
\end{hint}

\begin{solution}
\(2\cosh(L/2)\).
\end{solution}

\begin{exercise}[Vertical scale]\label{exr:vii36-ped-16}
How far must one move vertically to multiply height by \(10\)?
\end{exercise}

\begin{hint}
Use the logarithmic distance.
\end{hint}

\begin{solution}
Distance \(\log10\).
\end{solution}
