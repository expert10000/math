\section{Supplementary worked examples}
\label{sec:vii24-pedagogy}

\begin{example}[Euclidean path independence]\label{ex:vii24-ped-01}
In Euclidean space with the standard connection, a vector with constant Cartesian components is parallel along every curve.  Parallel transport therefore depends only on the endpoints and in fact is the identity after the canonical tangent-space identification.
\end{example}

\begin{example}[A parallel field along the equator]\label{ex:vii24-ped-02}
For the unit sphere equator \(\gamma(t)=(\cos t,\sin t,0)\), the constant ambient vector \(V=(0,0,1)\) is tangent to the sphere along the entire equator.  Its ambient derivative is zero, hence its tangential projection is zero, so \(V\) is parallel.
\end{example}

\begin{example}[Matrix transport]\label{ex:vii24-ped-03}
In a local frame, parallel transport is the fundamental matrix solution \(\dot U=-A(t)U\), \(U(a)=I\).  This makes transport a path-ordered linear evolution and explains concatenation as matrix multiplication.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Reverse path]\label{exr:vii24-ped-01}
What is the transport along the reversed curve?
\end{exercise}

\begin{hint}
Use uniqueness of the parallel ODE.
\end{hint}

\begin{solution}
\(P_{\gamma^{-1}}=P_\gamma^{-1}\).
\end{solution}

\begin{exercise}[Concatenation]\label{exr:vii24-ped-02}
Write the transport map for two concatenated paths.
\end{exercise}

\begin{hint}
Transport first along \(\gamma_1\), then \(\gamma_2\).
\end{hint}

\begin{solution}
\(P_{\gamma_2*\gamma_1}=P_{\gamma_2}\circ P_{\gamma_1}\).
\end{solution}

\begin{exercise}[Euclidean transport]\label{exr:vii24-ped-03}
Transport \((1,2)\) along any curve in \(\mathbb R^2\) with the standard connection.
\end{exercise}

\begin{hint}
Keep Cartesian components constant.
\end{hint}

\begin{solution}
The transported vector has components \((1,2)\) at every point.
\end{solution}

\begin{exercise}[Constant length]\label{exr:vii24-ped-04}
If \(V\) is Levi--Civita parallel and \(|V(a)|=3\), what is \(|V(b)|\)?
\end{exercise}

\begin{hint}
Use metric preservation.
\end{hint}

\begin{solution}
It is \(3\).
\end{solution}

\begin{exercise}[Fundamental matrix]\label{exr:vii24-ped-05}
Solve \(\dot U=-cU\), \(U(0)=1\), in rank one.
\end{exercise}

\begin{hint}
This is a scalar ODE.
\end{hint}

\begin{solution}
\(U(t)=e^{-ct}\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Linearity]\label{exr:vii24-ped-06}
Prove \(P_\gamma\) is linear.
\end{exercise}

\begin{hint}
Use linearity of the parallel ODE and uniqueness.
\end{hint}

\begin{solution}
Linear combinations of parallel solutions are parallel with corresponding initial data, so endpoint evaluation is linear.
\end{solution}

\begin{exercise}[Metric isometry]\label{exr:vii24-ped-07}
Prove Levi--Civita parallel transport preserves inner products.
\end{exercise}

\begin{hint}
Differentiate \(g(V,W)\).
\end{hint}

\begin{solution}
Metric compatibility and parallelism make its derivative zero.
\end{solution}

\begin{exercise}[Geodesic tangent]\label{exr:vii24-ped-08}
Prove a curve is geodesic exactly when its tangent field is parallel along itself.
\end{exercise}

\begin{hint}
Compare the two definitions.
\end{hint}

\begin{solution}
Both conditions are precisely \(D\dot\gamma/dt=0\).
\end{solution}

\begin{exercise}[Frame transport]\label{exr:vii24-ped-09}
Why does a transported orthonormal basis remain orthonormal?
\end{exercise}

\begin{hint}
Apply inner-product preservation pairwise.
\end{hint}

\begin{solution}
All pairwise inner products remain \(\delta_{ij}\).
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Endpoint-only transport]\label{exr:vii24-ped-10}
Test the claim: Levi--Civita transport between two points depends only on the endpoints.
\end{exercise}

\begin{hint}
Think of curved surfaces and loops.
\end{hint}

\begin{solution}
False in general.  Path dependence is measured by holonomy and curvature.
\end{solution}

\begin{exercise}[Parallel means constant components]\label{exr:vii24-ped-11}
Test the claim: a parallel vector field always has constant coordinate components.
\end{exercise}

\begin{hint}
Use a curvilinear frame.
\end{hint}

\begin{solution}
False.  Components must compensate for changing basis vectors through the connection coefficients.
\end{solution}

\begin{exercise}[All connections preserve length]\label{exr:vii24-ped-12}
Test the claim: parallel transport for any connection preserves a Riemannian metric.
\end{exercise}

\begin{hint}
Metric compatibility is an extra condition.
\end{hint}

\begin{solution}
False.  Only metric-compatible connections preserve the metric under parallel transport.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Attitude transport]\label{exr:vii24-ped-13}
Why is parallel transport relevant to moving a tangent frame over a curved surface in graphics or robotics?
\end{exercise}

\begin{hint}
It minimizes spurious frame rotation according to the chosen connection.
\end{hint}

\begin{solution}
It provides a canonical intrinsic comparison of orientations at nearby points while preserving tangent metric data.
\end{solution}

\begin{exercise}[Transport and texture]\label{exr:vii24-ped-14}
How can parallel transport help propagate a tangent direction field across a mesh or surface?
\end{exercise}

\begin{hint}
Move a seed direction along chosen paths.
\end{hint}

\begin{solution}
Transport gives a rule for carrying directions while accounting for surface curvature; path dependence warns where consistency defects may appear.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Equator field]\label{exr:vii24-ped-15}
Verify \((0,0,1)\) is tangent to the unit sphere along the equator.
\end{exercise}

\begin{hint}
Dot it with \(\gamma(t)\).
\end{hint}

\begin{solution}
Its dot product with \((\cos t,\sin t,0)\) is zero, so it lies in every equatorial tangent plane.
\end{solution}

\begin{exercise}[Closed-loop diagnostic]\label{exr:vii24-ped-16}
If transport around a small loop rotates a vector by a nonzero angle, what geometric quantity is being detected infinitesimally?
\end{exercise}

\begin{hint}
Connect transport failure with the next chapters.
\end{hint}

\begin{solution}
Curvature; finite loop effects are holonomy, and their infinitesimal generator is the curvature tensor.
\end{solution}
