\section{Supplementary worked examples}
\label{sec:vii04-pedagogy}

\begin{example}[Derivations recover Euclidean tangent vectors]\label{ex:vii04-ped-01}
At \(p\in\mathbb R^n\), a vector \(v=(v^1,\ldots,v^n)\) defines the derivation
\[
D_v f=\sum_i v^i\frac{\partial f}{\partial x^i}(p).
\]
Conversely, a derivation is determined by its values on the coordinate functions.  Thus the derivation definition of \(T_pM\) agrees with the usual vector-space picture in coordinates.
\end{example}

\begin{example}[Tangent space to a sphere as a kernel]\label{ex:vii04-ped-02}
For \(F(x)=\lVert x\rVert^2\), the sphere \(S^{n-1}=F^{-1}(1)\) is a regular level set.  Since \(dF_p(v)=2p\cdot v\),
\[
T_pS^{n-1}=\ker dF_p=\{v:p\cdot v=0\}.
\]
The abstract tangent space is therefore the familiar orthogonal hyperplane.
\end{example}

\begin{example}[Differential of a smooth map]\label{ex:vii04-ped-03}
For \(f:S^1\to S^1\), \(f(z)=z^k\), the differential multiplies angular tangent vectors by \(k\).  In an angle coordinate \(\theta\), the representative is \(\theta\mapsto k\theta\), so \(df\) records the first-order winding of the map.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Euclidean derivation]\label{exr:vii04-ped-01}
At \(p=(1,2)\), evaluate the derivation \(D=3\partial_x-2\partial_y\) on \(f(x,y)=x^2y\).
\end{exercise}
\begin{hint}
Compute the partial derivatives at \(p\).
\end{hint}
\begin{solution}
\(\partial_xf=2xy=4\) and \(\partial_yf=x^2=1\), so \(Df=3\cdot4-2\cdot1=10\).
\end{solution}

\begin{exercise}[Circle tangent]\label{exr:vii04-ped-02}
Find \(T_{(1,0)}S^1\subset\mathbb R^2\).
\end{exercise}
\begin{hint}
Use the level set \(x^2+y^2=1\).
\end{hint}
\begin{solution}
The tangent vectors satisfy \((1,0)\cdot(v_1,v_2)=0\), hence \(T_{(1,0)}S^1=\{(0,t):t\in\mathbb R\}\).
\end{solution}

\begin{exercise}[Sphere tangent]\label{exr:vii04-ped-03}
Find \(T_pS^2\) for \(p=(0,0,1)\).
\end{exercise}
\begin{hint}
Take the orthogonal complement of \(p\).
\end{hint}
\begin{solution}
It is the horizontal plane \(\{(a,b,0)\}\).
\end{solution}

\begin{exercise}[Differential of projection]\label{exr:vii04-ped-04}
For \(\pi:\mathbb R^m\times\mathbb R^n\to\mathbb R^m\), compute \(d\pi_{(p,q)}\).
\end{exercise}
\begin{hint}
Differentiate the coordinate formula.
\end{hint}
\begin{solution}
\(d\pi(v,w)=v\).
\end{solution}

\begin{exercise}[Differential of power map]\label{exr:vii04-ped-05}
For \(f:S^1\to S^1\), \(f(e^{i\theta})=e^{ik\theta}\), compute \(df\) in angular coordinates.
\end{exercise}
\begin{hint}
Differentiate \(k\theta\).
\end{hint}
\begin{solution}
The tangent scalar is multiplied by \(k\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Dimension of tangent space]\label{exr:vii04-ped-06}
Prove that \(T_pM\) has dimension \(n\) on an \(n\)-manifold.
\end{exercise}
\begin{hint}
Use a chart and the induced basis \(\partial/\partial x^i|_p\).
\end{hint}
\begin{solution}
A chart identifies derivations at \(p\) with derivations at a point of \(\mathbb R^n\), where the coordinate derivations form a basis of size \(n\).
\end{solution}

\begin{exercise}[Chain rule]\label{exr:vii04-ped-07}
Prove \(d(g\circ f)_p=dg_{f(p)}\circ df_p\).
\end{exercise}
\begin{hint}
Apply both sides to a smooth function using the derivation definition.
\end{hint}
\begin{solution}
For a tangent derivation \(v\), both sides act on a test function \(h\) as \(v(h\circ g\circ f)\), so they agree.
\end{solution}

\begin{exercise}[Tangent to a regular level set]\label{exr:vii04-ped-08}
Prove \(T_pF^{-1}(c)=\ker dF_p\) when \(c\) is a regular value.
\end{exercise}
\begin{hint}
Use the local normal form from the regular level-set theorem.
\end{hint}
\begin{solution}
Local coordinates make \(F\) a projection; vectors tangent to the level set are exactly those with zero components in the projected directions, which is the kernel of \(dF_p\).
\end{solution}

\begin{exercise}[Diffeomorphism induces tangent isomorphism]\label{exr:vii04-ped-09}
Prove that a diffeomorphism \(f:M\to N\) induces a linear isomorphism \(df_p:T_pM\to T_{f(p)}N\).
\end{exercise}
\begin{hint}
Differentiate \(f^{-1}\circ f=\mathrm{id}\).
\end{hint}
\begin{solution}
The chain rule gives \(d(f^{-1})_{f(p)}\,df_p=\mathrm{id}\) and the reverse composition, so the two differentials are inverses.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Tangent vectors depend on coordinates]\label{exr:vii04-ped-10}
Test the claim that tangent vectors themselves change when a chart changes.
\end{exercise}
\begin{hint}
Distinguish a vector from its coordinate components.
\end{hint}
\begin{solution}
False.  The abstract tangent vector is invariant; only its components transform by the Jacobian of the coordinate change.
\end{solution}

\begin{exercise}[Every ambient vector is tangent]\label{exr:vii04-ped-11}
Test the claim for \(S^2\subset\mathbb R^3\).
\end{exercise}
\begin{hint}
Use the normal vector \(p\).
\end{hint}
\begin{solution}
False.  At \(p\), the radial vector \(p\) is normal and satisfies \(p\cdot p\neq0\), so it is not tangent.
\end{solution}

\begin{exercise}[Zero differential means constant map]\label{exr:vii04-ped-12}
Test globally on a disconnected manifold.
\end{exercise}
\begin{hint}
Take different constants on different components.
\end{hint}
\begin{solution}
False without connectedness: a smooth function can be constant with different values on separate components and still have zero differential everywhere.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Velocity of a curve]\label{exr:vii04-ped-13}
Explain how a smooth curve \(\gamma:(-\epsilon,\epsilon)\to M\) defines a tangent vector at \(\gamma(0)\).
\end{exercise}
\begin{hint}
Differentiate test functions along the curve.
\end{hint}
\begin{solution}
The derivation \(f\mapsto \frac{d}{dt}|_0 f(\gamma(t))\) is a tangent vector; in coordinates it is the usual velocity.
\end{solution}

\begin{exercise}[Constraint kinematics]\label{exr:vii04-ped-14}
For motion constrained to \(S^2\), explain why velocity is orthogonal to position.
\end{exercise}
\begin{hint}
Differentiate \(\|\gamma(t)\|^2=1\).
\end{hint}
\begin{solution}
At each time, \(2\gamma\cdot\dot\gamma=0\), so \(\dot\gamma\in T_{\gamma}S^2\).
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Coordinate transformation law]\label{exr:vii04-ped-15}
Derive the transformation rule between bases \(\partial/\partial x^i\) and \(\partial/\partial y^j\).
\end{exercise}
\begin{hint}
Apply the chain rule to a test function.
\end{hint}
\begin{solution}
\(\frac{\partial}{\partial x^i}=\sum_j\frac{\partial y^j}{\partial x^i}\frac{\partial}{\partial y^j}\), evaluated at the point.
\end{solution}

\begin{exercise}[Tangent space of a matrix group]\label{exr:vii04-ped-16}
Show that \(T_I O(n)\) consists of skew-symmetric matrices.
\end{exercise}
\begin{hint}
Differentiate \(A(t)^TA(t)=I\) at \(t=0\).
\end{hint}
\begin{solution}
If \(X=A'(0)\), differentiation gives \(X^T+X=0\).  Conversely \(e^{tX}\in O(n)\) for skew-symmetric \(X\), so every such \(X\) occurs.
\end{solution}
