\section{Supplementary worked examples}
\label{sec:vii29-pedagogy}

\begin{example}[Null vector in Minkowski space]\label{ex:vii29-ped-01}
For \(\eta=-dt^2+dx^2+dy^2\), the nonzero vector \(v=\partial_t+\partial_x\) is null because \(\eta(v,v)=-1+1=0\).  Nondegeneracy therefore does not imply positive norm-square.
\end{example}

\begin{example}[Timelike orthogonal complement]\label{ex:vii29-ped-02}
For \(u=\partial_t\) in Minkowski space, \(u^\perp\) is spanned by the spatial coordinate vectors and the restricted metric is positive definite.  A timelike direction splits off one negative dimension.
\end{example}

\begin{example}[Null basis in two dimensions]\label{ex:vii29-ped-03}
In \(1+1\) Minkowski space, \(\ell_+=\partial_t+\partial_x\) and \(\ell_-=\partial_t-\partial_x\) are both null, but \(\eta(\ell_+,\ell_-)=-2\).  A basis can consist of null vectors even though the metric is nondegenerate.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Causal type]\label{exr:vii29-ped-01}
Classify \(v=(2,1,0)\) for \(-dt^2+dx^2+dy^2\).
\end{exercise}

\begin{hint}
Compute \(-4+1\).
\end{hint}

\begin{solution}
It is timelike.
\end{solution}

\begin{exercise}[Null type]\label{exr:vii29-ped-02}
Classify \((1,1,0)\).
\end{exercise}

\begin{hint}
Compute the norm-square.
\end{hint}

\begin{solution}
It is null.
\end{solution}

\begin{exercise}[Spacelike type]\label{exr:vii29-ped-03}
Classify \((1,2,0)\).
\end{exercise}

\begin{hint}
Compute \(-1+4\).
\end{hint}

\begin{solution}
It is spacelike.
\end{solution}

\begin{exercise}[Metric determinant]\label{exr:vii29-ped-04}
What is the determinant of \(\operatorname{diag}(-1,1,1)\)?
\end{exercise}

\begin{hint}
Multiply diagonal entries.
\end{hint}

\begin{solution}
It is \(-1\), so the form is nondegenerate.
\end{solution}

\begin{exercise}[Musical map]\label{exr:vii29-ped-05}
For Minkowski \(\eta\), what is \((\partial_t)^\flat\)?
\end{exercise}

\begin{hint}
Pair \(\partial_t\) with coordinate fields.
\end{hint}

\begin{solution}
\((\partial_t)^\flat=-dt\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Signature invariance]\label{exr:vii29-ped-06}
Why cannot a basis change convert a Lorentzian metric into a positive-definite one?
\end{exercise}

\begin{hint}
Use Sylvester's law of inertia.
\end{hint}

\begin{solution}
The numbers of positive and negative squares are invariant under real change of basis.
\end{solution}

\begin{exercise}[Null cone separation]\label{exr:vii29-ped-07}
Explain how the null cone separates causal types in Minkowski space.
\end{exercise}

\begin{hint}
Compare \(|\mathbf x|\) with \(|t|\).
\end{hint}

\begin{solution}
Inside the cone vectors are timelike, on it null, and outside it spacelike under the mostly-plus convention.
\end{solution}

\begin{exercise}[Timelike complement]\label{exr:vii29-ped-08}
Prove the orthogonal complement of a timelike vector is positive definite.
\end{exercise}

\begin{hint}
Choose a Lorentz-orthonormal basis with the timelike vector first.
\end{hint}

\begin{solution}
The complement is spanned by the positive-sign basis vectors, so the restriction is positive definite.
\end{solution}

\begin{exercise}[Signature locally constant]\label{exr:vii29-ped-09}
Why is metric signature locally constant for a smooth nondegenerate metric?
\end{exercise}

\begin{hint}
Eigenvalues cannot cross zero without degeneracy.
\end{hint}

\begin{solution}
Continuity of the symmetric-matrix eigenvalues preserves the count of positive and negative eigenvalues while no eigenvalue vanishes.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Zero norm means zero vector]\label{exr:vii29-ped-10}
Test this Riemannian statement in Lorentzian geometry.
\end{exercise}

\begin{hint}
Use a null vector.
\end{hint}

\begin{solution}
False.  Nonzero null vectors have zero norm-square.
\end{solution}

\begin{exercise}[Cauchy--Schwarz]\label{exr:vii29-ped-11}
Test the usual positive-definite Cauchy--Schwarz inequality for arbitrary Lorentzian vectors.
\end{exercise}

\begin{hint}
Indefinite inner products behave differently.
\end{hint}

\begin{solution}
It does not hold in the usual form for arbitrary causal types.
\end{solution}

\begin{exercise}[Metric determinant positive]\label{exr:vii29-ped-12}
Test the claim: a nondegenerate metric must have positive determinant.
\end{exercise}

\begin{hint}
Use Minkowski signature.
\end{hint}

\begin{solution}
False.  The sign depends on the number of negative directions; nondegeneracy requires only nonzero determinant.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Light cone]\label{exr:vii29-ped-13}
Why does the null cone encode a causal speed limit in Minkowski geometry?
\end{exercise}

\begin{hint}
Timelike directions lie inside it and null directions form the boundary.
\end{hint}

\begin{solution}
Continuous causal curves cannot cross from future timelike directions to spacelike ones without passing through the null boundary.
\end{solution}

\begin{exercise}[Index lowering]\label{exr:vii29-ped-14}
Why do raising and lowering indices still work for an indefinite metric?
\end{exercise}

\begin{hint}
Positivity is not required for invertibility.
\end{hint}

\begin{solution}
Nondegeneracy makes the metric matrix invertible, which is exactly what the musical isomorphisms need.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Null basis pairing]\label{exr:vii29-ped-15}
Compute \(\eta(\ell_+,\ell_-)\) for \(\ell_\pm=\partial_t\pm\partial_x\).
\end{exercise}

\begin{hint}
Expand bilinearly.
\end{hint}

\begin{solution}
It equals \(-2\).
\end{solution}

\begin{exercise}[Degeneracy boundary]\label{exr:vii29-ped-16}
Consider \(g_s=-dt^2+s^2dx^2\).  At what parameter does the metric become degenerate?
\end{exercise}

\begin{hint}
Look at the determinant.
\end{hint}

\begin{solution}
At \(s=0\), where the determinant vanishes.
\end{solution}
