\chapter{Indefinite Metrics}
\label{ch:vii29-indefinite-metrics}

Riemannian geometry assumes positive definiteness.  Many geometric constructions
require only nondegeneracy.  Replacing an inner product by a nondegenerate
symmetric bilinear form of arbitrary signature leads to pseudo-Riemannian
geometry.

\section{Pseudo-Riemannian metrics}

\begin{definition}[Pseudo-Riemannian metric]\label{def:vii29-pseudo}
A pseudo-Riemannian metric on a smooth manifold \(M\) is a smooth symmetric
\((0,2)\)-tensor \(g\) such that each bilinear form
\[
g_p:T_pM\times T_pM\to\mathbb R
\]
is nondegenerate.
\end{definition}

Nondegeneracy means
\[
g_p(v,w)=0\quad\text{for all }w
\qquad\Longrightarrow\qquad
v=0.
\]

\begin{warning}[Nonzero vectors may have zero norm-square]\label{warn:vii29-null}
Unlike the Riemannian case, nondegeneracy does not imply \(g(v,v)>0\) for
\(v\ne0\).  A nonzero vector may satisfy
\[
g(v,v)=0.
\]
Such vectors are called null.
\end{warning}

\section{Signature and Sylvester's law}

\begin{theorem}[Sylvester's law of inertia]\label{thm:vii29-sylvester}
For every nondegenerate symmetric bilinear form on a finite-dimensional real
vector space, there exists a basis in which the matrix is diagonal with entries
\[
\underbrace{+1,\dots,+1}_{p},
\underbrace{-1,\dots,-1}_{q}.
\]
The pair \((p,q)\) is independent of the diagonalizing basis.
\end{theorem}

\begin{definition}[Signature]\label{def:vii29-signature}
The ordered pair \((p,q)\) is the signature, with \(p+q=n\).
Some authors instead call \(p-q\) the signature; throughout this volume the pair
\((p,q)\) is used.
\end{definition}

\begin{proposition}[Signature is locally constant]\label{prop:vii29-local-constant}
For a smooth nondegenerate metric, the inertia pair \((p,q)\) is locally constant.
Hence on each connected component it is constant.
\end{proposition}

\section{Lorentzian metrics}

\begin{definition}[Lorentzian metric]\label{def:vii29-lorentzian}
A Lorentzian metric is a pseudo-Riemannian metric of index one: in dimension \(n\)
its signature is either
\[
(n-1,1)
\]
or, under the opposite sign convention, \((1,n-1)\).
We use the mostly-plus convention \((n-1,1)\).
\end{definition}

\begin{example}[Minkowski metric]\label{ex:vii29-minkowski}
On \(\mathbb R^n\) with coordinates
\[
(t,x^1,\dots,x^{n-1}),
\]
the flat Lorentzian metric is
\[
\eta
=
-dt^2+\sum_{i=1}^{n-1}(dx^i)^2.
\]
\end{example}

\section{Causal types}

Let \(v\ne0\) in a Lorentzian tangent space with mostly-plus convention.

\begin{definition}[Timelike, null, spacelike]\label{def:vii29-causal}
The vector \(v\) is
\[
\begin{array}{ll}
\text{timelike} & \text{if }g(v,v)<0,\\
\text{null} & \text{if }g(v,v)=0,\\
\text{spacelike} & \text{if }g(v,v)>0.
\end{array}
\]
\end{definition}

\begin{remark}
These inequalities reverse if one reverses the overall sign convention for the
metric.  The causal trichotomy itself is invariant.
\end{remark}

\begin{proposition}[Orthogonal complement of a timelike vector]\label{prop:vii29-timelike-orth}
If \(u\) is timelike in a Lorentzian vector space, then the restriction of \(g\) to
\[
u^\perp
\]
is positive definite.
\end{proposition}

\section{The null cone}

\begin{definition}[Null cone]\label{def:vii29-null-cone}
At \(p\), the null cone is
\[
\mathcal N_p
=
\{v\in T_pM\setminus\{0\}:g(v,v)=0\}.
\]
\end{definition}

In Minkowski space,
\[
-(v^0)^2+(v^1)^2+\cdots+(v^{n-1})^2=0.
\]
The null cone separates timelike from spacelike directions.

\section{Musical maps still exist}

\begin{proposition}[Flat and sharp]\label{prop:vii29-musical}
Nondegeneracy of \(g\) still gives an isomorphism
\[
\flat:T_pM\to T_p^*M,
\qquad
v^\flat=g(v,\cdot),
\]
with inverse \(\sharp\).
\end{proposition}

\begin{remark}
Positive definiteness is unnecessary for raising and lowering indices.  What is
essential is nondegeneracy.
\end{remark}

\section{Volume density}

In coordinates,
\[
(g_{ij})
\]
is invertible but may have negative determinant.

\begin{definition}[Pseudo-Riemannian density]\label{def:vii29-density}
The canonical metric density is
\[
dV_g
=
\sqrt{|\det(g_{ij})|}\,
|dx^1\cdots dx^n|.
\]
On an oriented pseudo-Riemannian manifold this determines a signed volume form
after an orientation is chosen.
\end{definition}

\section{Levi--Civita connection survives unchanged}

\begin{theorem}[Pseudo-Riemannian fundamental theorem]\label{thm:vii29-lc}
Every pseudo-Riemannian metric has a unique torsion-free metric-compatible
connection.
\end{theorem}

\begin{proof}
The Koszul formula from VII/22 requires only nondegeneracy of \(g\), not positive
definiteness.  It therefore determines a unique vector \(\nabla_XY\) from its
pairings with all \(Z\), and the standard verification remains valid.
\end{proof}

Consequently the Christoffel formula remains
\[
\Gamma^k_{ij}
=
\frac12g^{k\ell}
\left(
\partial_i g_{j\ell}
+\partial_j g_{i\ell}
-\partial_\ell g_{ij}
\right).
\]

\section{Geodesics and curvature}

The definitions
\[
\nabla_{\dot\gamma}\dot\gamma=0,
\qquad
R(X,Y)Z
=
\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z
\]
carry over verbatim.  Ricci, scalar, and Weyl curvature also remain meaningful.

\begin{warning}[What does not carry over verbatim]\label{warn:vii29-distance}
The positive-definite length and distance theory changes radically.  Null vectors
have zero norm-square, timelike vectors have negative norm-square, and the
Riemannian distance construction is no longer the correct basic global invariant.
\end{warning}

\section{Orthogonal groups of signature}

For the standard form
\[
I_{p,q}=\operatorname{diag}(1,\dots,1,-1,\dots,-1),
\]
define
\[
O(p,q)
=
\{A\in GL(n,\mathbb R):A^TI_{p,q}A=I_{p,q}\}.
\]

\begin{proposition}[Pseudo-orthogonal symmetry]\label{prop:vii29-opq}
Changes of pseudo-orthonormal frame take values in \(O(p,q)\).  In the Lorentzian
case the relevant group is \(O(n-1,1)\).
\end{proposition}

\section{A two-dimensional model}

\begin{example}[Lorentzian plane]\label{ex:vii29-plane}
On \(\mathbb R^2\),
\[
g=-dt^2+dx^2.
\]
The null lines are
\[
x=\pm t.
\]
They divide the tangent plane into timelike and spacelike sectors.
\end{example}

\section{Solved problem dossiers}
\begin{problem}[Signature problem]\label{prob:vii29-01}
Show two diagonalizations of a real nondegenerate symmetric form have the same numbers of positive and negative squares.
\end{problem}
\begin{solution}
This is Sylvester's law of inertia. One argument identifies the maximal dimension of a positive-definite subspace and of a negative-definite subspace; these dimensions are basis independent.
\end{solution}

\begin{problem}[Null-vector problem]\label{prob:vii29-02}
Give a nonzero null vector for the Lorentzian plane.
\end{problem}
\begin{solution}
For \(g=-dt^2+dx^2\), the vector \((1,1)\) satisfies \(-1+1=0\).
\end{solution}

\begin{problem}[Timelike complement problem]\label{prob:vii29-03}
Prove \(u^\perp\) is positive definite when \(u\) is timelike.
\end{problem}
\begin{solution}
Normalize \(u\) to one negative basis vector and extend it to a pseudo-orthonormal basis. The remaining basis vectors all carry positive squares.
\end{solution}

\begin{problem}[Signature-continuity problem]\label{prob:vii29-04}
Explain why signature cannot change without degeneracy.
\end{problem}
\begin{solution}
Eigenvalues of a symmetric matrix vary continuously. Changing the number of positive or negative eigenvalues requires an eigenvalue to pass through zero, making the form degenerate.
\end{solution}

\begin{problem}[Musical problem]\label{prob:vii29-05}
Prove \(v\mapsto g(v,\cdot)\) is an isomorphism for an indefinite nondegenerate form.
\end{problem}
\begin{solution}
Its kernel is exactly the radical of \(g\), which is zero. In finite dimensions injectivity implies bijectivity.
\end{solution}

\begin{problem}[Volume problem]\label{prob:vii29-06}
Why does the volume density use \(|\det g|\)?
\end{problem}
\begin{solution}
An indefinite metric matrix may have negative determinant. Taking the absolute value before the square root produces a real positive density with the correct Jacobian transformation law.
\end{solution}

\begin{problem}[Levi-Civita problem]\label{prob:vii29-07}
Why does the Koszul proof work without positive definiteness?
\end{problem}
\begin{solution}
The proof uses symmetry, nondegeneracy, torsion freeness, and metric compatibility. Nondegeneracy is enough to recover a vector from all of its metric pairings.
\end{solution}

\begin{problem}[Minkowski-geodesic problem]\label{prob:vii29-08}
Determine geodesics of flat Minkowski space in Cartesian coordinates.
\end{problem}
\begin{solution}
The metric coefficients are constant, so all Christoffel symbols vanish and the geodesic equations are \(\ddot x^\mu=0\); geodesics are affine lines.
\end{solution}

\begin{problem}[Causal-line problem]\label{prob:vii29-09}
Classify an affine line in Minkowski space by its constant tangent vector.
\end{problem}
\begin{solution}
If the tangent has negative, zero, or positive norm-square, the line is respectively timelike, null, or spacelike.
\end{solution}

\begin{problem}[Orthogonal-group problem]\label{prob:vii29-10}
Show pseudo-orthonormal changes of frame form \(O(p,q)\).
\end{problem}
\begin{solution}
If the metric matrix is \(I_{p,q}\) in both frames and the frame-change matrix is \(A\), then \(A^TI_{p,q}A=I_{p,q}\).
\end{solution}

\begin{problem}[Curvature problem]\label{prob:vii29-11}
Why do Riemann, Ricci, scalar, and Weyl curvature extend to pseudo-Riemannian metrics?
\end{problem}
\begin{solution}
The Levi--Civita connection exists, and all these tensors are obtained algebraically from the connection, the nondegenerate metric, and its inverse.
\end{solution}

\begin{problem}[Distance-failure problem]\label{prob:vii29-12}
Give a concrete reason a null curve cannot be treated by the usual positive length norm.
\end{problem}
\begin{solution}
A nonzero null tangent has \(g(\dot\gamma,\dot\gamma)=0\), so the positive-definite formula \(\sqrt{g(\dot\gamma,\dot\gamma)}\) assigns zero despite nonzero motion.
\end{solution}

\section{Challenge problems}
\begin{problem}[Witt decomposition preview]\label{prob:vii29-ch01}
Explain how null directions lead to a basis adapted to pairs of null vectors.
\end{problem}
\begin{solution}
In an indefinite space one can choose null vectors \(\ell,n\) with \(g(\ell,n)\ne0\), rescale to \(g(\ell,n)=-1\), and complement their span by a nondegenerate orthogonal subspace. This is the basic idea behind a Witt decomposition.
\end{solution}

\begin{problem}[Reverse Cauchy-Schwarz]\label{prob:vii29-ch02}
For future-directed timelike vectors in Minkowski space, what replaces the Euclidean Cauchy--Schwarz inequality?
\end{problem}
\begin{solution}
With normalized time orientation one obtains a reversed inequality \(-g(u,v)\ge\sqrt{-g(u,u)}\sqrt{-g(v,v)}\), with equality when the vectors are positively proportional.
\end{solution}

\begin{problem}[Null orthogonal complement]\label{prob:vii29-ch03}
Why is the orthogonal complement of a null vector degenerate?
\end{problem}
\begin{solution}
If \(\ell\) is null, then \(\ell\in\ell^\perp\). Moreover \(g(\ell,w)=0\) for every \(w\in\ell^\perp\), so \(\ell\) lies in the radical of the restricted form.
\end{solution}

\begin{problem}[Constant conformal scaling]\label{prob:vii29-ch04}
How does multiplying a pseudo-Riemannian metric by a positive constant affect its signature and Levi--Civita connection?
\end{problem}
\begin{solution}
The signature is unchanged because all signs are preserved. The constant factor cancels between \(g_{ij}\) and \(g^{ij}\) in the Christoffel formula, so the Levi--Civita connection is unchanged.
\end{solution}

\begin{problem}[Hyperboloid bridge]\label{prob:vii29-ch05}
Why will an indefinite ambient metric become useful again in hyperbolic geometry?
\end{problem}
\begin{solution}
The hyperboloid model realizes hyperbolic space as one sheet of a quadratic hypersurface in Minkowski space; the ambient Lorentzian form restricts to a positive-definite metric on each tangent space of the hyperboloid.
\end{solution}

\section*{Bridge to VII/30}
VII/30 keeps the same pseudo-Riemannian formalism but specializes to Lorentzian signature and compares it point by point with positive-definite Riemannian geometry.
