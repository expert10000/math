\section{Supplementary worked examples}
\label{sec:vii34-pedagogy}

\begin{example}[Parabolic translation]\label{ex:vii34-ped-01}
The matrix \(\begin{pmatrix}1&1\\0&1\end{pmatrix}\) has trace \(2\) and acts by \(z\mapsto z+1\).  It has the single ideal fixed point \(\infty\) and translation length zero that is not attained in \(\mathbb H\).
\end{example}

\begin{example}[Hyperbolic dilation]\label{ex:vii34-ped-02}
The diagonal lift \(\operatorname{diag}(e^{\ell/2},e^{-\ell/2})\) acts by \(z\mapsto e^\ell z\).  Its axis is the imaginary line, its ideal fixed points are \(0,\infty\), and its translation length is \(\ell\).
\end{example}

\begin{example}[Elliptic rotation]\label{ex:vii34-ped-03}
An element fixing \(i\) with \(|\operatorname{tr}A|<2\) acts as a hyperbolic rotation about that interior point.  Its translation length is zero and is attained exactly at its fixed point.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Trace classify 1]\label{exr:vii34-ped-01}
Classify an element with \(|\operatorname{tr}A|=3\).
\end{exercise}

\begin{hint}
Compare with \(2\).
\end{hint}

\begin{solution}
Hyperbolic.
\end{solution}

\begin{exercise}[Trace classify 2]\label{exr:vii34-ped-02}
Classify \(|\operatorname{tr}A|=2\) for a nonidentity element.
\end{exercise}

\begin{hint}
Use the three cases.
\end{hint}

\begin{solution}
Parabolic.
\end{solution}

\begin{exercise}[Trace classify 3]\label{exr:vii34-ped-03}
Classify \(|\operatorname{tr}A|=1\).
\end{exercise}

\begin{hint}
Use the three cases.
\end{hint}

\begin{solution}
Elliptic.
\end{solution}

\begin{exercise}[Translation length]\label{exr:vii34-ped-04}
If \(|\operatorname{tr}A|=3\), express \(\tau\).
\end{exercise}

\begin{hint}
Use the trace-length relation.
\end{hint}

\begin{solution}
\(\tau=2\operatorname{arcosh}(3/2)\).
\end{solution}

\begin{exercise}[Dilation length]\label{exr:vii34-ped-05}
What is the translation length of \(z\mapsto e^2z\)?
\end{exercise}

\begin{hint}
Read the exponent in the normal form.
\end{hint}

\begin{solution}
\(2\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Fixed-point discriminant]\label{exr:vii34-ped-06}
Derive the discriminant \((\operatorname{tr}A)^2-4\) of the fixed-point equation.
\end{exercise}

\begin{hint}
Expand \((d-a)^2+4bc\) and use \(ad-bc=1\).
\end{hint}

\begin{solution}
It simplifies to \((a+d)^2-4\).
\end{solution}

\begin{exercise}[Axis invariance]\label{exr:vii34-ped-07}
Why does a hyperbolic element preserve the geodesic joining its two boundary fixed points?
\end{exercise}

\begin{hint}
Isometries send geodesics to geodesics and fix the endpoints.
\end{hint}

\begin{solution}
Uniqueness of the geodesic with those ideal endpoints forces the axis to map to itself.
\end{solution}

\begin{exercise}[Trace sign]\label{exr:vii34-ped-08}
Why is only \(|\operatorname{tr}A|\) well defined on \(PSL(2,\mathbb R)\)?
\end{exercise}

\begin{hint}
Replace \(A\) by \(-A\).
\end{hint}

\begin{solution}
The trace changes sign while the projective isometry does not.
\end{solution}

\begin{exercise}[Parabolic infimum]\label{exr:vii34-ped-09}
Why is parabolic translation length zero but not attained?
\end{exercise}

\begin{hint}
Move high into the cusp for \(z\mapsto z+1\).
\end{hint}

\begin{solution}
Horizontal displacement becomes arbitrarily small in hyperbolic units as height grows, but no interior point is fixed.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Trace zero hyperbolic]\label{exr:vii34-ped-10}
Test the claim.
\end{exercise}

\begin{hint}
Compare \(|0|\) with \(2\).
\end{hint}

\begin{solution}
False.  A nonidentity trace-zero element is elliptic.
\end{solution}

\begin{exercise}[Two interior fixed points]\label{exr:vii34-ped-11}
Test the claim for a nonidentity orientation-preserving hyperbolic isometry.
\end{exercise}

\begin{hint}
Use the classification.
\end{hint}

\begin{solution}
False.  An elliptic has one interior fixed point; hyperbolic/parabolic fixed points lie on the boundary.
\end{solution}

\begin{exercise}[Hyperbolic tau zero]\label{exr:vii34-ped-12}
Test the claim.
\end{exercise}

\begin{hint}
Use the three translation-length cases.
\end{hint}

\begin{solution}
False.  Hyperbolic elements have positive translation length.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Closed geodesic length]\label{exr:vii34-ped-13}
How does the trace of a hyperbolic deck transformation determine a quotient closed-geodesic length?
\end{exercise}

\begin{hint}
Use its axis and translation length.
\end{hint}

\begin{solution}
For a primitive element, the closed geodesic length is \(2\operatorname{arcosh}(|\operatorname{tr}A|/2)\).
\end{solution}

\begin{exercise}[Dynamics]\label{exr:vii34-ped-14}
What happens to a generic boundary point under repeated powers of a hyperbolic element?
\end{exercise}

\begin{hint}
Use attracting and repelling fixed points.
\end{hint}

\begin{solution}
Forward iterates approach the attracting fixed point and backward iterates approach the repelling fixed point.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Eigenvalue length]\label{exr:vii34-ped-15}
A determinant-one hyperbolic lift has eigenvalues \(\lambda,\lambda^{-1}\) with \(\lambda>1\).  Express translation length.
\end{exercise}

\begin{hint}
Compare with \(e^{\ell/2}\).
\end{hint}

\begin{solution}
\(\tau=2\log\lambda\).
\end{solution}

\begin{exercise}[Conjugacy challenge]\label{exr:vii34-ped-16}
Why does conjugating an isometry preserve its elliptic/parabolic/hyperbolic type?
\end{exercise}

\begin{hint}
Conjugacy transports fixed points and preserves translation length.
\end{hint}

\begin{solution}
The number and location type of fixed points, and equivalently the absolute trace class, are unchanged under conjugation.
\end{solution}
