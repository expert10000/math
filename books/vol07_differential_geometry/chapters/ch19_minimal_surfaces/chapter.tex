\chapter{Minimal Surfaces}
\label{ch:vii19-minimal-surfaces}

Minimal surfaces are the critical points of area.  For an oriented regular surface in
\(\mathbb R^3\), the local condition is strikingly simple:
\[
\boxed{H=0.}
\]
The name ``minimal'' is historical: a critical surface need not globally minimize
area, but its first variation vanishes for every compactly supported normal
variation.

\section{The mean-curvature condition}

\begin{definition}[Minimal surface]\label{def:vii19-minimal}
An oriented regular surface \(S\subset\mathbb R^3\) is minimal if its mean curvature
vanishes identically:
\[
H\equiv 0.
\]
This condition is independent of orientation because reversing the normal changes
\(H\) to \(-H\).
\end{definition}

\begin{proposition}[Principal-curvature characterization]\label{prop:vii19-principal}
A surface is minimal exactly when
\[
k_1+k_2=0.
\]
At a nonplanar minimal point the principal curvatures have opposite signs, hence
\[
K=k_1k_2=-k_1^2<0.
\]
\end{proposition}

\begin{corollary}[Gaussian-curvature sign]\label{cor:vii19-k-nonpositive}
Every minimal surface in \(\mathbb R^3\) satisfies
\[
K\le 0,
\]
with equality at a point exactly when both principal curvatures vanish there.
\end{corollary}

\section{Minimal graphs}

Let a surface be given as a graph
\[
X(x,y)=(x,y,u(x,y)).
\]

\begin{theorem}[Minimal-surface equation]\label{thm:vii19-mse}
The graph of \(u\in C^2(\Omega)\) is minimal exactly when
\[
(1+u_y^2)u_{xx}-2u_xu_yu_{xy}+(1+u_x^2)u_{yy}=0.
\]
Equivalently,
\[
\operatorname{div}\!\left(
\frac{\nabla u}{\sqrt{1+|\nabla u|^2}}
\right)=0.
\]
\end{theorem}

\begin{proof}
For the upward unit normal
\[
N=\frac{(-u_x,-u_y,1)}{\sqrt{1+|\nabla u|^2}},
\]
the first and second fundamental forms give the displayed formula for \(2H\).
Setting \(H=0\) yields the equation.
\end{proof}

\begin{remark}[Linearization]\label{rem:vii19-linearization}
For small slopes \(|\nabla u|\ll1\), the minimal-surface equation linearizes to
\[
u_{xx}+u_{yy}=0.
\]
Thus harmonic functions are the infinitesimal model for minimal graphs.
\end{remark}

\section{Area functional and first variation}

For a graph \(u:\Omega\to\mathbb R\), its area is
\[
\mathcal A[u]
=
\int_\Omega \sqrt{1+|\nabla u|^2}\,dx\,dy.
\]

\begin{theorem}[Euler--Lagrange equation]\label{thm:vii19-euler-lagrange}
A \(C^2\) graph is a critical point of \(\mathcal A\) under compactly supported
variations \(u_t=u+t\varphi\) if and only if it satisfies the minimal-surface
equation.
\end{theorem}

\begin{proof}
Differentiate under the integral:
\[
\left.\frac{d}{dt}\right|_{0}\mathcal A[u+t\varphi]
=
\int_\Omega
\frac{\nabla u\cdot\nabla\varphi}
{\sqrt{1+|\nabla u|^2}}\,dx\,dy.
\]
Integration by parts, with \(\varphi\) compactly supported, gives
\[
-\int_\Omega
\varphi\,
\operatorname{div}\!\left(
\frac{\nabla u}{\sqrt{1+|\nabla u|^2}}
\right)\,dx\,dy.
\]
The fundamental lemma of the calculus of variations gives the result.
\end{proof}

\begin{theorem}[First variation in geometric form]\label{thm:vii19-first-variation}
For a compactly supported normal variation with variational field
\(V=\phi N\),
\[
\left.\frac{d}{dt}\right|_{0}\operatorname{Area}(S_t)
=
-2\int_S H\phi\,dA
\]
under the sign convention used in VII/17.
Hence \(H=0\) is precisely the stationarity condition for area.
\end{theorem}

\section{Conformal coordinates and harmonicity}

\begin{definition}[Conformal parametrization]\label{def:vii19-conformal}
A surface parametrization \(X(u,v)\) is conformal if
\[
E=G,\qquad F=0.
\]
\end{definition}

\begin{theorem}[Harmonic coordinate criterion]\label{thm:vii19-harmonic}
For a conformal regular parametrization,
\[
X_{uu}+X_{vv}=2EH\,N.
\]
Consequently the surface is minimal if and only if each coordinate function of
\(X\) is harmonic:
\[
X_{uu}+X_{vv}=0.
\]
\end{theorem}

\section{Classical examples}

\begin{example}[Plane]\label{ex:vii19-plane}
A plane has \(S=0\), hence \(H=0\).  It is the simplest minimal surface.
\end{example}

\begin{example}[Catenoid]\label{ex:vii19-catenoid}
For \(a>0\),
\[
X(u,v)=
\bigl(a\cosh(v/a)\cos u,\,
      a\cosh(v/a)\sin u,\,
      v\bigr)
\]
parametrizes a catenoid.  Direct computation gives \(H=0\).
It is the nonplanar minimal surface of revolution.
\end{example}

\begin{example}[Helicoid]\label{ex:vii19-helicoid}
The helicoid
\[
X(u,v)=(v\cos u,v\sin u,cu)
\]
has \(H=0\).  It is ruled but, unlike a developable surface, has negative Gaussian
curvature away from degenerate limiting cases.
\end{example}

\section{Minimal surfaces of revolution}

\begin{theorem}[Catenary profile]\label{thm:vii19-revolution}
A nonplanar minimal surface of revolution has a generating profile which, after
translation and scaling, is a catenary
\[
r(z)=a\cosh\!\left(\frac{z-z_0}{a}\right).
\]
Thus the corresponding surface is a catenoid.
\end{theorem}

\section{Maximum-principle intuition}

\begin{proposition}[No strict interior graph extrema]\label{prop:vii19-no-extrema}
A nonconstant minimal graph cannot have a strict interior maximum or minimum.
\end{proposition}

\begin{remark}
This reflects ellipticity of the minimal-surface equation.  A full nonlinear
maximum-principle treatment belongs to analysis; here the point is geometric:
minimal graphs behave like nonlinear harmonic functions.
\end{remark}

\section{Area minimizing versus stationary}

\begin{warning}[Stationary is not automatically minimizing]\label{warn:vii19-stationary}
The condition \(H=0\) says the first variation vanishes.  Determining whether a
minimal surface is stable or area minimizing requires second-variation information
and global comparison arguments.
\end{warning}

\section{Boundary-value viewpoint}

The Plateau problem asks for a surface of least area spanning a prescribed closed
curve.  Its solutions, when smooth and regular, satisfy \(H=0\) in the interior.
This global problem motivates much of minimal-surface theory, but existence and
regularity require tools beyond the present chapter.

\section{Solved problem dossiers}

\begin{problem}[Legacy minimal-surface problem]\label{prob:vii19-01}
Prove the equivalence \(H=0\iff k_1+k_2=0\) and derive \(K\le0\).
\end{problem}
\begin{solution}
Use the definitions of \(H\) and \(K\). If the sum vanishes then \(k_2=-k_1\), so \(K=-k_1^2\le0\).
\end{solution}

\begin{problem}[Legacy minimal-graph problem]\label{prob:vii19-02}
Derive the minimal-surface PDE for a graph.
\end{problem}
\begin{solution}
Compute \(I\), \(II\), and \(H\), or equivalently take the Euler--Lagrange equation of the graph area integrand \(\sqrt{1+|\nabla u|^2}\).
\end{solution}

\begin{problem}[Area-variation problem]\label{prob:vii19-03}
Show compactly supported critical graphs satisfy the divergence-form PDE.
\end{problem}
\begin{solution}
Differentiate the area functional, integrate by parts, and apply the fundamental lemma.
\end{solution}

\begin{problem}[Linearization problem]\label{prob:vii19-04}
Explain why Laplace's equation is the infinitesimal model of the minimal-surface equation.
\end{problem}
\begin{solution}
At zero slope, the nonlinear coefficients reduce to \(1,0,1\), leaving \(u_{xx}+u_{yy}=0\).
\end{solution}

\begin{problem}[Conformal-coordinate problem]\label{prob:vii19-05}
Prove a conformal immersion is minimal exactly when its coordinate functions are harmonic.
\end{problem}
\begin{solution}
For \(E=G,F=0\), the normal component of \(X_{uu}+X_{vv}\) is \(2EHN\), while its tangent component vanishes; hence the vector Laplacian is zero precisely when \(H=0\).
\end{solution}

\begin{problem}[Plane problem]\label{prob:vii19-06}
Show a plane is both minimal and globally area minimizing among graphs with the same affine boundary data.
\end{problem}
\begin{solution}
The plane has \(H=0\). For the graph functional, convexity of \(\sqrt{1+|p|^2}\) gives the comparison with the affine solution.
\end{solution}

\begin{problem}[Catenoid problem]\label{prob:vii19-07}
Explain why the catenary profile appears for a minimal surface of revolution.
\end{problem}
\begin{solution}
Substituting a rotational ansatz into \(H=0\) yields an ODE with first integral \(r/\sqrt{1+r'^2}=a\), whose solutions are \(r=a\cosh((z-z_0)/a)\).
\end{solution}

\begin{problem}[Helicoid problem]\label{prob:vii19-08}
Verify the helicoid has \(H=0\).
\end{problem}
\begin{solution}
Compute the first and second fundamental coefficients; the trace of \(I^{-1}II\) vanishes.
\end{solution}

\begin{problem}[Sphere comparison problem]\label{prob:vii19-09}
Why can no open piece of a round sphere be minimal?
\end{problem}
\begin{solution}
Its two principal curvatures are equal and nonzero, so their mean cannot vanish.
\end{solution}

\begin{problem}[Stationary-versus-stable problem]\label{prob:vii19-10}
Explain why \(H=0\) alone is insufficient for stability.
\end{problem}
\begin{solution}
It annihilates the first variation only; stability requires nonnegativity of the second variation for admissible variations.
\end{solution}

\begin{problem}[Plateau problem preview]\label{prob:vii19-11}
Explain why a smooth area-minimizing spanning surface must be minimal in its interior.
\end{problem}
\begin{solution}
An interior compactly supported variation preserves the boundary. A minimizer has zero first variation for all such variations, hence \(H=0\).
\end{solution}

\begin{problem}[Intrinsic/extrinsic problem]\label{prob:vii19-12}
Why is minimality extrinsic?
\end{problem}
\begin{solution}
Mean curvature depends on how the surface sits in ambient space. Isometric surfaces can have different mean curvature.
\end{solution}

\section{Challenge problems}

\begin{problem}[Bernstein preview]\label{prob:vii19-ch01}
State the classical Bernstein phenomenon for entire minimal graphs in low dimension and explain its geometric flavor.
\end{problem}
\begin{solution}
In \(\mathbb R^3\), every entire minimal graph over \(\mathbb R^2\) is a plane. The theorem says the nonlinear minimal graph equation is globally rigid under the entire-graph hypothesis.
\end{solution}

\begin{problem}[First integral for revolution]\label{prob:vii19-ch02}
Derive the first integral leading to the catenoid.
\end{problem}
\begin{solution}
For a rotational area functional \(2\pi\int r\sqrt{1+r'^2}\,dz\), the integrand has no explicit \(z\)-dependence. The Beltrami identity yields \(r/\sqrt{1+r'^2}=a\), which integrates to the catenary.
\end{solution}

\begin{problem}[Conjugate-surface preview]\label{prob:vii19-ch03}
Why do conformal harmonic parametrizations suggest a relation between the catenoid and helicoid?
\end{problem}
\begin{solution}
Harmonic coordinate functions locally admit harmonic conjugates. Rotating the complex differential of a conformal minimal immersion can produce an associated family; the catenoid and helicoid occur as conjugate members.
\end{solution}

\begin{problem}[Minimal and developable]\label{prob:vii19-ch04}
Show that a connected surface that is both minimal and developable must be planar wherever regular.
\end{problem}
\begin{solution}
Minimal gives \(k_2=-k_1\); developable gives \(k_1k_2=0\). Thus both principal curvatures vanish. Hence \(S=0\), so the Gauss map is locally constant and the surface is planar on connected regular pieces.
\end{solution}

\begin{problem}[Scaling law]\label{prob:vii19-ch05}
How do \(H\) and \(K\) change under dilation by a factor \(\lambda>0\), and why is minimality scale invariant?
\end{problem}
\begin{solution}
Principal curvatures scale as \(1/\lambda\), so \(H\mapsto H/\lambda\) and \(K\mapsto K/\lambda^2\). Therefore \(H=0\) is preserved by dilations.
\end{solution}

\section*{Bridge to VII/20}
The first fundamental form now becomes independent of any embedding: VII/20 develops Riemannian metrics on arbitrary smooth manifolds.
