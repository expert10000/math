\chapter{Geodesics}
\label{ch:vii23-geodesics}

Straight lines in Euclidean space have zero acceleration.  On a Riemannian
manifold the intrinsic replacement is zero covariant acceleration:
\[
\boxed{\nabla_{\dot\gamma}\dot\gamma=0.}
\]
Geodesics are simultaneously the autoparallel curves of the Levi--Civita
connection and the local critical curves of the energy functional.

\section{Definition}

\begin{definition}[Geodesic]\label{def:vii23-geodesic}
A smooth curve \(\gamma:I\to M\) is an affinely parametrized geodesic if
\[
\frac{D\dot\gamma}{dt}
=
\nabla_{\dot\gamma}\dot\gamma
=
0.
\]
\end{definition}

\begin{proposition}[Constant speed]\label{prop:vii23-constant-speed}
Every Levi--Civita geodesic has constant speed.
\end{proposition}

\begin{proof}
Metric compatibility gives
\[
\frac{d}{dt}|\dot\gamma|^2
=
2g\!\left(\frac{D\dot\gamma}{dt},\dot\gamma\right)=0.
\]
\end{proof}

\begin{warning}[Affine parameters]\label{warn:vii23-affine-parameter}
A nonlinear reparametrization of a geodesic generally does not satisfy the
geodesic equation.  Affine changes \(t\mapsto at+b\) preserve the equation.
\end{warning}

\section{Coordinate geodesic equation}

Write \(\gamma(t)=(x^1(t),\dots,x^n(t))\).

\begin{theorem}[Geodesic ODE]\label{thm:vii23-ode}
The geodesic equation is
\[
\ddot x^k
+
\Gamma^k_{ij}(x(t))\dot x^i\dot x^j
=
0,
\qquad
k=1,\dots,n.
\]
\end{theorem}

\begin{proof}
Expand
\[
\nabla_{\dot\gamma}\dot\gamma
=
\left(
\ddot x^k+\Gamma^k_{ij}\dot x^i\dot x^j
\right)\partial_k.
\]
\end{proof}

\section{Existence and uniqueness}

\begin{theorem}[Initial-value problem]\label{thm:vii23-ivp}
For every \(p\in M\) and \(v\in T_pM\), there exists a unique maximal geodesic
\(\gamma_v\) satisfying
\[
\gamma_v(0)=p,
\qquad
\dot\gamma_v(0)=v.
\]
It depends smoothly on the initial data wherever defined.
\end{theorem}

\begin{proof}
The coordinate geodesic equation is a smooth second-order ODE.  Rewrite it as a
first-order system on \(TM\) and apply standard existence, uniqueness, and smooth
dependence for ODEs.
\end{proof}

\section{The exponential map}

\begin{definition}[Exponential map]\label{def:vii23-exp}
For \(p\in M\), define
\[
\exp_p(v)=\gamma_v(1)
\]
whenever the geodesic with initial velocity \(v\) exists up to time \(1\).
\end{definition}

Equivalently,
\[
\exp_p(tv)=\gamma_v(t)
\]
whenever both sides are defined.

\begin{proposition}[Derivative at the origin]\label{prop:vii23-exp-derivative}
\[
d(\exp_p)_0=\operatorname{id}_{T_pM}.
\]
\end{proposition}

\begin{proof}
For \(w\in T_pM\), the curve \(s\mapsto\exp_p(sw)\) is the geodesic with initial
velocity \(w\).  Its derivative at \(s=0\) is \(w\).
\end{proof}

\begin{corollary}[Normal neighborhood]\label{cor:vii23-normal-neighborhood}
The inverse function theorem implies that \(\exp_p\) is a diffeomorphism from a
neighborhood of \(0\in T_pM\) onto a neighborhood of \(p\in M\).
\end{corollary}

\section{Normal coordinates}

\begin{definition}[Normal coordinates]\label{def:vii23-normal-coordinates}
Choose a basis of \(T_pM\) and use \(\exp_p\) to transfer linear coordinates near
\(0\) to coordinates near \(p\).  These are normal coordinates centered at \(p\).
\end{definition}

\begin{theorem}[Metric normalization]\label{thm:vii23-normal-metric}
For an orthonormal basis used in normal coordinates,
\[
g_{ij}(p)=\delta_{ij},
\qquad
\Gamma^k_{ij}(p)=0,
\qquad
\partial_\ell g_{ij}(p)=0.
\]
\end{theorem}

\begin{remark}
These identities hold at the center point, not throughout the coordinate
neighborhood.  Curvature appears in the higher-order deviations from Euclidean
behavior.
\end{remark}

\section{Energy and length}

\begin{definition}[Energy]\label{def:vii23-energy}
For a smooth curve \(\gamma:[a,b]\to M\),
\[
E(\gamma)
=
\frac12\int_a^b|\dot\gamma|^2\,dt.
\]
\end{definition}

\begin{proposition}[Energy--length inequality]\label{prop:vii23-energy-length}
\[
L(\gamma)^2
\le
2(b-a)E(\gamma),
\]
with equality exactly when the speed is constant.
\end{proposition}

\begin{proof}
Apply Cauchy--Schwarz to
\[
L(\gamma)=\int_a^b|\dot\gamma|\,dt.
\]
\end{proof}

\begin{theorem}[First variation of energy]\label{thm:vii23-first-variation}
Let \(\gamma_s(t)\) be a variation with fixed endpoints and variational field
\(V=\partial_s\gamma_s|_{s=0}\). Then
\[
\left.\frac d{ds}\right|_{0}E(\gamma_s)
=
-\int_a^b
g\!\left(V,\frac{D\dot\gamma}{dt}\right)\,dt.
\]
Consequently the fixed-endpoint critical curves of energy are precisely the
geodesics.
\end{theorem}

\section{Local minimizing property}

\begin{theorem}[Local minimization]\label{thm:vii23-local-min}
Every sufficiently short geodesic segment minimizes length among nearby curves
with the same endpoints.
\end{theorem}

\begin{warning}[Not globally minimizing forever]\label{warn:vii23-global}
A geodesic may cease to minimize after traveling far enough.  Global minimizing
behavior is governed by cut loci, conjugate points, and completeness, which belong
to later Riemannian geometry.
\end{warning}

\section{Examples}

\begin{example}[Euclidean space]\label{ex:vii23-euclidean}
Since all Cartesian Christoffel symbols vanish,
\[
\ddot x^k=0.
\]
Thus Euclidean geodesics are affine lines.
\end{example}

\begin{example}[Round sphere]\label{ex:vii23-sphere}
The geodesics of the round sphere are great circles, parametrized with constant
speed.
\end{example}

\begin{proof}
For a unit sphere embedded in Euclidean space, the intrinsic covariant derivative
is tangential projection.  A great circle has ambient acceleration normal to the
sphere, so its tangential acceleration vanishes.
\end{proof}

\begin{example}[Cylinder]\label{ex:vii23-cylinder}
A circular cylinder is locally isometric to the Euclidean plane.  Its geodesics
therefore correspond under unrolling to straight lines: circles, generators, and
helices appear as special cases.
\end{example}

\begin{example}[Euclidean polar coordinates]\label{ex:vii23-polar}
Using
\[
\Gamma^r_{\theta\theta}=-r,
\qquad
\Gamma^\theta_{r\theta}=\frac1r,
\]
the Euclidean geodesic equations become
\[
\ddot r-r\dot\theta^2=0,
\qquad
\ddot\theta+\frac2r\dot r\dot\theta=0.
\]
These equations still describe straight lines, merely in polar coordinates.
\end{example}

\section{Geodesic completeness preview}

\begin{definition}[Geodesically complete]\label{def:vii23-complete}
A Riemannian manifold is geodesically complete if every geodesic can be extended
for all \(t\in\mathbb R\).
\end{definition}

\begin{remark}
The Hopf--Rinow theorem later links geodesic completeness with completeness of the
Riemannian distance and existence of globally minimizing geodesics between points.
\end{remark}

\section{Solved problem dossiers}

\begin{problem}[Geodesic-ODE problem]\label{prob:vii23-01}
Derive the coordinate geodesic equation.
\end{problem}
\begin{solution}
Write \(\dot\gamma=\dot x^i\partial_i\), covariantly differentiate, and use \(\nabla_{\partial_i}\partial_j=\Gamma^k_{ij}\partial_k\).
\end{solution}


\begin{problem}[Initial-value problem]\label{prob:vii23-02}
Explain geodesic existence, uniqueness, and smooth dependence from standard ODE theory.
\end{problem}
\begin{solution}
Introduce velocities \(v^i=\dot x^i\) and rewrite the geodesic equation as a first-order smooth vector field on \(TM\).
\end{solution}


\begin{problem}[Exponential-map problem]\label{prob:vii23-03}
Prove \(d(\exp_p)_0=\operatorname{id}\).
\end{problem}
\begin{solution}
For \(w\in T_pM\), differentiate the radial curve \(s\mapsto\exp_p(sw)=\gamma_w(s)\) at zero.
\end{solution}


\begin{problem}[Normal-coordinate problem]\label{prob:vii23-04}
Explain why Christoffel symbols vanish at the center of normal coordinates.
\end{problem}
\begin{solution}
Every radial line \(t\mapsto tv\) in the tangent space maps to a geodesic. Substitution into the geodesic equation for arbitrary \(v\) forces the symmetric coefficients at the center to vanish.
\end{solution}


\begin{problem}[Energy problem]\label{prob:vii23-05}
Derive \(L^2\le2(b-a)E\).
\end{problem}
\begin{solution}
Apply Cauchy--Schwarz to \(1\cdot|\dot\gamma|\) on \([a,b]\).
\end{solution}


\begin{problem}[First-variation problem]\label{prob:vii23-06}
Derive the fixed-endpoint first variation of energy.
\end{problem}
\begin{solution}
Differentiate \(g(\partial_t\gamma_s,\partial_t\gamma_s)\), exchange covariant derivatives using torsion freeness, integrate by parts, and use the fixed endpoints.
\end{solution}


\begin{problem}[Euclidean problem]\label{prob:vii23-07}
Solve the Euclidean geodesic equation.
\end{problem}
\begin{solution}
All \(\Gamma^k_{ij}=0\), so \(\ddot x^k=0\), giving \(x^k(t)=a^kt+b^k\).
\end{solution}


\begin{problem}[Polar-coordinate problem]\label{prob:vii23-08}
Derive the Euclidean geodesic equations in polar coordinates.
\end{problem}
\begin{solution}
Insert the nonzero polar Christoffel symbols into the general equation.
\end{solution}


\begin{problem}[Sphere problem]\label{prob:vii23-09}
Show a unit-speed great circle is a geodesic on the round sphere.
\end{problem}
\begin{solution}
Its ambient acceleration points radially and therefore has zero tangential projection, which equals its intrinsic covariant acceleration.
\end{solution}


\begin{problem}[Cylinder problem]\label{prob:vii23-10}
Classify cylinder geodesics by unrolling.
\end{problem}
\begin{solution}
The cylinder is locally isometric to a strip in the plane. Straight planar lines descend to constant-slope helices, with horizontal and vertical limiting cases.
\end{solution}


\begin{problem}[Local-minimization problem]\label{prob:vii23-11}
Why is a sufficiently short radial geodesic in a normal neighborhood minimizing?
\end{problem}
\begin{solution}
Use Gauss-lemma geometry: radial motion contributes directly to length, while angular motion can only increase it. Hence the radial geodesic realizes the local distance.
\end{solution}


\begin{problem}[Completeness problem]\label{prob:vii23-12}
Distinguish local geodesic existence from geodesic completeness.
\end{problem}
\begin{solution}
ODE theory always gives a local solution. Completeness requires every maximal solution to extend for all real parameter values.
\end{solution}

\section{Challenge problems}

\begin{problem}[Affine reparametrization]\label{prob:vii23-ch01}
Show explicitly why \(t\mapsto at+b\) preserves the geodesic equation and a general nonlinear change does not.
\end{problem}
\begin{solution}
Under \(\widetilde\gamma(s)=\gamma(\phi(s))\), covariant acceleration becomes \((\phi')^2D\dot\gamma/dt+\phi''\dot\gamma\). For a geodesic the first term vanishes, leaving \(\phi''\dot\gamma\), so affine \(\phi\) is required.
\end{solution}


\begin{problem}[Geodesic spray]\label{prob:vii23-ch02}
Recast the geodesic equation as a vector field on \(TM\).
\end{problem}
\begin{solution}
In coordinates \((x^i,v^i)\), the geodesic spray is \(G=v^i\partial_{x^i}-\Gamma^k_{ij}v^iv^j\partial_{v^k}\). Its integral curves project to geodesics.
\end{solution}


\begin{problem}[Normal-coordinate metric expansion preview]\label{prob:vii23-ch03}
Why can curvature first appear only at second order in the metric coefficients in normal coordinates?
\end{problem}
\begin{solution}
At the center \(g_{ij}=\delta_{ij}\) and all first derivatives vanish. Therefore the first nontrivial Taylor data are second derivatives, which are constrained by the Riemann curvature tensor.
\end{solution}


\begin{problem}[Geodesic on an embedded surface]\label{prob:vii23-ch04}
Show that a unit-speed curve on a surface is geodesic exactly when its ambient acceleration is normal to the surface.
\end{problem}
\begin{solution}
The intrinsic acceleration is the tangential projection of the ambient acceleration by the induced Levi--Civita connection. It vanishes exactly when ambient acceleration has no tangent component.
\end{solution}


\begin{problem}[Constant-speed energy equivalence]\label{prob:vii23-ch05}
For fixed endpoints and fixed parameter interval, explain why minimizing energy and minimizing length agree among constant-speed curves.
\end{problem}
\begin{solution}
For constant speed equality holds in \(L^2\le2(b-a)E\), giving \(E=L^2/[2(b-a)]\). Thus the two functionals are monotone functions of one another on that class.
\end{solution}
