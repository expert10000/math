\section{Supplementary worked examples}
\label{sec:vii23-pedagogy}

\begin{example}[Euclidean geodesics]\label{ex:vii23-ped-01}
With zero Christoffel symbols, the geodesic equation reduces to \(\ddot x^k=0\), so affinely parametrized geodesics are straight lines \(x(t)=p+tv\).
\end{example}

\begin{example}[Great circles]\label{ex:vii23-ped-02}
On the round sphere, geodesics are great circles.  For \(p\in S^n\) and unit \(v\in T_pS^n\), \(\exp_p(tv)=\cos t\,p+\sin t\,v\).  This gives an explicit exponential map until the antipodal cut point.
\end{example}

\begin{example}[Affine versus nonlinear parameters]\label{ex:vii23-ped-03}
The image of a Euclidean line remains a geodesic curve after any monotone reparametrization, but the equation \(\nabla_{\dot\gamma}\dot\gamma=0\) is preserved only by affine parameter changes.  Parameter choice is therefore part of the differential equation.
\end{example}

\section{Graded supplementary exercises}
The following exercises add a deliberate progression from concrete calculation to structural proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Straight-line IVP]\label{exr:vii23-ped-01}
Solve the Euclidean geodesic IVP \(\gamma(0)=p\), \(\dot\gamma(0)=v\).
\end{exercise}

\begin{hint}
Use \(\ddot\gamma=0\).
\end{hint}

\begin{solution}
\(\gamma(t)=p+tv\).
\end{solution}

\begin{exercise}[Constant speed]\label{exr:vii23-ped-02}
Why does an affinely parametrized Riemannian geodesic have constant speed?
\end{exercise}

\begin{hint}
Differentiate its squared speed.
\end{hint}

\begin{solution}
Metric compatibility and zero covariant acceleration give derivative zero.
\end{solution}

\begin{exercise}[Polar radial geodesic]\label{exr:vii23-ped-03}
Check that \(\theta=\theta_0\), \(r=at+b\) solves the polar geodesic equations where \(r>0\).
\end{exercise}

\begin{hint}
Set \(\dot\theta=0\).
\end{hint}

\begin{solution}
Both equations reduce to \(\ddot r=0\) and \(0=0\).
\end{solution}

\begin{exercise}[Circle in plane]\label{exr:vii23-ped-04}
Is the Euclidean circle \(r=R\) a geodesic of the plane?
\end{exercise}

\begin{hint}
Insert \(\dot r=0\), \(\dot\theta\ne0\) in the radial equation.
\end{hint}

\begin{solution}
No.  The radial equation contains \(-R\dot\theta^2\ne0\).
\end{solution}

\begin{exercise}[Exponential derivative]\label{exr:vii23-ped-05}
What is \(d(\exp_p)_0\)?
\end{exercise}

\begin{hint}
Differentiate \(s\mapsto\exp_p(sw)\) at zero.
\end{hint}

\begin{solution}
It is the identity on \(T_pM\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Affine reparametrization]\label{exr:vii23-ped-06}
Prove \(t\mapsto\gamma(at+b)\) is geodesic if \(\gamma\) is and \(a\ne0\).
\end{exercise}

\begin{hint}
Differentiate twice covariantly.
\end{hint}

\begin{solution}
Covariant acceleration scales by \(a^2\), hence remains zero.
\end{solution}

\begin{exercise}[Nonlinear reparametrization]\label{exr:vii23-ped-07}
Show a nonlinear reparametrization generally introduces acceleration parallel to the tangent.
\end{exercise}

\begin{hint}
Apply the chain rule to \(\gamma(\phi(t))\).
\end{hint}

\begin{solution}
An extra term \(\phi''\dot\gamma\) appears, so the affine geodesic equation fails unless \(\phi''=0\).
\end{solution}

\begin{exercise}[Normal coordinates]\label{exr:vii23-ped-08}
Why do Christoffel symbols vanish at the center of normal coordinates?
\end{exercise}

\begin{hint}
Coordinate rays are geodesics through the center in every direction.
\end{hint}

\begin{solution}
The quadratic geodesic term must vanish for all initial velocities, forcing the symmetric Christoffel coefficients to vanish there.
\end{solution}

\begin{exercise}[Energy criticality]\label{exr:vii23-ped-09}
State the relation between geodesics and the energy functional with fixed endpoints.
\end{exercise}

\begin{hint}
Use first variation.
\end{hint}

\begin{solution}
Constant-speed geodesics are exactly the critical points of the energy under fixed-endpoint variations.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Geodesic equals minimizer]\label{exr:vii23-ped-10}
Test the claim: every geodesic segment globally minimizes distance.
\end{exercise}

\begin{hint}
Consider a long great-circle arc on a sphere.
\end{hint}

\begin{solution}
False.  Geodesics are locally minimizing only up to cut/conjugate phenomena; long arcs need not be global minimizers.
\end{solution}

\begin{exercise}[Unique geodesic globally]\label{exr:vii23-ped-11}
Test the claim: any two points on a connected Riemannian manifold have a unique geodesic joining them.
\end{exercise}

\begin{hint}
Use antipodal points on a sphere.
\end{hint}

\begin{solution}
False.  Antipodal sphere points are joined by infinitely many great-circle geodesics.
\end{solution}

\begin{exercise}[Zero Christoffels neighborhood]\label{exr:vii23-ped-12}
Test the claim: normal coordinates make the metric Euclidean throughout a neighborhood.
\end{exercise}

\begin{hint}
They normalize only low-order data at one point.
\end{hint}

\begin{solution}
False.  Curvature appears in higher-order terms away from the center.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Shooting method]\label{exr:vii23-ped-13}
Why is the exponential map central to numerical geodesic shooting?
\end{exercise}

\begin{hint}
It maps an initial velocity to the endpoint at unit time.
\end{hint}

\begin{solution}
A boundary-value problem can be converted into solving \(\exp_p(v)=q\) for the unknown initial velocity \(v\).
\end{solution}

\begin{exercise}[Geodesic interpolation]\label{exr:vii23-ped-14}
How does a short minimizing geodesic generalize linear interpolation?
\end{exercise}

\begin{hint}
Compare \((1-t)p+tq\) with \(\exp_p\).
\end{hint}

\begin{solution}
In a normal neighborhood one can write \(\gamma(t)=\exp_p(t\exp_p^{-1}(q))\), the intrinsic analogue of a straight segment.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Sphere exponential]\label{exr:vii23-ped-15}
For unit sphere data \(p\perp v\), \(|v|=1\), verify \(\gamma(t)=\cos t\,p+\sin t\,v\) stays on the sphere and has unit speed.
\end{exercise}

\begin{hint}
Compute norm and derivative.
\end{hint}

\begin{solution}
Both norms are \(1\) by orthogonality, so the curve lies on \(S^n\) and has unit speed.
\end{solution}

\begin{exercise}[Cut point challenge]\label{exr:vii23-ped-16}
On the unit sphere, at what distance from \(p\) does the radial exponential first cease to be one-to-one?
\end{exercise}

\begin{hint}
Think of antipodal points.
\end{hint}

\begin{solution}
At distance \(\pi\); all unit directions of length \(\pi\) map to the antipode.
\end{solution}
