\chapter{Hyperbolic Plane Models}
\label{ch:vii31-hyperbolic-models}

Hyperbolic geometry is the model geometry of constant negative curvature.  There is
not a single preferred coordinate picture.  Instead several equivalent models
make different structures visible:
\[
\boxed{
\text{hyperboloid}
\longleftrightarrow
\text{disk}
\longleftrightarrow
\text{upper half-plane}
\longleftrightarrow
\text{Klein disk}.
}
\]
This chapter develops the spaces and their explicit identifications.  VII/32
focuses on the Poincaré metric itself.

\section{The hyperboloid model}

Let \(\mathbb R^{2,1}\) carry the Lorentzian bilinear form
\[
\langle x,y\rangle_L
=
x_1y_1+x_2y_2-x_3y_3.
\]

\begin{definition}[Hyperboloid model]\label{def:vii31-hyperboloid}
The hyperboloid model is
\[
\mathbb H^2_{\mathrm{hyp}}
=
\left\{
x\in\mathbb R^{2,1}:
\langle x,x\rangle_L=-1,\ x_3>0
\right\}.
\]
\end{definition}

\begin{proposition}[Positive tangent metric]\label{prop:vii31-positive-tangent}
For \(x\in\mathbb H^2_{\mathrm{hyp}}\),
\[
T_x\mathbb H^2_{\mathrm{hyp}}
=
\{v:\langle v,x\rangle_L=0\}.
\]
The Lorentzian form restricted to this tangent plane is positive definite.
\end{proposition}

\begin{proof}
Differentiating \(\langle x,x\rangle_L=-1\) gives the tangent equation.  Since
\(x\) is timelike, its Lorentz-orthogonal complement is spacelike and positive
definite by VII/29.
\end{proof}

\section{Geodesics in the hyperboloid picture}

\begin{proposition}[Plane sections]\label{prop:vii31-hyp-geodesics}
Hyperbolic geodesics in the hyperboloid model are intersections of
\(\mathbb H^2_{\mathrm{hyp}}\) with two-dimensional linear subspaces through the
origin on which the Lorentzian form has signature \((1,1)\).
\end{proposition}

This is the hyperbolic analogue of great circles on the sphere.

\section{The Poincaré disk as a set}

\begin{definition}[Disk model]\label{def:vii31-disk}
The Poincaré disk is the open unit disk
\[
\mathbb D
=
\{z\in\mathbb C:|z|<1\}.
\]
\end{definition}

The Euclidean boundary circle
\[
S^1=\{|z|=1\}
\]
does not belong to the hyperbolic plane.  It represents ideal boundary points.

\begin{definition}[Ideal boundary]\label{def:vii31-ideal}
The ideal boundary of the disk model is
\[
\partial_\infty\mathbb H^2\cong S^1.
\]
\end{definition}

\section{Hyperboloid to disk}

\begin{theorem}[Stereographic-type projection]\label{thm:vii31-hyp-disk-map}
The map
\[
\Phi(x_1,x_2,x_3)
=
\frac{x_1+ix_2}{x_3+1}
\]
sends \(\mathbb H^2_{\mathrm{hyp}}\) diffeomorphically onto \(\mathbb D\).
Its inverse is
\[
\Phi^{-1}(z)
=
\left(
\frac{2\Re z}{1-|z|^2},
\frac{2\Im z}{1-|z|^2},
\frac{1+|z|^2}{1-|z|^2}
\right).
\]
\end{theorem}

\begin{proof}
Substituting the inverse formula gives Lorentz norm \(-1\) and third coordinate
positive.  Direct substitution in the two compositions gives the identity.
\end{proof}

\section{The upper half-plane}

\begin{definition}[Upper half-plane]\label{def:vii31-halfplane}
The upper half-plane model is
\[
\mathbb H
=
\{z=x+iy\in\mathbb C:y>0\}.
\]
\end{definition}

Its ideal boundary is
\[
\mathbb R\cup\{\infty\}.
\]

\section{Cayley transform}

\begin{theorem}[Disk--half-plane equivalence]\label{thm:vii31-cayley}
The Cayley transform
\[
C(z)
=
i\,\frac{1+z}{1-z}
\]
maps \(\mathbb D\) biholomorphically onto \(\mathbb H\).  Its inverse is
\[
C^{-1}(w)
=
\frac{w-i}{w+i}.
\]
\end{theorem}

\begin{proposition}[Boundary correspondence]\label{prop:vii31-boundary-map}
The Cayley transform extends to the ideal boundaries in the Möbius sense:
\[
S^1
\longleftrightarrow
\mathbb R\cup\{\infty\}.
\]
In particular,
\[
1\mapsto\infty,\qquad
-1\mapsto0,\qquad
i\mapsto-1.
\]
\end{proposition}

\section{Geodesics in conformal models}

\begin{theorem}[Disk geodesic shapes]\label{thm:vii31-disk-geodesics}
In the Poincaré disk, geodesics are Euclidean diameters and circular arcs meeting
the boundary circle orthogonally.
\end{theorem}

\begin{theorem}[Half-plane geodesic shapes]\label{thm:vii31-half-geodesics}
In the upper half-plane, geodesics are vertical lines and semicircles meeting the
real axis orthogonally.
\end{theorem}

\begin{remark}
VII/32 derives these facts from the Poincaré metric.  Here they provide the
geometric dictionary among models.
\end{remark}

\section{The Klein disk}

\begin{definition}[Klein model]\label{def:vii31-klein}
The Klein disk uses the same underlying open unit disk, but with a different
projective realization of hyperbolic geometry.
\end{definition}

\begin{theorem}[Central projection]\label{thm:vii31-klein-map}
The map
\[
K(x_1,x_2,x_3)
=
\left(\frac{x_1}{x_3},\frac{x_2}{x_3}\right)
\]
sends the hyperboloid onto the open Euclidean unit disk.
Its inverse is
\[
K^{-1}(u)
=
\frac{(u_1,u_2,1)}{\sqrt{1-|u|^2}}.
\]
\end{theorem}

\begin{proposition}[Klein geodesics]\label{prop:vii31-klein-geodesics}
Hyperbolic geodesics appear in the Klein model as Euclidean chords of the disk.
\end{proposition}

\begin{warning}[Klein is not conformal]\label{warn:vii31-klein}
The Klein model makes geodesics straight, but it does not preserve Euclidean
angles.  The Poincaré disk is conformal; the Klein disk is projective.
\end{warning}

\section{Disk--Klein conversion}

For \(z\in\mathbb D\), identify \(z\) with its vector in \(\mathbb R^2\).

\begin{proposition}[Radial conversion]\label{prop:vii31-disk-klein}
The Poincaré-disk coordinate \(z\) and Klein coordinate \(u\) for the same
hyperboloid point satisfy
\[
u=\frac{2z}{1+|z|^2}.
\]
Conversely,
\[
z
=
\frac{u}{1+\sqrt{1-|u|^2}}.
\]
\end{proposition}

\section{Ideal points and asymptotic geodesics}

\begin{definition}[Ideal endpoint]\label{def:vii31-ideal-endpoint}
A hyperbolic geodesic has two endpoints on
\[
\partial_\infty\mathbb H^2.
\]
These endpoints are not points of \(\mathbb H^2\), but encode the asymptotic
direction of the geodesic.
\end{definition}

Two distinct geodesics may share one ideal endpoint; they are then asymptotic.

\section{Model comparison}

\begin{proposition}[What each model makes easy]\label{prop:vii31-comparison}
The hyperboloid model makes Lorentz symmetry and curvature algebra transparent.
The Poincaré disk makes angles and ideal-boundary symmetry visual.  The upper
half-plane makes real Möbius transformations natural.  The Klein disk makes
geodesics straight.
\end{proposition}

\section{Negative curvature preview}

All these models describe the same simply connected complete surface of constant
sectional curvature \(-1\), once equipped with their corresponding hyperbolic
metrics.

\begin{warning}[Euclidean pictures are not Euclidean geometry]\label{warn:vii31-euclidean-picture}
The disk and half-plane are subsets of the Euclidean plane only as coordinate
models.  Euclidean lengths, areas, and straightness are generally not the
hyperbolic ones.
\end{warning}

\section{Solved problem dossiers}
\begin{problem}[Hyperboloid tangent problem]\label{prob:vii31-01}
Derive the tangent-space equation for the hyperboloid.
\end{problem}
\begin{solution}
For a curve \(x(t)\) on the hyperboloid, differentiate \(\langle x(t),x(t)\rangle_L=-1\) to obtain \(2\langle x,\dot x\rangle_L=0\).
\end{solution}

\begin{problem}[Positive-metric problem]\label{prob:vii31-02}
Prove the hyperboloid's induced metric is Riemannian.
\end{problem}
\begin{solution}
The position vector \(x\) is timelike. Its Lorentz-orthogonal complement is a positive-definite two-plane, and this complement is exactly the tangent space.
\end{solution}

\begin{problem}[Disk-map problem]\label{prob:vii31-03}
Verify the stated inverse \(\Phi^{-1}\) lands on the hyperboloid.
\end{problem}
\begin{solution}
Compute the Lorentz norm of \((2x,2y,1+r^2)/(1-r^2)\); the numerator becomes \(4r^2-(1+r^2)^2=-(1-r^2)^2\).
\end{solution}

\begin{problem}[Cayley problem]\label{prob:vii31-04}
Show \(C(z)=i(1+z)/(1-z)\) maps \(\mathbb D\) to the upper half-plane.
\end{problem}
\begin{solution}
A direct calculation gives \(\operatorname{Im}C(z)=(1-|z|^2)/|1-z|^2>0\).
\end{solution}

\begin{problem}[Boundary problem]\label{prob:vii31-05}
Explain the ideal-boundary correspondence between \(S^1\) and \(\mathbb R\cup\{\infty\}\).
\end{problem}
\begin{solution}
The Möbius Cayley transform extends to the Riemann sphere, carrying the unit circle to the extended real line.
\end{solution}

\begin{problem}[Disk-geodesic problem]\label{prob:vii31-06}
How does the Cayley transform carry disk geodesics to half-plane geodesics?
\end{problem}
\begin{solution}
Möbius maps send generalized circles to generalized circles and preserve angles; therefore circles orthogonal to \(S^1\) map to circles or lines orthogonal to the real axis.
\end{solution}

\begin{problem}[Klein-map problem]\label{prob:vii31-07}
Verify central projection maps the hyperboloid inside the unit disk.
\end{problem}
\begin{solution}
From \(x_1^2+x_2^2-x_3^2=-1\), one gets \((x_1^2+x_2^2)/x_3^2=1-1/x_3^2<1\).
\end{solution}

\begin{problem}[Klein-geodesic problem]\label{prob:vii31-08}
Explain why hyperboloid geodesics become chords in the Klein disk.
\end{problem}
\begin{solution}
A hyperboloid geodesic lies in a plane through the origin. Central projection sends that plane's intersection with \(x_3=1\) to a Euclidean line.
\end{solution}

\begin{problem}[Disk-Klein problem]\label{prob:vii31-09}
Derive \(u=2z/(1+|z|^2)\).
\end{problem}
\begin{solution}
Substitute the disk inverse \(\Phi^{-1}(z)\) into \(K(x)=(x_1/x_3,x_2/x_3)\).
\end{solution}

\begin{problem}[Ideal-endpoint problem]\label{prob:vii31-10}
Why are boundary points not points of the hyperbolic plane?
\end{problem}
\begin{solution}
In disk and half-plane coordinates the hyperbolic metric sends the boundary to infinite distance; the boundary records limiting directions rather than interior points.
\end{solution}

\begin{problem}[Model-choice problem]\label{prob:vii31-11}
Which model is best suited to straight-line convexity arguments?
\end{problem}
\begin{solution}
The Klein model, because its geodesics are Euclidean chords, though it sacrifices conformality.
\end{solution}

\begin{problem}[Model-synthesis problem]\label{prob:vii31-12}
Explain why the four pictures describe one geometry rather than four unrelated geometries.
\end{problem}
\begin{solution}
Explicit diffeomorphisms identify the underlying spaces, and when each is equipped with the transported hyperbolic metric these maps are isometries. Different models emphasize different features of the same constant-curvature surface.
\end{solution}

\section{Challenge problems}
\begin{problem}[Hyperboloid distance preview]\label{prob:vii31-ch01}
Guess the invariant quantity that will determine hyperbolic distance between \(x,y\) on the hyperboloid.
\end{problem}
\begin{solution}
The Lorentz product is preserved by \(O^+(2,1)\); for unit timelike points one has \(-\langle x,y\rangle_L\ge1\), and the distance is characterized by \(\cosh d(x,y)=-\langle x,y\rangle_L\).
\end{solution}

\begin{problem}[Geodesic endpoints]\label{prob:vii31-ch02}
Given two distinct ideal boundary points in the disk, why is there a unique hyperbolic geodesic joining them?
\end{problem}
\begin{solution}
There is a unique generalized Euclidean circle orthogonal to the unit circle through the two boundary points, with a diameter as the limiting straight-line case.
\end{solution}

\begin{problem}[Cross-model vertical line]\label{prob:vii31-ch03}
Under the inverse Cayley map, what kind of disk geodesic corresponds to a vertical half-plane geodesic?
\end{problem}
\begin{solution}
A vertical line is a generalized circle through \(\infty\) orthogonal to the real axis. Möbius inversion carries it to a generalized circle orthogonal to \(S^1\), hence a disk geodesic.
\end{solution}

\begin{problem}[Klein boundary behavior]\label{prob:vii31-ch04}
Why do Klein and Poincaré disk coordinates share the same ideal circle although interior points differ radially?
\end{problem}
\begin{solution}
The radial map \(u=2z/(1+|z|^2)\) tends to \(u=z\) as \(|z|\to1\), so it fixes every boundary direction in the limit.
\end{solution}

\begin{problem}[Lorentz group bridge]\label{prob:vii31-ch05}
Why should Lorentz transformations preserving the upper hyperboloid become hyperbolic isometries?
\end{problem}
\begin{solution}
They preserve the ambient Lorentz form and the hyperboloid equation, hence preserve the induced tangent metric. Later chapters identify these symmetries with Möbius transformations through the model maps.
\end{solution}
