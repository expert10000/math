\chapter*{Solutions for VI/03}
\addcontentsline{toc}{chapter}{Solutions for VI/03}

\paragraph{Exercise \ref{ex:vi03-kernel}.}
A polynomial \(f\) lies in the kernel exactly when its restriction is the zero function:
\[
f(a)=0
\]
for every \(a\in X\).  By definition this means \(f\in I(X)\).

\paragraph{Exercise \ref{ex:vi03-same-function}.}
The restrictions agree exactly when
\[
(f-g)|_X=0.
\]
By Exercise \ref{ex:vi03-kernel}, this holds exactly when
\[
f-g\in I(X).
\]

\paragraph{Exercise \ref{ex:vi03-point}.}
We have
\[
I(\{a\})=(x_1-a_1,\dots,x_n-a_n).
\]
Therefore
\[
k[\{a\}]
=
k[x_1,\dots,x_n]/(x_1-a_1,\dots,x_n-a_n)
\cong k
\]
via evaluation at \(a\).

\paragraph{Exercise \ref{ex:vi03-line}.}
Because \(b\neq0\),
\[
y=b^{-1}(c-ax)
\]
on \(L\).  Thus
\[
k[L]
=
k[x,y]/(ax+by-c).
\]
Define
\[
\phi:k[x,y]/(ax+by-c)\to k[t]
\]
by
\[
\bar x\mapsto t,
\qquad
\bar y\mapsto b^{-1}(c-at).
\]
The inverse sends \(t\mapsto\bar x\).

\paragraph{Exercise \ref{ex:vi03-parabola}.}
Define
\[
\phi:k[x,y]/(y-x^2)\to k[t]
\]
by
\[
\bar x\mapsto t,\qquad\bar y\mapsto t^2.
\]
Define
\[
\psi:k[t]\to k[x,y]/(y-x^2)
\]
by
\[
t\mapsto\bar x.
\]
Then
\[
(\psi\circ\phi)(\bar x)=\bar x,
\qquad
(\psi\circ\phi)(\bar y)=\bar x^2=\bar y,
\]
and
\[
(\phi\circ\psi)(t)=t.
\]
Hence the maps are inverse isomorphisms.

\paragraph{Exercise \ref{ex:vi03-hyperbola}.}
In
\[
A=k[x,y]/(xy-1),
\]
the classes satisfy
\[
\bar x\bar y=1.
\]
Thus \(\bar x\) is invertible.  The map
\[
k[t,t^{-1}]\to A,
\qquad
t\mapsto\bar x,\quad t^{-1}\mapsto\bar y
\]
is an isomorphism, with inverse given by
\[
\bar x\mapsto t,\qquad\bar y\mapsto t^{-1}.
\]

\paragraph{Exercise \ref{ex:vi03-reduced}.}
Suppose \([f]^m=0\) in \(k[X]\).  Then
\[
f^m\in I(X).
\]
The ideal \(I(X)\) is radical, so
\[
f\in I(X).
\]
Therefore \([f]=0\), and \(k[X]\) is reduced.

\paragraph{Exercise \ref{ex:vi03-generators}.}
Every element of \(k[X]\) is the class of a polynomial
\[
f(x_1,\dots,x_n),
\]
hence is
\[
f(\bar x_1,\dots,\bar x_n).
\]
Thus the coordinate classes generate.  Moreover,
\[
F(\bar x_1,\dots,\bar x_n)=0
\]
exactly when \(F\) lies in the kernel of the quotient map, which is \(I(X)\).

\paragraph{Exercise \ref{ex:vi03-evaluation}.}
The map
\[
\operatorname{ev}_a([f])=f(a)
\]
is well-defined because two representatives differ by an element of \(I(X)\).  It is
surjective because every \(c\in k\) is the value of the constant class \(c\).  Hence
\[
k[X]/\ker(\operatorname{ev}_a)\cong k,
\]
a field, so the kernel is maximal.

\paragraph{Exercise \ref{ex:vi03-closed-subset}.}
Since \(Y\subseteq X\),
\[
I(X)\subseteq I(Y).
\]
Therefore
\[
\frac{k[x_1,\dots,x_n]/I(X)}{I(Y)/I(X)}
\cong
k[x_1,\dots,x_n]/I(Y).
\]
The left side is \(k[X]/(I(Y)/I(X))\), while the right side is \(k[Y]\).

\paragraph{Exercise \ref{ex:vi03-two-points}.}
Since \(a\neq b\), the ideals
\[
(x-a),\qquad(x-b)
\]
are comaximal.  Also
\[
I(\{a,b\})=((x-a)(x-b)).
\]
The Chinese remainder theorem gives
\[
k[x]/((x-a)(x-b))
\cong
k[x]/(x-a)\times k[x]/(x-b)
\cong
k\times k.
\]

\paragraph{Exercise \ref{ex:vi03-finite-points}.}
The ideals
\[
(x-a_i)
\]
are pairwise comaximal, and
\[
I(X)=\left(\prod_{i=1}^r(x-a_i)\right).
\]
Therefore
\[
k[X]
\cong
\prod_{i=1}^r k[x]/(x-a_i)
\cong
k^r.
\]

\paragraph{Exercise \ref{ex:vi03-zero-divisors}.}
In
\[
k[X]=k[x,y]/(xy),
\]
the classes \(\bar x,\bar y\) are nonzero but satisfy
\[
\bar x\bar y=0.
\]
They are zero divisors.  Geometrically, \(X\) is the union
\[
V(x)\cup V(y)
\]
of two components.

\paragraph{Exercise \ref{ex:vi03-nilpotent-warning}.}
The first ring is
\[
A\cong k.
\]
In
\[
B=k[x]/(x^2),
\]
the class \(\bar x\) is nonzero but
\[
\bar x^2=0,
\]
so it is nilpotent.  A classical coordinate ring is reduced, so \(B\) cannot be the
coordinate ring of the classical one-point set \(V(x)\).

\paragraph{Exercise \ref{ex:vi03-finite-field}.}
Every \(a\in\mathbb F_q\) satisfies
\[
a^q=a,
\]
so \(x^q-x\) vanishes everywhere.  If \(f\) also vanishes everywhere, divide:
\[
f=q(x)(x^q-x)+r(x),
\qquad
\deg r<q.
\]
Then \(r\) vanishes at all \(q\) elements of \(\mathbb F_q\).  A nonzero polynomial of
degree \(<q\) cannot have \(q\) distinct roots, so \(r=0\).  Hence
\[
I(\mathbb A^1_{\mathbb F_q})=(x^q-x).
\]

\paragraph{Exercise \ref{ex:vi03-challenge-circle}.}
The defining relation gives
\[
\bar x^2+\bar y^2=1.
\]
Over \(\mathbb C\),
\[
(\bar x+i\bar y)(\bar x-i\bar y)
=
\bar x^2+\bar y^2
=
1.
\]
Thus
\[
u=\bar x+i\bar y
\]
is invertible with
\[
u^{-1}=\bar x-i\bar y.
\]
