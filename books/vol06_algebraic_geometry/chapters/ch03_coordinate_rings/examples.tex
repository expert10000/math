% Canonical worked examples for VI/03.

\begin{example}[The affine line]
For an infinite field \(k\),
\[
I(\mathbb A^1_k)=(0),
\]
so
\[
k[\mathbb A^1_k]=k[x].
\]
The element \(x\) is the coordinate function
\[
a\longmapsto a.
\]
\end{example}

\begin{example}[A point]
For \(a\in k\),
\[
I(\{a\})=(x-a),
\]
hence
\[
k[\{a\}]
=
k[x]/(x-a)
\cong k.
\]
Every polynomial function on a one-point space is constant.
\end{example}

\begin{example}[A parabola]
Let
\[
X=V(y-x^2)\subseteq\mathbb A_k^2.
\]
Assuming
\[
I(X)=(y-x^2),
\]
we obtain
\[
k[X]=k[x,y]/(y-x^2).
\]
Define
\[
\Phi:k[x,y]/(y-x^2)\to k[t]
\]
by
\[
\bar x\mapsto t,\qquad \bar y\mapsto t^2.
\]
This is an isomorphism, so the parabola has the same coordinate ring as the affine line.
\end{example}

\begin{example}[A hyperbola]
Let
\[
X=V(xy-1)\subseteq\mathbb A_k^2.
\]
Then
\[
k[X]=k[x,y]/(xy-1).
\]
Inside the quotient,
\[
\bar x\bar y=1,
\]
so \(\bar x\) is invertible with inverse \(\bar y\).  Therefore
\[
k[X]\cong k[t,t^{-1}],
\]
the Laurent polynomial ring.
\end{example}

\begin{example}[Two axes and zero divisors]
Let
\[
X=V(xy)\subseteq\mathbb A_k^2
\]
over an algebraically closed field.  Then
\[
k[X]=k[x,y]/(xy).
\]
The nonzero classes \(\bar x,\bar y\) satisfy
\[
\bar x\bar y=0.
\]
The ring detects the reducible geometry through zero divisors.
\end{example}

\begin{example}[A finite collection of points]
Let
\[
X=\{a_1,\dots,a_r\}\subseteq\mathbb A^1_k
\]
with distinct \(a_i\).  Then
\[
I(X)
=
\left(\prod_{i=1}^r(x-a_i)\right).
\]
By the Chinese remainder theorem,
\[
k[X]
\cong
\prod_{i=1}^r k.
\]
Thus a polynomial function on \(X\) is simply an arbitrary \(r\)-tuple of values.
\end{example}

\begin{example}[A closed subset as a quotient]
Let
\[
X=\mathbb A_k^2
\]
and
\[
Y=V(y-x^2).
\]
Then
\[
k[X]=k[x,y]
\]
and
\[
k[Y]=k[x,y]/(y-x^2).
\]
The inclusion
\[
Y\subseteq X
\]
corresponds to the quotient homomorphism
\[
k[x,y]\twoheadrightarrow k[x,y]/(y-x^2).
\]
\end{example}

\begin{example}[Finite-field affine line]
For \(k=\mathbb F_p\), every element satisfies \(a^p=a\), so
\[
I(\mathbb A^1_{\mathbb F_p})=(x^p-x).
\]
Hence
\[
\mathbb F_p[\mathbb A^1_{\mathbb F_p}]
=
\mathbb F_p[x]/(x^p-x).
\]
Because
\[
x^p-x=\prod_{a\in\mathbb F_p}(x-a),
\]
the Chinese remainder theorem gives
\[
\mathbb F_p[x]/(x^p-x)
\cong
\prod_{a\in\mathbb F_p}\mathbb F_p.
\]
This is the ring of all functions from the finite set \(\mathbb F_p\) to itself.
\end{example}
