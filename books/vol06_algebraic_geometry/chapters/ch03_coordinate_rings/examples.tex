% Worked examples for VI/03.

\section{Worked examples}
\label{sec:vi03-worked-examples}

\begin{example}[Affine space]
If \(k\) is infinite, then \(I(\mathbb A_k^n)=(0)\), so
\[
k[\mathbb A_k^n]=k[x_1,\dots,x_n].
\]
The coordinate functions are algebraically independent.
\end{example}

\begin{example}[A rational point]
For \(a=(a_1,\dots,a_n)\),
\[
k[\{a\}]
\cong
\frac{k[x_1,\dots,x_n]}{(x_1-a_1,\dots,x_n-a_n)}
\cong k.
\]
Every polynomial function on a one-point space is constant.
\end{example}

\begin{example}[A non-coordinate line]
Assume \(b\neq0\) and
\[
L=V(ax+by-c)\subseteq\mathbb A_k^2.
\]
Solving for \(y\) gives
\[
y=\frac{c-ax}{b},
\]
so \(k[L]\cong k[x]\cong k[t]\).  A linear change of ambient coordinates does not alter
the abstract coordinate ring of a line.
\end{example}

\begin{example}[Parabola]
For infinite \(k\),
\[
P=V(y-x^2),
\qquad
k[P]=k[x,y]/(y-x^2).
\]
The map \([x]\mapsto t\), \([y]\mapsto t^2\) gives an isomorphism \(k[P]\cong k[t]\).
Thus the plane parabola and affine line have isomorphic coordinate rings.
\end{example}

\begin{example}[Hyperbola and a Laurent polynomial ring]
Assume \(k\) is infinite and let \(H=V(xy-1)\).  On \(H\), \(x\) and \(y\) are mutually
inverse.  Hence
\[
k[H]\cong k[t,t^{-1}],
\]
with \(t\leftrightarrow\bar x\) and \(t^{-1}\leftrightarrow\bar y\).
\end{example}

\begin{example}[Two coordinate axes]
Over an algebraically closed field,
\[
X=V(xy),
\qquad
k[X]=k[x,y]/(xy).
\]
The classes \(\bar x\) and \(\bar y\) are nonzero, but their product is zero.  This is the
algebraic shadow of the union of two components.
\end{example}

\begin{example}[Two rational points]
For distinct \(a,b\in k\),
\[
X=\{a,b\}\subseteq\mathbb A_k^1,
\qquad
I(X)=((x-a)(x-b)).
\]
The factors are comaximal, so
\[
k[X]\cong k[x]/(x-a)\times k[x]/(x-b)\cong k\times k.
\]
The two idempotents produced by interpolation select the two points independently.
\end{example}

\begin{example}[A cusp as a monomial subalgebra]
Assume \(k\) is infinite and let \(C=V(y^2-x^3)\).  Substituting \(x=t^2\), \(y=t^3\)
gives
\[
k[C]\cong k[t^2,t^3]\subseteq k[t].
\]
The ring remembers that the available polynomial functions are generated by \(t^2\) and
\(t^3\), not by the parameter \(t\) itself.
\end{example}

\begin{example}[A closed subset as a quotient]
Let \(X=\mathbb A_k^2\) and \(Y=V(y-x^2)\).  Then
\[
k[X]=k[x,y]\twoheadrightarrow k[Y]=k[x,y]/(y-x^2).
\]
Geometric inclusion becomes a surjection of function rings.
\end{example}

\begin{example}[The affine line over a finite field]
For \(k=\mathbb F_q\),
\[
I(\mathbb A^1_k)=(x^q-x),
\qquad
k[\mathbb A^1_k]=\mathbb F_q[x]/(x^q-x).
\]
Since \(x^q-x=\prod_{a\in\mathbb F_q}(x-a)\) has distinct linear factors,
\[
k[\mathbb A^1_k]\cong\prod_{a\in\mathbb F_q}\mathbb F_q.
\]
Thus every set-theoretic function \(\mathbb F_q\to\mathbb F_q\) is represented by a
polynomial.
\end{example}
