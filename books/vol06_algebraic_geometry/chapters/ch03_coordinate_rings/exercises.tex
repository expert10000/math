% Canonical exercises for VI/03.

\begin{exercise}\label{ex:vi03-kernel}
Prove directly that the kernel of restriction
\[
k[x_1,\dots,x_n]\to\operatorname{Fun}(X,k)
\]
is \(I(X)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-same-function}
Show that \(f,g\in k[x_1,\dots,x_n]\) define the same polynomial function on \(X\) if and
only if \(f-g\in I(X)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-affine-space}
Assume \(k\) is infinite.  Prove \(k[\mathbb A_k^n]=k[x_1,\dots,x_n]\).
\end{exercise}

\begin{exercise}\label{ex:vi03-point}
For \(a=(a_1,\dots,a_n)\), compute \(I(\{a\})\) and \(k[\{a\}]\).
\end{exercise}

\begin{exercise}\label{ex:vi03-line}
Assume \(k\) is infinite.  Let \(L=V(ax+by-c)\subseteq\mathbb A_k^2\) with \(b\neq0\).  Construct an explicit
isomorphism \(k[L]\cong k[t]\).
\end{exercise}

\begin{exercise}\label{ex:vi03-parabola}
Assume \(k\) is infinite.  Construct mutually inverse homomorphisms
\[
k[x,y]/(y-x^2)\cong k[t].
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-hyperbola}
Assume \(k\) is infinite.  Show
\[
k[V(xy-1)]\cong k[t,t^{-1}].
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-generators-relations}
Prove \(k[X]=k[\bar x_1,\dots,\bar x_n]\), and show that the polynomial relations among the
\(\bar x_i\) are exactly the elements of \(I(X)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-reduced}
Prove that every classical affine coordinate ring is reduced.
\end{exercise}

\begin{exercise}\label{ex:vi03-evaluation}
For \(a\in X\), prove that evaluation \(\operatorname{ev}_a:k[X]\to k\) is a surjective
\(k\)-algebra homomorphism and that its kernel is maximal.
\end{exercise}

\begin{exercise}\label{ex:vi03-closed-subset}
Let \(Y\subseteq X\subseteq\mathbb A_k^n\) be algebraic.  Prove
\[
k[Y]\cong k[X]/(I(Y)/I(X)).
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-two-points}
For distinct \(a,b\in k\), prove
\[
k[\{a,b\}]\cong k\times k.
\]
Find the two idempotents corresponding to the characteristic functions of \(a\) and \(b\).
\end{exercise}

\begin{exercise}\label{ex:vi03-finite-points}
Let \(X=\{a_1,\dots,a_r\}\subseteq\mathbb A_k^1\) be distinct.  Prove
\[
k[X]\cong k^r.
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-zero-divisors}
Assume \(k\) is algebraically closed and \(X=V(xy)\).  Show that \(k[X]\) contains nonzero
zero divisors and identify a natural pair.
\end{exercise}

\begin{exercise}\label{ex:vi03-nilpotent-warning}
Compare \(A=k[x]/(x)\) and \(B=k[x]/(x^2)\).  Show that \(B\) has a nonzero nilpotent and
explain why \(B\) cannot be the classical coordinate ring of the one-point set \(V(x)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-finite-field-line}
Prove
\[
I(\mathbb A^1_{\mathbb F_q})=(x^q-x)
\]
and deduce \(\mathbb F_q[\mathbb A^1]\cong\mathbb F_q[x]/(x^q-x)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-product-finite}
Let \(X\) and \(Y\) be finite rational-point sets with \(|X|=r\), \(|Y|=s\).  Using only the
fact that every function on a finite rational-point set is polynomial, prove
\[
k[X\times Y]\cong k^{rs}.
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-circle-unit}
Let \(k=\mathbb C\) and \(X=V(x^2+y^2-1)\).  In \(k[X]\), put
\(u=\bar x+i\bar y\).  Show that \(u\) is a unit and compute \(u^{-1}\).
\end{exercise}
