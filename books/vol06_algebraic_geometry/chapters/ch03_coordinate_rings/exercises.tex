% Canonical exercises for VI/03.

\begin{exercise}\label{ex:vi03-kernel}
Let \(X\subseteq\mathbb A_k^n\).  Prove that the kernel of the restriction map
\[
k[x_1,\dots,x_n]\to\operatorname{Fun}(X,k)
\]
is \(I(X)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-same-function}
Prove that \(f,g\in k[x_1,\dots,x_n]\) define the same polynomial function on \(X\) if
and only if
\[
f-g\in I(X).
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-point}
Compute the coordinate ring of the point
\[
a=(a_1,\dots,a_n)\in\mathbb A_k^n.
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-line}
Let
\[
L=V(ax+by-c)\subseteq\mathbb A_k^2,
\]
with \(b\neq0\).  Show that
\[
k[L]\cong k[t].
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-parabola}
Prove explicitly that
\[
k[x,y]/(y-x^2)\cong k[t].
\]
Write down mutually inverse homomorphisms.
\end{exercise}

\begin{exercise}\label{ex:vi03-hyperbola}
Prove
\[
k[x,y]/(xy-1)\cong k[t,t^{-1}].
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-reduced}
Prove that the coordinate ring of a classical algebraic set is reduced.
\end{exercise}

\begin{exercise}\label{ex:vi03-generators}
Show that
\[
k[X]=k[\bar x_1,\dots,\bar x_n].
\]
Prove that a polynomial relation among the \(\bar x_i\) is exactly an element of \(I(X)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-evaluation}
For \(a\in X\), prove that
\[
\operatorname{ev}_a:k[X]\to k
\]
is a surjective \(k\)-algebra homomorphism and that its kernel is maximal.
\end{exercise}

\begin{exercise}\label{ex:vi03-closed-subset}
Let \(Y\subseteq X\subseteq\mathbb A_k^n\) be algebraic.  Prove
\[
k[Y]\cong k[X]/(I(Y)/I(X)).
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-two-points}
Let \(a\neq b\) in \(k\).  Prove
\[
k[\{a,b\}]
\cong
k\times k.
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-finite-points}
Let
\[
X=\{a_1,\dots,a_r\}\subseteq\mathbb A^1_k
\]
with distinct points.  Prove
\[
k[X]\cong k^r.
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-zero-divisors}
Assume \(k\) is algebraically closed and let
\[
X=V(xy).
\]
Show directly that \(k[X]\) has zero divisors.
What geometric feature of \(X\) do they reflect?
\end{exercise}

\begin{exercise}\label{ex:vi03-nilpotent-warning}
Compare
\[
A=k[x]/(x)
\qquad\text{and}\qquad
B=k[x]/(x^2).
\]
Show that \(B\) has a nonzero nilpotent and explain why \(B\) cannot be the classical
coordinate ring of the one-point algebraic set \(V(x)\).
\end{exercise}

\begin{exercise}\label{ex:vi03-finite-field}
Prove
\[
I(\mathbb A^1_{\mathbb F_q})=(x^q-x).
\]
Deduce that
\[
\mathbb F_q[\mathbb A^1_{\mathbb F_q}]
\cong
\mathbb F_q[x]/(x^q-x).
\]
\end{exercise}

\begin{exercise}\label{ex:vi03-challenge-circle}
Assume \(k\) has characteristic different from \(2\), and let
\[
X=V(x^2+y^2-1).
\]
Show that the classes \(\bar x,\bar y\in k[X]\) satisfy
\[
\bar x^2+\bar y^2=1.
\]
If \(k=\mathbb C\), define
\[
u=\bar x+i\bar y.
\]
Show that \(u\) is invertible and compute \(u^{-1}\).
\end{exercise}
