% Hints for VI/02. Not included in the default student build.

\chapter*{Hints for VI/02}
\addcontentsline{toc}{chapter}{Hints for VI/02}

\paragraph{Exercise \ref{ex:vi02-topology-axioms}.}
Use \(V((0))\), \(V((1))\), sums of ideals, and products of two ideals.

\paragraph{Exercise \ref{ex:vi02-relative-closed}.}
A closed subset of a subspace has the form \(X\cap Z\), with \(Z\) closed in the ambient
space.

\paragraph{Exercise \ref{ex:vi02-closure}.}
Show first that \(Y\subseteq V(I(Y))\), then prove it is contained in every closed set that
contains \(Y\).

\paragraph{Exercise \ref{ex:vi02-principal-intersection}.}
A product in a field is nonzero exactly when both factors are nonzero.

\paragraph{Exercise \ref{ex:vi02-principal-basis}.}
If \(x\) is not in a closed set \(V_X(S)\), at least one polynomial in \(S\) does not vanish
at \(x\).

\paragraph{Exercise \ref{ex:vi02-points-closed}.}
The displayed equations determine all coordinates independently.

\paragraph{Exercise \ref{ex:vi02-a1-cofinite}.}
A proper closed set is contained in the zero set of one nonzero polynomial.

\paragraph{Exercise \ref{ex:vi02-a1-not-hausdorff}.}
Two nonempty cofinite subsets of an infinite set must intersect.

\paragraph{Exercise \ref{ex:vi02-infinite-dense}.}
A proper closed subset is finite.

\paragraph{Exercise \ref{ex:vi02-finite-field-discrete}.}
Every singleton is closed and every subset of a finite set is a finite union of singletons.

\paragraph{Exercise \ref{ex:vi02-relative-parabola}.}
On \(X\), the equation \(x=a\) forces \(y=a^2\).

\paragraph{Exercise \ref{ex:vi02-euclidean-coarser}.}
Polynomial functions are Euclidean continuous.  In \(\mathbb A^1_{\mathbb R}\), use an
infinite proper Euclidean closed set.

\paragraph{Exercise \ref{ex:vi02-density-criterion}.}
Use
\[
\overline{Y}=V(I(Y)).
\]

\paragraph{Exercise \ref{ex:vi02-grid-dense}.}
View a polynomial \(f(x,y)\) first as a polynomial in \(y\) whose coefficients are
polynomials in \(x\).

\paragraph{Exercise \ref{ex:vi02-union-principal}.}
Take complements and apply De Morgan's law.

\paragraph{Exercise \ref{ex:vi02-challenge-graph-density}.}
One inclusion is immediate.  For the other, divide any polynomial vanishing on the graph by
\(y-p(x)\), treating it as a monic polynomial in \(y\).
