% Canonical exercises for VI/02.

\begin{exercise}\label{ex:vi02-topology-axioms}
Starting only from the identities for algebraic sets proved in VI/01, verify all three
closed-set axioms for the Zariski topology on \(\mathbb A^n_k\).
\end{exercise}

\begin{exercise}\label{ex:vi02-relative-closed}
Let \(X=V(I)\subseteq\mathbb A^n_k\).  Prove that every closed subset of \(X\) can be
written
\[
V_X(J)
\]
for some ideal \(J\subseteq k[x_1,\dots,x_n]\).
\end{exercise}

\begin{exercise}\label{ex:vi02-closure}
For \(Y\subseteq\mathbb A^n_k\), prove directly that
\[
V(I(Y))
\]
is the Zariski closure of \(Y\).
\end{exercise}

\begin{exercise}\label{ex:vi02-principal-intersection}
Prove
\[
D_X(f)\cap D_X(g)=D_X(fg).
\]
Generalize to finitely many polynomials.
\end{exercise}

\begin{exercise}\label{ex:vi02-principal-basis}
Prove that the principal opens \(D_X(f)\) form a basis for the Zariski topology on an
affine algebraic set \(X\).
\end{exercise}

\begin{exercise}\label{ex:vi02-points-closed}
For \(a=(a_1,\dots,a_n)\in\mathbb A^n_k\), prove
\[
\{a\}=V(x_1-a_1,\dots,x_n-a_n).
\]
Deduce that every finite subset is closed.
\end{exercise}

\begin{exercise}\label{ex:vi02-a1-cofinite}
Assume \(k\) is infinite.  Prove that every proper closed subset of
\(\mathbb A^1_k\) is finite.
\end{exercise}

\begin{exercise}\label{ex:vi02-a1-not-hausdorff}
Assume \(k\) is infinite.  Use the cofinite description to prove that
\(\mathbb A^1_k\) is not Hausdorff.
\end{exercise}

\begin{exercise}\label{ex:vi02-infinite-dense}
Assume \(k\) is infinite.  Prove that every infinite subset of \(\mathbb A^1_k\) is Zariski
dense.
\end{exercise}

\begin{exercise}\label{ex:vi02-finite-field-discrete}
Let \(k\) be finite.  Prove that the Zariski topology on the set \(\mathbb A^n_k=k^n\) is
discrete.
\end{exercise}

\begin{exercise}\label{ex:vi02-relative-parabola}
Let
\[
X=V(y-x^2)\subseteq\mathbb A^2_k.
\]
Describe explicitly the subsets
\[
V_X(x),\qquad D_X(x),\qquad V_X(x-1),\qquad D_X(x-1).
\]
\end{exercise}

\begin{exercise}\label{ex:vi02-euclidean-coarser}
For \(k=\mathbb R\) or \(\mathbb C\), prove that every Zariski closed subset of \(k^n\)
is Euclidean closed.  Give a Euclidean closed set that is not Zariski closed.
\end{exercise}

\begin{exercise}\label{ex:vi02-density-criterion}
Assume \(k\) is infinite.  Prove that
\[
Y\subseteq\mathbb A^n_k
\]
is Zariski dense if and only if the only polynomial vanishing on every point of \(Y\) is the
zero polynomial.
\end{exercise}

\begin{exercise}\label{ex:vi02-grid-dense}
Let \(A,B\subseteq k\) be infinite subsets of an infinite field \(k\).  Prove that
\[
A\times B
\]
is Zariski dense in \(\mathbb A^2_k\).
\end{exercise}

\begin{exercise}\label{ex:vi02-union-principal}
Show that
\[
D_X(f)\cup D_X(g)
=
X\setminus V_X(f,g).
\]
Explain why a finite union of principal opens need not itself be principal from this formula
alone.
\end{exercise}

\begin{exercise}\label{ex:vi02-challenge-graph-density}
Let \(k\) be infinite and \(p\in k[x]\).  Let
\[
\Gamma_p=\set{(t,p(t))\mid t\in k}\subseteq\mathbb A^2_k.
\]
Prove
\[
\overline{\Gamma_p}^{\,\mathrm{Zar}}
=
V(y-p(x)).
\]
\end{exercise}
