% VI/02 pedagogical exercises.

\subsection*{Exercise VI.02.1 --- Verify the topology axioms}
\label{ex:vi02-1}
Starting only from the identities for zero loci proved in VI/01, verify the three closed-set
axioms for the Zariski topology on \(\mathbb A_k^n\).

\subsection*{Exercise VI.02.2 --- Relative closed sets}
\label{ex:vi02-2}
Let \(X=V(I)\subseteq\mathbb A_k^n\).  Prove that every closed subset of \(X\) is
\(V_X(J)\) for some ideal \(J\subseteq k[x_1,\dots,x_n]\).

\subsection*{Exercise VI.02.3 --- Closure from equations}
\label{ex:vi02-3}
For \(Y\subseteq\mathbb A_k^n\), prove directly that \(V(I(Y))\) is the smallest Zariski
closed set containing \(Y\).

\subsection*{Exercise VI.02.4 --- Principal-open intersection calculus}
\label{ex:vi02-4}
Prove
\[
D_X(f)\cap D_X(g)=D_X(fg),
\]
and generalize to a finite family \(f_1,\dots,f_r\).

\subsection*{Exercise VI.02.5 --- Powers do not change principal opens}
\label{ex:vi02-5}
Prove \(D_X(f)=D_X(f^m)\) for every \(m\ge1\).  Compare with
\(V_X(f)=V_X(f^m)\).

\subsection*{Exercise VI.02.6 --- Closed points}
\label{ex:vi02-6}
For \(a=(a_1,\dots,a_n)\), prove
\[
\set{a}=V(x_1-a_1,\dots,x_n-a_n),
\]
and deduce that every finite subset of affine space is closed.

\subsection*{Exercise VI.02.7 --- The affine line is cofinite}
\label{ex:vi02-7}
Assume \(k\) is infinite.  Prove that every proper closed subset of \(\mathbb A_k^1\) is
finite and every finite subset is closed.

\subsection*{Exercise VI.02.8 --- Why the affine line is not Hausdorff}
\label{ex:vi02-8}
Assume \(k\) is infinite.  Show that any two nonempty open subsets of \(\mathbb A_k^1\)
intersect.  Deduce that distinct points cannot be separated by disjoint neighborhoods.

\subsection*{Exercise VI.02.9 --- Infinite subsets are dense}
\label{ex:vi02-9}
Assume \(k\) is infinite.  Prove that every infinite subset of \(\mathbb A_k^1\) is Zariski
dense.

\subsection*{Exercise VI.02.10 --- Finite fields are discrete}
\label{ex:vi02-10}
Let \(k\) be finite.  Prove that every subset of \(\mathbb A_k^n\) is both open and closed.

\subsection*{Exercise VI.02.11 --- Relative topology on a parabola}
\label{ex:vi02-11}
Let \(X=V(y-x^2)\subseteq\mathbb A_k^2\).  Describe
\[
V_X(x),\quad D_X(x),\quad V_X(x-1),\quad D_X(x-1).
\]

\subsection*{Exercise VI.02.12 --- Euclidean closed versus Zariski closed}
\label{ex:vi02-12}
For \(k=\mathbb R\) or \(\mathbb C\), prove every Zariski closed subset of \(k^n\) is
Euclidean closed.  Give an infinite Euclidean closed subset of \(\mathbb A_k^1\) that is not
Zariski closed.

\subsection*{Exercise VI.02.13 --- Density criterion}
\label{ex:vi02-13}
Assume \(k\) is infinite.  Prove that \(Y\subseteq\mathbb A_k^n\) is dense if and only if
no nonzero polynomial vanishes at every point of \(Y\).

\subsection*{Exercise VI.02.14 --- A dense grid}
\label{ex:vi02-14}
Let \(A_1,\dots,A_n\subseteq k\) be infinite subsets of an infinite field.  Prove
\[
A_1\times\cdots\times A_n
\]
is Zariski dense in \(\mathbb A_k^n\).

\subsection*{Exercise VI.02.15 --- Union of two principal opens}
\label{ex:vi02-15}
Show
\[
D_X(f)\cup D_X(g)=X\setminus V_X(f,g).
\]
Explain why this identity alone does not say that the union is principal.

\subsection*{Exercise VI.02.16 --- Product-open rectangles are Zariski open}
\label{ex:vi02-16}
Assume \(k\) is infinite.  Let \(U,V\subseteq\mathbb A_k^1\) be nonempty Zariski open
sets.  Prove that \(U\times V\) is Zariski open in \(\mathbb A_k^2\).

\subsection*{Exercise VI.02.17 --- A rank locus is closed}
\label{ex:vi02-17}
For \(m\times n\) matrices, prove that the locus \(\rank M\le r\) is Zariski closed by
writing explicit equations in minors.

\subsection*{Exercise VI.02.18 --- Dense torus}
\label{ex:vi02-18}
Assume \(k\) is infinite.  Prove that
\[
(k^\times)^n=D(x_1x_2\cdots x_n)
\]
is a dense open subset of \(\mathbb A_k^n\).
