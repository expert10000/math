% Canonical worked examples for VI/02.

\begin{example}[The coordinate axes]
Let
\[
X=V(xy)\subseteq\mathbb A^2_k.
\]
Then
\[
X=V(x)\cup V(y)
\]
is closed.  Inside \(X\), each axis is also closed:
\[
V_X(x)=V(x)\cap X=V(x),
\qquad
V_X(y)=V(y).
\]
Their complements in \(X\) are open.
\end{example}

\begin{example}[A principal open in the affine plane]
In \(\mathbb A^2_k\),
\[
D_{\mathbb A^2}(x)
=
\set{(a,b)\in k^2\mid a\neq0}.
\]
It is the complement of the \(y\)-axis \(V(x)\).
Similarly,
\[
D_{\mathbb A^2}(xy)
=
D_{\mathbb A^2}(x)\cap D_{\mathbb A^2}(y)
\]
is the set where both coordinates are nonzero.
\end{example}

\begin{example}[Closure of a finite subset]
Let
\[
F=\{a_1,\dots,a_r\}\subseteq\mathbb A^1_k.
\]
Then \(F\) is already closed:
\[
F
=
V\!\left(\prod_{j=1}^r(x-a_j)\right).
\]
Therefore
\[
\overline{F}^{\,\mathrm{Zar}}=F.
\]
\end{example}

\begin{example}[Closure of an infinite subset of the affine line]
Assume \(k\) is infinite and let \(Y\subseteq k\) be infinite.  A nonzero
\(f\in k[x]\) has only finitely many roots, so no nonzero polynomial vanishes on all of
\(Y\).  Hence
\[
I(Y)=(0),
\]
and
\[
\overline{Y}^{\,\mathrm{Zar}}
=
V((0))
=
\mathbb A^1_k.
\]
\end{example}

\begin{example}[Integers inside the complex line]
As a subset of \(\mathbb C\),
\[
\mathbb Z
\]
is Euclidean closed and discrete, but it is Zariski dense in
\(\mathbb A^1_{\mathbb C}\), because an ordinary complex polynomial that vanishes at every
integer has infinitely many roots and must be zero.
\end{example}

\begin{example}[A relative open subset of a parabola]
Let
\[
X=V(y-x^2)\subseteq\mathbb A^2_k.
\]
Then
\[
D_X(x)
=
X\setminus V_X(x)
\]
is the parabola with the origin removed, since \(x=0\) on \(X\) forces \(y=0\).
\end{example}

\begin{example}[Finite field]
Take \(k=\mathbb F_3\).  Then
\[
\mathbb A^1_k=\{0,1,2\}.
\]
Every singleton is closed, hence every subset is a finite union of closed points.  Therefore
the Zariski topology on the set of \(\mathbb F_3\)-rational points is discrete.
\end{example}

\begin{example}[Euclidean versus Zariski closure]
Inside \(\mathbb R\), the subset
\[
Y=\set{1,2,3,\dots}
\]
has no finite Euclidean accumulation point, so it is Euclidean closed.  Yet \(Y\) is
infinite, hence its Zariski closure is all of
\[
\mathbb A^1_{\mathbb R}.
\]
\end{example}
