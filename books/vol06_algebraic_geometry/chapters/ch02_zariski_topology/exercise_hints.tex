% VI/02 exercise hints.

\subsection*{Hint VI.02.1}
Use \(V(0)=\mathbb A^n\), \(V(1)=\varnothing\), arbitrary intersections as zero loci of
sums of ideals, and finite unions as zero loci of products.

\subsection*{Hint VI.02.2}
A closed subset of a subspace has the form \(X\cap V(J)\).

\subsection*{Hint VI.02.3}
Every polynomial in \(I(Y)\) vanishes on \(Y\); for minimality, compare with an arbitrary
closed set \(V(J)\) containing \(Y\).

\subsection*{Hint VI.02.4}
At a point, \(fg\neq0\) in a field exactly when both \(f\neq0\) and \(g\neq0\).

\subsection*{Hint VI.02.5}
A field has no nonzero nilpotents: \(f(x)^m\neq0\) iff \(f(x)\neq0\).

\subsection*{Hint VI.02.6}
The equations \(x_i-a_i=0\) force every coordinate.

\subsection*{Hint VI.02.7}
A nonzero one-variable polynomial has finitely many roots.  For the converse, multiply the
linear factors \(x-a_i\).

\subsection*{Hint VI.02.8}
Complements of nonempty opens are finite.  The union of two finite complements is finite.

\subsection*{Hint VI.02.9}
A proper closed subset of the affine line is finite.

\subsection*{Hint VI.02.10}
Every singleton is closed, and the ambient set itself is finite.

\subsection*{Hint VI.02.11}
On the parabola, \(x=0\) forces \(y=0\), and \(x=1\) forces \(y=1\).

\subsection*{Hint VI.02.12}
Polynomial maps are Euclidean continuous.  For a counterexample in one variable, use
\(\mathbb Z\) over \(\mathbb R\) or \(\mathbb C\).

\subsection*{Hint VI.02.13}
Use \(\overline Y=V(I(Y))\) and \(I(\mathbb A_k^n)=(0)\) for infinite \(k\).

\subsection*{Hint VI.02.14}
Induct on \(n\).  View a polynomial as a polynomial in the last variable whose coefficients
are polynomials in the first \(n-1\) variables.

\subsection*{Hint VI.02.15}
Take complements and use \(V_X(f)\cap V_X(g)=V_X(f,g)\).

\subsection*{Hint VI.02.16}
Write the finite complements of \(U\) and \(V\) as zero sets of one-variable polynomials
\(f(x)\) and \(g(y)\).

\subsection*{Hint VI.02.17}
A matrix has rank at most \(r\) exactly when every \((r+1)\times(r+1)\) minor vanishes.

\subsection*{Hint VI.02.18}
First identify the set as a principal open, then apply the proposition that every nonzero
principal open is dense over an infinite field.
