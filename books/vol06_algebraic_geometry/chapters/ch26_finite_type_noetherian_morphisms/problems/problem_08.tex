\begin{problem}[Base change of locally finite type]\label{prob:vi26-p08}
Prove that a base change of a locally finite type morphism is locally finite type.
\par\smallskip\noindent\textit{Provenance.} Canonical VI/26 dossier developed from the audited AG5/AG6 finiteness theory streams; no additional numbered legacy problem is claimed.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Consider a Cartesian square
\[
\begin{tikzcd}
X' \arrow[r] \arrow[d,"f'"'] & X \arrow[d,"f"]\\
S' \arrow[r] & S.
\end{tikzcd}
\]
The assertion is local on \(S'\), so choose affine opens \(V=\Spec A\subseteq S\) and
\(V'=\Spec A'\subseteq S'\) with \(V'\) mapping into \(V\).  Cover \(f^{-1}(V)\) by affine opens
\(U_j=\Spec B_j\) with
\[
B_j=A[b_{j1},\ldots,b_{jn_j}].
\]
Their pullbacks cover \(f'^{-1}(V')\), and each is
\[
U_j\times_VV'
=
\Spec(B_j\otimes_AA').
\]
But
\[
B_j\otimes_AA'
=
A'[b_{j1}\otimes1,\ldots,b_{jn_j}\otimes1],
\]
so every pulled-back chart is finite type over \(A'\).  Therefore \(f'\) is locally of finite type.
\end{solution}
\fi
