\begin{example}[Advanced: Hilbert basis mechanism]\label{ex:vi26-a01}
Starting from a Noetherian ring $A$, repeated application of Hilbert's basis theorem shows $A[x_1,\dots,x_n]$ Noetherian; quotients and localizations then remain Noetherian.
\end{example}

\begin{example}[Advanced: Finite affine local-global Noetherianity]\label{ex:vi26-a02}
If $D(f_1),\dots,D(f_r)$ cover $\Spec A$ and every $A_{f_i}$ is Noetherian, then $A$ is Noetherian.  Finite generation of an arbitrary ideal can be checked on this finite cover and descended.
\end{example}

\begin{example}[Advanced: Transitivity of algebra generation]\label{ex:vi26-a03}
If $B=A[b_1,\dots,b_m]$ and $C=B[c_1,\dots,c_n]$, then $C=A[b_1,\dots,b_m,c_1,\dots,c_n]$.  This is the affine core of composition stability.
\end{example}

\begin{example}[Advanced: Base change of generators]\label{ex:vi26-a04}
If $B=A[b_1,\dots,b_n]$, then $B\otimes_AA'$ is generated over $A'$ by the elements $b_i\otimes1$.
\end{example}

\begin{example}[Advanced: Base change of quasi-compactness]\label{ex:vi26-a05}
A quasi-compact morphism remains quasi-compact after base change.  Combined with local finite type stability this proves finite type is stable under base change.
\end{example}

\begin{example}[Advanced: Locally finite type over locally Noetherian]\label{ex:vi26-a06}
On affine charts $A\to B$ with $A$ Noetherian and $B$ finitely generated, Hilbert's basis theorem makes $B$ Noetherian.  Hence the source is locally Noetherian.
\end{example}

\begin{example}[Advanced: Finite type over Noetherian]\label{ex:vi26-a07}
If the base is Noetherian and the morphism finite type, then the source is locally Noetherian and quasi-compact, hence Noetherian.
\end{example}

\begin{example}[Advanced: Fiber finiteness]\label{ex:vi26-a08}
For $f$ locally finite type, $X_s\to\Spec\kappa(s)$ is locally finite type because it is a base change of $f$.  If $f$ is finite type, the fiber is finite type.
\end{example}

\begin{example}[Advanced: Why the infinite disjoint union fails]\label{ex:vi26-a09}
Every point component is an affine finite-type neighborhood, but no finite subcollection covers the source.  The missing hypothesis is exactly quasi-compactness.
\end{example}

\begin{example}[Advanced: Why infinitely many variables fail locally]\label{ex:vi26-a10}
At the generic point of $k[x_1,x_2,\ldots]$, any basic neighborhood inverts one polynomial involving only finitely many variables; infinitely many untouched variables remain algebraically independent.
\end{example}

\begin{example}[Advanced: Finite type does not mean finitely many relations]\label{ex:vi26-a11}
The quotient $A[x_1,\dots,x_n]/I$ is finite type for arbitrary $I$.  Finite generation of $I$ is a finite-presentation issue.
\end{example}

\begin{example}[Advanced: Localization preserves finite generation]\label{ex:vi26-a12}
If $B$ is finite type over $A$, then localizing at an element from $A$ or $B$ yields a finite-type algebra over the corresponding localized base.
\end{example}

\begin{example}[Advanced: Affine-open criterion]\label{ex:vi26-a13}
The AG6-P5 refinement argument converts one target affine cover into a statement over every affine open by simultaneous distinguished shrinking.
\end{example}

\begin{example}[Advanced: Quasi-compact source over an affine target]\label{ex:vi26-a14}
If $X\to\Spec A$ is locally finite type and $X$ is quasi-compact, finitely many finite-type affine source charts cover $X$, so the morphism is finite type.
\end{example}
