\chapter{Finite-Type and Noetherian Morphisms}\label{ch:vi26-finite-type-noetherian}

The preceding chapters built fiber products, base change, and fibers.  We now impose a finiteness condition on
morphisms.  Algebraically, a map of affine schemes
\[
\Spec B\longrightarrow\Spec A
\]
is of finite type when $B$ is generated by finitely many elements as an $A$-algebra.  Geometrically, this says
that the source is described over the base by finitely many coordinates, locally on the target.  The second
finiteness theme is Noetherianity: ascending chains of algebraic conditions eventually stabilize.  These two
notions interact through Hilbert's basis theorem and provide the standard bridge from arbitrary schemes to
manageable algebraic families.

\section{Locally of finite type and finite type}
\begin{definition}[Locally of finite type]\label{def:vi26-lft}
A morphism $f:X\to Y$ is \emph{locally of finite type} if $Y$ has an affine open cover $V_i=\Spec B_i$ such that
each $f^{-1}(V_i)$ has an affine open cover $U_{ij}=\Spec A_{ij}$ with every $A_{ij}$ a finitely generated
$B_i$-algebra.
\end{definition}

\begin{definition}[Finite type]\label{def:vi26-ft}
A morphism $f:X\to Y$ is \emph{of finite type} if it is locally of finite type and quasi-compact.  Equivalently,
over every affine open $V=\Spec B\subseteq Y$, the inverse image $f^{-1}(V)$ admits a \emph{finite} affine open
cover $\Spec A_j$ with each $A_j$ finitely generated over $B$.
\end{definition}

\section{The affine algebra behind finite type}
If $A\to B$ is a ring map, then $B$ is a finitely generated $A$-algebra precisely when there are finitely many
elements $b_1,\dots,b_n\in B$ and a surjection
\[
A[x_1,\dots,x_n]\twoheadrightarrow B,\qquad x_i\mapsto b_i.
\]
Thus every affine finite-type morphism factors as a closed immersion into affine $n$-space over the base followed
by the projection to the base.  No finite-generation hypothesis on the kernel is required here; that stronger
condition belongs to finite presentation, not finite type.

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_02}

\section{Checking locally finite type on every affine open}
The legacy Problems AG6-P5 and AG5-P5 establish an important invariance: the definition may begin with one
affine cover of the target, but once it holds, it holds over \emph{every} affine open $V=\Spec B\subseteq Y$.
The mechanism is simultaneous distinguished shrinking.  Around $y\in V\cap V_i$ choose
\[
W=D(b)\subseteq V,\qquad W=D(c)\subseteq V_i.
\]
Above $W$, the coordinate rings are localizations of finitely generated algebras and hence remain finitely
generated.  Since $B_b\cong B[t]/(bt-1)$ is itself of finite type over $B$, transitivity of finite generation
finishes the argument.

\begin{example}[Affine finite-type morphism]\label{ex:vi26-expand-01}
The projection $\mathbb A^n_A=\operatorname{Spec}A[x_1,\ldots,x_n]\to\operatorname{Spec}A$ is of finite type because its coordinate algebra is generated by finitely many variables.
\end{example}

\section{Locally Noetherian schemes}
\begin{definition}[Locally Noetherian and Noetherian]\label{def:vi26-noetherian-scheme}
A scheme $X$ is \emph{locally Noetherian} if it has an affine open cover by spectra of Noetherian rings.  It is
\emph{Noetherian} if it is locally Noetherian and quasi-compact.
\end{definition}

The recovered legacy Problem AG5-P4 proves that local Noetherianity is independent of the chosen affine cover:
$X$ is locally Noetherian if and only if every affine open $U=\Spec A\subseteq X$ has $A$ Noetherian.  The
proof uses distinguished refinements and the local-global criterion for finite generation of ideals.

\begin{theorem}[Affine-open criterion for locally Noetherian schemes]\label{thm:vi26-affine-noetherian}
A scheme $X$ is locally Noetherian if and only if every affine open $\Spec A\subseteq X$ has $A$ Noetherian.
\end{theorem}

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_04}

\section{Finite type equals local finite type plus quasi-compactness}
The distinction between locally finite type and finite type is global rather than algebraic.  Every point may have
a finite-coordinate neighborhood while infinitely many such neighborhoods are needed globally.  The standard
example is
\[
\coprod_{n\ge1}\Spec k\longrightarrow\Spec k,
\]
which is locally of finite type but not quasi-compact, hence not of finite type.

\begin{theorem}[Finite type criterion]\label{thm:vi26-ft-lft-qc}
A morphism is of finite type if and only if it is locally of finite type and quasi-compact.
\end{theorem}

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_05}

\section{A morphism that is not locally of finite type}
The opposite failure is genuinely algebraic.  The morphism
\[
\Spec k[x_1,x_2,x_3,\dots]\longrightarrow\Spec k
\]
is not locally of finite type.  No neighborhood of the generic point can be described by a finitely generated
$k$-algebra: localizing at one polynomial only involves finitely many variables and leaves infinitely many
independent variables present.

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_06}

\begin{example}[Closed subscheme finite type]\label{ex:vi26-expand-02}
If $B=A[x_1,\ldots,x_n]/I$, then $\operatorname{Spec}B\to\operatorname{Spec}A$ is finite type.  When $I$ is finitely generated it is finitely presented.
\end{example}

\section{Stability under composition}
If $A\to B\to C$ are ring maps with $B$ finitely generated over $A$ and $C$ finitely generated over $B$, then
$C$ is finitely generated over $A$.  After affine localization this proves that locally finite type morphisms are
closed under composition.  Combining this with quasi-compactness gives the same result for finite-type morphisms.

\begin{proposition}[Composition]\label{prop:vi26-composition}
If $X\to Y$ and $Y\to Z$ are locally of finite type, then $X\to Z$ is locally of finite type.  If both are of
finite type, then the composite is of finite type.
\end{proposition}

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_07}

\section{Stability under base change}
Let $f:X\to S$ be locally of finite type and let $S'\to S$ be arbitrary.  Affine-locally the base change replaces
an $A$-algebra $B$ by $B\otimes_AA'$.  If $B=A[b_1,\dots,b_n]$, then
\[
B\otimes_AA'=A'[b_1\otimes1,\dots,b_n\otimes1],
\]
so finite generation survives.  Quasi-compactness also survives base change.

\begin{theorem}[Base-change stability]\label{thm:vi26-base-change}
Locally finite type and finite type morphisms are stable under arbitrary base change.
\end{theorem}

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_08}

\section{Noetherian consequences of finite type}
Hilbert's basis theorem says that a finitely generated algebra over a Noetherian ring is Noetherian.  Therefore
finite-coordinate geometry over a Noetherian base remains Noetherian.

\begin{theorem}[Finite type over a Noetherian base]\label{thm:vi26-noetherian-consequence}
If $Y$ is locally Noetherian and $f:X\to Y$ is locally of finite type, then $X$ is locally Noetherian.  If $Y$ is
Noetherian and $f$ is of finite type, then $X$ is Noetherian.
\end{theorem}

\input{chapters/ch26_finite_type_noetherian_morphisms/figures/figure_09}

% BEGIN VOL06-EXPANSION VI26-example-03
\begin{example}[Noetherian inheritance]\label{ex:vi26-expand-03}
If $A$ is Noetherian and $B$ is a finitely generated $A$-algebra, Hilbert's basis theorem implies $B$ is Noetherian.  Thus affine finite-type schemes over a Noetherian base are Noetherian.
\end{example}
% END VOL06-EXPANSION VI26-example-03
\section{Fibers of a locally finite type morphism}
By VI/25 a fiber is a base change to $\Spec\kappa(s)$.  Base-change stability therefore gives an immediate
consequence: if $f:X\to S$ is locally of finite type, every fiber $X_s$ is locally of finite type over the field
$\kappa(s)$; if $f$ is of finite type, every fiber is a finite-type $\kappa(s)$-scheme.  This is the basic reason
that finite-type morphisms really do behave like algebraic families of finite-dimensional coordinate complexity.

\section{Boundary with adjacent chapters}
VI/23--VI/25 supplied fiber products, base change, and fibers.  VI/26 adds finiteness of morphisms and the
Noetherian consequences.  Integral schemes and function fields remain for VI/27, and normalization for VI/28.
We do not conflate finite type with finite presentation: finite type controls generators of an algebra, whereas
finite presentation also controls generators of the relations.

\section{Chapter synthesis}
Finite type is the condition that a morphism uses finitely many algebraic coordinates locally on its target;
quasi-compactness upgrades that local statement to a finite global description over each affine base chart.
Noetherianity is the companion stabilization condition on ideals.  Together they are robust under the operations
already developed in this volume -- restriction, composition, base change, and passage to fibers -- and they form
the finiteness framework for the scheme theory that follows.

\subsection*{Proofs}

\begin{exercise}[Composition finite type]\label{exr:vi26-expand-06}
Prove: Prove composition of finite-type morphisms is finite type.
\end{exercise}
\begin{hint}
Work directly from finite type, finite presentation, and Noetherian finiteness.
\end{hint}
\begin{solution}
Affinely, a finitely generated algebra over a finitely generated algebra is finitely generated over the base.
\end{solution}

\begin{exercise}[Base change finite type]\label{exr:vi26-expand-07}
Prove: Prove finite type is preserved by base change.
\end{exercise}
\begin{hint}
Work directly from finite type, finite presentation, and Noetherian finiteness.
\end{hint}
\begin{solution}
Tensor a finite generating set with the new base ring.
\end{solution}

\begin{exercise}[Closed subscheme]\label{exr:vi26-expand-08}
Prove: Prove a closed subscheme of a finite-type scheme over a base is finite type.
\end{exercise}
\begin{hint}
Work directly from finite type, finite presentation, and Noetherian finiteness.
\end{hint}
\begin{solution}
Affinely, quotient a finitely generated algebra.
\end{solution}

\begin{exercise}[Noetherian finite type]\label{exr:vi26-expand-09}
Prove: Prove a finite-type algebra over a Noetherian ring is Noetherian.
\end{exercise}
\begin{hint}
Work directly from finite type, finite presentation, and Noetherian finiteness.
\end{hint}
\begin{solution}
Apply Hilbert's basis theorem and quotient stability.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Algebraic families]\label{exr:vi26-expand-13}
Why is finite type the standard algebraicity condition for morphisms?
\end{exercise}
\begin{hint}
Translate the question into finite type, finite presentation, and Noetherian finiteness.
\end{hint}
\begin{solution}
It ensures only finitely many algebraic generators are needed locally.
\end{solution}

\begin{exercise}[Algorithmic finiteness]\label{exr:vi26-expand-14}
What does Noetherianity provide?
\end{exercise}
\begin{hint}
Translate the question into finite type, finite presentation, and Noetherian finiteness.
\end{hint}
\begin{solution}
It forces ideals and submodules to admit finite generating data.
\end{solution}
