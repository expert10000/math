% VI25_DETAILED_EXERCISE_SOLUTION_V1_01
\paragraph{Exercise VI.25.1.}
\begin{solution}
Let \(f:X\to S\) and let \(s\in S\).  The point \(s\) determines the canonical morphism
\[
\iota_s:\Spec\kappa(s)\longrightarrow S.
\]
The scheme-theoretic fiber is defined by the Cartesian square
\[
\begin{tikzcd}
X_s \arrow[r,"p"] \arrow[d,"f_s"'] & X \arrow[d,"f"]\\
\Spec\kappa(s) \arrow[r,"\iota_s"'] & S .
\end{tikzcd}
\]
Thus
\[
X_s=X\times_S\Spec\kappa(s).
\]
The word ``Cartesian'' means that for every scheme \(T\), a morphism \(T\to X_s\) is equivalent to a pair
\[
a:T\to X,\qquad b:T\to\Spec\kappa(s)
\]
whose composites to \(S\) agree.  This universal property is what gives the fiber its scheme structure; it is stronger than merely taking the set of points of \(X\) lying over \(s\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_02
\paragraph{Exercise VI.25.2.}
\begin{solution}
Let \(A\to B\) define \(\Spec B\to\Spec A\), and let \(\mathfrak p\in\Spec A\).  Since
\[
\Spec\kappa(\mathfrak p)\longrightarrow\Spec A
\]
is affine, the affine fiber-product theorem gives
\[
(\Spec B)_{\mathfrak p}
\cong
\Spec\!\bigl(B\otimes_A\kappa(\mathfrak p)\bigr).
\]
One can make the ring more explicit.  Put \(S=A\setminus\mathfrak p\).  Since
\[
\kappa(\mathfrak p)\cong S^{-1}(A/\mathfrak p),
\]
tensoring first by \(A/\mathfrak p\) and then localizing gives
\[
B\otimes_A\kappa(\mathfrak p)
\cong
S^{-1}(B/\mathfrak p B)
\cong
S^{-1}B/\mathfrak p\,S^{-1}B.
\]
Thus forming the fiber means first imposing the equations coming from \(\mathfrak p\), and then making every element of \(A\setminus\mathfrak p\) invertible.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_03
\paragraph{Exercise VI.25.3.}
\begin{solution}
Consider
\[
X=\Spec k[t,u]/(tu-1)\longrightarrow\Spec k[t].
\]
The fiber over \(t=0\) is obtained by tensoring with \(k[t]/(t)\cong k\):
\[
k[t,u]/(tu-1)\otimes_{k[t]}k
\cong
k[u]/(0\cdot u-1)
=
k[u]/(1)
=
0.
\]
The zero ring has empty spectrum, so
\[
X_0=\varnothing.
\]
Geometrically, the equation \(tu=1\) has no solution when \(t=0\).  The scheme calculation records this exactly by producing the zero coordinate ring.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_04
\paragraph{Exercise VI.25.4.}
\begin{solution}
For \(a\in k^\times\), the fiber is
\[
X_a
=
\Spec k[u]/(au-1).
\]
Because \(a\) is invertible, the relation forces \(u=a^{-1}\).  Evaluation at \(u=a^{-1}\) therefore induces an isomorphism
\[
k[u]/(au-1)\xrightarrow{\sim}k.
\]
Hence
\[
X_a\cong\Spec k,
\]
one reduced \(k\)-rational point.  In the same family, Exercise VI.25.3 gives an empty fiber at \(a=0\), so even the cardinality of the fiber may jump.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_05
\paragraph{Exercise VI.25.5.}
\begin{solution}
For
\[
X=\Spec k[t,u]/(u^2-t)\longrightarrow\Spec k[t],
\]
the fiber over \(t=0\) has coordinate ring
\[
k[t,u]/(u^2-t)\otimes_{k[t]}k[t]/(t)
\cong
k[u]/(u^2).
\]
Thus
\[
X_0=\Spec k[u]/(u^2).
\]
There is only one prime ideal, namely \((u)\), so the underlying topological space is one point.  However, the class of \(u\) is nonzero and satisfies \(u^2=0\).  Consequently the fiber is nonreduced: it is a doubled point, not the ordinary reduced point \(\Spec k\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_06
\paragraph{Exercise VI.25.6.}
\begin{solution}
For the DG4-P12 family
\[
X=V\bigl(y(y-tx-1)\bigr)\subset\mathbb A^3_k,
\]
the fiber over \(t=c\) is
\[
X_c=V\bigl(y(y-cx-1)\bigr)\subset\mathbb A^2_{x,y}.
\]
Its two components are
\[
L_1: y=0,\qquad L_2:y=cx+1.
\]
If \(c\ne0\), their intersection is obtained by substituting \(y=0\) into the second equation:
\[
0=cx+1,
\qquad
x=-c^{-1}.
\]
Therefore
\[
L_1\cap L_2=\{(-c^{-1},0)\}.
\]
Scheme-theoretically, the intersection ring is
\[
k[x,y]/(y,y-cx-1)\cong k,
\]
so the intersection consists of one reduced \(k\)-point.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_07
\paragraph{Exercise VI.25.7.}
\begin{solution}
At \(c=0\), the DG4-P12 fiber is
\[
X_0=V\bigl(y(y-1)\bigr).
\]
Hence its two components are
\[
L_1: y=0,\qquad L_2:y=1.
\]
Both are affine lines parallel to the \(x\)-axis.  Their ideals \((y)\) and \((y-1)\) are comaximal, because
\[
1=y-(y-1).
\]
The Chinese remainder theorem therefore gives
\[
k[x,y]/(y(y-1))
\cong
k[x,y]/(y)\times k[x,y]/(y-1)
\cong
k[x]\times k[x].
\]
Thus the special fiber is the disjoint union of two reduced affine lines.  This contrasts with the nonzero fibers, where the same two components meet in one point.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_08
\paragraph{Exercise VI.25.8.}
\begin{solution}
The generic point of \(\Spec\mathbb Z\) is \((0)\), whose residue field is \(\mathbb Q\).  Therefore the generic fiber of
\[
\Spec\mathbb Z[x]\longrightarrow\Spec\mathbb Z
\]
is
\[
\Spec\bigl(\mathbb Z[x]\otimes_{\mathbb Z}\mathbb Q\bigr).
\]
Since tensoring with \(\mathbb Q\) simply inverts every nonzero integer,
\[
\mathbb Z[x]\otimes_{\mathbb Z}\mathbb Q\cong\mathbb Q[x].
\]
Hence the generic fiber is
\[
\mathbb A^1_{\mathbb Q}=\Spec\mathbb Q[x].
\]
This is the characteristic-zero member of the arithmetic family.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_09
\paragraph{Exercise VI.25.9.}
\begin{solution}
At the closed point \((p)\in\Spec\mathbb Z\), the residue field is \(\mathbb F_p\).  Hence
\[
\bigl(\Spec\mathbb Z[x]\bigr)_p
=
\Spec\bigl(\mathbb Z[x]\otimes_{\mathbb Z}\mathbb F_p\bigr).
\]
Reduction of the coefficients modulo \(p\) gives
\[
\mathbb Z[x]\otimes_{\mathbb Z}\mathbb F_p\cong\mathbb F_p[x].
\]
Therefore
\[
\bigl(\Spec\mathbb Z[x]\bigr)_p\cong\mathbb A^1_{\mathbb F_p}.
\]
The same integral model thus contains one affine line in characteristic \(0\) and one affine line in every prime characteristic.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_10
\paragraph{Exercise VI.25.10.}
\begin{solution}
Write
\[
\mathbb Z[i]\cong\mathbb Z[T]/(T^2+1).
\]
The fiber at \(2\) is
\[
\Spec\!\left(\mathbb Z[T]/(T^2+1)\otimes_{\mathbb Z}\mathbb F_2\right)
\cong
\Spec\mathbb F_2[T]/(T^2+1).
\]
In characteristic \(2\),
\[
T^2+1=T^2+2T+1=(T+1)^2.
\]
Putting \(\varepsilon=T+1\) gives
\[
\mathbb F_2[T]/(T^2+1)\cong\mathbb F_2[\varepsilon]/(\varepsilon^2).
\]
Thus the \(2\)-fiber is one topological point carrying a nonzero square-zero element.  It is a nonreduced doubled point, the scheme-theoretic signature of ramification at \(2\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_11
\paragraph{Exercise VI.25.11.}
\begin{solution}
For \(p\equiv3\pmod4\), the polynomial \(T^2+1\) has no root in \(\mathbb F_p\): if \(a^2=-1\), then \(-1\) would be a quadratic residue, contrary to the standard criterion
\[
\left(\frac{-1}{p}\right)=(-1)^{(p-1)/2}=-1.
\]
Hence \(T^2+1\) is irreducible over \(\mathbb F_p\), and
\[
\mathbb F_p[T]/(T^2+1)\cong\mathbb F_{p^2}.
\]
Therefore the \(p\)-fiber of \(\Spec\mathbb Z[i]\) is
\[
\Spec\mathbb F_{p^2}.
\]
It has one scheme point, but its residue field has degree \(2\) over \(\mathbb F_p\).  After extending to an algebraic closure, that degree-two point splits into two geometric points.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_12
\paragraph{Exercise VI.25.12.}
\begin{solution}
Consider
\[
\Spec\mathbb Q\longrightarrow\Spec\mathbb Z.
\]
For a closed point \((p)\), the fiber ring is
\[
\mathbb Q\otimes_{\mathbb Z}\mathbb F_p.
\]
Since \(p\) is invertible in \(\mathbb Q\) but maps to \(0\) in \(\mathbb F_p\), the tensor product must vanish.  Explicitly,
\[
1\otimes1
=
p^{-1}p\otimes1
=
p^{-1}\otimes p
=
p^{-1}\otimes0
=
0.
\]
Thus the entire tensor product is the zero ring and the fiber is empty.  In contrast, the generic fiber is
\[
\Spec(\mathbb Q\otimes_{\mathbb Z}\mathbb Q)\cong\Spec\mathbb Q.
\]
So this morphism lives entirely over the generic point of \(\Spec\mathbb Z\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_13
\paragraph{Exercise VI.25.13.}
\begin{solution}
A standard example is the special fiber of
\[
X=\Spec k[t,u]/(u^2-t)\longrightarrow\Spec k[t]
\]
at \(t=0\).  As in Exercise VI.25.5,
\[
X_0=\Spec k[u]/(u^2).
\]
The ring \(k[u]/(u^2)\) has exactly one prime ideal, \((u)\), so its spectrum is irreducible: there is only one nonempty irreducible component.  Nevertheless, the class of \(u\) is a nonzero nilpotent.  Hence \(X_0\) is irreducible but nonreduced.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_14
\paragraph{Exercise VI.25.14.}
\begin{solution}
Take
\[
Y=\Spec(k\times k).
\]
A product of fields is reduced because an element \((a,b)\) is nilpotent only when both \(a\) and \(b\) are zero.  On the other hand,
\[
\Spec(k\times k)\cong\Spec k\amalg\Spec k,
\]
so the space has two disjoint nonempty open-and-closed components and is therefore reducible.  Such a fiber occurs, for example, in
\[
\Spec k[t,u]/(u^2-t^2)\to\Spec k[t]
\]
at any \(t=a\ne0\) when \(\operatorname{char}k\ne2\), because
\[
k[u]/(u^2-a^2)\cong k\times k.
\]
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_15
\paragraph{Exercise VI.25.15.}
\begin{solution}
Let \(f:X\to S\), let \(s\in S\), and choose a geometric point
\[
\bar s:\Spec\Omega\longrightarrow S
\]
whose image is \(s\), where \(\Omega\) is an algebraically closed extension of \(\kappa(s)\).  The geometric fiber is
\[
X_{\bar s}:=X\times_S\Spec\Omega.
\]
Associativity of fiber products gives the equivalent form
\[
X_{\bar s}
\cong
X_s\times_{\Spec\kappa(s)}\Spec\Omega.
\]
Thus one first forms the ordinary fiber over the residue field and then extends that field to an algebraically closed one.  This second step exposes splitting and inseparable phenomena that may be invisible over \(\kappa(s)\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_16
\paragraph{Exercise VI.25.16.}
\begin{solution}
Write one copy of \(\mathbb C\) as
\[
\mathbb C\cong\mathbb R[T]/(T^2+1).
\]
Then
\[
\mathbb C\otimes_{\mathbb R}\mathbb C
\cong
\mathbb C[T]/(T^2+1).
\]
Over \(\mathbb C\),
\[
T^2+1=(T-i)(T+i),
\]
and the two linear factors are coprime.  The Chinese remainder theorem yields
\[
\mathbb C[T]/(T^2+1)
\cong
\mathbb C[T]/(T-i)\times\mathbb C[T]/(T+i)
\cong
\mathbb C\times\mathbb C.
\]
Therefore
\[
\mathbb C\otimes_{\mathbb R}\mathbb C\cong\mathbb C\times\mathbb C.
\]
On spectra, a single complex point regarded over \(\mathbb R\) splits into two complex geometric points.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_17
\paragraph{Exercise VI.25.17.}
\begin{solution}
As an \(\mathbb R\)-scheme, \(\Spec\mathbb C\) has a single point, so it is irreducible.  To test geometric irreducibility, extend scalars to an algebraic closure of \(\mathbb R\), which may be taken to be \(\mathbb C\).  Exercise VI.25.16 gives
\[
(\Spec\mathbb C)_{\mathbb C}
=
\Spec(\mathbb C\otimes_{\mathbb R}\mathbb C)
\cong
\Spec(\mathbb C\times\mathbb C)
\cong
\Spec\mathbb C\amalg\Spec\mathbb C.
\]
This geometric fiber has two irreducible components.  Hence \(\Spec\mathbb C\) is irreducible over \(\mathbb R\) but not geometrically irreducible over \(\mathbb R\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_18
\paragraph{Exercise VI.25.18.}
\begin{solution}
Let \(S'\to S\), let \(s'\in S'\) map to \(s\in S\), and put \(X'=X\times_SS'\).  Then
\[
X'_{s'}
=
(X\times_SS')\times_{S'}\Spec\kappa(s').
\]
Associativity and cancellation of the intermediate pullback give
\[
X'_{s'}\cong X\times_S\Spec\kappa(s').
\]
The morphism \(S'\to S\) induces a residue-field homomorphism
\[
\kappa(s)\longrightarrow\kappa(s'),
\]
so \(\Spec\kappa(s')\to S\) factors through \(\Spec\kappa(s)\to S\).  Reassociating once more,
\[
X\times_S\Spec\kappa(s')
\cong
\bigl(X\times_S\Spec\kappa(s)\bigr)
\times_{\Spec\kappa(s)}
\Spec\kappa(s').
\]
Therefore
\[
\boxed{
X'_{s'}
\cong
X_s\times_{\kappa(s)}\kappa(s')
}.
\]
Taking a fiber after base change is thus the same as extending the residue field of the old fiber.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_19
\paragraph{Exercise VI.25.19.}
\begin{solution}
Let \(D(f)=\Spec A_f\hookrightarrow\Spec A\), and let \(\mathfrak p\in\Spec A\).  The fiber ring is
\[
A_f\otimes_A\kappa(\mathfrak p).
\]
If \(f\notin\mathfrak p\), then the image of \(f\) in the field \(\kappa(\mathfrak p)\) is nonzero and hence already invertible.  Localizing therefore changes nothing:
\[
A_f\otimes_A\kappa(\mathfrak p)\cong\kappa(\mathfrak p).
\]
If \(f\in\mathfrak p\), then \(f\) maps to \(0\) in \(\kappa(\mathfrak p)\), while localization requires it to be invertible.  Hence \(1=0\) in the tensor product and the ring is zero.  Thus
\[
(D(f))_{\mathfrak p}
\cong
\begin{cases}
\Spec\kappa(\mathfrak p),& f\notin\mathfrak p,\\
\varnothing,& f\in\mathfrak p.
\end{cases}
\]
This is exactly the pointwise membership criterion for \(D(f)\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_20
\paragraph{Exercise VI.25.20.}
\begin{solution}
For the closed immersion
\[
Z=\Spec(A/I)\hookrightarrow\Spec A,
\]
the fiber at \(\mathfrak p\) has coordinate ring
\[
(A/I)\otimes_A\kappa(\mathfrak p)
\cong
\kappa(\mathfrak p)/I\kappa(\mathfrak p).
\]
If \(I\subseteq\mathfrak p\), every element of \(I\) maps to zero in \(\kappa(\mathfrak p)\), so
\[
I\kappa(\mathfrak p)=0
\]
and the fiber is \(\Spec\kappa(\mathfrak p)\).  If \(I\nsubseteq\mathfrak p\), choose \(a\in I\setminus\mathfrak p\).  Its image in \(\kappa(\mathfrak p)\) is nonzero and therefore invertible, so \(I\kappa(\mathfrak p)=\kappa(\mathfrak p)\).  The quotient is then zero and the fiber is empty.  Therefore
\[
Z_{\mathfrak p}\ne\varnothing
\quad\Longleftrightarrow\quad
\mathfrak p\in V(I).
\]
\end{solution}

\paragraph{Exercise VI.25.21.}
\begin{solution}
Let
\[
X=\Spec\mathbb Z[x]/(x^2-2)\longrightarrow\Spec\mathbb Z.
\]
The fiber at \((2)\) is obtained by reducing coefficients modulo \(2\):
\[
X_2
\cong
\Spec\!\left(\mathbb F_2[x]/(x^2-2)\right)
=
\Spec\mathbb F_2[x]/(x^2).
\]
The ring has one prime ideal \((x)\), but the class of \(x\) is a nonzero square-zero element.  Hence \(X_2\) is a nonreduced doubled point.  By contrast, the generic fiber
\[
\Spec\mathbb Q[x]/(x^2-2)
\]
is reduced and is a degree-two point over \(\mathbb Q\).  The special fiber therefore records ramification at \(2\).
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_22
\paragraph{Exercise VI.25.22.}
\begin{solution}
Let
\[
k=\mathbb F_p(a^p),\qquad L=\mathbb F_p(a).
\]
Then \(L/k\) is purely inseparable of degree \(p\), and one may write
\[
L\cong k[T]/(T^p-a^p).
\]
After extending scalars from \(k\) to \(L\),
\[
L\otimes_kL
\cong
L[T]/(T^p-a^p).
\]
In characteristic \(p\),
\[
T^p-a^p=(T-a)^p.
\]
Putting \(\varepsilon=T-a\) yields
\[
L\otimes_kL\cong L[\varepsilon]/(\varepsilon^p).
\]
The class of \(\varepsilon\) is nonzero and nilpotent, so the scalar extension is nonreduced even though both copies of \(L\) are fields.  The repeated \(p\)th power is the algebraic signature of inseparability.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_23
\paragraph{Exercise VI.25.23.}
\begin{solution}
Let \(X\to S\) and \(Y\to S\), and let \(s\in S\).  By definition,
\[
(X\times_SY)_s
=
(X\times_SY)\times_S\Spec\kappa(s).
\]
Base change commutes with products, so
\[
(X\times_SY)\times_S\Spec\kappa(s)
\cong
\bigl(X\times_S\Spec\kappa(s)\bigr)
\times_{\Spec\kappa(s)}
\bigl(Y\times_S\Spec\kappa(s)\bigr).
\]
The two factors on the right are \(X_s\) and \(Y_s\).  Hence
\[
\boxed{
(X\times_SY)_s
\cong
X_s\times_{\kappa(s)}Y_s
}.
\]
Equivalently, both sides represent pairs of maps to \(X_s\) and \(Y_s\) over the same residue field.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_24
\paragraph{Exercise VI.25.24.}
\begin{solution}
Let \(K/k\) be a field extension of degree \(2\).  The scheme
\[
\Spec K
\]
has exactly one topological point, whose residue field is \(K\).  It is therefore a single degree-two point over \(k\), not two \(k\)-rational points.  By contrast,
\[
\Spec(k\times k)\cong\Spec k\amalg\Spec k
\]
has two topological points, and each has residue field \(k\).

The distinction can disappear after a suitable field extension.  If \(K/k\) is separable quadratic and \(\Omega\) contains both \(k\)-embeddings of \(K\), then
\[
K\otimes_k\Omega\cong\Omega\times\Omega.
\]
Thus one degree-two point can split into two geometric points.  Ordinary fibers remember the residue-field extension; geometric fibers reveal the points after that field extension has been split.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_25
\paragraph{Exercise VI.25.25.}
\begin{solution}
Geometric fibers detect whether component structure persists after extending the residue field.  For example,
\[
X=\Spec\mathbb C
\]
is a one-point irreducible scheme over \(\mathbb R\).  Its geometric scalar extension is
\[
X_{\mathbb C}
=
\Spec(\mathbb C\otimes_{\mathbb R}\mathbb C)
\cong
\Spec(\mathbb C\times\mathbb C),
\]
which has two irreducible components.  Thus the ordinary fiber is irreducible, while the geometric fiber is reducible.

A different phenomenon occurs for purely inseparable extensions: a reduced ordinary fiber can become nonreduced after extending the field.  Geometric fibers are designed precisely to expose such splitting and inseparability phenomena.
\end{solution}

% VI25_DETAILED_EXERCISE_SOLUTION_V1_26
\paragraph{Exercise VI.25.26.}
\begin{solution}
VI/24 studies \emph{arbitrary base change}: given \(X\to S\) and any morphism \(S'\to S\), it develops the calculus of
\[
X_{S'}=X\times_SS'
\]
and the properties preserved by this operation.  VI/25 specializes that calculus to the canonical maps
\[
\Spec\kappa(s)\to S
\]
and to geometric points \(\Spec\Omega\to S\).  Its subject is therefore not base change in general, but the geometry of the resulting ordinary and geometric fibers: their points, residue fields, nilpotents, components, splitting, and specialization behavior.
\end{solution}
