\begin{example}[Affine line over a field]\label{ex:vi25-r01}
For $\mathbb A^1_k\to\Spec k$, the only fiber is $\mathbb A^1_k$ itself.
\end{example}
\begin{example}[Projection from the affine plane]\label{ex:vi25-r02}
For $\Spec k[t,x]\to\Spec k[t]$, the fiber at $t=a$ is $\Spec k[x]\cong\mathbb A^1_k$.
\end{example}
\begin{example}[Hyperbola projection]\label{ex:vi25-r03}
For $X=\Spec k[t,u]/(tu-1)\to\Spec k[t]$, the fiber at $0$ is empty, while for $a\ne0$ it is $\Spec k$.
\end{example}
\begin{example}[A doubled special fiber]\label{ex:vi25-r04}
For $\Spec k[t,u]/(u^2-t)\to\Spec k[t]$, the fiber at $0$ is $\Spec k[u]/(u^2)$, a doubled point.
\end{example}
\begin{example}[Two reduced points]\label{ex:vi25-r05}
If $\operatorname{char}k\ne2$, the fiber of $u^2-t^2=0$ at $t=a\ne0$ is $\Spec(k\times k)$ after the two distinct linear factors are identified.
\end{example}
\begin{example}[Legacy line degeneration]\label{ex:vi25-r06}
For $y(y-tx-1)=0$, fibers at $t\ne0$ are two intersecting lines and the fiber at $0$ is two parallel lines.
\end{example}
\begin{example}[Affine line over the integers]\label{ex:vi25-r07}
$\Spec\mathbb Z[x]\to\Spec\mathbb Z$ has generic fiber $\mathbb A^1_{\mathbb Q}$ and $p$-fiber $\mathbb A^1_{\mathbb F_p}$.
\end{example}
\begin{example}[A fiber can be empty]\label{ex:vi25-r08}
$\Spec\mathbb Q\to\Spec\mathbb Z$ has empty fiber over every closed point $(p)$ because $\mathbb Q\otimes_{\mathbb Z}\mathbb F_p=0$.
\end{example}
\begin{example}[A point with larger residue field]\label{ex:vi25-r09}
The $\mathbb R$-scheme $\Spec\mathbb C$ is one ordinary fiber over $\Spec\mathbb R$, but its geometric base change to $\mathbb C$ splits into two points.
\end{example}
\begin{example}[Gaussian fiber at two]\label{ex:vi25-r10}
$\mathbb F_2[T]/(T^2+1)=\mathbb F_2[T]/((T+1)^2)$, so the fiber at $2$ of $\Spec\mathbb Z[i]$ is nonreduced.
\end{example}
\begin{example}[Gaussian split prime]\label{ex:vi25-r11}
For $p\equiv1\pmod4$, the $p$-fiber of $\Spec\mathbb Z[i]$ consists of two reduced points.
\end{example}
\begin{example}[Gaussian inert prime]\label{ex:vi25-r12}
For $p\equiv3\pmod4$, the $p$-fiber is $\Spec\mathbb F_{p^2}$: one point with a quadratic residue field.
\end{example}
\begin{example}[Closed immersion fiber]\label{ex:vi25-r13}
For $\Spec(A/I)\hookrightarrow\Spec A$, the fiber over $\mathfrak p$ is empty if $I\not\subseteq\mathfrak p$ and is $\Spec\kappa(\mathfrak p)$ if $I\subseteq\mathfrak p$.
\end{example}
\begin{example}[Open immersion fiber]\label{ex:vi25-r14}
For $D(f)\hookrightarrow\Spec A$, the fiber over $\mathfrak p$ is a point exactly when $f\notin\mathfrak p$, and is empty otherwise.
\end{example}
\begin{example}[Constant family]\label{ex:vi25-r15}
For a $k$-scheme $Y$, the projection $Y\times_kS\to S$ has fiber over $s$ equal to $Y\times_k\kappa(s)$.
\end{example}
\begin{example}[Fiber of an isomorphism]\label{ex:vi25-r16}
An isomorphism $X\to S$ has every fiber equal to the one-point scheme $\Spec\kappa(s)$.
\end{example}
\begin{example}[Fiber of a product family]\label{ex:vi25-r17}
The fiber of $X\times_SY\to S$ at $s$ is canonically $X_s\times_{\kappa(s)}Y_s$.
\end{example}
\begin{example}[Specializing a hypersurface]\label{ex:vi25-r18}
For $F(t,x,y)=0$, the fiber at $t=a$ is the hypersurface $F(a,x,y)=0$ over $\kappa(a)$.
\end{example}
\begin{example}[Generic point of an affine base]\label{ex:vi25-r19}
For a domain $A$ with fraction field $K$, the generic fiber of $\Spec B\to\Spec A$ is $\Spec(B\otimes_AK)$.
\end{example}
\begin{example}[Geometric fiber notation]\label{ex:vi25-r20}
Choosing an algebraic closure $\overline{\kappa(s)}$ gives $X_{\bar s}=X_s\times_{\kappa(s)}\overline{\kappa(s)}$.
\end{example}
