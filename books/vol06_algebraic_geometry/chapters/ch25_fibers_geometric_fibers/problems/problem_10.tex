\begin{problem}[Constructing a geometric fiber]\label{prob:vi25-p10}
    Let $f:X\to S$, $s\in S$, and $\Omega/\kappa(s)$ be algebraically closed. Show
\[
X\times_S\Spec\Omega\cong X_s\times_{\Spec\kappa(s)}\Spec\Omega.
\]
    \end{problem}
    \begin{solution}
    Both schemes are obtained from $X$ by the same composite base change $\Spec\Omega\to\Spec\kappa(s)\to S$.  Associativity of fiber products gives the canonical isomorphism.
    \end{solution}
