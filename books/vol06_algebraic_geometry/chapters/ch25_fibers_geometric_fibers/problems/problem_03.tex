\begin{problem}[Hyperbola projection: empty and one-point fibers]\label{prob:vi25-p03}
    For
\[
X=\Spec k[t,u]/(tu-1)\to\Spec k[t],
\]
compute every closed fiber.
    \end{problem}
    \begin{solution}
    At $t=a$ the fiber ring is $k[u]/(au-1)$.  For $a=0$ this is the zero ring, hence the fiber is empty.  For $a\ne0$, $u=a^{-1}$ and the quotient is $k$, so the fiber is one reduced $k$-point.
    \end{solution}
