\subsection*{VI.25.P7: The fibers of \(\Spec\mathbb Q\to\Spec\mathbb Z\)}
\begin{problem}\label{prob:vi25-p07}
Compute the fiber over \((0)\) and over every \((p)\).
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This example isolates a morphism whose image is only the generic point and shows how that fact appears scheme-theoretically in its fibers.

\paragraph{Useful ideas before starting.}
Use \(\mathbb Q\otimes_{\mathbb Z}\mathbb Q\) at the generic point and \(\mathbb Q\otimes_{\mathbb Z}\mathbb F_p\) at a closed point.

\paragraph{Guided hints.}
\begin{enumerate}
\item At \((0)\), localizing \(\mathbb Z\) gives \(\mathbb Q\).
\item At \((p)\), \(p\) is invertible in \(\mathbb Q\) but zero in \(\mathbb F_p\).
\item Show \(1\otimes1=0\) in the closed-fiber tensor product.
\end{enumerate}

\begin{solution}
At the generic point \((0)\), the residue field is \(\mathbb Q\), so
\[
X_{(0)}
=
\Spec(\mathbb Q\otimes_{\mathbb Z}\mathbb Q)
\cong
\Spec\mathbb Q.
\]
For a closed point \((p)\),
\[
X_p
=
\Spec(\mathbb Q\otimes_{\mathbb Z}\mathbb F_p).
\]
Because \(p\) is invertible in \(\mathbb Q\) and zero in \(\mathbb F_p\),
\[
1\otimes1
=
p^{-1}p\otimes1
=
p^{-1}\otimes p
=
0.
\]
Therefore
\[
\mathbb Q\otimes_{\mathbb Z}\mathbb F_p=0,
\qquad
X_p=\varnothing.
\]
The fiber calculation exactly reflects the image of \(\Spec\mathbb Q\to\Spec\mathbb Z\): the unique source point maps to the generic point, and no source point lies over any closed prime.
\end{solution}

\paragraph{What happened mathematically.}
Empty fibers detect base points missed by the morphism.

\paragraph{Extensions.}
The same argument applies to any localization \(\Spec S^{-1}A	o\Spec A\): the fiber is empty precisely when the prime meets \(S\).
