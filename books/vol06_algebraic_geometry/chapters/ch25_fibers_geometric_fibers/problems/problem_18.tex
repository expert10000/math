\begin{problem}[Fiber of a product family]\label{prob:vi25-p18}
    For $X\to S$ and $Y\to S$, prove
\[
(X\times_SY)_s\cong X_s\times_{\kappa(s)}Y_s.
\]
    \end{problem}
    \begin{solution}
    Both sides are obtained from $X\times_SY$ by base change along $\Spec\kappa(s)\to S$.  Products commute with base change, equivalently by associating the three relevant fiber products.
    \end{solution}
