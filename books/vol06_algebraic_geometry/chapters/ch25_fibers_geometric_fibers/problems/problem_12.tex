\begin{problem}[Fibers after base change]\label{prob:vi25-p12}
    Let $S'\to S$, let $s'\mapsto s$, and set $X'=X\times_SS'$. Prove
\[
X'_{s'}\cong X_s\times_{\kappa(s)}\kappa(s').
\]
    \end{problem}
    \begin{solution}
    Expand the left side and reassociate:
\[
(X\times_SS')\times_{S'}\Spec\kappa(s')
\cong X\times_S\Spec\kappa(s').
\]
The map $\Spec\kappa(s')\to S$ factors through $\Spec\kappa(s)$, giving the stated form by another application of associativity.
    \end{solution}
