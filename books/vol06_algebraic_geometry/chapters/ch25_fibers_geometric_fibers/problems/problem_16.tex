\subsection*{VI.25.P16: Geometric splitting of a real point}
\begin{problem}\label{prob:vi25-p16}
Show explicitly that the geometric fiber of
\[
\Spec\mathbb C\to\Spec\mathbb R
\]
over the unique point has two points.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This problem asks for the ring-level calculation behind the basic geometric splitting phenomenon.

\paragraph{Useful ideas before starting.}
Present \(\mathbb C\) as \(\mathbb R[T]/(T^2+1)\), tensor with \(\mathbb C\), factor, and apply CRT.

\paragraph{Guided hints.}
\begin{enumerate}
\item Compute \(\mathbb C\otimes_{\mathbb R}\mathbb C\).
\item Use \(T^2+1=(T-i)(T+i)\).
\item Take spectra of the product ring.
\end{enumerate}

\begin{solution}
The unique point of \(\Spec\mathbb R\) has residue field \(\mathbb R\).  A geometric point above it may be chosen as
\[
\Spec\mathbb C\to\Spec\mathbb R.
\]
The geometric fiber is therefore
\[
\Spec\mathbb C\times_{\Spec\mathbb R}\Spec\mathbb C
=
\Spec(\mathbb C\otimes_{\mathbb R}\mathbb C).
\]
Writing the second factor as \(\mathbb R[T]/(T^2+1)\),
\[
\mathbb C\otimes_{\mathbb R}\mathbb C
\cong
\mathbb C[T]/(T^2+1)
\cong
\mathbb C[T]/((T-i)(T+i)).
\]
The two factors are comaximal, so
\[
\mathbb C\otimes_{\mathbb R}\mathbb C
\cong
\mathbb C\times\mathbb C.
\]
Taking spectra,
\[
\Spec(\mathbb C\times\mathbb C)
\cong
\Spec\mathbb C\amalg\Spec\mathbb C.
\]
Hence the one ordinary point splits into exactly two geometric points.
\end{solution}

\paragraph{What happened mathematically.}
A separable residue-field extension can split into several geometric points after algebraic closure.

\paragraph{Extensions.}
Finite separable extensions split into one geometric point for each field embedding into the algebraic closure.
