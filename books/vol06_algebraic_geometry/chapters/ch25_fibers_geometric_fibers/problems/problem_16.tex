\begin{problem}[Geometric splitting of a real point]\label{prob:vi25-p16}
    Show explicitly that the geometric fiber of $\Spec\mathbb C\to\Spec\mathbb R$ over the unique point has two points.
    \end{problem}
    \begin{solution}
    Using $\mathbb C=\mathbb R[t]/(t^2+1)$,
\[
\mathbb C\otimes_{\mathbb R}\mathbb C\cong\mathbb C[t]/((t-i)(t+i))\cong\mathbb C\times\mathbb C
\]
by the Chinese remainder theorem.  Taking spectra gives two copies of $\Spec\mathbb C$.
    \end{solution}
