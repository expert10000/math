\begin{example}[Residue fields after geometric extension]\label{ex:vi25-a01}
If $x\in X_s$, points of the geometric fiber above $x$ are controlled by primes of $\kappa(x)\otimes_{\kappa(s)}\Omega$; one ordinary point can therefore split geometrically.
\end{example}
\begin{example}[Purely inseparable nonreduced geometric fiber]\label{ex:vi25-a02}
In characteristic $p$, let $k=\mathbb F_p(a^p)$ and $L=k(a)$. Then $L\otimes_kL$ contains nilpotents because $T^p-a^p=(T-a)^p$ after extending scalars to $L$.
\end{example}
\begin{example}[Fiber of a composition]\label{ex:vi25-a03}
For $X\to S'\to S$, the fiber over $s\in S$ can be studied by first base-changing $S'$ to $\kappa(s)$ and then pulling back $X$; this is associativity of fiber products.
\end{example}
\begin{example}[Fiber after base change]\label{ex:vi25-a04}
If $s'\mapsto s$, then $(X\times_SS')_{s'}\cong X_s\times_{\kappa(s)}\kappa(s')$.
\end{example}
\begin{example}[Generic fiber detects vertical equations]\label{ex:vi25-a05}
If an equation is annihilated by a nonzero element of the domain $A$, it can disappear after tensoring with $\operatorname{Frac}(A)$; special fibers may retain it.
\end{example}
\begin{example}[A family with constant dimension but changing incidence]\label{ex:vi25-a06}
DG4-P12 has two one-dimensional components in every fiber, yet the intersection pattern jumps at the special parameter.
\end{example}
\begin{example}[A family with changing number of points]\label{ex:vi25-a07}
For $u^2-t=0$ over a field of characteristic not $2$, a nonzero square parameter yields two rational points, a nonsquare may yield one degree-two point, and $t=0$ yields a doubled point.
\end{example}
\begin{example}[Geometric splitting removes arithmetic ambiguity]\label{ex:vi25-a08}
After passing to an algebraic closure, a separable polynomial splits into linear factors; the geometric fiber separates the corresponding geometric points.
\end{example}
\begin{example}[Nilpotents survive algebraic closure]\label{ex:vi25-a09}
If a fiber already contains a nonzero nilpotent, every faithfully flat field extension preserves its nonzeroness, so algebraic closure does not repair nonreducedness.
\end{example}
\begin{example}[Products of fibers]\label{ex:vi25-a10}
Associativity gives $(X\times_SY)_{\bar s}\cong X_{\bar s}\times_{\Omega}Y_{\bar s}$ for a geometric point with field $\Omega$.
\end{example}
\begin{example}[Empty geometric fibers]\label{ex:vi25-a11}
If $X_s=\varnothing$, every geometric fiber over $s$ is empty; scalar extension of the zero ring stays zero.
\end{example}
\begin{example}[Geometric connectedness warning]\label{ex:vi25-a12}
A connected scheme over a non-algebraically-closed field can become disconnected after scalar extension, as $\Spec\mathbb C$ over $\mathbb R$ demonstrates.
\end{example}
\begin{example}[Arithmetic factorization as fiber geometry]\label{ex:vi25-a13}
For monic $F\in\mathbb Z[T]$, the decomposition of $F$ modulo $p$ controls the components and residue fields of the $p$-fiber of $\Spec\mathbb Z[T]/(F)$.
\end{example}
\begin{example}[Fiber data is local on source and base]\label{ex:vi25-a14}
Fiber computations may be performed on affine neighborhoods of $s$ and affine opens of $X$ lying over them, then glued; the tensor formula is the local engine.
\end{example}
