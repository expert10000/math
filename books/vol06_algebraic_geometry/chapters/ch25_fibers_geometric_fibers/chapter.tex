\chapter{Fibers and Geometric Fibers}\label{ch:vi25-fibers-geometric-fibers}

A morphism $f:X\to S$ is a family of schemes parametrized by the points of $S$.  VI/23 constructed
fiber products and VI/24 developed base change.  The most important specialization of that calculus
is obtained by changing the base from $S$ to the residue field of one point $s\in S$.  The result is
the \emph{scheme-theoretic fiber}
\[
X_s:=X\times_S\Spec\kappa(s).
\]
It remembers more than the set of points above $s$: it remembers nilpotents, multiplicities, residue
fields, components, and the equations after specialization.  Extending $\kappa(s)$ further to an
algebraically closed field produces a geometric fiber and reveals geometry hidden over the original
residue field.

\section{The scheme-theoretic fiber}
Let $f:X\to S$ and $s\in S$.  The canonical point morphism $\Spec\kappa(s)\to S$ defines the Cartesian square in
\cref{fig:vi25-01}.  By definition, $X_s$ is a scheme over the field $\kappa(s)$.

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_01}

\begin{definition}[Fiber]\label{def:vi25-fiber}
The \emph{fiber of $f$ over $s$} is $X_s=X\times_S\Spec\kappa(s)$.
\end{definition}

The projection $X_s\to X$ lands on the set-theoretic preimage of $s$.  The scheme structure on
$X_s$, however, may contain nilpotents or residue-field information that the bare set cannot record.

\section{Affine fiber formula}
Suppose $S=\Spec A$, $X=\Spec B$, and $f$ is induced by $A\to B$.  For
$\mathfrak p\in\Spec A$,
\[
X_{\mathfrak p}\cong
\Spec\bigl(B\otimes_A\kappa(\mathfrak p)\bigr).
\]
Since
\[
\kappa(\mathfrak p)=\operatorname{Frac}(A/\mathfrak p),
\]
the computation has two steps: impose the equations in $\mathfrak p$, then invert every nonzero
element of $A/\mathfrak p$.

\begin{theorem}[Affine fiber formula]\label{thm:vi25-affine-fiber}
For an $A$-algebra $B$ and $\mathfrak p\in\Spec A$,
\[
\Spec B\times_{\Spec A}\Spec\kappa(\mathfrak p)
\cong\Spec(B\otimes_A\kappa(\mathfrak p)).
\]
\end{theorem}

\begin{example}[Fiber of a polynomial map]\label{ex:vi25-expand-01}
For $\mathbb A^1_k\to\mathbb A^1_k$ given by $t\mapsto t^2$, the fiber over $a\in k$ has coordinate ring $k[t]/(t^2-a)$.  Its splitting and multiplicity depend on $a$ and on the field.
\end{example}
% END VOL06-EXPANSION VI25-example-01
\section{Specializing equations}
For a family
\[
X=\Spec k[t,x_1,\dots,x_n]/I\longrightarrow\Spec k[t],
\]
the fiber over the closed point $(t-a)$ is obtained by substituting $t=a$:
\[
X_a\cong\Spec k[x_1,\dots,x_n]/I_a.
\]
This elementary operation can change factorization, incidence, reducedness, or the number of
components.  The legacy degeneration DG4-P12 is the cleanest example: all fibers are reduced unions
of two affine lines, but the two lines intersect for $a\ne0$ and become parallel at $a=0$.

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_03}

\section{Generic and special fibers over \texorpdfstring{$\Spec\mathbb Z$}{Spec Z}}
For $X\to\Spec\mathbb Z$, the generic point $(0)$ has residue field $\mathbb Q$, whereas the closed
point $(p)$ has residue field $\mathbb F_p$.  Hence
\[
X_{\mathbb Q}=X\times_{\Spec\mathbb Z}\Spec\mathbb Q,
\qquad
X_p=X\times_{\Spec\mathbb Z}\Spec\mathbb F_p.
\]
A single arithmetic scheme therefore packages a characteristic-zero generic fiber together with
fibers in every prime characteristic.

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_04}

\section{Reducedness and irreducibility can vary}
The ring of a fiber must be analyzed after tensoring with the residue field.  Typical outcomes are
\[
k,\qquad k\times k,\qquad k[\varepsilon]/(\varepsilon^2),\qquad 0.
\]
These correspond respectively to one reduced point, two reduced points, one nonreduced doubled point,
and the empty scheme.  Thus reducedness and irreducibility are independent questions.

\begin{remark}\label{rem:vi25-empty-reduced}
The empty scheme is reduced.  We do not call it irreducible under the standard convention that an
irreducible topological space is nonempty.
\end{remark}

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_05}

\section{Geometric fibers}
Let $s\in S$ and choose an algebraically closed extension field $\Omega$ of $\kappa(s)$.  The
\emph{geometric fiber over the chosen geometric point} $\bar s:\Spec\Omega\to S$ is
\[
X_{\bar s}:=X_s\times_{\Spec\kappa(s)}\Spec\Omega.
\]
It is the ordinary fiber with its ground field enlarged enough to expose algebraic splitting.
Different choices of algebraically closed overfield can be compared after passage to a common
overfield; the purpose is to distinguish geometry intrinsic after field extension from accidents of
the original residue field.

\begin{definition}[Geometric fiber]\label{def:vi25-geometric-fiber}
For a geometric point $\bar s:\Spec\Omega\to S$ with image $s$ and $\Omega$ algebraically closed,
$X_{\bar s}:=X\times_S\Spec\Omega\cong X_s\times_{\kappa(s)}\Omega$.
\end{definition}

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_06}

% BEGIN VOL06-EXPANSION VI25-example-02
\begin{example}[Nonreduced special fiber]\label{ex:vi25-expand-02}
For the family $\operatorname{Spec}k[t,x]/(x^2-t)\to\operatorname{Spec}k[t]$, the fiber at $t=0$ is $\operatorname{Spec}k[x]/(x^2)$, a nonreduced double point, while fibers at nonzero parameters can be reduced.
\end{example}
% END VOL06-EXPANSION VI25-example-02
\section{Why ordinary and geometric fibers differ}
Over $\mathbb R$, $\Spec\mathbb C$ is a single point and is irreducible.  After extension to
$\mathbb C$,
\[
\mathbb C\otimes_{\mathbb R}\mathbb C\cong\mathbb C\times\mathbb C,
\]
so its geometric fiber consists of two points.  Thus an ordinary fiber may be irreducible over its
residue field but become reducible after algebraic closure.  Purely inseparable field extensions can
also make a geometric fiber nonreduced.

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_07}

\section{Fibers after a further base change}
Suppose $S'\to S$, $s'\in S'$ maps to $s\in S$, and $X'=X\times_SS'$.  There is a canonical
isomorphism
\[
X'_{s'}\cong X_s\times_{\Spec\kappa(s)}\Spec\kappa(s').
\]
Thus taking a fiber after base change means first taking the old fiber and then extending its residue
field.  This formula is the bridge between VI/24 and the geometric-fiber viewpoint.

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_08}

% BEGIN VOL06-EXPANSION VI25-example-03
\begin{example}[Geometric fiber]\label{ex:vi25-expand-03}
The geometric fiber at $s\in S$ is obtained by extending the ordinary fiber from $\kappa(s)$ to an algebraic closure $\overline{\kappa(s)}$.  This separates field-arithmetic effects from intrinsic geometric structure.
\end{example}
% END VOL06-EXPANSION VI25-example-03
\section{Arithmetic fibers and factorization}
For
\[
X=\Spec\mathbb Z[T]/(T^2+1)\longrightarrow\Spec\mathbb Z,
\]
the fiber at $p$ is
\[
X_p=\Spec\mathbb F_p[T]/(T^2+1).
\]
At $p=2$ the polynomial is $(T+1)^2$ and the fiber is nonreduced.  If $p\equiv1\pmod4$ it splits into
two distinct linear factors, whereas if $p\equiv3\pmod4$ it remains irreducible and gives
$\Spec\mathbb F_{p^2}$.  Scheme-theoretic fibers therefore record ramification, splitting, and inertia
in one uniform construction.

\input{chapters/ch25_fibers_geometric_fibers/figures/figure_09}

\section{Boundary with adjacent chapters}
VI/23 supplies the universal fiber-product construction and VI/24 supplies arbitrary base change.
VI/25 specializes them to points and geometric points and analyzes the resulting schemes.  We do not
yet develop finite-type or Noetherian morphisms (VI/26), integral schemes and function fields (VI/27),
or normalization (VI/28).

\section{Chapter synthesis}
A fiber is not merely a set-theoretic inverse image.  It is a base change to a residue field, and its
coordinate ring can retain nilpotents, multiplicity, field extensions, and changing component
structure.  Geometric fibers then extend the residue field to expose splitting and inseparability.
Together the ordinary and geometric fibers turn a morphism into a family whose members can be
studied with the full algebraic structure intact.

\subsection*{Standard computations}

\begin{exercise}[Square map fiber]\label{exr:vi25-expand-01}
Write the coordinate ring of the fiber of $t\mapsto t^2$ over $a$.
\end{exercise}

\end{hint}

$k[t]/(t^2-a)$.
\end{solution}

\end{hint}
\begin{solution}
$k[x]/(x^2)$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Affine fiber formula]\label{exr:vi25-expand-06}
Prove: Prove the fiber of $\operatorname{Spec}B\to\operatorname{Spec}A$ over $\mathfrak p$ is $\operatorname{Spec}(B\otimes_A\kappa(\mathfrak p))$.
\end{exercise}
\begin{hint}
Work directly from scheme fibers, residue fields, and geometric fibers.
\end{hint}
\begin{solution}
Apply the affine fiber-product formula to $\operatorname{Spec}\kappa(\mathfrak p)\to\operatorname{Spec}A$.
\end{solution}

\begin{exercise}[Geometric base change]\label{exr:vi25-expand-07}
Prove: Prove the geometric fiber is a base change of the ordinary fiber.
\end{exercise}
\begin{hint}
Work directly from scheme fibers, residue fields, and geometric fibers.
\end{hint}
\begin{solution}
Associate the two fiber products along $\operatorname{Spec}\overline{\kappa(s)}\to\operatorname{Spec}\kappa(s)$.
\end{solution}

\begin{exercise}[Special nonreduced]\label{exr:vi25-expand-08}
Prove: Prove the fiber of $x^2-t$ at $t=0$ is nonreduced.
\end{exercise}
\begin{hint}
Work directly from scheme fibers, residue fields, and geometric fibers.
\end{hint}
\begin{solution}
The coordinate ring contains the nonzero nilpotent class of $x$.
\end{solution}

\begin{exercise}[Fiber compatibility]\label{exr:vi25-expand-09}
Prove: Prove fibers commute with further base change.
\end{exercise}
\begin{hint}
Work directly from scheme fibers, residue fields, and geometric fibers.
\end{hint}
\begin{solution}
Use associativity of fiber products.
\end{solution}

\subsection*{Applications and investigations}

How do fibers describe degeneration?
\end{exercise}
\begin{hint}
Translate the question into scheme fibers, residue fields, and geometric fibers.
\end{hint}
\begin{solution}
They compare the geometry of members of a family at different base points.
\end{solution}

Why inspect geometric fibers?
\end{exercise}

Translate the question into scheme fibers, residue fields, and geometric fibers.
\end{hint}
\begin{solution}
They test properties after removing residue-field arithmetic obstructions.
\end{solution}
