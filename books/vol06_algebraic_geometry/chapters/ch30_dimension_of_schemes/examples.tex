\begin{example}[A point]\label{ex:vi30-01}
$\Spec k$ has dimension $0$: there is one prime and no strict prime inclusion.
\end{example}

\begin{example}[The affine line]\label{ex:vi30-02}
$\mathbb A^1_k=\Spec k[t]$ has dimension $1$ from $(0)\subsetneq(t-a)$, and no longer prime chain exists.
\end{example}

\begin{example}[The affine plane]\label{ex:vi30-03}
$\mathbb A^2_k$ has dimension $2$, witnessed by $(0)\subsetneq(x)\subsetneq(x,y)$.
\end{example}

\begin{example}[Reduction]\label{ex:vi30-04}
$\Spec k[t]/(t^2)$ and $\Spec k[t]/(t)$ have the same one-point topology and both have dimension $0$.
\end{example}

\begin{example}[A closed line in the plane]\label{ex:vi30-05}
$V(x)\subset\mathbb A^2_k$ is isomorphic to $\mathbb A^1_k$, so its dimension drops from $2$ to $1$.
\end{example}

\begin{example}[A principal open of the affine plane]\label{ex:vi30-06}
$D(x)\subset\mathbb A^2_k$ has coordinate ring $k[x,x^{-1},y]$ and dimension $2$; this agrees with nonempty-open invariance for varieties.
\end{example}

\begin{example}[Two components of different dimensions]\label{ex:vi30-07}
$\Spec k\sqcup\mathbb A^1_k$ has dimension $1$, the maximum of component dimensions.
\end{example}

\begin{example}[Open component of smaller dimension]\label{ex:vi30-08}
In $\Spec k\sqcup\mathbb A^1_k$, the component $\Spec k$ is nonempty open of dimension $0$, strictly below the ambient dimension $1$.
\end{example}

\begin{example}[Affine three-space]\label{ex:vi30-09}
$\dim\mathbb A^3_k=3$ by the chain $(0)\subsetneq(x)\subsetneq(x,y)\subsetneq(x,y,z)$.
\end{example}

\begin{example}[An affine curve]\label{ex:vi30-10}
If $A=k[x,y]/(f)$ is a domain with nonconstant irreducible $f$, then $\Frac(A)$ has transcendence degree $1$, hence $\dim\Spec A=1$.
\end{example}

\begin{example}[An affine surface]\label{ex:vi30-11}
If $A=k[x,y,z]/(f)$ is a finitely generated domain with one nontrivial algebraic relation of the expected height, then $\operatorname{trdeg}_k\Frac(A)=2$ and $\dim\Spec A=2$.
\end{example}

\begin{example}[Normalization]\label{ex:vi30-12}
The cusp $\Spec k[t^2,t^3]$ and its normalization $\Spec k[t]$ both have dimension $1$.
\end{example}

\begin{example}[Finite field extension]\label{ex:vi30-13}
For a finite extension $L/k$, $\Spec L\to\Spec k$ is finite and both schemes have dimension $0$.
\end{example}

\begin{example}[Disjoint union of affine spaces]\label{ex:vi30-14}
$\coprod_{n\ge0}\mathbb A^n_k$ has infinite dimension because the component dimensions are unbounded.
\end{example}

\begin{example}[Localization can lower dimension]\label{ex:vi30-15}
For $A=k[x,y]_{(x)}$, $\dim\Spec A=1$ but $D(x)$ has only the generic point and dimension $0$.
\end{example}

\begin{example}[Dense open in a variety]\label{ex:vi30-16}
If $X$ is a variety and $U\subseteq X$ is nonempty open, then $U$ is dense and $\dim U=\dim X$.
\end{example}

\begin{example}[Function field of affine space]\label{ex:vi30-17}
$K(\mathbb A^n_k)=k(x_1,\ldots,x_n)$ has transcendence degree $n$, recovering $\dim\mathbb A^n_k=n$.
\end{example}

\begin{example}[Product with the affine line]\label{ex:vi30-18}
For a geometrically integral variety $X$, $\dim(X\times_k\mathbb A^1_k)=\dim X+1$.
\end{example}

\begin{example}[A zero-dimensional finite scheme]\label{ex:vi30-19}
Any finite $k$-scheme has dimension $0$, even if it is nonreduced.
\end{example}

\begin{example}[Closed subset inequality]\label{ex:vi30-20}
Every closed subscheme $Y\hookrightarrow X$ satisfies $\dim Y\le\dim X$ because its underlying irreducible-closed chains are chains in $X$.
\end{example}
