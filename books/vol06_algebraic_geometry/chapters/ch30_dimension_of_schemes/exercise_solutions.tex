% VI30 detailed exercise solutions -- VI/24+ proof-depth standard.

% VI30-DETAILED-EXERCISE-01
\paragraph{Solution VI.30.E1.}
The spectrum of a field \(k\) has one point, corresponding to the unique prime ideal \((0)\).
There is therefore no strict chain of two distinct irreducible closed subsets. Equivalently,
\[
\dim\Spec k=\dim k=0.
\]

% VI30-DETAILED-EXERCISE-02
\paragraph{Solution VI.30.E2.}
In \(k[x,y]\),
\[
(0)\subsetneq(x)\subsetneq(x,y)
\]
is a prime chain: the quotients by \((x)\) and \((x,y)\) are \(k[y]\) and \(k\), respectively.
It has two strict inclusions, hence length \(2\). Geometrically,
\[
V(x,y)\subsetneq V(x)\subsetneq\mathbb A^2_k.
\]

% VI30-DETAILED-EXERCISE-03
\paragraph{Solution VI.30.E3.}
Since
\[
\mathbb A^4_k=\Spec k[x_1,x_2,x_3,x_4]
\]
and the polynomial-ring dimension theorem gives
\[
\dim k[x_1,x_2,x_3,x_4]=4,
\]
we obtain
\[
\boxed{\dim\mathbb A^4_k=4}.
\]

% VI30-DETAILED-EXERCISE-04
\paragraph{Solution VI.30.E4.}
Every irreducible closed subset of a closed subspace \(Y\subseteq X\) is also irreducible and closed
in \(X\). Therefore every strict chain in \(Y\) is one in \(X\), so
\[
\boxed{\dim Y\le\dim X=2}.
\]
Equality can occur for a proper closed subset when \(X\) is reducible.

% VI30-DETAILED-EXERCISE-05
\paragraph{Solution VI.30.E5.}
A nilpotent thickening is a homeomorphism on underlying spaces. Affinely, if \(I\subseteq A\) is
nilpotent, every prime contains \(I\), and
\[
\Spec(A/I)\longrightarrow\Spec A
\]
is a homeomorphism. Since global dimension is defined from irreducible-closed chains,
\[
\boxed{\dim\Spec(A/I)=\dim\Spec A}.
\]

% VI30-DETAILED-EXERCISE-06
\paragraph{Solution VI.30.E6.}
An irreducible subset of a disjoint union lies in one component. Thus
\[
\dim(X\sqcup Y)=\max\{\dim X,\dim Y\}.
\]
Hence
\[
\boxed{\dim(\mathbb A^2_k\sqcup\mathbb A^5_k)=5}.
\]

% VI30-DETAILED-EXERCISE-07
\paragraph{Solution VI.30.E7.}
Take
\[
U=D(t)\subseteq\Spec k[t].
\]
Then \(U=\Spec k[t,t^{-1}]\). The chain
\[
(0)\subsetneq(t-1)
\]
survives localization, so \(\dim U\ge1\). Localization cannot increase dimension, so \(\dim U\le1\).
Therefore
\[
\boxed{\dim D(t)=1}.
\]

% VI30-DETAILED-EXERCISE-08
\paragraph{Solution VI.30.E8.}
Let \(A=k[x,y]_{(x)}\). Its primes correspond to primes of \(k[x,y]\) contained in \((x)\).
Since \((x)\) has height \(1\), these are \((0)\) and \((x)\). The open \(D(x)\) removes \((x)\),
leaving only the generic point \((0)\). Hence
\[
\boxed{\dim D(x)=0}.
\]

% VI30-DETAILED-EXERCISE-09
\paragraph{Solution VI.30.E9.}
For a variety, every nonempty open subset has the same dimension as the whole space. Thus if
\(\dim X=3\),
\[
\boxed{\dim U=3}
\]
for every nonempty open \(U\subseteq X\).

% VI30-DETAILED-EXERCISE-10
\paragraph{Solution VI.30.E10.}
If
\[
K(X)=k(s,t),
\]
then
\[
\operatorname{trdeg}_kK(X)=2.
\]
The dimension--transcendence-degree theorem gives
\[
\boxed{\dim X=2}.
\]

% VI30-DETAILED-EXERCISE-11
\paragraph{Solution VI.30.E11.}
If \(K(X)/k\) is algebraic, then
\[
\operatorname{trdeg}_kK(X)=0.
\]
Therefore for a variety
\[
\boxed{\dim X=0}.
\]

% VI30-DETAILED-EXERCISE-12
\paragraph{Solution VI.30.E12.}
A dominant morphism \(C\to S\) from a curve to a surface would induce
\[
K(S)\hookrightarrow K(C).
\]
Hence
\[
\operatorname{trdeg}_kK(S)\le\operatorname{trdeg}_kK(C),
\]
which by the dimension theorem says \(2\le1\), impossible. Therefore no such dominant morphism exists.

% VI30-DETAILED-EXERCISE-13
\paragraph{Solution VI.30.E13.}
For a dominant generically finite morphism \(X\to Y\), the extension
\[
K(Y)\hookrightarrow K(X)
\]
is algebraic. Algebraic extensions preserve transcendence degree, hence
\[
\dim X=\dim Y=4.
\]
Thus
\[
\boxed{\dim X=4}.
\]

% VI30-DETAILED-EXERCISE-14
\paragraph{Solution VI.30.E14.}
For a finite disjoint union the dimension is the maximum of component dimensions. Since
\(\dim\mathbb A^n_k=n\),
\[
\boxed{\dim\left(\coprod_{n=0}^{10}\mathbb A^n_k\right)=10}.
\]

% VI30-DETAILED-EXERCISE-15
\paragraph{Solution VI.30.E15.}
Each component \(\Spec k\) has dimension \(0\), and for an arbitrary disjoint union
\[
\dim\left(\coprod_iX_i\right)=\sup_i\dim X_i.
\]
Therefore
\[
\boxed{\dim\left(\coprod_{n\ge0}\Spec k\right)=0}.
\]

\[
X=\Spec k[x,y]_{(x)}.
\]

% VI30-DETAILED-EXERCISE-17
\paragraph{Solution VI.30.E17.}
Under the stated hypothesis, the product is a variety. Its function field is
\[
K(X)(t).
\]
Thus
\[
\operatorname{trdeg}_kK(X)(t)=\operatorname{trdeg}_kK(X)+1=d+1,
\]
so
\[
\boxed{\dim(X\times_k\mathbb A^1_k)=d+1}.
\]

% VI30-DETAILED-EXERCISE-18
\paragraph{Solution VI.30.E18.}
Reduction changes the structure sheaf while preserving the underlying topological space. For example
\[
\Spec k[\varepsilon]/(\varepsilon^2)
\quad\text{and}\quad
\Spec k
\]
both have dimension \(0\), but the first has a nonzero nilpotent global section. Hence equal dimensions do not
imply isomorphic schemes.

% VI30-DETAILED-EXERCISE-19
\paragraph{Solution VI.30.E19.}
Normalization is integral, and integral extensions preserve Krull dimension. Hence for an integral curve
\[
\boxed{\dim X^\nu=\dim X=1}.
\]
The normalization may change singular local rings while leaving the global chain length unchanged.

% VI30-DETAILED-EXERCISE-20
\paragraph{Solution VI.30.E20.}
If \(A\subseteq B\) is an integral extension, then
\[
\boxed{\dim B=\dim A}.
\]
For a general integral ring map \(A\to B\), one has
\[
\dim B=\dim(A/\ker(A\to B))\le\dim A.
\]
If the induced map on spectra is surjective, the kernel lies in the nilradical, so the quotient has the same
dimension as \(A\), and equality follows.

% VI30-DETAILED-EXERCISE-21
\paragraph{Solution VI.30.E21.}
A closed point \(x\) gives the irreducible closed subset \(\{x\}\). A one-point space has no strict nontrivial
chain of irreducible closed subsets, so
\[
\boxed{\dim\{x\}=0}.
\]

\[
D(t)\subsetneq\mathbb A^1_k

\[
\dim D(t)=1=\dim\mathbb A^1_k.

% VI30-DETAILED-EXERCISE-24
\paragraph{Solution VI.30.E24.}
Take
\[
X=\coprod_{n\ge0}\mathbb A^n_k.
\]
Then
\[
\dim X=\sup_{n\ge0}\dim\mathbb A^n_k=\sup_{n\ge0}n=\boxed{\infty}.
\]
No component is infinite-dimensional; the global value is an unbounded supremum.

% VI30-DETAILED-EXERCISE-25
\paragraph{Solution VI.30.E25.}
Global dimension
\[
\dim X
\]
measures irreducible-closed chains anywhere in \(X\). Local dimension at \(x\) is
\[
\dim\mathcal O_{X,x},
\]
which measures prime chains in the local ring. VI/30 treats the global invariant; the geometric local theory is
reserved for VI/32.

% VI30-DETAILED-EXERCISE-26
\paragraph{Solution VI.30.E26.}
Dimension is an absolute invariant of \(X\). Codimension is relative to an inclusion \(Y\subseteq X\) and measures
how deeply an irreducible closed subset sits in its ambient space. Under suitable hypotheses one later obtains
\[
\dim Y+\operatorname{codim}(Y,X)=\dim X.
\]
Systematic codimension theory belongs to VI/31.
