\chapter{Dimension of Schemes}\label{ch:vi30-dimension-schemes}

VI/29 built the affine engine: Krull dimension is the supremum of lengths of prime chains, equivalently chains of
irreducible closed subsets of an affine scheme.  VI/30 globalizes that invariant to schemes and isolates the extra
hypotheses under which dimension behaves uniformly on open subsets.  The central source theorem is the variety case:
if $X$ is an integral scheme of finite type over a field and $U\subseteq X$ is nonempty open, then
$\dim U=\dim X$.  This is false for general schemes, and even for some Noetherian integral schemes.

\section{Global dimension}\label{sec:vi30-global-definition}
\begin{definition}[Dimension of a scheme]\label{def:vi30-scheme-dimension}
For a nonempty scheme $X$, define
\[
\dim X=\sup\{n:Z_0\subsetneq Z_1\subsetneq\cdots\subsetneq Z_n
\text{ are irreducible closed subsets of }X\}.
\]
The empty scheme is assigned dimension $-\infty$ when a convention is needed.
\end{definition}
For $X=\Spec A$, the affine chain dictionary from VI/29 gives
\[
\dim X=\dim A.
\]
Thus the global definition extends Krull dimension without changing the affine case.
\input{chapters/ch30_dimension_of_schemes/figures/figure_01}

\section{Closed subsets, components, and reduction}\label{sec:vi30-closed-components}
If $Y\subseteq X$ is closed, every irreducible closed chain in $Y$ is one in $X$, hence
\begin{proposition}[Closed subsets do not increase dimension]\label{prop:vi30-closed-inequality}
\[
\dim Y\leq \dim X.
\]
\end{proposition}
If $X$ is Noetherian with irreducible components $X_1,\ldots,X_r$, every irreducible closed subset is contained in
some $X_i$, so
\begin{proposition}[Dimension from irreducible components]\label{prop:vi30-components-max}
\[
\dim X=\max_i\dim X_i.
\]
\end{proposition}
The reduction $X_{\mathrm{red}}\to X$ is a homeomorphism, so
\begin{proposition}[Nilpotents do not change global dimension]\label{prop:vi30-reduction}
\[
\dim X_{\mathrm{red}}=\dim X.
\]
\end{proposition}
\input{chapters/ch30_dimension_of_schemes/figures/figure_02}

\begin{example}[Affine space]\label{ex:vi30-expand-01}
The scheme $\mathbb A^n_k$ has dimension $n$ because its coordinate ring $k[x_1,\ldots,x_n]$ has Krull dimension $n$.
\end{example}

\section{Open subsets: the general inequality}\label{sec:vi30-open-inequality}
For an open subset $U\subseteq X$, the closure in $X$ of an irreducible closed subset of $U$ is irreducible, and a
strict chain in $U$ gives a strict chain of closures in $X$.  Therefore
\begin{proposition}[Open subsets cannot increase dimension]\label{prop:vi30-open-inequality}
\[
\dim U\leq\dim X.
\]
\end{proposition}
Equality need not hold.  The exact AG8 Problem 3 counterexample is
\[
X=\Spec k\sqcup\mathbb A^1_k,\qquad U=\Spec k.
\]
Then $U$ is nonempty and open, but $\dim U=0$ whereas $\dim X=1$.
\input{chapters/ch30_dimension_of_schemes/figures/figure_03}

\section{Why irreducibility alone is not enough}\label{sec:vi30-irreducible-warning}
Even Noetherian and integral do not force every nonempty open to have full dimension.  Let
\[
A=k[x,y]_{(x)}.
\]
Since $y$ is inverted, $A\cong k(y)[x]_{(x)}$ and $\dim A=1$.  But
\[
D(x)\subseteq\Spec A
\]
contains only the generic point, so $\dim D(x)=0$.  The missing hypothesis in the source theorem is the finite-type
variety setting, not merely irreducibility or Noetherianity.
\input{chapters/ch30_dimension_of_schemes/figures/figure_04}

\section{Varieties and nonempty open subsets}\label{sec:vi30-variety-open}
Here a variety over $k$ means an integral scheme of finite type over the field $k$.  If $V=\Spec A$ is a nonempty
affine open of a variety $X$, then $A$ is a finitely generated $k$-domain and its fraction field is $K(X)$.
The finite-type dimension theorem gives
\[
\dim A=\operatorname{trdeg}_k\operatorname{Frac}(A)
       =\operatorname{trdeg}_kK(X).
\]
Consequently every nonempty affine open has the same dimension.  Every nonempty open $U\subseteq X$ contains a
nonempty affine open $V$, so
\[
\dim X\geq\dim U\geq\dim V=\dim X.
\]
Thus:
\begin{theorem}[Nonempty-open invariance for varieties]\label{thm:vi30-open-variety}
If $X$ is a variety over $k$ and $\varnothing\neq U\subseteq X$ is open, then
\[
\dim U=\dim X.
\]
\end{theorem}
This is the VI/30 part of AG8 Problem 2.
\input{chapters/ch30_dimension_of_schemes/figures/figure_05}

\begin{example}[Hypersurface dimension]\label{ex:vi30-expand-02}
The plane curve $V(y-x^2)$ has coordinate ring $k[x]$, so its scheme dimension is $1$ even though it sits inside the two-dimensional affine plane.
\end{example}

\section{Dimension and the function field}\label{sec:vi30-trdeg}
VI/27 established that every nonempty affine chart of an integral scheme has the same fraction field $K(X)$.  In the
finite-type setting, combining that fact with the affine dimension theorem yields:
\begin{theorem}[Dimension--transcendence-degree theorem]\label{thm:vi30-trdeg}
For a variety $X$ over $k$,
\[
\dim X=\operatorname{trdeg}_k K(X).
\]
\end{theorem}
The function-field material itself remains canonically assigned to VI/27.  This theorem is its dimension-side
consequence and is not a second destination for that legacy fragment.
\input{chapters/ch30_dimension_of_schemes/figures/figure_06}

\section{Basic computations}\label{sec:vi30-computations}
Since $\dim k[x_1,\ldots,x_n]=n$,
\[
\dim\mathbb A^n_k=n.
\]
More generally, if $A$ is a finitely generated $k$-domain, then
\[
\dim\Spec A=\operatorname{trdeg}_k\Frac(A).
\]
For a finite disjoint union,
\[
\dim(X_1\sqcup\cdots\sqcup X_r)=\max_i\dim X_i,
\]
and for an arbitrary disjoint union the maximum is replaced by the supremum.
\input{chapters/ch30_dimension_of_schemes/figures/figure_07}

\section{Products of varieties}\label{sec:vi30-products}
For varieties $X,Y$ over $k$, every irreducible component of $X\times_kY$ has the expected dimension; in the standard
geometrically integral situation the product is itself a variety and
\[
\dim(X\times_kY)=\dim X+\dim Y.
\]
For affine integral charts this is the transcendence-degree computation for the relevant tensor-product component.
One must not suppress hypotheses: tensor products of domains over a non-algebraically-closed field need not remain
domains.
\input{chapters/ch30_dimension_of_schemes/figures/figure_08}

\begin{example}[Reducible dimension]\label{ex:vi30-expand-03}
For $X=V(xy)\subset\mathbb A^2$, both irreducible components are lines, so $X$ has dimension $1$.  Scheme dimension takes the supremum over all irreducible chains.
\end{example}
% END VOL06-EXPANSION VI30-example-03
\section{Dimension under integral and finite morphisms}\label{sec:vi30-integral-morphisms}
VI/29 proved that integral extensions preserve Krull dimension.  Hence an integral surjective affine morphism
$\Spec B\to\Spec A$ has source and target of the same dimension.  In particular normalization preserves dimension.
For finite surjective morphisms between Noetherian schemes, this affine statement globalizes componentwise:
\[
\dim X=\dim Y.
\]
This is a dimension consequence, not a reopening of the integral-extension or normalization chapters.
\input{chapters/ch30_dimension_of_schemes/figures/figure_09}

\section{Boundaries of VI/30}\label{sec:vi30-boundaries}
Global dimension must be kept distinct from two later invariants.  VI/31 studies codimension and the formula
$\dim Y+\operatorname{codim}(Y,X)=\dim X$ under suitable hypotheses.  VI/32 studies local dimension
$\dim\OO_{X,x}$ and tangent-space geometry.  Accordingly AG8 Problem 2(a)--(c) and the corresponding Problem 3
counterexamples are not consumed here.  VI/30 owns the global/open-dimension statement and the dimension-side
transcendence-degree theorem.

\section{Chapter synthesis}\label{sec:vi30-synthesis}
Scheme dimension is a topological chain invariant that agrees with Krull dimension on affine charts, but its global
behavior is sensitive to components and to the class of schemes under consideration.  Closed and open subsets can
only lower dimension.  In the finite-type integral setting over a field, every nonempty open has full dimension and
that common value is the transcendence degree of the function field.  The failures outside this setting are not
pathologies to hide: they explain exactly why the hypotheses in the variety theorem matter.

\subsection*{Proofs}

\begin{exercise}[Affine agreement]\label{exr:vi30-expand-06}
Prove: Prove topological scheme dimension of $\operatorname{Spec}A$ agrees with Krull dimension of $A$.
\end{exercise}
\begin{hint}
Work directly from scheme dimension and affine-local prime chains.
\end{hint}
\begin{solution}
Irreducible closed subsets are closures of prime points, and strict inclusion corresponds to strict reverse inclusion of prime ideals.
\end{solution}

\begin{exercise}[Open bound]\label{exr:vi30-expand-07}
Prove: Prove the dimension of an open subscheme is at most the dimension of the ambient scheme.
\end{exercise}
\begin{hint}
Work directly from scheme dimension and affine-local prime chains.
\end{hint}
\begin{solution}
Any chain of irreducible closed subsets in the open yields a corresponding chain in their ambient closures.
\end{solution}

\begin{exercise}[Component supremum]\label{exr:vi30-expand-08}
Prove: Prove dimension of a Noetherian scheme is the supremum of dimensions of irreducible components.
\end{exercise}
\begin{hint}
Work directly from scheme dimension and affine-local prime chains.
\end{hint}
\begin{solution}
Every irreducible chain lies inside some component, while component chains are ambient chains.
\end{solution}

\begin{exercise}[Parabola]\label{exr:vi30-expand-09}
Prove: Prove the parabola has dimension one via its coordinate-ring isomorphism with $k[x]$.
\end{exercise}
\begin{hint}
Work directly from scheme dimension and affine-local prime chains.
\end{hint}
\begin{solution}
Scheme isomorphisms preserve prime chains and dimension.
\end{solution}

\end{solution}
