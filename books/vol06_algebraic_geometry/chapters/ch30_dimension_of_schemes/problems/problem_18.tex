\begin{problem}[VI.30.P18: A finite union of closed subsets]\label{prob:vi30-18}
Suppose \(X=Y_1\cup\cdots\cup Y_r\) with each \(Y_i\) closed. Prove \(\dim X=\max_i\dim Y_i\) if the \(Y_i\) contain all irreducible components of \(X\).

\textbf{Provenance.} Canonical component argument.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
Z_0\subsetneq\cdots\subsetneq Z_n
\]
be an irreducible-closed chain in \(X\). The largest member \(Z_n\) is contained in some irreducible component
\(C\) of \(X\). By hypothesis, that component is contained in at least one \(Y_i\). Hence every \(Z_j\subseteq
Z_n\subseteq C\subseteq Y_i\), so
\[
n\le\dim Y_i\le\max_i\dim Y_i.
\]
Thus
\[
\dim X\le\max_i\dim Y_i.
\]
Conversely each \(Y_i\) is a closed subspace of \(X\), so
\[
\dim Y_i\le\dim X.
\]
Taking the maximum yields
\[
\boxed{\dim X=\max_i\dim Y_i}.
\]
\end{solution}
\fi
