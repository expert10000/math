% VI30-DETAILED-PROBLEM-17
\begin{problem}[VI.30.P17: Finite morphisms]\label{prob:vi30-17}
Let \(f:X\to Y\) be finite and surjective between affine schemes. Explain why \(\dim X=\dim Y\).

\textbf{Provenance.} Canonical bridge to finite morphisms.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Write
\[
Y=\Spec A,\qquad X=\Spec B,
\]
with ring map \(\varphi:A\to B\). Finiteness implies that \(B\) is a finite \(A\)-module and therefore integral
over the image
\[
A/\ker\varphi.
\]
Hence the integral-extension theorem gives
\[
\dim B=\dim(A/\ker\varphi).
\]

Surjectivity of
\[
\Spec B\to\Spec A
\]
means its image is all of \(\Spec A\). For an integral map the image is \(V(\ker\varphi)\), so
\[
V(\ker\varphi)=\Spec A.
\]
Thus \(\ker\varphi\subseteq\sqrt{(0)}\). Quotienting by an ideal contained in the nilradical does not alter the
underlying prime-poset dimension:
\[
\dim(A/\ker\varphi)=\dim A.
\]
Therefore
\[
\boxed{\dim X=\dim B=\dim A=\dim Y}.
\]
\end{solution}
\fi
