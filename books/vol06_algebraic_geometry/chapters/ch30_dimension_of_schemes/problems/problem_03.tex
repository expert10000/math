\begin{problem}[VI.30.P3: Affine compatibility]\label{prob:vi30-03}
Prove directly that the global topological dimension of \(\Spec A\) equals the Krull dimension of \(A\).

\textbf{Provenance.} Canonical VI/30 development from the VI/29 affine chain dictionary.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Every irreducible closed subset of \(\Spec A\) has the form
\[
V(\mathfrak p)
\]
for a unique prime ideal \(\mathfrak p\). Moreover,
\[
\mathfrak p\subsetneq\mathfrak q
\quad\Longleftrightarrow\quad
V(\mathfrak q)\subsetneq V(\mathfrak p).
\]
Therefore a prime chain
\[
\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_n
\]
corresponds, after reversing the order, to
\[
V(\mathfrak p_n)\subsetneq\cdots\subsetneq V(\mathfrak p_0).
\]
Both chains have \(n\) strict inclusions. The correspondence is reversible, so the sets of possible chain lengths
are identical. Taking suprema gives
\[
\boxed{\dim\Spec A=\dim A}.
\]
\end{solution}
\fi
