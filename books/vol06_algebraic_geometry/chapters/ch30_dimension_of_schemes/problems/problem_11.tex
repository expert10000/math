% VI30-DETAILED-PROBLEM-11
\begin{problem}[VI.30.P11: An integral Noetherian counterexample]\label{prob:vi30-11}
Let \(A=k[x,y]_{(x)}\). Show \(\dim\Spec A=1\) but \(\dim D(x)=0\).

\textbf{Provenance.} Canonical warning: finite type matters.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Localizing \(k[x,y]\) at the prime \((x)\) inverts every element not in \((x)\), in particular every nonzero
polynomial in \(y\). Thus
\[
A=k[x,y]_{(x)}\cong k(y)[x]_{(x)}.
\]
The latter is a one-dimensional local PID: its prime ideals are
\[
(0)\subsetneq(x).
\]
Hence
\[
\dim\Spec A=1.
\]

Now
\[
D(x)=\{\mathfrak p\in\Spec A:x\notin\mathfrak p\}.
\]
The maximal prime \((x)\) is excluded, leaving only \((0)\). Therefore \(D(x)\) is a one-point space and
\[
\boxed{\dim D(x)=0<1=\dim\Spec A}.
\]
Since \(A\) is a domain, \(D(x)\) is dense. This proves that integral and Noetherian are not sufficient for
nonempty-open dimension invariance.
\end{solution}
\fi
