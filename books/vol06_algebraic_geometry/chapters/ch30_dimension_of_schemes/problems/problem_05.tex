\begin{problem}[VI.30.P5: Closed subset inequality]\label{prob:vi30-05}
If $Y\subseteq X$ is closed, prove $\dim Y\leq\dim X$.

\textbf{Provenance.} Canonical global-dimension lemma.
\end{problem}
\begin{solution}
Every irreducible closed subset of $Y$ is also irreducible and closed in $X$. Therefore every strict chain in $Y$ is a strict chain in $X$, so its length is bounded by $\dim X$.
\end{solution}
