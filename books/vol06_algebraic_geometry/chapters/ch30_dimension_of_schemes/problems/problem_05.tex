\begin{problem}[VI.30.P5: Closed subset inequality]\label{prob:vi30-05}
If \(Y\subseteq X\) is closed, prove \(\dim Y\leq\dim X\).

\textbf{Provenance.} Canonical global-dimension lemma.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
Z_0\subsetneq Z_1\subsetneq\cdots\subsetneq Z_n
\]
be a chain of irreducible closed subsets of \(Y\). Since \(Y\) is closed in \(X\), every \(Z_i\) is also closed in
\(X\), and irreducibility is a property of the subspace topology. Thus the same display is an irreducible-closed
chain in \(X\). Every chain length occurring in \(Y\) therefore occurs in \(X\), and
\[
\boxed{\dim Y\le\dim X}.
\]
Strict inequality need not hold if \(X\) is reducible.
\end{solution}
\fi
