% VI30-DETAILED-PROBLEM-19
\begin{problem}[VI.30.P19: Product with affine space]\label{prob:vi30-19}
Let \(X\) be a geometrically integral variety over \(k\). Show \(\dim(X\times_k\mathbb A^m_k)=\dim X+m\).

\textbf{Provenance.} Canonical product computation under safe hypotheses.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Geometric integrality guarantees that
\[
X\times_k\mathbb A^m_k
\]
is again integral; it is clearly finite type over \(k\), hence a variety. Choose a nonempty affine open
\[
U=\Spec A\subseteq X.
\]
Then
\[
U\times_k\mathbb A^m_k
=
\Spec A[t_1,\ldots,t_m].
\]
Its fraction field is
\[
\Frac(A)(t_1,\ldots,t_m)
=
K(X)(t_1,\ldots,t_m).
\]
Therefore
\[
\operatorname{trdeg}_kK(X)(t_1,\ldots,t_m)
=
\operatorname{trdeg}_kK(X)+m.
\]
Applying the dimension theorem to the product and to \(X\),
\[
\boxed{
\dim(X\times_k\mathbb A^m_k)=\dim X+m.
}
\]
\end{solution}
\fi
