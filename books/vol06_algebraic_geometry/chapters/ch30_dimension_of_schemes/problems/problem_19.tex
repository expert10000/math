\begin{problem}[VI.30.P19: Product with affine space]\label{prob:vi30-19}
Let $X$ be a geometrically integral variety over $k$. Show $\dim(X\times_k\mathbb A^m_k)=\dim X+m$.

\textbf{Provenance.} Canonical product computation under safe hypotheses.
\end{problem}
\begin{solution}
On a nonempty affine chart $\Spec A\subseteq X$, the product chart has coordinate ring $A[t_1,\ldots,t_m]$. Its fraction field is $K(X)(t_1,\ldots,t_m)$, whose transcendence degree over $k$ is $\operatorname{trdeg}_kK(X)+m$. Apply the variety dimension theorem.
\end{solution}
