\begin{problem}[VI.30.P4: Affine space]\label{prob:vi30-04}
Compute \(\dim\mathbb A^n_k\).

\textbf{Provenance.} Canonical computation.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
By definition,
\[
\mathbb A^n_k=\Spec k[x_1,\ldots,x_n].
\]
The polynomial-ring dimension theorem gives
\[
\dim k[x_1,\ldots,x_n]=n.
\]
Using affine compatibility,
\[
\boxed{\dim\mathbb A^n_k=n}.
\]
The lower bound is visible in the coordinate chain
\[
(0)\subsetneq(x_1)\subsetneq\cdots\subsetneq(x_1,\ldots,x_n).
\]
\end{solution}
\fi
