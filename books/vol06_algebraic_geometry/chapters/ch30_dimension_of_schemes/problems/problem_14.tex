\begin{problem}[VI.30.P14: Generically finite morphisms]\label{prob:vi30-14}
Under the hypotheses of Problem 13, assume \(K(X)\) is algebraic over \(K(Y)\). Prove \(\dim X=\dim Y\).

\textbf{Provenance.} Canonical function-field application.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Because \(K(X)\) is algebraic over the embedded copy of \(K(Y)\), adjoining elements of \(K(X)\) creates no
new transcendence over \(k\). Therefore
\[
\operatorname{trdeg}_kK(X)
=
\operatorname{trdeg}_kK(Y).
\]
The dimension theorem yields
\[
\dim X
=
\operatorname{trdeg}_kK(X)
=
\operatorname{trdeg}_kK(Y)
=
\dim Y.
\]
Hence
\[
\boxed{\dim X=\dim Y}.
\]
This is the dimension-theoretic content of generic finiteness.
\end{solution}
\fi
