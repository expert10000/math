\begin{problem}[VI.30.P13: Dominant morphism inequality]\label{prob:vi30-13}
If $f:X\to Y$ is dominant between varieties over $k$, prove $\dim X\geq\dim Y$.

\textbf{Provenance.} Canonical morphism consequence.
\end{problem}
\begin{solution}
Dominance induces an embedding $K(Y)\hookrightarrow K(X)$. Transcendence degree cannot decrease when passing to a larger field, so $\operatorname{trdeg}_kK(X)\geq\operatorname{trdeg}_kK(Y)$. Apply the dimension--transcendence-degree theorem.
\end{solution}
