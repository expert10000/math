\begin{problem}[VI.30.P13: Dominant morphism inequality]\label{prob:vi30-13}
If \(f:X\to Y\) is dominant between varieties over \(k\), prove \(\dim X\geq\dim Y\).

\textbf{Provenance.} Canonical morphism consequence.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Dominance sends the generic point of \(X\) to that of \(Y\) and induces an injective field homomorphism
\[
f^*:K(Y)\hookrightarrow K(X).
\]
Any algebraically independent subset of \(K(Y)\) remains algebraically independent after applying an injective
field map. Hence
\[
\operatorname{trdeg}_kK(Y)
\le
\operatorname{trdeg}_kK(X).
\]
Applying the dimension theorem to both varieties gives
\[
\boxed{\dim Y\le\dim X}.
\]
In particular, a variety of smaller dimension cannot dominate one of larger dimension.
\end{solution}
\fi
