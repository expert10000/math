\begin{problem}[VI.30.P6: Finite disjoint unions]\label{prob:vi30-06}
Prove $\dim(X\sqcup Y)=\max\{\dim X,\dim Y\}$.

\textbf{Provenance.} Canonical component computation.
\end{problem}
\begin{solution}
An irreducible subset of a disjoint union lies entirely in one open-and-closed component. Thus every irreducible-closed chain lies wholly in $X$ or wholly in $Y$, while chains from either component are chains in the union.
\end{solution}
