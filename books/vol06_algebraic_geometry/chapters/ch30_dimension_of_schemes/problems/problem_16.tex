\begin{problem}[VI.30.P16: An infinite-dimensional scheme]\label{prob:vi30-16}
Show \(X=\coprod_{n\ge0}\mathbb A^n_k\) has infinite dimension.

\textbf{Provenance.} Canonical example.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
By Problem~15,
\[
\dim X
=
\sup_{n\ge0}\dim\mathbb A^n_k.
\]
Since
\[
\dim\mathbb A^n_k=n,
\]
we obtain
\[
\dim X=\sup_{n\ge0}n=\boxed{\infty}.
\]
This is an important kind of infinite dimension: each component has finite dimension, but the component dimensions
are unbounded.
\end{solution}
\fi
