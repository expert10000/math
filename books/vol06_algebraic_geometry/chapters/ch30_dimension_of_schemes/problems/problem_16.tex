\begin{problem}[VI.30.P16: An infinite-dimensional scheme]\label{prob:vi30-16}
Show $X=\coprod_{n\ge0}\mathbb A^n_k$ has infinite dimension.

\textbf{Provenance.} Canonical example.
\end{problem}
\begin{solution}
The previous problem gives $\dim X=\sup_{n\ge0}\dim\mathbb A^n_k=\sup_n n=\infty$.
\end{solution}
