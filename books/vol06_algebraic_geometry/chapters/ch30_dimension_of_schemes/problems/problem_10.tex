\begin{problem}[VI.30.P10: Open subset inequality]\label{prob:vi30-10}
Prove that every open \(U\subseteq X\) satisfies \(\dim U\leq\dim X\).

\textbf{Provenance.} Canonical global theorem.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
Z_0\subsetneq Z_1\subsetneq\cdots\subsetneq Z_n
\]
be irreducible closed subsets of \(U\). Their closures
\[
\overline{Z_i}^{\,X}
\]
are irreducible closed subsets of \(X\). The inclusions remain strict. Indeed, if
\[
\overline{Z_i}^{\,X}=\overline{Z_{i+1}}^{\,X},
\]
then intersecting with \(U\) gives
\[
Z_i=U\cap\overline{Z_i}^{\,X}
=
U\cap\overline{Z_{i+1}}^{\,X}
=
Z_{i+1},
\]
contradiction. Therefore every length-\(n\) chain in \(U\) produces a length-\(n\) chain in \(X\), and
\[
\boxed{\dim U\le\dim X}.
\]
\end{solution}
\fi
