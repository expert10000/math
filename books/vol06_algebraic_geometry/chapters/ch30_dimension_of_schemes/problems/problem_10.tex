\begin{problem}[VI.30.P10: Open subset inequality]\label{prob:vi30-10}
Prove that every open $U\subseteq X$ satisfies $\dim U\leq\dim X$.

\textbf{Provenance.} Canonical global theorem.
\end{problem}
\begin{solution}
For an irreducible closed $Z\subseteq U$, its closure $\overline Z^X$ is irreducible in $X$, and $Z=U\cap\overline Z^X$. A strict chain in $U$ yields a strict chain of closures in $X$: if two closures were equal, intersecting with $U$ would make the corresponding members equal.
\end{solution}
