\begin{problem}[VI.30.P8: Reduction invariance]\label{prob:vi30-08}
Prove \(\dim X_{\mathrm{red}}=\dim X\).

\textbf{Provenance.} Canonical consequence of VI/10 and VI/29.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The reduction morphism
\[
X_{\mathrm{red}}\to X
\]
is a homeomorphism on underlying topological spaces. Consequently it induces a bijection between irreducible closed
subsets that preserves strict inclusion. Since global scheme dimension is defined entirely from the lengths of
such chains,
\[
\boxed{\dim X_{\mathrm{red}}=\dim X}.
\]
This equality is purely topological; it does not imply that the two schemes have the same structure sheaf.
\end{solution}
\fi
