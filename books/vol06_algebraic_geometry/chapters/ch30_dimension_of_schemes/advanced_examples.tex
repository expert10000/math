\begin{example}[Advanced 1: Noetherian integral does not suffice]\label{ex:vi30-adv-01}
The integral Noetherian scheme $X=\Spec k[x,y]_{(x)}$ has dimension $1$, while $D(x)$ has dimension $0$. Thus the AG8 open-dimension theorem needs its variety hypothesis.
\end{example}

\begin{example}[Advanced 2: Non-equidimensional reducible scheme]\label{ex:vi30-adv-02}
For $X=\mathbb A^2_k\sqcup\mathbb A^1_k$, global dimension is $2$, yet an open-and-closed component has dimension $1$.
\end{example}

\begin{example}[Advanced 3: Infinite-dimensional scheme]\label{ex:vi30-adv-03}
$\Spec k[x_1,x_2,\ldots]$ has infinite dimension through chains $(0)\subsetneq(x_1)\subsetneq\cdots\subsetneq(x_1,\ldots,x_n)$ of arbitrary length.
\end{example}

\begin{example}[Advanced 4: Arbitrary disjoint union]\label{ex:vi30-adv-04}
For $X=\coprod_i X_i$, every irreducible subset lies in one component, so $\dim X=\sup_i\dim X_i$.
\end{example}

\begin{example}[Advanced 5: Dominant morphisms of varieties]\label{ex:vi30-adv-05}
A dominant morphism $f:X\to Y$ of varieties induces $K(Y)\hookrightarrow K(X)$ and therefore $\dim X\ge\dim Y$.
\end{example}

\begin{example}[Advanced 6: Generically finite dominant morphism]\label{ex:vi30-adv-06}
If $K(X)$ is algebraic over $K(Y)$ for a dominant morphism of varieties, then the transcendence degrees agree and $\dim X=\dim Y$.
\end{example}

\begin{example}[Advanced 7: Finite surjective morphism]\label{ex:vi30-adv-07}
A finite surjective morphism of Noetherian schemes is integral and preserves dimension componentwise; in the affine case this is the integral-extension theorem.
\end{example}

\begin{example}[Advanced 8: Tensor-product warning]\label{ex:vi30-adv-08}
The formula $\dim(X\times_kY)=\dim X+\dim Y$ requires care when $X\times_kY$ is reducible; the clean function-field proof applies componentwise under suitable geometric-integrality hypotheses.
\end{example}

\begin{example}[Advanced 9: A dense open can lose dimension outside finite type]\label{ex:vi30-adv-09}
In $X=\Spec k[x,y]_{(x)}$, the principal open $D(x)$ is dense because $X$ is integral, but its dimension is $0<1$. Density alone does not imply dimension equality.
\end{example}

\begin{example}[Advanced 10: Nilpotent thickening]\label{ex:vi30-adv-10}
If $X'\to X$ is a nilpotent thickening, the underlying spaces coincide, hence $\dim X'=\dim X$ even though tangent spaces and local rings may change.
\end{example}

\begin{example}[Advanced 11: Component formula in Noetherian schemes]\label{ex:vi30-adv-11}
Since a Noetherian scheme has finitely many irreducible components and every irreducible closed subset lies in one, global dimension is the maximum of component dimensions.
\end{example}

\begin{example}[Advanced 12: Affine charts need not all have ambient dimension]\label{ex:vi30-adv-12}
For general schemes an affine open chart can have smaller dimension than $X$; the equality of all nonempty affine charts is a special strength of varieties.
\end{example}

\begin{example}[Advanced 13: Normalization of a curve]\label{ex:vi30-adv-13}
For an integral finite-type curve, normalization is finite in the usual excellent setting and leaves dimension $1$ unchanged while correcting nonnormal local rings.
\end{example}

\begin{example}[Advanced 14: Transcendence basis viewpoint]\label{ex:vi30-adv-14}
If $X$ is a variety and $t_1,\ldots,t_d$ is a transcendence basis of $K(X)/k$, then $d=\dim X$. The integer $d$ is independent of the chosen basis and of the chosen affine chart.
\end{example}
