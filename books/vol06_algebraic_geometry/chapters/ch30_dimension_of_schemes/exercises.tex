\begin{exercise}[VI.30.E1]\label{exer:vi30-01}
Show $\dim\Spec k=0$.
\end{exercise}

\begin{exercise}[VI.30.E2]\label{exer:vi30-02}
Write a length-two prime chain in $k[x,y]$.
\end{exercise}

\begin{exercise}[VI.30.E3]\label{exer:vi30-03}
Compute $\dim\mathbb A^4_k$.
\end{exercise}

\begin{exercise}[VI.30.E4]\label{exer:vi30-04}
If $Y\subseteq X$ is closed and $\dim X=2$, what can be said about $\dim Y$?
\end{exercise}

\begin{exercise}[VI.30.E5]\label{exer:vi30-05}
Prove a nilpotent thickening does not change dimension.
\end{exercise}

\begin{exercise}[VI.30.E6]\label{exer:vi30-06}
Compute the dimension of $\mathbb A^2_k\sqcup\mathbb A^5_k$.
\end{exercise}

\begin{exercise}[VI.30.E7]\label{exer:vi30-07}
Give a nonempty open subset of $\Spec k[t]$ with dimension $1$.
\end{exercise}

\begin{exercise}[VI.30.E8]\label{exer:vi30-08}
Explain why $D(x)$ in $\Spec k[x,y]_{(x)}$ has only one point.
\end{exercise}

\begin{exercise}[VI.30.E9]\label{exer:vi30-09}
Let $X$ be a variety with $\dim X=3$. What is the dimension of every nonempty open subset?
\end{exercise}

\begin{exercise}[VI.30.E10]\label{exer:vi30-10}
If $K(X)=k(s,t)$ for a variety $X$, compute $\dim X$.
\end{exercise}

\begin{exercise}[VI.30.E11]\label{exer:vi30-11}
If $K(X)$ is algebraic over $k$, compute the dimension of the variety $X$.
\end{exercise}

\begin{exercise}[VI.30.E12]\label{exer:vi30-12}
Prove a dominant morphism from a curve to a surface cannot exist between varieties.
\end{exercise}

\begin{exercise}[VI.30.E13]\label{exer:vi30-13}
If $f:X\to Y$ is dominant and generically finite between varieties and $\dim Y=4$, compute $\dim X$.
\end{exercise}

\begin{exercise}[VI.30.E14]\label{exer:vi30-14}
Compute $\dim(\coprod_{n=0}^{10}\mathbb A^n_k)$.
\end{exercise}

\begin{exercise}[VI.30.E15]\label{exer:vi30-15}
Compute $\dim(\coprod_{n\ge0}\Spec k)$.
\end{exercise}

\begin{exercise}[VI.30.E16]\label{exer:vi30-16}
Can a dense open subset of an integral Noetherian scheme have smaller dimension?
\end{exercise}

\begin{exercise}[VI.30.E17]\label{exer:vi30-17}
Let $X$ be a variety of dimension $d$. Compute $\dim(X\times_k\mathbb A^1_k)$ under geometric-integrality hypotheses.
\end{exercise}

\begin{exercise}[VI.30.E18]\label{exer:vi30-18}
Why does $\dim X=\dim X_{\mathrm{red}}$ not imply the schemes are isomorphic?
\end{exercise}

\begin{exercise}[VI.30.E19]\label{exer:vi30-19}
If $X^\nu\to X$ is normalization of an integral curve, compare dimensions.
\end{exercise}

\begin{exercise}[VI.30.E20]\label{exer:vi30-20}
For an affine finite morphism $\Spec B\to\Spec A$ with $B$ integral over $A$, compare dimensions.
\end{exercise}

\begin{exercise}[VI.30.E21]\label{exer:vi30-21}
Show that a closed point is a zero-dimensional irreducible closed subset.
\end{exercise}

\begin{exercise}[VI.30.E22]\label{exer:vi30-22}
Does $\dim Y<\dim X$ hold for every proper closed subset $Y\subsetneq X$?
\end{exercise}

\begin{exercise}[VI.30.E23]\label{exer:vi30-23}
Does $\dim U<\dim X$ hold for every proper open subset?
\end{exercise}

\begin{exercise}[VI.30.E24]\label{exer:vi30-24}
Find a scheme whose dimension is the supremum $\infty$ of finite component dimensions.
\end{exercise}

\begin{exercise}[VI.30.E25]\label{exer:vi30-25}
State the distinction between global dimension and local dimension at $x$.
\end{exercise}

\begin{exercise}[VI.30.E26]\label{exer:vi30-26}
State the distinction between dimension and codimension.
\end{exercise}
