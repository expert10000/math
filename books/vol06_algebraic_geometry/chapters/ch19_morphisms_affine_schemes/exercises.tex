\begin{exercise}[Direction reversal]\label{ex:vi19-01}
For a ring map $\varphi:A\to B$, state the induced map of spectra and explain why its direction is reversed.
\end{exercise}

\begin{exercise}[Contraction is prime]\label{ex:vi19-02}
Prove that $\varphi^{-1}(\mfp)$ is prime whenever $\mfp\in\Spec B$.
\end{exercise}

\begin{exercise}[Principal-open inverse image]\label{ex:vi19-03}
Show $\Spec(\varphi)^{-1}(D(a))=D(\varphi(a))$.
\end{exercise}

\begin{exercise}[Continuity]\label{ex:vi19-04}
Deduce that $\Spec(\varphi)$ is continuous.
\end{exercise}

\begin{exercise}[Localized section map]\label{ex:vi19-05}
Write explicitly the map $A_a\to B_{\varphi(a)}$ induced by $\varphi$.
\end{exercise}

\begin{exercise}[Stalk map]\label{ex:vi19-06}
For $\mfp\in\Spec B$ and $\mfq=\varphi^{-1}(\mfp)$, write the induced map $A_{\mfq}\to B_{\mfp}$.
\end{exercise}

\begin{exercise}[Locality of the stalk map]\label{ex:vi19-07}
Prove the stalk map in the previous exercise is local.
\end{exercise}

\begin{exercise}[Identity functoriality]\label{ex:vi19-08}
Show $\Spec(\operatorname{id}_A)=\operatorname{id}_{\Spec A}$ including the sheaf map.
\end{exercise}

\begin{exercise}[Composition]\label{ex:vi19-09}
For $A\xrightarrow{\varphi}B\xrightarrow{\psi}C$, prove $\Spec(\psi\varphi)=\Spec(\varphi)\circ\Spec(\psi)$.
\end{exercise}

\begin{exercise}[Polynomial projection]\label{ex:vi19-10}
Compute the map $\A^2_k\to\A^1_k$ induced by $k[t]\to k[x,y]$, $t\mapsto x$, on closed $k$-points.
\end{exercise}

\begin{exercise}[Squaring map]\label{ex:vi19-11}
For $k[t]\to k[x]$, $t\mapsto x^2$, compute the image of the closed point $(x-a)$.
\end{exercise}

\begin{exercise}[Localization image]\label{ex:vi19-12}
Show that $A\to A_f$ induces a spectrum map with image $D(f)$.
\end{exercise}

\begin{exercise}[Quotient image]\label{ex:vi19-13}
Show that $A\to A/I$ induces an injective map on spectra with image $V(I)$.
\end{exercise}

\begin{exercise}[Reduction map]\label{ex:vi19-14}
Explain why $\Spec A_{\mathrm{red}}\to\Spec A$ is a homeomorphism but may fail to be an affine-scheme isomorphism.
\end{exercise}

\begin{exercise}[Maps to affine line]\label{ex:vi19-15}
Show that ring maps $\mathbb Z[t]\to B$ are in bijection with elements of $B$.
\end{exercise}

\begin{exercise}[Maps to affine $n$-space]\label{ex:vi19-16}
For a $k$-algebra $B$, show
\[
\operatorname{Hom}_{k\text{-alg}}(k[x_1,\dots,x_n],B)\cong B^n.
\]
Interpret the bijection as evaluation on the coordinate generators.
\end{exercise}

\begin{exercise}[Isomorphism criterion]\label{ex:vi19-17}
Prove that a morphism $\Spec B\to\Spec A$ is an isomorphism iff its corresponding ring map $A\to B$ is an isomorphism.
\end{exercise}

\begin{exercise}[Residue fields]\label{ex:vi19-18}
Construct the induced field map $\kappa(\varphi^{-1}(\mfp))\to\kappa(\mfp)$.
\end{exercise}

\begin{exercise}[Dense image criterion]\label{ex:vi19-19}
Show that the closure of the image of $\Spec B\to\Spec A$ is $V(\ker\varphi)$.
\end{exercise}

\begin{exercise}[Dense but noninjective]\label{ex:vi19-20}
Give a noninjective ring map whose induced map on spectra has dense image.
\end{exercise}

\begin{exercise}[Factor through a quotient]\label{ex:vi19-21}
For $\alpha:A\to B$, prove that $\Spec B\to\Spec A$ factors through $\Spec(A/I)$ iff $I\subseteq\ker\alpha$.
\end{exercise}

\begin{exercise}[Factor through a principal open]\label{ex:vi19-22}
For $\alpha:A\to B$, prove $\Spec B\to\Spec A$ factors through $D(f)=\Spec A_f$ iff $\alpha(f)$ is a unit in $B$.
\end{exercise}

\begin{exercise}[Same continuous map, different scheme maps]\label{ex:vi19-23}
Find two different endomorphisms of $\Spec k[\varepsilon]/(\varepsilon^2)$ with the same underlying continuous map.
\end{exercise}

\begin{exercise}[Recover a morphism from global sections]\label{ex:vi19-24}
Let $f,g:\Spec B\to\Spec A$ induce the same map $A\to B$ on global sections.  Prove $f=g$.
\end{exercise}

\begin{exercise}[Universal property for a locally ringed source]\label{ex:vi19-25}
Let $X$ be a locally ringed space and $s\in\Gamma(X,\OO_X)$. Describe explicitly the morphism $X\to\Spec\mathbb Z[t]$ corresponding to $t\mapsto s$.
\end{exercise}

\begin{exercise}[Boundary check]\label{ex:vi19-26}
State which neighboring legacy problems remain reserved after VI/19 and where they go.
\end{exercise}
