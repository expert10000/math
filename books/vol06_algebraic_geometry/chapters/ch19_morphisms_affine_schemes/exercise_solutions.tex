\paragraph{Exercise VI.19.1 -- Direction reversal.}
\begin{solution}
Let \(\varphi:A\to B\) be a ring homomorphism.  A point of \(\Spec B\) is a prime ideal \(\mathfrak p\subset B\).  Its contraction
\[
\varphi^{-1}(\mathfrak p)=\{a\in A:\varphi(a)\in\mathfrak p\}
\]
is a prime ideal of \(A\), so the induced point map is
\[
\Spec(\varphi):\Spec B\longrightarrow\Spec A,
\qquad
\mathfrak p\longmapsto\varphi^{-1}(\mathfrak p).
\]
The arrow reverses because prime ideals live in the codomain ring \(B\), while contraction carries them back to the domain ring \(A\).  Thus a ring arrow \(A\to B\) gives a geometric arrow \(\Spec B\to\Spec A\).
\end{solution}

\paragraph{Exercise VI.19.2 -- Contraction is prime.}
\begin{solution}
Put \(\mathfrak q=\varphi^{-1}(\mathfrak p)\).  Since \(\varphi(1)=1\notin\mathfrak p\), the ideal \(\mathfrak q\) is proper.  If \(ab\in\mathfrak q\), then
\[
\varphi(a)\varphi(b)=\varphi(ab)\in\mathfrak p.
\]
Because \(\mathfrak p\) is prime, either \(\varphi(a)\in\mathfrak p\) or \(\varphi(b)\in\mathfrak p\).  Hence either \(a\in\mathfrak q\) or \(b\in\mathfrak q\).  Therefore \(\mathfrak q\) is prime.
\end{solution}

\paragraph{Exercise VI.19.3 -- Principal-open inverse image.}
\begin{solution}
Let \(f=\Spec(\varphi)\).  For \(\mathfrak p\in\Spec B\),
\[
\begin{aligned}
\mathfrak p\in f^{-1}(D(a))
&\iff f(\mathfrak p)=\varphi^{-1}(\mathfrak p)\in D(a)\\
&\iff a\notin\varphi^{-1}(\mathfrak p)\\
&\iff \varphi(a)\notin\mathfrak p\\
&\iff \mathfrak p\in D(\varphi(a)).
\end{aligned}
\]
Thus
\[
\boxed{f^{-1}(D(a))=D(\varphi(a)).}
\]
This identity is the basic topological formula for affine morphisms.
\end{solution}

\paragraph{Exercise VI.19.4 -- Continuity.}
\begin{solution}
The distinguished opens \(D(a)\) form a basis for the Zariski topology on \(\Spec A\).  By Exercise VI.19.3, the inverse image of every such basis open is the distinguished open \(D(\varphi(a))\subseteq\Spec B\), hence is open.  A map whose inverse images of basis opens are open is continuous.  Therefore \(\Spec(\varphi)\) is continuous.
\end{solution}

\paragraph{Exercise VI.19.5 -- Localized section map.}
\begin{solution}
On \(D(a)\subseteq\Spec A\) the structure sheaf has section ring \(A_a\).  Its inverse image is \(D(\varphi(a))\), whose section ring is \(B_{\varphi(a)}\).  Since \(\varphi(a)\) is invertible in this localization, the universal property of localization gives a unique map
\[
\varphi_a:A_a\to B_{\varphi(a)},
\qquad
\frac r{a^n}\longmapsto \frac{\varphi(r)}{\varphi(a)^n}.
\]
Well-definedness is automatic from the localization universal property; directly, any relation annihilated by a power of \(a\) maps to a relation annihilated by the corresponding power of \(\varphi(a)\).
\end{solution}

\paragraph{Exercise VI.19.6 -- Stalk map.}
\begin{solution}
Fix \(\mathfrak p\in\Spec B\) and set \(\mathfrak q=\varphi^{-1}(\mathfrak p)\).  If \(s\notin\mathfrak q\), then \(\varphi(s)\notin\mathfrak p\), so \(\varphi(s)\) becomes invertible in \(B_{\mathfrak p}\).  Hence \(\varphi\) localizes to
\[
A_{\mathfrak q}\longrightarrow B_{\mathfrak p},
\qquad
\frac as\longmapsto\frac{\varphi(a)}{\varphi(s)}.
\]
This is exactly the stalk homomorphism induced by the sheaf map, since stalks are direct limits of sections over distinguished neighborhoods.
\end{solution}

\paragraph{Exercise VI.19.7 -- Locality of the stalk map.}
\begin{solution}
The maximal ideals of the two local rings are \(\mathfrak qA_{\mathfrak q}\) and \(\mathfrak pB_{\mathfrak p}\).  For \(a/s\in A_{\mathfrak q}\), its image lies in \(\mathfrak pB_{\mathfrak p}\) exactly when \(\varphi(a)\in\mathfrak p\), because \(\varphi(s)\notin\mathfrak p\) is a unit.  This is equivalent to \(a\in\varphi^{-1}(\mathfrak p)=\mathfrak q\).  Therefore
\[
(\varphi_{\mathfrak p})^{-1}(\mathfrak pB_{\mathfrak p})
=\mathfrak qA_{\mathfrak q},
\]
so the stalk map is local.
\end{solution}

\paragraph{Exercise VI.19.8 -- Identity functoriality.}
\begin{solution}
For the identity ring homomorphism \(\operatorname{id}_A\), contraction fixes every prime ideal, so the underlying point map is \(\operatorname{id}_{\Spec A}\).  On a distinguished open \(D(a)\), the induced localized map is
\[
A_a\longrightarrow A_a,\qquad r/a^n\longmapsto r/a^n,
\]
which is the identity.  These identity maps are compatible with all restrictions, so the induced sheaf map is also the identity.  Hence
\[
\Spec(\operatorname{id}_A)=\operatorname{id}_{\Spec A}
\]
as a morphism of locally ringed spaces.
\end{solution}

\paragraph{Exercise VI.19.9 -- Composition.}
\begin{solution}
Let \(A\xrightarrow{\varphi}B\xrightarrow{\psi}C\).  For \(\mathfrak p\in\Spec C\),
\[
(\psi\circ\varphi)^{-1}(\mathfrak p)
=\varphi^{-1}(\psi^{-1}(\mathfrak p)),
\]
which proves equality of the continuous maps.  On \(D(a)\subseteq\Spec A\), the direct localized map sends
\[
\frac r{a^n}\longmapsto
\frac{\psi(\varphi(r))}{\psi(\varphi(a))^n}.
\]
The two-step pullback through \(B_{\varphi(a)}\) gives the same fraction.  Thus the sheaf maps also agree and
\[
\Spec(\psi\circ\varphi)=\Spec(\varphi)\circ\Spec(\psi).
\]
\end{solution}

\paragraph{Exercise VI.19.10 -- Polynomial projection.}
\begin{solution}
The ring map \(k[t]\to k[x,y]\) is defined by \(t\mapsto x\).  A closed \(k\)-point \((a,b)\) of \(\mathbb A_k^2\) corresponds to the maximal ideal \((x-a,y-b)\).  Its contraction consists of those \(p(t)\) for which \(p(x)\in(x-a,y-b)\), equivalently \(p(a)=0\).  Thus the contraction is \((t-a)\), and the induced map on closed points is
\[
(a,b)\longmapsto a.
\]
So the contravariant coordinate pullback \(t\mapsto x\) is the usual first-coordinate projection geometrically.
\end{solution}

\paragraph{Exercise VI.19.11 -- Squaring map.}
\begin{solution}
For \(k[t]\to k[x]\), \(t\mapsto x^2\), the point \((x-a)\) maps to its contraction.  A polynomial \(p(t)\) lies in the contraction precisely when \(p(x^2)\in(x-a)\), i.e. when \(p(a^2)=0\).  Therefore the contraction is \((t-a^2)\), and the corresponding closed point is \(a^2\).  Thus the induced morphism on \(k\)-points is the squaring map \(a\mapsto a^2\).
\end{solution}

\paragraph{Exercise VI.19.12 -- Localization image.}
\begin{solution}
For the localization map \(A\to A_f\), primes of \(A_f\) correspond by contraction to prime ideals \(\mathfrak p\subset A\) disjoint from \(\{1,f,f^2,\ldots\}\).  This is equivalent to \(f\notin\mathfrak p\), so the image is exactly \(D(f)\).  Extension \(\mathfrak p\mapsto\mathfrak pA_f\) is inverse to contraction on these primes.  Hence the spectrum map is a bijection onto \(D(f)\); VI/18 strengthens this to an isomorphism of locally ringed spaces.
\end{solution}

\paragraph{Exercise VI.19.13 -- Quotient image.}
\begin{solution}
Every prime ideal of \(A/I\) has the form \(\mathfrak p/I\) for a unique prime \(\mathfrak p\subset A\) containing \(I\).  Under contraction along \(A\to A/I\), the prime \(\mathfrak p/I\) maps to \(\mathfrak p\).  Thus the spectrum map is injective and its image is
\[
\{\mathfrak p\in\Spec A:I\subseteq\mathfrak p\}=V(I).
\]
The correspondence also identifies the subspace topology on \(V(I)\) with the Zariski topology of \(\Spec(A/I)\).
\end{solution}

\paragraph{Exercise VI.19.14 -- Reduction map.}
\begin{solution}
Let \(N=\sqrt{(0)}\).  Every prime ideal of \(A\) contains \(N\), so prime correspondence for the quotient gives a bijective homeomorphism
\[
\Spec(A/N)\longrightarrow\Spec A.
\]
However, the induced map of structure sheaves can lose nilpotent sections.  On global sections it is the quotient homomorphism \(A\to A/N\), which need not be an isomorphism.  Since affine-scheme isomorphisms correspond to ring isomorphisms, the spectral homeomorphism need not be an isomorphism of affine schemes.
\end{solution}

\paragraph{Exercise VI.19.15 -- Maps to affine line.}
\begin{solution}
A unital ring homomorphism \(\mathbb Z[t]\to B\) must restrict on \(\mathbb Z\) to the canonical map, so it is determined entirely by the element \(b=\varphi(t)\).  Conversely, for each \(b\in B\), polynomial evaluation
\[
\sum_i n_i t^i\longmapsto\sum_i n_i b^i
\]
defines a unique unital ring homomorphism.  Hence
\[
\operatorname{Hom}_{\mathrm{Ring}}(\mathbb Z[t],B)\cong B.
\]
Under affine anti-equivalence, this identifies morphisms \(\Spec B\to\Spec\mathbb Z[t]\) with global sections of \(\Spec B\).
\end{solution}

\paragraph{Exercise VI.19.16 -- Maps to affine \(n\)-space.}
\begin{solution}
A \(k\)-algebra homomorphism
\[
\varphi:k[x_1,\ldots,x_n]\to B
\]
is uniquely determined by the tuple \((\varphi(x_1),\ldots,\varphi(x_n))\in B^n\), because every polynomial is built from the generators and coefficients in \(k\).  Conversely, any tuple \((b_1,\ldots,b_n)\) defines the evaluation homomorphism \(x_i\mapsto b_i\).  These assignments are inverse, giving
\[
\operatorname{Hom}_{k\text{-alg}}(k[x_1,\ldots,x_n],B)\cong B^n.
\]
Geometrically the entries are the pullbacks of the coordinate functions on \(\mathbb A_k^n\).
\end{solution}

\paragraph{Exercise VI.19.17 -- Isomorphism criterion.}
\begin{solution}
If \(\varphi:A\to B\) is a ring isomorphism with inverse \(\psi\), contravariant functoriality gives
\[
\Spec(\varphi)\circ\Spec(\psi)=\operatorname{id}_{\Spec B},
\qquad
\Spec(\psi)\circ\Spec(\varphi)=\operatorname{id}_{\Spec A}.
\]
Thus \(\Spec(\varphi)\) is an isomorphism.  Conversely, if \(f:\Spec B\to\Spec A\) is an affine-scheme isomorphism, applying global sections to \(f\) and its inverse gives mutually inverse ring maps \(A\leftrightarrows B\).  Therefore the corresponding ring map is an isomorphism.
\end{solution}

\paragraph{Exercise VI.19.18 -- Residue fields.}
\begin{solution}
Let \(\mathfrak p\in\Spec B\) and \(\mathfrak q=\varphi^{-1}(\mathfrak p)\).  The affine morphism gives a local homomorphism
\[
A_{\mathfrak q}\longrightarrow B_{\mathfrak p}.
\]
Locality implies that \(\mathfrak qA_{\mathfrak q}\) maps into \(\mathfrak pB_{\mathfrak p}\), so passing to quotients gives a unital field homomorphism
\[
\kappa(\mathfrak q)
=A_{\mathfrak q}/\mathfrak qA_{\mathfrak q}
\longrightarrow
B_{\mathfrak p}/\mathfrak pB_{\mathfrak p}
=\kappa(\mathfrak p).
\]
Because it is unital and the source is a field, this map is automatically injective.
\end{solution}

\paragraph{Exercise VI.19.19 -- Dense image criterion.}
\begin{solution}
Let \(f=\Spec(\varphi)\).  The ideal of all functions lying in every image prime is
\[
\bigcap_{\mathfrak p\in\Spec B}\varphi^{-1}(\mathfrak p)
=\varphi^{-1}\!\left(\sqrt{(0)_B}\right).
\]
An element \(a\in A\) belongs to this ideal iff \(\varphi(a)\) is nilpotent, iff \(a^n\in\ker\varphi\) for some \(n\), so the ideal is \(\sqrt{\ker\varphi}\).  Therefore the smallest closed subset containing the image is
\[
V(\sqrt{\ker\varphi})=V(\ker\varphi).
\]
In particular, the image is dense iff \(\ker\varphi\subseteq\sqrt{(0)_A}\).
\end{solution}

\paragraph{Exercise VI.19.20 -- Dense but noninjective.}
\begin{solution}
Take
\[
A=k[\varepsilon]/(\varepsilon^2),\qquad B=k,
\]
and let \(\varphi:A\to B\) send \(\varepsilon\mapsto0\).  The kernel \((\varepsilon)\) is nonzero, so \(\varphi\) is not injective.  But \((\varepsilon)\) is nilpotent, hence contained in \(\sqrt{(0)_A}\).  By Exercise VI.19.19, the induced map \(\Spec k\to\Spec A\) has dense image.  In fact both spectra have a single point, so the point map is surjective.
\end{solution}

\paragraph{Exercise VI.19.21 -- Factor through a quotient.}
\begin{solution}
A factorization
\[
\Spec B\longrightarrow\Spec(A/I)\longrightarrow\Spec A
\]
corresponds, under the affine Hom correspondence, to a reversed factorization
\[
A\longrightarrow A/I\longrightarrow B
\]
of \(\alpha:A\to B\).  By the universal property of the quotient, a map \(A/I\to B\) with composite \(\alpha\) exists exactly when every element of \(I\) maps to zero, i.e. when \(I\subseteq\ker\alpha\).  When it exists it is unique, so the geometric factorization is unique as well.
\end{solution}

\paragraph{Exercise VI.19.22 -- Factor through a principal open.}
\begin{solution}
Using \(D(f)\cong\Spec A_f\), a geometric factorization through \(D(f)\) is equivalent to a ring factorization
\[
A\longrightarrow A_f\longrightarrow B.
\]
The universal property of localization says that such a map \(A_f\to B\) exists uniquely exactly when \(\alpha(f)\) is a unit in \(B\).  Therefore \(\Spec B\to\Spec A\) factors through \(D(f)\) precisely when the pullback of \(f\) is invertible everywhere on \(\Spec B\).
\end{solution}

\paragraph{Exercise VI.19.23 -- Same continuous map, different scheme maps.}
\begin{solution}
Let \(A=k[\varepsilon]/(\varepsilon^2)\).  Consider the two \(k\)-algebra endomorphisms
\[
\varphi_1(\varepsilon)=\varepsilon,
\qquad
\varphi_0(\varepsilon)=0.
\]
The spectrum of \(A\) has only the prime \((\varepsilon)\), so both induced continuous self-maps are the identity on the one-point space.  Nevertheless the associated affine-scheme morphisms are distinct because their pullbacks on global sections are \(\varphi_1\) and \(\varphi_0\), which differ on \(\varepsilon\).  Thus the sheaf component of a scheme morphism contains data not visible topologically.
\end{solution}

\paragraph{Exercise VI.19.24 -- Recover a morphism from global sections.}
\begin{solution}
The affine Hom correspondence is a bijection
\[
\operatorname{Hom}_{\mathrm{AffSch}}(\Spec B,\Spec A)
\cong
\operatorname{Hom}_{\mathrm{Ring}}(A,B).
\]
If \(f\) and \(g\) induce the same pullback \(A\to B\), they correspond to the same element on the right-hand side and hence must be equal.  Concretely, the common global map determines the point map by contraction and determines the sheaf map on every distinguished open by localization, so no further choices remain.
\end{solution}

\paragraph{Exercise VI.19.25 -- Universal property for a locally ringed source.}
\begin{solution}
Let \(s\in\Gamma(X,\OO_X)\) and let \(\alpha:\mathbb Z[t]\to\Gamma(X,\OO_X)\) send \(t\mapsto s\).  At \(x\in X\), define
\[
f(x)=\{p(t)\in\mathbb Z[t]:p(s)_x\in\mathfrak m_x\}.
\]
This is a prime ideal.  For a polynomial \(p\), the inverse image \(f^{-1}(D(p))\) is the open set where the germ \(p(s)_x\) is a unit.  On that open, localization gives
\[
\mathbb Z[t]_p\longrightarrow\Gamma(f^{-1}D(p),\OO_X),
\qquad
\frac{q}{p^n}\longmapsto q(s)p(s)^{-n}.
\]
These maps glue to the unique locally-ringed-space morphism corresponding to the global section \(s\).
\end{solution}
