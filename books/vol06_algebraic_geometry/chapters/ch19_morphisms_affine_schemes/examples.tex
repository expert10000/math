\begin{example}[Identity map]
The identity ring map $A\to A$ induces $\operatorname{id}_{\Spec A}$.  Both the point map and all localized section maps are identities.
\end{example}

\begin{example}[A polynomial projection]
The inclusion $k[t]\hookrightarrow k[x,y]$, $t\mapsto x$, induces $\A^2_k\to\A^1_k$.  On closed $k$-points it is $(a,b)\mapsto a$.
\end{example}

\begin{example}[The squaring map on the affine line]
The homomorphism $k[t]\to k[x]$, $t\mapsto x^2$, induces $\A^1_k\to\A^1_k$.  Over an algebraically closed field it sends the closed point $(x-a)$ to $(t-a^2)$.
\end{example}

\begin{example}[A translation]
The automorphism $k[t]\to k[t]$, $t\mapsto t+c$, induces an automorphism of $\A^1_k$.  On a closed point $(t-a)$, contraction gives $(t-(a+c))$, so the corresponding geometric point moves by $a\mapsto a+c$.
\end{example}

\begin{example}[A quotient map]
For $A\to A/I$, the induced map $\Spec(A/I)\to\Spec A$ sends $\mfp/I$ to $\mfp$ and has image $V(I)$.
\end{example}

\begin{example}[A localization map]
The localization $A\to A_f$ induces $\Spec A_f\to\Spec A$ with image exactly $D(f)$; VI/18 identifies it with the inclusion of that open affine chart.
\end{example}

\begin{example}[An arithmetic map]
The inclusion $\mathbb Z\to\mathbb Z[i]$ induces $\Spec\mathbb Z[i]\to\Spec\mathbb Z$.  A prime of $\mathbb Z[i]$ is sent to its contraction in $\mathbb Z$.  The general structure-morphism viewpoint is reserved for VI/21.
\end{example}

\begin{example}[Dual numbers to a reduced point]
The quotient $k[\varepsilon]/(\varepsilon^2)\to k$ induces $\Spec k\to\Spec k[\varepsilon]/(\varepsilon^2)$.  Both spaces have one point, but the morphism is not an isomorphism because the ring map is not.
\end{example}

\begin{example}[Reduction map]
The quotient $A\to A_{\mathrm{red}}=A/\sqrt{(0)}$ induces a morphism $\Spec A_{\mathrm{red}}\to\Spec A$ that is a homeomorphism on underlying spaces, but need not be an isomorphism of affine schemes.
\end{example}

\begin{example}[Projection from a polynomial extension]
The inclusion $A\to A[t]$ induces $\Spec A[t]\to\Spec A$.  Fibers of this map are postponed to VI/25, but contractions of primes can already be computed.
\end{example}

\begin{example}[Evaluation at a scalar]
For $a\in k$, evaluation $k[t]\to k$, $t\mapsto a$, induces the map $\Spec k\to\A^1_k$ whose unique point lands at the closed point $(t-a)$.
\end{example}

\begin{example}[A map to affine two-space]
A pair $b_1,b_2\in B$ determines $k[x,y]\to B$ by $x\mapsto b_1$, $y\mapsto b_2$, hence a morphism $\Spec B\to\A^2_k$.
\end{example}

\begin{example}[Pulling back a principal open]
For $\varphi:A\to B$ and $a\in A$, the inverse image of $D(a)$ is $D(\varphi(a))$.  If $\varphi(a)$ is a unit, the entire source maps into $D(a)$.
\end{example}

\begin{example}[Pulling back a global function]
A morphism $f:\Spec B\to\Spec A$ corresponding to $\varphi:A\to B$ sends the global section $a\in A$ to $\varphi(a)\in B$.
\end{example}

\begin{example}[Residue field at an evaluation point]
For $k[t]\to k$, $t\mapsto a$, the induced point has residue-field map $k\to k$, the identity when the homomorphism fixes $k$.
\end{example}

\begin{example}[A noninjective ring map with dense spectrum map]
If $A=k[\varepsilon]/(\varepsilon^2)\to k$ kills $\varepsilon$, then the induced map has dense (indeed full) one-point image because the kernel is nilpotent.
\end{example}

\begin{example}[A surjective map has closed image]
If $A\twoheadrightarrow B$ has kernel $I$, then $\Spec B\to\Spec A$ has image $V(I)$, a closed subset.
\end{example}

\begin{example}[An injective map need not be surjective on spectra]
The inclusion $\mathbb Z\hookrightarrow\mathbb Q$ induces $\Spec\mathbb Q\to\Spec\mathbb Z$ whose image is only the generic point $(0)$.
\end{example}

\begin{example}[Composition]
If $A\to B\to C$, then the morphism $\Spec C\to\Spec A$ equals the composite $\Spec C\to\Spec B\to\Spec A$.
\end{example}

\begin{example}[Two morphisms with the same underlying point map]
On $A=k[\varepsilon]/(\varepsilon^2)$, the identity and the endomorphism $\varepsilon\mapsto0$ induce the same map of the one-point topological space $\Spec A$, but different morphisms of affine schemes.
\end{example}
