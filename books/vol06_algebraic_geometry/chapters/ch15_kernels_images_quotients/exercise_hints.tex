\subsection*{Hints for the exercises}
\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 1]
Use naturality of $\varphi$ with respect to restriction.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 2]
Glue in $\mathcal F$, then use locality in $\mathcal G$.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 3]
Send a germ represented by a kernel section to the same germ in $\mathcal F_x$.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 4]
Restrict a representative $\varphi_U(f)$.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 5]
Intersect a local lifting neighborhood with the smaller open.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 6]
First glue in $\mathcal G$; local liftability is checked inside one cover member.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 7]
Translate membership in each side into existence of a local representative lift.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 8]
Check that changing a representative by a section of $\mathcal F'$ remains so after restriction.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 9]
Map $[U,[s]]_x$ to $[s_x]$ and use shrinking for injectivity.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 10]
Combine sheafification-preserves-stalks with the ordinary cokernel calculation.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 11]
Check equality on each stalk, or prove both inclusions locally.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 12]
Compare stalks of both sheaves.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 13]
Work objectwise; no sheafification subtlety remains.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 14]
Use the stalk map $\mathbb Z\xrightarrow{n}\mathbb Z$.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 15]
Only the stalk at the support point is nonzero.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 16]
The projection is split by the inclusion of the second summand.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 17]
Use $(a,b)\mapsto b-a$.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 18]
Restriction commutes with germs and with sheafification.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 19]
This is exactly the definition of the local image sheaf.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 20]
Use $(\im\varphi)_x=\im\varphi_x$.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 21]
Factor $\varphi$ through the quotient by its kernel and then into the image.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 22]
Take the stalk at every point and apply the first isomorphism theorem for groups.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 23]
Every point of $S^1$ has an arc on which an argument can be chosen continuously.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 24]
A lift would make the degree-one map null-homotopic, or compare endpoint values after lifting a loop.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 25]
Use local coordinates and integrate a smooth coefficient function.
\end{hint}

\begin{hint}
% PEDAGOGY-ENRICHED-VI
[Exercise 26]
Multiplication descends because $\mathcal I(U)$ is an ideal; identify stalks using the quotient theorem.
\end{hint}
