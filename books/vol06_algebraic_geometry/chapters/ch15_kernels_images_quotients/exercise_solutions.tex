\begin{solution}[Exercise 1]
If $V\subseteq U$ and $s\in\ker\varphi_U$, then $\varphi_V(s|_V)=\varphi_U(s)|_V=0$, so restrictions land in the kernels; identity and composition are inherited from $\mathcal F$.
\end{solution}

\begin{solution}[Exercise 2]
Compatible $s_i\in\ker\varphi(U_i)$ glue to $s\in\mathcal F(U)$. Since $\varphi(s)$ restricts to zero on every $U_i$, locality in $\mathcal G$ gives $\varphi(s)=0$. Uniqueness comes from $\mathcal F$.
\end{solution}

\begin{solution}[Exercise 3]
The inclusion gives $(\ker\varphi)_x\to\mathcal F_x$ with image in $\ker\varphi_x$. If $s_x\in\ker\varphi_x$, then $\varphi(s)_x=0$, so after shrinking $\varphi(s)=0$ and $s_x$ is represented by a kernel section. This proves bijectivity onto $\ker\varphi_x$.
\end{solution}

\begin{solution}[Exercise 4]
If $g=\varphi_U(f)$ and $V\subseteq U$, then $g|_V=\varphi_V(f|_V)$. Hence images are stable under restriction and form a subpresheaf.
\end{solution}

\begin{solution}[Exercise 5]
For $g\in(\im\varphi)(U)$ and $W\subseteq U$, a local lift near $x\in W$ on $V\subseteq U$ restricts to a lift on $V\cap W$.
\end{solution}

\begin{solution}[Exercise 6]
Compatible $g_i$ glue uniquely to $g\in\mathcal G(U)$. For $x\in U$, choose $i$ with $x\in U_i$ and a neighborhood $V\subseteq U_i$ on which $g_i=\varphi(f)$. Then $g|_V=g_i|_V=\varphi(f)$, proving local liftability.
\end{solution}

\begin{solution}[Exercise 7]
A germ in $(\im\varphi)_x$ is represented by a locally liftable section, hence after shrinking by $\varphi(f)$, so it lies in $\im\varphi_x$. Conversely a germ $\varphi_x(f_x)$ is represented after shrinking by an actual image section.
\end{solution}

\begin{solution}[Exercise 8]
For $V\subseteq U$ define $[s]\mapsto[s|_V]$. If $s-s'\in\mathcal F'(U)$, then their restrictions differ by a section of $\mathcal F'(V)$, so the map is well defined; presheaf identities are immediate.
\end{solution}

\begin{solution}[Exercise 9]
Define $[U,[s]]_x\mapsto[s_x]$. Surjectivity is immediate. If $[s_x]=0$, then $s_x$ is represented by a germ from $\mathcal F'_x$; after shrinking, $s$ differs from an $\mathcal F'$-section by zero, so its quotient germ is zero. The same argument handles equality of two classes.
\end{solution}

\begin{solution}[Exercise 10]
The cokernel sheaf is the sheafification of the cokernel presheaf, which does not change stalks. The stalk of the cokernel presheaf is the cokernel of $\varphi_x$ by the same local representative argument as for quotients.
\end{solution}

\begin{solution}[Exercise 11]
Certainly $\mathcal F'\subseteq\ker q$. On stalks, $q_x:\mathcal F_x\to\mathcal F_x/\mathcal F'_x$ has kernel $\mathcal F'_x$. Thus the two subsheaves have equal stalks and hence are equal.
\end{solution}

\begin{solution}[Exercise 12]
The stalk of $\mathcal G/\im\varphi$ is $\mathcal G_x/\im\varphi_x$, which is $\coker\varphi_x$, the stalk of $\coker\varphi$. The natural comparison is therefore a stalkwise isomorphism.
\end{solution}

\begin{solution}[Exercise 13]
For $0$, kernel $=\mathcal F$, image $=0$, coimage $=0$, cokernel $=\mathcal G$. For $\id_\mathcal F$, kernel $=0$, image $=\mathcal F$, coimage $=\mathcal F$, cokernel $=0$.
\end{solution}

\begin{solution}[Exercise 14]
At every stalk the map is multiplication by $n$ on $\mathbb Z$. For $n\neq0$, kernel is zero, image is $n\underline{\mathbb Z}$, coimage is $\underline{\mathbb Z}$, and cokernel is $\underline{\mathbb Z/n\mathbb Z}$.
\end{solution}

\begin{solution}[Exercise 15]
At the support point take the ordinary kernel, image, and cokernel of $A\to B$; away from it every stalk is zero. These stalk descriptions determine the corresponding skyscraper sheaves.
\end{solution}

\begin{solution}[Exercise 16]
The projection $(f,g)\mapsto g$ has kernel $\mathcal F\oplus0\cong\mathcal F$ and is sectionwise surjective, so its cokernel is zero.
\end{solution}

\begin{solution}[Exercise 17]
The map $d:\mathcal F\oplus\mathcal F\to\mathcal F$, $d(a,b)=b-a$, is surjective and has kernel the diagonal. Hence it induces $(\mathcal F\oplus\mathcal F)/\Delta\mathcal F\cong\mathcal F$.
\end{solution}

\begin{solution}[Exercise 18]
At every point $x\in U$, both sides have the same stalk: for the image it is $\im\varphi_x$, and for the quotient it is $\mathcal F_x/\mathcal F'_x$. The natural comparison maps are therefore isomorphisms.
\end{solution}

\begin{solution}[Exercise 19]
If the image sheaf equals $\mathcal G$, every $g\in\mathcal G(U)$ is by definition locally in the image. Conversely, if all sections are locally liftable, then $(\im\varphi)(U)=\mathcal G(U)$ for every $U$.
\end{solution}

\begin{solution}[Exercise 20]
Using $(\im\varphi)_x=\im\varphi_x$, equality $\im\varphi=\mathcal G$ is equivalent to $\im\varphi_x=\mathcal G_x$ for all $x$, which is stalkwise surjectivity.
\end{solution}

\begin{solution}[Exercise 21]
Because $\varphi$ kills $\ker\varphi$, the quotient universal property gives $\mathcal F/\ker\varphi\to\mathcal G$. Its image lands in $\im\varphi$, producing the desired canonical map.
\end{solution}

\begin{solution}[Exercise 22]
At $x$ the canonical map is $\mathcal F_x/\ker\varphi_x\to\im\varphi_x$, $[a]\mapsto\varphi_x(a)$, an isomorphism by the first isomorphism theorem. Hence the sheaf map is an isomorphism.
\end{solution}

\begin{solution}[Exercise 23]
Choose an open arc $V\subseteq S^1$ small enough to avoid one point of the circle. The covering map $\mathbb R\to S^1$ admits a continuous local inverse on $V$. Composing a circle-valued function with this inverse gives a local argument and hence a local lift.
\end{solution}

\begin{solution}[Exercise 24]
If $F:S^1\to\mathbb R$ lifted the identity, then $e^{2\pi iF(z)}=z$. Parametrize $z=e^{2\pi it}$. A lift on $[0,1]$ must satisfy $F(e^{2\pi it})=t+k$ for a fixed integer $k$, so the values at $t=0$ and $t=1$ would differ by $1$ although they correspond to the same point of $S^1$, contradiction.
\end{solution}

\begin{solution}[Exercise 25]
In a coordinate interval a one-form is $a(t)\,dt$. Fix $t_0$ and set $F(t)=\int_{t_0}^t a(u)\,du$. Then $dF=a(t)\,dt$. Thus every germ of a one-form has a primitive and the image sheaf is all of $\Omega^1$.
\end{solution}

\begin{solution}[Exercise 26]
Define multiplication by $[a][b]=[ab]$ on sectionwise quotients. Because $\mathcal I(U)$ is an ideal this is well defined and survives sheafification. The quotient theorem gives $(\mathcal O/\mathcal I)_x\cong\mathcal O_x/\mathcal I_x$ as rings.
\end{solution}
