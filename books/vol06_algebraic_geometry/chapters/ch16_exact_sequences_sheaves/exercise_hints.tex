\begin{hint}[Exercise VI.16.1]
Translate membership in the sheaf image into existence of local preimages.
\end{hint}

\begin{hint}[Exercise VI.16.2]
Use $\im f\subseteq\ker g$.
\end{hint}

\begin{hint}[Exercise VI.16.3]
Take stalks of the equality and use the VI/15 kernel/image stalk formulas.
\end{hint}

\begin{hint}[Exercise VI.16.4]
Start with a kernel section and lift its germ at each point; then shrink.
\end{hint}

\begin{hint}[Exercise VI.16.5]
Use $(\ker f)_x=\ker(f_x)$ and that a sheaf is zero iff all stalks are zero.
\end{hint}

\begin{hint}[Exercise VI.16.6]
A section in the kernel of $f_U$ has zero germ at every point.
\end{hint}

\begin{hint}[Exercise VI.16.7]
Use $\im f=\G$ iff $(\im f)_x=\G_x$.
\end{hint}

\begin{hint}[Exercise VI.16.8]
Represent a preimage germ by a local section and shrink until equality of germs is equality of sections.
\end{hint}

\begin{hint}[Exercise VI.16.9]
Combine stalkwise injectivity and surjectivity, then use the stalkwise isomorphism criterion.
\end{hint}

\begin{hint}[Exercise VI.16.10]
Apply the stalkwise exactness theorem to all three nonzero terms.
\end{hint}

\begin{hint}[Exercise VI.16.11]
At every stalk use the first isomorphism theorem and then detect isomorphisms stalkwise.
\end{hint}

\begin{hint}[Exercise VI.16.12]
For $x\in U$, restriction does not change the stalk at $x$.
\end{hint}

\begin{hint}[Exercise VI.16.13]
Every point lies in some member of the cover, so every stalk sees an exact restricted sequence.
\end{hint}

\begin{hint}[Exercise VI.16.14]
Kernels are computed sectionwise; exactness at $\F(U)$ is therefore global.
\end{hint}

\begin{hint}[Exercise VI.16.15]
Use the degree-one map $z\mapsto z$ on $S^1$ and winding number.
\end{hint}

\begin{hint}[Exercise VI.16.16]
$e^{2\pi i f}=1$ exactly when $f$ is integer-valued; continuity makes it locally constant.
\end{hint}

\begin{hint}[Exercise VI.16.17]
A nonzero holomorphic function admits a holomorphic logarithm after shrinking to a simply connected neighborhood avoiding zero.
\end{hint}

\begin{hint}[Exercise VI.16.18]
Use local primitives and identify functions with zero derivative as locally constant.
\end{hint}

\begin{hint}[Exercise VI.16.19]
Use quotient stalks from VI/15 together with the kernel/image stalk identities.
\end{hint}

\begin{hint}[Exercise VI.16.20]
A sheaf is zero iff every stalk is zero.
\end{hint}

\begin{hint}[Exercise VI.16.21]
Take stalks: stalks commute with direct sums and direct sums are exact in $\mathbf{Ab}$.
\end{hint}

\begin{hint}[Exercise VI.16.22]
All stalks except one are zero; at the distinguished point recover the original group sequence.
\end{hint}

\begin{hint}[Exercise VI.16.23]
Every stalk is canonically the corresponding group in the original sequence.
\end{hint}

\begin{hint}[Exercise VI.16.24]
$(f_*\F)(V)=\F(f^{-1}V)$ and $\Gamma(f^{-1}V,-)$ is left exact.
\end{hint}

\begin{hint}[Exercise VI.16.25]
Check exactness at each $x$ using the stalk identity for inverse image.
\end{hint}

\begin{hint}[Exercise VI.16.26]
Stalks of sheafification equal stalks of the original presheaf; filtered colimits of abelian groups are exact.
\end{hint}
