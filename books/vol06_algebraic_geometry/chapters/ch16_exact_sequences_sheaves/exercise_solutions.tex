\begin{solution}[Exercise VI.16.1]
A section $s\in\G(U)$ with $g(s)=0$ lies in $\ker g(U)$.  Exactness means $\ker g=\im f$, so for every $x\in U$ there are $x\in V\subseteq U$ and $t\in\F(V)$ with $f(t)=s|_V$.  Conversely this local lifting condition says every kernel section lies in the sheaf image.
\end{solution}

\begin{solution}[Exercise VI.16.2]
Exactness gives $\im f=\ker g$, hence every local image of $f$ is killed by $g$.  Thus $gf$ has zero germ at every point, so $gf=0$.
\end{solution}

\begin{solution}[Exercise VI.16.3]
Taking the stalk at $x$ gives $(\im f)_x=(\ker g)_x$.  VI/15 identifies these with $\im f_x$ and $\ker g_x$, respectively.
\end{solution}

\begin{solution}[Exercise VI.16.4]
First $g_xf_x=0$ for every $x$, so $gf=0$.  If $s\in\ker g(U)$ and $x\in U$, exactness on the stalk gives $t_x$ with $f_x(t_x)=s_x$.  Represent $t_x$ locally and shrink until equality of germs is equality of sections.  Hence $s$ is locally in the image, so $s\in\im f(U)$.
\end{solution}

\begin{solution}[Exercise VI.16.5]
If $f$ is injective, $\ker f=0$, hence every $\ker f_x=0$.  Conversely if all $f_x$ are injective, then every stalk of $\ker f$ vanishes; therefore $\ker f=0$.
\end{solution}

\begin{solution}[Exercise VI.16.6]
If $f$ is injective as a sheaf morphism, its kernel sheaf is zero, so the sectionwise kernel is zero on every open.  Conversely, injectivity on all section groups certainly makes the kernel sheaf zero.
\end{solution}

\begin{solution}[Exercise VI.16.7]
If $f$ is surjective as a sheaf morphism, $\im f=\G$, so $\im f_x=\G_x$.  Conversely if every $f_x$ is onto, the image sheaf has the same stalk as $\G$ at every point and its inclusion into $\G$ is an isomorphism.
\end{solution}

\begin{solution}[Exercise VI.16.8]
If $f_x$ is onto and $g\in\G(U)$, lift $g_x$ to a germ $s_x$ and represent it by a section near $x$; shrink until $f(s)=g$.  Conversely a local lift immediately gives a preimage for every germ.
\end{solution}

\begin{solution}[Exercise VI.16.9]
If $f$ is an isomorphism, all stalk maps are.  Conversely, if every $f_x$ is an isomorphism, then the kernel and cokernel of $f$ have zero stalks, hence are zero; equivalently $f$ is both mono and epi and therefore an isomorphism.
\end{solution}

\begin{solution}[Exercise VI.16.10]
Apply the stalkwise criterion to the four-term sequence.  Exactness at $\F'$, $\F$, and $\F''$ is equivalent to exactness at the corresponding stalks for all $x$.
\end{solution}

\begin{solution}[Exercise VI.16.11]
Let $q:\F\to\F''$.  Exactness identifies $\ker q$ with $\F'$ and makes $q$ surjective.  The canonical coimage-image isomorphism of VI/15 gives $\F/\F'\cong\F/\ker q\cong\im q=\F''$.
\end{solution}

\begin{solution}[Exercise VI.16.12]
For $x\in U$, $(\F|_U)_x\cong\F_x$.  Thus the restricted stalk sequence is the original exact stalk sequence, and the stalkwise criterion proves exactness on $U$.
\end{solution}

\begin{solution}[Exercise VI.16.13]
Fix $x$.  Choose $U_i$ containing $x$.  The stalk of the restriction to $U_i$ at $x$ is the original stalk, so the stalk sequence is exact.  Since this holds for all $x$, the original sequence is exact.
\end{solution}

\begin{solution}[Exercise VI.16.14]
The first map on sections is injective because the sheaf inclusion is sectionwise injective.  If $s\in\F(U)$ maps to zero in $\F''(U)$, then $s$ belongs to the kernel sheaf $\F'$, hence comes from a unique element of $\F'(U)$.
\end{solution}

\begin{solution}[Exercise VI.16.15]
The identity map $z\mapsto z$ is a section of the circle-valued target.  A lift $a:S^1\to\mathbb R$ with $e^{2\pi ia(z)}=z$ would give a continuous argument around the circle and force degree zero, contradicting that the identity has degree one.
\end{solution}

\begin{solution}[Exercise VI.16.16]
If $e^{2\pi i f}=1$, then $f$ takes values in $\mathbb Z$.  A continuous integer-valued function is locally constant.  Conversely every locally constant integer-valued function lies in the kernel.
\end{solution}

\begin{solution}[Exercise VI.16.17]
Near any point where $g$ is holomorphic and nonzero, choose a small simply connected neighborhood on which $g$ avoids zero.  Then $g'/g$ has a primitive $h$ and $e^{-h}g$ is constant; adjusting the constant gives a logarithm.  Thus every germ of $\mathcal O^\times$ has an exponential preimage.
\end{solution}

\begin{solution}[Exercise VI.16.18]
The kernel of $d$ is the sheaf of locally constant real functions.  Every smooth one-form is locally $a(x)dx$ and has a local primitive $F(x)=\int a(x)dx$.  Hence the stalk maps of $d$ are surjective and the sequence is exact.
\end{solution}

\begin{solution}[Exercise VI.16.19]
Use $(\ker d_i)_x=\ker(d_i)_x$, $(\im d_{i-1})_x=\im(d_{i-1})_x$, and the quotient-stalk formula.  This gives the displayed ordinary homology group.
\end{solution}

\begin{solution}[Exercise VI.16.20]
Exactness at $\F_i$ is $\ker d_i=\im d_{i-1}$, which is equivalent to the quotient sheaf $\ker d_i/\im d_{i-1}$ being zero.  Equivalently, all its stalks vanish.
\end{solution}

\begin{solution}[Exercise VI.16.21]
At a point $x$, the stalk of the direct sum is the direct sum of the stalks.  Direct sums preserve exact sequences of abelian groups, so the stalkwise criterion gives exactness of the sheaf sequence.
\end{solution}

\begin{solution}[Exercise VI.16.22]
At $y\neq x$ all three skyscraper stalks vanish.  At $x$ the stalk sequence is exactly $0\to A'\to A\to A''\to0$.  Hence the sheaf sequence is exact by stalks.
\end{solution}

\begin{solution}[Exercise VI.16.23]
Each stalk of $\underline A$ is $A$ on a locally connected space.  Therefore the sheaf sequence has at each point the given short exact group sequence, so it is short exact.
\end{solution}

\begin{solution}[Exercise VI.16.24]
For every open $V\subseteq Y$, applying $f_*$ gives section groups over $f^{-1}V$.  The section functor is left exact, so kernels are preserved.  Hence $f_*$ is left exact.
\end{solution}

\begin{solution}[Exercise VI.16.25]
At $x\in X$, the stalk of the inverse-image sequence is the original sequence at $f(x)$, which is exact.  The stalkwise criterion gives exactness of the inverse-image sequence.
\end{solution}

\begin{solution}[Exercise VI.16.26]
An objectwise exact presheaf sequence gives an exact sequence after taking the filtered colimit over neighborhoods of any point.  Sheafification leaves these stalks unchanged, so the sheafified sequence is exact by the stalkwise criterion.
\end{solution}
