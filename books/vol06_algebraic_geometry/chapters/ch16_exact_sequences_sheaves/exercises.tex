\begin{exercise}[Exactness at a term]\label{ex:vi16-01}
Let $\F\xrightarrow{f}\G\xrightarrow{g}\Hcal$ be a sequence.  Rewrite exactness at $\G$ as a local lifting statement for sections in $\ker g$.
\end{exercise}

\begin{exercise}[Composition zero]\label{ex:vi16-02}
Prove that exactness at $\G$ implies $g\circ f=0$.
\end{exercise}

\begin{exercise}[Stalkwise forward direction]\label{ex:vi16-03}
Assuming $\im f=\ker g$, prove $\im f_x=\ker g_x$ for every $x$.
\end{exercise}

\begin{exercise}[Stalkwise converse]\label{ex:vi16-04}
Assume $\im f_x=\ker g_x$ for every $x$.  Prove $\im f=\ker g$.
\end{exercise}

\begin{exercise}[Injectivity on stalks]\label{ex:vi16-05}
Show that $f:\F\to\G$ is injective iff every stalk map $f_x$ is injective.
\end{exercise}

\begin{exercise}[Injectivity on sections]\label{ex:vi16-06}
Show that a sheaf morphism is injective iff $f_U$ is injective for every open $U$.
\end{exercise}

\begin{exercise}[Epimorphism criterion]\label{ex:vi16-07}
Prove that $f$ is surjective as a sheaf morphism iff every $f_x$ is surjective.
\end{exercise}

\begin{exercise}[Local lifting criterion]\label{ex:vi16-08}
Prove that stalkwise surjectivity is equivalent to local liftability of every target section.
\end{exercise}

\begin{exercise}[Isomorphism criterion]\label{ex:vi16-09}
Prove that $f$ is an isomorphism iff every $f_x$ is an isomorphism.
\end{exercise}

\begin{exercise}[Short exact stalks]\label{ex:vi16-10}
Show that a sequence $0\to\F'\to\F\to\F''\to0$ is short exact iff its stalk sequence is short exact for every point.
\end{exercise}

\begin{exercise}[Quotient recognition]\label{ex:vi16-11}
In a short exact sequence $0\to\F'\to\F\to\F''\to0$, prove $\F''\cong\F/\F'$.
\end{exercise}

\begin{exercise}[Restriction]\label{ex:vi16-12}
Prove directly from stalks that restriction to an open subset preserves exactness.
\end{exercise}

\begin{exercise}[Local exactness]\label{ex:vi16-13}
Show that if an open cover $\{U_i\}$ has exact restricted sequences on every $U_i$, then the original sequence is exact.
\end{exercise}

\begin{exercise}[Left exact sections]\label{ex:vi16-14}
From a short exact sequence prove $0\to\F'(U)\to\F(U)\to\F''(U)$ is exact.
\end{exercise}

\begin{exercise}[Failure at the right end]\label{ex:vi16-15}
For the continuous exponential sequence on $S^1$, exhibit a section of the target with no global lift.
\end{exercise}

\begin{exercise}[Continuous exponential kernel]\label{ex:vi16-16}
Show that the kernel of $f\mapsto e^{2\pi i f}$ is the locally constant integer-valued sheaf.
\end{exercise}

\begin{exercise}[Holomorphic logarithms]\label{ex:vi16-17}
Show locally that $\exp:\mathcal O\to\mathcal O^\times$ is surjective on stalks.
\end{exercise}

\begin{exercise}[Derivative sheaf]\label{ex:vi16-18}
On a smooth one-manifold prove $0\to\underline{\mathbb R}\to\mathcal C^\infty\xrightarrow{d}\Omega^1\to0$ is exact.
\end{exercise}

\begin{exercise}[Homology stalk]\label{ex:vi16-19}
For a complex, prove
\[
(\ker d_i/\im d_{i-1})_x\cong\ker((d_i)_x)/\im((d_{i-1})_x).
\]
\end{exercise}

\begin{exercise}[Vanishing homology]\label{ex:vi16-20}
Prove that a complex is exact at $\F_i$ iff its $i$th homology sheaf is zero.
\end{exercise}

\begin{exercise}[Direct sums]\label{ex:vi16-21}
Show that a direct sum of short exact sequences of sheaves of abelian groups is short exact.
\end{exercise}

\begin{exercise}[Skyscraper exactness]\label{ex:vi16-22}
Given an exact sequence of abelian groups, prove the induced skyscraper sequence at a closed point is exact.
\end{exercise}

\begin{exercise}[Locally constant exactness]\label{ex:vi16-23}
Given a short exact sequence of abelian groups, show the associated locally constant sheaves form a short exact sequence.
\end{exercise}

\begin{exercise}[Direct image left exactness]\label{ex:vi16-24}
For $f:X\to Y$, prove $f_*$ preserves kernels and therefore is left exact.
\end{exercise}

\begin{exercise}[Inverse image exactness]\label{ex:vi16-25}
Using $(f^{-1}\G)_x\cong\G_{f(x)}$, prove that $f^{-1}$ preserves exact sequences of sheaves of abelian groups.
\end{exercise}

\begin{exercise}[Sheafification exactness]\label{ex:vi16-26}
Show that sheafification carries an objectwise exact sequence of presheaves of abelian groups to an exact sequence of sheaves.
\end{exercise}
