\begin{example}[The zero sequence]
The sequence $0\to\F\xrightarrow{\id}\F\to0$ is exact.  Every stalk is the ordinary exact sequence $0\to\F_x\xrightarrow{\id}\F_x\to0$.
\end{example}

\begin{example}[A split short exact sequence]
For any $\F$ and $\G$,
\[
0\to\F\xrightarrow{s\mapsto(s,0)}\F\oplus\G\xrightarrow{(s,t)\mapsto t}\G\to0
\]
is short exact and even splits globally.
\end{example}

\begin{example}[Multiplication by an integer]
For $n\neq0$,
\[
0\to\underline{\mathbb Z}\xrightarrow{n}\underline{\mathbb Z}\to
\underline{\mathbb Z/n\mathbb Z}\to0
\]
is exact.  This follows immediately from the stalk sequence $0\to\mathbb Z\xrightarrow{n}\mathbb Z\to\mathbb Z/n\mathbb Z\to0$.
\end{example}

\begin{example}[A skyscraper short exact sequence]
If $0\to A'\to A\to A''\to0$ is exact and $i_x$ denotes a skyscraper sheaf at a closed point, then
$0\to i_x(A')\to i_x(A)\to i_x(A'')\to0$ is exact because the only nonzero stalk is at $x$.
\end{example}

\begin{example}[Exactness at the middle term]
If $\F\xrightarrow{f}\G\xrightarrow{g}\Hcal$ is exact at $\G$, then a section $s\in\G(U)$ with $g(s)=0$ need only be \emph{locally} of the form $f(t)$.  Exactness does not promise one global $t\in\F(U)$.
\end{example}

\begin{example}[Kernel detection]
A morphism $f:\F\to\G$ is injective if and only if $f_x$ is injective for every $x$.  If $f(s)=0$, all germs $s_x$ vanish, hence $s=0$ by the sheaf axiom.
\end{example}

\begin{example}[Local surjectivity]
A morphism $f:\F\to\G$ is surjective as a sheaf morphism exactly when every section of $\G$ admits preimages on a neighborhood of every point.  This is equivalent to surjectivity of all stalk maps.
\end{example}

\begin{example}[Restriction of a short exact sequence]
If $0\to\F'\to\F\to\F''\to0$ is short exact on $X$ and $U\subseteq X$ is open, then
$0\to\F'|_U\to\F|_U\to\F''|_U\to0$ is short exact on $U$.
\end{example}

\begin{example}[Finite discrete space]
On a finite discrete space, a sheaf is simply a family of stalk groups.  A sequence of sheaves is exact precisely when the corresponding sequence is exact at each point, and here sectionwise exactness over every open also follows componentwise.
\end{example}

\begin{example}[Continuous exponential]
The sequence
\[
0\to\underline{\mathbb Z}\to\mathcal C^0_{\mathbb R}
\xrightarrow{f\mapsto e^{2\pi i f}}\mathcal C^0_{S^1}\to0
\]
is exact as a sequence of sheaves: every circle-valued function has local continuous arguments.
\end{example}

\begin{example}[Derivative on an interval]
On an open interval $I\subseteq\mathbb R$, the map
$d:\mathcal C^\infty_I\to\Omega^1_I$ is surjective on sections because every smooth one-form has a global primitive on an interval.  Its kernel is the locally constant sheaf $\underline{\mathbb R}$.
\end{example}

\begin{example}[A complex with zero homology sheaf]
If $\F\xrightarrow{f}\G\xrightarrow{g}\Hcal$ is exact at $\G$, then
$\ker g/\im f=0$.  Conversely, the vanishing of this homology sheaf is exactly the statement of exactness at $\G$.
\end{example}

\begin{example}[Detecting nonexactness at one point]
If for some $x$ the sequence $\F_x\to\G_x\to\Hcal_x$ has a kernel element not in the preceding image, then the sheaf sequence is not exact.  One bad stalk is enough.
\end{example}

\begin{example}[Exactness on a neighborhood]
Suppose every $x\in X$ has an open neighborhood $U_x$ on which a given sequence restricts to an exact sequence.  Then the original sequence is exact, because each stalk can be computed inside one such neighborhood.
\end{example}

\begin{example}[Inclusion and quotient]
For a subsheaf $\F'\subseteq\F$, the quotient sequence
$0\to\F'\to\F\to\F/\F'\to0$ is exact.  This is the basic source of short exact sequences.
\end{example}

\begin{example}[Composition vanishes]
In any exact pair $\F\xrightarrow{f}\G\xrightarrow{g}\Hcal$, one has $gf=0$ because $\im f\subseteq\ker g$.
\end{example}

\begin{example}[Exactness and supports]
For a complex with homology sheaf $\HH_i$, exactness holds on an open set $U$ exactly when $\HH_i|_U=0$.  Thus $\Supp\HH_i$ records where exactness fails.
\end{example}

\begin{example}[Locally constant sheaves]
An exact sequence $0\to A'\to A\to A''\to0$ of abelian groups induces a short exact sequence of locally constant sheaves on a locally connected space.  Every stalk is the original group sequence.
\end{example}

\begin{example}[Left exactness of sections]
From $0\to\F'\to\F\to\F''\to0$, the sequence
$0\to\F'(U)\to\F(U)\to\F''(U)$ is exact.  A kernel section in $\F(U)$ is globally a section of the kernel subsheaf $\F'$.
\end{example}

\begin{example}[A sheaf epimorphism with no global lift]
For the continuous exponential on $S^1$, the identity map $z\mapsto z$ is a global section of $\mathcal C^0_{S^1}(S^1)$ with no global real-valued argument.  The map is nevertheless a sheaf epimorphism because local arguments exist.
\end{example}
