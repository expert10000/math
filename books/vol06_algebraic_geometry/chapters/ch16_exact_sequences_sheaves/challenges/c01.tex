\begin{problem}[Challenge: Short five lemma for sheaves]\label{chal:vi16-01}
Consider a commutative diagram of short exact sequences
\[
\begin{tikzcd}
0\arrow[r]&\F'\arrow[r]\arrow[d,"a"']&\F\arrow[r]\arrow[d,"b"']&\F''\arrow[r]\arrow[d,"c"']&0\\
0\arrow[r]&\G'\arrow[r]&\G\arrow[r]&\G''\arrow[r]&0.
\end{tikzcd}
\]
Prove that if $a$ and $c$ are isomorphisms, then $b$ is an isomorphism.
\end{problem}
\begin{hint}
Take stalks and apply the ordinary short five lemma, then use stalkwise detection of isomorphisms.
\end{hint}
