\chapter{\v{C}ech Cohomology}
\label{ch:vi47-cech-cohomology}

VI/46 introduced flabby sheaves and explained why their extension property kills higher cohomology.  The present chapter turns that abstract vanishing mechanism into a computational machine.  Given an open cover, one records sections on the opens, differences on pairwise overlaps, compatibility on triple overlaps, and so on.  The resulting alternating complex is the \v{C}ech complex.

The basic progression is
\[
\boxed{\begin{gathered}
\text{open cover}\Longrightarrow\text{cochains on overlaps}\\
\Downarrow\\
\text{cocycles modulo coboundaries}\Longrightarrow\check H^p
\end{gathered}}
\]
and the first major calculation is the standard two-chart cover of $\PP^1$, where line-bundle cohomology becomes a quotient of Laurent polynomials.

\section*{Standing conventions}
All sheaves are sheaves of abelian groups unless stated otherwise.  Covers are indexed by totally ordered sets so that signs in alternating formulas are unambiguous.  Empty intersections contribute the zero group.  When comparing \v{C}ech and sheaf cohomology, the comparison hypotheses will always be stated explicitly.

\section*{Learning goals}
By the end of the chapter, the reader should be able to:
\begin{itemize}
\item build the \v{C}ech cochain complex of an open cover;
\item prove $\delta^2=0$ and compute cocycles and coboundaries;
\item identify degree-zero \v{C}ech cohomology with global sections;
\item compute two-open and three-open examples explicitly;
\item use refinements and understand cover dependence;
\item state the Leray comparison criterion;
\item recover the flabby-sheaf vanishing result from VI/46;
\item compute $\check H^0$ and $\check H^1$ of $\mathcal O_{\PP^1}(n)$ on the standard cover;
\item reinterpret line-bundle transition functions as $\check H^1(\mathcal O_X^\times)$;
\item prepare for connecting homomorphisms and long exact sequences in VI/48.
\end{itemize}

\section{The \v{C}ech cochain complex}
\begin{definition}[Ordered open cover]\label{def:vi47-cover}
An \emph{ordered open cover} of a space $X$ is a family $\mathfrak U=\{U_i\}_{i\in I}$ whose union is $X$, together with a total order on $I$.  For $i_0<\cdots<i_p$ write
\[
U_{i_0\cdots i_p}=U_{i_0}\cap\cdots\cap U_{i_p}.
\]
\end{definition}

\begin{definition}[\v{C}ech cochains]\label{def:vi47-cochains}
For a sheaf $\mathcal F$ of abelian groups and an ordered open cover $\mathfrak U$, define
\[
C^p(\mathfrak U,\mathcal F)
=
\prod_{i_0<\cdots<i_p}\Gamma(U_{i_0\cdots i_p},\mathcal F).
\]
An element of $C^p$ is called a \emph{$p$-cochain}.
\end{definition}

\input{chapters/ch47_ech_cohomology/figures/figure_02}
\begin{proposition}[The alternating differential satisfies $\delta^2=0$]\label{prop:vi47-delta-square}
Define
\[
(\delta c)_{i_0\cdots i_{p+1}}
=
\sum_{r=0}^{p+1}(-1)^r
c_{i_0\cdots\widehat{i_r}\cdots i_{p+1}}
\big|_{U_{i_0\cdots i_{p+1}}}.
\]
Then $\delta^{p+1}\circ\delta^p=0$.
\begin{proof}
Every term obtained by deleting indices $i_r$ and $i_s$ occurs twice.  The two occurrences have opposite signs because deleting $i_r$ first shifts the position of $i_s$ by one exactly when $r<s$.  The corresponding restrictions agree by functoriality of restriction maps, so the terms cancel pairwise.
\end{proof}
\end{proposition}

\begin{definition}[\v{C}ech cocycles, coboundaries, and cohomology]\label{def:vi47-cohomology}
Set
\[
Z^p(\mathfrak U,\mathcal F)=\ker(\delta:C^p\to C^{p+1}),
\qquad
B^p(\mathfrak U,\mathcal F)=\operatorname{im}(\delta:C^{p-1}\to C^p).
\]
The $p$th \v{C}ech cohomology group of the cover is
\[
\check H^p(\mathfrak U,\mathcal F)=Z^p/B^p.
\]
\end{definition}

\begin{proposition}[Degree zero is global sections]\label{prop:vi47-h0}
For every open cover $\mathfrak U$ there is a canonical isomorphism
\[
\check H^0(\mathfrak U,\mathcal F)\cong\Gamma(X,\mathcal F).
\]
\begin{proof}
A zero-cocycle is a family $s_i\in\Gamma(U_i,\mathcal F)$ whose restrictions agree on every pairwise overlap.  The sheaf gluing axiom produces a unique global section, and every global section gives such a compatible family.
\end{proof}
\end{proposition}

\section{Two-open and three-open covers}
\input{chapters/ch47_ech_cohomology/figures/figure_03}
\begin{proposition}[The two-open complex]\label{prop:vi47-two-open}
If $X=U\cup V$, then
\[
C^0=\Gamma(U,\mathcal F)\oplus\Gamma(V,\mathcal F),
\qquad
C^1=\Gamma(U\cap V,\mathcal F),
\qquad
C^p=0\ (p\ge2),
\]
and
\[
\check H^1(\mathfrak U,\mathcal F)
\cong
\frac{\Gamma(U\cap V,\mathcal F)}
{\Gamma(U,\mathcal F)|_{U\cap V}-\Gamma(V,\mathcal F)|_{U\cap V}}.
\]
\begin{proof}
There are only two indices, so no strictly increasing triple exists.  The displayed quotient is therefore $C^1/\operatorname{im}\delta^0$.
\end{proof}
\end{proposition}

\begin{example}[An overlap obstruction]\label{ex:vi47-two-open-obstruction}
A section $g\in\Gamma(U\cap V,\mathcal F)$ determines a class in $\check H^1(\{U,V\},\mathcal F)$.  Its class is zero exactly when
\[
g=s_U|_{U\cap V}-s_V|_{U\cap V}
\]
for sections $s_U$ and $s_V$ on the two opens.  Thus first \v{C}ech cohomology records whether overlap data can be split into data coming from the members of the cover.
\end{example}

\input{chapters/ch47_ech_cohomology/figures/figure_04}
\begin{proposition}[The three-open complex]\label{prop:vi47-three-open}
For $X=U_0\cup U_1\cup U_2$,
\[
C^0=\prod_i\Gamma(U_i,\mathcal F),\qquad
C^1=\prod_{i<j}\Gamma(U_{ij},\mathcal F),\qquad
C^2=\Gamma(U_{012},\mathcal F).
\]
For $c=(c_{01},c_{02},c_{12})$,
\[
(\delta c)_{012}=c_{12}-c_{02}+c_{01}
\]
after restriction to the triple overlap.
\begin{proof}
This is the general alternating formula specialized to three indices.
\end{proof}
\end{proposition}

\section{Refinements and intrinsic \v{C}ech cohomology}
\input{chapters/ch47_ech_cohomology/figures/figure_05}
\begin{definition}[Refinement of a cover]\label{def:vi47-refinement}
A cover $\mathfrak V=\{V_j\}_{j\in J}$ \emph{refines} $\mathfrak U=\{U_i\}_{i\in I}$ if there is a function $\tau:J\to I$ such that
\[
V_j\subseteq U_{\tau(j)}
\]
for all $j$.  We write $\mathfrak V\prec\mathfrak U$.
\end{definition}

\begin{theorem}[Refinements induce maps on cohomology]\label{thm:vi47-refinement-map}
A refinement map $\tau$ induces a cochain map
\[
C^\bullet(\mathfrak U,\mathcal F)\longrightarrow C^\bullet(\mathfrak V,\mathcal F)
\]
by restriction, hence homomorphisms
\[
\check H^p(\mathfrak U,\mathcal F)\longrightarrow\check H^p(\mathfrak V,\mathcal F).
\]
Different choices of refinement map induce the same map on cohomology.
\begin{proof}
Restriction commutes with the alternating differential.  Two refinement choices are joined by the standard prism cochain homotopy, so their induced maps differ by a coboundary and therefore agree on cohomology.
\end{proof}
\end{theorem}

\begin{definition}[Global \v{C}ech cohomology]\label{def:vi47-global-cech}
The \emph{global \v{C}ech cohomology} of $\mathcal F$ is the filtered colimit over open covers ordered by refinement:
\[
\check H^p(X,\mathcal F)
=
\varinjlim_{\mathfrak U}\check H^p(\mathfrak U,\mathcal F).
\]
\end{definition}

\begin{remark}[Cover dependence before refinement]\label{remark:vi47-cover-dependence}
The group $\check H^p(\mathfrak U,\mathcal F)$ can depend strongly on the chosen cover.  Passing to refinements separates the intrinsic \v{C}ech theory of the space from artifacts of one coarse cover.
\end{remark}

% BEGIN VOL06-EXPANSION VI47-example-03
\begin{example}[Cocycle condition on triple overlaps]\label{ex:vi47-expand-03}
For three opens, a $1$-cochain $(g_{ij})$ is a cocycle when $g_{jk}-g_{ik}+g_{ij}=0$ on triple intersections.  The alternating signs are exactly the Cech differential.
\end{example}

\section{Flabby sheaves and Leray covers}
\input{chapters/ch47_ech_cohomology/figures/figure_06}
\begin{theorem}[Flabby sheaves are \v{C}ech-acyclic]\label{thm:vi47-flabby-vanish}
If $\mathcal F$ is flabby, then for every open cover $\mathfrak U$,
\[
\check H^p(\mathfrak U,\mathcal F)=0
\qquad(p>0).
\]
\begin{proof}
This is the exactness theorem developed in VI/46: successive extension of sections from intersections produces a contracting procedure for positive-degree cocycles.
\end{proof}
\end{theorem}

\begin{definition}[Acyclic (Leray) cover for a sheaf]\label{def:vi47-leray-cover}
An open cover $\mathfrak U=\{U_i\}$ is \emph{$\mathcal F$-acyclic} if every nonempty finite intersection $U_{i_0\cdots i_p}$ satisfies
\[
H^q(U_{i_0\cdots i_p},\mathcal F)=0
\qquad(q>0).
\]
\end{definition}

\begin{theorem}[Leray comparison criterion]\label{thm:vi47-leray}
If $\mathfrak U$ is $\mathcal F$-acyclic, then the natural comparison map is an isomorphism
\[
\check H^p(\mathfrak U,\mathcal F)\xrightarrow{\sim}H^p(X,\mathcal F)
\]
for all $p\ge0$.
\begin{proof}
Resolve $\mathcal F$ by flabby sheaves and form the double complex obtained by taking \v{C}ech cochains degreewise.  Exactness in one direction comes from flabbiness, while the acyclicity hypothesis collapses the other spectral filtration to the zeroth row.  Both filtrations therefore compute the same total cohomology.
\end{proof}
\end{theorem}

\begin{example}[Affine-cover preview for schemes]\label{ex:vi47-affine-preview}
If a scheme is covered by affine opens whose finite intersections are affine, then a quasi-coherent sheaf will be acyclic on those intersections once the affine vanishing theorem is available.  Thus the corresponding affine cover becomes a Leray cover.  The required affine vanishing result is proved systematically in VI/49.
\end{example}

\section{The standard cover of \texorpdfstring{$\PP^1$}{P1}}
\input{chapters/ch47_ech_cohomology/figures/figure_07}
\begin{definition}[The standard cover of $\PP^1$]\label{def:vi47-p1-cover}
Let
\[
\PP^1_k=\operatorname{Proj}k[x_0,x_1],
\qquad
U_0=D_+(x_0),\quad U_1=D_+(x_1).
\]
Writing $t=x_1/x_0$ on $U_0$ and $u=x_0/x_1=t^{-1}$ on $U_1$, we have
\[
U_0\cong\mathbb A^1_t,\qquad U_1\cong\mathbb A^1_u,\qquad U_{01}\cong\operatorname{Spec}k[t,t^{-1}].
\]
\end{definition}

\begin{proposition}[Transition function for $\mathcal O(n)$]\label{prop:vi47-on-transition}
Choose the standard trivializing generators $e_0=x_0^n$ on $U_0$ and $e_1=x_1^n$ on $U_1$.  On $U_{01}$,
\[
e_1=t^n e_0.
\]
Consequently a section $b(u)e_1$ from $U_1$ is represented in the $e_0$ trivialization by
\[
t^n b(t^{-1})e_0.
\]
\begin{proof}
The relation follows immediately from $x_1=tx_0$ on $D_+(x_0x_1)$.
\end{proof}
\end{proposition}

\begin{proposition}[The two-chart complex for $\mathcal O(n)$]\label{prop:vi47-on-complex}
With the $e_0$ trivialization on the overlap, the \v{C}ech complex is
\[
k[t]\oplus k[t^{-1}]
\longrightarrow
k[t,t^{-1}],
\qquad
(a(t),b(t^{-1}))\longmapsto t^n b(t^{-1})-a(t).
\]
Hence
\[
\check H^1(\mathfrak U,\mathcal O(n))
\cong
\frac{k[t,t^{-1}]}{k[t]+t^n k[t^{-1}]}.
\]
\begin{proof}
This is Proposition~\ref{prop:vi47-two-open} with the transition function inserted.
\end{proof}
\end{proposition}

\begin{theorem}[Degree-zero calculation for $\mathcal O(n)$]\label{thm:vi47-on-h0}
For the standard cover,
\[
\dim_k\check H^0(\mathfrak U,\mathcal O(n))
=
\begin{cases}
n+1,&n\ge0,\\
0,&n<0.
\end{cases}
\]
For $n\ge0$, a basis corresponds to the Laurent exponents $0,1,\dots,n$, equivalently to the homogeneous monomials
\[
x_0^n,x_0^{n-1}x_1,\dots,x_1^n.
\]
\begin{proof}
A zero-cocycle corresponds to an element of the intersection
\[
k[t]\cap t^n k[t^{-1}]\subset k[t,t^{-1}].
\]
For $n\ge0$ this is spanned by $1,t,\dots,t^n$; for $n<0$ the intersection is zero.
\end{proof}
\end{theorem}

\input{chapters/ch47_ech_cohomology/figures/figure_08}
\begin{theorem}[First \v{C}ech cohomology of $\mathcal O(n)$]\label{thm:vi47-on-h1}
For the standard cover,
\[
\check H^1(\mathfrak U,\mathcal O(n))=0
\qquad(n\ge-1),
\]
while for $n\le-2$ it has basis
\[
t^{n+1},t^{n+2},\dots,t^{-1}.
\]
Thus
\[
\dim_k\check H^1(\mathfrak U,\mathcal O(n))=-n-1
\qquad(n\le-2).
\]
\begin{proof}
The subspace $k[t]$ contains all Laurent monomials with exponent $\ge0$, while $t^n k[t^{-1}]$ contains all exponents $\le n$.  If $n\ge-1$, these two ranges cover every integer exponent.  If $n\le-2$, the missing exponents are exactly $n+1,\dots,-1$.
\end{proof}
\end{theorem}

\begin{corollary}[Dimension pattern]\label{cor:vi47-on-dimensions}
For the standard cover of $\PP^1$,
\[
\dim_k\check H^0(\mathcal O(n))=\max\{n+1,0\},
\qquad
\dim_k\check H^1(\mathcal O(n))=\max\{-n-1,0\}.
\]
Once affine acyclicity of quasi-coherent sheaves is established in VI/49, the Leray criterion identifies these groups with the sheaf cohomology groups $H^0$ and $H^1$.
\end{corollary}

\begin{example}[The first negative case]\label{ex:vi47-o-minus-two}
For $n=-2$,
\[
\check H^1(\mathfrak U,\mathcal O(-2))
\cong
k\cdot t^{-1}.
\]
The single missing Laurent mode $t^{-1}$ is the first nonzero line-bundle cohomology class visible in the standard two-chart calculation.
\end{example}

\section{Line bundles and comparison issues}
\begin{proposition}[Line bundles as unit-valued 1-cocycles]\label{prop:vi47-pic-cech}
For the sheaf of units $\mathcal O_X^\times$, a \v{C}ech 1-cocycle is a family of invertible transition functions $g_{ij}$.  On triple overlaps they satisfy
\[
g_{ij}g_{jk}=g_{ik}.
\]
Changing local trivializations multiplies the cocycle by a coboundary.  Therefore line bundles trivialized by the cover are classified by
\[
\check H^1(\mathfrak U,\mathcal O_X^\times).
\]
Passing to refinements recovers the usual \v{C}ech description of $\operatorname{Pic}(X)$.
\begin{proof}
This is precisely the gluing construction for line bundles from VI/42, rewritten in cohomological language.
\end{proof}
\end{proposition}

\begin{warning}[\v{C}ech and sheaf cohomology need not always coincide]\label{warning:vi47-cech-sheaf}
For an arbitrary topological space, arbitrary sheaf, and arbitrary cover, the groups $\check H^p(\mathfrak U,\mathcal F)$ need not equal the derived-functor groups $H^p(X,\mathcal F)$.  The Leray criterion gives a clean sufficient hypothesis.  This distinction is essential whenever a computation depends on the choice of cover.
\end{warning}

\begin{remark}[Bridge to exact sequences]\label{remark:vi47-next}
VI/48 studies what happens to cohomology under a short exact sequence
\[
0\to\mathcal F'\to\mathcal F\to\mathcal F''\to0.
\]
The connecting map $H^0(\mathcal F'')\to H^1(\mathcal F')$ will acquire a concrete interpretation: it is the \v{C}ech class of the failure of local lifts of a section of $\mathcal F''$ to glue globally.
\end{remark}
