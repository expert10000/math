% VI47-DETAILED-PROBLEM-01
\subsection*{Problem VI.47.1}
\begin{solution}
For two opens, \(C^2=0\), so every one-cochain is a cocycle.  Coboundaries are exactly differences of restrictions from the two opens.  Hence
\[
\check H^1=C^1/\operatorname{im}\delta^0.
\]
The zero class means the overlap datum splits into local pieces.
\end{solution}

% VI47-DETAILED-PROBLEM-02
\subsection*{Problem VI.47.2}
\begin{solution}
If \(c_{ij}=s_j-s_i\), then
\[
c_{12}-c_{02}+c_{01}=0.
\]
This is the direct triangle calculation showing every degree-one coboundary is a cocycle.
\end{solution}

% VI47-DETAILED-PROBLEM-03
\subsection*{Problem VI.47.3}
\begin{solution}
Two refinement maps are joined by the standard prism homotopy.  Writing the prism operator explicitly and expanding gives \(\delta K+K\delta=\tau^*-\sigma^*\), so the maps induce the same homomorphism on cohomology.
\end{solution}

% VI47-DETAILED-PROBLEM-04
\subsection*{Problem VI.47.4}
\begin{solution}
For a flabby sheaf, extend overlap components successively to the members of the cover and subtract the corresponding zero-cochain.  The cocycle identities preserve components already killed.  Eventually the cocycle becomes zero, so it was a coboundary.
\end{solution}

% VI47-DETAILED-PROBLEM-05
\subsection*{Problem VI.47.5}
\begin{solution}
For \(\mathcal O(3)\), \(H^0\) has basis \(1,t,t^2,t^3\), hence dimension four.  The denominator for \(H^1\) covers all Laurent exponents, so \(H^1=0\).
\end{solution}

% VI47-DETAILED-PROBLEM-06
\subsection*{Problem VI.47.6}
\begin{solution}
For \(\mathcal O(-5)\), \(H^0=0\).  The missing Laurent exponents are \(-4,-3,-2,-1\), so \(H^1\) has the corresponding four monomials as a basis.
\end{solution}

% VI47-DETAILED-PROBLEM-07
\subsection*{Problem VI.47.7}
\begin{solution}
The intersection \(k[t]\cap t^nk[t^{-1}]\) for \(n\ge0\) is spanned by \(1,t,\dots,t^n\).  Multiplying by \(x_0^n\) converts \(t^j\) to \(x_0^{n-j}x_1^j\), recovering all homogeneous polynomials of degree \(n\).
\end{solution}

% VI47-DETAILED-PROBLEM-08
\subsection*{Problem VI.47.8}
\begin{solution}
The transition functions are \(t^n\) and \(t^m\).  Tensor product multiplies them to \(t^{n+m}\), which is the transition function of \(\mathcal O(n+m)\).
\end{solution}

% VI47-DETAILED-PROBLEM-09
\subsection*{Problem VI.47.9}
\begin{solution}
The two standard opens and their intersection are affine, and \(\mathcal O(n)\) is quasi-coherent.  Once affine vanishing is available, every finite intersection is acyclic, so the standard cover is Leray and computes sheaf cohomology.
\end{solution}

% VI47-DETAILED-PROBLEM-10
\subsection*{Problem VI.47.10}
\begin{solution}
Choose local lifts \(t_i\) of a global quotient section.  The differences \(t_j-t_i\) lie in the kernel and satisfy the cocycle identity.  Changing lifts adds a coboundary, so the resulting class is well defined and becomes the connecting class.
\end{solution}

% VI47-DETAILED-PROBLEM-11
\subsection*{Problem VI.47.11}
\begin{solution}
The one-set cover has no positive cochains, while a nontrivial cover can have nonzero overlap classes.  Thus fixed-cover Cech cohomology depends on the cover; refinement or a comparison theorem is needed for intrinsic cohomology.
\end{solution}

% VI47-DETAILED-PROBLEM-12
\subsection*{Problem VI.47.12}
\begin{solution}
A line bundle trivialized on a cover has unit-valued transition functions satisfying the 1-cocycle condition.  Changing frames multiplies them by a coboundary.  Thus \(\check H^1(\mathcal O_X^\times)\) recovers the Picard classification from VI/42.
\end{solution}
