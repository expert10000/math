% VI47-DETAILED-EXERCISE-01
\subsection*{Exercise VI.47.1}
\begin{solution}
For \(X=U_0\cup U_1\), the alternating formula is
\[
(\delta s)_{01}=s_1|_{U_{01}}-s_0|_{U_{01}}.
\]
Using the opposite overall sign would give the same cohomology, but one convention must be kept consistently.
\end{solution}

% VI47-DETAILED-EXERCISE-02
\subsection*{Exercise VI.47.2}
\begin{solution}
If \(c_{01}=s_1-s_0\), \(c_{02}=s_2-s_0\), \(c_{12}=s_2-s_1\), then
\[
c_{12}-c_{02}+c_{01}=0
\]
on the triple overlap.  Thus \(\delta^1\delta^0=0\) directly.
\end{solution}

% VI47-DETAILED-EXERCISE-03
\subsection*{Exercise VI.47.3}
\begin{solution}
A zero-cocycle is a compatible family of local sections.  The sheaf gluing axiom produces a unique global section, and every global section produces such a family.  Therefore \(\check H^0(\mathfrak U,\mathcal F)\cong\Gamma(X,\mathcal F)\) for every cover.
\end{solution}

% VI47-DETAILED-EXERCISE-04
\subsection*{Exercise VI.47.4}
\begin{solution}
For the one-set cover, \(C^0=\Gamma(X,\mathcal F)\) and \(C^p=0\) for \(p>0\).  Hence \(\check H^0=\Gamma(X,\mathcal F)\) and all positive fixed-cover Cech groups vanish.
\end{solution}

% VI47-DETAILED-EXERCISE-05
\subsection*{Exercise VI.47.5}
\begin{solution}
A one-cochain \((c_{01},c_{02},c_{12})\) is a cocycle precisely when
\[
c_{12}-c_{02}+c_{01}=0
\]
on \(U_{012}\), after restricting all three terms.
\end{solution}

% VI47-DETAILED-EXERCISE-06
\subsection*{Exercise VI.47.6}
\begin{solution}
If \(U\cap V=\varnothing\), then \(C^1=\Gamma(\varnothing,\mathcal F)=0\).  Hence \(\check H^1=0\).
\end{solution}

% VI47-DETAILED-EXERCISE-07
\subsection*{Exercise VI.47.7}
\begin{solution}
A global section restricts to every member of either cover.  Refinement merely restricts those same sections further, so under \(\check H^0\cong\Gamma(X,\mathcal F)\) the refinement map is the identity.
\end{solution}

% VI47-DETAILED-EXERCISE-08
\subsection*{Exercise VI.47.8}
\begin{solution}
The nonempty intersections \(U_i\cap V_j\) cover \(X\) and refine both covers.  Passing to the filtered colimit therefore compares any two fixed-cover classes after a common refinement and removes dependence on a single coarse cover.
\end{solution}

% VI47-DETAILED-EXERCISE-09
\subsection*{Exercise VI.47.9}
\begin{solution}
Every overlap section \(g\) extends to one of the two opens by flabbiness.  Taking the local section on the other open to be zero makes \(g\) a coboundary up to sign.  Hence \(\check H^1=0\).
\end{solution}

% VI47-DETAILED-EXERCISE-10
\subsection*{Exercise VI.47.10}
\begin{solution}
A cover is Leray for \(\mathcal F\) when every nonempty finite intersection \(U_{i_0\cdots i_p}\) has
\[
H^q(U_{i_0\cdots i_p},\mathcal F)=0\quad(q>0).
\]
Then fixed-cover Cech cohomology agrees with sheaf cohomology.
\end{solution}

% VI47-DETAILED-EXERCISE-11
\subsection*{Exercise VI.47.11}
\begin{solution}
The overlap is \(D_+(x_0x_1)\).  Its degree-zero coordinate ring is generated by
\[
t=x_1/x_0,\qquad t^{-1}=x_0/x_1,
\]
so it is \(k[t,t^{-1}]\).
\end{solution}

% VI47-DETAILED-EXERCISE-12
\subsection*{Exercise VI.47.12}
\begin{solution}
For \(\mathcal O(2)\), the zero-cocycle intersection is
\[
k[t]\cap t^2k[t^{-1}]=\langle1,t,t^2\rangle.
\]
These correspond to \(x_0^2,x_0x_1,x_1^2\).
\end{solution}

% VI47-DETAILED-EXERCISE-13
\subsection*{Exercise VI.47.13}
\begin{solution}
For \(n=-1\), the denominator in
\[
k[t,t^{-1}]/(k[t]+t^{-1}k[t^{-1}])
\]
contains all exponents \(\ge0\) and all exponents \(\le-1\), hence every Laurent monomial.  The quotient is zero.
\end{solution}

% VI47-DETAILED-EXERCISE-14
\subsection*{Exercise VI.47.14}
\begin{solution}
For \(n=-3\), the denominator contains exponents \(\ge0\) and \(\le-3\).  The missing modes are \(t^{-2}\) and \(t^{-1}\), giving
\[
\check H^1(\mathcal O(-3))=k t^{-2}\oplus k t^{-1}.
\]
\end{solution}

% VI47-DETAILED-EXERCISE-15
\subsection*{Exercise VI.47.15}
\begin{solution}
The zero-cocycle space is \(k[t]\cap t^{-2}k[t^{-1}]\).  The two exponent ranges are \(\ge0\) and \(\le-2\), so the intersection is zero.
\end{solution}

% VI47-DETAILED-EXERCISE-16
\subsection*{Exercise VI.47.16}
\begin{solution}
At \(n=0\), \(k[t]\cap k[t^{-1}]=k\), hence \(\check H^0(\mathcal O)=k\).  The sum \(k[t]+k[t^{-1}]\) contains every Laurent monomial, so \(\check H^1=0\).
\end{solution}

% VI47-DETAILED-EXERCISE-17
\subsection*{Exercise VI.47.17}
\begin{solution}
Unit-valued cocycles \(g_{ij}\) are exactly transition functions for gluing trivial line bundles, and multiplicative coboundaries are exactly changes of local frames.  Therefore \(\check H^1(\mathfrak U,\mathcal O_X^\times)\) classifies line bundles trivialized by the cover.
\end{solution}

% VI47-DETAILED-EXERCISE-18
\subsection*{Exercise VI.47.18}
\begin{solution}
Tensor product multiplies transition functions:
\[
t^n t^m=t^{n+m}.
\]
Thus the transition cocycle of \(\mathcal O(n)\otimes\mathcal O(m)\) is the one for \(\mathcal O(n+m)\).
\end{solution}

% VI47-DETAILED-EXERCISE-19
\subsection*{Exercise VI.47.19}
\begin{solution}
The Leray hypothesis guarantees a spectral-sequence collapse proving the comparison isomorphism.  Its failure merely removes that sufficient argument; the comparison can still be an isomorphism for other reasons.  Hence the criterion is not necessary.
\end{solution}

% VI47-DETAILED-EXERCISE-20
\subsection*{Exercise VI.47.20}
\begin{solution}
The nonempty opens \(U_i\cap V_j\) form a cover.  Each lies in both \(U_i\) and \(V_j\), so this is a common refinement of the two original covers.
\end{solution}

% VI47-DETAILED-EXERCISE-21
\subsection*{Exercise VI.47.21}
\begin{solution}
For \(n=-4\), exponents \(\le-4\) and \(\ge0\) are killed.  The missing modes are
\[
t^{-3},t^{-2},t^{-1},
\]
which form a basis of \(\check H^1(\mathcal O(-4))\).
\end{solution}

% VI47-DETAILED-EXERCISE-22
\subsection*{Exercise VI.47.22}
\begin{solution}
If \(n\ge0\), the difference is \((n+1)-0=n+1\).  For \(n=-1\), both dimensions vanish.  For \(n\le-2\), the difference is \(0-(-n-1)=n+1\).  Thus the identity holds for every integer \(n\).
\end{solution}

% VI47-DETAILED-EXERCISE-23
\subsection*{Exercise VI.47.23}
\begin{solution}
Fixed-cover Cech cohomology depends on the chosen overlap complex, while derived sheaf cohomology is intrinsic.  They coincide under a comparison theorem such as Leray, but not automatically.  Therefore the replacement must be justified in each computation.
\end{solution}

% VI47-DETAILED-EXERCISE-24
\subsection*{Exercise VI.47.24}
\begin{solution}
Choose local lifts \(t_i\) of a global quotient section.  On overlaps \(t_j-t_i\) lies in the kernel sheaf and satisfies the triangle cocycle identity.  Changing the lifts changes this 1-cocycle by a coboundary, so its class is well defined and becomes the connecting obstruction in VI/48.
\end{solution}
