\chapter{Base Change}\label{ch:vi24-base-change}

Base change is the systematic operation of replacing the base of a scheme or morphism while
preserving all compatibility data.  If $X\to S$ is a scheme over $S$ and $S'\to S$ is another
morphism, the new scheme
\[
X_{S'}:=X\times_S S'
\]
is the same family viewed over the new base $S'$.  VI/23 constructed fiber products; this chapter
turns that construction into a working calculus.  In affine coordinates, base change is scalar
extension by tensor product.

\section{Definition and the base-change square}
Let $f:X\to S$ and $g:S'\to S$.  The \emph{base change of $X$ along $g$} is
\[
X_{S'}=X\times_S S'.
\]
The second projection gives the base-changed structure morphism
\[
f_{S'}:X_{S'}\longrightarrow S'.
\]
By definition the square
\[
\begin{tikzcd}
X_{S'} \arrow[r] \arrow[d,"f_{S'}"'] & X \arrow[d,"f"]\\
S' \arrow[r,"g"'] & S
\end{tikzcd}
\]
is Cartesian.

\section{Universal property after changing the base}
For every $T\to S'$, a morphism $T\to X_{S'}$ over $S'$ is equivalent to a morphism
$T\to X$ over $S$ whose structure map to $S$ factors through the prescribed $T\to S'\to S$.
Thus base change is not a new ad hoc construction: it is the universal pullback from VI/23, now
regarded as an operation depending on the map $S'\to S$.

\begin{proposition}[Identity base change]\label{prop:vi24-identity}
For every $X\to S$ there is a canonical isomorphism
\[
X\times_S S\cong X,
\]
where the second map $S\to S$ is the identity.
\end{proposition}

\begin{proposition}[Iterated base change]\label{prop:vi24-iterated}
If $S''\to S'\to S$, then there is a canonical isomorphism
\[
(X_{S'})_{S''}\cong X_{S''}.
\]
\end{proposition}

\input{chapters/ch24_base_change/figures/figure_02}

\section{Affine base change is scalar extension}
Suppose
\[
S=\Spec A,\qquad X=\Spec B,\qquad S'=\Spec A',
\]
with maps induced by $A\to B$ and $A\to A'$.  By the affine fiber-product formula,
\[
X_{S'}\cong\Spec(B\otimes_A A').
\]
The new coordinate ring is therefore obtained by extending scalars from $A$ to $A'$.

\begin{theorem}[Affine base-change formula]\label{thm:vi24-affine}
For an $A$-algebra $B$ and an $A$-algebra $A'$,
\[
\Spec B\times_{\Spec A}\Spec A'\cong\Spec(B\otimes_A A').
\]
\end{theorem}

\begin{example}[Field extension]\label{ex:vi24-expand-01}
Base changing $\mathbb A^1_{\mathbb R}$ to $\mathbb C$ gives $\mathbb A^1_{\mathbb C}$.  On rings this is $\mathbb C\otimes_{\mathbb R}\mathbb R[x]\cong\mathbb C[x]$.
\end{example}

\section{Changing coefficients in equations}
Let $k\to K$ be a field extension and
\[
X=\Spec k[x_1,\dots,x_n]/I.
\]
Then
\[
X_K:=X\times_{\Spec k}\Spec K
\cong
\Spec\bigl(K[x_1,\dots,x_n]/IK[x_1,\dots,x_n]\bigr).
\]
Thus base change from $k$ to $K$ leaves the equations formally unchanged while allowing their
coefficients, factorizations, and solutions to live in the larger field.  This is why geometric
features can become visible only after extending the base field.

\input{chapters/ch24_base_change/figures/figure_04}

\section{Quotients commute with scalar extension}
If $I\subseteq B$ is an ideal and $A\to A'$ is a ring map, right exactness of tensor product gives
\[
(B/I)\otimes_A A'\cong
(B\otimes_A A')/\operatorname{im}(I\otimes_A A').
\]
When $B$ is an $A$-algebra and the ideal is extended in the usual way, this is written
\[
(B/I)\otimes_AA'\cong(B\otimes_AA')/I(B\otimes_AA').
\]
This identity is the algebra behind base change of affine closed subschemes.

\section{Localization commutes with base change}
If $b\in B$, then
\[
B_b\otimes_AA'\cong(B\otimes_AA')_{b\otimes1}.
\]
Geometrically, pulling back the principal open $D(b)\subseteq\Spec B$ gives the principal open
$D(b\otimes1)$ in the base-changed affine scheme.  This is the local algebra behind stability of
open immersions under base change.

\begin{example}[Splitting after base change]\label{ex:vi24-expand-02}
The real scheme $\operatorname{Spec}\mathbb R[x]/(x^2+1)$ becomes $\operatorname{Spec}\mathbb C[x]/((x-i)(x+i))$ after base change to $\mathbb C$, revealing two complex points.
\end{example}

\section{Base change of morphisms}
Suppose $u:X\to Y$ is a morphism of schemes over $S$.  For $S'\to S$, the universal property gives
an induced morphism
\[
u_{S'}:X_{S'}\longrightarrow Y_{S'}.
\]
Identity morphisms and compositions are preserved:
\[
(\operatorname{id}_X)_{S'}=\operatorname{id}_{X_{S'}},
\qquad
(v\circ u)_{S'}=v_{S'}\circ u_{S'}.
\]
Hence base change defines a functor from schemes over $S$ to schemes over $S'$.

\section{Properties visibly preserved by base change}
Several fundamental classes of morphisms are stable for formal or affine-local reasons.
\begin{itemize}
\item Isomorphisms remain isomorphisms after any base change.
\item Open immersions remain open immersions, by localization on affine charts.
\item Closed immersions remain closed immersions, because quotient rings remain quotient rings after
scalar extension.
\item Monomorphisms remain monomorphisms, directly from the universal property.
\end{itemize}
Later chapters treat deeper finiteness and geometric properties.  VI/24 establishes the language and
the elementary stable classes without consuming VI/26.

\section{Products commute with base change}
If $X\to S$ and $Y\to S$, then for every $S'\to S$ there is a canonical isomorphism
\[
(X\times_S Y)\times_S S'
\cong
X_{S'}\times_{S'}Y_{S'}.
\]
Both sides represent triples of compatible maps to $X$, $Y$, and $S'$.  In affine coordinates the
identity is the associativity of tensor products.

\input{chapters/ch24_base_change/figures/figure_07}

\begin{example}[Base-changed morphism]\label{ex:vi24-expand-03}
If $X\to S$ is a morphism and $S'\to S$ is any map, the base change is $X_{S'}=X\times_S S'\to S'$.  Every construction is controlled by the same pullback square.
\end{example}
% END VOL06-EXPANSION VI24-example-03
\section{Diagonals and graphs under base change}
For $X\to S$, the diagonal of the base-changed scheme is canonically the pullback of the original
diagonal:
\[
\Delta_{X_{S'}/S'}\cong(\Delta_{X/S})_{S'}.
\]
Likewise, if $u:X\to Y$ is over $S$, then the graph of $u_{S'}$ is the base change of the graph of
$u$.  These identities are formal consequences of the Cartesian constructions in VI/23.

\input{chapters/ch24_base_change/figures/figure_08}

\section{Field extension can reveal hidden geometry}
The underlying shape of a scheme can change after extending the field.  For example,
\[
\Spec\mathbb C\times_{\Spec\mathbb R}\Spec\mathbb C
=
\Spec(\mathbb C\otimes_{\mathbb R}\mathbb C)
\cong
\Spec(\mathbb C\times\mathbb C),
\]
so one point over $\mathbb R$ splits into two points after base change to $\mathbb C$.  Conversely,
purely inseparable extensions can create nilpotents after base change.  Base change therefore
preserves the pullback structure, not every topological or reducedness property.

\input{chapters/ch24_base_change/figures/figure_09}

\section{Boundary with fibers and finiteness}
A fiber is a special base change
\[
X_s=X\times_S\Spec\kappa(s),
\]
but VI/25 is devoted to the detailed geometry of these schemes and their residue-field extensions.
Similarly, VI/26 studies finite-type and Noetherian conditions systematically.  Here these appear only
as forward references; the chapter's center is the operation of changing the base itself.

\section{Chapter synthesis}
Base change takes a geometric object over $S$ and transports it to a new base $S'$ by a Cartesian
square.  In affine charts this is exactly extension of scalars.  Identity and composition laws make
base change functorial; quotients and localizations explain the stability of closed and open
immersions; and field extension examples show why the operation is geometrically powerful.  The next
chapter specializes this calculus to fibers over points.

\subsection*{Standard computations}

\begin{exercise}[Affine line]\label{exr:vi24-expand-01}
Base change $\mathbb A^1_k$ along a field extension $K/k$.
\end{exercise}
\begin{hint}
Use base change of schemes and tensor products.
\end{hint}
\begin{solution}
It becomes $\mathbb A^1_K$.
\end{solution}

\begin{exercise}[Polynomial]\label{exr:vi24-expand-02}
Compute $K\otimes_k k[x]$.
\end{exercise}
\begin{hint}
Use base change of schemes and tensor products.
\end{hint}
\begin{solution}
$K[x]$.
\end{solution}

\begin{exercise}[Quadratic]\label{exr:vi24-expand-03}
What happens to $x^2+1$ from $\mathbb R$ to $\mathbb C$?
\end{exercise}
\begin{hint}
Use base change of schemes and tensor products.
\end{hint}
\begin{solution}
It factors as $(x-i)(x+i)$.
\end{solution}

\begin{exercise}[Notation]\label{exr:vi24-expand-04}
What is $X_{S'}$?
\end{exercise}
\begin{hint}
Use base change of schemes and tensor products.
\end{hint}
\begin{solution}
$X\times_S S'$.
\end{solution}

\begin{exercise}[Identity]\label{exr:vi24-expand-05}
What is base change along $S\to S$?
\end{exercise}
\begin{hint}
Use base change of schemes and tensor products.
\end{hint}
\begin{solution}
The original $X\to S$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Affine base change]\label{exr:vi24-expand-06}
Prove: Prove base change of $\operatorname{Spec}B\to\operatorname{Spec}A$ along $\operatorname{Spec}A'\to\operatorname{Spec}A$ is $\operatorname{Spec}(B\otimes_AA')$.
\end{exercise}
\begin{hint}
Work directly from base change of schemes and tensor products.
\end{hint}
\begin{solution}
Apply the affine fiber-product formula.
\end{solution}

\begin{exercise}[Composition]\label{exr:vi24-expand-07}
Prove: Prove successive base changes are canonically equivalent to base change along the composite.
\end{exercise}
\begin{hint}
Work directly from base change of schemes and tensor products.
\end{hint}
\begin{solution}
Use associativity of fiber products or tensor products.
\end{solution}

\begin{exercise}[Property transport example]\label{exr:vi24-expand-08}
Prove: Prove an open immersion remains an open immersion after base change.
\end{exercise}
\begin{hint}
Work directly from base change of schemes and tensor products.
\end{hint}
\begin{solution}
Check locally where it is localization; tensoring a localization gives the corresponding localization after base change.
\end{solution}

\begin{exercise}[Closed immersion stability]\label{exr:vi24-expand-09}
Prove: Prove a closed immersion remains closed after base change.
\end{exercise}
\begin{hint}
Work directly from base change of schemes and tensor products.
\end{hint}
\begin{solution}
Affinely, a surjection $A\to B$ remains surjective after tensoring with the new base.
\end{solution}
