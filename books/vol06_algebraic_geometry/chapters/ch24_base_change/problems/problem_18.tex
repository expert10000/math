\begin{problem}[Base change of a principal open]\label{prob:vi24-p18}
For $X=\Spec B$ and $D(b)\subseteq X$, identify its pullback to $X_{A'}$.
\end{problem}
\begin{solution}
It is
\[
D(b\otimes1)\subseteq\Spec(B\otimes_AA')
\]
because localization commutes with tensoring:
\[
B_b\otimes_AA'\cong(B\otimes_AA')_{b\otimes1}.
\]
\end{solution}
