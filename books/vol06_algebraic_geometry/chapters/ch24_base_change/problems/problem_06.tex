\begin{problem}[Base change of equations]\label{prob:vi24-p06}
Let $X=\Spec k[x_1,\dots,x_n]/I$ and $k\subseteq K$. Compute $X_K$.
\end{problem}
\begin{solution}
Using the quotient and polynomial-ring tensor identities,
\[
X_K\cong\Spec\left(K[x_1,\dots,x_n]/IK[x_1,\dots,x_n]\right).
\]
Thus the same equations define the scalar extension over $K$.
\end{solution}
