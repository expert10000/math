\begin{problem}[Challenge: Finite separable splitting]\label{chall:vi24-02}
Let $L/k$ be finite separable. Show after base change to an algebraic closure that $L\otimes_k\overline k$ is a product indexed by the $k$-embeddings $L\hookrightarrow\overline k$. Interpret the components geometrically.
\end{problem}
