\paragraph{Hint for VI.11.E01.} If there were two distinct maximal ideals, use their sum to write $1=a+b$ with $a,b$ both nonunits.

\paragraph{Hint for VI.11.E02.} A denominator is already nonzero at $a$; a fraction is invertible exactly when its numerator is also nonzero at $a$.

\paragraph{Hint for VI.11.E03.} Try the candidate inverse $s/a$ when $a\notin\mathfrak p$; for the converse use the prime ideal condition after clearing denominators.

\paragraph{Hint for VI.11.E04.} Use the prime correspondence for localization with $S=A\setminus\mathfrak p$.

\paragraph{Hint for VI.11.E05.} Localize the quotient $A/\mathfrak p$ at the image of $A\setminus\mathfrak p$.

\paragraph{Hint for VI.11.E06.} At $(0)$ invert every nonzero polynomial; at $(t-a)$ invert precisely the polynomials nonvanishing at $a$.

\paragraph{Hint for VI.11.E07.} The quotient $\RR[t]/(t^2+1)$ is already a field.

\paragraph{Hint for VI.11.E08.} A fraction $a/b$ is a unit in $\ZZ_{(p)}$ exactly when $p\nmid a$.

\paragraph{Hint for VI.11.E09.} Use that $\FF_q[t]/(f)$ is a field with $q^d$ elements.

\paragraph{Hint for VI.11.E10.} First identify $k[x,y]/(y-x^2)$ with $k[x]$.

\paragraph{Hint for VI.11.E11.} The classes of $x$ and $y$ remain nonzero locally at $(x,y)$ but their product is zero.

\paragraph{Hint for VI.11.E12.} Because $x\equiv a\neq0$, the class of $x$ becomes a unit and forces $y=0$.

\paragraph{Hint for VI.11.E13.} At $(x)$ the element $y$ is inverted; at $(y)$ the element $x$ is inverted.

\paragraph{Hint for VI.11.E14.} An element $a+b\varepsilon$ is a unit iff $a\neq0$.

\paragraph{Hint for VI.11.E15.} Use the maximal ideals corresponding to prime divisors of $n$ in $\ZZ/n\ZZ$.

\paragraph{Hint for VI.11.E16.} Both localizations invert the same elements of $A$ once $f$ is already invertible.

\paragraph{Hint for VI.11.E17.} Localize the exact sequence $I\to A\to A/I\to0$ at $A\setminus\mathfrak p$.

\paragraph{Hint for VI.11.E18.} After quotient-localization, quotient again by the maximal ideal and simplify.

\paragraph{Hint for VI.11.E19.} Recall the definition of equality of fractions in a localization.

\paragraph{Hint for VI.11.E20.} Every $a\notin\mathfrak p$ maps outside $\mathfrak q$, so the universal property of localization applies.

\paragraph{Hint for VI.11.E21.} A local map sends the maximal ideal into the maximal ideal and therefore descends to quotient fields.

\paragraph{Hint for VI.11.E22.} A prime of a product contains one idempotent factor; localization makes the complementary idempotent a unit.

\paragraph{Hint for VI.11.E23.} Modulo $(2,x-1)$, both $2$ and $x-1$ vanish; the quotient is $\FF_2$.

\paragraph{Hint for VI.11.E24.} Smaller primes have larger complements, so moving to a smaller prime inverts more elements.
