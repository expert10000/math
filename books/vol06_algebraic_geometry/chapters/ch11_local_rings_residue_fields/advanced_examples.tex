\begin{example}[Advanced: a discrete valuation model]
The local ring
\[
R=k[t]_{(t)}
\]
has maximal ideal $(t)$, and every nonzero element can be written uniquely as
\[
t^n u
\]
with $n\ge0$ and $u$ a unit.  Consequently every nonzero ideal of $R$ is $(t^n)$ for some
$n$.  This is the basic example of a discrete valuation ring, a notion used later in divisor
theory.
\end{example}

\begin{example}[Advanced: the Gaussian prime $(1+i)$]
Let $A=\ZZ[i]$ and $\mathfrak p=(1+i)$.  Since
\[
\ZZ[i]/(1+i)\cong\FF_2,
\]
the localization $A_{\mathfrak p}$ is local with residue field $\FF_2$.  This is an arithmetic
analogue of localizing a polynomial ring at a closed point.
\end{example}

\begin{example}[Advanced: a mixed arithmetic-geometric point]
Let
\[
A=\ZZ[x],\qquad \mathfrak m=(2,x-1).
\]
Then $A/\mathfrak m\cong\FF_2$, so
\[
\kappa(\mathfrak m)=\FF_2.
\]
The local ring $\ZZ[x]_{(2,x-1)}$ simultaneously remembers arithmetic localization at $2$
and geometric localization near $x=1$.
\end{example}

\begin{example}[Advanced: a product ring localizes to one factor]
Let $A=k\times k$.  Its two prime ideals are $\mathfrak p_1=0\times k$ and
$\mathfrak p_2=k\times0$.  Localizing at $\mathfrak p_1$ makes $(1,0)$ a unit and annihilates
the other factor, giving
\[
A_{\mathfrak p_1}\cong k.
\]
Likewise $A_{\mathfrak p_2}\cong k$.  A disconnected global spectrum has field-valued local
rings at both points.
\end{example}

\begin{example}[Advanced: the cusp at its singular point]
Let
\[
A=k[t^2,t^3]\cong k[x,y]/(y^2-x^3)
\]
and let $\mathfrak m=(t^2,t^3)$.  The local ring
\[
A_{\mathfrak m}=k[t^2,t^3]_{(t^2,t^3)}
\]
has residue field $k$.  It embeds in the larger local ring $k[t]_{(t)}$, but $t$ itself is
not in $A_{\mathfrak m}$.  This is a first glimpse of normalization: the parametrizing line
has a simpler local ring than the cusp.
\end{example}

\begin{example}[Advanced: a specialization chain and local maps]
In $A=k[x,y]$ consider
\[
(0)\subset(x)\subset(x,y).
\]
The natural maps of local rings run in the opposite direction:
\[
A_{(x,y)}\longrightarrow A_{(x)}\longrightarrow A_{(0)}=k(x,y).
\]
Moving toward a more generic point means inverting more elements.
\end{example}

\begin{example}[Advanced: a nonreduced local Artin ring]
For
\[
R_n=k[\varepsilon]/(\varepsilon^n),\qquad n\ge2,
\]
there is one maximal ideal $(\bar\varepsilon)$ and residue field $k$.  The filtration
\[
R_n\supset(\bar\varepsilon)\supset\cdots\supset(\bar\varepsilon^{n-1})\supset0
\]
records $n-1$ layers of infinitesimal thickness, all invisible to the one-point spectrum and
to the residue field alone.
\end{example}

\begin{example}[Advanced: localization at a minimal prime in a reduced ring]
Let $A=k[x,y]/(xy)$ and $\mathfrak p=(\bar x)$.  Since $A$ is reduced and $\mathfrak p$ is
minimal, localization kills the other component and gives the field $k(y)$.  This illustrates
a general phenomenon: for a reduced ring, localization at a minimal prime is a field.
\end{example}

\begin{example}[Advanced: $p$-adic integers as a local ring]
The ring $\ZZ_p$ of $p$-adic integers is local with maximal ideal $p\ZZ_p$ and residue field
\[
\ZZ_p/p\ZZ_p\cong\FF_p.
\]
Unlike $\ZZ_{(p)}$, it is complete in the $p$-adic topology.  Completion itself is developed
in Volume V; here the example is included to compare two important local rings with the same
residue field.
\end{example}

\begin{example}[Advanced: local exact sequence at a point]
For any prime $\mathfrak p$ there is an exact sequence
\[
0\to\mathfrak pA_{\mathfrak p}
\to A_{\mathfrak p}
\to\kappa(\mathfrak p)
\to0.
\]
For $A=k[t]$ and $\mathfrak p=(t-a)$ this becomes
\[
0\to(t-a)k[t]_{(t-a)}
\to k[t]_{(t-a)}
\to k
\to0.
\]
It formalizes the statement that local functions modulo functions vanishing at the point are
just their values.
\end{example}
