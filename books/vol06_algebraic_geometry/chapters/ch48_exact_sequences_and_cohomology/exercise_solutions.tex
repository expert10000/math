% VI48-DETAILED-EXERCISE-01
\subsection*{Exercise VI.48.1}
\begin{solution}
For a short exact sequence \(0\to\mathcal F'\to\mathcal F\to\mathcal F''\to0\), injectivity on sections gives the first exactness.  If \(s\in\Gamma(U,\mathcal F)\) maps to zero, every germ of \(s\) lies in the kernel sheaf.  These local preimages agree because \(\mathcal F'\to\mathcal F\) is a monomorphism, so they glue uniquely to a section of \(\mathcal F'\).  Hence \(\Gamma(U,-)\) is left exact.
\end{solution}

% VI48-DETAILED-EXERCISE-02
\subsection*{Exercise VI.48.2}
\begin{solution}
A sheaf epimorphism is only locally surjective.  For example, in a divisor/line-bundle sequence \(0\to\mathcal L(-D)\to\mathcal L\to\mathcal L|_D\to0\), a section on \(D\) need not extend globally when its connecting class in \(H^1(X,\mathcal L(-D))\) is nonzero.  Thus the last sheaf map is an epimorphism although the induced map on global sections can fail to be surjective.
\end{solution}

% VI48-DETAILED-EXERCISE-03
\subsection*{Exercise VI.48.3}
\begin{solution}
On \(U_{ijk}\),
\[
a_{jk}-a_{ik}+a_{ij}
=(t_k-t_j)-(t_k-t_i)+(t_j-t_i)=0.
\]
Each difference lies in \(\mathcal F'\) because the two lifts have the same image in \(\mathcal F''\).  Hence \((a_{ij})\) is a Čech \(1\)-cocycle with values in the kernel.
\end{solution}

% VI48-DETAILED-EXERCISE-04
\subsection*{Exercise VI.48.4}
\begin{solution}
Replace \(t_i\) by \(t_i+b_i\), where \(b_i\in\mathcal F'(U_i)\).  Then
\[
a'_{ij}=(t_j+b_j)-(t_i+b_i)=a_{ij}+b_j-b_i.
\]
Thus \(a'-a=\delta b\), so the cohomology class is independent of the chosen local lifts.
\end{solution}

% VI48-DETAILED-EXERCISE-05
\subsection*{Exercise VI.48.5}
\begin{solution}
If \(t\in\Gamma(X,\mathcal F)\) is a global lift of \(s\), choose \(t_i=t|_{U_i}\).  Then \(t_j-t_i=0\) on every overlap, so the obstruction cocycle is zero.  Therefore \(\delta(s)=0\).
\end{solution}

% VI48-DETAILED-EXERCISE-06
\subsection*{Exercise VI.48.6}
\begin{solution}
If the obstruction class vanishes, write \(a_{ij}=b_j-b_i\).  Replacing \(t_i\) by \(t_i-b_i\) makes the corrected local lifts agree on every overlap.  The sheaf gluing axiom then produces a global lift of \(s\).  This proves the converse lifting criterion.
\end{solution}

% VI48-DETAILED-EXERCISE-07
\subsection*{Exercise VI.48.7}
\begin{solution}
The beginning is
\[
0\to H^0(\mathcal F')\to H^0(\mathcal F)\to H^0(\mathcal F'')
 \xrightarrow{\delta^0}H^1(\mathcal F')\to H^1(\mathcal F)\to H^1(\mathcal F'').
\]
The next nonzero arrow is \(\delta^1:H^1(\mathcal F'')\to H^2(\mathcal F')\).  This displays explicitly where failure of surjectivity in degree zero is sent.
\end{solution}

% VI48-DETAILED-EXERCISE-08
\subsection*{Exercise VI.48.8}
\begin{solution}
If \(H^1(X,\mathcal F')=0\), the connecting map
\[
H^0(X,\mathcal F'')\to H^1(X,\mathcal F')
\]
has zero target.  Exactness therefore says every quotient section lies in the image of \(H^0(X,\mathcal F)\); hence every quotient section lifts globally.
\end{solution}

% VI48-DETAILED-EXERCISE-09
\subsection*{Exercise VI.48.9}
\begin{solution}
A splitting gives \(\mathcal F\simeq\mathcal F'\oplus\mathcal F''\).  Cohomology preserves finite direct sums, so in every degree the long exact sequence decomposes into
\[
0\to H^q(\mathcal F')\to H^q(\mathcal F')\oplus H^q(\mathcal F'')\to H^q(\mathcal F'')\to0.
\]
Consequently every connecting homomorphism is zero.
\end{solution}

% VI48-DETAILED-EXERCISE-10
\subsection*{Exercise VI.48.10}
\begin{solution}
For a fixed intersection \(U_I\), a sheaf epimorphism need not give a surjection
\(\Gamma(U_I,\mathcal F)\to\Gamma(U_I,\mathcal F'')\).  Therefore the sequence of Čech cochain groups may fail to be termwise exact.  One must refine until local lifts exist on all required intersections, impose a lifting hypothesis, or pass to suitable resolutions.
\end{solution}

% VI48-DETAILED-EXERCISE-11
\subsection*{Exercise VI.48.11}
\begin{solution}
Let \(c''\) be a \(q\)-cocycle and choose a cochain \(c\) mapping to it.  Since \(d c''=0\), the cochain \(dc\) maps to zero and therefore comes from the kernel complex.  Moreover \(d(dc)=0\).  The class of this kernel-valued \((q+1)\)-cocycle is the connecting class; changing the lift changes it by a coboundary.
\end{solution}

% VI48-DETAILED-EXERCISE-12
\subsection*{Exercise VI.48.12}
\begin{solution}
Choose local lifts \(t_i\) in the upper short exact sequence and apply the middle vertical morphism.  These are local lifts of the image quotient section downstairs.  Their overlap differences are exactly the images of \(t_j-t_i\) under the left vertical map.  Thus the resulting Čech classes commute with the morphism, proving naturality of \(\delta^0\).
\end{solution}

% VI48-DETAILED-EXERCISE-13
\subsection*{Exercise VI.48.13}
\begin{solution}
The ideal sequence of an effective Cartier divisor is
\[
0\to\mathcal I_D\to\mathcal O_X\to\mathcal O_D\to0.
\]
Because \(D\) is Cartier, \(\mathcal I_D\) is invertible and is canonically written \(\mathcal O_X(-D)\).  Substitution gives the standard divisor exact sequence.
\end{solution}

% VI48-DETAILED-EXERCISE-14
\subsection*{Exercise VI.48.14}
\begin{solution}
Tensor
\[
0\to\mathcal O_X(-D)\to\mathcal O_X\to\mathcal O_D\to0
\]
by \(\mathcal L\).  A line bundle is locally free of rank one, hence flat, so tensoring preserves exactness.  The result is
\[
0\to\mathcal L(-D)\to\mathcal L\to\mathcal L|_D\to0.
\]
\end{solution}

% VI48-DETAILED-EXERCISE-15
\subsection*{Exercise VI.48.15}
\begin{solution}
For \(s_D\in H^0(D,\mathcal L|_D)\), the long exact sequence sends it to
\[
\delta(s_D)\in H^1(X,\mathcal L(-D)).
\]
Exactness says \(s_D\) extends to a global section of \(\mathcal L\) exactly when \(\delta(s_D)=0\).  In particular, vanishing of the whole \(H^1\)-group forces extension of every prescribed divisor section.
\end{solution}

% VI48-DETAILED-EXERCISE-16
\subsection*{Exercise VI.48.16}
\begin{solution}
A \(k\)-rational point \(p\) is an effective Cartier divisor of degree one on \(\mathbb P^1\), with \(\mathcal O(-p)\simeq\mathcal O(-1)\).  Tensoring its ideal sequence by \(\mathcal O(n)\) yields
\[
0\to\mathcal O(n-1)\to\mathcal O(n)\to k_p\to0.
\]
\end{solution}

% VI48-DETAILED-EXERCISE-17
\subsection*{Exercise VI.48.17}
\begin{solution}
A skyscraper sheaf is flabby: restrictions are either identities, maps to zero, or zero maps between zero groups.  VI/46 gives flabby acyclicity, hence
\[
H^q(\mathbb P^1,k_p)=0\qquad(q>0).
\]
\end{solution}

% VI48-DETAILED-EXERCISE-18
\subsection*{Exercise VI.48.18}
\begin{solution}
For \(n\ge0\), evaluation \(H^0(\mathcal O(n))\to k_p\) is surjective.  In the point-sequence long exact sequence, the connecting map \(k_p\to H^1(\mathcal O(n-1))\) is therefore zero, and exactness identifies \(H^1(\mathcal O(n-1))\) with \(H^1(\mathcal O(n))\).  Starting at \(H^1(\mathcal O(-1))=0\) proves vanishing for every \(n\ge-1\).
\end{solution}

% VI48-DETAILED-EXERCISE-19
\subsection*{Exercise VI.48.19}
\begin{solution}
For \(m\ge2\), apply the point sequence to \(\mathcal O(-m+1)\).  Both relevant \(H^0\)-groups vanish, so exactness gives
\[
0\to k\to H^1(\mathcal O(-m))\to H^1(\mathcal O(-m+1))\to0.
\]
Starting from \(h^1(\mathcal O(-1))=0\), induction gives \(h^1(\mathcal O(-m))=m-1\).
\end{solution}

% VI48-DETAILED-EXERCISE-20
\subsection*{Exercise VI.48.20}
\begin{solution}
From \(0\to\mathcal F\to\mathcal I\to\mathcal Q\to0\) with \(\mathcal I\) flabby, the groups \(H^q(\mathcal I)\) vanish for \(q>0\).  Hence for \(q\ge1\) the long exact sequence contains
\[
0\to H^q(\mathcal Q)\xrightarrow{\sim}H^{q+1}(\mathcal F)\to0.
\]
This is the dimension-shifting isomorphism.
\end{solution}

% VI48-DETAILED-EXERCISE-21
\subsection*{Exercise VI.48.21}
\begin{solution}
The degree-zero part is
\[
H^0(\mathcal I)\to H^0(\mathcal Q)\to H^1(\mathcal F)\to H^1(\mathcal I)=0.
\]
Therefore
\[
H^1(\mathcal F)\cong H^0(\mathcal Q)/\operatorname{im}H^0(\mathcal I).
\]
First cohomology is thus the cokernel measuring failure of global lifting through the flabby embedding.
\end{solution}

% VI48-DETAILED-EXERCISE-22
\subsection*{Exercise VI.48.22}
\begin{solution}
Assuming finite-dimensional cohomology and a finite vanishing range, truncate the long exact sequence beyond the last nonzero term.  The alternating sum of dimensions of a finite exact sequence is zero.  Grouping terms by sheaf gives
\[
\chi(\mathcal F)=\chi(\mathcal F')+\chi(\mathcal F'').
\]
\end{solution}

% VI48-DETAILED-EXERCISE-23
\subsection*{Exercise VI.48.23}
\begin{solution}
If \(\mathcal F'\) and \(\mathcal F\) are acyclic, exactness first makes \(H^0(\mathcal F)\to H^0(\mathcal F'')\) surjective and then forces all \(H^{>0}(\mathcal F'')\) to vanish.  If instead \(\mathcal F\) and \(\mathcal F''\) are acyclic, the kernel automatically has \(H^q=0\) for \(q\ge2\), but \(H^1(\mathcal F')\) vanishes only when the map on global sections is surjective.  The long exact sequence distinguishes these cases.
\end{solution}

% VI48-DETAILED-EXERCISE-24
\subsection*{Exercise VI.48.24}
\begin{solution}
Vanishing replaces neighboring groups in a long exact sequence by zeros.  Exactness then forces adjacent maps to be injective, surjective, or isomorphisms and often splits the long sequence into short exact pieces.  VI/49 supplies systematic vanishing hypotheses, turning the formal machinery of VI/48 into concrete dimension and extension calculations.
\end{solution}
