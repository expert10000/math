% VI48-DETAILED-PROBLEM-01
\subsection*{Problem VI.48.1}
\begin{solution}
Choose an open cover on which the quotient section \(s\) lifts to sections \(t_i\) of the middle sheaf.  The differences \(a_{ij}=t_j-t_i\) lie in the kernel and satisfy the Čech cocycle identity.  Replacing \(t_i\) by another lift adds a kernel-valued \(0\)-cochain and changes \(a\) by its coboundary, so \([a]\in H^1(\mathcal F')\) is canonical.  Finally, \([a]=0\) exactly when the lifts can be corrected to agree and glue globally.
\end{solution}

% VI48-DETAILED-PROBLEM-02
\subsection*{Problem VI.48.2}
\begin{solution}
Given a commutative diagram of short exact sequences, start with a quotient section upstairs and choose local lifts.  Applying the middle vertical map produces valid local lifts of the image section downstairs.  Overlap differences are carried by the left vertical map, hence the lower obstruction cocycle is the image of the upper one.  Therefore the square of degree-zero boundary maps commutes without invoking resolutions.
\end{solution}

% VI48-DETAILED-PROBLEM-03
\subsection*{Problem VI.48.3}
\begin{solution}
Tensor the effective Cartier divisor sequence by \(\mathcal L\):
\[
0\to\mathcal L(-D)\to\mathcal L\to\mathcal L|_D\to0.
\]
For \(s_D\in H^0(D,\mathcal L|_D)\), exactness gives
\[
s_D\text{ extends }\Longleftrightarrow \delta(s_D)=0\in H^1(X,\mathcal L(-D)).
\]
Thus \(H^1(X,\mathcal L(-D))=0\) is a uniform extension criterion.
\end{solution}

% VI48-DETAILED-PROBLEM-04
\subsection*{Problem VI.48.4}
\begin{solution}
Use
\[
0\to\mathcal O(n-1)\to\mathcal O(n)\to k_p\to0.
\]
For \(n\ge0\), evaluation on \(H^0\) is onto, so \(H^1(\mathcal O(n-1))\simeq H^1(\mathcal O(n))\); from \(H^1(\mathcal O(-1))=0\) this gives vanishing for \(n\ge-1\).  For \(n<0\), the relevant \(H^0\)-groups vanish and each downward step contributes one copy of \(k\), giving \(h^1(\mathcal O(n))=-n-1\) for \(n\le-2\).
\end{solution}

% VI48-DETAILED-PROBLEM-05
\subsection*{Problem VI.48.5}
\begin{solution}
A sheaf splitting identifies the middle term with \(\mathcal F'\oplus\mathcal F''\).  Applying cohomology gives a direct sum in every degree, and the projection onto the quotient factor is already surjective.  Thus all obstruction/connecting maps vanish and the long exact sequence decomposes into split short exact sequences degree by degree.
\end{solution}

% VI48-DETAILED-PROBLEM-06
\subsection*{Problem VI.48.6}
\begin{solution}
In
\[
0\to\mathcal F\to\mathcal I\to\mathcal Q\to0
\]
with \(\mathcal I\) flabby, \(H^q(\mathcal I)=0\) for all \(q>0\).  Therefore
\[
H^{q+1}(\mathcal F)\cong H^q(\mathcal Q)\qquad(q\ge1),
\]
and in degree one
\[
H^1(\mathcal F)\cong H^0(\mathcal Q)/\operatorname{im}H^0(\mathcal I).
\]
Iterating this along a flabby resolution expresses all higher cohomology through successive cokernels.
\end{solution}

% VI48-DETAILED-PROBLEM-07
\subsection*{Problem VI.48.7}
\begin{solution}
The cover-lifting hypothesis gives a short exact sequence of Čech cochain complexes.  Its connecting map is the algebraic lift--differentiate--descend construction.  If the cover is Leray for all three sheaves, the natural Čech-to-derived comparison maps are isomorphisms.  Naturality of those comparison maps identifies the Čech boundary with the derived sheaf-cohomology boundary.
\end{solution}

% VI48-DETAILED-PROBLEM-08
\subsection*{Problem VI.48.8}
\begin{solution}
The divisor long exact sequence begins
\[
0\to H^0(\mathcal L(-D))\to H^0(\mathcal L)\to H^0(\mathcal L|_D)
\to H^1(\mathcal L(-D)).
\]
Hence the kernel of restriction is exactly \(H^0(\mathcal L(-D))\).  If \(H^1(\mathcal L(-D))=0\), the restriction map is surjective, so its cokernel is zero.
\end{solution}

% VI48-DETAILED-PROBLEM-09
\subsection*{Problem VI.48.9}
\begin{solution}
The target of the first boundary \(H^0(\mathcal F'')\to H^1(\mathcal F')\) is zero, so global sections are right-exact for this sequence.  In every positive degree, \(H^q(\mathcal F'')\) is trapped between vanishing groups from \(\mathcal F'\) and \(\mathcal F\).  Thus \(\mathcal F''\) is acyclic and
\[
0\to\Gamma(\mathcal F')\to\Gamma(\mathcal F)\to\Gamma(\mathcal F'')\to0
\]
is exact.
\end{solution}

% VI48-DETAILED-PROBLEM-10
\subsection*{Problem VI.48.10}
\begin{solution}
Euler characteristic is additive in short exact sequences.  Since a point skyscraper \(k_p\) has \(\chi(k_p)=1\), the point sequence gives
\[
\chi(\mathcal O(n))=\chi(\mathcal O(n-1))+1.
\]
Starting from \(\chi(\mathcal O)=1\), one obtains \(\chi(\mathcal O(n))=n+1\) recursively.
\end{solution}

% VI48-DETAILED-PROBLEM-11
\subsection*{Problem VI.48.11}
\begin{solution}
A sheaf epimorphism ensures local lifts, not a global lift over \(X\).  The long exact sequence gives
\[
\operatorname{coker}\bigl[H^0(\mathcal F)\to H^0(\mathcal F'')\bigr]
\simeq \operatorname{im}\delta^0\subseteq H^1(\mathcal F').
\]
Thus the failure of surjectivity on global sections is represented exactly by nonzero connecting obstruction classes.
\end{solution}

% VI48-DETAILED-PROBLEM-12
\subsection*{Problem VI.48.12}
\begin{solution}
A section of \(\mathcal O_X(D)|_D\) prescribes principal-part data along \(D\).  Locally one can realize it by meromorphic sections of \(\mathcal O_X(D)\); on overlaps these local choices differ by regular functions, producing an \(\mathcal O_X\)-valued Čech \(1\)-cocycle.  Its cohomology class is the connecting image and vanishes precisely when the local meromorphic sections glue to one global section with the prescribed polar data.
\end{solution}
