\begin{hint}[E1]
Intersect all closed sets $V(I)$ that contain $\mathfrak p$ and use
$I\subseteq\mathfrak p$.
\end{hint}

\begin{hint}[E2]
Substitute the closure formula from E1 into the definition of specialization.
\end{hint}

\begin{hint}[E3]
If $\mathfrak p\in V(I)$ and $\mathfrak p\subseteq\mathfrak q$, compare $I$ with
$\mathfrak q$.
\end{hint}

\begin{hint}[E4]
Take complements and use E3.
\end{hint}

\begin{hint}[E5]
For $\mathfrak p\neq\mathfrak q$, choose an element belonging to one prime but not the
other and use a principal open.
\end{hint}

\begin{hint}[E6]
A space is $T_1$ exactly when every singleton is closed.
\end{hint}

\begin{hint}[E7]
The closure of $\mathfrak p$ consists of all primes containing $\mathfrak p$.
\end{hint}

\begin{hint}[E8]
Use $A/\mathfrak p$ being a domain for irreducibility, then use E1 and the $T_0$ property.
\end{hint}

\begin{hint}[E9]
Replace $I$ by $\sqrt I$ and use the prime-ideal criterion for irreducibility.
\end{hint}

\begin{hint}[E10]
Apply E9 to $I=(0)$.
\end{hint}

\begin{hint}[E11]
Irreducible components are maximal irreducible closed subsets.  Inclusion of $V(\mathfrak p)$
reverses inclusion of primes.
\end{hint}

\begin{hint}[E12]
Use the prime-ideal list $(0),(2),(3),(5),\dots$ and containment.
\end{hint}

\begin{hint}[E13]
Use that $k[t]$ is a PID and nonzero prime ideals are generated by irreducibles and are
maximal.
\end{hint}

\begin{hint}[E14]
Irreducible real polynomials have degree one or two; the quadratic ones have negative
discriminant.
\end{hint}

\begin{hint}[E15]
A quotient by an irreducible polynomial is a field with $p^d$ elements.
\end{hint}

\begin{hint}[E16]
Compute the quotient by $(\bar x)$ and $(\bar y)$, then use containment into
$(\bar x,\bar y)$.
\end{hint}

\begin{hint}[E17]
Every nilpotent lies in $(\bar x)$, and the quotient by $(\bar x)$ is $k[y]$.
\end{hint}

\begin{hint}[E18]
If $\eta$ were outside $U$, the closed complement would contain the closure of $\eta$.
\end{hint}

\begin{hint}[E19]
Use the order-preserving bijection between primes of $A/I$ and primes of $A$ containing
$I$, together with the quotient-spectrum homeomorphism.
\end{hint}

\begin{hint}[E20]
If $\mathfrak q_1\subseteq\mathfrak q_2$ in $B$, contract both ideals to $A$.
\end{hint}
