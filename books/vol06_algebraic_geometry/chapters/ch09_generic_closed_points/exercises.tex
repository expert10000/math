\begin{exercise}[VI.09.E1 - Closure of a point]
For $\mathfrak p\in\Spec A$, prove directly that
\[
\overline{\{\mathfrak p\}}=V(\mathfrak p).
\]
\end{exercise}

\begin{exercise}[VI.09.E2 - Specialization order]
Show that $\mathfrak q$ is a specialization of $\mathfrak p$ if and only if
\[
\mathfrak p\subseteq\mathfrak q.
\]
\end{exercise}

\begin{exercise}[VI.09.E3 - Closed sets and specialization]
Prove that every Zariski closed subset of $\Spec A$ is stable under specialization.
\end{exercise}

\begin{exercise}[VI.09.E4 - Open sets and generalization]
Prove that every Zariski open subset of $\Spec A$ is stable under generalization.
\end{exercise}

\begin{exercise}[VI.09.E5 - The $T_0$ property]
Prove that $\Spec A$ is $T_0$.
\end{exercise}

\begin{exercise}[VI.09.E6 - The $T_1$ criterion]
Show that $\Spec A$ is $T_1$ if and only if every prime ideal is maximal.
\end{exercise}

\begin{exercise}[VI.09.E7 - Closed points]
Show directly from the closure formula that
\[
\mathfrak p\text{ is closed}\Longleftrightarrow\mathfrak p\text{ is maximal}.
\]
\end{exercise}

\begin{exercise}[VI.09.E8 - Generic point of $V(\mathfrak p)$]
If $\mathfrak p$ is prime, prove that $\mathfrak p$ is the unique generic point of
$V(\mathfrak p)$.
\end{exercise}

\begin{exercise}[VI.09.E9 - Irreducible $V(I)$]
Prove that $V(I)$ is irreducible if and only if $\sqrt I$ is prime, and identify its
generic point.
\end{exercise}

\begin{exercise}[VI.09.E10 - Generic point of the whole spectrum]
Show that $\Spec A$ has a generic point if and only if $\sqrt{(0)}$ is prime.  Identify
the generic point.
\end{exercise}

\begin{exercise}[VI.09.E11 - Minimal primes and components]
Show that minimal prime ideals correspond to irreducible components of $\Spec A$ and are
their generic points.
\end{exercise}

\begin{exercise}[VI.09.E12 - Arithmetic specializations]
Describe all generalizations and specializations of $(0)$ and $(p)$ in $\Spec\ZZ$.
\end{exercise}

\begin{exercise}[VI.09.E13 - Closed points of the affine line]
Let $k$ be any field.  Show that the closed points of $\Spec k[t]$ are precisely the
ideals $(f)$ with $f$ irreducible.
\end{exercise}

\begin{exercise}[VI.09.E14 - Real closed points]
List the closed points of $\Spec\RR[t]$ and distinguish the $\RR$-rational ones.
\end{exercise}

\begin{exercise}[VI.09.E15 - Finite-field closed points]
For an irreducible $f\in\FF_p[t]$ of degree $d$, show $(f)$ is a closed point and
\[
\FF_p[t]/(f)\cong\FF_{p^d}.
\]
Which such points are $\FF_p$-rational?
\end{exercise}

\begin{exercise}[VI.09.E16 - Crossing axes]
For $R=k[x,y]/(xy)$, show that $(\bar x)$ and $(\bar y)$ are the component-generic points
and that $(\bar x,\bar y)$ is a closed specialization of both.
\end{exercise}

\begin{exercise}[VI.09.E17 - A nonreduced generic point]
For $A=k[x,y]/(x^2)$, prove that $\sqrt{(0)}=(\bar x)$ is prime and hence is the generic
point of $\Spec A$.
\end{exercise}

\begin{exercise}[VI.09.E18 - Nonempty opens in an irreducible space]
Let $X$ be irreducible with generic point $\eta$.  Prove that every nonempty open
$U\subseteq X$ contains $\eta$ and that $\eta$ is generic in $U$.
\end{exercise}

\begin{exercise}[VI.09.E19 - Quotient and generic points]
Let $I\subseteq A$.  Under $\Spec(A/I)\cong V_A(I)$, show that a prime
$\mathfrak p/I$ is generic for an irreducible closed subset of $\Spec(A/I)$ exactly when
$\mathfrak p$ is generic for the corresponding irreducible closed subset of $V_A(I)$.
\end{exercise}

\begin{exercise}[VI.09.E20 - Maps preserve specialization]
For a ring map $\varphi:A\to B$, prove that the induced map
\[
\varphi^\ast:\Spec B\to\Spec A
\]
preserves specialization.
\end{exercise}
