\begin{example}[A field: one point that is both generic and closed]
Let $K$ be a field.  Then
\[
\Spec K=\{(0)\}.
\]
The zero ideal is maximal and prime, so the unique point is closed.  Its closure is the
whole one-point space, so it is also generic.  Thus ``generic'' and ``closed'' are not
mutually exclusive labels; they describe opposite closure behaviors only when the space
has more than one point.
\end{example}

\begin{example}[The affine line over an algebraically closed field]
Let $k$ be algebraically closed.  The points of $\Spec k[t]$ are
\[
(0)
\qquad\text{and}\qquad
(t-a),\ a\in k.
\]
The point $(0)$ is generic because $k[t]$ is a domain.  Each $(t-a)$ is maximal and hence
closed.  Moreover
\[
(0)\subset(t-a),
\]
so every classical point is a specialization of the generic point.
\end{example}

\begin{example}[The real affine line has non-rational closed points]
In $\Spec\RR[t]$, the nonzero prime ideals are generated by irreducible polynomials.  Thus
\[
(t-a),\qquad a\in\RR,
\]
are closed real-rational points, but so are ideals such as
\[
(t^2+1).
\]
The latter is maximal because
\[
\RR[t]/(t^2+1)\cong\CC
\]
is a field.  It is a closed point but not an $\RR$-rational point.
\end{example}

\begin{example}[The affine line over a finite field]
In $\Spec\FF_p[t]$, every irreducible polynomial $f(t)$ determines a closed point $(f)$.
If $\deg f=d$, then
\[
\FF_p[t]/(f)\cong\FF_{p^d}.
\]
Only the degree-one irreducibles correspond to $\FF_p$-rational points.  The spectrum has
far more closed points than the $p$ elements of the finite field.
\end{example}

\begin{example}[The arithmetic line $\Spec\ZZ$]
The points are
\[
(0),\ (2),\ (3),\ (5),\ldots.
\]
The point $(0)$ is generic, while every $(p)$ is closed.  Since
\[
(0)\subset(p),
\]
the generic point specializes to every prime-number point.
\end{example}

\begin{example}[Plane, line, point]
In $\Spec k[x,y]$,
\[
(0)\subset(x)\subset(x,y).
\]
Their closures are
\[
\Spec k[x,y],\qquad V(x),\qquad \{(x,y)\},
\]
respectively.  One chain of prime ideals therefore records three geometric scales:
the whole plane, a line, and a point.
\end{example}

\begin{example}[A parabola and its generic point]
Assume $k$ is algebraically closed and set
\[
\mathfrak p=(y-x^2)\subset k[x,y].
\]
Since
\[
k[x,y]/(y-x^2)\cong k[x]
\]
is a domain, $\mathfrak p$ is prime.  Hence $V(y-x^2)$ is irreducible and
$\mathfrak p$ is its generic point.  Every closed point
\[
(x-a,y-a^2)
\]
on the parabola contains $\mathfrak p$, so it is a specialization of the parabola's
generic point.
\end{example}

\begin{example}[Two crossing axes]
Let
\[
R=k[x,y]/(xy).
\]
The minimal primes are $(\bar x)$ and $(\bar y)$, so the two irreducible components have
distinct generic points.  The crossing point
\[
(\bar x,\bar y)
\]
contains both minimal primes and is therefore a closed specialization of both generic
points.
\end{example}

\begin{example}[A product of two fields]
For
\[
A=k\times k,
\]
the spectrum consists of two points,
\[
k\times0,\qquad 0\times k.
\]
Both are maximal, hence closed.  Each singleton is also an irreducible component, so each
point is generic for its own component.  The space is discrete and has no generic point
for the whole two-point spectrum.
\end{example}

\begin{example}[Dual numbers]
Let
\[
A=k[\varepsilon]/(\varepsilon^2).
\]
There is one prime ideal,
\[
(\bar\varepsilon).
\]
It is maximal, so the unique point is closed; it is also generic for the whole one-point
spectrum.  The nilpotent element $\bar\varepsilon$ changes the ring but not this point-set
topology, a phenomenon analyzed systematically in VI/10.
\end{example}

\begin{example}[A nonreduced irreducible spectrum whose generic point is nonzero]
Let
\[
A=k[x,y]/(x^2).
\]
The nilradical is
\[
\sqrt{(0)}=(\bar x).
\]
The quotient
\[
A/(\bar x)\cong k[y]
\]
is a domain, so $(\bar x)$ is prime.  Hence $\Spec A$ is irreducible with generic point
$(\bar x)$.  The zero ideal is not prime because $\bar x^2=0$ with $\bar x\neq0$.
\end{example}

\begin{example}[A generic point survives every nonempty open]
Let $A=k[x,y]$ and consider the generic point $(0)$.  Every nonempty principal open
$D(f)$ with $f\neq0$ contains $(0)$ because $f\notin(0)$.  Since principal opens form a
basis, every nonempty open subset of $\Spec k[x,y]$ contains $(0)$.
\end{example}

\begin{example}[Quotient spectra preserve the generic-point dictionary]
Let $A=k[x,y]$ and $I=(xy)$.  Under
\[
\Spec(A/I)\cong V_A(I),
\]
the minimal primes
\[
(\bar x),\qquad(\bar y)
\]
correspond to the primes $(x)$ and $(y)$ of $A$ containing $I$.  Thus the generic points
of the two components are transported exactly by the quotient-spectrum correspondence.
\end{example}

\begin{example}[Contraction preserves specialization]
Consider
\[
\varphi:k[t]\longrightarrow k[x],\qquad t\longmapsto x^2.
\]
In $\Spec k[x]$ one has
\[
(0)\subset(x-a).
\]
Contraction gives
\[
(0)\subseteq\varphi^{-1}(x-a).
\]
If $a\neq0$, the contraction is $(t-a^2)$; if $a=0$, it is $(t)$.  Thus the induced map
on spectra sends specialization relations to specialization relations.
\end{example}
