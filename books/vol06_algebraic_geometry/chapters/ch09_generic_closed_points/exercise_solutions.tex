\begin{solution}[E1]
A closed set $V(I)$ contains $\mathfrak p$ iff $I\subseteq\mathfrak p$.  Whenever this
holds,
\[
V(\mathfrak p)\subseteq V(I).
\]
Thus $V(\mathfrak p)$ is contained in every closed set containing $\mathfrak p$, and since
it is itself closed and contains $\mathfrak p$,
\[
\overline{\{\mathfrak p\}}=V(\mathfrak p).
\]
\end{solution}

\begin{solution}[E2]
By definition, $\mathfrak q$ is a specialization of $\mathfrak p$ iff
\[
\mathfrak q\in\overline{\{\mathfrak p\}}.
\]
By E1 this is equivalent to $\mathfrak q\in V(\mathfrak p)$, hence to
\[
\mathfrak p\subseteq\mathfrak q.
\]
\end{solution}

\begin{solution}[E3]
Let $\mathfrak p\in V(I)$ and suppose $\mathfrak p\subseteq\mathfrak q$.  Then
$I\subseteq\mathfrak p\subseteq\mathfrak q$, so $\mathfrak q\in V(I)$.
\end{solution}

\begin{solution}[E4]
Let $U$ be open, $\mathfrak q\in U$, and $\mathfrak p\subseteq\mathfrak q$.  If
$\mathfrak p\notin U$, then $\mathfrak p$ belongs to the closed complement.  By E3 the
specialization $\mathfrak q$ would also belong to the complement, contradiction.  Hence
$\mathfrak p\in U$.
\end{solution}

\begin{solution}[E5]
Take distinct primes $\mathfrak p,\mathfrak q$.  Assume there is
$f\in\mathfrak q\setminus\mathfrak p$.  Then
\[
\mathfrak p\in D(f),\qquad \mathfrak q\notin D(f).
\]
Thus the two points have different open neighborhoods, proving the $T_0$ property.
\end{solution}

\begin{solution}[E6]
A space is $T_1$ iff every singleton is closed.  In $\Spec A$, E7 below shows
$\{\mathfrak p\}$ is closed iff $\mathfrak p$ is maximal.  Therefore $\Spec A$ is $T_1$
iff every prime ideal is maximal.
\end{solution}

\begin{solution}[E7]
By E1,
\[
\overline{\{\mathfrak p\}}
=
\{\mathfrak q:\mathfrak p\subseteq\mathfrak q\}.
\]
This equals $\{\mathfrak p\}$ iff no prime properly contains $\mathfrak p$.  Since
$\mathfrak p$ is prime, this is equivalent to maximality.
\end{solution}

\begin{solution}[E8]
Because $A/\mathfrak p$ is a domain,
\[
V(\mathfrak p)\cong\Spec(A/\mathfrak p)
\]
is irreducible.  By E1 the closure of $\mathfrak p$ is $V(\mathfrak p)$, so
$\mathfrak p$ is generic.  If another point had the same closure, the $T_0$ property would
force equality.
\end{solution}

\begin{solution}[E9]
Since $V(I)=V(\sqrt I)$, we may replace $I$ by its radical.  The closed set is irreducible
iff the quotient by that radical is a domain, equivalently iff $\sqrt I$ is prime.  When
this occurs, its generic point is
\[
\sqrt I.
\]
\end{solution}

\begin{solution}[E10]
Apply E9 to $I=(0)$.  The whole spectrum is $V(0)$, so it is irreducible and has a generic
point exactly when
\[
\sqrt{(0)}
\]
is prime.  In that case the generic point is precisely $\sqrt{(0)}$.
\end{solution}

\begin{solution}[E11]
An irreducible component is a maximal irreducible closed set, so it is $V(\mathfrak p)$
for some prime $\mathfrak p$.  Since
\[
\mathfrak p\subseteq\mathfrak q
\Longrightarrow
V(\mathfrak q)\subseteq V(\mathfrak p),
\]
maximality of $V(\mathfrak p)$ is equivalent to minimality of $\mathfrak p$ among primes.
By E8, $\mathfrak p$ is its generic point.
\end{solution}

\begin{solution}[E12]
In $\Spec\ZZ$, every prime ideal is $(0)$ or $(p)$ for a prime number $p$.  The point
$(0)$ is contained in every $(p)$, so every $(p)$ is a specialization of $(0)$.  The only
generalization of $(0)$ is itself.  For a closed point $(p)$, its generalizations are
$(0)$ and $(p)$, while its only specialization is itself.
\end{solution}

\begin{solution}[E13]
The ring $k[t]$ is a PID.  Its nonzero prime ideals are exactly $(f)$ with $f$ irreducible.
Such an ideal is maximal because $k[t]/(f)$ is a field.  The zero ideal is prime but not
maximal.  Hence the closed points are precisely the $(f)$ with $f$ irreducible.
\end{solution}

\begin{solution}[E14]
Over $\RR$, irreducibles are linear polynomials $t-a$ and quadratics with negative
discriminant.  Thus the closed points are
\[
(t-a)
\]
and
\[
(q(t))
\]
for irreducible quadratics $q$.  The linear points have quotient $\RR$ and are
$\RR$-rational.  The irreducible quadratic points have quotient isomorphic to $\CC$ and
are not $\RR$-rational.
\end{solution}

\begin{solution}[E15]
If $f$ is irreducible, $(f)$ is maximal in the PID $\FF_p[t]$, hence closed.  The quotient
is a field of $p^d$ elements when $\deg f=d$, so it is isomorphic to $\FF_{p^d}$.  The
quotient equals $\FF_p$ exactly for $d=1$, so precisely the degree-one closed points are
$\FF_p$-rational.
\end{solution}

\begin{solution}[E16]
In
\[
R=k[x,y]/(xy),
\]
one has
\[
R/(\bar x)\cong k[y],\qquad R/(\bar y)\cong k[x],
\]
so $(\bar x)$ and $(\bar y)$ are prime.  They are minimal because their intersection is
the nilradical $(0)$ in this reduced ring.  Hence they are component-generic points.  The
ideal $(\bar x,\bar y)$ has quotient $k$, so it is maximal and closed, and it contains both
minimal primes.
\end{solution}

\begin{solution}[E17]
In $A=k[x,y]/(x^2)$, the class $\bar x$ is nilpotent, so $(\bar x)$ contains the
nilradical.  Conversely any nilpotent maps to zero in
\[
A/(\bar x)\cong k[y],
\]
which is reduced, hence belongs to $(\bar x)$.  Thus
\[
\sqrt{(0)}=(\bar x).
\]
The quotient is a domain, so this ideal is prime and therefore the generic point of
$\Spec A$.
\end{solution}

\begin{solution}[E18]
If $\eta\notin U$, then the closed set $X\setminus U$ contains $\eta$, hence contains
\[
\overline{\{\eta\}}=X,
\]
forcing $U=\varnothing$, contradiction.  Thus $\eta\in U$.  In the subspace topology,
\[
\overline{\{\eta\}}^{\,U}
=
U\cap\overline{\{\eta\}}^{\,X}
=
U,
\]
so $\eta$ is generic in $U$.
\end{solution}

\begin{solution}[E19]
Prime ideals of $A/I$ are exactly $\mathfrak p/I$ for primes $\mathfrak p\supseteq I$.
The homeomorphism
\[
\Spec(A/I)\longrightarrow V_A(I)
\]
sends $\mathfrak p/I$ to $\mathfrak p$ and preserves closures.  Therefore a point is
generic for an irreducible closed subset on one side exactly when its image is generic for
the corresponding subset on the other.
\end{solution}

\begin{solution}[E20]
Suppose $\mathfrak q_1\subseteq\mathfrak q_2$ in $\Spec B$.  Contraction preserves
containment:
\[
\varphi^{-1}(\mathfrak q_1)\subseteq\varphi^{-1}(\mathfrak q_2).
\]
By the specialization criterion, this says that if $\mathfrak q_2$ is a specialization
of $\mathfrak q_1$, then $\varphi^\ast(\mathfrak q_2)$ is a specialization of
$\varphi^\ast(\mathfrak q_1)$.
\end{solution}
