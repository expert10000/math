% VI37-DETAILED-PROBLEM-19
\begin{problem}[VI.37.P19: Reduced Proj from a nonreduced coordinate ring]\label{prob:vi37-19}
Let
\[
A=k[x_0,x_1,z]/(z^2,x_0z,x_1z).
\]
Show that \(A\) is nonreduced but \(\Proj A\) is reduced, and identify the projectively invisible nilpotent module.

\textbf{Provenance.} Canonical VI/37 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The class of \(z\) is nonzero and satisfies
\[
z^2=0,
\]
so \(A\) is not reduced.

Let
\[
N=(z)\subseteq A.
\]
The relations
\[
x_0z=x_1z=z^2=0
\]
show that
\[
A_+N=0.
\]
Thus \(N\) is irrelevant torsion. By VI/35,
\[
\widetilde N=0
\]
on \(\Proj A\).

On the standard charts, invertibility of \(x_0\) or \(x_1\) forces \(z=0\), giving exactly the chart rings of \(\mathbb P^1\). Hence
\[
\Proj A\cong\mathbb P^1_k,
\]
which is reduced. Therefore
\[
\boxed{
A\text{ nonreduced},\qquad
\Proj A\text{ reduced}.
}
\]
The nilpotent ideal \(N=(z)\) is precisely the algebra supported at the irrelevant vertex that projectivization discards.
\end{solution}
\fi
