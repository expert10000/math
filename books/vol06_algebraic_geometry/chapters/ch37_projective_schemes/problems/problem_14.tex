% VI37-DETAILED-PROBLEM-14
\begin{problem}[VI.37.P14: Saturated ideals are determined by projective chart ideals]\label{prob:vi37-14}
Let \(I,J\subseteq S=k[x_0,\ldots,x_n]\) be homogeneous. Assume
\[
IS_{x_i}=JS_{x_i}
\]
for every standard variable \(x_i\). Prove
\[
I^{\mathrm{sat}}=J^{\mathrm{sat}}.
\]
Deduce that two saturated homogeneous ideals with the same projective ideal sheaf are equal.

\textbf{Provenance.} Canonical VI/37 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
We prove
\[
I^{\mathrm{sat}}\subseteq J^{\mathrm{sat}}.
\]
Let \(g\in I^{\mathrm{sat}}\) be homogeneous. Since saturation does not change localization,
\[
g\in I^{\mathrm{sat}}S_{x_i}=IS_{x_i}=JS_{x_i}
\]
for every \(i\). Hence for each \(i\) there exists \(n_i\ge1\) such that
\[
x_i^{n_i}g\in J.
\]
Set
\[
N=\sum_{i=0}^n(n_i-1)+1.
\]
Every monomial of total degree \(N\) is divisible by at least one \(x_i^{n_i}\); otherwise all exponents would be at most \(n_i-1\), contradicting the definition of \(N\). Therefore
\[
S_+^Ng\subseteq J,
\]
so
\[
g\in J^{\mathrm{sat}}.
\]
By symmetry,
\[
\boxed{I^{\mathrm{sat}}=J^{\mathrm{sat}}}.
\]

If \(I\) and \(J\) are already saturated and define the same projective ideal sheaf, their localized chart ideals agree, so the argument gives
\[
I=I^{\mathrm{sat}}=J^{\mathrm{sat}}=J.
\]
Thus saturated homogeneous ideals are recovered from their projective ideal sheaves in the standard polynomial setting.
\end{solution}
\fi
