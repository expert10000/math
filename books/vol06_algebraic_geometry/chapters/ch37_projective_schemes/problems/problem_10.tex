\begin{problem}[VI.37.P10: Components from minimal primes]\label{prob:vi37-10}
Let
\[
X=\Proj\bigl(k[x_0,x_1,x_2]/(x_0x_1x_2)\bigr).
\]
Determine the irreducible components of \(X\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The principal ideal factors into distinct irreducibles:
\[
(x_0x_1x_2).
\]
Its minimal primes are
\[
(x_0),\qquad(x_1),\qquad(x_2).
\]
All are homogeneous and none contains the irrelevant ideal
\[
(x_0,x_1,x_2).
\]
Therefore each contributes one irreducible component of Proj:
\[
V_+(x_0),\qquad
V_+(x_1),\qquad
V_+(x_2).
\]
Each is a coordinate projective line. Hence
\[
\boxed{
X
=
V_+(x_0)\cup V_+(x_1)\cup V_+(x_2)
}
\]
with three irreducible components.
\end{solution}
\fi
