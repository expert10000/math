% VI37-DETAILED-PROBLEM-01
\begin{problem}[VI.37.P1: Quotient chart theorem]\label{prob:vi37-01}
Let \(S=k[x_0,\ldots,x_n]\), let \(I\) be homogeneous, and let \(f\) be homogeneous of positive degree. Prove the chart identity
\[
D_+(\bar f)\subseteq\Proj(S/I)\cong\Spec(S_{(f)}/I_{(f)}).
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The standard-open theorem from VI/35 gives
\[
D_+(\bar f)\cong\Spec (S/I)_{(f)}.
\]
Ordinary localization commutes with quotient:
\[
(S/I)_f\cong S_f/IS_f.
\]
Because the quotient and localization maps preserve grading, taking degree zero gives
\[
(S/I)_{(f)}
\cong
(S_f/IS_f)_0
\cong
(S_f)_0/(IS_f)_0.
\]
By definition
\[
(S_f)_0=S_{(f)},
\qquad
(IS_f)_0=I_{(f)}.
\]
Hence
\[
\boxed{
D_+(\bar f)
\cong
\Spec\bigl(S_{(f)}/I_{(f)}\bigr).
}
\]
The construction is compatible with further homogeneous localization. Thus these affine charts glue to the closed subscheme
\[
\Proj(S/I)\subseteq\Proj S.
\]
\end{solution}
\fi
