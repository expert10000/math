% VI37-DETAILED-PROBLEM-05
\begin{problem}[VI.37.P5: Saturation removes vertex torsion]\label{prob:vi37-05}
Let \(I=(x^2,xy,xz)\subset k[x,y,z]\). Compute \(I^{\rm sat}\) with respect to \((x,y,z)\) and compare the associated projective schemes.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
\mathfrak m=(x,y,z).
\]
Since
\[
\mathfrak m x=(x^2,xy,xz)=I,
\]
we have
\[
x\in I:\mathfrak m\subseteq I^{\mathrm{sat}}.
\]
Thus
\[
(x)\subseteq I^{\mathrm{sat}}.
\]
Conversely
\[
I\subseteq(x).
\]
The ideal \((x)\) is saturated with respect to \(\mathfrak m\): if \(\mathfrak m^r g\subseteq(x)\), then \(y^r g\in(x)\), and since \(x\) and \(y\) are relatively prime, \(x\mid g\). Hence
\[
I^{\mathrm{sat}}\subseteq(x).
\]
Therefore
\[
\boxed{I^{\mathrm{sat}}=(x)}.
\]

On every standard projective chart, \(I\) and \((x)\) localize to the same ideal, so
\[
\boxed{\Proj S/I\cong\Proj S/(x)}.
\]
Both define the same projective line, while their affine cones differ by embedded structure supported at the cone vertex.
\end{solution}
\fi
