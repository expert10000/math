% VI37-DETAILED-PROBLEM-04
\begin{problem}[VI.37.P4: A tangent line creates a fat intersection]\label{prob:vi37-04}
Intersect the conic \(XZ-Y^2=0\) with its tangent line \(Z=0\) at \([1:0:0]\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The conic ideal is
\[
I_C=(XZ-Y^2)
\]
and the tangent line has ideal
\[
I_L=(Z).
\]
Their scheme-theoretic intersection is defined by
\[
I_C+I_L
=
(Z,XZ-Y^2)
=
(Z,Y^2).
\]
Hence
\[
C\cap L
=
\Proj k[X,Y,Z]/(Z,Y^2).
\]
Its support is the single point \([1:0:0]\). On the chart \(X\neq0\), let
\[
y=Y/X,\qquad z=Z/X.
\]
The local ideal becomes
\[
(z,y^2),
\]
so the local intersection is
\[
\boxed{\Spec k[y]/(y^2)}.
\]
It has \(k\)-basis \(1,y\), hence length two. The nilpotent \(y\) records first-order tangency that the reduced point set would forget.
\end{solution}
\fi
