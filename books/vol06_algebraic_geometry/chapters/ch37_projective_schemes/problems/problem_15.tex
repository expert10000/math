% VI37-DETAILED-PROBLEM-15
\begin{problem}[VI.37.P15: A length-three projective fat point]\label{prob:vi37-15}
In \(\mathbb P^2\), let
\[
I=(x_1,x_2)^2.
\]
Show that \(\Proj S/I\) is supported at \([1:0:0]\), compute its affine local ring there, and determine its length.

\textbf{Provenance.} Canonical VI/37 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The radical is
\[
\sqrt I=(x_1,x_2),
\]
the homogeneous ideal of the point
\[
p=[1:0:0].
\]
Thus the projective scheme is supported at \(p\).

On the chart \(x_0\neq0\), set
\[
u=x_1/x_0,\qquad v=x_2/x_0.
\]
The ideal dehomogenizes to
\[
(u,v)^2=(u^2,uv,v^2).
\]
Therefore
\[
X\cap D_+(x_0)
=
\Spec k[u,v]/(u,v)^2.
\]
Since the support is entirely inside this chart, this affine ring is the whole local Artin model.

As a \(k\)-vector space,
\[
k[u,v]/(u,v)^2
\]
has basis
\[
1,u,v.
\]
Hence
\[
\boxed{\operatorname{length}(X)=3}.
\]
The maximal ideal \((u,v)\) squares to zero and has dimension two, so the fat point retains both first-order tangent directions at \(p\).
\end{solution}
\fi
