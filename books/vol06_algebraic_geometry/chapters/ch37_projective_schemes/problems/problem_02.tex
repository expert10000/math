% VI37-DETAILED-PROBLEM-02
\begin{problem}[VI.37.P2: Projective conic charts]\label{prob:vi37-02}
For \(X=V_+(XZ-Y^2)\subseteq\mathbb P^2_k\), compute all three standard affine charts.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
F=XZ-Y^2.
\]
On \(X\neq0\), write
\[
y=Y/X,\qquad z=Z/X.
\]
Dividing \(F\) by \(X^2\) gives
\[
z-y^2=0,
\]
so this chart is
\[
\Spec k[y,z]/(z-y^2)\cong\Spec k[y].
\]

On \(Y\neq0\), write
\[
x=X/Y,\qquad z=Z/Y.
\]
Dividing by \(Y^2\) gives
\[
xz-1=0,
\]
so
\[
X\cap D_+(Y)\cong\Spec k[x,z]/(xz-1)\cong\mathbb G_m.
\]

On \(Z\neq0\), write
\[
x=X/Z,\qquad y=Y/Z.
\]
Dividing by \(Z^2\) gives
\[
x-y^2=0,
\]
again an affine line. The transition maps are restrictions of the standard projective coordinate changes, so the three affine equations glue to the same projective conic.
\end{solution}
\fi
