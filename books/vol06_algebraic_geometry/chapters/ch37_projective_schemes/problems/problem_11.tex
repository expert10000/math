% VI37-DETAILED-PROBLEM-11
\begin{problem}[VI.37.P11: Same Proj, different graded rings]\label{prob:vi37-11}
Let \(A=k[x_0,x_1,z]/(z^2,x_0z,x_1z)\). Prove \(\Proj A\cong\mathbb P^1_k\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The class of \(z\) is nilpotent, so every prime of \(A\) contains \(z\). A projectively relevant prime must avoid at least one of \(x_0,x_1\); otherwise it would contain all of \(A_+\). Thus
\[
D_+(x_0)\cup D_+(x_1)=\Proj A.
\]

On \(D_+(x_0)\), the relation
\[
x_0z=0
\]
and invertibility of \(x_0\) imply \(z=0\). Hence
\[
A_{(x_0)}
\cong
k[x_1/x_0].
\]
Similarly
\[
A_{(x_1)}
\cong
k[x_0/x_1].
\]
On their overlap the transition is the usual inversion
\[
x_0/x_1=(x_1/x_0)^{-1}.
\]
These are precisely the two affine charts and overlap map of \(\mathbb P^1_k\). Therefore
\[
\boxed{\Proj A\cong\mathbb P^1_k}.
\]
The nilpotent \(z\) lives only at the irrelevant cone vertex and disappears after projectivization.
\end{solution}
\fi
