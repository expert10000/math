% VI37-DETAILED-PROBLEM-12
\begin{problem}[VI.37.P12: A saturated zero-dimensional scheme]\label{prob:vi37-12}
In \(\mathbb P^2\), let \(I=(x_2,x_0x_1)\). Describe \(\Proj S/I\), and show \(I\) is saturated.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Inside the line
\[
x_2=0\cong\mathbb P^1,
\]
the remaining equation is
\[
x_0x_1=0.
\]
Hence the projective scheme consists of the two coordinate points
\[
p=[1:0:0],
\qquad
q=[0:1:0].
\]
Their homogeneous ideals are
\[
I_p=(x_1,x_2),
\qquad
I_q=(x_0,x_2).
\]
A direct intersection computation gives
\[
I_p\cap I_q=(x_2,x_0x_1)=I.
\]
Each point ideal is saturated. If \(g\notin I_p\cap I_q\), then it fails membership in at least one saturated ideal, so no power of the irrelevant ideal can force it into the intersection. Equivalently saturation commutes with this finite intersection here. Thus
\[
\boxed{I^{\mathrm{sat}}=I}.
\]
Therefore
\[
\boxed{\Proj S/I=p\sqcup q}
\]
as two reduced points.
\end{solution}
\fi
