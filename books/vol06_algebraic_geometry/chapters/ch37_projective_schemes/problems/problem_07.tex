\begin{problem}[VI.37.P7: Two lines and their union]\label{prob:vi37-07}
Let \(L_0=V_+(x_0)\) and \(L_1=V_+(x_1)\) in \(\mathbb P^2\). Compute their union and intersection scheme-theoretically.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The line ideals are
\[
I_0=(x_0),
\qquad
I_1=(x_1).
\]
The scheme-theoretic union is defined by
\[
I_0\cap I_1.
\]
Since \(x_0,x_1\) are relatively prime,
\[
I_0\cap I_1=(x_0x_1).
\]
Thus
\[
\boxed{
L_0\cup L_1
=
\Proj S/(x_0x_1).
}
\]

The intersection is defined by the sum:
\[
I_0+I_1=(x_0,x_1),
\]
so
\[
\boxed{
L_0\cap L_1
=
\Proj S/(x_0,x_1)
=
\{[0:0:1]\}.
}
\]
The union is reducible and reduced, while the intersection is one reduced point.
\end{solution}
\fi
