% VI37-DETAILED-PROBLEM-08
\begin{problem}[VI.37.P8: Empty projective scheme]\label{prob:vi37-08}
Show that \(\Proj k[x_0,x_1]/(x_0,x_1)^2\) is empty although the quotient ring is nonzero.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
A=k[x_0,x_1]/(x_0,x_1)^2.
\]
The ring is nonzero because the class of \(1\) survives, and \(x_0,x_1\) themselves are nonzero in general. But the irrelevant ideal
\[
A_+=(\bar x_0,\bar x_1)
\]
satisfies
\[
A_+^2=0.
\]
Every prime ideal contains every nilpotent element, so every prime of \(A\) contains \(A_+\). Therefore no homogeneous prime satisfies the defining condition for Proj. Hence
\[
\boxed{\Proj A=\varnothing}.
\]
This is the model example of a nonzero graded ring whose entire positive-degree geometry is concentrated at the irrelevant vertex.
\end{solution}
\fi
