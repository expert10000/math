% VI37-DETAILED-PROBLEM-09
\begin{problem}[VI.37.P9: A nonreduced projective line]\label{prob:vi37-09}
Describe \(X=\Proj k[x_0,x_1,x_2]/(x_2^2)\) and compare it with \(V_+(x_2)\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The radical of \((x_2^2)\) is \((x_2)\), so the underlying topological space is the line
\[
V_+(x_2)\cong\mathbb P^1.
\]
But the structure is thicker.

On \(D_+(x_0)\), set
\[
u=x_1/x_0,\qquad v=x_2/x_0.
\]
Then
\[
X\cap D_+(x_0)
\cong
\Spec k[u,v]/(v^2).
\]
The class \(v\) is a nonzero nilpotent normal direction along the line. Similarly on \(D_+(x_1)\), the ratio \(x_2/x_1\) has square zero.

Thus
\[
\boxed{
X
}
\]
is a double structure on the projective line, with
\[
X_{\mathrm{red}}\cong V_+(x_2)\cong\mathbb P^1.
\]
It is not isomorphic to the reduced line because its structure sheaf contains nonzero nilpotents.
\end{solution}
\fi
