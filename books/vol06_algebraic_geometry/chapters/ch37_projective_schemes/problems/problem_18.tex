% VI37-DETAILED-PROBLEM-18
\begin{problem}[VI.37.P18: Equivalent criteria for empty Proj]\label{prob:vi37-18}
For homogeneous \(I\subseteq S=k[x_0,\ldots,x_n]\), prove the equivalence
\[
\Proj(S/I)=\varnothing
\iff
I^{\mathrm{sat}}=S
\iff
S_+^N\subseteq I\text{ for some }N.
\]

\textbf{Provenance.} Canonical VI/37 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
If
\[
S_+^N\subseteq I,
\]
then every prime containing \(I\) contains \(S_+\), so no such prime lies in Proj. Hence
\[
S_+^N\subseteq I\Longrightarrow\Proj(S/I)=\varnothing.
\]

If
\[
S_+^N\subseteq I,
\]
then
\[
1\in I:S_+^N\subseteq I^{\mathrm{sat}},
\]
so
\[
I^{\mathrm{sat}}=S.
\]

Conversely, if
\[
I^{\mathrm{sat}}=S,
\]
then \(1\in I:S_+^N\) for some \(N\), which says exactly
\[
S_+^N\subseteq I.
\]

Finally, if Proj is empty, every homogeneous prime containing \(I\) contains \(S_+\), hence
\[
S_+\subseteq\sqrt I.
\]
Since \(S_+\) is finitely generated, a power \(S_+^N\) lies in \(I\). Thus all three conditions are equivalent.
\end{solution}
\fi
