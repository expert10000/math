\begin{example}[A projective point]\label{ex:vi37-01}
In \(\mathbb P^2_k\), the homogeneous ideal \((x_1,x_2)\) gives \(\Proj k[x_0]\cong\Spec k\), the point \([1:0:0]\).
\end{example}

\begin{example}[A projective line]\label{ex:vi37-02}
The ideal \((x_2)\subset k[x_0,x_1,x_2]\) defines \(\Proj k[x_0,x_1]\cong\mathbb P^1_k\).
\end{example}

\begin{example}[A conic]\label{ex:vi37-03}
The quotient \(k[X,Y,Z]/(XZ-Y^2)\) defines a projective conic in \(\mathbb P^2\).  On \(X\ne0\) the equation becomes \(z-y^2=0\).
\end{example}

\begin{example}[A cubic curve]\label{ex:vi37-04}
The homogeneous cubic \(Y^2Z-X^3-XZ^2\) defines a projective plane cubic.  On \(Z\ne0\) it becomes \(y^2=x^3+x\).
\end{example}

\begin{example}[Two coordinate lines]\label{ex:vi37-05}
The ideal \((x_0x_1)\subset k[x_0,x_1,x_2]\) defines the union of the two projective lines \(x_0=0\) and \(x_1=0\).
\end{example}

\begin{example}[Double line]\label{ex:vi37-06}
The ideals \((x_0)\) and \((x_0^2)\) have the same underlying projective line, but \(S/(x_0^2)\) retains a nilpotent normal direction.
\end{example}

\begin{example}[Intersection of two lines]\label{ex:vi37-07}
In \(\mathbb P^2\), \(V_+(x_0)\cap V_+(x_1)=V_+(x_0,x_1)\), the point \([0:0:1]\).
\end{example}

\begin{example}[A saturated hyperplane ideal]\label{ex:vi37-08}
For \(I=x_0(x_0,x_1,x_2)\), one has \(I^{\mathrm{sat}}=(x_0)\), so both ideals define the same projective line.
\end{example}

\begin{example}[An empty Proj]\label{ex:vi37-09}
The quotient \(k[x_0,x_1]/(x_0,x_1)^3\) is nonzero, but its Proj is empty because a power of the irrelevant ideal vanishes.
\end{example}

\begin{example}[Closure of an affine parabola]\label{ex:vi37-10}
The affine parabola \(y=x^2\) homogenizes to \(x_0x_2-x_1^2=0\) in \(\mathbb P^2\), producing the projective conic closure.
\end{example}

\begin{example}[Points at infinity]\label{ex:vi37-11}
The projective closure of \(xy=1\) is \(x_1x_2-x_0^2=0\).  On \(x_0=0\) it meets infinity at \([0:1:0]\) and \([0:0:1]\).
\end{example}

\begin{example}[Cone over a conic]\label{ex:vi37-12}
The affine cone over \(XZ-Y^2=0\subset\mathbb P^2\) is the quadric cone \(V(XZ-Y^2)\subset\mathbb A^3\), singular at the origin when the characteristic does not create exceptional behavior.
\end{example}

\begin{example}[A reducible projective scheme]\label{ex:vi37-13}
\(\Proj k[x_0,x_1,x_2]/(x_0x_1)\) has two irreducible components, corresponding to the homogeneous minimal primes \((x_0)\) and \((x_1)\).
\end{example}

\begin{example}[Nonreduced ring, reduced Proj]\label{ex:vi37-14}
For \(A=k[x_0,x_1,z]/(z^2,x_0z,x_1z)\), the class of \(z\) vanishes after localizing at either \(x_0\) or \(x_1\), so \(\Proj A\cong\mathbb P^1_k\).
\end{example}

\begin{example}[Homogeneous coordinate ring changes with embedding]\label{ex:vi37-15}
The projective line has coordinate ring \(k[s,t]\) in its standard embedding, while the quadratic Veronese presentation has coordinate ring \(k[X,Y,Z]/(XZ-Y^2)\).  The schemes are isomorphic although the graded rings are not.
\end{example}

\begin{example}[A zero-dimensional projective scheme]\label{ex:vi37-16}
In \(\mathbb P^1\), the homogeneous polynomial \(x_0x_1\) defines two reduced points when the factors are distinct; the projective scheme has dimension zero.
\end{example}
