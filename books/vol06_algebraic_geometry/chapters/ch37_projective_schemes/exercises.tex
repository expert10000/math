\begin{exercise}[Quotient chart]\label{exr:vi37-01}
Prove directly that \((S/I)_{(f)}\cong S_{(f)}/I_{(f)}\) for homogeneous \(I\) and \(f\).
\end{exercise}

\begin{exercise}[Underlying closed set]\label{exr:vi37-02}
Show that homogeneous primes of \(S/I\) defining Proj correspond exactly to points of \(V_+(I)\subseteq\Proj S\).
\end{exercise}

\begin{exercise}[Coordinate line]\label{exr:vi37-03}
Identify \(\Proj k[x_0,x_1,x_2]/(x_2)\).
\end{exercise}

\begin{exercise}[Conic chart]\label{exr:vi37-04}
Dehomogenize \(XZ-Y^2\) on each standard chart of \(\mathbb P^2\).
\end{exercise}

\begin{exercise}[Hypersurface dimension]\label{exr:vi37-05}
Use the principal ideal theorem to justify the dimension formula for a nonzero projective hypersurface in \(\mathbb P^n\).
\end{exercise}

\begin{exercise}[Intersection by sum]\label{exr:vi37-06}
Compute the scheme-theoretic intersection of the lines \(x_0=0\) and \(x_1=0\) in \(\mathbb P^2\).
\end{exercise}

\begin{exercise}[Union by intersection]\label{exr:vi37-07}
Show that \((x_0)\cap(x_1)=(x_0x_1)\) in a polynomial ring and interpret the corresponding projective union.
\end{exercise}

\begin{exercise}[Double hyperplane]\label{exr:vi37-08}
Show that \(\Proj S/(x_0^2)\) and \(\Proj S/(x_0)\) have the same underlying topological space but different structure sheaves.
\end{exercise}

\begin{exercise}[Saturation calculation]\label{exr:vi37-09}
For \(I=x_0(x_0,x_1,x_2)\subset k[x_0,x_1,x_2]\), prove \(I^{\mathrm{sat}}=(x_0)\).
\end{exercise}

\begin{exercise}[Empty Proj]\label{exr:vi37-10}
Prove that \(\Proj S/I=\varnothing\) when \(S_+^r\subseteq I\) for some \(r\).
\end{exercise}

\begin{exercise}[Converse empty criterion]\label{exr:vi37-11}
For \(S=k[x_0,\ldots,x_n]\), prove that empty Proj implies some power of \(S_+\) lies in \(I\).
\end{exercise}

\begin{exercise}[Homogenize a parabola]\label{exr:vi37-12}
Homogenize \(y-x^2\) and identify the point(s) of its projective closure at infinity.
\end{exercise}

\begin{exercise}[Homogenize a hyperbola]\label{exr:vi37-13}
Homogenize \(xy-1\) and determine its points at infinity.
\end{exercise}

\begin{exercise}[Affine cone]\label{exr:vi37-14}
Write the affine cone of the projective conic \(XZ-Y^2=0\) and identify its vertex.
\end{exercise}

\begin{exercise}[Reducedness descends]\label{exr:vi37-15}
Prove that if a graded ring \(A\) is reduced, then \(\Proj A\) is reduced.
\end{exercise}

\begin{exercise}[Domain descends]\label{exr:vi37-16}
Prove that if \(A\) is a domain and \(\Proj A\ne\varnothing\), then \(\Proj A\) is integral.
\end{exercise}

\begin{exercise}[Projectively invisible nilpotent]\label{exr:vi37-17}
Verify that \(z\) vanishes on the standard charts of \(\Proj k[x_0,x_1,z]/(z^2,x_0z,x_1z)\).
\end{exercise}

\begin{exercise}[Components]\label{exr:vi37-18}
Determine the projective components of \(\Proj k[x_0,x_1,x_2]/(x_0x_1)\).
\end{exercise}

\begin{exercise}[Two coordinate points]\label{exr:vi37-19}
Find a homogeneous ideal defining the two points \([1:0:0]\) and \([0:1:0]\) in \(\mathbb P^2\).
\end{exercise}

\begin{exercise}[A fat point]\label{exr:vi37-20}
Describe the projective scheme defined in \(\mathbb P^2\) by \((x_1,x_2)^2\) near \([1:0:0]\).
\end{exercise}

\begin{exercise}[Coordinate ring dependence]\label{exr:vi37-21}
Explain why \(k[s,t]\) and \(k[X,Y,Z]/(XZ-Y^2)\) can both serve as homogeneous coordinate rings of schemes isomorphic to \(\mathbb P^1\).
\end{exercise}

\begin{exercise}[Chartwise equality after saturation]\label{exr:vi37-22}
Prove directly that \(I\) and \(I^{\mathrm{sat}}\) have the same localized ideal after inverting any variable \(x_i\).
\end{exercise}

\begin{exercise}[Intersection with infinity]\label{exr:vi37-23}
Let \(F\) be the homogenization of an affine polynomial.  Explain how setting \(x_0=0\) computes the scheme-theoretic intersection of its projective closure with the hyperplane at infinity, subject to the saturation caveat.
\end{exercise}

\begin{exercise}[Why radicals lose information]\label{exr:vi37-24}
Compare \(S/(x_0^2)\) and \(S/(x_0)\) on a standard chart and identify the nilpotent killed by passing to the radical.
\end{exercise}
