% VI37-DETAILED-CHALLENGE-01
\subsection*{Challenge VI.37.C1 solution}
\begin{solution}
Let
\[
S=k[x_0,\ldots,x_n],\qquad \mathfrak m=S_+.
\]
The homogeneous ideal \(I\) determines on each standard chart \(U_i=D_+(x_i)\) the ideal
\[
I_{(x_i)}=(IS_{x_i})_0\subseteq S_{(x_i)}.
\]
VI/34 gives
\[
IS_{x_i}=I^{\mathrm{sat}}S_{x_i}.
\]
Taking degree-zero parts yields
\[
\boxed{
I_{(x_i)}
=
I^{\mathrm{sat}}_{(x_i)}
}
\]
for every \(i\).

By the affine correspondence of VI/33, these two modules define the same quasi-coherent ideal sheaf on every \(U_i\). The standard opens cover \(\mathbb P^n\), and the identifications are induced by the same localization maps on overlaps. Therefore the projective ideal sheaves are equal:
\[
\boxed{
\widetilde I=\widetilde{I^{\mathrm{sat}}}
\subseteq\mathcal O_{\mathbb P^n}.
}
\]

This is stronger than equality of underlying closed sets. Equality of closed sets would only imply equality after taking radicals of the corresponding local ideals. Here the actual ideal sheaves agree, so all nilpotent and infinitesimal scheme structure visible on projective charts is preserved. Saturation removes only algebra supported at the irrelevant vertex, which no projective chart sees.
\end{solution}

% VI37-DETAILED-CHALLENGE-02
\subsection*{Challenge VI.37.C2 solution}
\begin{solution}
Take the affine ideal
\[
J=(x^2-y,\ x^3-xy-x)\subseteq k[x,y].
\]
Let
\[
f_1=x^2-y,\qquad f_2=x^3-xy-x.
\]
Then
\[
f_2-xf_1=-x,
\]
so \(x\in J\). Since
\[
y=x^2-f_1,
\]
also \(y\in J\). Hence
\[
\boxed{J=(x,y)},
\]
so the affine scheme is the origin in \(\mathbb A^2\).

Introduce the homogenizing variable \(Z\), with projective coordinates \([Z:X:Y]\). The displayed generators homogenize to
\[
F_1=X^2-YZ,
\]
\[
F_2=X^3-XYZ-XZ^2.
\]
Let
\[
H=(F_1,F_2)\subseteq k[Z,X,Y].
\]
The relation
\[
F_2-XF_1=-XZ^2
\]
shows
\[
XZ^2\in H.
\]
Therefore
\[
X\in H:Z^\infty.
\]
Using
\[
F_1=X^2-YZ
\]
and the fact that \(X\in H:Z^\infty\), we obtain \(YZ\in H:Z^\infty\), hence
\[
Y\in H:Z^\infty.
\]
Thus
\[
(X,Y)\subseteq H:Z^\infty.
\]
Conversely \(H\subseteq(X,Y)\), and \((X,Y)\) is \(Z\)-saturated because \(Z\) is a non-zero-divisor modulo \((X,Y)\). Hence
\[
\boxed{
H:Z^\infty=(X,Y).
}
\]

The saturated ideal \((X,Y)\) defines the single projective point
\[
[1:0:0],
\]
the closure of the affine origin.

The naive ideal \(H\) has extra structure at infinity. Setting \(Z=0\) gives
\[
H+(Z)=(Z,X^2,X^3)=(Z,X^2),
\]
so it retains a nonreduced point supported at
\[
[0:0:1].
\]
That point is not in the closure of the original affine origin. Saturation by \(Z\) removes this spurious component/embedded structure and recovers the correct projective closure.
\end{solution}

% VI37-DETAILED-CHALLENGE-03
\subsection*{Challenge VI.37.C3 solution}
\begin{solution}
Use the smooth conic
\[
C=V_+(XZ-Y^2)\subseteq\mathbb P^2_k
\]
and the rational point
\[
p=[1:0:0].
\]
The partial derivatives of
\[
F=XZ-Y^2
\]
are
\[
F_X=Z,\qquad F_Y=-2Y,\qquad F_Z=X.
\]
At \(p\), the gradient is
\[
(0,0,1),
\]
so the tangent line is
\[
L:Z=0.
\]
For this model no characteristic restriction is needed at \(p\).

The scheme-theoretic intersection is defined by
\[
(F,Z)=(Z,Y^2).
\]
On the affine chart \(X\neq0\), set
\[
y=Y/X,\qquad z=Z/X.
\]
Then
\[
C:z-y^2=0,
\qquad
L:z=0.
\]
Adding the local ideals gives
\[
(z-y^2,z)=(z,y^2).
\]
Eliminating \(z\) yields
\[
\boxed{
C\cap L
\cong
\Spec k[y]/(y^2).
}
\]
Writing \(\varepsilon\) for the class of \(y\),
\[
\boxed{
C\cap L
\cong
\Spec k[\varepsilon]/(\varepsilon^2).
}
\]
The underlying point is only \(p\), but the length is two. Thus the nonreduced intersection records tangency.
\end{solution}

% VI37-DETAILED-CHALLENGE-04
\subsection*{Challenge VI.37.C4 solution}
\begin{solution}
The quadratic Veronese conic has homogeneous coordinate ring
\[
A=k[X,Y,Z]/(XZ-Y^2).
\]
Its affine cone is
\[
C=\Spec A\subseteq\mathbb A^3_k.
\]
The cone vertex is the origin
\[
o=(0,0,0).
\]

For
\[
F=XZ-Y^2,
\]
the Jacobian row is
\[
\left(
\frac{\partial F}{\partial X},
\frac{\partial F}{\partial Y},
\frac{\partial F}{\partial Z}
\right)
=
(Z,-2Y,X).
\]
At the origin this becomes
\[
(0,0,0).
\]
Since the cone is a hypersurface of expected dimension \(2\), the vanishing Jacobian at the vertex shows that the Zariski tangent space there has dimension \(3\), strictly larger than the local dimension \(2\). Therefore
\[
\boxed{o\text{ is singular on the affine cone}.}
\]

Now consider the projective conic
\[
Q=V_+(XZ-Y^2)\subseteq\mathbb P^2.
\]
A projective singular point would have to satisfy
\[
F=F_X=F_Y=F_Z=0.
\]
If \(\operatorname{char}k\neq2\), the derivative equations give
\[
X=Y=Z=0,
\]
which is impossible projectively.

In characteristic \(2\), the derivative conditions reduce to
\[
X=Z=0.
\]
Then the equation \(XZ-Y^2=0\) gives
\[
Y^2=0,
\]
so \(Y=0\), again impossible in projective space. Thus
\[
\boxed{Q\text{ is smooth in every characteristic}.}
\]

The contrast is fundamental: projectivization removes the cone vertex, and the singularity concentrated there disappears. A smooth projective variety can therefore have a singular affine cone.
\end{solution}

% VI37-DETAILED-CHALLENGE-05
\subsection*{Challenge VI.37.C5 solution}
\begin{solution}
The corrected phenomenon is exhibited by
\[
A=k[x,y]/(x^2,xy)
\]
with the standard grading.

The defining ideal is
\[
(x^2,xy)=x(x,y).
\]
Its radical is
\[
(x),
\]
so the unique homogeneous minimal prime is
\[
\mathfrak p=(x).
\]
It does not contain the irrelevant ideal
\[
A_+=(x,y),
\]
because \(y\notin(x)\).

Now consider the nonzero class of \(x\in A\). Its annihilator is
\[
\operatorname{Ann}_A(x)=(x,y)=A_+,
\]
because
\[
x^2=xy=0,
\]
while an element with nonzero constant term cannot annihilate \(x\). Hence
\[
\mathfrak m=A_+
\]
is an associated prime. Since
\[
\mathfrak p\subsetneq\mathfrak m,
\]
it is embedded rather than minimal.

The minimal prime contributes the unique irreducible projective component. The embedded associated prime does not even define a point of Proj because it contains \(A_+\).

Chartwise, any projective point must avoid \(y\), since every prime contains \(x\) but a projective prime cannot contain both \(x\) and \(y\). Thus
\[
\Proj A=D_+(y).
\]
After inverting \(y\), the relation
\[
xy=0
\]
forces
\[
x=0.
\]
The degree-zero chart ring is therefore simply
\[
k.
\]
Hence
\[
\boxed{\Proj A\cong\Spec k}.
\]

This is exactly the projective phenomenon the original challenge was aiming at: an embedded piece supported at the irrelevant cone vertex disappears, while the genuine minimal projective component survives.
\end{solution}
