\subsection*{VI.23.P11: Pullback of a closed immersion in the affine case}
\begin{problem}\label{prob:vi23-p11}
For $A\to B$ and $I\subseteq A$, compute
\[
\Spec B\times_{\Spec A}\Spec(A/I).
\]
\par\textbf{Solution.} We have $B\otimes_AA/I\cong B/IB$. Hence the fiber product is $\Spec(B/IB)$, a closed subscheme of $\Spec B$.
\end{problem}
\paragraph{Provenance.} Canonical affine pullback computation; no general base-change preservation theorem is claimed here.
