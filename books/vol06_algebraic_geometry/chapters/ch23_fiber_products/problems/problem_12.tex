\subsection*{VI.23.P12: The affine diagonal}
\begin{problem}\label{prob:vi23-p12}
Let $A\to B$. Identify the ring map corresponding to $\Delta:\Spec B\to\Spec B\times_{\Spec A}\Spec B$.
\par\textbf{Solution.} The product is $\Spec(B\otimes_AB)$. The diagonal has both projections equal to the identity, so contravariantly it is induced by multiplication $\mu:B\otimes_AB\to B$, $b_1\otimes b_2\mapsto b_1b_2$.
\end{problem}
\paragraph{Provenance.} Canonical diagonal computation.
