\subsection*{VI.23.P5: Product of affine spaces}
\begin{problem}\label{prob:vi23-p05}
Prove $\mathbb A^m_k\times_k\mathbb A^n_k\cong\mathbb A^{m+n}_k$.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This recovers the familiar Cartesian product of affine coordinate spaces from the scheme-theoretic universal property.

\paragraph{Useful ideas before starting.}
Use the affine fiber-product theorem and identify the tensor product of two polynomial rings over the same field.

\paragraph{Guided hints.}
\begin{enumerate}
\item Write both affine spaces as spectra.
\item Compute the tensor product over $k$.
\item Identify the two projection ring maps.
\end{enumerate}

\begin{solution}
Write
\[
\mathbb A^m_k=\Spec k[x_1,\dots,x_m],\qquad
\mathbb A^n_k=\Spec k[y_1,\dots,y_n].
\]
By the affine fiber-product theorem the product over $\Spec k$ is
\[
\Spec\!\left(k[x_1,\dots,x_m]\otimes_k k[y_1,\dots,y_n]\right).
\]
Define a $k$-algebra map
\[
k[x_1,\dots,x_m,y_1,\dots,y_n]
\to
k[x_1,\dots,x_m]\otimes_k k[y_1,\dots,y_n]
\]
by $x_i\mapsto x_i\otimes1$ and $y_j\mapsto1\otimes y_j$.  The polynomial-ring universal
property gives the inverse to the multiplication-on-monomials map, so this is an isomorphism.
Hence the product is
\[
\Spec k[x_1,\dots,x_m,y_1,\dots,y_n]=\mathbb A^{m+n}_k.
\]
Under this identification the two projections simply forget the last $n$ and first $m$
coordinates, respectively.
\end{solution}

\paragraph{What happened mathematically.}
A geometric assertion was reduced to a representability statement, so the canonical map was forced by universality.

\paragraph{Extensions.}
Repeat the argument after restricting to affine opens and compare with the tensor-product formula.
