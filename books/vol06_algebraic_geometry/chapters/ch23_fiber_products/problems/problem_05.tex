\subsection*{VI.23.P5: Product of affine spaces}
\begin{problem}\label{prob:vi23-p05}
Prove $\mathbb A^m_k\times_k\mathbb A^n_k\cong\mathbb A^{m+n}_k$.
\par\textbf{Solution.} Write the factors as spectra of $k[x_1,\dots,x_m]$ and $k[y_1,\dots,y_n]$. Their tensor product over $k$ is canonically $k[x_1,\dots,x_m,y_1,\dots,y_n]$. Taking spectra proves the claim.
\end{problem}
\paragraph{Provenance.} Canonical affine application.
