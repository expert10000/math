\subsection*{VI.23.P17: Points are not merely compatible pairs}
\begin{problem}\label{prob:vi23-p17}
Explain why the underlying set of $X\times_SY$ cannot in general be identified with $|X|\times_{|S|}|Y|$.
\par\textbf{Solution.} A product point $z$ determines a compatible pair $(x,y)$, but above a fixed pair there may be several choices of prime in the relevant residue-field tensor product. For instance, if $K/k$ is a finite separable extension that splits after extending scalars to a sufficiently large field $\Omega$, then $K\otimes_k\Omega$ is a product of fields and its spectrum has several points, although $\Spec K$, $\Spec\Omega$, and $\Spec k$ each have one point.
\end{problem}
\paragraph{Provenance.} Canonical point-theoretic refinement; detailed fibers are reserved for VI/25.
