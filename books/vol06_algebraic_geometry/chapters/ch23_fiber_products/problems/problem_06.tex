\subsection*{VI.23.P6: Fiber product with the base}
\begin{problem}\label{prob:vi23-p06}
Show $X\times_SS\cong X$ when the second map is $\operatorname{id}_S$.
\par\textbf{Solution.} A test map $T\to X\times_SS$ is a pair $(a:T\to X,b:T\to S)$ with $f\circ a=b$. Thus $b$ is forced by $a$, and the universal data are exactly maps $T\to X$.
\end{problem}
\paragraph{Provenance.} Canonical universal-property derivative.
