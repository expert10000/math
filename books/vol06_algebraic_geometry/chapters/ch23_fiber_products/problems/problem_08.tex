\subsection*{VI.23.P8: Associativity}
\begin{problem}\label{prob:vi23-p08}
For $X,Y,Z\to S$, prove
\[
(X\times_SY)\times_SZ\cong X\times_S(Y\times_SZ).
\]
\par\textbf{Solution.} Both schemes represent triples of maps $T\to X,Y,Z$ whose three composites to $S$ are equal. Since they represent the same functor, the desired isomorphism is unique and canonical.
\end{problem}
\paragraph{Provenance.} Canonical formal derivative.
