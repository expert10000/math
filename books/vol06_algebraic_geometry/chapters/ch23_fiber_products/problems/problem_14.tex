\subsection*{VI.23.P14: Equalizer as a pullback of the diagonal}
\begin{problem}\label{prob:vi23-p14}
Let $f,g:T\rightrightarrows X$ be morphisms over $S$. Show that the pullback of $(f,g):T\to X\times_SX$ along $\Delta_{X/S}$ represents maps $Q\to T$ on which $f$ and $g$ become equal.
\par\textbf{Solution.} A map into the pullback is a map $h:Q\to T$ together with a map $Q\to X$ such that $(f\circ h,g\circ h)$ lies on the diagonal. This condition is exactly $f\circ h=g\circ h$, and the map to $X$ is then uniquely forced.
\end{problem}
\paragraph{Provenance.} Canonical equality-condition derivative.
