\subsection*{VI.23.P1: The AG7 diagonal Cartesian square}
\begin{problem}\label{prob:vi23-p01}
Suppose $X\to S\to T$ and $Y\to S\to T$. Prove that
\[
\begin{tikzcd}
X\times_SY \arrow[r] \arrow[d] & S \arrow[d,"\Delta_{S/T}"]\\
X\times_TY \arrow[r] & S\times_TS
\end{tikzcd}
\]
is Cartesian.
\par\textbf{Solution.} A map $Q\to X\times_TY$ is a pair $Q\to X$, $Q\to Y$ whose images agree in $T$. A compatible map $Q\to S$ into the displayed pullback requires the induced pair of maps $Q\to S$ to land on the diagonal, hence to be equal already in $S$. That is exactly the universal condition for a map $Q\to X\times_SY$. Therefore both upper-left objects represent the same functor and are uniquely isomorphic.
\end{problem}
