\subsection*{VI.23.P19: Functor of points}
\begin{problem}\label{prob:vi23-p19}
Prove that for every scheme $T$,
\[
(X\times_SY)(T)\cong X(T)\times_{S(T)}Y(T).
\]
\par\textbf{Solution.} This is exactly the universal property written with $T$ as the test object. Naturality in $T$ follows because precomposition preserves the compatibility equation.
\end{problem}
\paragraph{Provenance.} Canonical categorical reformulation.
