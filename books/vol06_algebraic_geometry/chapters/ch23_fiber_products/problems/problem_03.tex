\subsection*{VI.23.P3: Affine fiber product}
\begin{problem}\label{prob:vi23-p03}
Let $A\to B$ and $A\to C$ be ring maps. Prove
\[
\Spec B\times_{\Spec A}\Spec C\cong\Spec(B\otimes_AC).
\]
\par\textbf{Solution.} For every affine test scheme $\Spec R$, maps to the right correspond to maps $B\otimes_AC\to R$. The tensor-product universal property identifies these with pairs $B\to R$, $C\to R$ agreeing on $A$. Reversing arrows yields precisely compatible maps $\Spec R\to\Spec B$ and $\Spec R\to\Spec C$. The same description for arbitrary test schemes follows locally, so the right-hand side represents the scheme fiber-product functor.
\end{problem}
\paragraph{Provenance.} Canonical affine realization of the AG7 fiber-product theory stream.
