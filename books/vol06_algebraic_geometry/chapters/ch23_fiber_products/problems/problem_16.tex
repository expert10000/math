\subsection*{VI.23.P16: Compatible affine refinements}
\begin{problem}\label{prob:vi23-p16}
Explain why shrinking $\Spec A$ to $D(f)$ and the factors to the corresponding inverse images does not change the local fiber-product construction except by localization.
\par\textbf{Solution.} The local coordinate ring becomes $(B\otimes_AC)_f$. Canonically
\[
(B\otimes_AC)_f\cong B_f\otimes_{A_f}C_f,
\]
with the appropriate images of $f$ inverted. Hence the constructions agree on common principal refinements.
\end{problem}
\paragraph{Provenance.} Canonical local-compatibility derivative.
