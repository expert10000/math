\subsection*{VI.23.P9: Closed intersections by fiber product}
\begin{problem}\label{prob:vi23-p09}
Let $I,J\subseteq A$. Compute
\[
\Spec(A/I)\times_{\Spec A}\Spec(A/J).
\]
\par\textbf{Solution.} The affine formula gives $\Spec((A/I)\otimes_A(A/J))$. The canonical multiplication map identifies this tensor product with $A/(I+J)$. Therefore the fiber product is the scheme-theoretic intersection $V(I+J)$.
\end{problem}
\paragraph{Provenance.} Canonical bridge to VI/22 closed subschemes, without re-migrating VI/22.
