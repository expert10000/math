\subsection*{VI.23.P4: Uniqueness of fiber products}
\begin{problem}\label{prob:vi23-p04}
Given two fiber products $P$ and $P'$ of $X\to S\leftarrow Y$, construct the unique projection-compatible isomorphism $P\cong P'$.
\par\textbf{Solution.} The projections from $P$ form a compatible pair, so the universal property of $P'$ gives a unique $u:P\to P'$. Similarly obtain $v:P'\to P$. Both $v\circ u$ and $\operatorname{id}_P$ have the same two projected components, so uniqueness forces $v\circ u=\operatorname{id}_P$. Likewise $u\circ v=\operatorname{id}_{P'}$.
\end{problem}
\paragraph{Provenance.} Canonical categorical derivative.
