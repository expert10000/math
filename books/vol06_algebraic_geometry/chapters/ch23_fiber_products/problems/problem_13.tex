\subsection*{VI.23.P13: The graph in affine coordinates}
\begin{problem}\label{prob:vi23-p13}
Let $f:\Spec B\to\Spec C$ be a morphism over $\Spec A$, corresponding to $C\to B$. Identify the graph map on rings.
\par\textbf{Solution.} Since $\Spec B\times_A\Spec C=\Spec(B\otimes_AC)$, the graph corresponds to $B\otimes_AC\to B$, $b\otimes c\mapsto b\,f^\#(c)$. Its compositions with the two canonical maps recover $\operatorname{id}_B$ and $f^\#$.
\end{problem}
\paragraph{Provenance.} Canonical graph computation.
