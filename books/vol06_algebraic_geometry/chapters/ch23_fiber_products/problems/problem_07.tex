\subsection*{VI.23.P7: Symmetry}
\begin{problem}\label{prob:vi23-p07}
Construct the canonical isomorphism $X\times_SY\cong Y\times_SX$ and prove its square is the identity.
\par\textbf{Solution.} Swap the two projections. The swapped pair is compatible over $S$, so universality gives a unique map. Swapping again produces a map with the original projections; uniqueness forces it to be the identity.
\end{problem}
\paragraph{Provenance.} Canonical formal derivative.
