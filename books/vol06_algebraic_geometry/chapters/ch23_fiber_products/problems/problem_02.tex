\subsection*{VI.23.P2: The AG7 graph Cartesian square}
\begin{problem}\label{prob:vi23-p02}
Let $f:X\to Y$ be a morphism over $S$. Prove that
\[
\begin{tikzcd}
X \arrow[r,"f"] \arrow[d,"\Gamma_f"'] & Y \arrow[d,"\Delta_{Y/S}"]\\
X\times_SY \arrow[r,"h"'] & Y\times_SY
\end{tikzcd}
\]
is Cartesian.
\par\textbf{Solution.} A map $Q\to X\times_SY$ is a pair $(a,b)$ with compatible images in $S$. Landing in the pullback of the diagonal means $(f\circ a,b)$ factors through $\Delta_Y$, equivalently $b=f\circ a$. Thus the datum reduces uniquely to $a:Q\to X$, and the induced map to $X\times_SY$ is $(a,f\circ a)=\Gamma_f\circ a$.
\end{problem}
