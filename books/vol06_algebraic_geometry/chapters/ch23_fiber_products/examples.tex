\begin{example}[Product of two affine lines]\label{ex:vi23-r01}
Over a field $k$, $k[x]\otimes_k k[y]\cong k[x,y]$, so $\mathbb A^1_k\times_k\mathbb A^1_k\cong\mathbb A^2_k$.
\end{example}

\begin{example}[Product with a point]\label{ex:vi23-r02}
For a $k$-scheme $X$, $X\times_k\Spec k\cong X$ by the universal property.
\end{example}

\begin{example}[A polynomial surface as a pullback]\label{ex:vi23-r03}
The maps $\Spec k[x,z]\to\Spec k[t]$ with $t\mapsto x$ and $\Spec k[y]\to\Spec k[t]$ with $t\mapsto y^2$ give fiber product $\Spec k[x,y,z]/(x-y^2)$.
\end{example}

\begin{example}[Two quotient maps]\label{ex:vi23-r04}
For $A$ and ideals $I,J$, $\Spec(A/I)\times_{\Spec A}\Spec(A/J)\cong\Spec(A/(I+J))$.
\end{example}

\begin{example}[An open restriction]\label{ex:vi23-r05}
For $X=\Spec B\to\Spec A$ and $D(f)\subseteq\Spec A$, the pullback is $\Spec(B\otimes_AA_f)\cong\Spec B_{\varphi(f)}$. This identity is used here only as a fiber-product computation; systematic base change is VI/24.
\end{example}

\begin{example}[Symmetry]\label{ex:vi23-r06}
The flip $b\otimes c\mapsto c\otimes b$ induces $\Spec(B\otimes_AC)\cong\Spec(C\otimes_AB)$, matching symmetry of fiber products.
\end{example}

\begin{example}[Associativity]\label{ex:vi23-r07}
The canonical associativity isomorphism $(B\otimes_AC)\otimes_AD\cong B\otimes_A(C\otimes_AD)$ realizes associativity of affine fiber products.
\end{example}

\begin{example}[The diagonal of affine line]\label{ex:vi23-r08}
For $X=\mathbb A^1_k$, $X\times_kX=\Spec k[x_1,x_2]$, and $\Delta$ corresponds to $k[x_1,x_2]\to k[x]$, $x_1,x_2\mapsto x$.
\end{example}

\begin{example}[Graph of a square map]\label{ex:vi23-r09}
For $f:\mathbb A^1_k\to\mathbb A^1_k$, $t\mapsto t^2$, the graph is $\Spec k[x,y]/(y-x^2)$ inside $\mathbb A^2_k$.
\end{example}

\begin{example}[Identity pullback]\label{ex:vi23-r10}
The square obtained by pulling $X\to S$ against $\operatorname{id}_S$ has upper-left object canonically isomorphic to $X$.
\end{example}

\begin{example}[Empty scheme]\label{ex:vi23-r11}
Because the empty scheme is initial, $\varnothing\times_SY\cong\varnothing$.
\end{example}

\begin{example}[Product over Spec Z]\label{ex:vi23-r12}
$\Spec\mathbb Z[x]\times_{\Spec\mathbb Z}\Spec\mathbb Z[y]\cong\Spec\mathbb Z[x,y]$.
\end{example}

\begin{example}[Common zero locus]\label{ex:vi23-r13}
Inside $\Spec A$, the scheme-theoretic intersection of $V(I)$ and $V(J)$ is the fiber product and is cut out by $I+J$.
\end{example}

\begin{example}[Diagonal imposes equality]\label{ex:vi23-r14}
A map $T\to X\times_SX$ factors through $\Delta_{X/S}$ exactly when its two component maps $T\to X$ are equal.
\end{example}

\begin{example}[Graph criterion]\label{ex:vi23-r15}
A map $T\to X\times_SY$ factors through $\Gamma_f$ exactly when its $Y$-component equals $f$ composed with its $X$-component.
\end{example}

\begin{example}[A field extension product]\label{ex:vi23-r16}
If $K/k$ is a field extension, $\Spec K\times_k\mathbb A^1_k\cong\Spec K[t]$.
\end{example}

\begin{example}[Two copies over a common affine base]\label{ex:vi23-r17}
Maps $A\to B$ and $A\to C$ may identify equations from the base; the tensor product $B\otimes_AC$ is precisely the quotient that enforces those identifications universally.
\end{example}

\begin{example}[A repeated factor]\label{ex:vi23-r18}
$X\times_SX$ carries two projections to $X$ and a diagonal section; these are distinct unless $X\to S$ is itself an isomorphism in a very strong sense.
\end{example}

\begin{example}[Fiber product of open charts]\label{ex:vi23-r19}
If $U\subseteq X$ and $V\subseteq Y$ map into a common open $W\subseteq S$, then $U\times_WV$ is an open subscheme of $X\times_SY$.
\end{example}

\begin{example}[Uniqueness in practice]\label{ex:vi23-r20}
To construct an isomorphism between two candidate pullbacks, use the projections to obtain maps in both directions; uniqueness in the universal property forces the composites to be identities.
\end{example}
