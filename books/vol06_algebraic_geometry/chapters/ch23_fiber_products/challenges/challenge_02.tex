\subsection*{Challenge 2: Residue-field point classification}
\begin{problem}\label{chal:vi23-02}
Develop a precise description of the points above a compatible pair $x\in X$, $y\in Y$ over $s\in S$ in terms of primes of a suitable tensor product of residue fields. Explain what changes when the maps $\kappa(s)\to\kappa(x),\kappa(y)$ are not embeddings in a naive sense.
\end{problem}
