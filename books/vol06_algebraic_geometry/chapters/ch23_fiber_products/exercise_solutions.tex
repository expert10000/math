\paragraph{Exercise VI.23.1.}
\begin{solution}
A fiber product of $X\xrightarrow fS\xleftarrow gY$ is a scheme $P$ together with projections
$p_X:P\to X$ and $p_Y:P\to Y$ satisfying $f p_X=g p_Y$, with the following universal property.
For every scheme $T$ and every compatible pair $a:T\to X$, $b:T\to Y$ satisfying
$f a=g b$, there is a unique $u:T\to P$ such that $p_Xu=a$ and $p_Yu=b$.  Equivalently,
naturally in $T$,
\[
\Hom(T,P)\cong\Hom(T,X)\times_{\Hom(T,S)}\Hom(T,Y).
\]
No choice of affine coordinates enters this definition.
\end{solution}

\paragraph{Exercise VI.23.2.}
\begin{solution}
Let $P$ be the upper-left corner of a Cartesian square and let $\alpha:P'\xrightarrow{\sim}P$ be
an isomorphism compatible with the two outgoing arrows.  For a test scheme $T$, composition with
$\alpha$ gives a bijection $\Hom(T,P')\to\Hom(T,P)$.  Since the original square is Cartesian,
the latter Hom-set is the fiber product of the two relevant Hom-sets.  Hence $P'$ represents the
same functor and the transported square is Cartesian.
\end{solution}

\paragraph{Exercise VI.23.3.}
\begin{solution}
Let $f:X\to S$.  A $T$-point of $X\times_S S$, where the second map is $\id_S$, is a pair
$(a:T\to X,b:T\to S)$ with $f a=b$.  The second component is therefore uniquely determined by
$a$.  Thus
\[
\Hom(T,X\times_SS)\cong\Hom(T,X)
\]
naturally in $T$.  The representing objects are canonically isomorphic, with the first projection
giving $X\times_SS\to X$ and $(\id_X,f)$ giving the inverse.
\end{solution}

\paragraph{Exercise VI.23.4.}
\begin{solution}
If $(P,p_X,p_Y)$ is $X\times_SY$, then $(P,p_Y,p_X)$ satisfies the universal property for
$Y\times_SX$.  Hence universality gives a canonical flip
$\tau:X\times_SY\to Y\times_SX$.  Applying the same construction again gives a composite whose
two projections are the original $p_X,p_Y$.  The identity has the same two projections, so
uniqueness forces $\tau^2=\id$.
\end{solution}

\paragraph{Exercise VI.23.5.}
\begin{solution}
Write $\mathbb A^1_k=\Spec k[x]$ and $\mathbb A^2_k=\Spec k[y,z]$.  Since
\[
k[x]\otimes_k k[y,z]\cong k[x,y,z],
\]
the affine fiber-product theorem gives
\[
\mathbb A^1_k\times_k\mathbb A^2_k
\cong\Spec k[x,y,z]=\mathbb A^3_k.
\]
The two projections correspond to the inclusions of the $x$ variables and the $y,z$ variables.
\end{solution}

\paragraph{Exercise VI.23.6.}
\begin{solution}
The affine formula gives
\[
\Spec(A/I)\times_{\Spec A}\Spec(A/J)
=\Spec\bigl((A/I)\otimes_A(A/J)\bigr).
\]
Define $(A/I)\otimes_A(A/J)\to A/(I+J)$ by
$\bar a\otimes\bar b\mapsto\overline{ab}$.  It is an isomorphism; equivalently tensor the exact
presentation $A\to A/I$ with $A/J$.  Therefore
\[
\Spec(A/I)\times_{\Spec A}\Spec(A/J)\cong\Spec A/(I+J).
\]
This is the scheme-theoretic intersection of the two closed subschemes.
\end{solution}

\paragraph{Exercise VI.23.7.}
\begin{solution}
Let $\varphi:A\to B$ and write $D(f)=\Spec A_f$.  Then
\[
\Spec B\times_{\Spec A}D(f)
\cong\Spec(B\otimes_AA_f).
\]
The universal property of localization gives
\[
B\otimes_AA_f\cong B_{\varphi(f)}.
\]
Hence the pullback is the principal open
$D(\varphi(f))\subseteq\Spec B$.  This computation is the affine prototype for the base-change
statements of VI/24.
\end{solution}

\paragraph{Exercise VI.23.8.}
\begin{solution}
There is a unique morphism $\varnothing\to Z$ for every scheme $Z$.  Thus for every test scheme
$T$, a map $T\to\varnothing\times_SY$ can exist only when there is a map $T\to\varnothing$,
which forces $T=\varnothing$.  The empty scheme itself, with its unique maps to $\varnothing$ and
$Y$, has exactly this universal property.  Therefore
$\varnothing\times_SY\cong\varnothing$.
\end{solution}

\paragraph{Exercise VI.23.9.}
\begin{solution}
For $S=\Spec A$ and $X=\Spec B$,
\[
X\times_SX=\Spec(B\otimes_AB).
\]
The diagonal has both components equal to $\id_X$.  Contravariantly its ring map must therefore
restrict to the identity on both copies of $B$.  The unique such $A$-algebra map is multiplication
\[
\mu:B\otimes_AB\longrightarrow B,\qquad b_1\otimes b_2\longmapsto b_1b_2.
\]
Thus $\Delta_{X/S}=\Spec(\mu)$.
\end{solution}

\paragraph{Exercise VI.23.10.}
\begin{solution}
Let $f:\Spec B\to\Spec C$ correspond to an $A$-algebra map
$f^\#:C\to B$.  Since
\[
\Spec B\times_{\Spec A}\Spec C=\Spec(B\otimes_AC),
\]
the graph $\Gamma_f=(\id,f)$ corresponds to
\[
\gamma:B\otimes_AC\longrightarrow B,\qquad
b\otimes c\longmapsto b\,f^\#(c).
\]
Indeed $\gamma(b\otimes1)=b$ and $\gamma(1\otimes c)=f^\#(c)$, which recover the two graph
components.
\end{solution}

\paragraph{Exercise VI.23.11.}
\begin{solution}
The equality
\[
f\circ p_X=g\circ p_Y
\]
is part of the defining data of a fiber product.  It is needed before the universal property can
even be stated: only pairs of maps to $X$ and $Y$ whose composites to $S$ agree are allowed to
factor through $X\times_SY$.  Thus the two projection composites agree identically, not merely
pointwise.
\end{solution}

\paragraph{Exercise VI.23.12.}
\begin{solution}
Let $\tau:X\times_SY\to Y\times_SX$ be the unique map with components $(p_Y,p_X)$.
Let $\tau'$ be the corresponding flip in the other direction.  Then
$p_X\tau'\tau=p_X$ and $p_Y\tau'\tau=p_Y$.  The identity on $X\times_SY$ has the same
components.  By uniqueness in the fiber-product universal property,
$\tau'\tau=\id$.  The same argument gives $\tau\tau'=\id$.
\end{solution}

\paragraph{Exercise VI.23.13.}
\begin{solution}
A map
\[
T\to (X\times_SY)\times_SZ
\]
is a map $T\to X\times_SY$ together with $T\to Z$, with equal composites to $S$.  The first map
is itself a compatible pair $T\to X,T\to Y$.  Hence the total datum is a triple
$(a,b,c)$ satisfying
\[
f_Xa=f_Yb=f_Zc.
\]
The other bracketing $X\times_S(Y\times_SZ)$ represents exactly the same triples, naturally in
$T$.  Therefore the two iterated fiber products are canonically isomorphic.
\end{solution}

\paragraph{Exercise VI.23.14.}
\begin{solution}
This is the same tensor-product computation as Exercise VI.23.5:
\[
k[x]\otimes_k k[y,z]\cong k[x,y,z].
\]
Therefore
\[
\Spec k[x]\times_k\Spec k[y,z]\cong\Spec k[x,y,z].
\]
\end{solution}

\paragraph{Exercise VI.23.15.}
\begin{solution}
Iterating the affine fiber-product theorem gives
\[
(A/I)\otimes_A(A/J)\otimes_A(A/K)
\cong A/(I+J+K).
\]
Therefore the threefold fiber product is
\[
\Spec A/(I+J+K).
\]
It is the closed subscheme on which all equations from $I$, $J$, and $K$ vanish simultaneously.
Associativity makes the answer independent of the chosen bracketing.
\end{solution}

\paragraph{Exercise VI.23.16.}
\begin{solution}
Let $h:T\to X\times_SX$ have components $h_1,h_2:T\to X$.
If $h=\Delta\circ q$, then $h_1=q=h_2$.  Conversely, if $h_1=h_2=q$, the pair defining $h$
is $(q,q)$, which is exactly the pair defining $\Delta\circ q$.  By uniqueness of maps into the
fiber product, $h=\Delta q$.  Thus factorization through the diagonal is equivalent to equality
of the two components.
\end{solution}

\paragraph{Exercise VI.23.17.}
\begin{solution}
Write a map $h:T\to X\times_SY$ as its components $(a,b)$.  If
$h=\Gamma_f\circ q$, then the first component gives $q=a$ and the second gives $b=f a$.
Conversely, if $b=f a$, then $(a,b)=(a,f a)$ is exactly the compatible pair defining
$\Gamma_f\circ a$.  Uniqueness into the fiber product proves the factorization.
\end{solution}

\paragraph{Exercise VI.23.18.}
\begin{solution}
A point $z\in X\times_SY$ certainly determines a compatible pair
$(x,y)$ with common image $s\in S$.  The converse is not unique in general.  Above a fixed pair
$(x,y)$, the remaining point data are measured by
\[
\Spec\bigl(\kappa(x)\otimes_{\kappa(s)}\kappa(y)\bigr).
\]
This spectrum may have several primes, or in degenerate cases none.  Therefore forgetting residue
fields loses information, and the underlying set of the fiber product need not be the naive
set-theoretic fiber product.
\end{solution}

\paragraph{Exercise VI.23.19.}
\begin{solution}
Take $k=\mathbb R$ and $K=\mathbb C$.  Both $\Spec\mathbb C$ factors have one point over the
single point of $\Spec\mathbb R$, but
\[
\mathbb C\otimes_{\mathbb R}\mathbb C\cong\mathbb C\times\mathbb C.
\]
Hence
\[
\Spec\mathbb C\times_{\Spec\mathbb R}\Spec\mathbb C
\]
has two points.  Thus the unique compatible pair of underlying points lifts to two different
fiber-product points.
\end{solution}

\paragraph{Exercise VI.23.20.}
\begin{solution}
Define
\[
\Phi:B\otimes_AA\to B,\qquad b\otimes a\mapsto b\,\varphi(a),
\]
and
\[
\Psi:B\to B\otimes_AA,\qquad b\mapsto b\otimes1.
\]
Then $\Phi\Psi=\id_B$.  For a pure tensor,
\[
\Psi\Phi(b\otimes a)=b\varphi(a)\otimes1=b\otimes a
\]
by the balancing relation.  Hence $\Phi$ and $\Psi$ are inverse $A$-algebra isomorphisms.
\end{solution}

\paragraph{Exercise VI.23.21.}
\begin{solution}
Let $f\in A$ and denote its images in $B,C$, and $B\otimes_AC$, by the same symbol.  Localization
commutes with tensor products:
\[
(B\otimes_AC)_f
\cong (B\otimes_AA_f)\otimes_{A_f}(C\otimes_AA_f)
\cong B_f\otimes_{A_f}C_f.
\]
One may verify the isomorphism on pure tensors by
$(b\otimes c)/f^n\mapsto (b/f^n)\otimes(c/1)$; the inverse follows from the common universal
property of inverting $f$.
\end{solution}

\paragraph{Exercise VI.23.22.}
\begin{solution}
The absolute product is taken over $\Spec\mathbb Z$.  Since
\[
\mathbb Z[x]\otimes_{\mathbb Z}\mathbb Z[y]
\cong\mathbb Z[x,y],
\]
the affine theorem yields
\[
\Spec\mathbb Z[x]\times_{\Spec\mathbb Z}\Spec\mathbb Z[y]
\cong\Spec\mathbb Z[x,y].
\]
\end{solution}

\paragraph{Exercise VI.23.23.}
\begin{solution}
When local affine pullbacks are compared on a triple overlap, two different chains of restriction
produce two isomorphisms between the same two local fiber products.  Each candidate is compatible
with both projections to the restricted factors.  But the universal property says that there is
only one projection-compatible morphism between two realizations of the same fiber product.
Therefore the two composites agree.  These forced equalities are precisely the cocycle identities
needed for VI/20 gluing.
\end{solution}

\paragraph{Exercise VI.23.24.}
\begin{solution}
For a commutative square with upper-left object $P$,
\[
\begin{tikzcd}
P\arrow[r]\arrow[d]&Y\arrow[d]\\
X\arrow[r]&S,
\end{tikzcd}
\]
the square is Cartesian exactly when, for every scheme $T$, composition with the two projections
induces a natural bijection
\[
\Hom(T,P)\xrightarrow{\sim}
\Hom(T,X)\times_{\Hom(T,S)}\Hom(T,Y).
\]
The word ``natural'' means that these bijections commute with precomposition $T'\to T$.
\end{solution}

Questions about its coordinate ring, its points, geometric fibers, and how fibers vary in families

\end{solution}
