\chapter{Fiber Products}\label{ch:vi23-fiber-products}

Fiber products are the basic mechanism for imposing compatibility between two maps to a common
base.  They simultaneously generalize ordinary products, intersections defined by equations,
graphs, diagonals, and the pullback squares that dominate later scheme theory.  The defining
principle is categorical: the fiber product is characterized not by coordinates but by a universal
property.  Coordinates enter afterward, and for affine schemes they turn the construction into a
tensor product of rings.

\section{The universal property}
Let $f:X\to S$ and $g:Y\to S$ be morphisms of schemes.  A \emph{fiber product} of $X$ and $Y$
over $S$ is a scheme $P$ with morphisms $p_X:P\to X$ and $p_Y:P\to Y$ such that
\[
 f\circ p_X=g\circ p_Y,
\]
and such that for every scheme $T$ with morphisms $a:T\to X$ and $b:T\to Y$ satisfying
$f\circ a=g\circ b$, there exists a unique $u:T\to P$ with
\[
 p_X\circ u=a,\qquad p_Y\circ u=b.
\]
When it exists, we write $P=X\times_S Y$.

\begin{theorem}[Uniqueness up to unique isomorphism]\label{thm:vi23-unique}
Any two fiber products of the same pair $X\to S\leftarrow Y$ are uniquely isomorphic in a way
compatible with both projections.
\end{theorem}

\section{Cartesian squares}
A commutative square
\[
\begin{tikzcd}
P \arrow[r] \arrow[d] & Y \arrow[d,"g"]\\
X \arrow[r,"f"'] & S
\end{tikzcd}
\]
is \emph{Cartesian} if $P$, together with the displayed maps, is a fiber product $X\times_S Y$.
Equivalently, for every $T$ one has a natural bijection
\[
\operatorname{Hom}(T,P)
\cong
\operatorname{Hom}(T,X)
\times_{\operatorname{Hom}(T,S)}
\operatorname{Hom}(T,Y).
\]
This Hom-set formula is the main recognition tool throughout the chapter.

\section{The affine construction}
Suppose
\[
S=\Spec A,\qquad X=\Spec B,\qquad Y=\Spec C,
\]
where $B$ and $C$ are $A$-algebras.  Then
\[
X\times_S Y\cong\Spec(B\otimes_A C).
\]
The two projections correspond to the canonical maps
\[
B\longrightarrow B\otimes_A C,\quad b\mapsto b\otimes1,
\qquad
C\longrightarrow B\otimes_A C,\quad c\mapsto1\otimes c.
\]

\begin{theorem}[Affine fiber product]\label{thm:vi23-affine}
For affine schemes over an affine base,
\[
\Spec B\times_{\Spec A}\Spec C\cong\Spec(B\otimes_A C).
\]
\end{theorem}

\section{Why the tensor product appears}
For any ring $R$, a morphism
\[
\Spec R\to\Spec(B\otimes_A C)
\]
corresponds to a ring homomorphism $B\otimes_A C\to R$.  By the tensor-product universal
property, this is equivalent to a pair of ring maps
\[
B\to R,\qquad C\to R
\]
whose restrictions to $A$ agree.  Reversing arrows gives exactly the universal property of the
scheme fiber product.

\section{Existence for arbitrary schemes}
Fiber products of schemes always exist.  The construction is local on the three schemes involved.
Choose affine opens $U=\Spec A\subseteq S$, $V=\Spec B\subseteq f^{-1}(U)$, and
$W=\Spec C\subseteq g^{-1}(U)$.  Form
\[
V\times_U W=\Spec(B\otimes_A C).
\]
On smaller common affine opens these constructions agree by localization and the uniqueness of
fiber products.  The affine pieces therefore glue, using VI/20, to a scheme $X\times_S Y$.

\begin{theorem}[Existence of fiber products of schemes]\label{thm:vi23-existence}
The category of schemes admits fiber products.
\end{theorem}

\section{Products as fiber products}
Because $\Spec\mathbb Z$ is final by VI/21, the categorical product of two schemes is
\[
X\times Y:=X\times_{\Spec\mathbb Z}Y.
\]
If $X$ and $Y$ are schemes over a field $k$, their product over $k$ means
\[
X\times_kY:=X\times_{\Spec k}Y.
\]
For example,
\[
\mathbb A^m_k\times_k\mathbb A^n_k\cong\mathbb A^{m+n}_k.
\]

\section{Diagonal morphisms}
For $X\to S$, the pair $(\operatorname{id}_X,\operatorname{id}_X)$ induces the diagonal
\[
\Delta_{X/S}:X\longrightarrow X\times_SX.
\]
It is the universal morphism imposing equality of the two projections.  In the affine case
$X=\Spec B$, $S=\Spec A$, it corresponds to multiplication
\[
B\otimes_A B\longrightarrow B,\qquad b_1\otimes b_2\longmapsto b_1b_2.
\]
The later theory of separated morphisms will place geometric conditions on this diagonal; here we
only establish the construction.

\input{chapters/ch23_fiber_products/figures/figure_06}

\section{Graphs as pullbacks of diagonals}
Let $f:X\to Y$ be a morphism over $S$.  Its graph is
\[
\Gamma_f=(\operatorname{id}_X,f):X\to X\times_SY.
\]
The square
\[
\begin{tikzcd}
X \arrow[r,"f"] \arrow[d,"\Gamma_f"'] & Y \arrow[d,"\Delta_{Y/S}"]\\
X\times_SY \arrow[r,"h"'] & Y\times_SY
\end{tikzcd}
\]
is Cartesian.  This is one of the central Cartesian constructions in the AG7 source stream.

\input{chapters/ch23_fiber_products/figures/figure_07}

\section{Changing the equality condition}
Suppose $X\to S\to T$ and $Y\to S\to T$.  The diagonal $S\to S\times_TS$ forces two maps to
$S$ that already agree over $T$ to become equal over $S$.  Consequently
\[
X\times_SY\cong(X\times_TY)\times_{S\times_TS}S.
\]
Equivalently, the square
\[
\begin{tikzcd}
X\times_SY \arrow[r] \arrow[d] & S \arrow[d,"\Delta_{S/T}"]\\
X\times_TY \arrow[r] & S\times_TS
\end{tikzcd}
\]
is Cartesian.  This is the second central Cartesian construction in the AG7 source stream.

\input{chapters/ch23_fiber_products/figures/figure_08}

\section{The points of a fiber product}
For schemes, $X\times_SY$ is not generally just the set of pairs $(x,y)$ with equal image in $S$.
If $x\mapsto s$ and $y\mapsto s$, points of the fiber product lying over the pair $(x,y)$ are
controlled by
\[
\Spec\bigl(\kappa(x)\otimes_{\kappa(s)}\kappa(y)\bigr),
\]
where the tensor product uses the residue-field homomorphisms induced by the two structure maps.
Thus one compatible pair can give zero, one, or several scheme points.  The fields need not be
literally presented as common subfields: the two maps out of $\kappa(s)$ are the structure data
that define the tensor product.  This residue-field refinement is essential whenever field
extensions are present.

\section{Elementary formal identities}
The universal property gives canonical isomorphisms
\[
X\times_SY\cong Y\times_SX,
\]
\[
(X\times_SY)\times_SZ\cong X\times_S(Y\times_SZ),
\]
and
\[
X\times_SS\cong X.
\]
These isomorphisms are not arbitrary coordinate tricks: each is the unique map induced by the same
compatible families of projections.

\section{Boundary with the next chapters}
This chapter develops the fiber-product construction itself.  VI/24 will study \emph{base change}
as a systematic operation on morphisms and properties.  VI/25 will specialize the construction to
scheme-theoretic fibers
\[
X_s=X\times_S\Spec\kappa(s)
\]
and analyze their geometry.  The reserved CA13 tensor-product exercises and the detailed AG7 fiber
examples therefore remain untouched here.

\section{Chapter synthesis}
Fiber products convert compatibility conditions into geometric objects.  Their categorical universal
property explains uniqueness and formal identities; their affine incarnation as tensor products makes
them computable; and gluing makes them available for arbitrary schemes.  Diagonals and graphs show
how equality conditions become pullback squares, while the residue-field description warns that the
underlying points carry more information than a naive set-theoretic product.
