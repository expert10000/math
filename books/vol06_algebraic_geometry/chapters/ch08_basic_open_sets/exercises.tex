\begin{exercise}[VI.08.E1 - Elementary identities]
Prove
\[
D(0)=\varnothing,\qquad D(1)=\Spec A,\qquad D(fg)=D(f)\cap D(g).
\]
\end{exercise}

\begin{exercise}[VI.08.E2 - Powers]
Show $D(f^n)=D(f)$ for every $n\ge1$.
\end{exercise}

\begin{exercise}[VI.08.E3 - Principal-open basis]
If $U=\Spec A\setminus V(I)$, prove
\[
U=\bigcup_{f\in I}D(f).
\]
\end{exercise}

\begin{exercise}[VI.08.E4 - Finite generators]
If $I=(f_1,\dots,f_r)$, show
\[
\Spec A\setminus V(I)=D(f_1)\cup\cdots\cup D(f_r).
\]
\end{exercise}

\begin{exercise}[VI.08.E5 - Inclusion of basic opens]
Prove
\[
D(f)\subseteq D(g)\Longleftrightarrow f\in\sqrt{(g)}.
\]
\end{exercise}

\begin{exercise}[VI.08.E6 - Equality of basic opens]
Prove
\[
D(f)=D(g)\Longleftrightarrow \sqrt{(f)}=\sqrt{(g)}.
\]
\end{exercise}

\begin{exercise}[VI.08.E7 - Units]
Show
\[
D(f)=\Spec A\Longleftrightarrow f\in A^\times.
\]
\end{exercise}

\begin{exercise}[VI.08.E8 - Nilpotents]
Show
\[
D(f)=\varnothing\Longleftrightarrow f\text{ is nilpotent}.
\]
\end{exercise}

\begin{exercise}[VI.08.E9 - Cover of the entire spectrum]
For $f_1,\dots,f_r\in A$, prove
\[
\Spec A=\bigcup_{i=1}^rD(f_i)
\Longleftrightarrow
(f_1,\dots,f_r)=A.
\]
\end{exercise}

\begin{exercise}[VI.08.E10 - Relative cover criterion]
Let $g_1,\dots,g_r,f\in A$.  Prove
\[
D(f)\subseteq\bigcup_{i=1}^rD(g_i)
\Longleftrightarrow
f\in\sqrt{(g_1,\dots,g_r)}.
\]
\end{exercise}

\begin{exercise}[VI.08.E11 - Quasi-compactness]
Prove directly from E10 that every $D(f)$ is quasi-compact.
\end{exercise}

\begin{exercise}[VI.08.E12 - Arithmetic basic opens]
For nonzero $n\in\ZZ$, describe $D(n)\subseteq\Spec\ZZ$ and identify the points removed.
\end{exercise}

\begin{exercise}[VI.08.E13 - Punctured line]
For algebraically closed $k$, describe $D(t)\subseteq\Spec k[t]$ and compute $k[t]_t$.
\end{exercise}

\begin{exercise}[VI.08.E14 - Nested localization]
Prove
\[
(A_f)_{g/1}\cong A_{fg}.
\]
\end{exercise}

\begin{exercise}[VI.08.E15 - Prime correspondence for $A_f$]
Show that prime ideals of $A_f$ correspond bijectively to primes $\mathfrak p\subseteq A$
with $f\notin\mathfrak p$.
\end{exercise}

\begin{exercise}[VI.08.E16 - Topology of the localization correspondence]
Upgrade E15 to a homeomorphism
\[
\Spec A_f\cong D_A(f).
\]
\end{exercise}

\begin{exercise}[VI.08.E17 - Preimage of a basic open]
For $\varphi:A\to B$, prove
\[
(\varphi^\ast)^{-1}(D_A(f))=D_B(\varphi(f)).
\]
\end{exercise}

\begin{exercise}[VI.08.E18 - Quotient then localize]
For $I\subseteq A$, prove
\[
(A/I)_{\bar f}\cong A_f/IA_f
\]
and identify its spectrum with $V(I)\cap D(f)$.
\end{exercise}

\begin{exercise}[VI.08.E19 - Crossing axes]
For $R=k[x,y]/(xy)$, compute $R_{\bar x}$ and $R_{\bar y}$ and interpret the corresponding
basic opens geometrically.
\end{exercise}

\begin{exercise}[VI.08.E20 - Dual numbers]
For $A=k[\varepsilon]/(\varepsilon^2)$, compute $D(\varepsilon)$,
$D(1+\varepsilon)$, and $A_\varepsilon$.
\end{exercise}
