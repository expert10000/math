\begin{solution}[VI.08.E1]
Every prime contains $0$, so $D(0)=\varnothing$, and no proper ideal contains $1$, so $D(1)=\Spec A$.
Moreover $fg\notin\mathfrak p$ iff neither $f$ nor $g$ belongs to the prime $\mathfrak p$, hence
$D(fg)=D(f)\cap D(g)$.
\end{solution}

\begin{solution}[VI.08.E2]
Primality implies $f^n\in\mathfrak p$ iff $f\in\mathfrak p$.  Taking complements gives $D(f^n)=D(f)$.
\end{solution}

\begin{solution}[VI.08.E3]
A prime lies outside $V(I)$ iff it does not contain $I$, iff some $f\in I$ is absent from the prime,
iff it lies in some $D(f)$.  Hence $U=\bigcup_{f\in I}D(f)$.
\end{solution}

\begin{solution}[VI.08.E4]
A prime contains $I=(f_1,\dots,f_r)$ iff it contains every $f_i$.  Therefore the complement of
$V(I)$ is the union of the $D(f_i)$.
\end{solution}

\begin{solution}[VI.08.E5]
We have
\[
D(f)\subseteq D(g)\iff V(g)\subseteq V(f)
\iff \sqrt{(f)}\subseteq\sqrt{(g)}
\iff f\in\sqrt{(g)}.
\]
\end{solution}

\begin{solution}[VI.08.E6]
Equality means both inclusions.  By E5 this is equivalent to
$f\in\sqrt{(g)}$ and $g\in\sqrt{(f)}$, which is equivalent to equality of the two radicals.
\end{solution}

\begin{solution}[VI.08.E7]
If $f$ is a unit, no prime contains it.  Conversely, if $D(f)=\Spec A$, then no maximal ideal
contains $f$.  If $f$ were a nonunit, the proper ideal $(f)$ would lie in a maximal ideal, contradiction.
Thus $f$ is a unit.
\end{solution}

\begin{solution}[VI.08.E8]
$D(f)=\varnothing$ iff $f$ belongs to every prime ideal, iff
$f\in\bigcap_{\mathfrak p}\mathfrak p=\sqrt{(0)}$, iff $f$ is nilpotent.
\end{solution}

\begin{solution}[VI.08.E9]
The union covers iff its complement is empty:
\[
V(f_1,\dots,f_r)=\varnothing.
\]
No prime contains the ideal iff the ideal is the whole ring, since every proper ideal lies in a maximal ideal.
Hence the cover condition is $(f_1,\dots,f_r)=A$.
\end{solution}

\begin{solution}[VI.08.E10]
Taking complements gives
\[
V(g_1,\dots,g_r)\subseteq V(f).
\]
By the closed-set/radical criterion this is equivalent to
$f\in\sqrt{(g_1,\dots,g_r)}$.
\end{solution}

\begin{solution}[VI.08.E11]
Suppose $D(f)\subseteq\bigcup_{\lambda}D(g_\lambda)$.  E10 in its arbitrary-family form gives
$f^N\in(g_\lambda:\lambda)$ for some $N$.  Any expression for $f^N$ as an element of that ideal
uses finitely many $g_{\lambda_i}$, and E10 then gives a finite subcover.
\end{solution}

\begin{solution}[VI.08.E12]
The primes of $\ZZ$ are $(0)$ and $(p)$ for rational primes $p$.  A nonzero integer $n$ belongs to
$(p)$ iff $p\mid n$, and it never belongs to $(0)$.  Thus
\[
D(n)=\{(0)\}\cup\{(p):p\nmid n\}.
\]
\end{solution}

\begin{solution}[VI.08.E13]
For algebraically closed $k$, the points are $(0)$ and $(t-a)$.  The ideal $(t)$ is the only closed point
containing $t$, so $D(t)$ is the line with that point removed.  Localization gives
\[
k[t]_t=k[t,t^{-1}].
\]
\end{solution}

\begin{solution}[VI.08.E14]
The ring $(A_f)_{g/1}$ receives a map from $A$ in which both $f$ and $g$ are units.  Conversely any map from
$A$ inverting both factors uniquely factors through it.  The same universal property characterizes $A_{fg}$,
because if $fg$ is a unit then both $f$ and $g$ are units in a commutative ring.  Hence the rings are canonically isomorphic.
\end{solution}

\begin{solution}[VI.08.E15]
Prime ideals of $S^{-1}A$ correspond to prime ideals of $A$ disjoint from $S$.  For
$S=\{1,f,f^2,\dots\}$, disjointness is equivalent to $f\notin\mathfrak p$.  Extension and contraction are inverse.
\end{solution}

\begin{solution}[VI.08.E16]
Under the bijection of E15, a principal open $D_{A_f}(g/1)$ corresponds to
\[
\{\mathfrak p\in D_A(f):g\notin\mathfrak p\}=D_A(fg).
\]
These sets form bases on both sides, so the bijection and its inverse are continuous.  Hence it is a homeomorphism.
\end{solution}

\begin{solution}[VI.08.E17]
For $\mathfrak q\in\Spec B$,
\[
\mathfrak q\in(\varphi^\ast)^{-1}(D_A(f))
\iff f\notin\varphi^{-1}(\mathfrak q)
\iff \varphi(f)\notin\mathfrak q.
\]
This is exactly $\mathfrak q\in D_B(\varphi(f))$.
\end{solution}

\begin{solution}[VI.08.E18]
The natural map $A\to(A/I)_{\bar f}$ kills $I$ and inverts $f$, so it factors through $A_f/IA_f$.
Conversely the latter ring has the same universal property, giving the canonical isomorphism.  Taking spectra and using
$\Spec(A/I)\cong V(I)$ yields
\[
\Spec((A/I)_{\bar f})\cong V(I)\cap D(f).
\]
\end{solution}

\begin{solution}[VI.08.E19]
In $R_{\bar x}$, $\bar x$ is invertible and $\bar x\bar y=0$, hence $\bar y=0$.  Therefore
\[
R_{\bar x}\cong k[x,x^{-1}].
\]
Similarly $R_{\bar y}\cong k[y,y^{-1}]$.  Geometrically each principal open keeps the punctured component on which the
chosen coordinate is nonzero and removes the other component entirely.
\end{solution}

\begin{solution}[VI.08.E20]
Since $\varepsilon^2=0$, the element $\varepsilon$ is nilpotent, so $D(\varepsilon)=\varnothing$.
Also $(1+\varepsilon)(1-\varepsilon)=1$, so $D(1+\varepsilon)=\Spec A$.  Localizing at a nilpotent element forces
$0=\varepsilon^2$ to become a unit; hence the localized ring is the zero ring:
\[
A_\varepsilon=0.
\]
\end{solution}
