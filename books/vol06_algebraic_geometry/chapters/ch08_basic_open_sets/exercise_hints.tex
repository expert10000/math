\begin{hint}[VI.08.E1]
Translate membership in $D(f)$ into non-membership of $f$ in a prime ideal and use primality for $fg$.
\end{hint}
\begin{hint}[VI.08.E2]
A prime contains $f^n$ iff it contains $f$.
\end{hint}
\begin{hint}[VI.08.E3]
$\mathfrak p\notin V(I)$ means $I\not\subseteq\mathfrak p$.
\end{hint}
\begin{hint}[VI.08.E4]
If a prime contains every generator, it contains the whole ideal.
\end{hint}
\begin{hint}[VI.08.E5]
Take complements and use the radical description of closed-set containment.
\end{hint}
\begin{hint}[VI.08.E6]
Apply E5 in both directions.
\end{hint}
\begin{hint}[VI.08.E7]
If $f$ is not a unit, $(f)$ is contained in a maximal ideal.
\end{hint}
\begin{hint}[VI.08.E8]

\end{hint}
\begin{hint}[VI.08.E9]

\end{hint}
\begin{hint}[VI.08.E10]
Take complements and translate inclusion of closed sets into radical containment.
\end{hint}
\begin{hint}[VI.08.E11]
An ideal-membership relation $f^N\in(g_\lambda)$ uses only finitely many generators.
\end{hint}
\begin{hint}[VI.08.E12]
$(p)$ contains $n$ exactly when $p\mid n$; $(0)$ never contains nonzero $n$.
\end{hint}
\begin{hint}[VI.08.E13]

\end{hint}
\begin{hint}[VI.08.E14]

\end{hint}
\begin{hint}[VI.08.E15]

\end{hint}
\begin{hint}[VI.08.E16]

\end{hint}
\begin{hint}[VI.08.E17]

\end{hint}
\begin{hint}[VI.08.E18]
Both rings are universal for maps killing $I$ and inverting $f$.
\end{hint}
\begin{hint}[VI.08.E19]
Once $\bar x$ is a unit, the equation $\bar x\bar y=0$ forces $\bar y=0$.
\end{hint}
\begin{hint}[VI.08.E20]

\end{hint}
