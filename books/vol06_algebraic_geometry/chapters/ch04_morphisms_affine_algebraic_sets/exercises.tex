% Canonical exercises for VI/04.

\begin{exercise}\label{ex:vi04-target-test}
Let \(Y=V(I)\subseteq\mathbb A_k^m\) and \(\varphi=(f_1,\dots,f_m):X\to\mathbb A_k^m\).
Prove that \(\varphi(X)\subseteq Y\) iff \(g(f_1,\dots,f_m)=0\) in \(k[X]\) for every
\(g\in I\).
\end{exercise}

\begin{exercise}\label{ex:vi04-pullback}
Prove directly that \(\varphi^*(h)=h\circ\varphi\) defines a \(k\)-algebra homomorphism
\(k[Y]\to k[X]\).
\end{exercise}

\begin{exercise}\label{ex:vi04-coordinates}
If \(\varphi=(f_1,\dots,f_m):X\to Y\), show that \(\varphi^*(\bar y_i)=f_i\).  Deduce that
\(\varphi^*\) is uniquely determined by the coordinate functions of \(\varphi\).
\end{exercise}

\begin{exercise}\label{ex:vi04-contravariance}
For \(X\xrightarrow{\varphi}Y\xrightarrow{\psi}Z\), prove
\((\psi\circ\varphi)^*=\varphi^*\circ\psi^*\).
\end{exercise}

\begin{exercise}\label{ex:vi04-reconstruct}
Given a \(k\)-algebra homomorphism \(\Phi:k[Y]\to k[X]\), reconstruct the corresponding
morphism \(X\to Y\) from the images of the coordinate classes on \(Y\).
\end{exercise}

\begin{exercise}\label{ex:vi04-continuity}
Prove \(\varphi^{-1}(V_Y(S))=V_X(\varphi^*S)\), and deduce that every morphism is Zariski
continuous.
\end{exercise}

\begin{exercise}\label{ex:vi04-principal-open}
For \(f\in k[Y]\), prove \(\varphi^{-1}(D_Y(f))=D_X(\varphi^*f)\).
\end{exercise}

\begin{exercise}\label{ex:vi04-parabola}
Prove that \(t\mapsto(t,t^2)\) is an isomorphism \(\mathbb A^1\cong V(y-x^2)\).  Write both
inverse morphisms and both pullbacks.
\end{exercise}

\begin{exercise}\label{ex:vi04-axis}
For \(i(t)=(t,0):\mathbb A^1\to\mathbb A^2\), compute \(i^*\), its kernel, and its image;
verify algebraically that \(i\) is a closed embedding.
\end{exercise}

\begin{exercise}\label{ex:vi04-affine-line-automorphism}
For \(\varphi(t)=at+b\) with \(a\neq0\), compute the inverse morphism and the inverse
\(k\)-algebra automorphism.
\end{exercise}

\begin{exercise}\label{ex:vi04-squaring}
Assume \(k\) is infinite.  For \(s(t)=t^2\), compute \(s^*:k[u]\to k[t]\) and show that it
is injective but not surjective.
\end{exercise}

\begin{exercise}\label{ex:vi04-maps-to-a1}
Assume \(k\) is infinite.  Prove \(\operatorname{Mor}(X,\mathbb A^1)\cong k[X]\).
\end{exercise}

\begin{exercise}\label{ex:vi04-iso-ring}
If \(\varphi:X\to Y\) has \(\varphi^*\) a \(k\)-algebra isomorphism, construct the inverse
morphism to \(\varphi\).
\end{exercise}

\begin{exercise}\label{ex:vi04-closed-quotient}
If \(Y\subseteq X\) is closed, show that inclusion induces \(k[X]\twoheadrightarrow k[Y]\)
and identify its kernel.
\end{exercise}

\begin{exercise}\label{ex:vi04-product-pair}
Show that a morphism \(Z\to X\times Y\) is equivalent to a pair of morphisms \(Z\to X\) and
\(Z\to Y\).  Express the construction in coordinates.
\end{exercise}

\begin{exercise}\label{ex:vi04-constant-map}
Let \(X\neq\varnothing\) and \(y\in Y\).  Show that the constant map \(c_y:X\to Y\) is a
morphism with pullback \(h\mapsto h(y)\cdot1\).  Conversely, if the image of \(\varphi^*\)
is contained in the constant subfield \(k\subseteq k[X]\), show that \(\varphi\) is constant.
\end{exercise}

\begin{exercise}\label{ex:vi04-graph}
For \(f=(f_1,\dots,f_m):X\to\mathbb A^m\), show that its graph in \(X\times\mathbb A^m\)
is cut out by \(y_i-f_i=0\).  Deduce that the graph map is a closed embedding.
\end{exercise}

\begin{exercise}\label{ex:vi04-kernel-image}
Show directly that \(\ker\varphi^*\) consists exactly of the polynomial functions on \(Y\)
that vanish on \(\varphi(X)\).  Deduce
\(V_Y(\ker\varphi^*)=\overline{\varphi(X)}^{\mathrm{Zar}}\).
\end{exercise}
