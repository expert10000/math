% Canonical exercises for VI/04.

\begin{exercise}\label{ex:vi04-target-test}
Let
\[
Y=V(I)\subseteq\mathbb A_k^m
\]
and let
\[
\varphi=(f_1,\dots,f_m):X\to\mathbb A_k^m.
\]
Prove that \(\varphi(X)\subseteq Y\) if and only if
\[
g(f_1,\dots,f_m)=0
\]
in \(k[X]\) for every \(g\in I\).
\end{exercise}

\begin{exercise}\label{ex:vi04-pullback}
Prove directly that
\[
\varphi^\ast(h)=h\circ\varphi
\]
defines a \(k\)-algebra homomorphism
\[
k[Y]\to k[X].
\]
\end{exercise}

\begin{exercise}\label{ex:vi04-coordinates}
If
\[
\varphi=(f_1,\dots,f_m):X\to Y,
\]
show that
\[
\varphi^\ast(\bar y_i)=f_i.
\]
Deduce that \(\varphi^\ast\) is determined uniquely by the coordinate functions of
\(\varphi\).
\end{exercise}

\begin{exercise}\label{ex:vi04-contravariance}
For
\[
X\xrightarrow{\varphi}Y\xrightarrow{\psi}Z,
\]
prove
\[
(\psi\circ\varphi)^\ast=\varphi^\ast\circ\psi^\ast.
\]
\end{exercise}

\begin{exercise}\label{ex:vi04-reconstruct}
Let
\[
\Phi:k[Y]\to k[X]
\]
be a \(k\)-algebra homomorphism.  Using the coordinate classes
\(\bar y_1,\dots,\bar y_m\), construct a morphism
\[
\varphi_\Phi:X\to Y
\]
whose pullback is \(\Phi\).
\end{exercise}

\begin{exercise}\label{ex:vi04-continuity}
Prove that every morphism
\[
\varphi:X\to Y
\]
is Zariski continuous by showing
\[
\varphi^{-1}(V_Y(S))
=
V_X(\varphi^\ast S).
\]
\end{exercise}

\begin{exercise}\label{ex:vi04-principal-open}
For \(f\in k[Y]\), prove
\[
\varphi^{-1}(D_Y(f))
=
D_X(\varphi^\ast f).
\]
\end{exercise}

\begin{exercise}\label{ex:vi04-parabola}
Prove that
\[
t\mapsto(t,t^2)
\]
defines an isomorphism
\[
\mathbb A_k^1\cong V(y-x^2).
\]
Give both inverse morphisms and both pullback homomorphisms.
\end{exercise}

\begin{exercise}\label{ex:vi04-axis}
For
\[
i:\mathbb A_k^1\to\mathbb A_k^2,
\qquad
t\mapsto(t,0),
\]
compute \(i^\ast\), its kernel, and its image.  Verify algebraically that \(i\) is a closed
embedding.
\end{exercise}

\begin{exercise}\label{ex:vi04-affine-line-automorphism}
Let
\[
\varphi(t)=at+b
\]
with \(a\neq0\).  Compute the inverse morphism and the inverse \(k\)-algebra automorphism
of \(k[t]\).
\end{exercise}

\begin{exercise}\label{ex:vi04-squaring}
Assume \(k\) is infinite.  For
\[
s(t)=t^2,
\]
compute
\[
s^\ast:k[u]\to k[t].
\]
Show that it is injective but not surjective.
\end{exercise}



\begin{exercise}\label{ex:vi04-maps-to-a1}
Assume \(k\) is infinite.  Prove
\[
\operatorname{Mor}(X,\mathbb A_k^1)\cong k[X].
\]
\end{exercise}

\begin{exercise}\label{ex:vi04-iso-ring}
Let
\[
\varphi:X\to Y
\]
be a morphism.  Assuming \(\varphi^\ast\) is a \(k\)-algebra isomorphism, construct the
inverse morphism to \(\varphi\).
\end{exercise}

\begin{exercise}\label{ex:vi04-closed-quotient}
Suppose \(Y\subseteq X\) is closed.  Show that the inclusion induces
\[
k[X]\twoheadrightarrow k[Y].
\]
Identify its kernel.
\end{exercise}

\begin{exercise}\label{ex:vi04-challenge-product}
Let \(X\subseteq\mathbb A_k^n\) and \(Y\subseteq\mathbb A_k^m\) be affine algebraic
sets.  Show that a map
\[
Z\to X\times Y
\]
is the same as a pair of morphisms
\[
Z\to X,\qquad Z\to Y.
\]
Express the statement in coordinate functions.
\end{exercise}


\subsection*{Recovered legacy problems}

The following problems are retained from the legacy affine-algebraic-geometry source because
they test the central material of this chapter.  Their wording has been normalized to the
notation and hypotheses of the canonical text.

% Legacy source: theory-of-differential-geometry-4.tex, Problem 7.
\begin{exercise}[Legacy problem: the hyperbola \texorpdfstring{$xy=1$}{xy=1}]
\label{ex:vi04-legacy-hyperbola}
Let
\[
H=V(xy-1)\subseteq\mathbb A_k^2.
\]
Show that \(H\) is not isomorphic to \(\mathbb A_k^1\), and determine all morphisms
\[
\mathbb A_k^1\longrightarrow H.
\]
\end{exercise}

% Legacy source: theory-of-differential-geometry-4.tex, Problem 9.
\begin{exercise}[Legacy problem: products of affine algebraic sets]
\label{ex:vi04-legacy-products}
Let
\[
X\subseteq\mathbb A_k^m,
\qquad
Y\subseteq\mathbb A_k^n
\]
be affine algebraic sets.  Prove that
\[
X\times Y\subseteq\mathbb A_k^{m+n}
\]
is an affine algebraic set.  Prove the natural coordinate-ring isomorphism
\[
k[X\times Y]\cong k[X]\otimes_k k[Y]
\]
under the standard hypotheses used in this chapter.

If \(k\) is algebraically closed and \(X,Y\) are irreducible varieties, prove as a harder
part that \(X\times Y\) is irreducible.
\end{exercise}

% Legacy source: theory-of-differential-geometry-4.tex, Problem 10.
% The legacy solution implicitly assumes algebraically closed k and char(k) != 2;
% the canonical statement makes those assumptions explicit.
\begin{exercise}[Legacy problem: no nonconstant map to a cubic]
\label{ex:vi04-legacy-cubic-map}
Assume that \(k\) is algebraically closed and
\[
\operatorname{char}(k)\neq2.
\]
Let
\[
E=V(y^2-x^3+x)\subseteq\mathbb A_k^2.
\]
Show that every morphism
\[
\mathbb A_k^1\longrightarrow E
\]
is constant.

\emph{Hint.}  Apply the induced homomorphism
\[
k[E]\longrightarrow k[t]
\]
to \(x\) and \(y\), and use unique factorization in \(k[t]\).
\end{exercise}
