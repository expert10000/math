% Hints for VI/04.

\chapter*{Hints for VI/04}
\addcontentsline{toc}{chapter}{Hints for VI/04}

\paragraph{Exercise \ref{ex:vi04-target-test}.}
Unpack the definition of \(V(I)\).

\paragraph{Exercise \ref{ex:vi04-pullback}.}
Composition of functions respects addition and multiplication.

\paragraph{Exercise \ref{ex:vi04-coordinates}.}
Apply the definition of pullback to the coordinate function \(y_i\).

\paragraph{Exercise \ref{ex:vi04-contravariance}.}
Evaluate both sides on an arbitrary \(h\in k[Z]\).

\paragraph{Exercise \ref{ex:vi04-reconstruct}.}
Set
\[
f_i=\Phi(\bar y_i).
\]
Use every relation in \(I(Y)\).

\paragraph{Exercise \ref{ex:vi04-continuity}.}
A point lies in the inverse image exactly when every defining function vanishes after
composition.

\paragraph{Exercise \ref{ex:vi04-principal-open}.}
Take complements of the closed-set formula or test nonvanishing directly.

\paragraph{Exercise \ref{ex:vi04-parabola}.}
Projection onto the first coordinate is the inverse.

\paragraph{Exercise \ref{ex:vi04-axis}.}
The variable \(y\) must go to zero.

\paragraph{Exercise \ref{ex:vi04-affine-line-automorphism}.}
Solve \(u=at+b\) for \(t\).

\paragraph{Exercise \ref{ex:vi04-squaring}.}
The image is \(k[t^2]\).

\paragraph{Exercise \ref{ex:vi04-maps-to-a1}.}
A map to the affine line has one polynomial coordinate.

\paragraph{Exercise \ref{ex:vi04-iso-ring}.}
Use the affine correspondence on \((\varphi^\ast)^{-1}\).

\paragraph{Exercise \ref{ex:vi04-closed-quotient}.}
Use
\[
I(X)\subseteq I(Y).
\]

\paragraph{Exercise \ref{ex:vi04-challenge-product}.}
A map to the product is the concatenation of the two coordinate tuples.


\paragraph{Exercise \ref{ex:vi04-legacy-hyperbola}.}
Use
\[
k[H]\cong k[x,x^{-1}].
\]
Compare the units of this Laurent polynomial ring with the units of \(k[t]\).

\paragraph{Exercise \ref{ex:vi04-legacy-products}.}
Put the defining equations of \(X\) in the \(x\)-variables and those of \(Y\) in the
\(y\)-variables.  For the coordinate ring, use the quotient presentation of a tensor
product of finitely generated \(k\)-algebras.

\paragraph{Exercise \ref{ex:vi04-legacy-cubic-map}.}
Write
\[
a=\varphi^*(x),\qquad b=\varphi^*(y).
\]
Then
\[
b^2=a(a-1)(a+1).
\]
The three factors on the right are pairwise coprime when \(\operatorname{char}(k)\neq2\).
