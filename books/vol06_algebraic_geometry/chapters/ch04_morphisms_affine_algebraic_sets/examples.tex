% Canonical worked examples for VI/04.

\begin{example}[Parabola isomorphism]
Let \(P=V(y-x^2)\).  The maps
\[
\alpha:\mathbb A^1\to P,\quad t\mapsto(t,t^2),
\qquad
\beta:P\to\mathbb A^1,\quad(x,y)\mapsto x
\]
are inverse morphisms.  On rings,
\[
\alpha^*(\bar x)=t,\quad \alpha^*(\bar y)=t^2,
\qquad \beta^*(t)=\bar x.
\]
\end{example}

\begin{example}[Projection from a curve]
For \(C=V(y^2-x^3)\), projection
\[
\pi:C\to\mathbb A^1,\qquad(x,y)\mapsto x
\]
has pullback \(\pi^*:k[t]\to k[C]\), \(t\mapsto\bar x\).  Its image is the subalgebra
\(k[\bar x]\subseteq k[C]\).
\end{example}

\begin{example}[A closed embedding]
The map \(i:\mathbb A^1\to\mathbb A^2\), \(t\mapsto(t,0)\), has pullback
\[
i^*:k[x,y]\to k[t],\qquad x\mapsto t,\ y\mapsto0.
\]
It is surjective with kernel \((y)\), so the image is the closed axis \(V(y)\).
\end{example}

\begin{example}[Morphisms into the hyperbola]
For \(H=V(xy-1)\), a pair \((f,g)\in k[X]^2\) defines \(X\to H\) exactly when
\(fg=1\).  Thus morphisms into \(H\) are controlled by the unit group \(k[X]^\times\).
\end{example}

\begin{example}[Squaring on the affine line]
For \(s(t)=t^2\),
\[
s^*:k[u]\to k[t],\qquad u\mapsto t^2.
\]
When \(k\) is infinite this is injective but not surjective, so \(s\) is not an
isomorphism even though its image may be Zariski dense.
\end{example}

\begin{example}[Translation]
For \(a\in k\), \(\tau_a(t)=t+a\) is an automorphism and
\[
\tau_a^*(t)=t+a,
\qquad
(\tau_a^{-1})^*(t)=t-a.
\]
\end{example}

\begin{example}[Composition is substitution]
Let \(\varphi(t)=(t,t^2)\) and \(\psi(x,y)=x+y\).  Then
\[
(\psi\circ\varphi)(t)=t+t^2
\]
and
\[
(\varphi^*\circ\psi^*)(u)=\varphi^*(x+y)=t+t^2.
\]
\end{example}

\begin{example}[A map constrained by a target relation]
For \(Y=V(v-u^2)\), choosing \(f\in k[X]\) and setting
\[
u\mapsto f,\qquad v\mapsto f^2
\]
automatically defines a morphism \(X\to Y\), since the target relation becomes zero.
\end{example}

\begin{example}[The cusp: bijective but not isomorphic]
Let \(C=V(y^2-x^3)\) and
\[
\nu:\mathbb A^1\to C,\qquad t\mapsto(t^2,t^3).
\]
The map is bijective on \(k\)-rational points: if \(x\neq0\), then \(t=y/x\); the origin
comes from \(t=0\).  But
\[
\nu^*:k[C]\hookrightarrow k[t]
\]
has image \(k[t^2,t^3]\), which does not contain \(t\).  Thus \(\nu\) is not an
isomorphism.
\end{example}

\begin{example}[The graph of a morphism]
For \(f:\mathbb A^1\to\mathbb A^1\), \(f(t)=t^3-t\), the graph is
\[
\Gamma_f=V(y-x^3+x)\subseteq\mathbb A^2.
\]
The graph embedding \(t\mapsto(t,t^3-t)\) is closed, and projection to the first coordinate
is its inverse onto \(\Gamma_f\).
\end{example}
