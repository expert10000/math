% Canonical worked examples for VI/04.

\begin{example}[Parabola isomorphism]
Let
\[
P=V(y-x^2).
\]
Define
\[
\alpha:\mathbb A_k^1\to P,
\qquad
t\mapsto(t,t^2),
\]
and
\[
\beta:P\to\mathbb A_k^1,
\qquad
(x,y)\mapsto x.
\]
Both are morphisms, and
\[
\beta\circ\alpha=\operatorname{id}_{\mathbb A^1},
\qquad
\alpha\circ\beta=\operatorname{id}_{P}.
\]
Thus \(P\cong\mathbb A_k^1\).

On rings,
\[
\alpha^\ast(\bar x)=t,\qquad
\alpha^\ast(\bar y)=t^2,
\]
while
\[
\beta^\ast(t)=\bar x.
\]
\end{example}

\begin{example}[Projection from a curve]
Let
\[
C=V(y^2-x^3)\subseteq\mathbb A_k^2.
\]
The map
\[
\pi:C\to\mathbb A_k^1,
\qquad
(x,y)\mapsto x
\]
is a morphism.  Its pullback is
\[
\pi^\ast:k[t]\to k[C],
\qquad
t\mapsto\bar x.
\]
\end{example}

\begin{example}[A closed embedding]
The map
\[
i:\mathbb A_k^1\to\mathbb A_k^2,
\qquad
t\mapsto(t,0)
\]
has image \(V(y)\).  The pullback
\[
i^\ast:k[x,y]\to k[t]
\]
is
\[
x\mapsto t,\qquad y\mapsto0.
\]
It is surjective with kernel \((y)\).
\end{example}

\begin{example}[A map into a hyperbola]
Let
\[
H=V(xy-1)\subseteq\mathbb A_k^2.
\]
A pair of polynomial functions \(f,g\in k[X]\) defines a morphism
\[
X\to H
\]
exactly when
\[
fg=1
\]
in \(k[X]\).  Thus morphisms into \(H\) correspond to units of \(k[X]\): choose a unit
\(u\), then send
\[
x\mapsto u,\qquad y\mapsto u^{-1}.
\]
\end{example}

\begin{example}[Squaring on the affine line]
Let
\[
s:\mathbb A_k^1\to\mathbb A_k^1,
\qquad
t\mapsto t^2.
\]
Then
\[
s^\ast:k[u]\to k[t],
\qquad
u\mapsto t^2.
\]
The pullback is injective when \(k\) is infinite, but not surjective, since \(t\) is not a
polynomial in \(t^2\).  Thus \(s\) is not an isomorphism.
\end{example}

\begin{example}[Translation]
For \(a\in k\),
\[
\tau_a:\mathbb A_k^1\to\mathbb A_k^1,
\qquad
t\mapsto t+a
\]
is an automorphism.  Its pullback is
\[
\tau_a^\ast:k[t]\to k[t],
\qquad
t\mapsto t+a.
\]
The inverse pullback sends
\[
t\mapsto t-a.
\]
\end{example}

\begin{example}[Composition and substitution]
Let
\[
\varphi:\mathbb A_k^1\to\mathbb A_k^2,
\qquad
t\mapsto(t,t^2),
\]
and
\[
\psi:\mathbb A_k^2\to\mathbb A_k^1,
\qquad
(x,y)\mapsto x+y.
\]
Then
\[
(\psi\circ\varphi)(t)=t+t^2.
\]
On rings,
\[
\psi^\ast(u)=x+y,
\]
and
\[
\varphi^\ast(x)=t,\qquad
\varphi^\ast(y)=t^2.
\]
Hence
\[
(\varphi^\ast\circ\psi^\ast)(u)=t+t^2,
\]
exactly as contravariance predicts.
\end{example}

\begin{example}[A polynomial map constrained by relations]
Let
\[
Y=V(v-u^2)\subseteq\mathbb A_k^2.
\]
To define a morphism
\[
X\to Y,
\]
it is enough to choose one function \(f\in k[X]\) and set
\[
u\mapsto f,\qquad v\mapsto f^2.
\]
The defining relation is automatically respected:
\[
f^2-f^2=0.
\]
\end{example}
