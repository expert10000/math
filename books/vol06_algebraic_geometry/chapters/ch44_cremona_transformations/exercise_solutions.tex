% VI44-DETAILED-EXERCISE-01
\subsection*{Exercise VI.44.1}
\begin{solution}
The three forms \(xz,yz,z^2\) share the common factor \(z\).  On the dense open set \(z\neq0\),
\[
[xz:yz:z^2]=[x:y:z].
\]
A rational map is determined on a dense open set, and primitive reduction cancels the common factor, so this triple represents the identity map rather than a map with a genuine base line.
\end{solution}

% VI44-DETAILED-EXERCISE-02
\subsection*{Exercise VI.44.2}
\begin{solution}
The common zero equations are \(xy=xz=yz=0\).  If two coordinates were nonzero, their product would be nonzero, so at least two coordinates vanish.  Projectively this gives exactly
\[
[1:0:0],\ [0:1:0],\ [0:0:1].
\]
No other point lies in the base locus.
\end{solution}

% VI44-DETAILED-EXERCISE-03
\subsection*{Exercise VI.44.3}
\begin{solution}
Direct substitution gives
\[
\sigma[1:2:3]=[6:3:2].
\]
Applying \(\sigma\) again gives \([6:12:18]\), which rescales to \([1:2:3]\).  The example is a numerical instance of the rational identity \(\sigma^2=\mathrm{id}\).
\end{solution}

% VI44-DETAILED-EXERCISE-04
\subsection*{Exercise VI.44.4}
\begin{solution}
On \(x=0\) and away from \(p_2,p_3\), both \(y,z\) are nonzero.  Hence
\[
\sigma[0:y:z]=[yz:0:0]=[1:0:0]=p_1.
\]
Thus the whole open part of the coordinate line is contracted to one point; its closure is an exceptional curve.
\end{solution}

% VI44-DETAILED-EXERCISE-05
\subsection*{Exercise VI.44.5}
\begin{solution}
A target point lies on \(X+Y+Z=0\) exactly when the pulled-back coordinates satisfy
\[
yz+xz+xy=0.
\]
Thus the inverse image is the conic \(V_+(xy+xz+yz)\).  It passes through the three base points because every term vanishes there.
\end{solution}

% VI44-DETAILED-EXERCISE-06
\subsection*{Exercise VI.44.6}
\begin{solution}
The vector space of all conics has basis \(x^2,y^2,z^2,xy,xz,yz\).  Vanishing at \(p_1,p_2,p_3\) forces the coefficients of \(x^2,y^2,z^2\) to be zero, leaving
\[
\langle xy,xz,yz\rangle.
\]
Hence the vector-space dimension is three and the projectivized system is a net \(\mathbb P^2\).
\end{solution}

% VI44-DETAILED-EXERCISE-07
\subsection*{Exercise VI.44.7}
\begin{solution}
A projective automorphism is represented by an invertible \(3\times3\) matrix of linear forms.  If all three coordinate forms vanished at a projective point, the matrix would send a nonzero vector to zero, contradicting invertibility.  Hence its base locus is empty.
\end{solution}

% VI44-DETAILED-EXERCISE-08
\subsection*{Exercise VI.44.8}
\begin{solution}
Substituting the quadratic triple into itself yields
\[
[x^2yz:xy^2z:xyz^2]=xyz[x:y:z].
\]
The naive degree is four, but the primitive triple is obtained after cancelling the common cubic factor \(xyz\), leaving degree one.  This is the algebraic shadow of exceptional curves being sent to inverse base points.
\end{solution}

% VI44-DETAILED-EXERCISE-09
\subsection*{Exercise VI.44.9}
\begin{solution}
A general line has \(d=1\) and multiplicities \(m_i=0\).  Therefore
\[
\deg\sigma_*(L)=2.
\]
Since \(\sigma\) is an isomorphism on a dense torus, the transform is birational to the original line; explicitly it is a conic through the three inverse base points.
\end{solution}

% VI44-DETAILED-EXERCISE-10
\subsection*{Exercise VI.44.10}
\begin{solution}
For a line through \(p_1\) only, \(d=1\) and \((m_1,m_2,m_3)=(1,0,0)\).  Thus
\[
\deg\sigma_*(L)=2-1=1.
\]
Provided the line is not exceptional, its transform is another line.
\end{solution}

% VI44-DETAILED-EXERCISE-11
\subsection*{Exercise VI.44.11}
\begin{solution}
An irreducible conic through all three base points has \(d=2\) and \(m_1=m_2=m_3=1\).  Hence
\[
\deg\sigma_*(Q)=4-3=1.
\]
Indeed such a conic is a member of the defining net and is the pullback of a target line.
\end{solution}

% VI44-DETAILED-EXERCISE-12
\subsection*{Exercise VI.44.12}
\begin{solution}
The line \(x=0\) contains two base points and is itself exceptional.  It is contracted to \(p_1\), so the image is not a curve.  The formal number
\[
2\cdot1-1-1=0
\]
correctly signals contraction, but the curve-degree formula is stated for noncontracted transforms and should not be interpreted as a degree-zero plane curve.
\end{solution}

% VI44-DETAILED-EXERCISE-13
\subsection*{Exercise VI.44.13}
\begin{solution}
For a cubic through each base point simply,
\[
d=3,\qquad m_1=m_2=m_3=1,
\]
so
\[
\deg\sigma_*(C)=6-3=3.
\]
Thus the plane degree stays three.  The restriction is birational on the nonexceptional part, so normalization preserves the function field of the cubic.
\end{solution}

% VI44-DETAILED-EXERCISE-14
\subsection*{Exercise VI.44.14}
\begin{solution}
If a smooth cubic avoids all three base points, \(m_i=0\), so its image has degree \(6\).  The image is birational to a genus-one curve.  A smooth plane sextic has genus \(10\), so the sextic image must acquire singularities whose delta-invariants account for the genus drop.
\end{solution}

% VI44-DETAILED-EXERCISE-15
\subsection*{Exercise VI.44.15}
\begin{solution}
On \(z=1\),
\[
\sigma[x:y:1]=[y:x:xy].
\]
When \(xy\neq0\), divide by \(xy\):
\[
[y:x:xy]=[1/x:1/y:1].
\]
Thus on the dense affine torus the map is \((x,y)\mapsto(x^{-1},y^{-1})\).
\end{solution}

% VI44-DETAILED-EXERCISE-16
\subsection*{Exercise VI.44.16}
\begin{solution}
The induced pullback on the rational function field sends
\[
u\mapsto u^{-1},\qquad v\mapsto v^{-1}.
\]
Applying this substitution twice fixes both algebraically independent generators, so it is an automorphism of \(k(u,v)\) of order two.  Hence the corresponding plane map is birational.
\end{solution}

% VI44-DETAILED-EXERCISE-17
\subsection*{Exercise VI.44.17}
\begin{solution}
Projective automorphisms \(A,B\) have degree one and regular inverses.  Therefore \(B\sigma A\) is birational with inverse
\[
A^{-1}\sigma B^{-1}.
\]
Linear changes of source and target coordinates preserve the fact that the primitive coordinate forms have degree two, so the conjugate is again quadratic.
\end{solution}

% VI44-DETAILED-EXERCISE-18
\subsection*{Exercise VI.44.18}
\begin{solution}
Three noncollinear points impose three independent linear conditions on conics, leaving the complete net with no fixed curve component.  If the three points are collinear, the geometry degenerates: the line through them can enter as a fixed component of relevant sublinear systems, and no projective automorphism carries the configuration to the coordinate triangle.  Thus the standard proper-base-point model requires noncollinearity.
\end{solution}

% VI44-DETAILED-EXERCISE-19
\subsection*{Exercise VI.44.19}
\begin{solution}
Insert \(d=4\), \((m_1,m_2,m_3)=(2,1,1)\):
\[
\deg C'=8-2-1-1=4.
\]
So this multiplicity pattern exactly compensates the quadratic degree increase and leaves the plane degree unchanged.
\end{solution}

% VI44-DETAILED-EXERCISE-20
\subsection*{Exercise VI.44.20}
\begin{solution}
For \(d=5\), \((m_1,m_2,m_3)=(3,2,1)\),
\[
\deg C'=10-3-2-1=4.
\]
The conclusion assumes no exceptional coordinate line occurs as a component; otherwise fixed components must first be separated.
\end{solution}

% VI44-DETAILED-EXERCISE-21
\subsection*{Exercise VI.44.21}
\begin{solution}
A general line avoiding the base points maps birationally to a conic.  The source and image have the same function field and are both rational, yet their plane degrees are \(1\) and \(2\).  Therefore plane degree depends on the chosen embedding and cannot be a birational invariant of abstract curves.
\end{solution}

% VI44-DETAILED-EXERCISE-22
\subsection*{Exercise VI.44.22}
\begin{solution}
Suppose \(\psi\) contracts a curve into a base point of \(\phi\).  When the homogeneous coordinates of \(\psi\) are substituted into those of \(\phi\), the exceptional component occurs in all resulting forms and becomes a common factor.  Primitive reduction cancels it, producing
\[
\deg(\phi\circ\psi)<\deg\phi\,\deg\psi.
\]
The cancellation in \(\sigma^2\) is the model case.
\end{solution}

% VI44-DETAILED-EXERCISE-23
\subsection*{Exercise VI.44.23}
\begin{solution}
If \(\phi:X\dashrightarrow Y\) and \(\psi:Y\dashrightarrow X\) are rational inverses, pullback of rational functions gives field maps
\[
\phi^*:k(Y)\to k(X),\qquad \psi^*:k(X)\to k(Y).
\]
On the dense open sets where the maps are inverse morphisms, the composites are identities.  Hence the function fields are isomorphic.
\end{solution}

% VI44-DETAILED-EXERCISE-24
\subsection*{Exercise VI.44.24}
\begin{solution}
Near a base point, different approach directions can have different limiting images.  Lines through a point are parametrized by their slopes together with the vertical direction, i.e. by a projective line \(\mathbb P^1\).  Replacing the point by this \(\mathbb P^1\) records the missing direction data and is exactly the geometric idea of the blow-up.
\end{solution}
