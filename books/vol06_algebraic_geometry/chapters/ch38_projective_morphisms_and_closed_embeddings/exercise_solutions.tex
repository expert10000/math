% VI/38 detailed exercise solutions -- dossier-preserving refinement.

% VI38-DETAILED-EXERCISE-01
\paragraph{Solution VI.38.E1.}
By definition,
\[
U_\varphi
=
\{\mathfrak q\in\Proj T:\varphi^{-1}(\mathfrak q)\not\supseteq S_+\}.
\]
Thus \(\mathfrak q\in U_\varphi\) exactly when there is a homogeneous element
\(f\in S_+\) such that \(\varphi(f)\notin\mathfrak q\).  For such an \(f\),
\(\mathfrak q\in D_+(\varphi(f))\). Conversely, membership in one of these standard
opens gives such an \(f\). Hence
\[
\boxed{U_\varphi=\bigcup_{f\in S_+}D_+(\varphi(f))},
\]
and this is open because every \(D_+(\varphi(f))\) is open in \(\Proj T\).

% VI38-DETAILED-EXERCISE-02
\paragraph{Solution VI.38.E2.}
Let \(\varphi:k[x,y]\to k[u,v]\) satisfy \(x\mapsto u\) and \(y\mapsto0\).
The induced Proj map is defined at \(\mathfrak q\in\Proj k[u,v]\) precisely when
\(\varphi((x,y))\not\subseteq\mathfrak q\).  Since
\[
\varphi((x,y))=(u),
\]
the bad locus is
\[
V_+(u)=\{(u)\}\subseteq\Proj k[u,v]\cong\mathbb P^1_k.
\]
In homogeneous coordinates this is the point \([0:1]\). Therefore the maximal
domain is
\[
\boxed{D_+(u)=\mathbb P^1_k\setminus\{[0:1]\}}.
\]
On that domain contraction lands in \(D_+(x)\subseteq\mathbb P^1\).

% VI38-DETAILED-EXERCISE-03
\paragraph{Solution VI.38.E3.}
Assume
\[
T_+\subseteq\sqrt{\varphi(S_+)T}.
\]
If the Proj morphism were undefined at a homogeneous prime \(\mathfrak q\in\Proj T\),
then \(\varphi(S_+)\subseteq\mathfrak q\), hence
\(\varphi(S_+)T\subseteq\mathfrak q\).  Since \(\mathfrak q\) is prime, it contains
the radical:
\[
\sqrt{\varphi(S_+)T}\subseteq\mathfrak q.
\]
The assumed inclusion would then give \(T_+\subseteq\mathfrak q\), contradicting
\(\mathfrak q\in\Proj T\).  Thus the domain is all of \(\Proj T\).

% VI38-DETAILED-EXERCISE-04
\paragraph{Solution VI.38.E4.}
Suppose \(\varphi(S_r)\subseteq T_{dr}\) for a fixed \(d>0\).  Define the
\(d\)-th Veronese regrading
\[
T^{(d)}=\bigoplus_{r\ge0}T_{dr},
\qquad
(T^{(d)})_r=T_{dr}.
\]
Then for every \(r\),
\[
\varphi(S_r)\subseteq (T^{(d)})_r,
\]
so the same ring homomorphism is degree preserving when regarded as
\[
\boxed{\varphi:S\longrightarrow T^{(d)}}.
\]
This is the precise graded-ring mechanism behind homogeneous coordinate formulas
whose coordinates all have degree \(d\).

% VI38-DETAILED-EXERCISE-05
\paragraph{Solution VI.38.E5.}
Let \(F_0,\ldots,F_m\in k[x_0,\ldots,x_n]_d\).  They determine the graded map
\[
k[y_0,\ldots,y_m]\longrightarrow k[x_0,\ldots,x_n]^{(d)},
\qquad y_i\longmapsto F_i.
\]
At a homogeneous prime \(\mathfrak q\) of the source projective space the induced
Proj map is defined iff the images of the target irrelevant ideal are not all in
\(\mathfrak q\), i.e. iff not all \(F_i\) vanish there. Consequently the maximal
domain is
\[
\boxed{\mathbb P^n\setminus V_+(F_0,\ldots,F_m)}.
\]
On \(D_+(F_i)\) the map is given affinely by the regular ratios
\(F_j/F_i\); these formulas agree on overlaps, so they glue.

% VI38-DETAILED-EXERCISE-06
\paragraph{Solution VI.38.E6.}
Write \(B=A/I\).  The map
\[
\Spec B\longrightarrow\Spec A
\]
sends a prime \(\mathfrak p/I\) to \(\mathfrak p\).  Prime ideals of \(B\) are in
bijection with primes of \(A\) containing \(I\), so the map is a homeomorphism onto
\[
V(I)\subseteq\Spec A,
\]
which is closed.  For a basic open \(D(f)\subseteq\Spec A\), localization gives
\[
A_f\twoheadrightarrow B_{\bar f}\cong A_f/IA_f.
\]
Thus the map of structure sheaves is locally surjective.  Hence
\[
\boxed{\Spec(A/I)\hookrightarrow\Spec A}
\]
is a closed immersion.

% VI38-DETAILED-EXERCISE-07
\paragraph{Solution VI.38.E7.}
Let \(I\subseteq S\) be homogeneous and let \(f\) be homogeneous. Ordinary
localization commutes with quotient:
\[
(S/I)_{\bar f}\cong S_f/IS_f.
\]
This is an isomorphism of graded rings. Taking degree-zero pieces yields a
surjection
\[
S_{(f)}=(S_f)_0\twoheadrightarrow ((S/I)_{\bar f})_0=(S/I)_{(\bar f)}.
\]
Its kernel is
\[
(IS_f)_0=:I_{(f)}.
\]
Therefore
\[
\boxed{(S/I)_{(\bar f)}\cong S_{(f)}/I_{(f)}}.
\]
This is the chartwise algebra used for projective closed immersions.

% VI38-DETAILED-EXERCISE-08
\paragraph{Solution VI.38.E8.}
Take
\[
S=k[x_0,\ldots,x_n],\qquad I=(x_n).
\]
Then
\[
S/I\cong k[x_0,\ldots,x_{n-1}].
\]
The quotient is homogeneous and surjective, so contravariance of Proj gives a
closed immersion
\[
\Proj(S/I)\hookrightarrow\Proj S.
\]
Identifying the two Proj schemes gives
\[
\boxed{\mathbb P^{n-1}_k\hookrightarrow\mathbb P^n_k},
\]
with image \(V_+(x_n)\), the standard hyperplane.

% VI38-DETAILED-EXERCISE-09
\paragraph{Solution VI.38.E9.}
Closed immersions are affine-local on the target.  Let a closed immersion over an
affine chart be
\[
\Spec(A/I)\hookrightarrow\Spec A.
\]
Base change along \(\Spec A'\to\Spec A\) gives
\[
\Spec((A/I)\otimes_AA')\longrightarrow\Spec A'.
\]
Using right exactness of tensor product,
\[
(A/I)\otimes_AA'\cong A'/IA'.
\]
Hence the base-changed map is the quotient closed immersion
\[
\boxed{\Spec(A'/IA')\hookrightarrow\Spec A'}.
\]
The affine calculations glue, proving stability under arbitrary base change.

% VI38-DETAILED-EXERCISE-10
\paragraph{Solution VI.38.E10.}
Choose a projective factorization
\[
X\hookrightarrow\mathbb P^n_Y\longrightarrow Y.
\]
For any \(Y'\to Y\), base change gives a closed immersion
\[
i_{Y'}:X\times_YY'\hookrightarrow\mathbb P^n_Y\times_YY'.
\]
By the base-change identity for relative projective space,
\[
\mathbb P^n_Y\times_YY'\cong\mathbb P^n_{Y'}.
\]
Thus
\[
X_{Y'}\hookrightarrow\mathbb P^n_{Y'}\to Y'
\]
is a projective factorization. Therefore projective morphisms are stable under
arbitrary base change.

% VI38-DETAILED-EXERCISE-11
\paragraph{Solution VI.38.E11.}
Consider
\[
\theta:k[Y_0,Y_1,Y_2]\longrightarrow k[s,t]^{(2)},
\quad
Y_0\mapsto s^2,\ Y_1\mapsto st,\ Y_2\mapsto t^2.
\]
The target is generated in its regraded degree one by \(s^2,st,t^2\), so \(\theta\)
is surjective.  The polynomial
\[
Q=Y_0Y_2-Y_1^2
\]
lies in the kernel.  Modulo \(Q\), every monomial can be reduced until it contains
not both \(Y_0\) and \(Y_2\); hence it has one of the forms
\(Y_0^aY_1^b\) or \(Y_1^bY_2^c\). Their images are distinct monomials in \(s,t\)
unless the source monomials coincide. Thus no further linear relation remains and
\[
\boxed{\ker\theta=(Y_0Y_2-Y_1^2)}.
\]
Consequently the quadratic Veronese image is the conic \(V_+(Q)\subseteq\mathbb P^2\).

% VI38-DETAILED-EXERCISE-12
\paragraph{Solution VI.38.E12.}
Let
\[
x_0^{a_0}\cdots x_n^{a_n}
\]
have total degree \(rd\).  Regard the exponent vector as a multiset containing
\(a_i\) copies of \(x_i\). Partition this multiset into \(r\) groups of exactly
\(d\) entries. Each group determines a monomial \(M_j\) of degree \(d\), and
\[
x_0^{a_0}\cdots x_n^{a_n}=M_1\cdots M_r.
\]
Hence the Veronese ring
\[
S^{(d)}=\bigoplus_{r\ge0}S_{rd}
\]
is generated, in its regraded degree one, by the degree-\(d\) monomials of \(S\).

% VI38-DETAILED-EXERCISE-13
\paragraph{Solution VI.38.E13.}
For \(n=2,d=2\), the number of degree-two monomials in three variables is
\[
\binom{n+d}{d}=\binom42=6.
\]
Thus the target is \(\mathbb P^5\), so \(N=5\).  The six homogeneous coordinates
are
\[
\boxed{x_0^2,\
 x_0x_1,\ x_0x_2,\ x_1^2,\ x_1x_2,\ x_2^2}.
\]
Accordingly
\[
\nu_2([x_0:x_1:x_2])
=[x_0^2:x_0x_1:x_0x_2:x_1^2:x_1x_2:x_2^2].
\]

% VI38-DETAILED-EXERCISE-14
\paragraph{Solution VI.38.E14.}
Let \(A=k[x_0,\ldots,x_m]\), \(B=k[y_0,\ldots,y_n]\), and
\(A\#B=\bigoplus_{d\ge0}A_d\otimes_kB_d\).  Put \(z_{ab}=x_a\otimes y_b\).
On \(D_+(z_{ij})\), degree-zero fractions include
\[
\frac{z_{aj}}{z_{ij}}=\frac{x_a}{x_i},
\qquad
\frac{z_{ib}}{z_{ij}}=\frac{y_b}{y_j}.
\]
Every degree-zero fraction is a polynomial in these ratios, so
\[
(A\#B)_{(z_{ij})}
\cong
A_{(x_i)}\otimes_k B_{(y_j)}.
\]
Therefore
\[
D_+(z_{ij})\cong D_+(x_i)\times D_+(y_j)\cong\mathbb A^m\times\mathbb A^n.
\]

% VI38-DETAILED-EXERCISE-15
\paragraph{Solution VI.38.E15.}
Under \(z_{ij}\mapsto x_i y_j\), a \(2\times2\) minor maps to
\[
z_{ij}z_{\ell r}-z_{ir}z_{\ell j}
\longmapsto
x_i y_jx_\ell y_r-x_i y_rx_\ell y_j=0.
\]
Thus every minor belongs to the kernel of the Segre coordinate map.  Geometrically,
the matrix \((z_{ij})\) arising from a Segre point is the outer product
\((x_i)(y_j)\), hence has rank at most one.

% VI38-DETAILED-EXERCISE-16
\paragraph{Solution VI.38.E16.}
Let \(J\) be the ideal generated by the \(2\times2\) minors.  The relation
\[
z_{ij}z_{\ell r}\equiv z_{ir}z_{\ell j}\pmod J
\]
permits an interchange of two column indices while leaving the multiset of row
indices fixed.  Hence any two monomials with the same row-index multiset and the
same column-index multiset are congruent modulo \(J\).

The image of
\[
z_{i_1j_1}\cdots z_{i_dj_d}
\]
is
\[
(x_{i_1}\cdots x_{i_d})(y_{j_1}\cdots y_{j_d}),
\]
which records exactly those two multisets.  Group any polynomial in the kernel by
its image monomial.  Within each group the coefficients sum to zero, so the group
is a linear combination of differences of monomials with the same image; each such
difference lies in \(J\) by repeated swaps. Therefore
\[
\boxed{\ker\theta=J}.
\]

% VI38-DETAILED-EXERCISE-17
\paragraph{Solution VI.38.E17.}
For \(\mathbb P^1\times\mathbb P^1\), the Segre coordinate matrix is
\[
\begin{pmatrix}
Z_{00}&Z_{01}\\
Z_{10}&Z_{11}
\end{pmatrix}.
\]
There is only one \(2\times2\) minor, so the Segre ideal is
\[
(Z_{00}Z_{11}-Z_{01}Z_{10}).
\]
By the Segre kernel theorem the homogeneous coordinate ring of the image is
\[
k[Z_{00},Z_{01},Z_{10},Z_{11}]/(Z_{00}Z_{11}-Z_{01}Z_{10}).
\]
Hence
\[
\boxed{\mathbb P^1\times\mathbb P^1
\cong V_+(Z_{00}Z_{11}-Z_{01}Z_{10})\subseteq\mathbb P^3}.
\]

% VI38-DETAILED-EXERCISE-18
\paragraph{Solution VI.38.E18.}
The Segre morphism and its equations are defined over \(\mathbb Z\):
\[
Z_{ij}\longmapsto X_iY_j,
\qquad
Z_{ij}Z_{\ell r}-Z_{ir}Z_{\ell j}.
\]
For an arbitrary base scheme \(Y\), the relative construction is obtained by base
change from \(\Spec\mathbb Z\). Since closed immersions are preserved by base change,
we obtain
\[
\boxed{\mathbb P^m_Y\times_Y\mathbb P^n_Y
\hookrightarrow\mathbb P^{(m+1)(n+1)-1}_Y}.
\]
On every affine open of \(Y\), the same determinantal equations define the image,
showing explicitly that the equations commute with base change.

% VI38-DETAILED-EXERCISE-19
\paragraph{Solution VI.38.E19.}
Suppose
\[
X\hookrightarrow\mathbb P^m_Y,
\qquad
Y\hookrightarrow\mathbb P^n_Z
\]
exhibit \(X\to Y\) and \(Y\to Z\) as projective.  Since
\[
\mathbb P^m_Y\cong\mathbb P^m_Z\times_ZY,
\]
base change of the second closed immersion gives
\[
\mathbb P^m_Y\hookrightarrow
\mathbb P^m_Z\times_Z\mathbb P^n_Z.
\]
Then relative Segre gives
\[
\mathbb P^m_Z\times_Z\mathbb P^n_Z
\hookrightarrow\mathbb P^N_Z,
\qquad
N=(m+1)(n+1)-1.
\]
Thus the required chain is
\[
\boxed{X\hookrightarrow\mathbb P^m_Y
\hookrightarrow\mathbb P^m_Z\times_Z\mathbb P^n_Z
\hookrightarrow\mathbb P^N_Z}.
\]

% VI38-DETAILED-EXERCISE-20
\paragraph{Solution VI.38.E20.}
Relative projective zero-space is canonically
\[
\mathbb P^0_Y\cong Y.
\]
If \(i:X\hookrightarrow Y\) is a closed immersion, identify its target with
\(\mathbb P^0_Y\). Then
\[
X\hookrightarrow\mathbb P^0_Y\longrightarrow Y
\]
is exactly a projective factorization. Thus every closed immersion is a projective
morphism.

% VI38-DETAILED-EXERCISE-21
\paragraph{Solution VI.38.E21.}
Let \(\nu_d:\mathbb P^n\hookrightarrow\mathbb P^N\) be the \(d\)-uple Veronese
embedding.  On a standard chart \(D_+(x_i)\), use the Veronese coordinate
\(x_i^d\).  If \(x_j\neq0\) as well, the transition from the trivialization based
on \(x_i^d\) to that based on \(x_j^d\) is
\[
\frac{x_j^d}{x_i^d}=\left(\frac{x_j}{x_i}\right)^d.
\]
The transition function for \(\OO(1)\) on \(\mathbb P^n\) is \(x_j/x_i\); its
\(d\)-th tensor power therefore has the displayed transition law. Hence
\[
\boxed{\nu_d^*\OO_{\mathbb P^N}(1)\cong\OO_{\mathbb P^n}(d)}.
\]

% VI38-DETAILED-EXERCISE-22
\paragraph{Solution VI.38.E22.}
An element \(A\in\mathrm{GL}_{n+1}(k)\) sends the vector space spanned by
\(x_0,\ldots,x_n\) to itself by an invertible linear map. By the universal property
of the symmetric algebra, this extends uniquely to a graded automorphism of
\[
k[x_0,\ldots,x_n].
\]
Contravariance of Proj gives an automorphism of \(\mathbb P^n_k\). Scalar matrices
multiply all homogeneous coordinates by the same nonzero scalar and hence act
trivially on projective space. Therefore the action factors through
\[
\boxed{\mathrm{PGL}_{n+1}(k)=\mathrm{GL}_{n+1}(k)/k^\times}.
\]

% VI38-DETAILED-EXERCISE-23
\paragraph{Solution VI.38.E23.}
Take
\[
\Phi:\mathbb P^2\dashrightarrow\mathbb P^1,
\qquad
[x_0:x_1:x_2]\longmapsto[x_0x_1:x_1x_2].
\]
Both coordinates have degree two. Their common vanishing locus is
\[
V_+(x_0x_1,x_1x_2)
=V_+(x_1)\cup V_+(x_0,x_2).
\]
Thus the maximal domain is
\[
U=\mathbb P^2\setminus\bigl(V_+(x_1)\cup V_+(x_0,x_2)\bigr).
\]
On \(U\), necessarily \(x_1\neq0\), so
\[
[x_0x_1:x_1x_2]=[x_0:x_2].
\]
The original homogeneous formula is therefore a regular morphism on \(U\) but not
a global morphism because its base locus is nonempty.

% VI38-DETAILED-EXERCISE-24
\paragraph{Solution VI.38.E24.}
A closed immersion requires more than an injective map of underlying rational
points.  Affine-locally it is equivalent to a surjective ring map
\[
A\twoheadrightarrow B,
\]
so it remembers the complete scheme structure, including nilpotents and tangent
directions.

For example, \(\Spec k\) and \(\Spec k[\varepsilon]/(\varepsilon^2)\) each have one
\(k\)-point, but the latter has nontrivial nilpotent structure. A pointwise test
cannot detect whether the corresponding local ring map is surjective with the
correct kernel. Thus injectivity on \(k\)-points cannot establish a closed
immersion; one must verify the scheme-theoretic affine/local criterion.
