\begin{example}[A graded map defined only on one chart]\label{ex:vi38-01}
Let \(S=k[x,y]\), \(T=k[u,v]\), and \(x\mapsto u,\ y\mapsto0\).  The point \((u)\in\Proj T\) contracts to \((x,y)=S_+\), so no image point exists there.  The induced map is defined on \(D_+(u)\), not on all of \(\mathbb P^1\).
\end{example}

\begin{example}[A homogeneous quotient]\label{ex:vi38-02}
The map \(k[X,Y,Z]\to k[s,t]\), \(X\mapsto s^2\), \(Y\mapsto st\), \(Z\mapsto t^2\), has image exactly the second Veronese subring \(k[s,t]^{(2)}\).  Viewed as \(k[X,Y,Z]\twoheadrightarrow k[s,t]^{(2)}\), it is a surjective degree-preserving map and gives the conic embedding.
\end{example}

\begin{example}[Linear change of coordinates]\label{ex:vi38-03}
A matrix in \(\mathrm{GL}_{n+1}(k)\) acts on the variables and extends to a graded automorphism of the polynomial ring, hence induces an automorphism of \(\mathbb P^n\).
\end{example}

\begin{example}[A morphism from two quadratic forms]\label{ex:vi38-04}
The forms \(s^2,t^2\) have no common zero on \(\mathbb P^1\), so
\([s:t]\mapsto[s^2:t^2]\) defines a morphism \(\mathbb P^1\to\mathbb P^1\).
\end{example}

\begin{example}[A base point]\label{ex:vi38-05}
The forms \(x_0x_1,x_1^2\) vanish simultaneously at \([1:0]\), so the formula \([x_0:x_1]\mapsto[x_0x_1:x_1^2]\) is undefined there.  On \(D_+(x_1)\) it reduces to \([x_0:x_1]\mapsto[x_0:x_1]\).
\end{example}

\begin{example}[A hyperplane closed immersion]\label{ex:vi38-06}
The quotient \(k[x_0,\ldots,x_n]\twoheadrightarrow k[x_0,\ldots,x_n]/(x_n)\) gives \(\mathbb P^{n-1}\hookrightarrow\mathbb P^n\) as the hyperplane \(x_n=0\).
\end{example}

\begin{example}[A nonreduced closed immersion]\label{ex:vi38-07}
The quotient by \((x_n^2)\) gives a double hyperplane embedded as a closed subscheme of \(\mathbb P^n\).  Closed immersion does not imply reduced image.
\end{example}

\begin{example}[Projective space over a base]\label{ex:vi38-08}
For any scheme \(Y\), the projection \(\mathbb P^n_Y\to Y\) is projective.  Its fiber over \(y\in Y\) is \(\mathbb P^n_{\kappa(y)}\).
\end{example}

\begin{example}[Base change of a projective family]\label{ex:vi38-09}
If \(X\hookrightarrow\mathbb P^n_Y\) is cut out by homogeneous equations with coefficients on \(Y\), then after \(Y'\to Y\) the same equations with base-changed coefficients cut out \(X\times_Y Y'\subseteq\mathbb P^n_{Y'}\).
\end{example}

\begin{example}[The quadratic Veronese conic]\label{ex:vi38-10}
The map \([s:t]\mapsto[s^2:st:t^2]\) is a closed immersion of \(\mathbb P^1\) into \(\mathbb P^2\), with image \(Y_0Y_2-Y_1^2=0\).
\end{example}

\begin{example}[The cubic rational normal curve]\label{ex:vi38-11}
The third Veronese embedding sends \([s:t]\) to \([s^3:s^2t:st^2:t^3]\in\mathbb P^3\).  Its ideal contains the \(2\times2\) minors of \(\begin{pmatrix}Y_0&Y_1&Y_2\\Y_1&Y_2&Y_3\end{pmatrix}\).
\end{example}

\begin{example}[The quadric surface]\label{ex:vi38-12}
The Segre map \(\mathbb P^1\times\mathbb P^1\to\mathbb P^3\) has image \(Z_{00}Z_{11}-Z_{01}Z_{10}=0\), identifying the product with a smooth projective quadric surface.
\end{example}

\begin{example}[\(\mathbb P^1\times\mathbb P^2\)]\label{ex:vi38-13}
The Segre image in \(\mathbb P^5\) is defined by the \(2\times2\) minors of a \(2\times3\) matrix \((Z_{ij})\).
\end{example}

\begin{example}[A closed point]\label{ex:vi38-14}
A \(k\)-rational point \([1:0:\cdots:0]\) is the closed subscheme cut out by \((x_1,\ldots,x_n)\), hence its inclusion into \(\mathbb P^n\) is a closed immersion.
\end{example}

\begin{example}[Composition via Segre]\label{ex:vi38-15}
If \(X\subseteq\mathbb P^m_Y\) and \(Y\subseteq\mathbb P^n_Z\), then \(X\) embeds into \(\mathbb P^m_Z\times_Z\mathbb P^n_Z\) and hence, by relative Segre, into one projective space over \(Z\).
\end{example}

\begin{example}[A very ample twist]\label{ex:vi38-16}
On \(\mathbb P^n_k\), \(\OO(1)\) is very ample by the identity embedding.  The line bundle \(\OO(d)\) for \(d\ge1\) is very ample because the \(d\)-uple Veronese embedding pulls back \(\OO_{\mathbb P^N}(1)\) to \(\OO_{\mathbb P^n}(d)\).
\end{example}
