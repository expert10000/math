% VI38-DETAILED-PROBLEM-05
\begin{problem}[VI.38.P5: Relative projectivity survives base change]\label{prob:vi38-05}
Let \(X\to Y\) be projective and \(Y'\to Y\) arbitrary.  Prove directly from a chosen projective embedding that \(X_{Y'}\to Y'\) is projective.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
By projectivity there is an integer \(n\) and a closed immersion over \(Y\),
\[
i:X\hookrightarrow\mathbb P^n_Y,
\]
such that the composite with \(\mathbb P^n_Y\to Y\) is the given map \(X\to Y\).
Base change along \(g:Y'\to Y\) gives the Cartesian diagram
\[
\begin{array}{ccc}
X_{Y'} & \xrightarrow{\ i'\ } & \mathbb P^n_Y\times_Y Y'\\
\downarrow && \downarrow\\
X & \xrightarrow{\ i\ } & \mathbb P^n_Y
\end{array}
\]
Closed immersions are stable under arbitrary base change, so \(i'\) is closed.
Relative projective space also commutes with base change:
\[
\mathbb P^n_Y\times_Y Y'
\cong
\Proj_{Y'}\OO_{Y'}[x_0,\ldots,x_n]
=
\mathbb P^n_{Y'}.
\]
Consequently we have a factorization
\[
\boxed{X_{Y'}\hookrightarrow\mathbb P^n_{Y'}\longrightarrow Y'}
\]
with first arrow a closed immersion.  This is exactly the definition that
\(X_{Y'}\to Y'\) is projective.

Notice that neither flatness nor finite type of \(Y'\to Y\) is required: the proof
uses only base-change stability of closed immersions and the universal relative
construction of projective space.
\end{solution}
\fi
