% VI38-DETAILED-PROBLEM-08
\begin{problem}[VI.38.P8: Proj of a Segre product]\label{prob:vi38-08}
Prove \(\Proj(A\#B)\cong\Proj A\times\Proj B\) for polynomial rings \(A=k[x_0,\ldots,x_m]\), \(B=k[y_0,\ldots,y_n]\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Define the Segre product
\[
A\#B=\bigoplus_{d\ge0}A_d\otimes_kB_d.
\]
It is generated in degree one by
\[
z_{ij}=x_i\otimes y_j,
\qquad 0\le i\le m,
\quad 0\le j\le n.
\]
Hence the standard opens \(D_+(z_{ij})\) cover \(\Proj(A\#B)\).

Fix \(i,j\).  In the degree-zero localization at \(z_{ij}\), we have
\[
\frac{x_a\otimes y_j}{x_i\otimes y_j}=\frac{x_a}{x_i},
\qquad
\frac{x_i\otimes y_b}{x_i\otimes y_j}=\frac{y_b}{y_j}.
\]
More formally, the assignment
\[
A_{(x_i)}\otimes_kB_{(y_j)}
\longrightarrow
(A\#B)_{(z_{ij})}
\]
sends
\[
\frac{a}{x_i^r}\otimes\frac{b}{y_j^s}
\]
to a common-denominator degree-zero fraction after multiplying numerator and
denominator so the two degrees agree.  Conversely every degree-zero fraction in
the Segre localization separates into a product of an \(x\)-ratio and a
\(y\)-ratio. Thus
\[
\boxed{(A\#B)_{(z_{ij})}\cong A_{(x_i)}\otimes_kB_{(y_j)}}.
\]
Therefore
\[
D_+(z_{ij})
\cong
D_+(x_i)\times D_+(y_j).
\]

On an overlap, say after also inverting \(z_{\ell r}\), both identifications are
obtained by the same further localization. Hence the chart isomorphisms commute
with the transition maps.  They glue to a global isomorphism
\[
\boxed{\Proj(A\#B)\xrightarrow{\sim}\Proj A\times_k\Proj B}.
\]
Since \(\Proj A=\mathbb P^m\) and \(\Proj B=\mathbb P^n\), this realizes the product
through a single standard graded coordinate ring.
\end{solution}
\fi
