% VI38-DETAILED-PROBLEM-01
\begin{problem}[VI.38.P1: Domain theorem for Proj]\label{prob:vi38-01}
Let \(\varphi:S\to T\) be degree preserving.  Prove carefully that the standard-open maps \(\Spec T_{(\varphi(f))}\to\Spec S_{(f)}\) glue on \(U_\varphi\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Set
\[
U_\varphi
=
\{\mathfrak q\in\Proj T:\varphi^{-1}(\mathfrak q)\not\supseteq S_+\}
=
\bigcup_{f\in S_+}D_+(\varphi(f)).
\]
For a homogeneous \(f\in S_+\), localization of \(\varphi\) gives a graded map
\[
S_f\longrightarrow T_{\varphi(f)},
\qquad
\frac{s}{f^r}\longmapsto\frac{\varphi(s)}{\varphi(f)^r}.
\]
Taking degree-zero parts gives
\[
\varphi_{(f)}:S_{(f)}\longrightarrow T_{(\varphi(f))}.
\]
Contravariance of \(\Spec\) therefore produces the affine morphism
\[
\psi_f:
D_+(\varphi(f))=\Spec T_{(\varphi(f))}
\longrightarrow
D_+(f)=\Spec S_{(f)}.
\]

We must prove compatibility on overlaps.  Let \(f,g\in S_+\) be homogeneous.  The
overlap of the two source charts is
\[
D_+(\varphi(f))\cap D_+(\varphi(g))
=D_+(\varphi(fg)).
\]
Localization in stages identifies
\[
\bigl(T_{(\varphi(f))}\bigr)_{(\varphi(g)/1)}
\cong T_{(\varphi(fg))}
\cong
\bigl(T_{(\varphi(g))}\bigr)_{(\varphi(f)/1)},
\]
and similarly on \(S\).  Under these identifications, the restrictions of both
\(\varphi_{(f)}\) and \(\varphi_{(g)}\) are the degree-zero part of the single
localized map
\[
S_{fg}\longrightarrow T_{\varphi(fg)}.
\]
Hence
\[
\psi_f|_{D_+(\varphi(fg))}
=
\psi_g|_{D_+(\varphi(fg))}.
\]
The affine gluing theorem therefore gives a unique morphism
\[
\boxed{\psi_\varphi:U_\varphi\longrightarrow\Proj S}
\]
whose restriction to each standard open is \(\psi_f\).

Finally, on points the affine maps are contraction of primes.  If
\(\mathfrak q\in D_+(\varphi(f))\), the image prime on this chart is induced by
\(\varphi^{-1}(\mathfrak q)\), and the fact that \(f\notin\varphi^{-1}(\mathfrak q)\)
shows that the contraction defines a point of \(D_+(f)\).  Thus the glued point map
is precisely the expected contraction map on its maximal projective domain.
\end{solution}
\fi
