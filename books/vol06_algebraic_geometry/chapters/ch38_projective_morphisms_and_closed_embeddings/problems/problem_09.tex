% VI38-DETAILED-PROBLEM-09
\begin{problem}[VI.38.P9: Segre ideal]\label{prob:vi38-09}
Prove that the kernel of \(k[z_{ij}]\to A\#B\), \(z_{ij}\mapsto x_i\otimes y_j\), is generated by the \(2\times2\) minors.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
R=k[z_{ij}:0\le i\le m,\ 0\le j\le n]
\]
and
\[
\theta:R\longrightarrow A\#B,
\qquad z_{ij}\longmapsto x_i\otimes y_j.
\]
The map is surjective because the Segre product is generated by its degree-one
elements \(x_i\otimes y_j\).

Let \(J\) be the ideal generated by the \(2\times2\) minors
\[
z_{ij}z_{\ell r}-z_{ir}z_{\ell j}.
\]
Each minor maps to
\[
(x_ix_\ell)\otimes(y_jy_r)-(x_ix_\ell)\otimes(y_ry_j)=0,
\]
so \(J\subseteq\ker\theta\).

For the reverse inclusion, consider a degree-\(d\) monomial
\[
M=z_{i_1j_1}\cdots z_{i_dj_d}.
\]
Its image is
\[
\theta(M)
=(x_{i_1}\cdots x_{i_d})\otimes(y_{j_1}\cdots y_{j_d}),
\]
which is determined exactly by the multiset of row indices
\(\{i_1,\ldots,i_d\}\) and the multiset of column indices
\(\{j_1,\ldots,j_d\}\).

Modulo \(J\), the minor relation gives the elementary swap
\[
z_{ij}z_{\ell r}\equiv z_{ir}z_{\ell j}\pmod J.
\]
By repeated swaps, the column indices can be permuted arbitrarily among the fixed
row factors. Therefore any two monomials having the same row multiset and column
multiset are congruent modulo \(J\).

Now let \(F\in\ker\theta\). Group its monomials according to the monomial of
\(A\#B\) to which they map. Since distinct monomials in the polynomial rings
\(A\) and \(B\) are linearly independent, in every group the sum of coefficients
must be zero. Fixing one monomial \(M_0\) in such a group, the whole group is a
linear combination of differences \(M-M_0\). Each difference belongs to \(J\) by
the swap argument. Hence every group lies in \(J\), and so does \(F\). Therefore
\[
\boxed{\ker\theta=J},
\]
proving scheme-theoretically that the Segre image is cut out by all
\(2\times2\) minors.
\end{solution}
\fi
