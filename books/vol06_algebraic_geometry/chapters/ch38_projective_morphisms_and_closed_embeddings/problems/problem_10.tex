% VI38-DETAILED-PROBLEM-10
\begin{problem}[VI.38.P10: Segre quadric]\label{prob:vi38-10}
Identify \(\mathbb P^1\times\mathbb P^1\) with a closed quadric in \(\mathbb P^3\) and compute its four standard product charts.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The Segre morphism is
\[
\sigma:\mathbb P^1\times\mathbb P^1\longrightarrow\mathbb P^3,
\qquad
([s:t],[u:v])\longmapsto[su:sv:tu:tv].
\]
Write the target coordinates as
\[
\begin{pmatrix}
Z_{00}&Z_{01}\\
Z_{10}&Z_{11}
\end{pmatrix}.
\]
By the Segre ideal theorem, in the \(2\times2\) case the kernel has the single
generator
\[
Q=Z_{00}Z_{11}-Z_{01}Z_{10}.
\]
Hence
\[
\boxed{\mathbb P^1\times\mathbb P^1
\cong V_+(Q)\subseteq\mathbb P^3}.
\]

Now compute the four standard product charts.

On \(D_+(Z_{00})\), corresponding to \(s\neq0\), \(u\neq0\), put
\[
a=\frac{Z_{10}}{Z_{00}}=\frac ts,
\qquad
b=\frac{Z_{01}}{Z_{00}}=\frac vu.
\]
The quadric relation gives
\[
\frac{Z_{11}}{Z_{00}}=ab,
\]
so the chart ring is \(k[a,b]\).

On \(D_+(Z_{01})\), with \(s\neq0\), \(v\neq0\), take
\[
a=\frac{Z_{11}}{Z_{01}}=\frac ts,
\qquad
b=\frac{Z_{00}}{Z_{01}}=\frac uv,
\]
and again the remaining ratio is their product. Thus the ring is \(k[a,b]\).
Similarly,
\[
D_+(Z_{10})\cong\Spec k[s/t,v/u]\cong\mathbb A^2,
\]
and
\[
D_+(Z_{11})\cong\Spec k[s/t,u/v]\cong\mathbb A^2.
\]
Equivalently the four charts are
\[
D_+(s)\times D_+(u),\quad
D_+(s)\times D_+(v),\quad
D_+(t)\times D_+(u),\quad
D_+(t)\times D_+(v).
\]
They cover the product and the quadric, and their transition maps are exactly the
usual inversions in the two \(\mathbb P^1\) coordinates.
\end{solution}
\fi
