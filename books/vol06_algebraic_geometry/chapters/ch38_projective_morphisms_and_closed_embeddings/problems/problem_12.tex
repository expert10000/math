% VI38-DETAILED-PROBLEM-12
\begin{problem}[VI.38.P12: Very ample twists on projective space]\label{prob:vi38-12}
Show that \(\OO_{\mathbb P^n}(d)\) is very ample for every \(d\ge1\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Recall the geometric criterion: an invertible sheaf \(L\) on a projective scheme is
very ample if there is a closed immersion
\[
i:X\hookrightarrow\mathbb P^N
\]
such that
\[
L\cong i^*\OO_{\mathbb P^N}(1).
\]
For \(d\ge1\), use the \(d\)-uple Veronese closed immersion
\[
\nu_d:\mathbb P^n\hookrightarrow\mathbb P^N,
\qquad
N=\binom{n+d}{d}-1.
\]
We show
\[
\nu_d^*\OO_{\mathbb P^N}(1)\cong\OO_{\mathbb P^n}(d).
\]

On \(U_i=D_+(x_i)\subseteq\mathbb P^n\), choose the Veronese coordinate
\(x_i^d\), which is nonzero on \(U_i\). This trivializes the pullback of
\(\OO_{\mathbb P^N}(1)\). On \(U_i\cap U_j\), the transition factor between the
trivializations based on \(x_i^d\) and \(x_j^d\) is
\[
\frac{x_j^d}{x_i^d}
=
\left(\frac{x_j}{x_i}\right)^d.
\]
But \(x_j/x_i\) is the standard transition factor for \(\OO_{\mathbb P^n}(1)\), so
its \(d\)-th power is precisely the transition factor for
\(\OO_{\mathbb P^n}(d)=\OO(1)^{\otimes d}\). Hence
\[
\boxed{\nu_d^*\OO_{\mathbb P^N}(1)\cong\OO_{\mathbb P^n}(d)}.
\]
Since \(\nu_d\) is a closed immersion, this proves that
\[
\boxed{\OO_{\mathbb P^n}(d)\text{ is very ample for every }d\ge1.}
\]
The argument also identifies the complete degree-\(d\) monomial coordinate system
that realizes the very ample embedding.
\end{solution}
\fi
