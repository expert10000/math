% VI38-DETAILED-PROBLEM-02
\begin{problem}[VI.38.P2: A map with a base point]\label{prob:vi38-02}
Analyze the map of graded rings \(k[a,b]\to k[x,y]^{(2)}\) given by \(a\mapsto x^2\), \(b\mapsto xy\).  Determine the induced projective domain and the pointwise formula.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
S=k[a,b],\qquad T=k[x,y]^{(2)}.
\]
The irrelevant ideal of \(S\) is \(S_+=(a,b)\), whose image generates
\[
(x^2,xy)=x(x,y)
\]
in the Veronese ring.  The induced Proj map is defined at a homogeneous prime
\(\mathfrak q\in\Proj T\) exactly when the images \(x^2,xy\) are not both contained
in \(\mathfrak q\).  Under the canonical isomorphism
\[
\Proj k[x,y]^{(2)}\cong\Proj k[x,y]=\mathbb P^1_k,
\]
the common projective zero locus is
\[
V_+(x^2,xy)=V_+(x)=\{[0:1]\}.
\]
Therefore
\[
\boxed{U_\varphi=\mathbb P^1_k\setminus\{[0:1]\}=D_+(x)}.
\]

On this domain \(x\neq0\), so the homogeneous coordinate formula is
\[
[x:y]\longmapsto[x^2:xy]=[x:y].
\]
This can also be checked on affine coordinate rings.  On \(D_+(a)\subseteq\Proj S\),
with coordinate \(b/a\), the corresponding source chart is \(D_+(x^2)=D_+(x)\), and
\[
\frac ba\longmapsto\frac{xy}{x^2}=\frac yx.
\]
Thus the induced map is the identity in the usual affine coordinate \(y/x\).

At \([0:1]\), both proposed projective coordinates vanish. Algebraically the
corresponding projective prime contains \(x^2\) and \(xy\), so its contraction
contains \((a,b)=S_+\) and therefore is not a point of \(\Proj S\).  Hence the
failure at \([0:1]\) is exactly the base-point/irrelevant-ideal obstruction.
\end{solution}
\fi
