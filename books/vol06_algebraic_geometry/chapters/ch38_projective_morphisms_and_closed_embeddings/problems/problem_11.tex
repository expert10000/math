% VI38-DETAILED-PROBLEM-11
\begin{problem}[VI.38.P11: Composition of projective morphisms]\label{prob:vi38-11}
Prove projective morphisms are closed under composition, explicitly identifying the relative projective ambient dimension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
f:X\to Y,
\qquad
g:Y\to Z
\]
be projective. Choose projective factorizations
\[
X\hookrightarrow\mathbb P^m_Y\to Y,
\qquad
Y\hookrightarrow\mathbb P^n_Z\to Z.
\]
Since relative projective space commutes with base change,
\[
\mathbb P^m_Y
\cong
\mathbb P^m_Z\times_ZY.
\]
Base-changing the closed immersion \(j\) along \(\mathbb P^m_Z\to Z\) produces a
closed immersion
\[
\mathbb P^m_Z\times_ZY
\hookrightarrow
\mathbb P^m_Z\times_Z\mathbb P^n_Z.
\]
Consequently
\[
X\hookrightarrow
\mathbb P^m_Z\times_Z\mathbb P^n_Z
\]
is a closed immersion.

The relative Segre morphism is a closed immersion
\[
\sigma:
\mathbb P^m_Z\times_Z\mathbb P^n_Z
\hookrightarrow
\mathbb P^N_Z,
\qquad
N=(m+1)(n+1)-1.
\]
Indeed it is obtained by base change from the universal Segre embedding over
\(\mathbb Z\), or equivalently is defined locally over \(Z\) by the universal
\(2\times2\) minors.

Composing all closed immersions gives
\[
\boxed{X\hookrightarrow\mathbb P^{(m+1)(n+1)-1}_Z}.
\]
Its composite with the structural projection to \(Z\) is exactly \(g\circ f\).
Therefore \(g\circ f\) is projective, with the displayed relative projective space
as an explicit ambient space.
\end{solution}
\fi
