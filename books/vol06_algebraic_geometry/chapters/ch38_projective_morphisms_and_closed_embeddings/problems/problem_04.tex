% VI38-DETAILED-PROBLEM-04
\begin{problem}[VI.38.P4: Morphism from same-degree forms]\label{prob:vi38-04}
Let \(F_0,\ldots,F_m\in k[x_0,\ldots,x_n]_d\).  Prove that if their projective common zero locus is empty, they define a morphism \(\mathbb P^n\to\mathbb P^m\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Put
\[
S=k[y_0,\ldots,y_m],
\qquad
T=k[x_0,\ldots,x_n],
\]
and regrade by the \(d\)-th Veronese
\[
T^{(d)}=\bigoplus_{r\ge0}T_{rd}.
\]
Then
\[
\varphi:S\longrightarrow T^{(d)},
\qquad y_i\longmapsto F_i
\]
is degree preserving.  Its projective domain is
\[
U_\varphi
=
\Proj T^{(d)}\setminus V_+(F_0,\ldots,F_m).
\]
By hypothesis the projective common zero locus is empty, so
\[
U_\varphi=\Proj T^{(d)}.
\]
The canonical Veronese isomorphism
\[
\Proj T\xrightarrow{\sim}\Proj T^{(d)}
\]
therefore turns the induced Proj morphism into a global morphism
\[
F:\mathbb P^n\longrightarrow\mathbb P^m.
\]

To identify it with the usual coordinate formula, work over
\(D_+(y_i)\subseteq\mathbb P^m\).  Its inverse image is \(D_+(F_i)\).  The target
affine coordinate \(y_j/y_i\) maps to
\[
\frac{F_j}{F_i},
\]
which is regular on \(D_+(F_i)\) because numerator and denominator have the same
degree. Thus on points
\[
\boxed{[x_0:\cdots:x_n]\longmapsto
[F_0(x):\cdots:F_m(x)]}.
\]
The sets \(D_+(F_i)\) cover \(\mathbb P^n\) precisely because the common zero locus
is empty, and the ratio formulas agree on overlaps.  Hence the homogeneous formula
is a genuine everywhere-defined morphism.
\end{solution}
\fi
