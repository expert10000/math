% VI38-DETAILED-PROBLEM-07
\begin{problem}[VI.38.P7: General Veronese embedding]\label{prob:vi38-07}
Give the complete ring-theoretic proof that the \(d\)-uple map \(\mathbb P^n\to\mathbb P^N\) is a closed immersion.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
S=k[x_0,\ldots,x_n]
\]
and let \(M_0,\ldots,M_N\) be all monomials of total degree \(d\). Their number is
\[
N+1=\binom{n+d}{d}.
\]
The \(d\)-th Veronese ring
\[
S^{(d)}=\bigoplus_{r\ge0}S_{rd}
\]
is regraded so that \(S_d\) has degree one. Every degree-\(rd\) monomial factors
into \(r\) degree-\(d\) monomials, so \(S^{(d)}\) is generated in degree one by the
\(M_\alpha\). Hence the graded homomorphism
\[
\theta:R=k[Z_0,\ldots,Z_N]\longrightarrow S^{(d)},
\qquad Z_\alpha\longmapsto M_\alpha
\]
is surjective.

Put \(I_d=\ker\theta\).  By the chartwise quotient criterion, the surjection induces
a closed immersion
\[
\Proj S^{(d)}\cong\Proj(R/I_d)\hookrightarrow\Proj R=\mathbb P^N.
\]
On the other hand, Veronese invariance gives a canonical isomorphism
\[
\eta_d:\Proj S\xrightarrow{\sim}\Proj S^{(d)}.
\]
The composite
\[
\Proj S\xrightarrow{\eta_d}\Proj S^{(d)}\hookrightarrow\mathbb P^N
\]
is therefore a closed immersion.

To identify its coordinates, on the chart where \(M_\alpha\neq0\) the affine map
sends the ratio \(Z_\beta/Z_\alpha\) to
\[
M_\beta/M_\alpha.
\]
Thus on homogeneous points the composite is exactly
\[
\boxed{\nu_d([x_0:\cdots:x_n])
=[M_0(x):\cdots:M_N(x)]}.
\]
Since at any projective point some \(x_i\neq0\), the monomial \(x_i^d\) is nonzero,
so these coordinates have no common projective zero.  We have therefore obtained
the global \(d\)-uple morphism, and the surjective graded presentation proves that
it is scheme-theoretically a closed immersion.  Its homogeneous image ideal is
precisely \(I_d\), the ideal of algebraic relations among degree-\(d\) monomials.
\end{solution}
\fi
