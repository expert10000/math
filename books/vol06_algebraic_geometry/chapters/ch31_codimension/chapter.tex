\chapter{Codimension}\label{ch:vi31-codimension}

VI/29 measured the absolute length of prime chains and VI/30 globalized dimension to schemes.  Codimension measures the
same chains \emph{relative to a chosen closed locus}.  For an irreducible closed subset $Y\subseteq X$, it records how
many strict irreducible specializations can occur between an ambient component and $Y$.  On an affine scheme this is
height, and at the generic point of $Y$ it is the dimension of the corresponding local ring.  For varieties over a
field, codimension and dimension fit the clean formula
\[
 \dim Y+\operatorname{codim}(Y,X)=\dim X.
\]
The hypotheses matter: AG8 Problem 3 asks for a general scheme where this additivity fails, and that exact counterexample
is preserved here.

\section*{Learning goals}
By the end of the chapter, the reader should be able to:
\begin{itemize}
\item compute codimension from chains of irreducible closed subsets;
\item translate codimension on an affine scheme into the height of a prime ideal;
\item identify codimension with the Krull dimension of the local ring at the generic point;
\item use the dimension--codimension formula correctly for varieties and recognize why it can fail for general schemes;
\item distinguish codimension from the number of equations or generators used to present a closed locus;
\item recognize codimension-one geometry as the numerical precursor of divisor theory.
\end{itemize}

\section*{Conceptual roadmap}
The chapter moves through one dictionary in three forms:
\[
\text{prime chains below }\mathfrak p
\quad\longleftrightarrow\quad
\text{irreducible closed chains above }V(\mathfrak p)
\quad\longleftrightarrow\quad
\dim A_{\mathfrak p}.
\]
The variety case then adds a fourth term, $\dim(A/\mathfrak p)$, producing the familiar decomposition of ambient dimension
into dimension along the subvariety plus codimension transverse to it.

\section{Relative chains}\label{sec:vi31-relative-chains}
\begin{definition}[Codimension of an irreducible closed subset]\label{def:vi31-codim-irreducible}
Let $Y\subseteq X$ be irreducible and closed. Define
\[
\operatorname{codim}(Y,X)=\sup\{n:Y=Z_n\subsetneq Z_{n-1}\subsetneq\cdots\subsetneq Z_0,
\ Z_i\text{ irreducible closed in }X\}.
\]
Thus codimension zero means that $Y$ is an irreducible component of $X$.
\end{definition}
For a closed subset $W\subseteq X$ with irreducible components $W_\alpha$, we use
\begin{definition}[Codimension of a closed subset]\label{def:vi31-codim-closed}
\[
 \operatorname{codim}(W,X)=\inf_\alpha\operatorname{codim}(W_\alpha,X).
\]
\end{definition}
This convention records the component of $W$ lying closest to an ambient component.
\input{chapters/ch31_codimension/figures/figure_01}

\section{Affine translation: codimension is height}\label{sec:vi31-affine-height}
Let $X=\Spec A$ and let $Y=V(\mathfrak p)$ for a prime $\mathfrak p$.  Reversing prime inclusions gives
\[
\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_n=\mathfrak p
\quad\Longleftrightarrow\quad
V(\mathfrak p)=V(\mathfrak p_n)\subsetneq\cdots\subsetneq V(\mathfrak p_0).
\]
Hence
\begin{proposition}[Affine codimension--height dictionary]\label{prop:vi31-height}
\[
 \operatorname{codim}\bigl(V(\mathfrak p),\Spec A\bigr)=\operatorname{ht}(\mathfrak p).
\]
\end{proposition}
This is the relative counterpart of $\dim\Spec A=\dim A$.
\input{chapters/ch31_codimension/figures/figure_02}

\begin{example}[A line in the plane]\label{ex:vi31-expand-01}
The closed irreducible subset $V(x)\subset\mathbb A^2_k$ has dimension $1$ inside a $2$-dimensional irreducible ambient space, so its codimension is $1$.
\end{example}

\section{The generic-point local ring}\label{sec:vi31-generic-local}
Every irreducible closed subset $Y$ of a scheme has a unique generic point $\eta_Y$.  Choose an affine neighborhood
$U=\Spec A$ of $\eta_Y$ and let $\mathfrak p$ be the corresponding prime.  Any generalization of $\eta_Y$ also lies in
$U$, so every chain relevant to the codimension can be read inside this one affine neighborhood.  Therefore
\begin{theorem}[Codimension at the generic point]\label{thm:vi31-generic-local}
For every irreducible closed $Y\subseteq X$,
\[
 \operatorname{codim}(Y,X)=\dim\OO_{X,\eta_Y}.
\]
\end{theorem}
Indeed $\OO_{X,\eta_Y}\cong A_{\mathfrak p}$ and $\dim A_{\mathfrak p}=\operatorname{ht}(\mathfrak p)$.
\input{chapters/ch31_codimension/figures/figure_03}

\section{The infimum formula}\label{sec:vi31-infimum}
If $y\in Y$, then $y$ is a specialization of $\eta_Y$.  In an affine neighborhood of $y$, the prime of
$\eta_Y$ is contained in the prime of $y$, so every chain below the former is also a chain below the latter.  Thus
\[
 \dim\OO_{X,y}\geq \dim\OO_{X,\eta_Y}.
\]
Since equality occurs at $y=\eta_Y$, we obtain the codimension statement singled out in AG8 Problem 2:
\begin{theorem}[Local-ring characterization of codimension]\label{thm:vi31-inf-local}
For irreducible closed $Y\subseteq X$,
\[
 \operatorname{codim}(Y,X)=\inf_{y\in Y}\dim\OO_{X,y}.
\]
For a general closed $W$, the same formula holds with the convention of \cref{def:vi31-codim-closed}.
\end{theorem}
VI/32 will study $\dim\OO_{X,x}$ as a local invariant in its own right; here it appears only as the exact bridge to
codimension.
\input{chapters/ch31_codimension/figures/figure_04}

\section{Varieties: dimension plus codimension}\label{sec:vi31-variety-additivity}
Let $X$ be a variety over $k$ and $Y\subseteq X$ an irreducible closed subvariety. Choose a nonempty affine open
$U=\Spec A$ meeting $Y$, with $Y\cap U=V(\mathfrak p)$.  The finitely generated $k$-domain $A$ satisfies the dimension
formula
\[
 \operatorname{ht}(\mathfrak p)+\dim(A/\mathfrak p)=\dim A.
\]
VI/30 identifies the dimensions of the nonempty affine opens with the dimensions of the varieties. Hence
\begin{theorem}[Dimension--codimension additivity for varieties]\label{thm:vi31-additivity}
If $X$ is a variety over $k$ and $Y\subseteq X$ is irreducible closed, then
\[
 \boxed{\dim Y+\operatorname{codim}(Y,X)=\dim X.}
\]
\end{theorem}
For a reducible closed subset $W$ of a variety, the convention using the smallest component codimension gives the same
identity $\dim W+\operatorname{codim}(W,X)=\dim X$.
\input{chapters/ch31_codimension/figures/figure_05}

\begin{example}[A point in the plane]\label{ex:vi31-expand-02}
A closed point $(a,b)$ in $\mathbb A^2_k$ has dimension $0$, hence codimension $2$.  Algebraically the corresponding maximal ideal sits at the end of a prime chain of length two.
\end{example}

\section{The AG8 counterexample: additivity can fail}\label{sec:vi31-ag8-counterexample}
Set
\[
 X=\Spec k\sqcup\mathbb A^1_k,
 \qquad Y=\Spec k\subseteq X.
\]
Then $Y$ is an irreducible component, so $\operatorname{codim}(Y,X)=0$.  Also $\dim Y=0$ while $\dim X=1$. Thus
\[
 \dim Y+\operatorname{codim}(Y,X)=0\neq1=\dim X.
\]
This is precisely the part-(c) counterexample requested by AG8 Problem 3.  It shows that the variety theorem is not a
formal identity for arbitrary schemes; non-equidimensional components already destroy it.
\input{chapters/ch31_codimension/figures/figure_06}

\section{Codimension one}\label{sec:vi31-codim-one}
Codimension-one irreducible closed subsets are especially important because they become prime divisors in the later
divisor chapters.  In $\mathbb A^n_k$, the coordinate hyperplane $V(x_1)$ has codimension one. More generally, if
$A$ is a Noetherian domain and $0\neq f\in A$ is a nonunit, every prime minimal over $(f)$ has height one by the
principal ideal theorem.  Thus every irreducible component of a genuine principal hypersurface has codimension one.
This chapter uses the fact only as a codimension computation; divisor theory remains reserved for the later divisor chapters.
\input{chapters/ch31_codimension/figures/figure_07}

\section{Nested subvarieties and chain additivity}\label{sec:vi31-nested}
For arbitrary schemes and irreducible closed subsets $Z\subseteq Y\subseteq X$, concatenating chains gives the general
inequality
\begin{proposition}[Concatenation inequality]\label{prop:vi31-concatenation}
\[
 \operatorname{codim}(Z,X)\geq
 \operatorname{codim}(Z,Y)+\operatorname{codim}(Y,X)
\]
whenever the displayed codimensions are finite.
\end{proposition}
Equality is not formal.  For varieties over a field, however, applying \cref{thm:vi31-additivity} three times yields
\begin{corollary}[Additivity along subvarieties]\label{cor:vi31-nested-varieties}
If $Z\subseteq Y\subseteq X$ are irreducible closed subvarieties of a variety, then
\[
 \operatorname{codim}(Z,X)=\operatorname{codim}(Z,Y)+\operatorname{codim}(Y,X).
\]
\end{corollary}
\input{chapters/ch31_codimension/figures/figure_08}

\begin{example}[A hypersurface]\label{ex:vi31-expand-03}
For a nonzero nonunit irreducible polynomial $f\in k[x_1,\ldots,x_n]$, the hypersurface $V(f)$ has codimension $1$ in affine $n$-space under the standard dimension theorem.
\end{example}

\section{Standard computations}\label{sec:vi31-computations}
For the coordinate linear subspace
\[
 L=V(x_1,\ldots,x_r)\subseteq\mathbb A^n_k
\]
we have $\dim L=n-r$ and
\[
 \operatorname{codim}(L,\mathbb A^n_k)=r.
\]
A closed point of $\mathbb A^n_k$ has codimension $n$.  The diagonal
$\Delta\subseteq\mathbb A^n_k\times_k\mathbb A^n_k$ has dimension $n$ inside ambient dimension $2n$, hence
codimension $n$.  For subvarieties $Y_i\subseteq X_i$ of varieties, the same calculation gives
\[
 \operatorname{codim}(Y_1\times_k Y_2,X_1\times_k X_2)
 =\operatorname{codim}(Y_1,X_1)+\operatorname{codim}(Y_2,X_2)
\]
whenever the indicated products are irreducible, so that the relative codimensions are being taken for irreducible
closed subsets.  In particular, this applies when the factors involved are geometrically integral over $k$.
\input{chapters/ch31_codimension/figures/figure_09}

\section{Boundaries of VI/31}\label{sec:vi31-boundaries}
VI/31 owns height as the affine model for codimension, the generic-point local-ring formula, the AG8 codimension and
additivity statements, and the AG8 Problem 3(c) failure.  It does not absorb the deeper local geometry of VI/32:
closed-point local dimension, tangent spaces, embedding dimension, regularity, and the remaining Problem 3(a)
counterexample stay there.  Nor does codimension one yet mean a divisor: Weil and Cartier divisor structures belong to
the later divisor arc.

\section{Common pitfalls}\label{sec:vi31-pitfalls}
\begin{itemize}
\item \textbf{Codimension is not the number of defining equations.}  Redundant equations can leave codimension unchanged,
and non-complete-intersection loci need not have codimension equal to a chosen number of generators.
\item \textbf{Codimension is relative to the ambient scheme.}  The same irreducible scheme can have different
codimensions in different embeddings.
\item \textbf{The formula $\dim Y+\operatorname{codim}(Y,X)=\dim X$ is not formal.}  It is valid in the variety setting used
here, but can fail immediately on non-equidimensional schemes.
\item \textbf{Codimension one is not yet a divisor.}  A codimension-one irreducible closed subset becomes a prime divisor
only after the hypotheses and structures of the divisor chapters are introduced.
\item \textbf{Integral extensions require care prime by prime.}  Going-up and incomparability control chains under an
integral map, but they do not by themselves say that every chosen prime upstairs has the same height as its contraction.
\end{itemize}

\section{Chapter synthesis}\label{sec:vi31-synthesis}
Codimension is the relative form of Krull dimension.  An irreducible closed subset $Y$ is measured by the height of its
generic prime, equivalently by $\dim\OO_{X,\eta_Y}$.  In varieties, finite-type dimension theory forces the clean
identity $\dim Y+\operatorname{codim}(Y,X)=\dim X$ and makes codimension additive along nested subvarieties.  Outside
that setting, component structure can break the identity immediately.  The distinction is essential preparation for
local dimension in VI/32 and for codimension-one divisor theory later in the volume.

\subsection*{Proofs}

\begin{exercise}[Height chain]\label{exr:vi31-expand-06}

\end{exercise}
\begin{hint}
Work directly from codimension and chains inside irreducible spaces.
\end{hint}
\begin{solution}
Translate prime chains into reverse chains of irreducible closed subsets from the ambient component down to the closure of the point.
\end{solution}

\begin{exercise}[Hyperplane codimension]\label{exr:vi31-expand-07}
Prove: Prove a coordinate hyperplane in affine $n$-space has codimension one.
\end{exercise}
\begin{hint}
Work directly from codimension and chains inside irreducible spaces.
\end{hint}
\begin{solution}
Its coordinate ring is a polynomial ring in $n-1$ variables.
\end{solution}

\begin{exercise}[Point codimension]\label{exr:vi31-expand-08}

Work directly from codimension and chains inside irreducible spaces.
\end{hint}
\begin{solution}
Use the chain $(0)\subset(x_1)\subset(x_1,x_2)\subset\cdots\subset(x_1,\ldots,x_n)$.
\end{solution}

\end{exercise}

Work directly from codimension and chains inside irreducible spaces.
\end{hint}

Isomorphisms preserve irreducible closed subsets, dimensions, and inclusion chains.
\end{solution}
