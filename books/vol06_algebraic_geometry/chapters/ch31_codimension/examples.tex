\begin{example}[A component has codimension zero]\label{ex:vi31-01}
If $Y$ is an irreducible component of $X$, no strictly larger irreducible closed subset lies between $Y$ and an ambient component. Hence $\operatorname{codim}(Y,X)=0$.
\end{example}

\begin{example}[A line in the affine plane]\label{ex:vi31-02}
For $Y=V(x)\subset\mathbb A^2_k$, the prime $(x)$ has height $1$, so $\operatorname{codim}(Y,\mathbb A^2_k)=1$.
\end{example}

\begin{example}[The origin in the affine plane]\label{ex:vi31-03}
The maximal ideal $(x,y)$ has height $2$ in $k[x,y]$. Thus the origin has codimension $2$ in $\mathbb A^2_k$.
\end{example}

\begin{example}[A coordinate plane in affine four-space]\label{ex:vi31-04}
$V(x_1,x_2)\subset\mathbb A^4_k$ has dimension $2$ and codimension $2$.
\end{example}

\begin{example}[A hypersurface]\label{ex:vi31-05}
If $f\in k[x_1,\ldots,x_n]$ is nonconstant and irreducible, then $V(f)$ has codimension $1$ in $\mathbb A^n_k$.
\end{example}

\begin{example}[The generic point]\label{ex:vi31-06}
For an integral scheme $X$, the closed irreducible subset $X$ itself has generic point $\eta$ and codimension $0$; correspondingly $\OO_{X,\eta}=K(X)$ has dimension $0$.
\end{example}

\begin{example}[A closed point on a curve]\label{ex:vi31-07}
Every closed point of an integral curve has codimension $1$.
\end{example}

\begin{example}[A closed point on a surface]\label{ex:vi31-08}
For a surface variety, a closed point has codimension $2$.
\end{example}

\begin{example}[The diagonal in the affine line squared]\label{ex:vi31-09}
The diagonal $V(x-y)\subset\mathbb A^1_k\times\mathbb A^1_k\cong\mathbb A^2_k$ is a hypersurface, hence has codimension $1$.
\end{example}

\begin{example}[The diagonal in affine n-space]\label{ex:vi31-10}
$\Delta\subset\mathbb A^n_k\times\mathbb A^n_k$ is cut out by $x_i-y_i$ for $1\le i\le n$ and has codimension $n$.
\end{example}

\begin{example}[A codimension-two curve in three-space]\label{ex:vi31-11}
The $z$-axis $V(x,y)\subset\mathbb A^3_k$ has codimension $2$.
\end{example}

\begin{example}[A union with codimension one]\label{ex:vi31-12}
In $\mathbb A^2_k$, $V(xy)=V(x)\cup V(y)$ has two irreducible components, each codimension $1$, so the closed subset has codimension $1$.
\end{example}

\begin{example}[A mixed closed subset]\label{ex:vi31-13}
In $\mathbb A^2_k$, $V(x)\cup\{(1,0)\}$ has codimension $1$: the line component has codimension $1$, while the isolated point has codimension $2$, and we take the infimum.
\end{example}

\begin{example}[Localization at a height-one prime]\label{ex:vi31-14}
For $A=k[x,y]$ and $\mathfrak p=(x)$, $A_{\mathfrak p}$ has dimension $1$, exactly the codimension of $V(x)$.
\end{example}

\begin{example}[Localization at the origin]\label{ex:vi31-15}
$k[x,y]_{(x,y)}$ has dimension $2$, equal to the codimension of the origin in $\mathbb A^2_k$.
\end{example}

\begin{example}[Nested coordinate spaces]\label{ex:vi31-16}
Inside $\mathbb A^3_k$, the chain $V(x,y)\subset V(x)\subset\mathbb A^3_k$ has codimensions $1+1=2$.
\end{example}

\begin{example}[The AG8 disconnected counterexample]\label{ex:vi31-17}
For $X=\Spec k\sqcup\mathbb A^1_k$ and $Y=\Spec k$, one has $\operatorname{codim}(Y,X)=0$, but $\dim Y+0=0<1=\dim X$.
\end{example}

\begin{example}[Finite field extension]\label{ex:vi31-18}
$\Spec L$ inside itself has codimension $0$ for every field extension $L/k$.
\end{example}

\begin{example}[Reduction preserves codimension]\label{ex:vi31-19}
A nilpotent thickening does not change the underlying irreducible closed subsets, so corresponding closed subsets have the same codimension before and after reduction.
\end{example}

\begin{example}[Affine three-space chain]\label{ex:vi31-20}
The chain $(0)\subsetneq(x)\subsetneq(x,y)\subsetneq(x,y,z)$ shows successive geometric codimensions $1,1,1$ and total codimension $3$ for the origin.
\end{example}
