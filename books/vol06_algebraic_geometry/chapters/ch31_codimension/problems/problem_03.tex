% VI31-DETAILED-PROBLEM-03
\begin{problem}[VI.31.P3: Codimension equals height on affine schemes]\label{prob:vi31-03}
Let \(A\) be a ring and \(\mathfrak p\in\Spec A\). Prove
\[
\operatorname{codim}(V(\mathfrak p),\Spec A)=\operatorname{ht}(\mathfrak p).
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
A prime chain ending at \(\mathfrak p\),
\[
\mathfrak p_0\subsetneq\mathfrak p_1\subsetneq\cdots\subsetneq\mathfrak p_n=\mathfrak p,
\]
corresponds under the affine closed-set dictionary to
\[
V(\mathfrak p)
=
V(\mathfrak p_n)
\subsetneq
V(\mathfrak p_{n-1})
\subsetneq\cdots\subsetneq
V(\mathfrak p_0).
\]
The order is reversed, but the number \(n\) of strict inclusions is unchanged. Conversely every chain of
irreducible closed subsets beginning at \(V(\mathfrak p)\) arises uniquely from such a prime chain. Therefore the
two sets of possible chain lengths coincide. Taking suprema gives
\[
\boxed{
\operatorname{codim}(V(\mathfrak p),\Spec A)
=
\operatorname{ht}(\mathfrak p).
}
\]
\end{solution}
\fi
