\begin{problem}[VI.31.P9: The diagonal]\label{prob:vi31-09}
Compute the codimension of the diagonal
\[
\Delta\subseteq\mathbb A^n_k\times_k\mathbb A^n_k.
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Identify the ambient product with
\[
\mathbb A^{2n}_k
=
\Spec k[x_1,\ldots,x_n,y_1,\ldots,y_n].
\]
The diagonal is cut out by
\[
x_1-y_1,\ldots,x_n-y_n.
\]
Its coordinate ring is obtained by setting \(y_i=x_i\):
\[
k[x_1,\ldots,x_n,y_1,\ldots,y_n]/(x_1-y_1,\ldots,x_n-y_n)
\cong
k[x_1,\ldots,x_n].
\]
Thus
\[
\dim\Delta=n,
\qquad
\dim(\mathbb A^n\times\mathbb A^n)=2n.
\]
The variety additivity theorem therefore gives
\[
\boxed{\operatorname{codim}(\Delta,\mathbb A^n\times\mathbb A^n)=n}.
\]
\end{solution}
\fi
