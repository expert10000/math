% VI31-DETAILED-PROBLEM-17
\begin{problem}[VI.31.P17: Additivity for nested subvarieties]\label{prob:vi31-17}
Let \(Z\subseteq Y\subseteq X\) be irreducible closed subvarieties of a variety. Prove exact codimension additivity.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The variety dimension--codimension theorem gives
\[
\operatorname{codim}(Z,X)=\dim X-\dim Z,
\]
\[
\operatorname{codim}(Y,X)=\dim X-\dim Y,
\]
and, since \(Y\) is itself a variety,
\[
\operatorname{codim}(Z,Y)=\dim Y-\dim Z.
\]
Therefore
\[
\begin{aligned}
\operatorname{codim}(Z,Y)+\operatorname{codim}(Y,X)
&=(\dim Y-\dim Z)+(\dim X-\dim Y)\\
&=\dim X-\dim Z.
\end{aligned}
\]
Hence
\[
\boxed{
\operatorname{codim}(Z,X)
=
\operatorname{codim}(Z,Y)+\operatorname{codim}(Y,X).
}
\]
Thus the general concatenation inequality becomes equality in the variety setting.
\end{solution}
\fi
