\begin{problem}[VI.31.P6: Closed points of affine space]\label{prob:vi31-06}
Show that every \(k\)-rational point of \(\mathbb A^n_k\) has codimension \(n\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
a=(a_1,\ldots,a_n)\in\mathbb A^n(k).
\]
Its maximal ideal is
\[
\mathfrak m_a=(x_1-a_1,\ldots,x_n-a_n).
\]
Translation \(x_i\mapsto x_i+a_i\) is an automorphism of the polynomial ring and carries \(\mathfrak m_a\) to
\[
(x_1,\ldots,x_n).
\]
The chain
\[
(0)\subsetneq(x_1)\subsetneq\cdots\subsetneq(x_1,\ldots,x_n)
\]
has length \(n\), so \(\operatorname{ht}(\mathfrak m_a)\ge n\). Since the whole polynomial ring has dimension
\(n\), equality follows. Therefore
\[
\boxed{
\operatorname{codim}(\{a\},\mathbb A^n_k)=n.
}
\]
\end{solution}
\fi
