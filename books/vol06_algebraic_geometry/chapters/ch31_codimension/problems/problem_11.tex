% VI31-DETAILED-PROBLEM-11
\begin{problem}[VI.31.P11: A reducible closed subset]\label{prob:vi31-11}
Let
\[
W=V(x)\cup V(y,z)\subseteq\mathbb A^3_k.
\]
Compute \(\dim W\) and \(\operatorname{codim}(W,\mathbb A^3_k)\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The irreducible components are
\[
W_1=V(x)
\qquad\text{and}\qquad
W_2=V(y,z).
\]
The first is a plane with coordinate ring \(k[y,z]\), so
\[
\dim W_1=2,
\qquad
\operatorname{codim}(W_1,\mathbb A^3)=1.
\]
The second is a line with coordinate ring \(k[x]\), so
\[
\dim W_2=1,
\qquad
\operatorname{codim}(W_2,\mathbb A^3)=2.
\]

The dimension of a finite union of irreducible components is the maximum component dimension:
\[
\boxed{\dim W=2}.
\]
By the chapter convention for a reducible closed subset,
\[
\operatorname{codim}(W,\mathbb A^3)
=
\inf\{1,2\}
=
\boxed{1}.
\]
Thus the component of largest dimension is exactly the one of smallest codimension.
\end{solution}
\fi
