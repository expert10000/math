\begin{problem}[VI.31.P11: A reducible closed subset]\label{prob:vi31-11}
Let $W=V(x)\cup V(y,z)\subseteq\mathbb A^3_k$. Compute $\dim W$ and $\operatorname{codim}(W,\mathbb A^3_k)$.
\end{problem}
\begin{solution}
The plane $V(x)$ has dimension two and codimension one; the line $V(y,z)$ has dimension one and codimension two. Thus $\dim W=2$ and, by the component-infimum convention, $\operatorname{codim}(W,\mathbb A^3_k)=1$.
\end{solution}
