% VI31-DETAILED-PROBLEM-01
\begin{problem}[VI.31.P1: AG8 Problem 2: codimension statements]\label{prob:vi31-01}
Let \(X\) be a variety over \(k\) and let \(Y\subseteq X\) be a closed irreducible subvariety. Prove
\[
\operatorname{codim}(Y,X)=\inf_{y\in Y}\dim\mathcal O_{X,y}
\quad\text{and}\quad
\dim Y+\operatorname{codim}(Y,X)=\dim X.
\]

\textbf{Provenance.} Legacy: AG8-P2b, the codimension portion of Problem 2. Global/open dimension is VI/30; the deeper local-dimension portion remains VI/32.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let \(\eta=\eta_Y\) be the generic point of \(Y\). Choose an affine open
\[
U=\Spec A\subseteq X
\]
meeting \(Y\). Since \(Y\) is irreducible, every nonempty open subset of \(Y\) contains \(\eta\); hence
\(\eta\in U\). Write
\[
Y\cap U=V(\mathfrak p)
\]
for the prime corresponding to \(\eta\).

The affine codimension--height dictionary gives
\[
\operatorname{codim}(Y,X)
=
\operatorname{ht}(\mathfrak p)
=
\dim A_{\mathfrak p}
=
\dim\mathcal O_{X,\eta}.
\]
For any \(y\in Y\), the point \(y\) is a specialization of \(\eta\). In an affine neighborhood of \(y\), the
corresponding primes satisfy
\[
\mathfrak p_\eta\subseteq\mathfrak p_y,
\]
so every chain below \(\mathfrak p_\eta\) is also a chain below \(\mathfrak p_y\). Therefore
\[
\dim\mathcal O_{X,y}\ge\dim\mathcal O_{X,\eta}.
\]
Equality occurs for \(y=\eta\), and consequently
\[
\boxed{
\operatorname{codim}(Y,X)
=
\inf_{y\in Y}\dim\mathcal O_{X,y}.
}
\]

For the additivity formula, the finite-type domain \(A\) satisfies
\[
\dim A
=
\operatorname{ht}(\mathfrak p)
+
\dim(A/\mathfrak p).
\]
By VI/30, the nonempty affine open \(U\subseteq X\) has dimension \(\dim X\), while
\(Y\cap U\) is a nonempty open of the variety \(Y\) and has dimension \(\dim Y\). Thus
\[
\dim X
=
\operatorname{codim}(Y,X)+\dim Y.
\]
Hence
\[
\boxed{\dim Y+\operatorname{codim}(Y,X)=\dim X}.
\]
\end{solution}
\fi
