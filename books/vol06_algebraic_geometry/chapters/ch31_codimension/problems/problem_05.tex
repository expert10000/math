\begin{problem}[VI.31.P5: Coordinate subspaces]\label{prob:vi31-05}
For $L=V(x_1,\ldots,x_r)\subseteq\mathbb A^n_k$, compute $\dim L$ and $\operatorname{codim}(L,\mathbb A^n_k)$.
\end{problem}
\begin{solution}
The quotient is $k[x_{r+1},\ldots,x_n]$, so $\dim L=n-r$. The prime $(x_1,\ldots,x_r)$ has height $r$, hence the codimension is $r$.
\end{solution}
