% VI31-DETAILED-PROBLEM-05
\begin{problem}[VI.31.P5: Coordinate subspaces]\label{prob:vi31-05}
For
\[
L=V(x_1,\ldots,x_r)\subseteq\mathbb A^n_k,
\]
compute \(\dim L\) and \(\operatorname{codim}(L,\mathbb A^n_k)\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The coordinate ring of \(L\) is
\[
k[x_1,\ldots,x_n]/(x_1,\ldots,x_r)
\cong
k[x_{r+1},\ldots,x_n].
\]
Therefore
\[
\boxed{\dim L=n-r}.
\]

The ideal
\[
\mathfrak p=(x_1,\ldots,x_r)
\]
is prime, and the chain
\[
(0)\subsetneq(x_1)\subsetneq(x_1,x_2)\subsetneq\cdots\subsetneq(x_1,\ldots,x_r)
\]
has length \(r\). The polynomial-ring dimension formula
\[
\dim k[x_1,\ldots,x_n]
=
\operatorname{ht}(\mathfrak p)+\dim(A/\mathfrak p)
\]
gives
\[
n=\operatorname{ht}(\mathfrak p)+(n-r),
\]
so
\[
\operatorname{ht}(\mathfrak p)=r.
\]
Thus
\[
\boxed{\operatorname{codim}(L,\mathbb A^n_k)=r}.
\]
\end{solution}
\fi
