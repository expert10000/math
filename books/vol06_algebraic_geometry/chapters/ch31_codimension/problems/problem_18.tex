% VI31-DETAILED-PROBLEM-18
\begin{problem}[VI.31.P18: Codimension-one center]\label{prob:vi31-18}
Let \(A\) be a Noetherian domain and \(f\neq0\) a nonunit. If \(\mathfrak p\) is minimal over \((f)\), show \(V(\mathfrak p)\) has codimension one in \(\Spec A\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Because \(\mathfrak p\) is minimal over the principal ideal \((f)\), the principal ideal theorem gives
\[
\operatorname{ht}(\mathfrak p)\le1.
\]
Since \(A\) is a domain,
\[
(0)
\]
is prime. The assumption \(f\ne0\) implies \(f\notin(0)\), while \(f\in\mathfrak p\), so
\[
(0)\subsetneq\mathfrak p.
\]
Therefore
\[
\operatorname{ht}(\mathfrak p)\ge1.
\]
Combining the two inequalities,
\[
\operatorname{ht}(\mathfrak p)=1.
\]
Using affine codimension equals height,
\[
\boxed{
\operatorname{codim}(V(\mathfrak p),\Spec A)=1.
}
\]
No divisor or valuation structure is needed for this computation.
\end{solution}
\fi
