% VI31-DETAILED-PROBLEM-04
\begin{problem}[VI.31.P4: Codimension from the generic stalk]\label{prob:vi31-04}
Let \(Y\subseteq X\) be irreducible closed with generic point \(\eta\). Prove
\[
\operatorname{codim}(Y,X)=\dim\mathcal O_{X,\eta}.
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Choose an affine neighborhood
\[
U=\Spec A
\]
of \(\eta\), and let \(\mathfrak p\in\Spec A\) correspond to \(\eta\). Because open subsets are stable under
generalization, every generalization of \(\eta\) lies in \(U\). A codimension chain beginning at \(Y\) corresponds
to a chain of generic points generalizing \(\eta\), so the entire chain is visible inside \(U\).

Within \(U\), the closed subset \(Y\cap U\) is
\[
V(\mathfrak p),
\]
and Problem~3 gives
\[
\operatorname{codim}(Y,X)
=
\operatorname{ht}(\mathfrak p).
\]
The local ring at \(\eta\) is
\[
\mathcal O_{X,\eta}\cong A_{\mathfrak p},
\]
and VI/29 gives
\[
\dim A_{\mathfrak p}=\operatorname{ht}(\mathfrak p).
\]
Hence
\[
\boxed{
\operatorname{codim}(Y,X)=\dim\mathcal O_{X,\eta}.
}
\]
\end{solution}
\fi
