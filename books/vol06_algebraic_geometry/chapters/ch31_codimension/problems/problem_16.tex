\begin{problem}[VI.31.P16: Concatenation inequality]\label{prob:vi31-16}
For irreducible closed \(Z\subseteq Y\subseteq X\) of finite codimension, prove the concatenation inequality for codimensions.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Set
\[
a=\operatorname{codim}(Z,Y),
\qquad
b=\operatorname{codim}(Y,X),
\]
and assume both are finite. Since codimensions are suprema of subsets of the nonnegative integers and are finite,
the suprema are maxima. Choose chains
\[
Z=Z_a\subsetneq Z_{a-1}\subsetneq\cdots\subsetneq Z_0=Y
\]
inside \(Y\), and
\[
Y=Y_b\subsetneq Y_{b-1}\subsetneq\cdots\subsetneq Y_0
\]
inside \(X\), of lengths \(a\) and \(b\).

Because \(Y\) is closed in \(X\), each \(Z_i\), closed in \(Y\), is also closed in \(X\). Concatenating at the
common term \(Y\) produces
\[
Z\subsetneq\cdots\subsetneq Y\subsetneq\cdots\subsetneq Y_0,
\]
a chain in \(X\) of length \(a+b\). Therefore
\[
\boxed{
\operatorname{codim}(Z,X)
\ge
\operatorname{codim}(Z,Y)+\operatorname{codim}(Y,X).
}
\]
Equality requires additional hypotheses and is not formal.
\end{solution}
\fi
