\begin{problem}[VI.31.P7: Irreducible hypersurface]\label{prob:vi31-07}
Let \(0\neq f\in k[x_1,\ldots,x_n]\) be irreducible and nonconstant. Compute the codimension of \(V(f)\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Because \(f\) is irreducible in the UFD \(k[x_1,\ldots,x_n]\), the ideal
\[
(f)
\]
is prime. Since \(f\ne0\), this prime is nonzero, while \((0)\) is prime because the polynomial ring is a domain.
Thus
\[
(0)\subsetneq(f),
\]
so
\[
\operatorname{ht}(f)\ge1.
\]
The principal ideal theorem gives
\[
\operatorname{ht}(f)\le1.
\]
Hence
\[
\operatorname{ht}(f)=1.
\]
By the affine codimension--height dictionary,
\[
\boxed{\operatorname{codim}(V(f),\mathbb A^n_k)=1}.
\]
\end{solution}
\fi
