\begin{problem}[VI.31.P10: Nested coordinate chain]\label{prob:vi31-10}
In \(\mathbb A^4_k\), let
\[
Z=V(x_1,x_2,x_3),\qquad Y=V(x_1).
\]
Verify codimension additivity through \(Z\subseteq Y\subseteq X\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The ambient variety
\[
X=\mathbb A^4_k
\]
has dimension \(4\). The hyperplane
\[
Y=V(x_1)
\]
has dimension \(3\), while
\[
Z=V(x_1,x_2,x_3)
\]
has dimension \(1\).

Hence
\[
\operatorname{codim}(Y,X)=4-3=1,
\]
and, viewing \(Z\) inside the three-dimensional affine space \(Y\),
\[
\operatorname{codim}(Z,Y)=3-1=2.
\]
Directly,
\[
\operatorname{codim}(Z,X)=4-1=3.
\]
Therefore
\[
\boxed{
\operatorname{codim}(Z,X)
=
\operatorname{codim}(Z,Y)+\operatorname{codim}(Y,X)
=
2+1.
}
\]
\end{solution}
\fi
