\begin{problem}[VI.31.P10: Nested coordinate chain]\label{prob:vi31-10}
In $\mathbb A^4_k$, let $Z=V(x_1,x_2,x_3)$ and $Y=V(x_1)$. Verify codimension additivity through $Z\subseteq Y\subseteq X$.
\end{problem}
\begin{solution}
The dimensions are $1,3,4$. Hence
\[
\operatorname{codim}(Z,Y)=2,\qquad
\operatorname{codim}(Y,X)=1,
\]
and therefore $\operatorname{codim}(Z,X)=3=2+1$.
\end{solution}
