\begin{problem}[VI.31.P14: Product of coordinate subspaces]\label{prob:vi31-14}
Let \(L_r\subset\mathbb A^m\) and \(M_s\subset\mathbb A^n\) be coordinate subspaces of codimensions \(r\) and \(s\). Compute the codimension of \(L_r\times M_s\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
After relabeling coordinates, write
\[
L_r=V(x_1,\ldots,x_r)\subseteq\mathbb A^m,
\]
and
\[
M_s=V(y_1,\ldots,y_s)\subseteq\mathbb A^n.
\]
Their product lies in
\[
\mathbb A^{m+n}
\]
and is defined by the \(r+s\) coordinate equations
\[
x_1,\ldots,x_r,y_1,\ldots,y_s.
\]
Its coordinate ring is
\[
k[x_{r+1},\ldots,x_m,y_{s+1},\ldots,y_n],
\]
of dimension
\[
(m-r)+(n-s)=m+n-(r+s).
\]
Since the ambient dimension is \(m+n\),
\[
\boxed{
\operatorname{codim}(L_r\times M_s,\mathbb A^{m+n})=r+s.
}
\]
\end{solution}
\fi
