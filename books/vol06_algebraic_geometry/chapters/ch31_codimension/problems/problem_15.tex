% VI31-DETAILED-PROBLEM-15
\begin{problem}[VI.31.P15: Infimum over a closed set]\label{prob:vi31-15}
Let \(W\) be a closed subset with finitely many irreducible components \(W_i\). Show
\[
\inf_{w\in W}\dim\mathcal O_{X,w}
=
\min_i\operatorname{codim}(W_i,X).
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Write
\[
W=W_1\cup\cdots\cup W_r.
\]
For each irreducible component \(W_i\), let \(\eta_i\) be its generic point. The generic-point local-ring theorem
gives
\[
\operatorname{codim}(W_i,X)=\dim\mathcal O_{X,\eta_i}.
\]
Moreover, for every \(w\in W_i\),
\[
\dim\mathcal O_{X,w}\ge\dim\mathcal O_{X,\eta_i},
\]
because \(w\) is a specialization of \(\eta_i\). Hence
\[
\inf_{w\in W_i}\dim\mathcal O_{X,w}
=
\operatorname{codim}(W_i,X).
\]

Taking the infimum over the finite union,
\[
\inf_{w\in W}\dim\mathcal O_{X,w}
=
\min_i
\inf_{w\in W_i}\dim\mathcal O_{X,w},
\]
and therefore
\[
\boxed{
\inf_{w\in W}\dim\mathcal O_{X,w}
=
\min_i\operatorname{codim}(W_i,X).
}
\]
\end{solution}
\fi
