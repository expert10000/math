% Problem dossier: VI.01.P6
\subsection*{Problem VI.01.P6 --- All polynomial functions on a finite field}
\label{prob:vi01-p6}

\paragraph{Problem.}
Let \(k=\mathbb F_q\).  Consider the evaluation map
\[
\operatorname{ev}:k[x]\longrightarrow \operatorname{Fun}(k,k),
\qquad
f\longmapsto(a\mapsto f(a)).
\]
Prove that \(\operatorname{ev}\) is surjective and that
\[
\ker(\operatorname{ev})=(x^q-x).
\]
Deduce
\[
k[x]/(x^q-x)\cong\operatorname{Fun}(k,k).
\]

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
Over finite fields, formal polynomials contain redundancy but still represent every
possible function.  This problem identifies that redundancy exactly.

\paragraph{Useful ideas before starting.}
Use the functions
\[
\delta_a(x)=1-(x-a)^{q-1}
\]
and Fermat's identity in \(\mathbb F_q\).  For the kernel, divide by \(x^q-x\).

\paragraph{Guided hints.}
\textbf{Hint 1.} Check that \(\delta_a(a)=1\) and \(\delta_a(b)=0\) when \(b\neq a\).

\textbf{Hint 2.} For a function \(\varphi:k\to k\), try
\[
\sum_{a\in k}\varphi(a)\delta_a(x).
\]

\textbf{Hint 3.} If \(f\) vanishes everywhere, write
\[
f=(x^q-x)g+r,
\qquad \deg r<q.
\]

\paragraph{Solution.}
For \(a\in k\), define
\[
\delta_a(x)=1-(x-a)^{q-1}.
\]
At \(x=a\) this equals \(1\).  If \(b\neq a\), then \(b-a\in k^\times\) and
\((b-a)^{q-1}=1\), so \(\delta_a(b)=0\).  Hence every function \(\varphi:k\to k\) is
represented by
\[
f_\varphi(x)=\sum_{a\in k}\varphi(a)\delta_a(x),
\]
proving surjectivity.

The polynomial \(x^q-x\) vanishes at every element of \(k\), so
\[
(x^q-x)\subseteq\ker(\operatorname{ev}).
\]
Conversely, if \(f\) vanishes everywhere, divide by \(x^q-x\):
\[
f=(x^q-x)g+r,
\qquad \deg r<q.
\]
The remainder \(r\) also vanishes at all \(q\) elements of \(k\).  A nonzero polynomial of
degree less than \(q\) cannot have \(q\) roots, so \(r=0\).  Therefore the kernel is
exactly \((x^q-x)\).  The first isomorphism theorem gives the quotient description.

\paragraph{What happened mathematically.}
The relation \(x^q=x\) is precisely the universal relation obeyed by the coordinate
function on the finite set \(\mathbb F_q\).  Modding out by it turns formal polynomials into
actual functions.

\paragraph{Extensions.}
\begin{enumerate}
\item Count both sides and verify that each has \(q^q\) elements.
\item Generalize to \(\mathbb F_q^n\); see Challenge VI.01.C1.
\item Compare with interpolation over an arbitrary finite subset of an infinite field.
\end{enumerate}

\paragraph{Provenance.}
New canonical finite-field problem extending the elementary legacy observation that
\(x^p-x\) vanishes everywhere.
