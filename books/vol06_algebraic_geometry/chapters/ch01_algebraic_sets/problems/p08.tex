% Problem dossier: VI.01.P8
\subsection*{Problem VI.01.P8 --- Vanishing ideals of unions and intersections}
\label{prob:vi01-p8}

\paragraph{Problem.}
For arbitrary subsets \(X,Y\subseteq\mathbb A_k^n\), prove
\[
I(X\cup Y)=I(X)\cap I(Y).
\]
Also prove
\[
I(X)+I(Y)\subseteq I(X\cap Y).
\]
Show that the second inclusion can be strict: assume \(k\) is infinite and take
\[
X=V(x),
\qquad
Y=V(x-y^2)
\]
in \(\mathbb A_k^2\).  Compute all three relevant ideals.

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
The first identity is exact, while the second can lose information because geometric
intersection forgets multiplicity.  This is one of the earliest places where radicals
appear naturally rather than cosmetically.

\paragraph{Useful ideas before starting.}
Use only definitions for the general inclusions.  For the example, identify \(X\cap Y\)
and use the infinite-field graph argument to compute \(I(Y)\).

\paragraph{Guided hints.}
\textbf{Hint 1.} A polynomial vanishes on \(X\cup Y\) exactly when it vanishes on both
pieces.

\textbf{Hint 2.} Every sum \(f+g\) with \(f\in I(X)\), \(g\in I(Y)\) vanishes on
\(X\cap Y\).

\textbf{Hint 3.} In the example,
\[
X\cap Y=\{(0,0)\},
\]
while
\[
I(X)+I(Y)=(x,y^2).
\]

\paragraph{Solution.}
A polynomial belongs to \(I(X\cup Y)\) exactly when it vanishes at every point of \(X\)
and every point of \(Y\).  Therefore
\[
I(X\cup Y)=I(X)\cap I(Y).
\]
If \(f\in I(X)\) and \(g\in I(Y)\), then for any \(p\in X\cap Y\),
\[
(f+g)(p)=0.
\]
Multiplying by arbitrary polynomials preserves vanishing, so the whole ideal sum satisfies
\[
I(X)+I(Y)\subseteq I(X\cap Y).
\]

Now assume \(k\) infinite.  We have
\[
I(X)=(x),
\qquad
I(Y)=(x-y^2),
\]
where the second equality follows by the graph/division argument.  Thus
\[
I(X)+I(Y)=(x,x-y^2)=(x,y^2).
\]
On the other hand, solving \(x=0\) and \(x=y^2\) gives \(y=0\), hence
\[
X\cap Y=\{(0,0)\}
\]
and
\[
I(X\cap Y)=(x,y).
\]
Therefore
\[
(x,y^2)\subsetneq(x,y).
\]
The radical of the smaller ideal is the larger one.

\paragraph{What happened mathematically.}
Set-theoretic intersection remembers only the point \((0,0)\), not the doubled contact
encoded by \(y^2\).  Passing from equations to ordinary zero sets discards multiplicity.

\paragraph{Extensions.}
\begin{enumerate}
\item Over an algebraically closed field, use Nullstellensatz later to show
      \(I(X\cap Y)=\sqrt{I(X)+I(Y)}\) for algebraic \(X,Y\).
\item Find examples with higher-order contact producing \((x,y^m)\).
\item Interpret the missing multiplicity scheme-theoretically.
\end{enumerate}

\paragraph{Provenance.}
New canonical synthesis problem emphasizing the boundary between classical algebraic sets
and later scheme structure.
