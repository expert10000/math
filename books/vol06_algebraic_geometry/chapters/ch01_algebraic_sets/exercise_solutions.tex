% VI/01 exercise solutions - one stable solution block per exercise.

\subsection*{Solution VI.01.1}
We have
\[
V(0)=\mathbb A_k^n,
\qquad
V(1)=\varnothing.
\]
In \(\mathbb A_k^2\), \(V(x)\) is the \(y\)-axis and \(V(x,y)=\{(0,0)\}\).  Finally,
\[
V(x(y-1))=V(x)\cup V(y-1).
\]

\subsection*{Solution VI.01.2}
If \(f(a)=g(a)=0\), then
\[
(g+hf)(a)=0.
\]
Conversely, if \(f(a)=0\) and \((g+hf)(a)=0\), then
\[
g(a)=(g+hf)(a)-h(a)f(a)=0.
\]
Hence the two zero loci agree.  More generally, replacing generators by another list that
generates the same ideal leaves the zero locus unchanged.

\subsection*{Solution VI.01.3}
The intersection is the point \((0,1)\), and
\[
(x)+(y-1)=(x,y-1),
\]
so
\[
X\cap Y=V(x,y-1).
\]
For arbitrary \(I,J\), a point lies in \(V(I)\cap V(J)\) exactly when it annihilates every
element of both ideals, equivalently every element of \(I+J\).

\subsection*{Solution VI.01.4}
The equations are
\[
xy=0,
\qquad
xz=0.
\]
If \(x=0\), there is no condition on \(y,z\), giving \(V(x)\).  If \(x\neq0\), then
\(y=z=0\), giving \(V(y,z)\).  Hence
\[
V(xy,xz)=V(x)\cup V(y,z).
\]

\subsection*{Solution VI.01.5}
Clearly \((y,z)\subseteq I(L)\).  Conversely, write
\[
f(x,y,z)=a_0(x)+yq(x,y,z)+zr(x,y,z).
\]
If \(f\in I(L)\), then \(a_0(t)=f(t,0,0)=0\) for every \(t\in k\).  Since \(k\) is
infinite, the one-variable polynomial \(a_0\) is zero.  Thus \(f\in(y,z)\).

\subsection*{Solution VI.01.6}
If \(I\subseteq J\), every common zero of \(J\) is a common zero of \(I\), hence
\(V(J)\subseteq V(I)\).  If \(X\subseteq Y\), every polynomial vanishing on \(Y\) vanishes
on \(X\), hence \(I(Y)\subseteq I(X)\).  Algebraic constraints and geometric points move
in opposite directions.

\subsection*{Solution VI.01.7}
Extensivity is the basic inclusion \(X\subseteq V(I(X))\).  If \(X\subseteq Y\), then
\(I(Y)\subseteq I(X)\), so applying \(V\) gives
\[
V(I(X))\subseteq V(I(Y)).
\]
For idempotence, the closure inequality gives
\[
I(X)\subseteq I(V(I(X))).
\]
Since \(X\subseteq V(I(X))\), order reversal gives the opposite inclusion.  Therefore
\[
I(V(I(X)))=I(X),
\]
and applying \(V\) proves idempotence.

\subsection*{Solution VI.01.8}
Because \(I\subseteq\sqrt I\), order reversal gives
\[
V(\sqrt I)\subseteq V(I).
\]
If \(a\in V(I)\) and \(f\in\sqrt I\), choose \(m\) with \(f^m\in I\).  Then
\[
f(a)^m=0.
\]
A field has no nonzero nilpotent elements, so \(f(a)=0\).  Hence
\(a\in V(\sqrt I)\).

\subsection*{Solution VI.01.9}
The element \(x\) lies in \((x,y)\) but not in \((x^2,xy,y^2)\), so the ideals differ.
If \((a,b)\) annihilates \(x^2\) and \(y^2\), then \(a=b=0\), hence both zero loci are
\(\{(0,0)\}\).  Moreover \(x^2,y^2\in J\), so \(x,y\in\sqrt J\), giving
\[
\sqrt J=(x,y).
\]

\subsection*{Solution VI.01.10}
The zero loci are
\[
\varnothing\text{ over }\mathbb R,
\qquad
\{i,-i\}\text{ over }\mathbb C,
\]
\[
\{1\}\text{ over }\mathbb F_2,
\qquad
\varnothing\text{ over }\mathbb F_3,
\qquad
\{2,3\}\text{ over }\mathbb F_5.
\]
In \(\mathbb F_3\), the squares are \(0,1\), while \(-1=2\).

\subsection*{Solution VI.01.11}
For every \(a\in\mathbb F_p\), Fermat's identity gives \(a^p=a\).  Thus
\[
(x^p-x)(a)=0
\]
for every point.  Since \(x^p-x\) is not the zero polynomial, the evaluation map from
polynomials to functions is not injective.

\subsection*{Solution VI.01.12}
Let
\[
f(x)=\prod_{j=1}^r(x-a_j).
\]
Then \(f(a_j)=0\) for every \(j\).  Conversely, if \(f(a)=0\), the product of the field
elements \(a-a_j\) is zero, so one factor is zero and \(a=a_j\) for some \(j\).

\subsection*{Solution VI.01.13}
The two equations imply \(y=x\) and \(x=x^2\), so
\[
x(x-1)=0.
\]
A field has no zero divisors, hence \(x=0\) or \(x=1\), giving
\[
\{(0,0),(1,1)\}.
\]
In characteristic \(2\), the points remain distinct because every field still has
\(0\neq1\).

\subsection*{Solution VI.01.14}
Substitution gives
\[
(t^3)^2-(t^2)^3=0.
\]
If \(y^2=x^3\) and \(x\neq0\), let \(t=y/x\).  Then
\[
t^2=\frac{y^2}{x^2}=x,
\qquad
t^3=t\,t^2=y.
\]

\subsection*{Solution VI.01.15}
Every multiple of \(y\) vanishes on \(S\), so \((y)\subseteq I(S)\).  Write
\[
f(x,y)=a_0(x)+yq(x,y).
\]
If \(f\in I(S)\), then \(a_0(t)=0\) for all \(t\in k\).  Since \(k\) is infinite,
\(a_0=0\), so \(f\in(y)\).  Thus \(I(S)=(y)\) and its zero locus is the whole line
\(V(y)\).

\subsection*{Solution VI.01.16}
By Hilbert's basis theorem the ideal \((S)\) has generators \(g_1,\dots,g_m\).  Each
\(g_i\) is a finite combination of elements of \(S\).  Let \(S_0\subseteq S\) be the
finite union of all source elements that occur in those combinations.  Then
\[
(S_0)=(S),
\]
so
\[
V(S_0)=V(S).
\]

\subsection*{Solution VI.01.17}
For \(a\in k^n\), evaluating a polynomial with coefficients in \(k\) gives the same value
whether computed in \(k\) or in \(K\).  Hence the \(k\)-rational solutions are precisely
\(V_K(I)\cap k^n\).  For example,
\[
V_{\mathbb R}(x^2+1)=\varnothing
\]
but
\[
V_{\mathbb C}(x^2+1)=\{i,-i\}.
\]

\subsection*{Solution VI.01.18}
The dictionary is
\[
V(I+J)=V(I)\cap V(J),
\]
\[
V(IJ)=V(I\cap J)=V(I)\cup V(J),
\]
\[
V(\sqrt I)=V(I).
\]
Also \(X\subseteq Y\) implies \(I(Y)\subseteq I(X)\), and directly from the definition
\[
I(X\cup Y)=I(X)\cap I(Y).
\]
Each identity follows by testing what it means for a point to satisfy all equations, or for
a polynomial to vanish at all points.
