% VI/01 pedagogical exercises.
% These are new or substantially redesigned; legacy problems are audited separately.

\subsection*{Exercise VI.01.1 --- Zero-locus warm-up}
\label{ex:vi01-1}
Describe the following subsets of affine space:
\[
V(0),\qquad V(1),\qquad V(x)\subseteq\mathbb A_k^2,
\qquad V(x,y)\subseteq\mathbb A_k^2,
\qquad V\bigl(x(y-1)\bigr)\subseteq\mathbb A_k^2.
\]

\subsection*{Exercise VI.01.2 --- Changing generators without changing solutions}
\label{ex:vi01-2}
Let \(f,g,h\in k[x_1,\dots,x_n]\).  Prove
\[
  V(f,g)=V(f,g+hf).
\]
Explain why elementary row operations on a finite list of equations do not change its
common zero locus when they preserve the generated ideal.

\subsection*{Exercise VI.01.3 --- Intersections are sums of ideals}
\label{ex:vi01-3}
Let
\[
  X=V(x),
  \qquad
  Y=V(y-1)
\]
in \(\mathbb A_k^2\).  Compute \(X\cap Y\) directly and verify
\[
  X\cap Y=V\bigl((x)+(y-1)\bigr).
\]
Then prove the corresponding formula for arbitrary ideals \(I,J\).

\subsection*{Exercise VI.01.4 --- A union in three variables}
\label{ex:vi01-4}
Describe
\[
  V(xy,xz)\subseteq\mathbb A_k^3
\]
as a union of simpler algebraic sets, and verify your answer using the identities for
finite unions.

\subsection*{Exercise VI.01.5 --- Vanishing ideal of a coordinate subspace}
\label{ex:vi01-5}
Let
\[
  L=\{(t,0,0):t\in k\}\subseteq\mathbb A_k^3.
\]
Assume \(k\) is infinite.  Prove
\[
  I(L)=(y,z).
\]

\subsection*{Exercise VI.01.6 --- Order reversal in both directions}
\label{ex:vi01-6}
Prove
\[
  I\subseteq J\Longrightarrow V(J)\subseteq V(I),
\]
and
\[
  X\subseteq Y\Longrightarrow I(Y)\subseteq I(X).
\]
Give a geometric sentence explaining why both arrows reverse.

\subsection*{Exercise VI.01.7 --- The algebraic-envelope operator}
\label{ex:vi01-7}
Define
\[
  \operatorname{Alg}(X):=V(I(X)).
\]
Prove that this operator is extensive, monotone, and idempotent:
\[
  X\subseteq\operatorname{Alg}(X),
\]
\[
  X\subseteq Y\Longrightarrow\operatorname{Alg}(X)\subseteq\operatorname{Alg}(Y),
\]
and
\[
  \operatorname{Alg}(\operatorname{Alg}(X))=\operatorname{Alg}(X).
\]

\subsection*{Exercise VI.01.8 --- Radicals do not change zero loci}
\label{ex:vi01-8}
Prove
\[
  V(I)=V(\sqrt I)
\]
for every ideal \(I\subseteq k[x_1,\dots,x_n]\).  Identify exactly where the fact that a
field has no nonzero nilpotent elements is used.

\subsection*{Exercise VI.01.9 --- Same set, different ideals}
\label{ex:vi01-9}
Show that
\[
  (x,y)\neq(x^2,xy,y^2)
\]
but that the two ideals define the same subset of \(\mathbb A_k^2\).  Compute the radical
of the second ideal.

\subsection*{Exercise VI.01.10 --- A field-dependence table}
\label{ex:vi01-10}
Determine \(V(x^2+1)\subseteq\mathbb A_k^1\) for
\[
  k=\mathbb R,\qquad \mathbb C,\qquad \mathbb F_2,\qquad \mathbb F_3,\qquad \mathbb F_5.
\]

\subsection*{Exercise VI.01.11 --- Polynomial functions over a finite field}
\label{ex:vi01-11}
Over \(\mathbb F_p\), prove that \(x^p-x\) vanishes at every point.  Deduce that the map
from formal polynomials to polynomial functions on \(\mathbb F_p\) is not injective.

\subsection*{Exercise VI.01.12 --- Every finite subset of the affine line is algebraic}
\label{ex:vi01-12}
Let \(a_1,\dots,a_r\in k\) be distinct.  Show
\[
  \{a_1,\dots,a_r\}=V\!\left(\prod_{j=1}^r(x-a_j)\right)
  \subseteq\mathbb A_k^1.
\]
No algebraic-closedness hypothesis is needed.

\subsection*{Exercise VI.01.13 --- Line meets parabola}
\label{ex:vi01-13}
Determine
\[
  V(y-x)\cap V(y-x^2)
\]
in \(\mathbb A_k^2\).  Explain why the answer is valid in every field and what changes,
if anything, in characteristic \(2\).

\subsection*{Exercise VI.01.14 --- Parametrizing the cusp away from the origin}
\label{ex:vi01-14}
Let
\[
  C=V(y^2-x^3).
\]
Verify that \((t^2,t^3)\in C\) for every \(t\in k\).  If \((x,y)\in C\) and
\(x\neq0\), prove that \(t=y/x\) recovers the point.

\subsection*{Exercise VI.01.15 --- Infinite points force a line}
\label{ex:vi01-15}
Assume \(k\) is infinite and let
\[
  S=\{(t,0):t\in k\}\subseteq\mathbb A_k^2.
\]
Prove
\[
  I(S)=(y)
\]
and hence
\[
  V(I(S))=V(y).
\]

\subsection*{Exercise VI.01.16 --- Infinitely many equations reduce to finitely many}
\label{ex:vi01-16}
Assume Hilbert's basis theorem.  If \(S\subseteq k[x_1,\dots,x_n]\) is arbitrary, prove
that there exists a finite subset \(S_0\subseteq S\) such that
\[
  V(S)=V(S_0).
\]

\subsection*{Exercise VI.01.17 --- Extending the ground field}
\label{ex:vi01-17}
Let \(K/k\) be a field extension and let \(I\subseteq k[x_1,\dots,x_n]\).  Show that
\[
  V_k(I)=V_K(I)\cap k^n,
\]
where the same polynomials are evaluated in \(K\).  Give an example where
\(V_K(I)\) is strictly larger.

\subsection*{Exercise VI.01.18 --- Reconstruct the dictionary}
\label{ex:vi01-18}
Without looking back at the summary table, fill in the geometric counterpart of each
algebraic operation:
\[
  I+J,\qquad IJ,\qquad I\cap J,\qquad \sqrt I,
\]
and the algebraic counterpart of
\[
  X\subseteq Y,\qquad X\cup Y.
\]
Prove every identity you use.
