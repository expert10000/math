% VI/01 exercises

\section{Exercises}
\label{sec:vi01-exercises}

\begin{exercise}[Extreme zero loci]
Prove directly from the definition that
\[
  V(0)=\mathbb A_k^n,
  \qquad
  V(1)=\varnothing.
\]
\end{exercise}

\begin{exercise}[Generated ideals]
Let \(S\subseteq k[x_1,\dots,x_n]\).  Prove that
\[
  V(S)=V((S)).
\]
Then explain why multiplying one defining equation by a nonzero scalar does not change the
zero locus.
\end{exercise}

\begin{exercise}[Intersections]
For ideals \(I_\lambda\subseteq k[x_1,\dots,x_n]\), prove
\[
  \bigcap_\lambda V(I_\lambda)
  =
  V\!\left(\sum_\lambda I_\lambda\right).
\]
\end{exercise}

\begin{exercise}[Unions]
Prove
\[
  V(I)\cup V(J)=V(IJ)=V(I\cap J).
\]
Identify exactly where the fact that \(k\) has no zero divisors is used.
\end{exercise}

\begin{exercise}[Coordinate axes]
Describe
\[
  V(xy,xz)\subseteq\mathbb A_k^3
\]
as a union of simpler algebraic sets.
\end{exercise}

\begin{exercise}[Vanishing ideals are ideals]
For an arbitrary subset \(X\subseteq\mathbb A_k^n\), prove that \(I(X)\) is an ideal.
\end{exercise}

\begin{exercise}[Order reversal]
Prove both implications:
\[
  I\subseteq J\Longrightarrow V(J)\subseteq V(I),
\]
and
\[
  X\subseteq Y\Longrightarrow I(Y)\subseteq I(X).
\]
Give a one-sentence geometric interpretation of each.
\end{exercise}

\begin{exercise}[The algebraic closure operator]
Show that
\[
  X\subseteq V(I(X))
\]
for every subset \(X\), and prove that if \(Y\) is algebraic and \(X\subseteq Y\), then
\[
  V(I(X))\subseteq Y.
\]
\end{exercise}

\begin{exercise}[Radicals]
Prove
\[
  V(I)=V(\sqrt I).
\]
Deduce that
\[
  \sqrt I\subseteq I(V(I)).
\]
\end{exercise}

\begin{exercise}[Same set, different ideals]
In \(k[x,y]\), compare the ideals
\[
  I=(x,y)
  \quad\text{and}\quad
  J=(x^2,xy,y^2).
\]
Show that \(I\neq J\), but
\[
  V(I)=V(J).
\]
\end{exercise}

\begin{exercise}[Field dependence]
Determine \(V(x^2+1)\) in \(\mathbb A_k^1\) for
\[
  k=\mathbb R,\qquad
  k=\mathbb C,\qquad
  k=\mathbb F_2,\qquad
  k=\mathbb F_5.
\]
\end{exercise}

\begin{exercise}[Finite-field polynomial functions]
Over \(\mathbb F_p\), prove that
\[
  x^p-x
\]
vanishes at every point of \(\mathbb A_{\mathbb F_p}^1\).  Explain why this shows that the
map ``polynomial \(\mapsto\) polynomial function'' need not be injective over a finite field.
\end{exercise}

\begin{exercise}[Algebraic sets in one variable]
Assume \(k\) is algebraically closed.  Prove that every proper algebraic subset of
\(\mathbb A_k^1\) is finite.
\end{exercise}

\begin{exercise}[A parametrized curve]
Let
\[
  C=V(y^2-x^3)\subseteq\mathbb A_k^2.
\]
Verify that every point of the form
\[
  (t^2,t^3)
\]
lies in \(C\).  If \(k\) is a field and \((x,y)\in C\) with \(x\neq0\), show that
\[
  t=\frac{y}{x}
\]
satisfies \(x=t^2\) and \(y=t^3\).
\end{exercise}

\begin{exercise}[Finite equations from Noetherianity]
Assume Hilbert's basis theorem.  Prove that every algebraic subset of
\(\mathbb A_k^n\), even one initially defined by infinitely many equations, can be defined by
finitely many equations.
\end{exercise}

\subsection*{Further challenge}

\begin{exercise}[The \(V\)-\(I\) adjunction pattern]
For an ideal \(J\subseteq k[x_1,\dots,x_n]\) and a subset \(X\subseteq\mathbb A_k^n\),
prove the equivalence
\[
  X\subseteq V(J)
  \quad\Longleftrightarrow\quad
  J\subseteq I(X).
\]
Use it to rederive the two closure inequalities
\[
  J\subseteq I(V(J)),
  \qquad
  X\subseteq V(I(X)).
\]
\end{exercise}
