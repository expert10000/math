% VI/01 worked examples

\section{Worked examples}
\label{sec:vi01-worked-examples}

\begin{example}[A point in the affine plane]
Fix \(a,b\in k\).  Then
\[
  V(x-a,y-b)=\{(a,b)\}.
\]
Indeed, both equations vanish exactly when \(x=a\) and \(y=b\).
\end{example}

\begin{example}[A line as a zero locus]
Over any field \(k\),
\[
  V(y-2x-1)
\]
is the graph of the affine-linear function \(y=2x+1\).  The same geometric set can be
defined by any nonzero scalar multiple of the equation.
\end{example}

\begin{example}[A union encoded by a product]
In \(\mathbb A_k^2\),
\[
  V(x(y-1))
  =
  V(x)\cup V(y-1).
\]
Thus a product equation records a union: the vertical axis \(x=0\) together with the
horizontal line \(y=1\).
\end{example}

\begin{example}[A cusp]
The algebraic set
\[
  C=V(y^2-x^3)\subseteq\mathbb A_k^2
\]
is a cusp when \(k=\mathbb R\).  The parametrization
\[
  t\longmapsto (t^2,t^3)
\]
lands in \(C\) because
\[
  (t^3)^2-(t^2)^3=0.
\]
The origin is geometrically special.  Later chapters will distinguish smooth and singular
points using local algebra and tangent spaces.
\end{example}

\begin{example}[Different equations, same set]
In \(\mathbb A_k^2\),
\[
  V(x,y)
  =
  V(x^2,xy,y^2)
  =
  \{(0,0)\}.
\]
The ideals
\[
  (x,y)
  \quad\text{and}\quad
  (x^2,xy,y^2)
\]
are different, but their radicals coincide:
\[
  \sqrt{(x^2,xy,y^2)}=(x,y).
\]
The underlying algebraic set therefore cannot see the difference between them.
\end{example}

\begin{example}[Intersection of a line and a parabola]
Let
\[
  L=V(y-x),
  \qquad
  P=V(y-x^2)
\]
in \(\mathbb A_k^2\).  Their intersection is
\[
  L\cap P
  =
  V(y-x,y-x^2).
\]
Subtracting the two equations gives
\[
  x^2-x=x(x-1)=0.
\]
Thus, whenever \(0\neq1\) in \(k\),
\[
  L\cap P=\{(0,0),(1,1)\}.
\]
This illustrates how geometric intersection translates into simultaneous polynomial
equations.
\end{example}

\begin{example}[Real and complex geometry differ]
For
\[
  X=V(x^2+y^2+1),
\]
we have
\[
  X(\mathbb R)=\varnothing,
\]
because \(x^2+y^2+1>0\) for real \(x,y\).  Over \(\mathbb C\), however, points exist; for
example
\[
  (i,0)\in X(\mathbb C).
\]
The formula is unchanged, but the set of available coordinates is different.
\end{example}

\begin{example}[An infinite set with no extra polynomial equations]
Let \(k\) be an infinite field and consider
\[
  X=\{(t,0):t\in k\}\subseteq\mathbb A_k^2.
\]
Then
\[
  I(X)=(y).
\]
Certainly \(y\) vanishes on \(X\), so \((y)\subseteq I(X)\).  Conversely, write
\[
  f(x,y)=q(x,y)y+r(x),
\]
where \(r(x)=f(x,0)\).  If \(f\) vanishes on all of \(X\), then \(r(t)=0\) for every
\(t\in k\).  Since \(k\) is infinite, a one-variable polynomial with infinitely many roots is
zero, so \(r=0\).  Hence \(f\in(y)\).
\end{example}
