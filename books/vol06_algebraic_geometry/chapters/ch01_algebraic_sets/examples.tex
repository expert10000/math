\section{Worked examples}
\label{sec:vi01-worked-examples}

\begin{example}[From equations to a single point]
Let \(a,b\in k\).  The equations
\[
  x-a=0,
  \qquad
  y-b=0
\]
cut out exactly one point:
\[
  V(x-a,y-b)=\{(a,b)\}.
\]
This is the simplest illustration of simultaneous equations producing an algebraic set.
\end{example}

\begin{example}[One equation can encode several pieces]
In \(\mathbb A_k^2\),
\[
  V\bigl(x(y-1)\bigr)=V(x)\cup V(y-1).
\]
Geometrically this is the union of the vertical line \(x=0\) and the horizontal line
\(y=1\).  The factorization of the polynomial is already predicting the union identity
\(V(IJ)=V(I)\cup V(J)\).
\end{example}

\begin{example}[Intersection by adding equations]
Let
\[
  L=V(y-x),
  \qquad
  P=V(y-x^2).
\]
Then
\[
  L\cap P=V(y-x,y-x^2).
\]
Subtracting the equations gives
\[
  x^2-x=x(x-1)=0.
\]
Hence, in every field with \(0\neq1\),
\[
  L\cap P=\{(0,0),(1,1)\}.
\]
The geometric intersection is obtained by imposing both systems at once.
\end{example}

\begin{example}[The coordinate axes in three-space]
Consider
\[
  X=V(xy,xz,yz)\subseteq\mathbb A_k^3.
\]
At a point of \(X\), no two coordinates can be simultaneously nonzero.  Therefore
\[
  X=V(y,z)\cup V(x,z)\cup V(x,y),
\]
the union of the three coordinate axes.  This example will return as a harder problem when
we ask for its full vanishing ideal.
\end{example}

\begin{example}[Different ideals, identical geometry]
The ideals
\[
  I=(x,y),
  \qquad
  J=(x^2,xy,y^2)
\]
are not equal, but
\[
  V(I)=V(J)=\{(0,0)\}.
\]
Indeed, \(J\subsetneq I\) while \(\sqrt J=I\).  Ordinary algebraic sets remember where
polynomials vanish, but not how strongly they vanish.
\end{example}

\begin{example}[The algebraic envelope of infinitely many points]
Assume \(k\) is infinite and put
\[
  S=\{(t,0):t\in k\}\subseteq\mathbb A_k^2.
\]
Every polynomial vanishing on \(S\) is divisible by \(y\).  To see the key point, write
\[
  f(x,y)=a_0(x)+a_1(x)y+\cdots+a_m(x)y^m.
\]
If \(f(t,0)=0\) for every \(t\in k\), then \(a_0(t)=0\) for infinitely many \(t\), so
\(a_0=0\).  Hence
\[
  I(S)=(y),
  \qquad
  V(I(S))=V(y).
\]
Thus an infinite set of points can force an entire algebraic line.
\end{example}

\begin{example}[A parametrized cusp]
Let
\[
  C=V(y^2-x^3)\subseteq\mathbb A_k^2.
\]
The map
\[
  t\longmapsto(t^2,t^3)
\]
lands in \(C\) because
\[
  (t^3)^2-(t^2)^3=0.
\]
Conversely, if \((x,y)\in C\) and \(x\neq0\), then
\[
  t=\frac yx
\]
satisfies \(x=t^2\) and \(y=t^3\).  This is a first glimpse of how equations and
parametrizations can describe the same geometric object in different ways.
\end{example}

\begin{example}[The ground field changes the point set]
For the equation
\[
  x^2+1=0,
\]
we obtain
\[
  V_{\mathbb R}(x^2+1)=\varnothing,
  \qquad
  V_{\mathbb C}(x^2+1)=\{i,-i\}.
\]
Over \(\mathbb F_5\), the solutions are \(2\) and \(3\), because
\[
  2^2=3^2=4=-1.
\]
The polynomial expression has not changed; the geometry has.
\end{example}

\begin{example}[A finite-field polynomial that vanishes everywhere]
Over \(\mathbb F_p\), Fermat's identity gives
\[
  a^p-a=0
\]
for every \(a\in\mathbb F_p\).  Hence the nonzero polynomial
\[
  x^p-x
\]
induces the zero function on \(\mathbb A_{\mathbb F_p}^1\).  This is why formal
polynomials and polynomial functions must be distinguished over finite fields.
\end{example}

\begin{example}[A closure calculation without topology]
Let \(k\) be infinite and
\[
  S=\{(t,t^2):t\in k\}\subseteq\mathbb A_k^2.
\]
Certainly
\[
  y-x^2\in I(S),
\]
so
\[
  V(I(S))\subseteq V(y-x^2).
\]
The reverse inclusion follows once one proves
\[
  I(S)=(y-x^2),
\]
which is developed in Problem VI.01.P2.  Thus the smallest algebraic set containing the
parametrized point set is exactly the parabola.
\end{example}
