% VI/01 exercise hints - one stable hint block per exercise.

\subsection*{Hint VI.01.1}
Use the definitions, then factor \(x(y-1)\).

\subsection*{Hint VI.01.2}
At a common zero of \(f\) and \(g\), the expression \(g+hf\) also vanishes.  Reverse the
argument by writing \(g=(g+hf)-hf\).

\subsection*{Hint VI.01.3}
The ideal sum is generated by all equations from both systems.

\subsection*{Hint VI.01.4}
Separate the cases \(x=0\) and \(x\neq0\).

\subsection*{Hint VI.01.5}
Regard a polynomial as a polynomial in \(x\) with coefficients in \(k[y,z]\), then evaluate
at \((t,0,0)\).  Infinite fields force the pure \(x\)-part to vanish identically.

\subsection*{Hint VI.01.6}
More equations leave fewer points; more points leave fewer equations that vanish everywhere.

\subsection*{Hint VI.01.7}
For idempotence, first prove
\[
  I(V(I(X)))=I(X).
\]
Use both order reversal and the closure inequalities.

\subsection*{Hint VI.01.8}
If \(f^m\in I\) and \(a\in V(I)\), then \(f(a)^m=0\) in the field \(k\).

\subsection*{Hint VI.01.9}
The square of every element of \((x,y)\) need not lie in the smaller ideal, but every
generator \(x,y\) lies in its radical.

\subsection*{Hint VI.01.10}
Solve \(x^2=-1\) separately in each field.

\subsection*{Hint VI.01.11}
Use Fermat's identity \(a^p=a\) for \(a\in\mathbb F_p\).

\subsection*{Hint VI.01.12}
A product is zero in a field exactly when at least one factor is zero.

\subsection*{Hint VI.01.13}
Substitute \(y=x\) into \(y=x^2\), obtaining \(x(x-1)=0\).

\subsection*{Hint VI.01.14}
For the converse, compute \((y/x)^2\) using \(y^2=x^3\).

\subsection*{Hint VI.01.15}
Write
\[
  f(x,y)=a_0(x)+yq(x,y).
\]
If \(f(t,0)=0\) for infinitely many \(t\), then \(a_0=0\).

\subsection*{Hint VI.01.16}
The ideal \((S)\) is finitely generated.  Each chosen generator is a finite linear
combination of elements of \(S\); collect all elements of \(S\) that occur.

\subsection*{Hint VI.01.17}
A point with coordinates in \(k\) satisfies a polynomial equation in \(k\) whether the
calculation is performed in \(k\) or in the larger field \(K\).

\subsection*{Hint VI.01.18}
Use the propositions on arbitrary intersections, finite unions, radicals, and order
reversal.  For \(X\cup Y\), start directly from the definition of a vanishing ideal.
