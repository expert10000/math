The zero polynomial evaluates to \(0\) at every point, hence \(V(0)=\mathbb A_k^n\).
The constant polynomial \(1\) evaluates to \(1\neq0\) at every point, hence
\(V(1)=\varnothing\).
\end{solution}

\begin{solution}
A point \(a\) belongs to \(\bigcap_\lambda V(I_\lambda)\) exactly when every polynomial in
every \(I_\lambda\) vanishes at \(a\).  Then every finite sum of such polynomials also
vanishes, which is exactly the condition
\[
  a\in V\!\left(\sum_\lambda I_\lambda\right).
\]
The converse is immediate because every \(I_\lambda\) lies in the sum.
\end{solution}

\begin{solution}
The inclusions
\[
  V(I)\cup V(J)\subseteq V(I\cap J)\subseteq V(IJ)
\]
follow from \(IJ\subseteq I\cap J\subseteq I,J\) and order reversal.  For the reverse
inclusion, suppose \(a\notin V(I)\cup V(J)\).  Choose \(f\in I\), \(g\in J\) with
\(f(a)\neq0\) and \(g(a)\neq0\).  Then \(fg\in IJ\), and because \(k\) is a field,
\[
  f(a)g(a)\neq0.
\]
Hence \(a\notin V(IJ)\).  The no-zero-divisor property is used in this last implication.
\end{solution}

\begin{solution}
The equations are
\[
  xy=0,\qquad xz=0.
\]
If \(x=0\), there is no restriction on \(y,z\), giving the plane \(V(x)\).  If \(x\neq0\),
then \(y=z=0\), giving the \(x\)-axis \(V(y,z)\).  Therefore
\[
  V(xy,xz)=V(x)\cup V(y,z).
\]
\end{solution}

\begin{solution}
The zero polynomial belongs to \(I(X)\).  If \(f,g\in I(X)\), then
\[
  (f+g)(a)=0
\]
for every \(a\in X\).  If \(h\) is arbitrary and \(f\in I(X)\), then
\[
  (hf)(a)=h(a)f(a)=0
\]
for every \(a\in X\).  Hence \(I(X)\) is an ideal.
\end{solution}

\begin{solution}
If \(I\subseteq J\), a point satisfying all equations in \(J\) satisfies all equations in
\(I\), so \(V(J)\subseteq V(I)\).  If \(X\subseteq Y\), a polynomial vanishing on \(Y\)
vanishes on \(X\), so \(I(Y)\subseteq I(X)\).  More equations mean fewer points; more
points mean fewer equations.
\end{solution}

\begin{solution}
Every polynomial in \(I(X)\) vanishes on every point of \(X\), so
\[
  X\subseteq V(I(X)).
\]
If \(Y=V(J)\) and \(X\subseteq Y\), then each polynomial in \(J\) vanishes on \(X\), hence
\(J\subseteq I(X)\).  Applying \(V\) reverses inclusion:
\[
  V(I(X))\subseteq V(J)=Y.
\]
\end{solution}

\begin{solution}
Since \(I\subseteq\sqrt I\),
\[
  V(\sqrt I)\subseteq V(I).
\]
If \(a\in V(I)\) and \(f\in\sqrt I\), choose \(m\) with \(f^m\in I\).  Then
\[
  0=f(a)^m,
\]
so \(f(a)=0\).  Therefore \(a\in V(\sqrt I)\), proving equality.  Every
\(f\in\sqrt I\) consequently vanishes on \(V(I)\), so
\[
  \sqrt I\subseteq I(V(I)).
\]
\end{solution}

\begin{solution}
Clearly \(J\subseteq I\), but \(x\notin J\), so \(I\neq J\).  Both zero loci consist only
of the origin:
\[
  V(I)=\{(0,0)\}.
\]
If \((a,b)\in V(J)\), then \(a^2=b^2=0\), hence \(a=b=0\), so
\[
  V(J)=\{(0,0)\}=V(I).
\]
\end{solution}

\begin{solution}
Over \(\mathbb R\), \(x^2=-1\) has no solution.  Over \(\mathbb C\), the solutions are
\(\pm i\).  In \(\mathbb F_2\), \(-1=1\), and \(1^2=1\), so the zero locus is
\(\{1\}\).  In \(\mathbb F_5\), \(-1=4\), and \(2^2=3^2=4\), so the zero locus is
\(\{2,3\}\).
\end{solution}

\begin{solution}
For every \(a\in\mathbb F_p\), Fermat's theorem gives \(a^p=a\).  Hence
\[
  a^p-a=0
\]
for every \(a\).  The polynomial \(x^p-x\) is nonzero in \(\mathbb F_p[x]\), yet its
induced function is the zero function.  Thus the polynomial-to-function map is not
injective.
\end{solution}

\begin{solution}
Let \(X=V(I)\subsetneq\mathbb A_k^1\).  If \(I=(0)\), then \(X=\mathbb A_k^1\), contrary
to properness.  Choose a nonzero \(f\in I\).  Then
\[
  X\subseteq V(f).
\]
A nonzero one-variable polynomial has finitely many roots, so \(X\) is finite.  Over an
algebraically closed field every finite subset is itself algebraic, since
\[
  \{a_1,\dots,a_r\}
  =
  V\!\left(\prod_{j=1}^r(x-a_j)\right).
\]
\end{solution}

\begin{solution}
For every \(t\),
\[
  (t^3)^2-(t^2)^3=t^6-t^6=0,
\]
so \((t^2,t^3)\in C\).  Conversely, if \(y^2=x^3\) and \(x\neq0\), let \(t=y/x\).  Then
\[
  t^2=\frac{y^2}{x^2}=\frac{x^3}{x^2}=x,
\]
and therefore
\[
  t^3=t\,t^2=\frac{y}{x}x=y.
\]
\end{solution}

\begin{solution}
If \(X=V(S)\), then \(X=V((S))\).  By Hilbert's basis theorem,
\[
  (S)=(f_1,\dots,f_r)
\]
for finitely many polynomials.  Hence
\[
  X=V(f_1,\dots,f_r).
\]
\end{solution}

\begin{solution}
The statement \(X\subseteq V(J)\) means that for every \(a\in X\) and every \(f\in J\),
\[
  f(a)=0.
\]
This is exactly the statement that every \(f\in J\) belongs to \(I(X)\), i.e.
\[
  J\subseteq I(X).
\]
Putting \(X=V(J)\) gives \(J\subseteq I(V(J))\).  Putting \(J=I(X)\) gives
\(X\subseteq V(I(X))\).
\end{solution}
