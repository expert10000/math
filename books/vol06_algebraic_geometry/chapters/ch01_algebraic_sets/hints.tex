% VI/01 hints

\chapter*{Hints for Chapter VI/01}
\addcontentsline{toc}{chapter}{Hints for Chapter VI/01}

\begin{hint}
Use only the definitions of \(V(0)\) and \(V(1)\).
\end{hint}

\begin{hint}
Evaluate a general finite combination
\[
  h_1f_1+\cdots+h_rf_r
\]
at a common zero of the \(f_i\).
\end{hint}

\begin{hint}
A point lies in every \(V(I_\lambda)\) exactly when every generator coming from every
\(I_\lambda\) vanishes there.
\end{hint}

\begin{hint}
For the difficult inclusion, prove the contrapositive.  If a point belongs to neither
\(V(I)\) nor \(V(J)\), choose one polynomial from each ideal that does not vanish there and
multiply them.
\end{hint}

\begin{hint}
Factor the conditions as
\[
  x(y,z)=(0,0).
\]
Separate the cases \(x=0\) and \(x\neq0\).
\end{hint}

\begin{hint}
Check closure under addition and under multiplication by an arbitrary polynomial.
\end{hint}

\begin{hint}
Both statements say that imposing more constraints reverses inclusion on the other side of
the algebra--geometry correspondence.
\end{hint}

\begin{hint}
Write \(Y=V(J)\).  From \(X\subseteq V(J)\), show \(J\subseteq I(X)\), then reverse
inclusion with \(V\).
\end{hint}

\begin{hint}
If \(f^m\) vanishes at a point, then \(f\) vanishes at that point because the target is a
field.
\end{hint}

\begin{hint}
Observe that \(x\in I\) but \(x\notin J\), while
\(\sqrt J=I\).
\end{hint}

\begin{hint}
Check squares in each field.  In \(\mathbb F_5\), note that \(2^2=4=-1\).
\end{hint}

\begin{hint}
Use Fermat's identity \(a^p=a\) for every \(a\in\mathbb F_p\).
\end{hint}

\begin{hint}
A proper algebraic set is \(V(I)\) for a nonzero ideal \(I\).  Pick a nonzero
\(f\in I\) and use the fact that a nonzero one-variable polynomial has finitely many roots.
\end{hint}

\begin{hint}
From \(y^2=x^3\) and \(t=y/x\), compute \(t^2\) and \(t^3\).
\end{hint}

\begin{hint}
Replace the defining family \(S\) by the ideal \((S)\), then apply Noetherianity.
\end{hint}

\begin{hint}
Unpack both sides: ``every point of \(X\) kills every polynomial of \(J\)'' is the same
statement as ``every polynomial of \(J\) vanishes on every point of \(X\)''.
\end{hint}
