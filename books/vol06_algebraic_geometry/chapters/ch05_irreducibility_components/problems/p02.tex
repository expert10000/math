% Problem dossier: VI.05.P2
\subsection*{Problem VI.05.P2 --- Uniqueness of irreducible decomposition}
\label{prob:vi-05-p2}

\paragraph{Problem.}
\label{prob:vi05-p2}
Suppose
\[
X=X_1\cup\cdots\cup X_r
\]
and
\[
X=X'_1\cup\cdots\cup X'_s
\]
are two decompositions of an affine algebraic set into irreducible closed subsets.  Assume
there are no redundant containments:
\[
X_i\not\subseteq X_j,
\qquad
X'_i\not\subseteq X'_j
\qquad(i\neq j).
\]
Prove that
\[
r=s,
\]
and after reindexing,
\[
X_i=X'_i
\]
for all \(i\).

\ifdefined\FullProblemDossiers

\paragraph{Why this problem matters.}
Existence of components is not enough: a canonical decomposition is useful only if it is
unique.  This problem proves the uniqueness mechanism used throughout algebraic geometry.

\paragraph{Useful ideas before starting.}
\begin{itemize}
\item an irreducible space contained in a finite union of closed sets lies in one of them;
\item irreducible components are maximal irreducible closed subsets;
\item the decompositions are assumed irredundant.
\end{itemize}

\paragraph{Guided hints.}
\textbf{Hint 1.} Fix \(X_i\).  Since it lies in the finite union
\(X'_1\cup\cdots\cup X'_s\), irreducibility forces it into one \(X'_j\).

\textbf{Hint 2.} Apply the same argument in the opposite direction and use the
irredundancy assumptions to turn inclusions into equalities.

\paragraph{Solution.}
Fix \(i\).  Since
\[
X_i\subseteq X'_1\cup\cdots\cup X'_s
\]
and \(X_i\) is irreducible, there exists \(j(i)\) such that
\[
X_i\subseteq X'_{j(i)}.
\]
Applying the same argument to \(X'_{j(i)}\), there exists \(\ell\) with
\[
X'_{j(i)}\subseteq X_\ell.
\]
Hence
\[
X_i\subseteq X_\ell.
\]
The noncontainment assumption implies \(\ell=i\), so
\[
X_i=X'_{j(i)}.
\]
Thus every \(X_i\) appears among the \(X'_j\).  By symmetry, every \(X'_j\) appears among
the \(X_i\).  Therefore the two collections coincide, and in particular \(r=s\).

\paragraph{What happened mathematically.}
The proof is a finite-containment argument: irreducibility converts a finite cover into
one containment, and maximality/irredundancy upgrades containment to equality.  This
mechanism reappears for minimal primes in commutative algebra.

\paragraph{Extensions.}
\begin{enumerate}
\item Translate the argument into a statement about minimal prime ideals.
\item Prove uniqueness for irreducible components in any Noetherian topological space.
\item Give a non-Noetherian space with infinitely many irreducible components.
\end{enumerate}
