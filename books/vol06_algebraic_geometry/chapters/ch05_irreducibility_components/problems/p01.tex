% Problem dossier: VI.05.P1
\subsection*{Problem VI.05.P1 --- Dense open subsets of irreducible sets}
\label{prob:vi-05-p1}

\paragraph{Problem.}
\label{prob:vi05-p1}
Let \(X\) be an irreducible algebraic set and let
\[
\varnothing\neq U\subseteq X
\]
be open.  Prove that \(U\) is both dense in \(X\) and irreducible.

\ifdefined\FullProblemDossiers

\paragraph{Why this problem matters.}
This is the basic topological engine behind irreducible geometry.  It explains why
arguments may safely shrink to a nonempty open subset without losing the generic behavior
of an irreducible space.

\paragraph{Useful ideas before starting.}
\begin{itemize}
\item the open-set criterion for irreducibility;
\item closure and density;
\item the subspace topology.
\end{itemize}

\paragraph{Guided hints.}
\textbf{Hint 1.} To prove density, assume the closure of \(U\) is proper and look at
its open complement.

\textbf{Hint 2.} To prove irreducibility of \(U\), represent two nonempty opens in \(U\) as
\(U\cap V_1\) and \(U\cap V_2\).

\paragraph{Solution.}
Let \(U\subseteq X\) be nonempty open.

To prove density, suppose
\[
\overline U\neq X.
\]
Then
\[
X\setminus\overline U
\]
is a nonempty open subset disjoint from \(U\), contradicting irreducibility.  Hence
\[
\overline U=X.
\]

To prove \(U\) irreducible, let \(U_1,U_2\subseteq U\) be nonempty open subsets.  They are
of the form
\[
U_i=U\cap V_i
\]
with \(V_i\subseteq X\) open.  Then \(U\cap V_1\) and \(U\cap V_2\) are nonempty open
subsets of \(X\), so they intersect.  Thus
\[
U_1\cap U_2\neq\varnothing.
\]
Therefore \(U\) is irreducible.

\paragraph{What happened mathematically.}
Irreducibility is a statement about every nonempty open set, not merely about the
whole space.  This is why ``generic'' arguments in algebraic geometry can often be made on
any convenient dense open subset.

\paragraph{Extensions.}
\begin{enumerate}
\item Show that the intersection of finitely many nonempty open subsets of an irreducible
space is nonempty.
\item Prove that every dense open subset has the same generic point whenever a generic point
exists.
\item Give an example showing that the intersection of infinitely many nonempty opens may be
empty.
\end{enumerate}
