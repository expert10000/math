\subsection*{Exercise VI.05.1 --- Three quick irreducibility tests}
\label{ex:vi05-1}
Assume \(k\) is algebraically closed.  Decide which of the following algebraic sets are
irreducible, and justify each answer using the coordinate ring:
\[
V(y-x^2),\qquad
V(xy),\qquad
V(xy-1)
\subseteq \mathbb A_k^2.
\]

\subsection*{Exercise VI.05.2 --- Dense principal opens}
\label{ex:vi05-2}
Let \(X\) be irreducible and let \(f\in k[X]\) be nonzero.  Assume
\[
D_X(f)\neq\varnothing.
\]
Prove that \(D_X(f)\) is dense in \(X\).

\subsection*{Exercise VI.05.3 --- Polynomial identities from dense subsets}
\label{ex:vi05-3}
Let \(X\) be an irreducible affine algebraic set and \(U\subseteq X\) a nonempty open
subset.  If \(f,g\in k[X]\) satisfy
\[
f|_U=g|_U,
\]
prove that
\[
f=g
\]
in \(k[X]\).

\subsection*{Exercise VI.05.4 --- A parametrized cusp}
\label{ex:vi05-4}
Assume \(k\) is infinite.  Let
\[
C=V(y^2-x^3)\subseteq\mathbb A_k^2.
\]
Use the polynomial map
\[
\phi:\mathbb A_k^1\to C,
\qquad
t\mapsto(t^2,t^3),
\]
to show that \(C\) is irreducible.

\subsection*{Exercise VI.05.5 --- Reading components from a factorization}
\label{ex:vi05-5}
Determine the irreducible components of
\[
X=V\bigl(xy(x-y)(x+y)\bigr)\subseteq\mathbb A_k^2,
\]
assuming \(\operatorname{char}k\neq2\).  Is \(X\) connected?

\subsection*{Exercise VI.05.6 --- A reducible but connected surface}
\label{ex:vi05-6}
Consider
\[
X=V(xy)\subseteq\mathbb A_k^3.
\]
Describe its irreducible components, their intersection, and the component-intersection
graph.  Deduce that \(X\) is connected but reducible.

\subsection*{Exercise VI.05.7 --- Zero divisors versus idempotents}
\label{ex:vi05-7}
In
\[
A=k[x,y]/(xy),
\]
exhibit nonzero zero divisors.  Then prove that \(A\) has no nontrivial idempotents when
\(k\) is algebraically closed.  Explain the geometric distinction detected by these two
facts.

\subsection*{Exercise VI.05.8 --- Two disjoint lines}
\label{ex:vi05-8}
Let
\[
X=V(x(x-1))\subseteq\mathbb A_k^2.
\]
Use the Chinese remainder theorem to prove
\[
k[X]\cong k[y]\times k[y].
\]
Construct an explicit nontrivial idempotent and identify the corresponding clopen
decomposition of \(X\).

\subsection*{Exercise VI.05.9 --- A chain of components}
\label{ex:vi05-9}
Let
\[
X=X_1\cup X_2\cup X_3\cup X_4,
\]
where every \(X_i\) is irreducible and
\[
X_1\cap X_2\neq\varnothing,\qquad
X_2\cap X_3\neq\varnothing,\qquad
X_3\cap X_4\neq\varnothing.
\]
No assumption is made about the other pairwise intersections.  Prove that \(X\) is
connected.

\subsection*{Exercise VI.05.10 --- Continuous images of irreducible spaces}
\label{ex:vi05-10}
Prove that the continuous image of an irreducible topological space is irreducible.
Deduce that if
\[
\phi:X\to Y
\]
is a morphism and \(X\) is irreducible, then
\[
\overline{\phi(X)}
\]
is irreducible.

\subsection*{Exercise VI.05.11 --- Removing a proper closed set}
\label{ex:vi05-11}
Let \(X\) be irreducible and \(Z\subsetneq X\) closed.  Prove that
\[
X\setminus Z
\]
is nonempty, irreducible, and dense in \(X\).

\subsection*{Exercise VI.05.12 --- Compare four notions}
\label{ex:vi05-12}
For each of the following affine algebraic sets over an algebraically closed field, determine
whether it is irreducible and whether it is connected:
\[
\mathbb A_k^1,\qquad
V(xy)\subseteq\mathbb A_k^2,\qquad
V(x(x-1))\subseteq\mathbb A_k^2,\qquad
\{0,1,2\}\subseteq\mathbb A_k^1.
\]
For each example, state the corresponding feature of its coordinate ring.
