% VI/05 pedagogical exercises.
% These are intentionally new or substantially redesigned.
% Legacy problems are preserved separately as self-contained dossiers.

\subsection*{Exercise VI.05.1 --- Three quick irreducibility tests}
\label{ex:vi05-1}
Assume \(k\) is algebraically closed.  Decide which of
\[
V(y-x^2),\qquad V(xy),\qquad V(xy-1)\subseteq\mathbb A_k^2
\]
are irreducible.  Justify each answer using the coordinate ring.

\subsection*{Exercise VI.05.2 --- Dense principal opens}
\label{ex:vi05-2}
Let \(X\) be irreducible and let \(f\in k[X]\).  If \(D_X(f)\neq\varnothing\), prove
that \(D_X(f)\) is dense in \(X\).

\subsection*{Exercise VI.05.3 --- Polynomial identities from dense subsets}
\label{ex:vi05-3}
Let \(X\) be irreducible and \(U\subseteq X\) a nonempty open subset.  If
\(f,g\in k[X]\) agree on \(U\), prove that \(f=g\) in \(k[X]\).

\subsection*{Exercise VI.05.4 --- A parametrized cusp}
\label{ex:vi05-4}
Assume \(k\) is infinite.  Let
\[
C=V(y^2-x^3)\subseteq\mathbb A_k^2.
\]
Use \(t\mapsto(t^2,t^3)\) to show that \(C\) is irreducible.

\subsection*{Exercise VI.05.5 --- Reading components from a factorization}
\label{ex:vi05-5}
Assume \(\operatorname{char}k\neq2\).  Determine the irreducible components of
\[
X=V\bigl(xy(x-y)(x+y)\bigr)\subseteq\mathbb A_k^2.
\]
Is \(X\) connected?

\subsection*{Exercise VI.05.6 --- A reducible but connected surface}
\label{ex:vi05-6}
For \(X=V(xy)\subseteq\mathbb A_k^3\), describe the irreducible components, their
intersection, and the component graph.  Deduce that \(X\) is connected but reducible.

\subsection*{Exercise VI.05.7 --- Zero divisors versus idempotents}
\label{ex:vi05-7}
In \(A=k[x,y]/(xy)\), exhibit nonzero zero divisors.  Assuming \(k\) algebraically closed,
prove that \(A\) has no nontrivial idempotents.  Explain the two different geometric
phenomena detected by these facts.

\subsection*{Exercise VI.05.8 --- Two disjoint lines}
\label{ex:vi05-8}
Let \(X=V(x(x-1))\subseteq\mathbb A_k^2\).  Use the Chinese remainder theorem to prove
\[
k[X]\cong k[y]\times k[y].
\]
Construct an explicit nontrivial idempotent and identify the corresponding clopen pieces.

\subsection*{Exercise VI.05.9 --- A chain of components}
\label{ex:vi05-9}
Let \(X=X_1\cup X_2\cup X_3\cup X_4\), where each \(X_i\) is irreducible and
\[
X_1\cap X_2\neq\varnothing,\quad
X_2\cap X_3\neq\varnothing,\quad
X_3\cap X_4\neq\varnothing.
\]
Prove that \(X\) is connected.

\subsection*{Exercise VI.05.10 --- Continuous images of irreducible spaces}
\label{ex:vi05-10}
Prove that the continuous image of an irreducible topological space is irreducible.  Deduce
that if \(\phi:X\to Y\) is a morphism and \(X\) is irreducible, then
\(\overline{\phi(X)}\) is irreducible.

\subsection*{Exercise VI.05.11 --- Removing a proper closed set}
\label{ex:vi05-11}
Let \(X\) be irreducible and \(Z\subsetneq X\) closed.  Prove that \(X\setminus Z\) is
nonempty, irreducible, and dense in \(X\).

\subsection*{Exercise VI.05.12 --- Compare four one-piece behaviors}
\label{ex:vi05-12}
Assume \(k\) is algebraically closed and choose three distinct elements \(a,b,c\in k\).
Determine irreducibility and connectedness for
\[
\mathbb A_k^1,\qquad V(xy),\qquad V(x(x-1)),\qquad \{a,b,c\}.
\]
For each, state the corresponding feature of its coordinate ring.

\subsection*{Exercise VI.05.13 --- Minimal primes of a plane plus a line}
\label{ex:vi05-13}
Show
\[
(xy,xz)=(x)\cap(y,z)\subseteq k[x,y,z].
\]
Deduce the irreducible components of \(V(xy,xz)\), and decide whether the set is connected.

\subsection*{Exercise VI.05.14 --- Three disjoint vertical lines}
\label{ex:vi05-14}
Let \(a\in k\setminus\{0,1\}\) and
\[
X=V\bigl(x(x-1)(x-a)\bigr)\subseteq\mathbb A_k^2.
\]
Describe its irreducible and connected components.  Show that \(k[X]\cong k[y]^3\).

\subsection*{Exercise VI.05.15 --- Continuous images of connected spaces}
\label{ex:vi05-15}
Prove that a continuous image of a connected space is connected.  Deduce that the image of
a connected affine algebraic set under a morphism is connected in its image topology.

\subsection*{Exercise VI.05.16 --- Order reversal and components}
\label{ex:vi05-16}
Assume \(k\) algebraically closed and
\[
I=\mathfrak p_1\cap\cdots\cap\mathfrak p_r
\]
is an irredundant intersection of distinct prime ideals, none containing another.  Prove
that the irreducible components of \(V(I)\) are exactly \(V(\mathfrak p_i)\).

\subsection*{Exercise VI.05.17 --- The graph of the legacy three-component example}
\label{ex:vi05-17}
For
\[
Y=V(x^2-yz,\;xz-x)\subseteq\mathbb A_k^3,
\]
use the three components identified in Problem VI.05.P3 to compute all pairwise
intersections.  Draw the component graph and decide whether \(Y\) is connected.

\subsection*{Exercise VI.05.18 --- Finite intersections of dense opens}
\label{ex:vi05-18}
Let \(X\) be irreducible and let \(U_1,\dots,U_r\subseteq X\) be nonempty open subsets.
Prove that
\[
U_1\cap\cdots\cap U_r
\]
is nonempty and dense in \(X\).  Apply this to principal opens
\(D_X(f_1),\dots,D_X(f_r)\).
