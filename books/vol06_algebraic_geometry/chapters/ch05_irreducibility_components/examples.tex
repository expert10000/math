% VI/05 worked examples.

\subsection*{Example 1: Affine space is irreducible}

Assume \(k\) is infinite.  Since
\[
I(\mathbb A_k^n)=(0)
\]
and
\[
k[x_1,\dots,x_n]
\]
is an integral domain, \cref{thm:vi05-prime-criterion} gives
\[
\mathbb A_k^n
\]
irreducible.

Thus any two nonempty Zariski open subsets of affine space intersect.

\subsection*{Example 2: A parabola}

Let
\[
P=V(y-x^2)\subseteq\mathbb A_k^2.
\]
Its coordinate ring is
\[
k[P]\cong k[t],
\]
a domain.  Hence \(P\) is irreducible.

This recovers geometrically the fact that the parabola is one algebraic component.

\subsection*{Example 3: Crossing axes}

Let
\[
X=V(xy)\subseteq\mathbb A_k^2.
\]
Then
\[
X=V(x)\cup V(y),
\]
so \(X\) is reducible.

However,
\[
V(x)\cap V(y)=\{(0,0)\},
\]
so the component-intersection graph has two vertices joined by an edge.  Hence \(X\) is
connected.

Algebraically,
\[
k[X]=k[x,y]/(xy)
\]
has zero divisors but no nontrivial idempotent.

\subsection*{Example 4: Two parallel components}

Let
\[
X=V(x(x-1))\subseteq\mathbb A_k^2.
\]
Then
\[
X=V(x)\cup V(x-1),
\]
the union of two disjoint vertical lines.

The ideals
\[
(x)
\qquad\text{and}\qquad
(x-1)
\]
are comaximal, so the Chinese remainder theorem gives
\[
k[X]
\cong
k[x,y]/(x(x-1))
\cong
k[y]\times k[y].
\]
Thus \(X\) is disconnected.  The class of \(x\) is idempotent:
\[
x^2-x=x(x-1)=0
\]
in \(k[X]\).

\subsection*{Example 5: A plane plus a line}

Consider
\[
X=V(xy,xz)\subseteq\mathbb A_k^3.
\]
Factoring the equations gives
\[
X=V(x)\cup V(y,z).
\]
Thus the components are the plane \(x=0\) and the \(x\)-axis.

They meet at the origin, so \(X\) is connected but reducible.

\subsection*{Example 6: Three lines through the origin}

Let
\[
X=V\bigl(xy(x-y)\bigr)\subseteq\mathbb A_k^2.
\]
Then
\[
X=V(x)\cup V(y)\cup V(x-y).
\]
Each component meets the other two at the origin, so the component-intersection graph is a
triangle.  Hence \(X\) is connected.

\subsection*{Example 7: A chain of components}

Suppose
\[
X=X_1\cup X_2\cup X_3
\]
with
\[
X_1\cap X_2\neq\varnothing,\qquad
X_2\cap X_3\neq\varnothing,\qquad
X_1\cap X_3=\varnothing.
\]
The component-intersection graph is a path
\[
X_1---X_2---X_3.
\]
It is connected, so \(X\) is connected even though not every pair of components meets.

\subsection*{Example 8: A disconnected finite algebraic set}

Let
\[
X=\{a,b\}\subseteq\mathbb A_k^1,
\qquad a\neq b.
\]
Then
\[
k[X]\cong k\times k.
\]
The element
\[
e=(1,0)
\]
is a nontrivial idempotent.

The two singletons are simultaneously open and closed, so
\[
X=\{a\}\sqcup\{b\}
\]
is disconnected.
