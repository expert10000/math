% VI40-DETAILED-PROBLEM-01
\begin{solution}
Let \(t\) be the affine coordinate on \(U_0=\PP^1\setminus\{\infty\}\), and let \(u=1/t\) on \(U_\infty=\PP^1\setminus\{0\}\).  The divisor \([a]-[\infty]\) is principal, with global rational equation
\[
f=t-a.
\]
Thus one may take
\[
f_0=t-a\quad\text{on }U_0,
\qquad
f_\infty=\frac{1-au}{u}\quad\text{on }U_\infty.
\]
These are the same rational function written in the two coordinates, so on the overlap their ratio is \(1\), certainly a regular unit.  At \(a\), \(t-a\) is a uniformizer, giving coefficient \(+1\).  At infinity, \(u\) is a uniformizer and \((1-au)/u\) has order \(-1\), because \(1-au\) is a unit at \(u=0\).  Hence the associated Weil divisor is \([a]-[\infty]\), as required.
\end{solution}

% VI40-DETAILED-PROBLEM-02
\begin{solution}
The Cartier divisor \(2V(x)\) is represented by the regular non-zero-divisor \(x^2\).  Hence the corresponding closed subscheme is
\[
D=V(x^2)\subset\AA^2_k,
\qquad
\IIcal_D=(x^2).
\]
Its reduced subscheme is \(V(x)\), since \(\sqrt{(x^2)}=(x)\), but the scheme is nonreduced.  Locally along the line \(x=0\), the quotient
\[
k[x,y]/(x^2)
\]
contains the nonzero nilpotent class of \(x\).  Thus \(2V(x)\) is the first infinitesimal thickening of the line in the normal direction, and its Cartier multiplicity is encoded scheme-theoretically by the square of the defining equation.
\end{solution}

% VI40-DETAILED-PROBLEM-03
\begin{solution}
The effective Cartier divisor \(V(t)\subset\AA^1_t\) has equation \(t\).  Pulling back along
\[
g:\AA^1_u\to\AA^1_t,
\qquad
t=u^3(u-1)^2,
\]
gives the nonzero polynomial
\[
g^*t=u^3(u-1)^2.
\]
Since \(k[u]\) is a domain, this is a non-zero-divisor, so the pullback is defined and effective.  Factorization in the UFD \(k[u]\) gives
\[
g^*V(t)=3V(u)+2V(u-1)=3[0]+2[1].
\]
Equivalently, the orders of the pulled-back equation are \(3\) at \(u=0\) and \(2\) at \(u=1\), and zero at every other closed point.
\end{solution}

% VI40-DETAILED-PROBLEM-04
\begin{solution}
Let \(X=\AA^2_{t,u}\), let \(D=V(t)\), and let \(i:Y=V(t)\hookrightarrow X\).  The local Cartier equation for \(D\) is \(t\).  In \(\OO_Y\) its pullback is
\[
i^\#(t)=0.
\]
But a Cartier equation must determine an element of \(\mathcal K_Y^\times\), the unit group of the total quotient sheaf.  Zero is not invertible, so the defining condition for Cartier pullback fails.  This is precisely the geometric obstruction: the whole source \(Y\) is contained in \(|D|\).  Thus one cannot interpret the scheme-theoretic self-intersection here merely as a Cartier pullback without additional intersection-theoretic machinery.
\end{solution}

% VI40-DETAILED-PROBLEM-05
\begin{solution}
Let \(X=\Spec A\), with \(A\) a UFD, and write
\[
f=\frac{p_1^{a_1}\cdots p_r^{a_r}}
        {q_1^{b_1}\cdots q_s^{b_s}},
\]
where the displayed irreducibles are pairwise nonassociate after cancellation.  Every height-one prime of a UFD is generated by an irreducible element.  At the prime \((p_i)\), all other displayed factors are units in \(A_{(p_i)}\), so
\[
\operatorname{ord}_{V(p_i)}(f)=a_i.
\]
Likewise,
\[
\operatorname{ord}_{V(q_j)}(f)=-b_j.
\]
All remaining height-one orders vanish.  Hence
\[
[(f)_{\mathrm{Cart}}]_{\mathrm W}
=\operatorname{div}(f)
=\sum_{i=1}^r a_i[V(p_i)]-
  \sum_{j=1}^s b_j[V(q_j)].
\]
This is exactly the compatibility of the Cartier-to-Weil map with principal divisors.
\end{solution}

% VI40-DETAILED-PROBLEM-06
\begin{solution}
Let
\[
A=k[x,y,z]/(xy-z^2),\qquad
\mathfrak m=(x,y,z),\qquad
I=(x,z)A_{\mathfrak m}.
\]
If the prime divisor \(P=V(x,z)\) were Cartier at the vertex, then \(I\) would be principal.  By Nakayama's lemma, the minimal number of generators of \(I\) is
\[
\mu(I)=\dim_k I/\mathfrak m I.
\]
The relation \(xy-z^2\) has degree two.  Therefore modulo \(\mathfrak m I\), which is generated by terms of degree at least two, the classes of \(x\) and \(z\) remain linearly independent.  Hence
\[
\dim_k I/\mathfrak m I=2,
\]
so \(I\) cannot be principal.  Thus \(P\) is not Cartier at the vertex.

On the other hand, VI/39 computed the height-one valuation along \(P\).  Since \(y\) is a unit in \(A_P\) and \(xy=z^2\),
\[
x=\frac{z^2}{y},
\]
so \(\operatorname{ord}_P(x)=2\).  The zero locus of \(x\) has only the prime component \(P\), because
\[
A/(x)\cong k[y,z]/(z^2)
\]
has the unique minimal prime \((z)\).  Consequently
\[
\operatorname{div}(x)=2P.
\]
Thus \(2P\) is principal Cartier even though \(P\) itself is not Cartier.  This is the local phenomenon that VI/41 will record in the divisor class group.
\end{solution}

% VI40-DETAILED-PROBLEM-07
\begin{solution}
Let \(D\) and \(E\) have local equations \(f_i\) and \(g_i\) on a common cover.  Put
\[
e_i=f_i^{-1}\quad\text{and}\quad h_i=g_i^{-1}.
\]
These are local generators of \(\OO(D)\) and \(\OO(E)\).  On overlaps,
\[
e_i=\frac{f_j}{f_i}e_j,
\qquad
h_i=\frac{g_j}{g_i}h_j.
\]
Therefore
\[
e_i\otimes h_i
=\frac{f_jg_j}{f_i g_i}(e_j\otimes h_j).
\]
But \((f_i g_i)^{-1}\) is the local generator of \(\OO(D+E)\), and its transition factor is exactly
\[
\frac{f_jg_j}{f_i g_i}.
\]
Hence the map defined locally by multiplication,
\[
e_i\otimes h_i\longmapsto(f_i g_i)^{-1},
\]
respects transition functions and glues to a canonical isomorphism
\[
\OO_X(D)\otimes\OO_X(E)\xrightarrow{\sim}\OO_X(D+E).
\]
\end{solution}

% VI40-DETAILED-PROBLEM-08
\begin{solution}
Under the \(d\)-uple Veronese map
\[
\nu_d:\PP^n\longrightarrow\PP^N,
\]
a homogeneous coordinate \(Y_\alpha\) on the target corresponds to a degree-\(d\) monomial
\[
M_\alpha(x)=x_0^{\alpha_0}\cdots x_n^{\alpha_n},
\qquad \sum_i\alpha_i=d.
\]
The coordinate hyperplane \(H_\alpha=V_+(Y_\alpha)\) therefore pulls back to
\[
\nu_d^*H_\alpha=V_+(M_\alpha)
=\sum_{i=0}^n \alpha_i H_i
\]
as an effective Cartier divisor, where \(H_i=V_+(x_i)\).  Its total degree is \(d\).

The divisor line-bundle construction is compatible with defined pullback, so
\[
\nu_d^*\OO_{\PP^N}(1)
\cong\OO_{\PP^n}(\nu_d^*H_\alpha)
\cong\OO_{\PP^n}(d).
\]
This identifies both the multiplicity of the pulled-back hyperplane and the standard Veronese twisting formula.
\end{solution}

% VI40-DETAILED-PROBLEM-09
\begin{solution}
Choose a trivializing cover \(U_i\) of \(\mathcal L\) and generators \(e_i\).  Write
\[
s=f_i e_i,
\qquad f_i\in\OO_X(U_i).
\]
By hypothesis each \(f_i\) is a non-zero-divisor.  If \(e_i=u_{ij}e_j\) on \(U_i\cap U_j\), with \(u_{ij}\in\OO_X^\times\), then equality of the section gives
\[
f_j=u_{ij}f_i.
\]
Thus \(f_i/f_j=u_{ij}^{-1}\) is a unit, so the \(f_i\) are compatible Cartier local equations.  Because they are regular non-zero-divisors, they define an effective Cartier divisor
\[
D=Z(s).
\]
The local generator of \(\OO_X(D)\) is \(f_i^{-1}\).  Sending \(f_i^{-1}\) to \(e_i\) glues to an isomorphism
\[
\OO_X(D)\xrightarrow{\sim}\mathcal L,
\]
under which the canonical section \(1=f_i(f_i^{-1})\) corresponds exactly to \(s=f_i e_i\).
\end{solution}

% VI40-DETAILED-PROBLEM-10
\begin{solution}
Let \(D\) be represented by local equations \(f_i\) on \(U_i\).  Then
\[
\OO_X(D)|_{U_i}=f_i^{-1}\OO_{U_i}.
\]
Restricting to \(U\cap U_i\) gives
\[
j^*\OO_X(D)|_{U\cap U_i}
=f_i^{-1}\OO_{U\cap U_i}.
\]
But the restricted Cartier divisor \(D|_U\) is represented by the same rational equations restricted to \(U\cap U_i\), so by definition
\[
\OO_U(D|_U)|_{U\cap U_i}=f_i^{-1}\OO_{U\cap U_i}.
\]
The transition functions are unchanged.  Hence the local identities glue canonically to
\[
j^*\OO_X(D)\cong\OO_U(D|_U).
\]
\end{solution}

% VI40-DETAILED-PROBLEM-11
\begin{solution}
The assertion is local.  Let \(X=\Spec A\) near a point of \(D\), and let
\[
D=V(f)
\]
with \(f\in A\) a non-zero-divisor.  Let \(\mathfrak p\) be a minimal prime of an irreducible component of \(X\) that meets the component of \(D\) under consideration.  Because \(f\) is a non-zero-divisor, it lies in no associated prime of \(A\), in particular in no minimal prime, so its image in the domain-like component ring \(A/\mathfrak p\) is nonzero.

If \(\mathfrak q\) is minimal over \((\mathfrak p,f)\), then the principal ideal theorem applied in \(A/\mathfrak p\) gives
\[
\operatorname{ht}(\mathfrak q/\mathfrak p)\le1.
\]
The height cannot be zero, because \(f\notin\mathfrak p\).  Hence
\[
\operatorname{ht}(\mathfrak q/\mathfrak p)=1.
\]
Thus every irreducible component of the effective Cartier divisor has pure codimension one inside the irreducible component of \(X\) on which it lies.
\end{solution}

% VI40-DETAILED-PROBLEM-12
\begin{solution}
The line bundle \(\OO_X(D)\) forgets the chosen rational trivialization.  Indeed, for any global rational unit \(f\),
\[
\OO_X(D+(f))
\cong\OO_X(D)\otimes\OO_X((f))
\cong\OO_X(D),
\]
because a principal divisor has trivial associated line bundle.  Therefore every divisor in the principal-equivalence class of \(D\) produces an isomorphic line bundle.

To recover the specific divisor one must retain a rational section.  For the pair \((\OO_X(D),s_D)\), the canonical rational section \(s_D=1\) has divisor exactly \(D\).  Replacing \(s_D\) by \(f s_D\) changes the divisor to
\[
D+(f).
\]
Thus the line bundle alone remembers only the class that will become Picard data, whereas the line bundle together with its rational trivialization remembers the actual Cartier divisor.
\end{solution}
