% Detailed solutions for the five canonical VI/40 challenges.

% VI40-DETAILED-CHALLENGE-01
\begin{solution}
Let \(X=\Spec A\) with \(A\) a normal Noetherian domain and fraction field \(K\).  We first justify the height-one intersection formula
\[
A=\bigcap_{\operatorname{ht}\mathfrak p=1}A_{\mathfrak p}\subset K.
\]
A normal Noetherian domain satisfies Serre's conditions \(R_1\) and \(S_2\).  Let \(f\in K\) lie in every \(A_{\mathfrak p}\) with \(\operatorname{ht}\mathfrak p=1\).  There is a largest open set \(U\subseteq X\) on which \(f\) is regular.  By hypothesis every codimension-one point belongs to \(U\), so \(X\setminus U\) has codimension at least two.  The \(S_2\) extension property for a normal Noetherian scheme gives
\[
\Gamma(X,\OO_X)\xrightarrow{\sim}\Gamma(U,\OO_U).
\]
Hence \(f\) extends to a global regular function, i.e. \(f\in A\).  This proves the intersection identity; the reverse inclusion is immediate.

Now let a Cartier divisor \(D\) map to zero as a Weil divisor.  On an affine open \(U=\Spec A\), choose a Cartier equation \(f\in K^\times\).  Vanishing of the Weil image means
\[
\operatorname{ord}_{\mathfrak p}(f)=0
\qquad
\text{for every height-one prime }\mathfrak p.
\]
Thus both \(f\) and \(f^{-1}\) lie in every DVR \(A_{\mathfrak p}\).  By the intersection formula,
\[
f,f^{-1}\in A,
\]
so \(f\in A^\times\).  Therefore the Cartier divisor is zero on \(U\).  Since this holds on an affine cover, \(D=0\) globally.  Hence the Cartier-to-Weil homomorphism is injective.
\end{solution}

% VI40-DETAILED-CHALLENGE-02
\begin{solution}
Let \(g:Y\to X\) be a morphism with \(Y\) locally Noetherian, and let \(D\subset X\) be effective Cartier.  Work locally on \(X\), where \(D=V(f)\) and \(f\) is a non-zero-divisor.  The pullback closed subscheme on \(g^{-1}(U)\) is cut out by
\[
h=g^\#(f)\in\OO_Y(g^{-1}(U)).
\]
For a Noetherian ring, an element is a zero-divisor exactly when it lies in an associated prime.  Equivalently, a section is a non-zero-divisor at all stalks exactly when it avoids all associated points.

By hypothesis no associated point \(\eta\in\operatorname{Ass}(Y)\) maps into \(|D|\).  Since \(|D|=V(f)\), this says that the image of \(f\) in \(\OO_{Y,\eta}\) is not contained in the associated prime corresponding to \(\eta\).  Hence \(h\) lies in no associated prime of \(Y\), so multiplication by \(h\) is injective.  Therefore \(h\) is a non-zero-divisor.

On overlaps, two equations for \(D\) differ by a regular unit, and pullback of that unit remains a regular unit.  Thus the pulled-back non-zero-divisors glue to compatible Cartier equations.  Consequently \(g^*D\) is defined and is again an effective Cartier divisor.  The hypothesis is sharp in the expected sense: if an associated point maps into \(|D|\), the pulled-back equation may become a zero-divisor or even zero.
\end{solution}

% VI40-DETAILED-CHALLENGE-03
\begin{solution}
Put
\[
A=k[x,y,z]/(xy-z^2),\qquad
\mathfrak m=(x,y,z),\qquad
P=(x,z).
\]
At the vertex, consider \(I=P A_{\mathfrak m}\).  If \(I\) were principal, then Nakayama's lemma would give
\[
\dim_k I/\mathfrak m I=1.
\]
But \(I\) is generated by \(x,z\), and \(\mathfrak m I\) consists of classes represented by terms of degree at least two.  The defining relation \(xy-z^2\) is also quadratic, so it gives no nontrivial linear relation between the images of \(x\) and \(z\) in \(I/\mathfrak m I\).  Therefore
\[
\dim_k I/\mathfrak m I=2.
\]
Thus \(I\) is not principal and \(P\) is not Cartier at the vertex.

For the Weil multiplicity, localize at the height-one prime \(P\).  The element \(y\notin P\) is a unit, and
\[
x=\frac{z^2}{y}.
\]
The normal surface has DVR \(A_P\), with \(z\) a uniformizer, hence
\[
\operatorname{ord}_P(z)=1,
\qquad
\operatorname{ord}_P(x)=2.
\]
Moreover \(A/(x)\cong k[y,z]/(z^2)\) has only one minimal prime, namely \((z)\), corresponding to \(P\).  Therefore
\[
\operatorname{div}(x)=2P.
\]
So the class of \(P\) is killed by \(2\) in the divisor class group.  Since \(P\) is not Cartier/principal at the vertex, this is the first concrete evidence for the nontrivial \(2\)-torsion class that VI/41 will compute.
\end{solution}

% VI40-DETAILED-CHALLENGE-04
\begin{solution}
Let \(F\in k[x_0,\ldots,x_n]\) be nonzero homogeneous of degree \(d\).  On the standard chart \(U_i=D_+(x_i)\), define
\[
f_i=\frac{F}{x_i^d}.
\]
This is a regular function on \(U_i\).  Since \(\PP^n_k\) is integral and \(F\neq0\), it is a non-zero-divisor.  On an overlap,
\[
\frac{f_i}{f_j}
=\left(\frac{x_j}{x_i}\right)^d,
\]
a regular unit.  Thus the \(f_i\) define an effective Cartier divisor whose zero subscheme is exactly \(V_+(F)\).

The associated divisor sheaf is locally generated by
\[
f_i^{-1}=\frac{x_i^d}{F}.
\]
On \(U_i\cap U_j\), these generators satisfy
\[
\frac{f_i^{-1}}{f_j^{-1}}
=\left(\frac{x_i}{x_j}\right)^d,
\]
which are the standard transition functions of \(\OO_{\PP^n}(d)\).  Hence
\[
\OO_{\PP^n}(V_+(F))\cong\OO_{\PP^n}(d).
\]

If
\[
F=c\prod_{r=1}^m F_r^{e_r}
\]
with distinct irreducible homogeneous factors, then the local equation factors in the same way.  Therefore the Cartier-to-Weil image is
\[
[V_+(F)]_{\mathrm W}=\sum_{r=1}^m e_r[V_+(F_r)].
\]
Thus the scheme-theoretic divisor remembers the same factor multiplicities as the corresponding Weil divisor.
\end{solution}

% VI40-DETAILED-CHALLENGE-05
\begin{solution}
Assume \(X\) is integral with function field \(K(X)\).  From a Cartier divisor \(D\) with local equations \(f_i\), form the invertible sheaf
\[
\mathcal L=\OO_X(D),
\qquad
\mathcal L|_{U_i}=f_i^{-1}\OO_{U_i}\subset K(X).
\]
The rational section \(s=1\) generates \(\mathcal L\otimes K(X)\).  Thus \(D\) determines a pair \((\mathcal L,s)\).

Conversely, let \(\mathcal L\) be invertible and let \(s\) be a nonzero rational section generating \(\mathcal L\otimes K(X)\).  Choose local bases \(e_i\) and write
\[
s=f_i e_i,
\qquad f_i\in K(X)^\times.
\]
If \(e_i=u_{ij}e_j\), with \(u_{ij}\in\OO_X^\times\), then
\[
f_j=u_{ij}f_i,
\]
so
\[
\frac{f_i}{f_j}=u_{ij}^{-1}\in\OO_X^\times.
\]
Hence the \(f_i\) define a Cartier divisor \(D_s\).  The map \(f_i^{-1}\mapsto e_i\) identifies \(\OO_X(D_s)\) with \(\mathcal L\) and carries the canonical rational section \(1\) to \(s\).  These two constructions are inverse up to isomorphism of pairs.

Finally, for \(f\in K(X)^\times\), replacing \(s\) by \(fs\) replaces every local coefficient \(f_i\) by \(ff_i\).  Therefore
\[
\operatorname{div}(fs)
=\operatorname{div}(s)+(f)_{\mathrm{Cart}}.
\]
Thus multiplying the rational trivialization changes the Cartier divisor by a principal divisor, exactly explaining why forgetting the rational section retains only the line-bundle isomorphism class.
\end{solution}
