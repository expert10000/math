% Detailed solutions for the 12 canonical VI/41 dossier problems.

% VI41-PROBLEM-SOLUTION-01
\subsection*{Solution to Problem VI.41.01 — Class group of a localization}
\begin{solution}
The localization theorem gives
\[
\Cl(A_f)\cong \Cl(A)\Big/\left\langle[\mathfrak p]:
\height\mathfrak p=1,\ f\in\mathfrak p\right\rangle.
\]
Thus \([D_1]\) certainly dies. The divisor \(D_2\) itself survives because its height-one
prime does not contain \(f\). However, one must still account for every other
height-one prime containing \(f\): its class could impose an additional relation
involving \([D_2]\). If the full killed subgroup is exactly \(\mathbb Z[D_1]\), then
\[
\Cl(A_f)\cong \mathbb Z/3[D_2].
\]
So localization kills classes through the complete set of codimension-one components
of \(V(f)\), not merely through a preferred choice of class-group generators.
\end{solution}

% VI41-PROBLEM-SOLUTION-02
\subsection*{Solution to Problem VI.41.02 — The cone relation from \(x/z\)}
\begin{solution}
Using
\[
\operatorname{div}(x)=2L,\qquad \operatorname{div}(z)=L+M,
\]
we get
\[
\operatorname{div}(x/z)=2L-(L+M)=L-M.
\]
Hence \(L\sim M\) and \([L]=[M]\). Since \(\operatorname{div}(z)\) also yields
\([L]+[M]=0\), it follows that \(2[L]=0\).
\end{solution}

% VI41-PROBLEM-SOLUTION-03
\subsection*{Solution to Problem VI.41.03 — Why \(D(x)\) forgets the cone class}
\begin{solution}
Algebraically,
\[
A_x\cong k[x,x^{-1},z],
\]
a UFD, so \(\Cl(A_x)=0\). Geometrically, \(X\setminus D(x)=V(x)\) has the unique
codimension-one component \(L=V(x,z)\). Therefore
\[
\Cl(D(x))\cong\Cl(X)/\langle[L]\rangle.
\]
Since \([L]\) generates \(\Cl(X)\), the quotient is zero. The UFD computation and the
deleted-divisor computation are the same localization phenomenon.
\end{solution}

% VI41-PROBLEM-SOLUTION-04
\subsection*{Solution to Problem VI.41.04 — Why deleting the vertex does not forget the cone class}
\begin{solution}
The vertex is codimension two in the normal surface. Hence deleting it removes no
prime divisor. Intersection and closure identify \(\Div(X)\) and \(\Div(U)\), while
the function field and all height-one local rings are unchanged. Principal divisors
therefore correspond, giving
\[
\Cl(X)\xrightarrow{\sim}\Cl(U).
\]
Thus the order-two class survives on the smooth punctured cone.
\end{solution}

% VI41-PROBLEM-SOLUTION-05
\subsection*{Solution to Problem VI.41.05 — A factoriality test}
\begin{solution}
Since \(A_f\) is a UFD, \(\Cl(A_f)=0\). The localization sequence shows that the
classes \([\mathfrak p_1],\ldots,[\mathfrak p_r]\) generate \(\Cl(A)\). Define
\[
\Phi:\mathbb Z^r\to\Cl(A),\qquad e_i\mapsto[\mathfrak p_i].
\]
Its kernel consists of the vectors
\[
(v_{\mathfrak p_1}(g),\ldots,v_{\mathfrak p_r}(g))
\]
for rational functions \(g\) whose principal divisors are supported on these primes.
Therefore
\[
\Cl(A)\cong\mathbb Z^r/R,
\]
where \(R\) is the lattice generated by those valuation vectors. Concrete computations
amount to finding generators for \(R\).
\end{solution}

% VI41-PROBLEM-SOLUTION-06
\subsection*{Solution to Problem VI.41.06 — Hyperplanes and rational functions}
\begin{solution}
The rational function \(x_0/x_1\) is well defined on projective space because numerator
and denominator have the same degree. It has a simple zero along
\(H_0=V_+(x_0)\), a simple pole along \(H_1=V_+(x_1)\), and no other
codimension-one contribution. Hence
\[
\operatorname{div}(x_0/x_1)=H_0-H_1,
\]
so \(H_0\sim H_1\).
\end{solution}

% VI41-PROBLEM-SOLUTION-07
\subsection*{Solution to Problem VI.41.07 — A degree-\(d\) hypersurface}
\begin{solution}
As a Weil divisor,
\[
V_+(F)=\sum_i e_iV_+(F_i).
\]
Each irreducible hypersurface satisfies
\[
[V_+(F_i)]=(\deg F_i)[H].
\]
Consequently
\[
[V_+(F)]=\sum_i e_i\deg(F_i)[H]=(\deg F)[H].
\]
Thus factorization controls the prime decomposition and multiplicities, while total
degree is the resulting class-group coordinate.
\end{solution}

% VI41-PROBLEM-SOLUTION-08
\subsection*{Solution to Problem VI.41.08 — Local versus global factoriality}
\begin{solution}
Smoothness makes every local ring regular and hence a UFD. Therefore every Weil divisor
is locally principal, i.e. Cartier. But global principalness requires one rational
function whose divisor equals the given divisor everywhere. Local equations may glue
only up to units and need not come from one global equation. Thus smoothness removes
the local obstruction without forcing \(\Cl(X)\) to vanish; for example
\[
\Cl(\PP^n)\cong\mathbb Z.
\]
\end{solution}

% VI41-PROBLEM-SOLUTION-09
\subsection*{Solution to Problem VI.41.09 — Torsion and a Cartier multiple}
\begin{solution}
The equality \(m[D]=0\) means exactly that \(mD\) is principal:
\[
mD=\operatorname{div}(f)
\]
for some \(f\in K(X)^\times\). Every principal divisor is Cartier, so \(mD\) is
Cartier. Hence \(D\) is \(\mathbb Q\)-Cartier. The converse distinction matters:
a Cartier multiple need not be principal.
\end{solution}

% VI41-PROBLEM-SOLUTION-10
\subsection*{Solution to Problem VI.41.10 — Codimension-one deletion}
\begin{solution}
For \(X=\PP^n\), \(U=D_+(x_0)\cong\AA^n\), and
\(H=X\setminus U\), the exact sequence gives
\[
\mathbb Z[H]\longrightarrow\Cl(\PP^n)\longrightarrow\Cl(U)\longrightarrow0.
\]
Because \([H]\) generates \(\Cl(\PP^n)\cong\mathbb Z\), the first map is surjective.
Therefore
\[
\Cl(U)=0.
\]
\end{solution}

% VI41-PROBLEM-SOLUTION-11
\subsection*{Solution to Problem VI.41.11 — The local cone obstruction}
\begin{solution}
Let \(\mathfrak m=(x,y,z)\) and
\(I=(x,z)A_{\mathfrak m}\). The ideal \(I\) is a height-one prime. Modulo
\(\mathfrak mI\), the classes of \(x\) and \(z\) are linearly independent because
\(\mathfrak mI\) contains only terms of degree at least two. Hence
\[
\dim_k I/\mathfrak mI=2.
\]
By Nakayama's lemma \(I\) needs two generators and is not principal. In a UFD every
height-one prime is principal, so \(A_{\mathfrak m}\) is not a UFD. Thus the cone is
not locally factorial at the vertex.
\end{solution}

% VI41-PROBLEM-SOLUTION-12
\subsection*{Solution to Problem VI.41.12 — A divisor-class diagnostic}
\begin{solution}
For a height-one prime \(\mathfrak p\), the class \([\mathfrak p]\) vanishes exactly when
there is \(f\in\Frac(A)^\times\) with
\[
\operatorname{div}(f)=[\mathfrak p].
\]
If no such rational function exists, then \([\mathfrak p]\neq0\), so
\(\Cl(A)\neq0\). Since \(A\) is normal Noetherian, the UFD criterion implies that
\(A\) is not factorial.
\end{solution}
