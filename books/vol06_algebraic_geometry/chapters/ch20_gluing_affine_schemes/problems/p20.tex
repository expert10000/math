\begin{problem}[Affine atlas as localization data]\label{prob:vi20-p20}
Show an affine atlas may be refined so every transition is described locally by isomorphisms between localizations $A_f\cong B_g$.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This packages P2, VI/18, and VI/19 into a computational atlas formalism.

\paragraph{Useful ideas before starting.}
Refine every pairwise overlap by P2.

\paragraph{Guided hints.}
\begin{enumerate}
\item Refine every pairwise overlap by P2.
\end{enumerate}

% VI20_FULL_SOLUTION_REFINEMENT P20
\begin{solution}
Let $X=\bigcup_iU_i$ be an affine atlas with $U_i=\Spec A_i$.  For a pair $U_i,U_j$, apply VI.20.P2 to the intersection.  It yields a cover by opens $W_\lambda$ such that, for suitable $f_\lambda\in A_i$ and $g_\lambda\in A_j$,
\[
W_\lambda=D(f_\lambda)\subseteq U_i,
\qquad
W_\lambda=D(g_\lambda)\subseteq U_j.
\]
By the principal-open theorem,
\[
W_\lambda\cong\Spec(A_i)_{f_\lambda}
\qquad\text{and}\qquad
W_\lambda\cong\Spec(A_j)_{g_\lambda}.
\]
The identity transition on the geometric overlap therefore corresponds, by affine anti-equivalence, to an isomorphism of localized rings
\[
(A_j)_{g_\lambda}\xrightarrow{\sim}(A_i)_{f_\lambda}.
\]

Perform this refinement for every pair of charts.  On triple intersections, refine again by common distinguished opens if needed.  Since all transition maps come from restrictions of the identity on $X$, their localized ring isomorphisms satisfy the inverse and cocycle identities automatically.  Thus every affine atlas admits a refinement whose transition functions are locally explicit isomorphisms between localizations of the chart rings.
\end{solution}


\paragraph{What happened mathematically.}
General scheme atlases can be made algebraically explicit chart by chart.
