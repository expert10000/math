\begin{problem}[Common refinement of two affine atlases]\label{prob:vi20-p12}
Given two affine covers of one scheme, construct a common affine refinement using VI.20.P2.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This is a practical comparison theorem for atlases.

\paragraph{Useful ideas before starting.}
Intersect a chart from each cover, then cover the intersection by common distinguished opens.

\paragraph{Guided hints.}
\begin{enumerate}
\item Intersect a chart from each cover, then cover the intersection by common distinguished opens.
\end{enumerate}

% VI20_FULL_SOLUTION_REFINEMENT P12
\begin{solution}
Let $X=\bigcup_iU_i=\bigcup_jV_j$ be two affine covers, with $U_i=\Spec A_i$ and $V_j=\Spec B_j$.  The intersections $U_i\cap V_j$ form an open cover of $X$.  Fix a point $x\in U_i\cap V_j$.  By VI.20.P2 there is an open neighborhood $W_{ij,x}$ of $x$ that is distinguished in both charts:
\[
W_{ij,x}=D(f_{ij,x})\subseteq U_i,
\qquad
W_{ij,x}=D(g_{ij,x})\subseteq V_j.
\]
As $i,j,x$ vary, these opens cover $X$, so they form an affine refinement of both original covers.

Moreover, the principal-open theorem gives
\[
W_{ij,x}\cong\Spec(A_i)_{f_{ij,x}}
\cong\Spec(B_j)_{g_{ij,x}}.
\]
Hence the identity of the geometric open $W_{ij,x}$ corresponds to an isomorphism of localized rings
\[
(A_i)_{f_{ij,x}}\cong(B_j)_{g_{ij,x}}.
\]
Thus the common refinement not only compares the two atlases topologically; it expresses their transition functions entirely by localization isomorphisms.
\end{solution}


\paragraph{What happened mathematically.}
Any two affine atlases can be compared through localization charts.

\paragraph{Extensions.}
Translate transition maps to localized rings.
