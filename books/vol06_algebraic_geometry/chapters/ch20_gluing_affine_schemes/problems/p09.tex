\begin{problem}[Global sections of the inversion gluing]\label{prob:vi20-p09}
Compute the global sections of the two affine lines glued by inversion.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This illustrates the equalizer and shows why the result is not affine line.

\paragraph{Useful ideas before starting.}
A global section is $p(t)$ and $q(u)$ with $p(t)=q(t^{-1})$ in $k[t,t^{-1}]$.

\paragraph{Guided hints.}
\begin{enumerate}
\item A global section is $p(t)$ and $q(u)$ with $p(t)=q(t^{-1})$ in $k[t,t^{-1}]$.
\end{enumerate}

% VI20_FULL_SOLUTION_REFINEMENT P09
\begin{solution}
Let $X$ be the inversion gluing of $X_0=\Spec k[t]$ and $X_1=\Spec k[u]$.  A global regular function is a compatible pair
\[
p(t)\in k[t],\qquad q(u)\in k[u].
\]
On the overlap the transition is $u=t^{-1}$, so compatibility means
\[
p(t)=q(t^{-1})\qquad\text{in }k[t,t^{-1}].
\]
The left-hand side contains only nonnegative powers of $t$, while the right-hand side contains only nonpositive powers.  Therefore their common Laurent polynomial can have no nonzero term of positive or negative degree; it must be a constant $c\in k$.

Conversely, every constant gives the compatible pair $(c,c)$.  Hence
\[
\Gamma(X,\OO_X)\cong k.
\]
This computation also illustrates why the global ring alone cannot recover this non-affine two-chart scheme: affine reconstruction is valid only under the additional hypothesis that the source is affine.
\end{solution}


\paragraph{What happened mathematically.}
Overlap compatibility can drastically reduce global functions.

Generalize to other monomial transition maps.
