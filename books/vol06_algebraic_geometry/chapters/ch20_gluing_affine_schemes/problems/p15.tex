\begin{problem}[Equalizer for two affine charts]\label{prob:vi20-p15}
Let $X=U\cup V$ with $U=\Spec A$, $V=\Spec B$, and affine overlap $W=\Spec C$. Prove $\Gamma(X,\OO_X)$ is the equalizer of $A\times B\rightrightarrows C$.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This is the computational form of the sheaf axiom on a glued scheme.

\paragraph{Useful ideas before starting.}
A global section is a compatible pair.

\paragraph{Guided hints.}
\begin{enumerate}
\item A global section is a compatible pair.
\end{enumerate}

% VI20_FULL_SOLUTION_REFINEMENT P15
\begin{solution}
Let $X=U\cup V$, with
\[
U=\Spec A,\qquad V=\Spec B,\qquad W=U\cap V=\Spec C.
\]
Let $\rho_U:A\to C$ and $\rho_V:B\to C$ denote the two restriction maps, where the second is understood through the chosen identification of the overlap.

Restriction gives
\[
\Gamma(X,\OO_X)\longrightarrow A\times B,
\qquad s\longmapsto(s|_U,s|_V).
\]
The sheaf locality axiom makes this map injective.  If $s$ is global, its two restrictions agree on $W$, hence its image satisfies
\[
\rho_U(a)=\rho_V(b).
\]
Conversely, if $(a,b)\in A\times B$ satisfies this equality, then the corresponding sections on $U$ and $V$ agree on the overlap.  The sheaf gluing axiom produces a unique $s\in\Gamma(X,\OO_X)$ restricting to them.  Therefore
\[
\Gamma(X,\OO_X)
\cong
\{(a,b)\in A\times B:\rho_U(a)=\rho_V(b)\},
\]
which is precisely the equalizer of the two arrows $A\times B\rightrightarrows C$.
\end{solution}


\paragraph{What happened mathematically.}
Global functions are assembled by an algebraic equalizer.

\paragraph{Extensions.}
Apply to the doubled origin and inversion gluing.
