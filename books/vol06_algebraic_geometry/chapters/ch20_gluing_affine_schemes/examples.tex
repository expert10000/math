\begin{example}[Disjoint union]
If every $U_{ij}$ with $i\ne j$ is empty, the gluing is simply the disjoint union $\coprod_i X_i$ as a scheme.
\end{example}

\begin{example}[Gluing identical charts]
Gluing two copies of a scheme $X$ along all of $X$ by the identity produces a scheme canonically isomorphic to $X$; the quotient identifies corresponding points and the sheaves agree everywhere.
\end{example}

\begin{example}[Two affine lines along the punctured line]
Two copies of $\Spec k[t]$ glued along $D(t)$ by the identity give the doubled-origin line.  The nonzero loci are identified while the origins remain distinct.
\end{example}

\begin{example}[Two affine lines by inversion]
Glue $\Spec k[t]$ and $\Spec k[u]$ by $u=t^{-1}$ on the punctured lines.  The result has two affine charts whose overlap ring is $k[t,t^{-1}]$.
\end{example}

\begin{example}[A redundant chart]
If $X_3$ is an open subscheme already contained in $X_1$ and the transition maps are the obvious inclusions, adding $X_3$ to the gluing data does not change the resulting scheme up to unique isomorphism.
\end{example}

\begin{example}[Global sections on a union]
For $X=U\cup V$, a global function is a pair $(s_U,s_V)$ with equal restrictions to $U\cap V$.  This is the sheaf equalizer formula.
\end{example}

\begin{example}[Common distinguished neighborhood]
If $x\in U\cap V$ with $U=\Spec A$ and $V=\Spec B$, AG5-P2 gives an affine neighborhood $W$ of $x$ with $W=D(f)$ in $U$ and $W=D(g)$ in $V$.
\end{example}

\begin{example}[Empty overlap]
Gluing $X_1$ and $X_2$ along the empty open gives $X_1\amalg X_2$; the cocycle conditions are vacuous.
\end{example}

\begin{example}[Full overlap]
If $U_{12}=X_1$, $U_{21}=X_2$, and $\varphi_{12}$ is an isomorphism, the gluing identifies the two whole charts and yields either one up to canonical isomorphism.
\end{example}

\begin{example}[Gluing principal opens]
An isomorphism $A_f\cong B_g$ determines an isomorphism $D(g)\subseteq\Spec B\cong D(f)\subseteq\Spec A$ and hence a two-chart gluing.
\end{example}

\begin{example}[A chart stalk]
At a point represented by $x_i\in X_i$, the stalk of the glued sheaf is $\OO_{X_i,x_i}$; choosing another representative gives the same local ring through the transition isomorphism.
\end{example}

\begin{example}[Compatible maps glue]
If $f_i:X_i\to Y$ agree on overlaps after the transition identifications, there is exactly one $f:X\to Y$ restricting to all $f_i$.
\end{example}

\begin{example}[Cocycle on three copies]
For three charts, choosing $\varphi_{13}=\varphi_{23}\circ\varphi_{12}$ on the triple overlap ensures that identifying through chart 2 or directly through chart 3 gives the same equivalence class.
\end{example}

\begin{example}[Failure without cocycle]
If three transition maps disagree on a triple overlap, the attempted chart identifications are not coherent; one cannot simultaneously recover the prescribed pairwise identifications as embeddings into a single glued scheme.
\end{example}

\begin{example}[Restriction of the glued sheaf]
On $\psi_i(X_i)$ the compatible-family definition of $\OO_X$ recovers $\OO_{X_i}$ because every other component is forced from the $i$th component on overlaps.
\end{example}

\begin{example}[Affine cover reconstruction]
Given an affine open cover $X=\bigcup U_i$, gluing copies of the $U_i$ along their literal intersections using identity maps reconstructs $X$.
\end{example}

\begin{example}[Refinement does not change the gluing]
Replacing an overlap by a cover of smaller opens and using the restricted transition maps produces the same glued scheme once the pieces collectively encode the same identifications.
\end{example}

\begin{example}[Doubled-origin global functions]
For the doubled-origin line, compatible pairs of polynomials satisfy $p=q$ in $k[t,t^{-1}]$, hence $p=q$ already in $k[t]$; therefore $\Gamma(X,\OO_X)\cong k[t]$.
\end{example}

\begin{example}[Overlap ring by localization]
If $W=D(f)\subseteq\Spec A$, then $\Gamma(W,\OO)=A_f$.  A gluing isomorphism of such overlaps is therefore encoded by a ring isomorphism of localizations.
\end{example}

\begin{example}[Local property survives gluing]
Any property defined purely by the local rings and stable under isomorphism can be checked chartwise on a glued scheme, because every stalk comes from a chart stalk.
\end{example}
