\paragraph{VI.20.E1. Identity transition.}
\begin{solution}
Set $U_{ii}=X_i$ and $\varphi_{ii}=\operatorname{id}_{X_i}$.  Then every point $x\in X_i$ is related to itself, so reflexivity of the gluing relation is built into the same notation as the off-diagonal identifications.  The convention also makes the cocycle formula
\[
\varphi_{ik}=\varphi_{jk}\circ\varphi_{ij}
\]
valid when two of the indices coincide: for example, when $j=i$ it reduces to $\varphi_{ik}=\varphi_{ik}\circ\operatorname{id}$.  Thus identity transitions remove exceptional cases without changing the gluing data.
\end{solution}

\paragraph{VI.20.E2. Symmetry of the relation.}
\begin{solution}
Suppose $x\in X_i$ and $y\in X_j$ satisfy $x\sim y$.  By definition, $x\in U_{ij}$ and $y=\varphi_{ij}(x)$.  Since $\varphi_{ji}=\varphi_{ij}^{-1}$, we have $y\in U_{ji}$ and
\[
\varphi_{ji}(y)=\varphi_{ij}^{-1}(\varphi_{ij}(x))=x.
\]
Hence $y\sim x$.  This proves symmetry.
\end{solution}

\paragraph{VI.20.E3. Transitivity.}
\begin{solution}
Assume $x\in X_i$, $y\in X_j$, and $z\in X_k$ with $x\sim y$ and $y\sim z$.  Then
$y=\varphi_{ij}(x)$ and $z=\varphi_{jk}(y)$, and $x$ lies in the relevant triple overlap on which the cocycle is defined.  Therefore
\[
z=\varphi_{jk}(\varphi_{ij}(x))=\varphi_{ik}(x).
\]
Thus $x\sim z$.  Together with the identity and inverse conditions, the cocycle is exactly what turns the prescribed pairwise identifications into an equivalence relation.
\end{solution}

\paragraph{VI.20.E4. Open chart image.}
\begin{solution}
Let $\psi_i:X_i\to X$ be the quotient map.  By the definition of the quotient/final topology, a subset $W\subseteq X$ is open iff every $\psi_j^{-1}(W)$ is open in $X_j$.  For the chart image $W=\psi_i(X_i)$, a point of $X_j$ maps into $W$ precisely when it is identified with a point of $X_i$, hence precisely when it lies in $U_{ji}$.  Thus
\[
\psi_j^{-1}(\psi_i(X_i))=U_{ji},
\]
which is open by hypothesis for every $j$.  Hence $\psi_i(X_i)$ is open in $X$.
\end{solution}

\paragraph{VI.20.E5. Chart injectivity.}
\begin{solution}
Suppose $x,x'\in X_i$ have the same image in the quotient.  Any chain of transition identifications joining $x$ to $x'$ can be shortened, using the cocycle condition, to the direct transition from chart $i$ back to chart $i$.  That transition is $\varphi_{ii}=\operatorname{id}_{X_i}$.  Hence $x'=x$.  Therefore $\psi_i$ is injective.  Since the quotient topology restricts to the original topology on $X_i$, $\psi_i$ is a homeomorphism onto its open image.
\end{solution}

\paragraph{VI.20.E6. Overlap identity.}
\begin{solution}
If a point $p\in X$ belongs to both $\psi_i(X_i)$ and $\psi_j(X_j)$, choose representatives $x\in X_i$ and $y\in X_j$.  Equality of their quotient classes means $x\sim y$, so $x\in U_{ij}$ and $y=\varphi_{ij}(x)$.  Thus $p\in\psi_i(U_{ij})$.  Conversely, every $x\in U_{ij}$ has the same quotient class as $\varphi_{ij}(x)\in U_{ji}$, so its class lies in both chart images.  Hence
\[
\psi_i(X_i)\cap\psi_j(X_j)=\psi_i(U_{ij})=\psi_j(U_{ji}).
\]
\end{solution}

\paragraph{VI.20.E7. Componentwise restrictions.}
\begin{solution}
A section of the glued sheaf on $V$ is a family $(s_i)$ satisfying the transition equalities on every $V_i\cap U_{ij}$.  If $W\subseteq V$, restriction sends it to $(s_i|_{W_i})$.  Because $\varphi_{ij}^{\sharp}$ is a morphism of sheaves, it commutes with restriction.  Restricting the compatibility equality on $V_i\cap U_{ij}$ therefore gives the corresponding equality on $W_i\cap U_{ij}$.  Hence componentwise restriction preserves the compatibility condition and defines the presheaf restriction maps.
\end{solution}

\paragraph{VI.20.E8. Sheaf locality.}
\begin{solution}
Let $V=\bigcup_\alpha V^\alpha$ and suppose two glued sections $(s_i)$ and $(t_i)$ have equal restrictions to every $V^\alpha$.  For fixed $i$, the opens $V_i^\alpha=\psi_i^{-1}(V^\alpha)$ cover $V_i$.  The equalities on the glued cover imply
$s_i|_{V_i^\alpha}=t_i|_{V_i^\alpha}$ for every $\alpha$.  Since $\OO_{X_i}$ is a sheaf, locality gives $s_i=t_i$.  This holds for every $i$, so the compatible families are equal.  Thus the glued presheaf satisfies the locality axiom.
\end{solution}

\paragraph{VI.20.E9. Sheaf gluing.}
\begin{solution}
Suppose compatible sections $(s_i^\alpha)$ are given on an open cover $V=\bigcup_\alpha V^\alpha$ and agree on pairwise intersections.  For each fixed chart $i$, the sections $s_i^\alpha$ agree on the cover $V_i=\bigcup_\alpha V_i^\alpha$, so the sheaf axiom in $X_i$ produces a unique $s_i\in\OO_{X_i}(V_i)$.  On an overlap $V_i\cap U_{ij}$, the two candidate sections
$s_i$ and $\varphi_{ij}^{\sharp}(s_j)$ agree after restriction to every $V_i^\alpha\cap U_{ij}$, because each local family was compatible.  By locality in $\OO_{X_i}$ they agree globally.  Therefore $(s_i)$ is a compatible glued section, and uniqueness follows componentwise.
\end{solution}

\paragraph{VI.20.E10. Stalk identification.}
\begin{solution}
Let $x=\psi_i(x_i)$.  The chart image $\psi_i(X_i)$ is an open neighborhood of $x$, and the gluing construction identifies the restricted sheaf $\OO_X|_{\psi_i(X_i)}$ with $\OO_{X_i}$ through $\psi_i$.  Stalks may be computed using any neighborhood basis, so restricting the direct limit to neighborhoods contained in the chart gives
\[
\OO_{X,x}\cong \OO_{X_i,x_i}.
\]
If another chart represents the same point, the transition isomorphism gives the same local-ring identification, so the stalk is intrinsic.
\end{solution}

\paragraph{VI.20.E11. Locality.}
\begin{solution}
Every point $x$ of the glued space lies in some chart image $\psi_i(X_i)$.  By the preceding stalk identification,
\[
\OO_{X,x}\cong\OO_{X_i,x_i}.
\]
Because $X_i$ is a scheme, $\OO_{X_i,x_i}$ is a local ring.  An isomorphic ring is local as well.  Hence every stalk of the glued ringed space is local, so the gluing is a locally ringed space.
\end{solution}

\paragraph{VI.20.E12. Scheme condition.}
\begin{solution}
The images $\psi_i(X_i)$ form an open cover of the glued locally ringed space, and each is isomorphic as a locally ringed space to the scheme $X_i$.  Given any point $x$, choose such a chart and then choose an affine open neighborhood of the corresponding point inside $X_i$.  Transporting it through $\psi_i$ gives an affine open neighborhood of $x$ in the glued space.  Therefore the glued locally ringed space is locally affine, hence is a scheme.
\end{solution}

\paragraph{VI.20.E13. Two-chart global sections.}
\begin{solution}
For $X=U\cup V$, restriction gives a map
\[
\Gamma(X,\OO_X)\longrightarrow \Gamma(U,\OO_X)\times\Gamma(V,\OO_X).
\]
By the sheaf locality axiom it is injective.  Its image consists precisely of pairs $(s_U,s_V)$ whose restrictions to $U\cap V$ coincide; the sheaf gluing axiom says every such pair has a unique global preimage.  Hence
\[
\Gamma(X,\OO_X)
=\operatorname{Eq}\!\left(
\Gamma(U)\times\Gamma(V)\rightrightarrows\Gamma(U\cap V)
\right).
\]
This is an equalizer in rings, not merely in sets.
\end{solution}

\paragraph{VI.20.E14. Doubled origin sections.}
\begin{solution}
For the doubled-origin line, both chart rings are $k[t]$ and the overlap ring is $k[t,t^{-1}]$.  A global section is therefore a pair $(p,q)\in k[t]^2$ with the same image in the localization.  The localization map $k[t]\to k[t,t^{-1}]$ is injective because $k[t]$ is a domain, so equality after localization implies $p=q$.  Thus
\[
\Gamma(X,\OO_X)\cong k[t],\qquad p\longmapsto(p,p).
\]
\end{solution}

\paragraph{VI.20.E15. Doubled origin non-affine.}
\begin{solution}
From the preceding exercise, $\Gamma(X,\OO_X)\cong k[t]$.  If $X$ were affine, affine reconstruction would produce an isomorphism
\[
X\cong\Spec\Gamma(X,\OO_X)\cong\Spec k[t].
\]
But the gluing leaves the two origins of the two charts distinct, whereas $\Spec k[t]$ has only one prime $(t)$ corresponding to the origin.  An isomorphism of schemes is in particular a bijection of underlying points, so this is impossible.  Therefore the doubled-origin line is not affine.
\end{solution}

\paragraph{VI.20.E16. Inversion overlap.}
\begin{solution}
The punctured chart $D(t)\subseteq\Spec k[t]$ has ring $k[t,t^{-1}]$, while $D(u)\subseteq\Spec k[u]$ has ring $k[u,u^{-1}]$.  Define
\[
\theta:k[u,u^{-1}]\longrightarrow k[t,t^{-1}],\qquad u\longmapsto t^{-1}.
\]
Because $t^{-1}$ is a unit, the map is well defined.  Its inverse sends $t\mapsto u^{-1}$, so $\theta$ is an isomorphism.  Contravariantly it gives the overlap isomorphism used in the two-chart gluing.
\end{solution}

\paragraph{VI.20.E17. Common distinguished refinement.}
\begin{solution}
Fix $x\in U\cap V$ with $U=\Spec A$ and $V=\Spec B$.  Choose $D(f_0)\subseteq U$ with $x\in D(f_0)\subseteq U\cap V$.  Since $D(f_0)$ is open in $V$, choose a distinguished $D(g_0)\subseteq V$ with $x\in D(g_0)\subseteq D(f_0)$.  Now regard $D(g_0)$ as an open neighborhood inside $D(f_0)\cong\Spec A_{f_0}$ and choose a distinguished neighborhood $D(h/f_0^n)$ contained in it.  This is a distinguished open of $U$ after clearing denominators.  Repeating the same shrinking argument inside $V$ if necessary gives an open $W$ containing $x$ that is distinguished in both charts.  Doing this for every $x$ covers $U\cap V$ by common distinguished opens.
\end{solution}

\paragraph{VI.20.E18. Common overlap ring.}
\begin{solution}
Suppose the same open $W$ is written as $D(f)\subseteq U=\Spec A$ and as $D(g)\subseteq V=\Spec B$.  The principal-open theorem gives canonical scheme identifications
\[
W\cong\Spec A_f\qquad\text{and}\qquad W\cong\Spec B_g.
\]
Taking global sections of these two identifications gives
\[
A_f\cong\Gamma(W,\OO_W)\cong B_g.
\]
The resulting isomorphism is the algebraic transition map between the two affine coordinates on $W$.
\end{solution}

\paragraph{VI.20.E19. Compatible maps.}
\begin{solution}
Let $f_i:X_i\to Y$ satisfy $f_i|_{U_{ij}}=f_j\circ\varphi_{ij}$.  Define a point map on the quotient by $f([x])=f_i(x)$ for $x\in X_i$.  If $x$ and $y$ represent the same quotient point, then $y=\varphi_{ij}(x)$ and compatibility gives $f_j(y)=f_i(x)$, so the definition is independent of representatives.  Continuity follows because its restriction to every open chart is the continuous map $f_i$.  The morphisms of structure sheaves agree on chart overlaps by the same compatibility condition, hence glue uniquely to a sheaf map.  Locality of stalk maps is checked inside a chart.  Thus the compatible family determines a morphism $X\to Y$.
\end{solution}

\paragraph{VI.20.E20. Uniqueness of a glued map.}
\begin{solution}
Suppose $f,g:X\to Y$ have the same restrictions to every chart image.  Since the chart images cover $X$, the underlying continuous maps agree at every point.  For the sheaf maps, on each chart the morphisms $f^\sharp$ and $g^\sharp$ agree by assumption.  Morphisms of sheaves are local on the source: two sheaf morphisms agreeing on an open cover agree globally.  Hence $f^\sharp=g^\sharp$, so $f=g$.  This proves uniqueness in the mapping property.
\end{solution}

\paragraph{VI.20.E21. Reconstruct from affine cover.}
\begin{solution}
Let $X=\bigcup_iU_i$ be an affine open cover.  Regard each literal intersection $U_i\cap U_j$ as an open subscheme of both $U_i$ and $U_j$, and take the identity of that intersection as the transition map.  The inverse and cocycle laws are automatic.  The inclusions $U_i\hookrightarrow X$ form a compatible family, so the mapping property gives a morphism from the glued scheme $X'$ to $X$.  Its restriction to each chart is the identity inclusion, hence an isomorphism onto the corresponding open subset.  These opens cover both source and target, so the morphism is an isomorphism.  Thus the original scheme is recovered from its affine atlas.
\end{solution}

\paragraph{VI.20.E22. Disjoint union as gluing.}
\begin{solution}
Take $U_{ij}=\varnothing$ for $i\neq j$.  Then no point in one chart is identified with a point in another chart, so the quotient set and topology are exactly those of the topological disjoint union.  The overlap compatibility conditions for sections are vacuous.  Consequently, on an open set whose chart pieces are $V_i$, sections are simply the product $\prod_i\OO_{X_i}(V_i)$, which is the usual sheaf on a disjoint union.  Hence this gluing is the scheme-theoretic disjoint union.
\end{solution}

\paragraph{VI.20.E23. Full-overlap gluing.}
\begin{solution}
Suppose $U_{12}=X_1$, $U_{21}=X_2$, and $\varphi:X_1\xrightarrow{\sim}X_2$ is the transition.  Every point of $X_1$ is identified with exactly its image under $\varphi$, so either chart maps bijectively onto the entire quotient.  The topology transported from the charts is the original topology, and the glued sheaf is transported by the sheaf isomorphism $\varphi^\sharp$.  Therefore either chart map is an isomorphism from its chart onto the glued scheme.  The result is canonically isomorphic to both $X_1$ and $X_2$.
\end{solution}

\paragraph{VI.20.E24. Triple cocycle.}
\begin{solution}
On a triple overlap, the direct transition from chart $1$ to chart $3$ prescribes the image $\varphi_{13}(x)$.  Passing instead through chart $2$ prescribes $\varphi_{23}(\varphi_{12}(x))$.  If these two points differ, an equivalence relation containing all pairwise identifications would force them to become equivalent in chart $3$.  Then the chart-$3$ map to the quotient would cease to be injective, contradicting the requirement that every chart embed as an open subscheme.  Thus the cocycle condition is necessary for coherent gluing.
\end{solution}

\paragraph{VI.20.E25. Inversion gluing global sections.}
\begin{solution}
A global section of the inversion gluing is a pair
\[
p(t)\in k[t],\qquad q(u)\in k[u]
\]
with $p(t)=q(t^{-1})$ in $k[t,t^{-1}]$.  Hence the common Laurent polynomial lies in both subrings $k[t]$ and $k[t^{-1}]$.  An element of the first has only nonnegative powers of $t$, while an element of the second has only nonpositive powers.  Their intersection is therefore $k$.  Thus every global regular function is constant and
\[
\Gamma(X,\OO_X)\cong k.
\]
\end{solution}

\paragraph{VI.20.E26. Boundary audit.}
\begin{solution}

Closed-subscheme formalism remains in VI/22, while fiber products and base change remain in VI/23--VI/24.  Keeping these boundaries prevents the gluing chapter from silently assuming later universal constructions.
\end{solution}
