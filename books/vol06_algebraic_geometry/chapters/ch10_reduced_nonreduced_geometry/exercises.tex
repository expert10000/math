\begin{exercise}[Nilpotents form an ideal]
Prove directly that the nilpotent elements of a commutative ring form an ideal.
\end{exercise}

\begin{exercise}[Squarefree integers]
Prove that $\ZZ/n\ZZ$ is reduced if and only if $n$ is squarefree.
\end{exercise}

\begin{exercise}[Reduction is reduced]
Prove that $A/\Nil(A)$ is reduced without using the proposition in the text.
\end{exercise}

\begin{exercise}[Universal property]
Let $B$ be reduced.  Show that every ring map $A\to B$ kills $\Nil(A)$ and therefore factors
uniquely through $A_{\mathrm{red}}$.
\end{exercise}

\begin{exercise}[The same spectrum]
Show directly that prime ideals of $A$ and $A_{\mathrm{red}}$ correspond bijectively and
that the correspondence preserves inclusions.
\end{exercise}

\begin{exercise}[Reduced quotient criterion]
Prove that $A/I$ is reduced if and only if $I$ is radical.
\end{exercise}

\begin{exercise}[Compute three reductions]
Compute the reductions of
\[
k[t]/(t^5),\qquad
k[x,y]/(x^3,xy),\qquad
\ZZ/72\ZZ.
\]
\end{exercise}

\begin{exercise}[Products]
Prove that $A\times B$ is reduced if and only if both $A$ and $B$ are reduced.
\end{exercise}

\begin{exercise}[Polynomial extensions]
Prove that $A[t]$ is reduced if and only if $A$ is reduced.
\end{exercise}

\begin{exercise}[Localization preserves reducedness]
Let $S$ be multiplicatively closed.  Prove that $A$ reduced implies $S^{-1}A$ reduced.
\end{exercise}

\begin{exercise}[Reduction and localization]
Construct the natural isomorphism
\[
S^{-1}(A_{\mathrm{red}})\cong(S^{-1}A)_{\mathrm{red}}.
\]
\end{exercise}

\begin{exercise}[A principal-open local test]
Assume $D(f_1),\dots,D(f_r)$ cover $\Spec A$ and every $A_{f_i}$ is reduced.  Prove that
$A$ is reduced.
\end{exercise}

\begin{exercise}[Squarefree one-variable quotients]
Let $k$ be a field and $0\ne f\in k[t]$.  Show that $k[t]/(f)$ is reduced if and only if
$f$ has no repeated irreducible factor.
\end{exercise}

\begin{exercise}[Crossing axes]
Prove directly that $(xy)$ is radical in $k[x,y]$ and conclude that
$k[x,y]/(xy)$ is reduced.
\end{exercise}

\begin{exercise}[An irreducible nonreduced spectrum]
Show that $\Spec k[x,y]/(x^2)$ is irreducible and identify its generic point.
\end{exercise}

\begin{exercise}[All thickened points have the same spectrum]
For $n\ge1$, prove that
\[
\Spec k[t]/(t^n)
\]
has one point and that all these spectra are homeomorphic.
\end{exercise}

\begin{exercise}[Maps to reduced rings]
Let $N\subseteq A$ be any ideal consisting entirely of nilpotent elements.  Show that every
map from $A$ to a reduced ring kills $N$.
\end{exercise}

\begin{exercise}[Nil ideals do not change the spectrum]
If $I\subseteq\Nil(A)$, prove that
\[
\Spec(A/I)\to\Spec A
\]
is a homeomorphism.
\end{exercise}

\begin{exercise}[Prime-local criterion]
Prove that $A$ is reduced if and only if $A_{\mathfrak p}$ is reduced for every
$\mathfrak p\in\Spec A$.
\end{exercise}

\begin{exercise}[Dual-number maps]
Describe all $k$-algebra homomorphisms
\[
k[t]\longrightarrow k[\varepsilon]/(\varepsilon^2).
\]
Which of them induce the same map on underlying topological points?
\end{exercise}
