\begin{solution}
Let $a^m=0$ and $b^n=0$.  In the binomial expansion of $(a+b)^{m+n}$, every term
$a^i b^{m+n-i}$ has either $i\ge m$ or $m+n-i\ge n$, hence vanishes.  Thus $a+b$ is
nilpotent.  Also $(ra)^m=r^m a^m=0$, so nilpotents form an ideal.
\end{solution}

\begin{solution}
If $p^2\mid n$, write $n=p^2m$.  The class of $pm=n/p$ in $\ZZ/n\ZZ$ is nonzero,
while
\[
(pm)^2=p^2m^2=n m\equiv0\pmod n.
\]
Hence $\ZZ/n\ZZ$ is not reduced.  Conversely, if $n=p_1\cdots p_r$ is squarefree,
the Chinese remainder theorem gives
\[
\ZZ/n\ZZ\cong\prod_i\FF_{p_i},
\]
a product of fields, hence a reduced ring.
\end{solution}

\begin{solution}
Every prime $\mathfrak p$ of $A$ contains $\Nil(A)$, so quotient correspondence gives
\[
\mathfrak p\longleftrightarrow \mathfrak p/\Nil(A).
\]
Containment is preserved because quotienting by the same ideal preserves inclusions.
The same correspondence sends $V_A(J)$ to
$V_{A_{\mathrm{red}}}(J/\Nil(A))$ whenever $J\supseteq\Nil(A)$, hence is a
homeomorphism.
\end{solution}

\begin{solution}

\begin{solution}
Localization commutes with quotient:
\[
S^{-1}(A/\Nil(A))\cong S^{-1}A/S^{-1}\Nil(A).
\]
An element $a/s$ is nilpotent in $S^{-1}A$ exactly when some multiple from $S$ makes a
power of $a$ zero, which shows
\[
\Nil(S^{-1}A)=S^{-1}\Nil(A).
\]
The displayed quotient is therefore $(S^{-1}A)_{\mathrm{red}}$.
\end{solution}

\begin{solution}
Let $a^n=0$.  Since $A_{f_i}$ is reduced, $a/1=0$ in every $A_{f_i}$.  Thus for each $i$
there is $N_i$ with $f_i^{N_i}a=0$.  The cover condition implies
\[
\sqrt{(f_1,\dots,f_r)}=(1),
\]
and the same is true after replacing the generators by positive powers.  Hence there are
$c_i\in A$ with
\[
1=\sum_i c_i f_i^{N_i}.
\]
Multiplying by $a$ gives $a=0$.
\end{solution}

\begin{solution}
In $A=k[x,y]/(x^2)$, every prime contains $\bar x$, so the nilradical is $(\bar x)$.
The reduction is
\[
A/(\bar x)\cong k[y],

whose spectrum is irreducible.  Since reduction preserves the underlying spectrum,

\begin{solution}
Prime ideals of $A/I$ correspond to primes $\mathfrak p$ of $A$ with $I\subseteq\mathfrak p$.
If $I\subseteq\Nil(A)$, every prime contains $I$, so this is all of $\Spec A$.  Closed sets
correspond under quotient, giving a homeomorphism.

\begin{solution}
If $A$ is reduced, every localization is reduced.  Conversely suppose each
$A_{\mathfrak p}$ is reduced and let $a^n=0$.  Then $a/1=0$ in every $A_{\mathfrak p}$.
If $a\neq0$, its annihilator $\operatorname{Ann}(a)$ is proper; choose a prime
$\mathfrak p$ containing it.  The equality $a/1=0$ in $A_{\mathfrak p}$ would give some
$s\notin\mathfrak p$ with $sa=0$, hence $s\in\operatorname{Ann}(a)\subseteq\mathfrak p$,
a contradiction.  Thus $a=0$.
\end{solution}

\begin{solution}
A $k$-algebra map is determined by the image of $t$, and every element of the target is
uniquely $a+b\varepsilon$.  Hence the maps are exactly
\[
\varphi_{a,b}:t\mapsto a+b\varepsilon.
\]
On spectra, the dual numbers have one point and its contraction is $(t-a)$.  Therefore all
maps with the same $a$ but arbitrary $b$ induce the same underlying topological point map.
\end{solution}
