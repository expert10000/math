% VI46-DETAILED-EXERCISE-01
\subsection*{Exercise VI.46.1}
\begin{solution}
Every restriction map of the zero sheaf is the unique homomorphism \(0\to0\), which is surjective.  Thus the zero sheaf is flabby.
\end{solution}

% VI46-DETAILED-EXERCISE-02
\subsection*{Exercise VI.46.2}
\begin{solution}
If every global-to-open restriction is onto, then a section on \(V\subseteq U\) first extends to \(X\) and then restricts to \(U\).  The converse is the defining condition with \(U=X\).
\end{solution}

% VI46-DETAILED-EXERCISE-03
\subsection*{Exercise VI.46.3}
\begin{solution}
A skyscraper has section group \(A\) on opens containing its support and \(0\) otherwise.  Restrictions are \(A\to A\), \(A\to0\), or \(0\to0\), all surjective.
\end{solution}

% VI46-DETAILED-EXERCISE-04
\subsection*{Exercise VI.46.4}
\begin{solution}
Given \(f:V\to A\) and \(V\subseteq U\), extend it to \(U\) by setting the value equal to \(0\) on \(U\setminus V\).  Hence every restriction is surjective.
\end{solution}

% VI46-DETAILED-EXERCISE-05
\subsection*{Exercise VI.46.5}
\begin{solution}
On \(V=(-2,-1)\cup(1,2)\), choose different locally constant values on the two components.  A locally constant section on connected \(\mathbb R\) must be constant, so this section cannot extend globally.  The constant sheaf is therefore not flabby.
\end{solution}

% VI46-DETAILED-EXERCISE-06
\subsection*{Exercise VI.46.6}
\begin{solution}
Restrictions in a product are componentwise products of the individual restriction maps.  Given a target tuple, choose an extension in every component.  The tuple of extensions proves surjectivity.
\end{solution}

% VI46-DETAILED-EXERCISE-07
\subsection*{Exercise VI.46.7}
\begin{solution}
\[
\Gamma(U,f_*\mathcal F)=\Gamma(f^{-1}U,\mathcal F).
\]
For \(V\subseteq U\), the relevant restriction is the restriction from \(f^{-1}U\) to \(f^{-1}V\), which is surjective because \(\mathcal F\) is flabby.
\end{solution}

% VI46-DETAILED-EXERCISE-08
\subsection*{Exercise VI.46.8}
\begin{solution}
A quotient section generally lifts only locally.  On overlaps two lifts differ by a section of the kernel, and without an extension property for that kernel the differences may not be correctable globally.  Thus flabbiness of the middle term alone is insufficient.
\end{solution}

% VI46-DETAILED-EXERCISE-09
\subsection*{Exercise VI.46.9}
\begin{solution}
Partially order compatible lifts by inclusion of their domains.  A maximal lift exists by Zorn.  If a point is missing, choose a local lift there; on the overlap its difference from the maximal lift lies in the flabby kernel and can be extended and corrected, producing a larger lift—a contradiction.  Hence the lift exists on all of the open.
\end{solution}

% VI46-DETAILED-EXERCISE-10
\subsection*{Exercise VI.46.10}
\begin{solution}
A quotient section over an open lifts to the middle sheaf by the preceding argument, then extends globally because the middle sheaf is flabby.  Descending that global lift gives a global extension of the quotient section, proving the quotient is flabby.
\end{solution}

% VI46-DETAILED-EXERCISE-11
\subsection*{Exercise VI.46.11}
\begin{solution}
For \(X=U\cup V\), the map is
\[
(s,t)\mapsto s|_{U\cap V}-t|_{U\cap V}.
\]
Its kernel is exactly the set of pairs that agree on the overlap and therefore glue.
\end{solution}

% VI46-DETAILED-EXERCISE-12
\subsection*{Exercise VI.46.12}
\begin{solution}
Given \(a\in\Gamma(U\cap V,\mathcal F)\), extend \(a\) to \(s\in\Gamma(U,\mathcal F)\) and take \(t=0\).  Then the difference of restrictions equals \(a\) up to the chosen sign.
\end{solution}

% VI46-DETAILED-EXERCISE-13
\subsection*{Exercise VI.46.13}
\begin{solution}
For a zero-cochain \((s_i)\),
\[
(\delta s)_{ij}=s_j|_{U_{ij}}-s_i|_{U_{ij}}.
\]
This is the alternating restriction formula.
\end{solution}

% VI46-DETAILED-EXERCISE-14
\subsection*{Exercise VI.46.14}
\begin{solution}
For a one-cochain \((t_{ij})\),
\[
(\delta t)_{ijk}=t_{jk}-t_{ik}+t_{ij}
\]
after restriction to \(U_{ijk}\).
\end{solution}

% VI46-DETAILED-EXERCISE-15
\subsection*{Exercise VI.46.15}
\begin{solution}
For two opens, \(C^2=0\), so every overlap section is a cocycle.  Surjectivity of the difference map says every such cocycle is a coboundary.  Therefore \(\check H^1=0\).
\end{solution}

% VI46-DETAILED-EXERCISE-16
\subsection*{Exercise VI.46.16}
\begin{solution}
For every cover \(\mathfrak U\) and every \(p>0\),
\[
\check H^p(\mathfrak U,\mathcal F)=0
\]
when \(\mathcal F\) is flabby.  Equivalently, the augmented Cech complex is exact.
\end{solution}

% VI46-DETAILED-EXERCISE-17
\subsection*{Exercise VI.46.17}
\begin{solution}
A restriction \(G^0(\mathcal F)(U)\to G^0(\mathcal F)(V)\) simply forgets the coordinates at points of \(U\setminus V\).  Any family on \(V\) extends by choosing arbitrary stalk elements at the new points, so \(G^0(\mathcal F)\) is flabby.
\end{solution}

% VI46-DETAILED-EXERCISE-18
\subsection*{Exercise VI.46.18}
\begin{solution}
If every germ of a section is zero, each point has a neighborhood on which the section vanishes.  These neighborhoods cover the open, and the locality axiom forces the section itself to be zero.  Hence the germ map is injective.
\end{solution}

% VI46-DETAILED-EXERCISE-19
\subsection*{Exercise VI.46.19}
\begin{solution}
Begin with \(\mathcal F\hookrightarrow G^0(\mathcal F)\), let \(\mathcal Q^1\) be the cokernel, then embed \(\mathcal Q^1\hookrightarrow G^0(\mathcal Q^1)\).  These are the first two flabby terms of the Godement resolution.
\end{solution}

% VI46-DETAILED-EXERCISE-20
\subsection*{Exercise VI.46.20}
\begin{solution}
A sheaf is \(\Gamma\)-acyclic when
\[
H^p(X,\mathcal F)=0\quad(p>0),
\]
i.e. all positive right-derived functors of global sections vanish.
\end{solution}

% VI46-DETAILED-EXERCISE-21
\subsection*{Exercise VI.46.21}
\begin{solution}
If \(W'\subseteq W\subseteq U\) are opens, their restriction map for \(\mathcal F|_U\) is exactly the corresponding restriction map for \(\mathcal F\).  Hence it is surjective, so restriction preserves flabbiness.
\end{solution}

% VI46-DETAILED-EXERCISE-22
\subsection*{Exercise VI.46.22}
\begin{solution}
A section on \(V\subset U\) can be represented by a morphism from an extension-by-zero free sheaf.  The source sheaf for \(V\) embeds into the one for \(U\).  Injectivity extends the morphism, producing the desired section over \(U\).
\end{solution}

% VI46-DETAILED-EXERCISE-23
\subsection*{Exercise VI.46.23}
\begin{solution}
Every sheaf admits a flabby resolution and every flabby sheaf is acyclic.  Therefore the acyclic-resolution theorem permits computing derived global sections from a flabby resolution exactly as from an injective resolution.
\end{solution}

% VI46-DETAILED-EXERCISE-24
\subsection*{Exercise VI.46.24}
\begin{solution}
For every open \(V\subseteq Y\), the restricted sheaf on \(f^{-1}V\) is flabby and has vanishing positive cohomology.  Hence the local groups defining \(R^pf_*\mathcal F\) vanish for \(p>0\), so the higher direct-image sheaves are zero.
\end{solution}
