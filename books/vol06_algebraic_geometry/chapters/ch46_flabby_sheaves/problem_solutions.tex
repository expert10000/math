% VI46-DETAILED-PROBLEM-SOLUTIONS
% VI46-DETAILED-PROBLEM-01
\subsection*{Problem VI.46.1}
\begin{solution}
On every nonempty open of an integral scheme, \(\mathcal K_X\) has section group \(K(X)\) and identity restrictions, so it is flabby.  Therefore \(H^p(X,\mathcal K_X)=0\) for all \(p>0\).
\end{solution}

% VI46-DETAILED-PROBLEM-02
\subsection*{Problem VI.46.2}
\begin{solution}
Give different locally constant values on the two components of \((-2,-1)\cup(1,2)\).  Any globally locally constant function on connected \(\mathbb R\) is constant, so the section cannot extend; hence the constant sheaf is not flabby.
\end{solution}

% VI46-DETAILED-PROBLEM-03
\subsection*{Problem VI.46.3}
\begin{solution}
Use Zorn on compatible local lifts.  A missing point admits a new local lift; the difference on the overlap lies in the flabby kernel and can be extended and used to correct one lift.  The corrected lifts glue, contradicting maximality.  Thus a lift exists over the whole open.
\end{solution}

% VI46-DETAILED-PROBLEM-04
\subsection*{Problem VI.46.4}
\begin{solution}
For \(V\subseteq U\subseteq Y\), the restriction for \(f_*\mathcal F\) is exactly
\[
\Gamma(f^{-1}U,\mathcal F)\to\Gamma(f^{-1}V,\mathcal F),
\]
which is surjective by flabbiness.
\end{solution}

% VI46-DETAILED-PROBLEM-05
\subsection*{Problem VI.46.5}
\begin{solution}
The augmented two-open complex is
\[
0\to\Gamma(X,\mathcal F)\to\Gamma(U,\mathcal F)\oplus\Gamma(V,\mathcal F)\to\Gamma(U\cap V,\mathcal F)\to0.
\]
Sheaf gluing gives exactness through the middle and flabbiness gives surjectivity at the last term.
\end{solution}

% VI46-DETAILED-PROBLEM-06
\subsection*{Problem VI.46.6}
\begin{solution}
For three opens,
\[
C^0=\prod_i\Gamma(U_i,\mathcal F),\quad
C^1=\prod_{i<j}\Gamma(U_{ij},\mathcal F),\quad
C^2=\Gamma(U_{012},\mathcal F).
\]
The differentials are the usual alternating restrictions.
\end{solution}

% VI46-DETAILED-PROBLEM-07
\subsection*{Problem VI.46.7}
\begin{solution}
Iterate \(G^0\) on successive cokernels.  Morphisms induce stalk maps, hence maps at every stage, so the construction is functorial.  Each \(G^0\)-term is flabby because restrictions are coordinate projections.
\end{solution}

% VI46-DETAILED-PROBLEM-08
\subsection*{Problem VI.46.8}
\begin{solution}
A quotient section only lifts locally; differences of lifts lie in the kernel.  Unless kernel sections can be extended, those differences may prevent gluing.  Flabbiness of the middle term alone does not supply this correction; flabbiness of the kernel does.
\end{solution}

% VI46-DETAILED-PROBLEM-09
\subsection*{Problem VI.46.9}
\begin{solution}
Injectivity extends the morphism encoding a section across the monomorphism associated to an open inclusion, proving injective sheaves are flabby.  Since there are enough injectives, successively embedding cokernels in injectives produces a flabby resolution.
\end{solution}

% VI46-DETAILED-PROBLEM-10
\subsection*{Problem VI.46.10}
\begin{solution}
Vanishing only on \(X\) is one global statement.  Flabbiness remains true on every open, so higher cohomology vanishes on every open.  This local acyclicity is what higher direct images and cover arguments require.
\end{solution}

% VI46-DETAILED-PROBLEM-11
\subsection*{Problem VI.46.11}
\begin{solution}
Apply global sections termwise to a flabby resolution and take ordinary cohomology of the resulting complex.  Acyclicity of the flabby terms identifies these groups with \(H^p(X,\mathcal F)\).
\end{solution}

% VI46-DETAILED-PROBLEM-12
\subsection*{Problem VI.46.12}
\begin{solution}
VI/46 proves that extension kills positive Cech classes.  VI/47 studies the same overlap complex without assuming extension, so cocycles can survive; refinements, Leray covers, and explicit calculations then measure the obstruction.
\end{solution}
