\subsection*{VI.39.P1: Factorization on an affine factorial scheme}
% VI39-DETAILED-PROBLEM-01
\begin{solution}
Set \(A=k[x,y,z]\).  Since \(A\) is a UFD, its height-one prime ideals are generated by irreducible elements, and for an irreducible \(p\) the coefficient of \([V(p)]\) in a principal divisor is the exponent of \(p\) in the factorization of the rational function.

Here
\[
 f=x^3(y-z)^2y^{-4}z^{-1}.
\]
The four elements \(x\), \(y-z\), \(y\), and \(z\) are irreducible and pairwise nonassociate.  Therefore their orders are respectively
\[
 3,\qquad 2,\qquad -4,\qquad -1,
\]
and every other height-one prime has order zero.  Thus
\[
 \boxed{\operatorname{div}(f)
 =3[V(x)]+2[V(y-z)]-4[V(y)]-[V(z)].}
\]
The positive coefficients are zeros and the negative coefficients are poles.
\end{solution}

\subsection*{VI.39.P2: Restriction changes the visible divisor}
% VI39-DETAILED-PROBLEM-02
\begin{solution}
On \(X=\mathbb P^1_k\), the rational coordinate \(t\) has a simple zero at \(0\) and a simple pole at \(\infty\).  Hence
\[
 \operatorname{div}_X(t)=[0]-[\infty].
\]
Now \(U=X\setminus\{0,\infty\}\cong\mathbb G_m\).  Restriction of a Weil divisor to an open subset deletes every prime component lying entirely in the complement.  Both components of \(\operatorname{div}_X(t)\) are deleted, so
\[
 \boxed{\operatorname{div}_U(t)=0.}
\]
Equivalently, \(t\) and \(t^{-1}\) are both regular on \(U\), so \(t\in\Gamma(U,\mathcal O_U)^\times\).  This example shows explicitly that a principal divisor depends on the ambient scheme in which zeros and poles are being recorded.
\end{solution}

\subsection*{VI.39.P3: A projective ratio}
% VI39-DETAILED-PROBLEM-03
\begin{solution}
The numerator and denominator have the same homogeneous degree, so
\[
 f=\frac{x_0x_1}{x_2^2}
\]
is a rational function on \(\mathbb P^2_k\).  Let \(H_i=V_+(x_i)\).  At the generic point of \(H_0\), the coordinates \(x_1,x_2\) are units, so \(x_0\) contributes order one.  Similarly \(H_1\) occurs with order one.  Along \(H_2\), the denominator contains \(x_2^2\), so the order is \(-2\).  No other irreducible homogeneous factor occurs.  Therefore
\[
 \boxed{\operatorname{div}(f)=[H_0]+[H_1]-2[H_2].}
\]
As a useful check, all three hyperplanes have degree one, so the divisor has total degree \(1+1-2=0\).
\end{solution}

\subsection*{VI.39.P4: Orders on the quadric cone}
% VI39-DETAILED-PROBLEM-04
\begin{solution}
Let
\[
 A=k[x,y,z]/(xy-z^2),\qquad P=(x,z).
\]
At \(P\), the element \(y\notin P\), hence \(y\) is a unit in \(A_P\).  The relation gives
\[
 x=\frac{z^2}{y}.
\]
Thus \(z\) is a uniformizer of the DVR \(A_P\), so
\[
 \operatorname{ord}_P(z)=1,
 \qquad
 \operatorname{ord}_P(x)=2.
\]
For
\[
 f=\frac{x^3}{z^5}
\]
we obtain
\[
 \operatorname{ord}_P(f)
 =3\operatorname{ord}_P(x)-5\operatorname{ord}_P(z)
 =3\cdot2-5\cdot1=1.
\]
Hence \(f\) has a simple zero along the prime divisor \(P\):
\[
 \boxed{\operatorname{ord}_P(f)=1.}
\]
Notice that the exponent of \(x\) in the displayed rational function is not itself the order along \(P\); the singular relation doubles the order of \(x\).
\end{solution}

\subsection*{VI.39.P5: Degree check on \(\mathbb P^1\)}
% VI39-DETAILED-PROBLEM-05
\begin{solution}
Factor
\[
 p=c\prod_i p_i^{a_i},\qquad q=d\prod_j q_j^{b_j}
\]
into distinct monic irreducibles.  Let \(P_i,Q_j\) be the corresponding closed points of \(\mathbb A^1_k\).  Then
\[
 \operatorname{div}_{\mathbb P^1}(p)
 =\sum_i a_i[P_i]-r[\infty],
\]
\[
 \operatorname{div}_{\mathbb P^1}(q)
 =\sum_j b_j[Q_j]-s[\infty].
\]
Therefore
\[
 \operatorname{div}(p/q)
 =\sum_i a_i[P_i]-\sum_j b_j[Q_j]+(s-r)[\infty].
\]
Taking weighted degree and using
\[
 \sum_i a_i[\kappa(P_i):k]=r,
 \qquad
 \sum_j b_j[\kappa(Q_j):k]=s,
\]
we get
\[
 \deg\operatorname{div}(p/q)=r-s+(s-r)=0.
\]
Thus the finite zeros and poles are exactly balanced by the order at infinity.
\end{solution}

\subsection*{VI.39.P6: Effective principal divisors on a proper curve}
% VI39-DETAILED-PROBLEM-06
\begin{solution}
Write
\[
 \operatorname{div}(f)=\sum_P n_P[P],\qquad n_P\ge0.
\]
Because \(C\) is a normal projective integral curve, every principal divisor has degree zero:
\[
 0=\deg\operatorname{div}(f)
 =\sum_P n_P[\kappa(P):k].
\]
Each residue-field degree \([\kappa(P):k]\) is a positive integer, and every \(n_P\) is nonnegative.  A sum of nonnegative integers can equal zero only if every summand is zero.  Hence \(n_P=0\) for every \(P\), so
\[
 \boxed{\operatorname{div}(f)=0.}
\]
In words: on a proper curve a nonzero rational function cannot have zeros without also having poles of matching total degree.
\end{solution}

\subsection*{VI.39.P7: Detecting a unit at a generic point}
% VI39-DETAILED-PROBLEM-07
\begin{solution}
The valuation is additive, so
\[
 \operatorname{ord}_Z(fg)
 =\operatorname{ord}_Z(f)+\operatorname{ord}_Z(g)
 =3+(-3)=0.
\]
In the DVR \(\mathcal O_{X,\eta_Z}\), order zero is equivalent to being a unit.  Therefore
\[
 \boxed{fg\in\mathcal O_{X,\eta_Z}^{\times}.}
\]
The zero of order three contributed by \(f\) and the pole of order three contributed by \(g\) cancel exactly along \(Z\).  This is a statement only at the generic point of \(Z\); other prime divisors may still contribute to \(\operatorname{div}(fg)\).
\end{solution}

\subsection*{VI.39.P8: Why the cusp is not a DVR computation}
% VI39-DETAILED-PROBLEM-08
\begin{solution}
Let
\[
 A=k[t^2,t^3],\qquad \mathfrak m=(t^2,t^3),\qquad R=A_{\mathfrak m}.
\]
If \(t^2\) were a uniformizer, the maximal ideal of the one-dimensional local ring \(R\) would be generated by \(t^2\).  In particular \(t^3\in(t^2)R\), so
\[
 t^3=t^2u
\]
for some \(u\in R\).  In the fraction field this forces \(u=t\).  But \(t\notin R\): indeed \(t\) is integral over \(A\) because it satisfies \(T^2-t^2=0\), yet it is not contained in \(A\) or in its localization at \(\mathfrak m\).  Thus \(R\) is not integrally closed and cannot be a DVR.

Equivalently, the value semigroup of elements of the cusp ring at the singular point begins
\[
 0,2,3,4,5,\ldots
\]
rather than containing an element of order one.  The normalization \(k[t]_{(t)}\) restores the missing uniformizer \(t\).
\end{solution}

\subsection*{VI.39.P9: Same support, different multiplicity}
% VI39-DETAILED-PROBLEM-09
\begin{solution}
In the UFD \(k[x,y]\), the prime divisors determined by \(x\) and \(y\) are the coordinate axes.  Factorization gives
\[
 \operatorname{div}(xy)=[V(x)]+[V(y)]
\]
and
\[
 \operatorname{div}(x^2y^3)=2[V(x)]+3[V(y)].
\]
Thus both divisors have the same underlying support
\[
 V(x)\cup V(y),
\]
but their coefficients differ.  The first records multiplicities \((1,1)\), while the second records \((2,3)\).  Hence Weil divisors contain strictly more information than their supports.
\end{solution}

\subsection*{VI.39.P10: A rational function with no divisor on an open torus}
% VI39-DETAILED-PROBLEM-10
\begin{solution}
The open set
\[
 U=D(xy)=D(x)\cap D(y)
\]
has coordinate ring
\[
 k[x,y]_{xy}=k[x^{\pm1},y^{\pm1}].
\]
Both \(x\) and \(y\) are units there, so
\[
 \frac{x^5}{y^2}\in\Gamma(U,\mathcal O_U)^\times.
\]
A unit has order zero at every codimension-one point.  Therefore
\[
 \boxed{\operatorname{div}_U(x^5/y^2)=0.}
\]
Viewed on \(\mathbb A^2\), the same rational function would have divisor \(5[V(x)]-2[V(y)]\); both components disappear after restricting to \(D(xy)\).
\end{solution}

\subsection*{VI.39.P11: Principal pullback only}
% VI39-DETAILED-PROBLEM-11
\begin{solution}
Let
\[
 u=f/h\in K(X)^\times.
\]
The hypothesis \(\operatorname{div}_X(f)=\operatorname{div}_X(h)\) gives
\[
 \operatorname{div}_X(u)=0.
\]
Because the chapter works with normal Noetherian integral schemes, a rational function with zero Weil divisor is a global unit.  Indeed, on every affine open \(V=\operatorname{Spec}A\subset X\), normality makes \(A\) a Krull domain and
\[
 A=\bigcap_{\operatorname{ht}\mathfrak p=1}A_{\mathfrak p}
 \subset K(X).
\]
All height-one orders of both \(u\) and \(u^{-1}\) are nonnegative (in fact zero), so \(u,u^{-1}\in A\).  Hence \(u\in\Gamma(X,\mathcal O_X)^\times\).

Pullback of a global unit is a global unit, so \(g^*u\in\Gamma(Y,\mathcal O_Y)^\times\) and
\[
 \operatorname{div}_Y(g^*u)=0.
\]
Using \(g^*u=g^*f/g^*h\), we conclude
\[
 \boxed{\operatorname{div}_Y(g^*f)=\operatorname{div}_Y(g^*h).}
\]
This does \emph{not} create a pullback operation for arbitrary Weil divisors under an arbitrary morphism; it uses only rational-function pullback plus the normal codimension-one extension property.
\end{solution}

\subsection*{VI.39.P12: Cartier coefficients are well defined}
% VI39-DETAILED-PROBLEM-12
\begin{solution}
Suppose \(f=ug\) in a neighborhood of \(\eta_Z\), with
\[
 u\in\mathcal O_{X,\eta_Z}^{\times}.
\]
The unit has zero valuation, so
\[
 \operatorname{ord}_Z(f)
 =\operatorname{ord}_Z(u)+\operatorname{ord}_Z(g)
 =0+\operatorname{ord}_Z(g).
\]
Hence
\[
 \boxed{\operatorname{ord}_Z(f)=\operatorname{ord}_Z(g).}
\]
Therefore the coefficient assigned to \(Z\) depends only on the Cartier divisor, not on the chosen local equation.  This is the local well-definedness step behind the Cartier-to-Weil map developed in VI/40.
\end{solution}
