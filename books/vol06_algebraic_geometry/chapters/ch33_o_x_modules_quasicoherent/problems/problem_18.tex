% VI33-DETAILED-PROBLEM-18
\begin{problem}[VI.33.P18: Coherent kernels on a Noetherian affine scheme]\label{prob:vi33-18}
Let \(A\) be Noetherian and \(u:M\to N\) a morphism of finitely generated \(A\)-modules. Prove that the kernel,
image, and cokernel sheaves of \(\widetilde u\) are coherent.

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Since \(A\) is Noetherian and \(M,N\) are finitely generated, every submodule of a finitely generated module is
finitely generated. Therefore
\[
\ker u\subseteq M
\]
and
\[
\operatorname{im}u\subseteq N
\]
are finitely generated. The quotient
\[
\operatorname{coker}u=N/\operatorname{im}u
\]
is also finitely generated.

Problem~14 identifies the corresponding sheaf operations:
\[
\ker(\widetilde u)\cong\widetilde{\ker u},
\]
\[
\operatorname{im}(\widetilde u)\cong\widetilde{\operatorname{im}u},
\]
and
\[
\operatorname{coker}(\widetilde u)\cong\widetilde{\operatorname{coker}u}.
\]
Under the locally Noetherian convention of VI/33, tilde sheaves of finitely generated modules are coherent.
Hence all three sheaves are coherent.
\end{solution}
\fi
