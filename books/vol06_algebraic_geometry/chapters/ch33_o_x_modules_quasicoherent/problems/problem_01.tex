% VI33-DETAILED-PROBLEM-01
\begin{problem}[VI.33.P1: Affine reconstruction for quasi-coherent sheaves]\label{prob:vi33-01}
Let \(X=\operatorname{Spec}A\) and let \(\mathcal F\) be quasi-coherent. Prove that the canonical map
\[
\widetilde{\Gamma(X,\mathcal F)}\longrightarrow\mathcal F
\]
is an isomorphism.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Because \(X\) itself is affine and \(\mathcal F\) is quasi-coherent, there exists an \(A\)-module \(M\) and an
isomorphism
\[
\alpha:\widetilde M\xrightarrow{\sim}\mathcal F.
\]
Taking global sections gives
\[
\Gamma(X,\mathcal F)
\cong
\Gamma(X,\widetilde M)
\cong
M.
\]
Under this identification, the canonical reconstruction morphism becomes the identity model
\[
\widetilde M\longrightarrow\widetilde M.
\]

One can also verify the canonical map directly on the distinguished-open basis.  On \(D(f)\),
quasi-coherence gives
\[
\mathcal F(D(f))
\cong
\Gamma(X,\mathcal F)_f.
\]
But the left side of the reconstruction map has sections
\[
\widetilde{\Gamma(X,\mathcal F)}(D(f))
=
\Gamma(X,\mathcal F)_f.
\]
Thus the map is an isomorphism on every \(D(f)\).  Equivalently, it is an isomorphism on every stalk. Hence
\[
\boxed{\widetilde{\Gamma(X,\mathcal F)}\cong\mathcal F}.
\]

This is the essential-surjectivity half of the affine equivalence
\(A\text{-}\mathrm{Mod}\simeq\operatorname{QCoh}(X)\).
\end{solution}
\fi
