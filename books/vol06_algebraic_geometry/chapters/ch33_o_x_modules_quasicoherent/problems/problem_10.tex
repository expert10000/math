% VI33-DETAILED-PROBLEM-10
\begin{problem}[VI.33.P10: Local criterion for zero quasi-coherent sheaf]\label{prob:vi33-10}
Let \(X=\operatorname{Spec}A\) and \(\mathcal F=\widetilde M\). Prove that
\[
\mathcal F=0\iff M=0,
\]
and that it is enough to check vanishing on a distinguished cover.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
If \(M=0\), then every localization \(M_f\) is zero, so \(\widetilde M=0\).

Conversely, if \(\widetilde M=0\), then taking global sections gives
\[
M\cong\Gamma(X,\widetilde M)=0.
\]
Thus
\[
\boxed{\widetilde M=0\iff M=0}.
\]

Now suppose
\[
X=\bigcup_iD(f_i)
\]
and
\[
\widetilde M|_{D(f_i)}=0
\]
for every \(i\). Equivalently,
\[
M_{f_i}=0
\]
for every \(i\).  Given any prime \(\mathfrak p\), choose \(i\) with
\[
\mathfrak p\in D(f_i).
\]
Then
\[
M_{\mathfrak p}
\cong
(M_{f_i})_{\mathfrak p}
=
0.
\]
Hence every stalk of \(\widetilde M\) vanishes, so \(\widetilde M=0\), and therefore \(M=0\).  This proves that
vanishing is affine-local, and in fact distinguished-local.
\end{solution}
\fi
