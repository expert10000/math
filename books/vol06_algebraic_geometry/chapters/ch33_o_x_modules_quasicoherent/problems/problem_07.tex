% VI33-DETAILED-PROBLEM-07
\begin{problem}[VI.33.P7: Tensor product and scheme-theoretic intersection]\label{prob:vi33-07}
For ideals \(I,J\subseteq A\), prove
\[
\widetilde{A/I}\otimes\widetilde{A/J}
\cong
\widetilde{A/(I+J)}
\]
and interpret the support.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Compatibility of tilde with tensor products gives
\[
\widetilde{A/I}\otimes_{\mathcal O_X}\widetilde{A/J}
\cong
\widetilde{(A/I)\otimes_A(A/J)}.
\]
The standard quotient-tensor calculation is
\[
(A/I)\otimes_A(A/J)
\cong
A/(I+J),
\]
because tensoring the sequence \(I\to A\to A/I\to0\) with \(A/J\) produces the quotient of \(A/J\) by the image
of \(I\), namely \((I+J)/J\). Hence
\[
\boxed{
\widetilde{A/I}\otimes\widetilde{A/J}
\cong
\widetilde{A/(I+J)}.
}
\]

The support is
\[
V(I+J).
\]
Set-theoretically,
\[
V(I+J)=V(I)\cap V(J).
\]
Scheme-theoretically, \(A/(I+J)\) is precisely the coordinate ring of the fiber-product intersection of the two
closed subschemes inside \(\Spec A\).
\end{solution}
\fi
