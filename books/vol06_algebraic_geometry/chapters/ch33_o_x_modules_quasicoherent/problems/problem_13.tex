% VI33-DETAILED-PROBLEM-13
\begin{problem}[VI.33.P13: Restriction to a principal affine open]\label{prob:vi33-13}
Let \(X=\Spec A\), \(M\) an \(A\)-module, and \(f\in A\). Prove
\[
\widetilde M|_{D(f)}
\cong
\widetilde{M_f}
\]
where the right side is formed as an \(A_f\)-module sheaf on \(\Spec A_f=D(f)\).

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The distinguished opens of \(\Spec A_f\) have the form
\[
D_{A_f}(g/f^n),
\]
and they agree with intersections of distinguished opens of \(\Spec A\) with \(D(f)\). It is enough to check a
basis of opens of the form
\[
D(f)\cap D(g)=D(fg).
\]
On such an open, the restriction of \(\widetilde M\) has sections
\[
M_{fg}.
\]
On the other hand, the \(A_f\)-module \(M_f\) localized at \(g\) has
\[
(M_f)_g\cong M_{fg}.
\]
The identifications commute with restriction, so the sheaves agree:
\[
\boxed{
\widetilde M|_{D(f)}
\cong
\widetilde{M_f}.
}
\]
This is the precise sheaf-theoretic expression of the slogan that restriction on an affine scheme is localization.
\end{solution}
\fi
