% VI33-DETAILED-PROBLEM-05
\begin{problem}[VI.33.P5: Pullback is extension of scalars]\label{prob:vi33-05}
Let \(A\to B\) and
\[
f:\operatorname{Spec}B\to\operatorname{Spec}A.
\]
Prove
\[
f^*\widetilde M\cong\widetilde{B\otimes_A M}.
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
By definition,
\[
f^*\widetilde M
=
\mathcal O_{\Spec B}
\otimes_{f^{-1}\mathcal O_{\Spec A}}
f^{-1}\widetilde M.
\]
The result is quasi-coherent on \(\Spec B\). To identify its corresponding \(B\)-module, take global sections on
the affine target.  The scalar extension induced by \(A\to B\) gives
\[
B\otimes_A M.
\]

A local check is equally concrete.  At a prime \(\mathfrak q\subseteq B\), let
\[
\mathfrak p=\mathfrak q\cap A.
\]
Then
\[
(f^*\widetilde M)_{\mathfrak q}
\cong
B_{\mathfrak q}\otimes_{A_{\mathfrak p}}M_{\mathfrak p}.
\]
On the other hand,
\[
\left(\widetilde{B\otimes_A M}\right)_{\mathfrak q}
\cong
(B\otimes_A M)_{\mathfrak q}
\cong
B_{\mathfrak q}\otimes_{A_{\mathfrak p}}M_{\mathfrak p}.
\]
The canonical map is therefore an isomorphism on every stalk, giving
\[
\boxed{f^*\widetilde M\cong\widetilde{B\otimes_A M}}.
\]
\end{solution}
\fi
