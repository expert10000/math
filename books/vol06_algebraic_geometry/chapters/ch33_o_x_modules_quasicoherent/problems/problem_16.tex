% VI33-DETAILED-PROBLEM-16
\begin{problem}[VI.33.P16: Isomorphisms detected by localization]\label{prob:vi33-16}
Let \(u:M\to N\). Prove that the following are equivalent:
\[
\begin{gathered}
u\text{ is an isomorphism}\\
\Longleftrightarrow\ 
u_{\mathfrak p}\text{ is an isomorphism for every }\mathfrak p\\
\Longleftrightarrow\
\widetilde u\text{ is an isomorphism}.
\end{gathered}
\]

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
If \(u\) is an isomorphism, every localization \(u_{\mathfrak p}\) is an isomorphism.

Suppose every \(u_{\mathfrak p}\) is an isomorphism.  Then
\[
(\widetilde u)_{\mathfrak p}
=
u_{\mathfrak p}
\]
under the affine stalk identification. Hence \(\widetilde u\) is an isomorphism on every stalk, so it is an
isomorphism of sheaves.

Finally, suppose \(\widetilde u\) is an isomorphism. Taking global sections on \(X=\Spec A\) gives
\[
\Gamma(X,\widetilde u)=u.
\]
Since global sections preserve isomorphisms, \(u\) is an isomorphism.

Thus
\[
\boxed{
u\text{ iso}
\iff
u_{\mathfrak p}\text{ iso for all }\mathfrak p
\iff
\widetilde u\text{ iso}.
}
\]
The same pattern applies to injectivity and surjectivity after using kernels and cokernels.
\end{solution}
\fi
