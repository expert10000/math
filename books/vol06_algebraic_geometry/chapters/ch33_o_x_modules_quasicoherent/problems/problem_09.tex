% VI33-DETAILED-PROBLEM-09
\begin{problem}[VI.33.P9: A nilpotent thickening as a coherent module]\label{prob:vi33-09}
Let
\[
A=k[t,\varepsilon]/(\varepsilon^2)
\quad\text{and}\quad
M=A/(t).
\]
Describe \(\operatorname{Supp}(\widetilde M)\) and explain what module data remain invisible in the support.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The module is cyclic, so
\[
\operatorname{Ann}_A(M)=(t).
\]
Indeed \(M\cong k[\varepsilon]/(\varepsilon^2)\), and \(t\) acts by zero while \(\varepsilon\) does not. Therefore
\[
\operatorname{Supp}(\widetilde M)
=
V(t).
\]
This closed subset consists of the prime(s) of the thickened affine line containing \(t\); topologically it is the
single point over \(t=0\).

However, support records only whether the stalk is zero.  It forgets the internal module structure at that point.
In particular, on \(M\) the element
\[
\varepsilon\ne0
\]
acts nilpotently:
\[
\varepsilon^2=0.
\]
For example, the support cannot distinguish \(M\) from \(A/(t,\varepsilon)\).  The first module still carries
a nontrivial nilpotent action. Hence
\[
\boxed{\operatorname{Supp}(\widetilde M)=V(t)}
\]
does not record nilpotent thickness or the action of \(\varepsilon\).
\end{solution}
\fi
