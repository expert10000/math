% VI33-DETAILED-PROBLEM-15
\begin{problem}[VI.33.P15: Quasi-coherent kernels and cokernels]\label{prob:vi33-15}
Let \(\varphi:\mathcal F\to\mathcal G\) be a morphism of quasi-coherent sheaves on a scheme \(X\). Prove that
\(\ker\varphi\) and \(\operatorname{coker}\varphi\) are quasi-coherent.

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Quasi-coherence is affine-local. Choose an affine open
\[
U=\Spec A\subseteq X
\]
on which
\[
\mathcal F|_U\cong\widetilde M,
\qquad
\mathcal G|_U\cong\widetilde N.
\]
By full faithfulness of the affine equivalence, the restricted morphism
\[
\varphi|_U
\]
is induced by an \(A\)-linear map
\[
u:M\to N.
\]
Problem~14 gives
\[
(\ker\varphi)|_U
\cong
\widetilde{\ker u}
\]
and
\[
(\operatorname{coker}\varphi)|_U
\cong
\widetilde{\operatorname{coker}u}.
\]
Thus both kernel and cokernel are represented by modules on every affine chart. Therefore
\[
\boxed{
\ker\varphi,\ \operatorname{coker}\varphi\in\operatorname{QCoh}(X).
}
\]
In particular, the category of quasi-coherent sheaves on a scheme is closed under these abelian-category operations.
\end{solution}
\fi
