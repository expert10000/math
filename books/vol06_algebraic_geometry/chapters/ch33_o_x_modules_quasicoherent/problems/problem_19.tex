% VI33-DETAILED-PROBLEM-19
\begin{problem}[VI.33.P19: Stalk of affine pullback]\label{prob:vi33-19}
Let \(A\to B\), let \(f:\Spec B\to\Spec A\), and let \(\mathfrak q\in\Spec B\) with
\(\mathfrak p=\mathfrak q\cap A\). Prove
\[
(f^*\widetilde M)_{\mathfrak q}
\cong
B_{\mathfrak q}\otimes_{A_{\mathfrak p}}M_{\mathfrak p}.
\]

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
By affine pullback,
\[
f^*\widetilde M
\cong
\widetilde{B\otimes_A M}.
\]
Taking the stalk at \(\mathfrak q\) gives
\[
(f^*\widetilde M)_{\mathfrak q}
\cong
(B\otimes_A M)_{\mathfrak q}.
\]
Localization commutes with tensor product:
\[
(B\otimes_A M)_{\mathfrak q}
\cong
B_{\mathfrak q}\otimes_A M.
\]
Every element of \(A\setminus\mathfrak p\) maps to an element of \(B\setminus\mathfrak q\), hence becomes a unit in
\(B_{\mathfrak q}\).  Therefore the tensor product factors uniquely through \(A_{\mathfrak p}\), and
\[
B_{\mathfrak q}\otimes_A M
\cong
B_{\mathfrak q}\otimes_{A_{\mathfrak p}}M_{\mathfrak p}.
\]
Thus
\[
\boxed{
(f^*\widetilde M)_{\mathfrak q}
\cong
B_{\mathfrak q}\otimes_{A_{\mathfrak p}}M_{\mathfrak p}.
}
\]
\end{solution}
\fi
