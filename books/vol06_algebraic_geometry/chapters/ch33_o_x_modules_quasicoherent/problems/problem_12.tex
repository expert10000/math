% VI33-DETAILED-PROBLEM-12
\begin{problem}[VI.33.P12: Bridge to graded modules]\label{prob:vi33-12}
Let \(S\) be a graded ring and \(M\) a graded \(S\)-module. Explain which part of the affine construction must change
on \(\operatorname{Proj}S\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
For an affine scheme \(\Spec A\), the tilde construction uses
\[
D(f)\longmapsto M_f.
\]
On \(\Proj S\), the basic opens are
\[
D_+(f)
\]
for homogeneous \(f\in S_+\). Their coordinate rings are not the whole graded localization \(S_f\), because that
ring still contains all degrees.  Instead,
\[
\mathcal O_{\Proj S}(D_+(f))
=
S_{(f)}
:=
(S_f)_0.
\]
Therefore the module assigned to \(D_+(f)\) should be
\[
M_{(f)}
:=
(M_f)_0.
\]

If \(D_+(g)\subseteq D_+(f)\), homogeneous localization gives the restriction map, followed by degree zero.
On overlaps,
\[
D_+(f)\cap D_+(g)=D_+(fg),
\]
and the iterated homogeneous localizations agree.  Thus the affine recipe changes in exactly one algebraic place:
\[
\boxed{
M_f\quad\rightsquigarrow\quad(M_f)_0.
}
\]
The sheaf-gluing mechanism itself is the one already developed in VI/33.
\end{solution}
\fi
