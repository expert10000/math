% VI33-DETAILED-PROBLEM-02
\begin{problem}[VI.33.P2: Exactness of the tilde functor]\label{prob:vi33-02}
Prove that \(M\mapsto\widetilde M\) is exact.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
0\longrightarrow M'\xrightarrow{u}M\xrightarrow{v}M''\longrightarrow0
\]
be an exact sequence of \(A\)-modules.  For every prime \(\mathfrak p\subseteq A\), localization is exact:
\[
0\longrightarrow M'_{\mathfrak p}
\xrightarrow{u_{\mathfrak p}}
M_{\mathfrak p}
\xrightarrow{v_{\mathfrak p}}
M''_{\mathfrak p}
\longrightarrow0.
\]
The affine stalk formula identifies these modules with the stalks of the associated sheaves:
\[
(\widetilde{M'})_{\mathfrak p}\cong M'_{\mathfrak p},
\quad
(\widetilde M)_{\mathfrak p}\cong M_{\mathfrak p},
\quad
(\widetilde{M''})_{\mathfrak p}\cong M''_{\mathfrak p}.
\]
Therefore
\[
0\to(\widetilde{M'})_{\mathfrak p}
\to(\widetilde M)_{\mathfrak p}
\to(\widetilde{M''})_{\mathfrak p}
\to0
\]
is exact for every \(\mathfrak p\).  Exactness of sheaves of modules is stalkwise, so
\[
\boxed{
0\to\widetilde{M'}\to\widetilde M\to\widetilde{M''}\to0
}
\]
is exact.

The same argument works for any finite exact sequence. Thus the functor
\[
\boxed{M\longmapsto\widetilde M}
\]
preserves kernels, cokernels, images, and short exact sequences.  The key input is not sheafification by itself,
but the localization formula for the stalks.
\end{solution}
\fi
