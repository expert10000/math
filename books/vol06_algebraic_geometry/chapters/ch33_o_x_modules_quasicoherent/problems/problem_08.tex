% VI33-DETAILED-PROBLEM-08
\begin{problem}[VI.33.P8: Affine exactness of global sections]\label{prob:vi33-08}
Let \(X\) be affine and
\[
0\to\mathcal F'\to\mathcal F\to\mathcal F''\to0
\]
an exact sequence of quasi-coherent sheaves. Prove that global sections remain exact on the right.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Write
\[
X=\Spec A.
\]
By affine reconstruction there are modules
\[
M'=\Gamma(X,\mathcal F'),
\qquad
M=\Gamma(X,\mathcal F),
\qquad
M''=\Gamma(X,\mathcal F'')
\]
with
\[
\mathcal F'\cong\widetilde{M'},
\quad
\mathcal F\cong\widetilde M,
\quad
\mathcal F''\cong\widetilde{M''}.
\]
Because the affine equivalence is exact, the sheaf sequence corresponds to an exact module sequence
\[
0\to M'\to M\to M''\to0.
\]
But global sections of an affine module sheaf recover its module:
\[
\Gamma(X,\widetilde M)=M.
\]
Therefore
\[
\boxed{
0\to\Gamma(X,\mathcal F')
\to\Gamma(X,\mathcal F)
\to\Gamma(X,\mathcal F'')
\to0
}
\]
is exact.

The point is that right exactness of global sections is special here.  For arbitrary sheaves, a section of the
quotient may only admit local lifts and need not have a global lift.
\end{solution}
\fi
