% VI33-DETAILED-PROBLEM-03
\begin{problem}[VI.33.P3: Closed subscheme sequence]\label{prob:vi33-03}
Let \(I\subseteq A\), let
\[
i:Z=\operatorname{Spec}(A/I)\hookrightarrow X=\operatorname{Spec}A,
\]
and derive the exact sequence
\[
0\to\widetilde I\to\mathcal O_X\to i_*\mathcal O_Z\to0.
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Begin with the exact sequence of \(A\)-modules
\[
0\longrightarrow I\longrightarrow A\longrightarrow A/I\longrightarrow0.
\]
Applying the exact tilde functor gives
\[
0\longrightarrow\widetilde I
\longrightarrow\widetilde A
\longrightarrow\widetilde{A/I}
\longrightarrow0.
\]
Since
\[
\widetilde A\cong\mathcal O_X,
\]
it remains to identify the quotient sheaf.

For a distinguished open \(D(f)\subseteq X\),
\[
\widetilde{A/I}(D(f))
=
(A/I)_f
\cong
A_f/IA_f.
\]
The inverse image
\[
Z\cap D(f)
\]
is the affine scheme
\[
\Spec(A_f/IA_f),
\]
so
\[
(i_*\mathcal O_Z)(D(f))
=
\mathcal O_Z(Z\cap D(f))
\cong
A_f/IA_f.
\]
These identifications commute with restrictions. Hence
\[
\widetilde{A/I}\cong i_*\mathcal O_Z,
\]
and therefore
\[
\boxed{
0\to\widetilde I\to\mathcal O_X\to i_*\mathcal O_Z\to0.
}
\]
Thus the ideal \(I\) becomes the quasi-coherent ideal sheaf cutting out the closed subscheme \(Z\).
\end{solution}
\fi
