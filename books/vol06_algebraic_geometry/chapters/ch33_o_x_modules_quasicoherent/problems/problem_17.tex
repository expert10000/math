% VI33-DETAILED-PROBLEM-17
\begin{problem}[VI.33.P17: Modules living on a closed subscheme]\label{prob:vi33-17}
Let \(I\subseteq A\) and \(M\) an \(A\)-module with \(IM=0\). Show that \(M\) is naturally an \(A/I\)-module and
that \(\widetilde M\) on \(\Spec A\) is the pushforward of the corresponding quasi-coherent sheaf on
\(Z=\Spec(A/I)\).

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The condition
\[
IM=0
\]
means that the action map \(A\times M\to M\) factors through the quotient \(A/I\). Thus \(M\) is canonically an
\(A/I\)-module.

Let
\[
i:Z=\Spec(A/I)\hookrightarrow X=\Spec A.
\]
Write \(\widetilde M_Z\) for the sheaf associated with \(M\) as an \(A/I\)-module.  On a distinguished open
\(D(f)\subseteq X\),
\[
(i_*\widetilde M_Z)(D(f))
=
\widetilde M_Z(D(\bar f))
=
M_{\bar f}.
\]
Localizing \(M\) as an \(A/I\)-module at \(\bar f\) is the same as localizing it as an \(A\)-module at \(f\):
\[
M_{\bar f}\cong M_f.
\]
But
\[
\widetilde M_X(D(f))=M_f.
\]
Therefore
\[
\boxed{
\widetilde M_X\cong i_*\widetilde M_Z.
}
\]
So modules annihilated by \(I\) are exactly affine quasi-coherent modules living scheme-theoretically on the
closed subscheme cut out by \(I\).
\end{solution}
\fi
