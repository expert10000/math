% VI33-DETAILED-PROBLEM-20
\begin{problem}[VI.33.P20: Support of a tensor product of finite modules]\label{prob:vi33-20}
Let \(M,N\) be finitely generated \(A\)-modules. Prove
\[
\operatorname{Supp}\bigl(\widetilde{M\otimes_A N}\bigr)
=
\operatorname{Supp}(\widetilde M)\cap\operatorname{Supp}(\widetilde N).
\]

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
At a prime \(\mathfrak p\),
\[
(M\otimes_A N)_{\mathfrak p}
\cong
M_{\mathfrak p}\otimes_{A_{\mathfrak p}}N_{\mathfrak p}.
\]
If either factor vanishes, the tensor product vanishes. Hence
\[
\operatorname{Supp}(M\otimes N)
\subseteq
\operatorname{Supp}M\cap\operatorname{Supp}N.
\]

For the reverse inclusion, suppose
\[
M_{\mathfrak p}\ne0,
\qquad
N_{\mathfrak p}\ne0.
\]
These are nonzero finite modules over the local ring
\[
R=A_{\mathfrak p}
\]
with residue field \(\kappa(\mathfrak p)\). By Nakayama,
\[
M_{\mathfrak p}/\mathfrak pM_{\mathfrak p}\ne0,
\qquad
N_{\mathfrak p}/\mathfrak pN_{\mathfrak p}\ne0.
\]
Their tensor product over the field \(\kappa(\mathfrak p)\) is nonzero. But
\[
(M_{\mathfrak p}\otimes_RN_{\mathfrak p})
\otimes_R\kappa(\mathfrak p)
\cong
(M_{\mathfrak p}\otimes_R\kappa(\mathfrak p))
\otimes_{\kappa(\mathfrak p)}
(N_{\mathfrak p}\otimes_R\kappa(\mathfrak p)),
\]
so \(M_{\mathfrak p}\otimes_RN_{\mathfrak p}\ne0\). Therefore
\[
\boxed{
\operatorname{Supp}(\widetilde{M\otimes_A N})
=
\operatorname{Supp}(\widetilde M)\cap\operatorname{Supp}(\widetilde N).
}
\]
\end{solution}
\fi
