% VI33-DETAILED-PROBLEM-04
\begin{problem}[VI.33.P4: Support of a finite module]\label{prob:vi33-04}
Let \(M\) be a finitely generated \(A\)-module. Prove
\[
\operatorname{Supp}(\widetilde M)=V(\operatorname{Ann}M).
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
A prime \(\mathfrak p\) lies in the support exactly when
\[
M_{\mathfrak p}\ne0.
\]

Suppose first that
\[
\operatorname{Ann}M\not\subseteq\mathfrak p.
\]
Choose
\[
s\in\operatorname{Ann}M\setminus\mathfrak p.
\]
Then \(s\) becomes a unit in \(A_{\mathfrak p}\) but still annihilates \(M_{\mathfrak p}\). Hence
\[
M_{\mathfrak p}=0.
\]

Conversely, suppose \(M_{\mathfrak p}=0\). Choose generators
\[
M=(m_1,\ldots,m_r).
\]
For every \(i\), the equality \(m_i/1=0\) in \(M_{\mathfrak p}\) means there exists
\[
s_i\notin\mathfrak p
\]
with
\[
s_i m_i=0.
\]
The product
\[
s=s_1\cdots s_r
\]
is still outside \(\mathfrak p\) and annihilates every generator, hence every element of \(M\). Thus
\[
s\in\operatorname{Ann}M\setminus\mathfrak p.
\]

Therefore
\[
M_{\mathfrak p}\ne0
\iff
\operatorname{Ann}M\subseteq\mathfrak p,
\]
which is exactly
\[
\boxed{\operatorname{Supp}(\widetilde M)=V(\operatorname{Ann}M)}.
\]
Finite generation is used in passing from separate annihilators \(s_i\) to one common annihilator.
\end{solution}
\fi
