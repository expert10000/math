% VI33-DETAILED-PROBLEM-11
\begin{problem}[VI.33.P11: Morphisms are module maps on affine schemes]\label{prob:vi33-11}
For \(X=\operatorname{Spec}A\), prove
\[
\operatorname{Hom}_{\mathcal O_X}(\widetilde M,\widetilde N)
\cong
\operatorname{Hom}_A(M,N).
\]
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Define
\[
\Phi:
\operatorname{Hom}_{\mathcal O_X}(\widetilde M,\widetilde N)
\to
\operatorname{Hom}_A(M,N)
\]
by taking global sections.  Since
\[
\Gamma(X,\widetilde M)=M,
\qquad
\Gamma(X,\widetilde N)=N,
\]
this is well-defined.

Conversely, an \(A\)-linear map
\[
u:M\to N
\]
localizes for every \(f\in A\):
\[
u_f:M_f\to N_f.
\]
These maps are compatible with further localization, so they define a morphism of sheaves
\[
\widetilde u:\widetilde M\to\widetilde N.
\]
Taking global sections of \(\widetilde u\) returns \(u\).

Finally, if a sheaf morphism \(\varphi\) has global-section map \(u\), then on every distinguished open
quasi-coherence forces
\[
\varphi_{D(f)}=u_f.
\]
Hence \(\varphi=\widetilde u\). Therefore
\[
\boxed{
\operatorname{Hom}_{\mathcal O_X}(\widetilde M,\widetilde N)
\cong
\operatorname{Hom}_A(M,N).
}
\]
This is full faithfulness of the affine equivalence.
\end{solution}
\fi
