% VI33-DETAILED-PROBLEM-14
\begin{problem}[VI.33.P14: Kernels and cokernels under tilde]\label{prob:vi33-14}
Let \(u:M\to N\) be \(A\)-linear. Prove
\[
\ker(\widetilde u)\cong\widetilde{\ker u},
\qquad
\operatorname{coker}(\widetilde u)\cong\widetilde{\operatorname{coker}u}.
\]

\textbf{Provenance.} Canonical VI/33 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The module sequences
\[
0\to\ker u\to M\xrightarrow{u}N
\]
and
\[
M\xrightarrow{u}N\to\operatorname{coker}u\to0
\]
are exact. Applying the exact tilde functor gives
\[
0\to\widetilde{\ker u}
\to\widetilde M
\xrightarrow{\widetilde u}
\widetilde N
\]
and
\[
\widetilde M
\xrightarrow{\widetilde u}
\widetilde N
\to
\widetilde{\operatorname{coker}u}
\to0.
\]
Exactness identifies the first sheaf with the kernel and the last with the cokernel:
\[
\boxed{
\ker(\widetilde u)\cong\widetilde{\ker u},
\qquad
\operatorname{coker}(\widetilde u)\cong\widetilde{\operatorname{coker}u}.
}
\]
The same reasoning identifies the sheaf image with the tilde of \(\operatorname{im}u\).
\end{solution}
\fi
