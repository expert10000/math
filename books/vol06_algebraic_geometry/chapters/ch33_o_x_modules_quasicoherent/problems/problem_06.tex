% VI33-DETAILED-PROBLEM-06
\begin{problem}[VI.33.P6: Finite locally free versus projective]\label{prob:vi33-06}
Let \(P\) be a finitely generated \(A\)-module. Show that \(P\) is projective if and only if
\(\widetilde P\) is finite locally free on \(\operatorname{Spec}A\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Assume first that \(P\) is finitely generated projective.  For every prime \(\mathfrak p\),
\[
P_{\mathfrak p}
\]
is a finitely generated projective module over the local ring \(A_{\mathfrak p}\), hence is free.  A basis at
\(\mathfrak p\) extends after inverting some
\[
f\notin\mathfrak p,
\]
so
\[
P_f\cong A_f^{\oplus r}.
\]
Thus
\[
\widetilde P|_{D(f)}
\cong
\mathcal O_{D(f)}^{\oplus r}.
\]
Hence \(\widetilde P\) is finite locally free.

Conversely, suppose \(\widetilde P\) is finite locally free.  The affine scheme \(\Spec A\) is quasi-compact.
Choose finitely many trivializing distinguished opens
\[
D(f_1),\ldots,D(f_n)
\]
on which
\[
P_{f_i}\cong A_{f_i}^{\oplus r_i}.
\]
Thus \(P\) is locally free of finite rank at every prime.  The finite distinguished cover also implies that \(P\)
is finitely presented: presentations can be chosen locally and patched by clearing finitely many denominators.
Each localization \(P_{\mathfrak p}\) is free, hence flat; a finitely presented flat module is projective.
Therefore \(P\) is finitely generated projective.

So
\[
\boxed{
P\text{ finitely generated projective}
\iff
\widetilde P\text{ finite locally free}.
}
\]
\end{solution}
\fi
