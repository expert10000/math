\begin{example}[The structure sheaf]
For $X=\operatorname{Spec}A$, one has $\widetilde A\cong\mathcal O_X$.
\end{example}

\begin{example}[A free module]
For $M=A^{\oplus r}$, the associated sheaf is $\widetilde M\cong\mathcal O_X^{\oplus r}$.
\end{example}

\begin{example}[A principal ideal]
For $I=(f)\subseteq A$, the sheaf $\widetilde I$ is the image of multiplication by $f$ on $\mathcal O_X$.
\end{example}

\begin{example}[A quotient module]
For $M=A/I$, the associated sheaf is supported on $V(I)$ and is the structure sheaf of the affine closed subscheme $\operatorname{Spec}(A/I)$.
\end{example}

\begin{example}[A closed point of the affine line]
Let $A=k[t]$ and $M=A/(t-a)$.  Then $\widetilde M$ is supported at the closed point $(t-a)$.
\end{example}

\begin{example}[Localization]
For $M=A/(x)$ on $X=\operatorname{Spec}k[x,y]$, restriction to $D(x)$ is zero because $M_x=0$.
\end{example}

\begin{example}[Restriction to $D(y)$]
For the same module $M=A/(x)$, one has $M_y\cong k[y,y^{-1}]$ as a module over $k[x,y]_y/(x)$.
\end{example}

\begin{example}[A torsion module]
For $A=k[t]$ and $M=A/(t^n)$, the associated coherent sheaf is supported at the origin but remembers the order-$n$ nilpotent thickness.
\end{example}

\begin{example}[Multiplication by a function]
The module map $A\xrightarrow{\cdot f}A$ gives a sheaf morphism $\mathcal O_X\xrightarrow{\cdot f}\mathcal O_X$ whose cokernel is $\widetilde{A/(f)}$.
\end{example}

\begin{example}[Tensoring quotient modules]
For ideals $I,J\subseteq A$,
\[
(A/I)\otimes_A(A/J)\cong A/(I+J),
\]
so the tensor product of the corresponding affine quasi-coherent sheaves is supported on $V(I+J)=V(I)\cap V(J)$.
\end{example}

\begin{example}[Support of a free sheaf]
If $M=A^{\oplus r}$ with $r>0$, then $\operatorname{Supp}(\widetilde M)=\operatorname{Spec}A$.
\end{example}

\begin{example}[Finite locally free from projective modules]
If $P$ is a finitely generated projective $A$-module, then $\widetilde P$ is finite locally free on $\operatorname{Spec}A$.
\end{example}

\begin{example}[Pullback along a quotient]
For $A\to A/I$ and an $A$-module $M$, the pullback to $\operatorname{Spec}(A/I)$ corresponds to
\[
(A/I)\otimes_A M\cong M/IM.
\]
\end{example}

\begin{example}[Pullback along localization]
For $A\to A_f$, pullback of $\widetilde M$ to $D(f)$ is $\widetilde{M_f}$.
\end{example}

\begin{example}[Coherent sheaf on a Noetherian affine scheme]
If $A$ is Noetherian and $M$ is finitely generated, then $\widetilde M$ is coherent.
\end{example}

\begin{example}[Global sections on affine space]
For $X=\mathbb A^n_k$ and $M=k[x_1,\ldots,x_n]/I$, global sections of $\widetilde M$ recover the module $M$ itself.
\end{example}
