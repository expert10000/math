% C07 conceptual visual: specialization in Spec.
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[>=Stealth,node distance=1.8cm]
  \node (p0) {$\mathfrak p$};
  \node[below=of p0] (p1) {$\mathfrak q$};
  \node[below=of p1] (p2) {$\mathfrak m$};
  \draw[->] (p0) -- node[right] {$\mathfrak p\subseteq\mathfrak q$} (p1);
  \draw[->] (p1) -- node[right] {$\mathfrak q\subseteq\mathfrak m$} (p2);
  \node[left=2.4cm of p1,align=center] {$\mathfrak q\in\overline{\{\mathfrak p\}}$\\
  means $\mathfrak p\subseteq\mathfrak q$};
\end{tikzpicture}
\caption{Specialization in $\operatorname{Spec}A$ runs from a smaller prime to a larger
prime. Thus the closure of $\{\mathfrak p\}$ consists exactly of the primes containing
$\mathfrak p$.}
\label{fig:vi07-c07-specialization}
\end{figure}
