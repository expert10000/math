\begin{hint}[VI.07.E1]
Translate membership in $V(I)$ into ideal containment.  For the union, use primality on $IJ$.
\end{hint}
\begin{hint}[VI.07.E2]
A prime ideal contains $I$ iff it contains every element whose power lies in $I$.
\end{hint}
\begin{hint}[VI.07.E3]
Use E2 and the identity $\sqrt I=\bigcap_{\mathfrak p\supseteq I}\mathfrak p$.
\end{hint}
\begin{hint}[VI.07.E4]
List the closed sets containing $\mathfrak p$ and identify the smallest one.
\end{hint}
\begin{hint}[VI.07.E5]
$V(\mathfrak p)=\{\mathfrak p\}$ exactly when no strictly larger prime contains $\mathfrak p$.
\end{hint}
\begin{hint}[VI.07.E6]
Distinct primes have distinct closures.
\end{hint}
\begin{hint}[VI.07.E7]
Remember that a prime ideal must be proper.
\end{hint}
\begin{hint}[VI.07.E8]
In the PID $k[t]$, nonzero prime ideals are generated by irreducible polynomials.
\end{hint}
\begin{hint}[VI.07.E9]
Irreducibles in $\RR[t]$ have degree one or two.
\end{hint}
\begin{hint}[VI.07.E10]
Every ideal of $\ZZ$ is $(n)$; compute $V(n)$ by divisibility.
\end{hint}
\begin{hint}[VI.07.E11]
A prime containing $xy$ must contain $x$ or $y$.
\end{hint}
\begin{hint}[VI.07.E12]
Use prime correspondence for $\ZZ\to\ZZ/n\ZZ$.
\end{hint}
\begin{hint}[VI.07.E13]
Use $\bar x\bar y=0$ and then the quotients by $(\bar x)$ and $(\bar y)$.
\end{hint}
\begin{hint}[VI.07.E14]
A closed set $V(J)$ contains $S$ exactly when $J$ lies in every prime of $S$.
\end{hint}
\begin{hint}[VI.07.E15]
Track a closed set $V_{A/I}(J/I)$ under prime correspondence.
\end{hint}
\begin{hint}[VI.07.E16]
Expand $I\subseteq\varphi^{-1}(\mathfrak q)$.
\end{hint}
\begin{hint}[VI.07.E17]
Surjectivity gives a bijection between primes of $B$ and primes of $A$ containing the kernel.
\end{hint}
\begin{hint}[VI.07.E18]
Use the image-closure theorem with $\ker\varphi=(0)$.
\end{hint}
\begin{hint}[VI.07.E19]
A prime ideal of $A\times B$ contains $(1,0)$ or $(0,1)$.
\end{hint}
\begin{hint}[VI.07.E20]
All primes containing $(x^m)$ also contain $(x)$.
\end{hint}
