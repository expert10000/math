\begin{example}[The spectrum of a field]
If $K$ is a field, its only proper ideal is $(0)$, and that ideal is prime. Hence
\[
\Spec K=\{(0)\}.
\]
The unique point is closed.
\end{example}

\begin{example}[The empty spectrum]
The zero ring has no proper ideals, hence
\[
\Spec(0)=\varnothing.
\]
This edge case makes $V(A)=\varnothing$ consistent with the topology.
\end{example}

\begin{example}[The affine line over an algebraically closed field]
If $k$ is algebraically closed,
\[
\Spec k[t]=\{(0)\}\cup\{(t-a):a\in k\}.
\]
The closure of $(0)$ is the whole spectrum, while each $(t-a)$ is closed.
\end{example}

\begin{example}[The real affine line has extra closed points]
In $\RR[t]$, the nonzero prime ideals are generated by irreducible polynomials:
linear factors and irreducible quadratics.  Thus $(t^2+1)$ is a closed point of
$\Spec\RR[t]$ even though it is not an $\RR$-rational point of the classical line.
\end{example}

\begin{example}[Finite-field affine line]
The points of $\Spec\FF_q[t]$ are $(0)$ and $(f)$ for monic irreducible polynomials $f$.
Only degree-one irreducibles correspond to $\FF_q$-rational points; higher-degree primes
record finite field extensions.
\end{example}

\begin{example}[Arithmetic line]
The points of $\Spec\ZZ$ are
\[
(0),(2),(3),(5),\dots.
\]
For $n\neq0$,
\[
V(n)=\{(p):p\mid n\}.
\]
Thus arithmetic divisibility appears directly as Zariski closed geometry.
\end{example}

\begin{example}[A finite arithmetic spectrum]
Prime ideals of $\ZZ/12\ZZ$ correspond to prime ideals of $\ZZ$ containing $(12)$.
Hence
\[
\Spec(\ZZ/12\ZZ)=\{(2)/(12),(3)/(12)\}.
\]
Both points are closed; a finite $T_1$ space is discrete.
\end{example}

\begin{example}[Two crossing components]
Let $R=k[x,y]/(xy)$.  The primes $(\bar x)$ and $(\bar y)$ are minimal.  The maximal ideal
$(\bar x,\bar y)$ contains both, so the origin is a specialization of both component points.
\end{example}

\begin{example}[A product ring]
Every prime of $A\times B$ is either $\mathfrak p\times B$ or $A\times\mathfrak q$.
Consequently,
\[
\Spec(A\times B)\cong\Spec A\sqcup\Spec B.
\]
\end{example}

\begin{example}[A nilpotent thickening with the same topology]
For $R=k[\varepsilon]/(\varepsilon^2)$, the only prime ideal is $(\bar\varepsilon)$.
Thus $\Spec R$ is a one-point space, just like $\Spec k$.  VI/10 explains why these
schemes are nevertheless different.
\end{example}

\begin{example}[A quotient as a closed subspace]
Let
\[
A=k[x,y],\qquad I=(y-x^2).
\]
Then
\[
\Spec(A/I)\cong V_A(y-x^2).
\]
The spectrum of the coordinate ring of the parabola is literally the corresponding
closed subspace of $\Spec A$.
\end{example}

\begin{example}[Three specialization levels in the affine plane]
For algebraically closed $k$,
\[
(0)\subsetneq (f)\subsetneq(x-a,y-b)
\]
can occur when $f$ is irreducible and $f(a,b)=0$.  Topologically this is
\[
\text{plane generic point}\rightsquigarrow
\text{curve generic point}\rightsquigarrow
\text{closed point}.
\]
\end{example}

\begin{example}[A map induced by squaring]
The homomorphism
\[
k[t]\longrightarrow k[x],\qquad t\longmapsto x^2
\]
induces
\[
\Spec k[x]\longrightarrow\Spec k[t].
\]
A closed point $(x-a)$ contracts to $(t-a^2)$, while $(0)$ contracts to $(0)$.
\end{example}

\begin{example}[The generic arithmetic point from a field inclusion]
The inclusion
\[
\ZZ\hookrightarrow\QQ
\]
induces
\[
\Spec\QQ\longrightarrow\Spec\ZZ.
\]
The unique point $(0)$ of $\Spec\QQ$ maps to the generic point $(0)$ of $\Spec\ZZ$.
\end{example}
