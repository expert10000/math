\begin{exercise}[VI.07.E1 - Verify the Zariski topology]
Prove directly that the sets $V(I)$ satisfy the closed-set axioms:
\[
V((0))=\Spec A,\quad V(A)=\varnothing,\quad
V(I)\cup V(J)=V(IJ),\quad
\bigcap_\lambda V(I_\lambda)=V(\sum_\lambda I_\lambda).
\]
\end{exercise}

\begin{exercise}[VI.07.E2 - Radical invariance]
Show
\[
V(I)=V(\sqrt I)
\]
for every ideal $I\subseteq A$.
\end{exercise}

\begin{exercise}[VI.07.E3 - Equality of closed sets]
Prove
\[
V(I)=V(J)\Longleftrightarrow \sqrt I=\sqrt J.
\]
\end{exercise}

\begin{exercise}[VI.07.E4 - Closure of one point]
For $\mathfrak p\in\Spec A$, prove
\[
\overline{\{\mathfrak p\}}=V(\mathfrak p).
\]
\end{exercise}

\begin{exercise}[VI.07.E5 - Closed points]
Show that $\{\mathfrak p\}$ is closed in $\Spec A$ iff $\mathfrak p$ is maximal.
\end{exercise}

\begin{exercise}[VI.07.E6 - The $T_0$ property]
Prove every spectrum is $T_0$.
\end{exercise}

\begin{exercise}[VI.07.E7 - Spectrum of a field and of the zero ring]
Compute $\Spec K$ for a field $K$ and $\Spec(0)$ for the zero ring.
\end{exercise}

\begin{exercise}[VI.07.E8 - Legacy CA13: points of a polynomial-ring spectrum]
Describe all points of $\Spec k[t]$.  State the answer first for arbitrary $k$, then simplify
when $k$ is algebraically closed.
\end{exercise}

\begin{exercise}[VI.07.E9 - The real affine line]
Describe the points and point closures of $\Spec\RR[t]$.
\end{exercise}

\begin{exercise}[VI.07.E10 - Legacy CA13: closed sets of $\Spec\ZZ$]
Describe all Zariski closed subsets of $\Spec\ZZ$.
\end{exercise}

\begin{exercise}[VI.07.E11 - Legacy CA13: $V(xy)$]
In $k[x,y]$, prove
\[
V(xy)=V(x)\cup V(y).
\]
Interpret the two sides through prime containment.
\end{exercise}

\begin{exercise}[VI.07.E12 - Finite arithmetic spectra]
For $n\ge2$, determine the points of $\Spec(\ZZ/n\ZZ)$ and show that the underlying
topological space is finite and discrete.
\end{exercise}

\begin{exercise}[VI.07.E13 - Crossing axes]
For $R=k[x,y]/(xy)$, prove every prime contains $\bar x$ or $\bar y$, and describe the
closures of $(\bar x)$, $(\bar y)$, and $(\bar x,\bar y)$.
\end{exercise}

\begin{exercise}[VI.07.E14 - Closure of a subset]
For $S\subseteq\Spec A$, prove
\[
\overline S
=
V\!\left(\bigcap_{\mathfrak p\in S}\mathfrak p\right).
\]
\end{exercise}

\begin{exercise}[VI.07.E15 - Quotient spectrum]
Show that the prime correspondence gives a homeomorphism
\[
\Spec(A/I)\cong V_A(I).
\]
\end{exercise}

\begin{exercise}[VI.07.E16 - Continuity of contraction]
For $\varphi:A\to B$, prove
\[
(\varphi^\ast)^{-1}(V_A(I))=V_B(IB).
\]
\end{exercise}

\begin{exercise}[VI.07.E17 - Surjections]
If $\varphi:A\twoheadrightarrow B$, show that $\varphi^\ast$ is a homeomorphism
$\Spec B\cong V(\ker\varphi)$.
\end{exercise}

\begin{exercise}[VI.07.E18 - Dense image from an injection]
If $\varphi:A\hookrightarrow B$ is injective, prove the image of
$\Spec B\to\Spec A$ is dense.
\end{exercise}

\begin{exercise}[VI.07.E19 - Product spectra]
Prove
\[
\Spec(A\times B)\cong\Spec A\sqcup\Spec B
\]
as topological spaces.
\end{exercise}

\begin{exercise}[VI.07.E20 - Topology ignores nilpotent powers]
Compare
\[
\Spec k[x]/(x)
\qquad\text{and}\qquad
\Spec k[x]/(x^m),\quad m\ge2,
\]
as topological spaces.  Do not yet claim the corresponding schemes are the same.
\end{exercise}
