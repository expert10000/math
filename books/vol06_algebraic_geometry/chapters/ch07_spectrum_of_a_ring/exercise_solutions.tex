\begin{solution}[VI.07.E1]
The extremes follow because every prime contains $(0)$ and no proper prime contains $1$.
A prime contains $IJ$ iff it contains $I$ or $J$, giving $V(IJ)=V(I)\cup V(J)$.
A prime contains $\sum I_\lambda$ iff it contains every $I_\lambda$, giving the intersection formula.
\end{solution}

\begin{solution}[VI.07.E2]
If $\mathfrak p$ contains $I$ and $a\in\sqrt I$, then $a^n\in I\subseteq\mathfrak p$ for some $n$.
Since $\mathfrak p$ is prime, $a\in\mathfrak p$.  Thus $\mathfrak p$ contains $\sqrt I$.
The reverse implication is immediate from $I\subseteq\sqrt I$.
\end{solution}

\begin{solution}[VI.07.E3]
By E2, equality of radicals implies equality of closed sets. Conversely,
\[
\sqrt I=\bigcap_{\mathfrak p\in V(I)}\mathfrak p,
\]
and similarly for $J$, so equal closed sets give equal intersections.
\end{solution}

\begin{solution}[VI.07.E4]
A closed set $V(I)$ contains $\mathfrak p$ iff $I\subseteq\mathfrak p$.  Then
$V(\mathfrak p)\subseteq V(I)$ by inclusion reversal, so $V(\mathfrak p)$ is the smallest
closed set containing $\mathfrak p$.
\end{solution}

\begin{solution}[VI.07.E5]
The point is closed iff $V(\mathfrak p)=\{\mathfrak p\}$.  This means there is no prime
properly containing $\mathfrak p$, which is equivalent to maximality.
\end{solution}

\begin{solution}[VI.07.E6]
If $\mathfrak p\neq\mathfrak q$, then their closures $V(\mathfrak p)$ and $V(\mathfrak q)$
are different because prime ideals are radical and a closed set determines its radical ideal.
Thus the points are topologically distinguishable, which is the $T_0$ condition.
\end{solution}

\begin{solution}[VI.07.E7]
A field has exactly one proper ideal, $(0)$, and it is prime.  The zero ring has no proper
ideals, so its spectrum is empty.
\end{solution}

\begin{solution}[VI.07.E8]
Since $k[t]$ is a PID, its primes are $(0)$ and $(f)$ with $f$ irreducible.  If $k$ is
algebraically closed, the irreducibles are linear, so the nonzero primes are $(t-a)$.
\end{solution}

\begin{solution}[VI.07.E9]
The primes are $(0)$, the ideals $(t-a)$, and ideals generated by irreducible quadratics.
The closure of $(0)$ is the whole spectrum.  Every nonzero prime is maximal in the PID
$\RR[t]$, hence is closed.
\end{solution}

\begin{solution}[VI.07.E10]
The closed sets are $V(0)=\Spec\ZZ$, $V(1)=\varnothing$, and for $n\neq0$,
\[
V(n)=\{(p):p\text{ prime and }p\mid n\}.
\]
Hence every proper closed subset is a finite set of prime-number points.
\end{solution}

\begin{solution}[VI.07.E11]
A prime $\mathfrak p$ contains $xy$ iff $x\in\mathfrak p$ or $y\in\mathfrak p$.
Therefore $\mathfrak p\in V(x)$ or $\mathfrak p\in V(y)$, proving the identity.
\end{solution}

\begin{solution}[VI.07.E12]
Primes of $\ZZ/n\ZZ$ correspond to primes $(p)\subseteq\ZZ$ containing $(n)$, i.e. to
prime divisors $p$ of $n$.  All such primes are maximal.  Thus the spectrum is finite and $T_1$,
hence discrete.
\end{solution}

\begin{solution}[VI.07.E13]
Every prime contains $\bar x$ or $\bar y$.  Primes containing $\bar x$ correspond to primes
of $k[y]$ and those containing $\bar y$ to primes of $k[x]$.  The closure of $(\bar x)$ is
$V(\bar x)$, similarly for $(\bar y)$, while $(\bar x,\bar y)$ is maximal and hence closed.
\end{solution}

\begin{solution}[VI.07.E14]
A closed set $V(J)$ contains $S$ iff $J\subseteq\mathfrak p$ for every $\mathfrak p\in S$,
iff $J\subseteq\bigcap_{\mathfrak p\in S}\mathfrak p$.  Inclusion reversal now shows the
smallest closed set containing $S$ is the asserted $V$.
\end{solution}

\begin{solution}[VI.07.E15]
Primes of $A/I$ are exactly $\mathfrak p/I$ with $\mathfrak p\supseteq I$.  A closed set
$V_{A/I}(J/I)$ corresponds to $V_A(J)$ inside $V_A(I)$, so the bijection preserves closed
sets and is a homeomorphism onto $V_A(I)$.
\end{solution}

\begin{solution}[VI.07.E16]
For $\mathfrak q\in\Spec B$,
\[
\varphi^\ast(\mathfrak q)\in V_A(I)
\iff I\subseteq\varphi^{-1}(\mathfrak q)
\iff \varphi(I)\subseteq\mathfrak q
\iff IB\subseteq\mathfrak q.
\]
\end{solution}

\begin{solution}[VI.07.E17]
Prime correspondence for the quotient $A/\ker\varphi\cong B$ gives the bijection onto
$V(\ker\varphi)$; E15 proves it is a homeomorphism.
\end{solution}

\begin{solution}[VI.07.E18]
The closure of the image equals $V(\ker\varphi)$.  If $\varphi$ is injective, this is
$V(0)=\Spec A$.
\end{solution}

\begin{solution}[VI.07.E19]
If $P$ is prime in $A\times B$, then
\[
(1,0)(0,1)=0\in P,
\]
so $(1,0)\in P$ or $(0,1)\in P$.  This forces $P=\mathfrak p\times B$ or
$P=A\times\mathfrak q$.  The two families are clopen and carry the original spectra's
topologies.
\end{solution}

\begin{solution}[VI.07.E20]
Primes of $k[x]/(x^m)$ correspond to primes of $k[x]$ containing $(x^m)$, equivalently
containing $(x)$.  Thus there is one point, as for $k[x]/(x)\cong k$.  The topological
spaces agree, although the rings differ when $m\ge2$.
\end{solution}
