\begin{exercise}[Homogeneous decomposition]\label{exr:vi34-01}
Prove uniqueness of the homogeneous decomposition of an element of a graded ring.
\end{exercise}

\begin{exercise}[Degrees of products]\label{exr:vi34-02}
For nonzero homogeneous \(a,b\) with \(ab\neq0\), prove \(\deg(ab)=\deg a+\deg b\).
\end{exercise}

\begin{exercise}[Standard grading dimensions]\label{exr:vi34-03}
For \(S=k[x,y,z]\), compute \(\dim_kS_d\).
\end{exercise}

\begin{exercise}[Component criterion]\label{exr:vi34-04}
Prove that an ideal generated by homogeneous elements contains every homogeneous component of each of its elements.
\end{exercise}

\begin{exercise}[Nonhomogeneous ideal]\label{exr:vi34-05}
Show \((x+1)\subset k[x]\) is not homogeneous.
\end{exercise}

\begin{exercise}[Graded quotient]\label{exr:vi34-06}
Verify that a quotient by a homogeneous ideal inherits a grading.
\end{exercise}

\begin{exercise}[Homogeneous radical]\label{exr:vi34-07}
Prove \(\sqrt I\) is homogeneous when \(I\) is homogeneous.
\end{exercise}

\begin{exercise}[Homogeneous prime test]\label{exr:vi34-08}
Prove the homogeneous primality test from the chapter.
\end{exercise}

\begin{exercise}[Irrelevant ideal]\label{exr:vi34-09}
Compute \(S_+\) for \(k[x_0,\ldots,x_n]\) with standard grading.
\end{exercise}

\begin{exercise}[Weighted irrelevant ideal]\label{exr:vi34-10}
For \(k[x,y]\) with \(\deg x=2,\deg y=3\), compute \(S_+\) and \(S_1\).
\end{exercise}

\begin{exercise}[Graded kernels]\label{exr:vi34-11}
Show the kernel of a degree-zero map of graded modules is homogeneous.
\end{exercise}

\begin{exercise}[Shift convention]\label{exr:vi34-12}
Under \(M(d)_n=M_{n+d}\), find the new degree of an element originally in \(M_r\).
\end{exercise}

\begin{exercise}[Multiplication map]\label{exr:vi34-13}
For homogeneous \(f\in S_d\), verify \(S(-d)\xrightarrow{\cdot f}S\) has degree zero.
\end{exercise}

\begin{exercise}[Localization degree]\label{exr:vi34-14}
Show the grading on \(S_f\) is independent of the homogeneous fraction representation.
\end{exercise}

\begin{exercise}[Compute \(S_{(x)}\)]\label{exr:vi34-15}
Prove \(k[x,y]_{(x)}=k[y/x]\).
\end{exercise}

\begin{exercise}[Three-variable chart]\label{exr:vi34-16}
Compute \(k[x_0,x_1,x_2]_{(x_2)}\).
\end{exercise}

\begin{exercise}[Quadric cone chart]\label{exr:vi34-17}
For \(S=k[x,y,z]/(xy-z^2)\), compute \(S_{(x)}\).
\end{exercise}

\begin{exercise}[Module localization]\label{exr:vi34-18}
Show \(M_{(f)}\) is an \(S_{(f)}\)-module.
\end{exercise}

\begin{exercise}[Exactness after localization]\label{exr:vi34-19}
Show homogeneous localization is exact in graded modules.
\end{exercise}

\begin{exercise}[Degree-zero functor]\label{exr:vi34-20}
Show taking the degree-zero part of an exact sequence of graded modules with degree-zero maps is exact.
\end{exercise}

\begin{exercise}[Second Veronese]\label{exr:vi34-21}
Show \(k[x,y]^{(2)}\cong k[u,v,w]/(uw-v^2)\).
\end{exercise}

\begin{exercise}[Weighted degree 12]\label{exr:vi34-22}
With \(\deg x=2,\deg y=3\), list monomials of weighted degree \(12\).
\end{exercise}

\begin{exercise}[Cone invariance]\label{exr:vi34-23}
Prove the zero locus of a homogeneous ideal is stable under scalar multiplication.
\end{exercise}

\begin{exercise}[Why degree zero?]\label{exr:vi34-24}
Explain why \(S_f\) is too large for a projective chart and \((S_f)_0\) is natural.
\end{exercise}
