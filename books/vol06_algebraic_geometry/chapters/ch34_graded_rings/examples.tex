\begin{example}[Standard polynomial grading]\label{ex:vi34-01}
For \(S=k[x,y,z]\) with \(\deg x=\deg y=\deg z=1\), \(x^2+yz\) is homogeneous of degree two, while \(x+y^2\) is not.
\end{example}

\begin{example}[A homogeneous ideal]\label{ex:vi34-02}
The ideal \(I=(x^2,xy,y^3)\subset k[x,y]\) is homogeneous, so the quotient inherits the standard grading.
\end{example}

\begin{example}[A nonhomogeneous ideal]\label{ex:vi34-03}
The ideal \((x+1)\subset k[x]\) is not homogeneous because it contains \(x+1\) but not its components \(x\) and \(1\).
\end{example}

\begin{example}[Homogeneous prime]\label{ex:vi34-04}
In \(k[x,y,z]\), \((x,y)\) is homogeneous and prime since the quotient is \(k[z]\).
\end{example}

\begin{example}[Irrelevant ideal]\label{ex:vi34-05}
For \(k[x_0,x_1,x_2]\) standard graded, \(S_+=(x_0,x_1,x_2)\) and \(S/S_+\cong k\).
\end{example}

\begin{example}[A graded cone quotient]\label{ex:vi34-06}
The quotient \(k[x,y,z]/(xy-z^2)\) is graded because the relation is homogeneous of degree two.
\end{example}

\begin{example}[Shifted free module]\label{ex:vi34-07}
With \(S(-2)_n=S_{n-2}\), the element \(1\in S_0\) has degree \(2\) in \(S(-2)\). Multiplication by a quadratic gives a degree-zero map \(S(-2)\to S\).
\end{example}

\begin{example}[Graded hypersurface sequence]\label{ex:vi34-08}
If homogeneous \(f\in S_d\) is a non-zero-divisor, then
\[
0\to S(-d)\xrightarrow{\cdot f}S\to S/(f)\to0
\]
is exact in graded modules.
\end{example}

\begin{example}[Homogeneous localization]\label{ex:vi34-09}
For \(S=k[x,y]\), \(y^3/x^2\in S_x\) has degree \(1\), while \(y^2/x^2\) has degree \(0\).
\end{example}

\begin{example}[A standard chart ring]\label{ex:vi34-10}
For \(S=k[x_0,x_1,x_2]\),
\[
S_{(x_0)}=k[x_1/x_0,x_2/x_0]\cong k[u,v].
\]
\end{example}

\begin{example}[A quadratic cone chart]\label{ex:vi34-11}
For \(S=k[x,y,z]/(xy-z^2)\), localizing at \(x\) and setting \(u=z/x\) gives \(y/x=u^2\), hence \(S_{(x)}\cong k[u]\).
\end{example}

\begin{example}[Localized shifted module]\label{ex:vi34-12}
For \(S=k[x,y]\) and \(M=S(-1)\), \(M_{(x)}\) is free of rank one over \(k[y/x]\).
\end{example}

\begin{example}[Second Veronese]\label{ex:vi34-13}
\[
k[x,y]^{(2)}=k[x^2,xy,y^2]\cong k[u,v,w]/(uw-v^2).
\]
\end{example}

\begin{example}[Weighted homogeneous polynomial]\label{ex:vi34-14}
If \(\deg x=2\) and \(\deg y=3\), then \(x^3+y^2\) is weighted homogeneous of degree six.
\end{example}

\begin{example}[Cone scaling]\label{ex:vi34-15}
If \(F\) is homogeneous of degree \(d\), then \(F(\lambda a)=\lambda^dF(a)\), so \(V(F)\) is stable under scalar multiplication.
\end{example}

\begin{example}[Ordinary versus degree-zero localization]\label{ex:vi34-16}
\(k[x,y]_x\) contains \(x^{-1}\) of degree \(-1\), whereas \(k[x,y]_{(x)}\) retains only degree-zero expressions such as \(y/x\).
\end{example}
