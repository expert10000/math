% VI34-DETAILED-PROBLEM-07
\begin{problem}[VI.34.P7: Exact degree-zero localization]\label{prob:vi34-07}
Prove
\[
0\to M'_{(f)}\to M_{(f)}\to M''_{(f)}\to0
\]
for a short exact sequence of graded modules.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Start with
\[
0\to M'\to M\to M''\to0
\]
with degree-zero maps. Ordinary localization is exact:
\[
0\to M'_f\to M_f\to M''_f\to0.
\]
Because \(f\) is homogeneous, each localization is graded and the maps still have degree zero. Exactness of a
graded sequence is degreewise, so taking the degree-zero pieces gives
\[
\boxed{
0\to(M'_f)_0\to(M_f)_0\to(M''_f)_0\to0.
}
\]
By definition these are \(M'_{(f)},M_{(f)},M''_{(f)}\).
\end{solution}
\fi
