% VI34-DETAILED-PROBLEM-09
\begin{problem}[VI.34.P9: Weighted localization]\label{prob:vi34-09}
Let \(S=k[x,y]\) with \(\deg x=2,\deg y=3\). Describe \(S_{(x)}\) and \(S_{(y)}\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
A degree-zero monomial in \(S_x\) has the form
\[
\frac{x^ay^b}{x^r}
\]
with
\[
2a+3b=2r.
\]
Thus \(3b\) is even, so \(b=2c\). Then
\[
2a+6c=2r,\qquad r=a+3c,
\]
and
\[
\frac{x^ay^{2c}}{x^{a+3c}}
=
\left(\frac{y^2}{x^3}\right)^c.
\]
Hence
\[
\boxed{S_{(x)}=k[y^2/x^3]}.
\]

Similarly a degree-zero monomial over a power of \(y\) satisfies
\[
2a+3b=3r,
\]
so \(a=3c\), and every such fraction is a power of \(x^3/y^2\). Therefore
\[
\boxed{S_{(y)}=k[x^3/y^2]}.
\]
\end{solution}
\fi
