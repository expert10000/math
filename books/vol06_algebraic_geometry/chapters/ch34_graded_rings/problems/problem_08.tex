% VI34-DETAILED-PROBLEM-08
\begin{problem}[VI.34.P8: Veronese quadratic cone]\label{prob:vi34-08}
Prove
\[
k[x,y]^{(2)}\cong k[u,v,w]/(uw-v^2)
\]
and interpret the affine spectrum.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The second Veronese consists of even-degree forms and is generated, after regrading, by
\[
x^2,\quad xy,\quad y^2.
\]
Define
\[
\phi:k[u,v,w]\to k[x,y]^{(2)}
\]
by \(u\mapsto x^2\), \(v\mapsto xy\), \(w\mapsto y^2\). It is surjective and
\[
uw-v^2
\]
lies in the kernel.

Modulo \(uw-v^2\), every monomial can be reduced so that it contains either no \(u\) or no \(w\). Their images are
distinct monomials in \(x,y\), so there is no further relation. Hence
\[
\boxed{k[x,y]^{(2)}\cong k[u,v,w]/(uw-v^2)}.
\]
Its spectrum is the affine quadric cone
\[
V(uw-v^2)\subseteq\mathbb A^3_k.
\]
\end{solution}
\fi
