% VI34-DETAILED-PROBLEM-04
\begin{problem}[VI.34.P4: Standard chart algebra]\label{prob:vi34-04}
For \(S=k[x_0,\ldots,x_n]\), prove \(S_{(x_i)}\cong k[t_1,\ldots,t_n]\) by explicit inverse maps.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let the variables \(t_j\) correspond to the indices \(j\neq i\). Define
\[
\Phi:k[t_j:j\neq i]\longrightarrow S_{(x_i)},
\qquad
t_j\longmapsto \frac{x_j}{x_i}.
\]
Each image has degree zero.

For the inverse, take a degree-zero fraction
\[
\frac{F}{x_i^d},
\qquad
F\in S_d.
\]
Set
\[
\Psi\!\left(\frac{F}{x_i^d}\right)
=
F(t_0,\ldots,t_{i-1},1,t_{i+1},\ldots,t_n),
\]
where \(x_j/x_i\) is replaced by \(t_j\). Homogeneity of \(F\) ensures this exactly rewrites \(F/x_i^d\).
The formula is compatible with fraction equality and gives a ring map.

Clearly \(\Psi\Phi\) fixes every \(t_j\), and \(\Phi\Psi\) fixes every homogeneous degree-zero fraction. Hence
\[
\boxed{S_{(x_i)}\cong k[t_1,\ldots,t_n]}.
\]
\end{solution}
\fi
