\begin{problem}[VI.34.P4: Standard chart algebra]\label{prob:vi34-04}
For \(S=k[x_0,\ldots,x_n]\), prove \(S_{(x_i)}\cong k[t_1,\ldots,t_n]\) by explicit inverse maps.
\begin{solution}
Send the polynomial variables to \(x_j/x_i\). Conversely write a degree-zero fraction \(F/x_i^d\) as \(F(x_0/x_i,\ldots,1,\ldots,x_n/x_i)\).
\end{solution}
\end{problem}
