% VI34-DETAILED-PROBLEM-17
\begin{problem}[VI.34.P17: A weighted chart that is not affine space]\label{prob:vi34-17}
Let
\[
S=k[x,y,z],\qquad \deg x=\deg y=1,\quad \deg z=2.
\]
Compute \(S_{(z)}\) and identify its affine spectrum.

\textbf{Provenance.} Canonical VI/34 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Consider a degree-zero monomial fraction whose denominator is \(z^r\). Its numerator has weighted degree
\(2r\). Such numerators are generated by the degree-\(2\) monomials
\[
x^2,\quad xy,\quad y^2,\quad z.
\]
After division by \(z\), set
\[
u=\frac{x^2}{z},\qquad
v=\frac{xy}{z},\qquad
w=\frac{y^2}{z}.
\]
These satisfy
\[
uw=v^2.
\]
Every degree-zero fraction is a polynomial in \(u,v,w\), and the same monomial-reduction argument as for the second
Veronese shows this is the only relation. Hence
\[
\boxed{
S_{(z)}
\cong
k[u,v,w]/(uw-v^2).
}
\]
Thus the \(z\)-chart is the affine quadric cone, not affine \(2\)-space. Weighted projective charts can therefore
carry quotient/singularity phenomena absent from the standard polynomial grading.
\end{solution}
\fi
