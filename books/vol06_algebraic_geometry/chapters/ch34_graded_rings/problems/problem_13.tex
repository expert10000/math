% VI34-DETAILED-PROBLEM-13
\begin{problem}[VI.34.P13: Homogeneous localization in stages]\label{prob:vi34-13}
Let \(f,g\) be homogeneous of positive degrees. Prove the equal-power overlap formula that compares the degree-zero
rings obtained by localizing first at \(f\) and then along the \(g\)-direction.

\textbf{Provenance.} Canonical VI/34 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Write
\[
\deg f=e,\qquad \deg g=d.
\]
Choose positive integers \(a,b\) such that
\[
ae=bd.
\]
Then
\[
\frac{g^b}{f^a}
\]
has degree zero in \(S_f\), hence lies in \(S_{(f)}\).

Localizing \(S_{(f)}\) at this element permits denominators built from both \(f\) and \(g\), while every resulting
fraction still has net degree zero. On the other hand, localizing \(S\) at the homogeneous product \(f^ag^b\)
inverts both \(f\) and \(g\), and taking degree zero imposes exactly the same condition. The mutually inverse maps
are induced by the identity on \(S\). Hence
\[
\boxed{
(S_{(f)})_{g^b/f^a}\cong S_{(f^ag^b)}.
}
\]
When \(d=e\), take \(a=b=1\) to recover
\[
(S_{(f)})_{g/f}\cong S_{(fg)}.
\]
\end{solution}
\fi
