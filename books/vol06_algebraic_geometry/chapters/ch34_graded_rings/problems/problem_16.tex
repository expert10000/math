% VI34-DETAILED-PROBLEM-16
\begin{problem}[VI.34.P16: Veronese chart equality]\label{prob:vi34-16}
Let \(S^{(d)}=\bigoplus_{n\ge0}S_{nd}\). If \(f\in S_{md}\) is homogeneous, compare the chart rings obtained from
\(S\) and \(S^{(d)}\).

\textbf{Provenance.} Canonical VI/34 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Regard \(f\) as an element of degree \(m\) in the regraded Veronese ring \(S^{(d)}\). Then
\[
(S^{(d)})_{(f)}
=
\left((S^{(d)})_f\right)_0.
\]
A degree-zero fraction on the left has the form
\[
\frac a{f^r}
\]
with
\[
a\in(S^{(d)})_{rm}=S_{rmd}.
\]
But this is exactly the condition that the same fraction have degree zero in \(S_f\), since
\[
\deg_S(f^r)=rmd.
\]
Thus the two degree-zero subrings consist of the same fractions:
\[
\boxed{
(S^{(d)})_{(f)}\cong S_{(f)}.
}
\]

If \(h\in S\) has degree not divisible by \(d\), then \(h^d\in S^{(d)}\), and
\[
S_{(h)}=S_{(h^d)}\cong(S^{(d)})_{(h^d)}.
\]
This is the chart-level reason Veronese regrading does not change the later projective geometry.
\end{solution}
\fi
