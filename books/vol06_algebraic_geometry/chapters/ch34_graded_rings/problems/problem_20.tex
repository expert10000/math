% VI34-DETAILED-PROBLEM-20
\begin{problem}[VI.34.P20: Degree zero as the intrinsic invariant piece]\label{prob:vi34-20}
Explain intrinsically why the degree-zero subring is the part of a graded localization invariant under the scaling
action, without relying on the size of the ground field.

\textbf{Provenance.} Canonical VI/34 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
A \(\mathbb Z\)-grading on a \(k\)-algebra \(R=\bigoplus_nR_n\) determines the algebraic scaling coaction
\[
\rho:R\longrightarrow R[t,t^{-1}],
\qquad
r_n\longmapsto r_nt^n
\]
for homogeneous \(r_n\in R_n\). This is the coordinate-ring form of the \(\mathbb G_m\)-action.

An element \(r=\sum_nr_n\) is invariant exactly when
\[
\rho(r)=r\otimes1.
\]
But
\[
\rho(r)=\sum_nr_nt^n,
\]
and uniqueness of Laurent-polynomial coefficients forces \(r_n=0\) for every \(n\neq0\). Hence the invariant
subring is exactly
\[
\boxed{R^{\mathbb G_m}=R_0}.
\]

Applying this to the graded localization \(R=S_f\) gives
\[
\boxed{S_{(f)}=(S_f)_0=(S_f)^{\mathbb G_m}}.
\]
This formulation remains correct over every base field, including finite fields, and avoids accidental
pointwise-scaling exceptions.
\end{solution}
\fi
