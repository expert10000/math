% VI34-DETAILED-PROBLEM-06
\begin{problem}[VI.34.P6: Graded hypersurface sequence]\label{prob:vi34-06}
Let \(f\in S_d\) be homogeneous and a non-zero-divisor. Prove
\[
0\to S(-d)\xrightarrow{\cdot f}S\to S/(f)\to0
\]
is exact in graded modules.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
As an ungraded sequence, injectivity of multiplication by \(f\) follows from the non-zero-divisor hypothesis,
its image is \((f)\), and the cokernel is \(S/(f)\). Hence the sequence is exact.

It remains to check degrees. Under the convention
\[
S(-d)_n=S_{n-d},
\]
an element \(a\in S(-d)_n\) lies originally in \(S_{n-d}\). Multiplication gives
\[
fa\in S_{d+n-d}=S_n.
\]
Thus the first map has degree zero. The quotient map is degree zero because \((f)\) is homogeneous. Therefore the
sequence is exact in the graded category.
\end{solution}
\fi
