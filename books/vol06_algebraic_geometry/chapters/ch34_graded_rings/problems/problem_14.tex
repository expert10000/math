% VI34-DETAILED-PROBLEM-14
\begin{problem}[VI.34.P14: Module overlap algebra]\label{prob:vi34-14}
Under the hypotheses of P13, prove the corresponding formula
\[
(M_{(f)})_{g^b/f^a}\cong M_{(f^ag^b)}
\]
for a graded \(S\)-module \(M\).

\textbf{Provenance.} Canonical VI/34 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Start from
\[
M_{(f)}=(M_f)_0.
\]
Because \(g^b/f^a\) has degree zero, localizing \(M_{(f)}\) by this element is an ordinary localization over the
ring \(S_{(f)}\). A typical fraction has the form
\[
\frac{m/f^r}{(g^b/f^a)^N}.
\]
Rewriting it in \(M_{f^ag^b}\) gives the same element with both \(f\) and \(g\) inverted. Since the localizing
element has degree zero, the degree of the numerator fraction is unchanged, so the degree-zero condition is exactly
the one defining \(M_{(f^ag^b)}\).

The identity on \(M\) therefore induces inverse maps and yields
\[
\boxed{
(M_{(f)})_{g^b/f^a}\cong M_{(f^ag^b)}.
}
\]
These module identifications are compatible with the ring identifications of P13, which is the algebraic input
needed for sheaf gluing on projective overlaps.
\end{solution}
\fi
