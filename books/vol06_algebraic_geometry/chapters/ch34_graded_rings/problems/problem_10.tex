% VI34-DETAILED-PROBLEM-10
\begin{problem}[VI.34.P10: Cone vertex and irrelevant ideal]\label{prob:vi34-10}
For standard graded \(S=k[x_0,\ldots,x_n]/I\) over a field with \(I\) homogeneous and proper, prove \(S_+\) cuts out the cone vertex.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The positive-degree ideal of \(S\) is the image of
\[
(x_0,\ldots,x_n).
\]
Since \(I\) is homogeneous and proper, \(I\cap k=0\), so
\[
S/S_+\cong k.
\]
Thus \(S_+\) is maximal. Under the affine cone embedding
\[
\Spec S\hookrightarrow\mathbb A^{n+1}_k,
\]
this maximal ideal is the image of the origin ideal \((x_0,\ldots,x_n)\). Hence its point is the cone vertex:
\[
\boxed{V(S_+)=\{\text{vertex}\}}
\]
in the standard affine-cone situation.
\end{solution}
\fi
