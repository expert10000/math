% VI34-DETAILED-PROBLEM-05
\begin{problem}[VI.34.P5: A conic chart]\label{prob:vi34-05}
Let \(S=k[x,y,z]/(xz-y^2)\). Compute \(S_{(x)}\), \(S_{(z)}\), and their common overlap.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
All variables have degree \(1\). On the \(x\)-chart set
\[
u=\frac yx.
\]
Dividing the relation by \(x^2\) gives
\[
\frac zx=u^2.
\]
Hence
\[
S_{(x)}\cong k[u].
\]

On the \(z\)-chart set
\[
v=\frac yz.
\]
Dividing by \(z^2\) gives
\[
\frac xz=v^2,
\]
so
\[
S_{(z)}\cong k[v].
\]

On the overlap \(x\neq0\) and \(z\neq0\), the relation \(xz=y^2\) gives
\[
uv=\frac yx\frac yz=\frac{y^2}{xz}=1.
\]
Thus \(v=u^{-1}\), and both chart rings localize to
\[
\boxed{k[u,u^{-1}]}.
\]
This is the algebraic overlap that VI/35 will glue.
\end{solution}
\fi
