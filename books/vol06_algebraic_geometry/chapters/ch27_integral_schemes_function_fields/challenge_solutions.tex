% VI27-DETAILED-CHALLENGE-01
\subsection*{Challenge 1 solution}
\begin{solution}
Let
\[
X=U_1\cup\cdots\cup U_r,\qquad U_i=\Spec A_i,
\]
be an affine cover of the integral scheme \(X\). Every nonempty overlap \(U_i\cap U_j\) contains the generic
point \(\eta\). Choose a nonempty affine refinement
\[
W_{ij}=\Spec C_{ij}\subseteq U_i\cap U_j.
\]
The restriction maps
\[
A_i\to C_{ij},\qquad A_j\to C_{ij}
\]
are injective, because regular functions on a nonempty open of an integral scheme inject into the generic stalk.
Moreover \(W_{ij}\) is a nonempty open of each integral affine chart, so localization does not change the fraction
field:
\[
\Frac(A_i)\cong\Frac(C_{ij})\cong\Frac(A_j).
\]
Define \(\phi_{ij}\) by this composite.

The map is independent of the chosen affine refinement. Two refinements have nonempty intersection containing
\(\eta\), and after restricting to a common principal subopen both constructions give the same germ at \(\eta\).
Likewise, on a triple overlap the three identifications factor through a common affine refinement, so
\[
\phi_{jk}\circ\phi_{ij}=\phi_{ik}.
\]
Thus the \(\phi_{ij}\) satisfy the cocycle condition.

Consequently the fraction fields of the charts canonically identify with one field. This field is exactly the
generic stalk:
\[
K(X)=\mathcal O_{X,\eta}.
\]
Hence \(K(X)\) is reconstructed from the affine atlas and its restriction maps alone.
\end{solution}

% VI27-DETAILED-CHALLENGE-02
\subsection*{Challenge 2 solution}
\begin{solution}
Let \(A\) be a domain with \(K=\Frac(A)\). Clearly
\[
A\subseteq\bigcap_{\mathfrak m\in\operatorname{Max}A}A_{\mathfrak m}
\]
inside \(K\). We prove the reverse inclusion.

Take \(x\in K\setminus A\), and define
\[
I_x=\{a\in A:ax\in A\}.
\]
This is a proper ideal, since \(1\in I_x\) would imply \(x\in A\). Choose a maximal ideal
\(\mathfrak m\supseteq I_x\). If \(x\in A_{\mathfrak m}\), write
\[
x=\frac as,\qquad s\notin\mathfrak m.
\]
Then \(sx=a\in A\), so \(s\in I_x\subseteq\mathfrak m\), contradiction. Therefore
\(x\notin A_{\mathfrak m}\). Thus every element of \(K\setminus A\) is excluded from at least one maximal
localization, and
\[
\boxed{
A=\bigcap_{\mathfrak m\in\operatorname{Max}A}A_{\mathfrak m}
}
\]
inside \(K\).

Geometrically, a rational function on the integral affine scheme \(\Spec A\) is globally regular exactly when it
is regular at every closed point.
\end{solution}

% VI27-DETAILED-CHALLENGE-03
\subsection*{Challenge 3 solution}
\begin{solution}
Dominance gives an injective \(k\)-field map
\[
f^*:K(Y)\hookrightarrow K(X).
\]
Let \(y_1,\ldots,y_r\in K(Y)\). If
\[
P(y_1,\ldots,y_r)=0
\]
for some nonzero \(P\in k[T_1,\ldots,T_r]\), then applying \(f^*\) gives the same relation among the images.
Conversely, a polynomial relation among the images pulls back through the injective map and therefore was already
a relation in \(K(Y)\). Hence algebraic independence over \(k\) is preserved and reflected by \(f^*\).

A transcendence basis of \(K(Y)/k\) therefore maps to an algebraically independent subset of \(K(X)\). Hence
\[
\operatorname{trdeg}_kK(Y)\le\operatorname{trdeg}_kK(X).
\]
Equality need not hold: the projection
\[
\mathbb A^2_k\to\mathbb A^1_k
\]
gives \(k(t)\hookrightarrow k(t,u)\). If \(K(X)\) is algebraic over the image of \(K(Y)\), then the
transcendence degrees are equal. This is a field-theoretic statement and does not use the VI/30 dimension theorem.
\end{solution}

% VI27-DETAILED-CHALLENGE-04
\subsection*{Challenge 4 solution}
\begin{solution}
Take \(k=\mathbb R\) and
\[
X=\Spec\mathbb C
\]
with its natural \(\mathbb R\)-scheme structure. Since \(\mathbb C\) is a field, \(X\) is integral and
\[
K(X)=\mathbb C.
\]
After base change to \(\mathbb C\),
\[
\mathbb C\otimes_{\mathbb R}\mathbb C
\cong
\mathbb C[T]/(T^2+1)
\cong
\mathbb C[T]/((T-i)(T+i))
\cong
\mathbb C\times\mathbb C.
\]
Therefore
\[
X_{\mathbb C}\cong\Spec\mathbb C\amalg\Spec\mathbb C,
\]
which is reducible. Thus \(X\) is integral but not geometrically integral.

The fact that \(K(X)\) is a field only records ordinary integrality over the original ground field. Geometric
integrality is a statement about scalar extensions. One must inspect the base-changed coordinate ring, or
equivalently here
\[
K(X)\otimes_{\mathbb R}\mathbb C,
\]
which splits.
\end{solution}

% VI27-DETAILED-CHALLENGE-05
\subsection*{Challenge 5 solution}
\begin{solution}
Let
\[
A=k[t^2,t^3]\subseteq k[t]\subseteq k(t).
\]
Since
\[
t=\frac{t^3}{t^2},
\]
we have
\[
K(X)=\Frac(A)=k(t).
\]
The element \(t\) is not in \(A\), because the exponents appearing in polynomials in \(t^2,t^3\) lie in
\(\{0,2,3,4,\ldots\}\). However \(t\) is integral over \(A\), since it satisfies
\[
T^2-t^2=0
\]
with \(t^2\in A\). Hence
\[
A[t]=k[t]
\]
is integral over \(A\).

Now let \(z\in k(t)\) be integral over \(A\). The same monic equation witnessing integrality over \(A\) also has
coefficients in the larger ring \(k[t]\), so \(z\) is integral over \(k[t]\). The PID \(k[t]\) is integrally
closed in its fraction field \(k(t)\), hence \(z\in k[t]\). Conversely, every element of \(k[t]=A[t]\) is integral
over \(A\), because \(t\) is integral. Therefore
\[
\boxed{
\overline A^{\,k(t)}=k[t].
}
\]
Thus the affine normalization predicted from the function field is
\[
\Spec k[t]\longrightarrow\Spec k[t^2,t^3].
\]
VI/28 proves this integral-closure construction is the general normalization.
\end{solution}
