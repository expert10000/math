\begin{example}[Advanced 1: Cusp]\label{advex:vi27-01}
For $A=k[t^2,t^3]\subset k[t]$, $A$ is a domain and $K(\Spec A)=k(t)$.  The fact that $A$ is not integrally closed belongs to VI/28.
\end{example}

\begin{example}[Advanced 2: Node as an integral curve]\label{advex:vi27-02}
If $y^2-x^2(x+1)$ is irreducible over $k$, the coordinate ring of the nodal affine curve is a domain even though the curve is singular.
\end{example}

\begin{example}[Advanced 3: Field extension as function-field extension]\label{advex:vi27-03}
A dominant morphism of integral curves can induce a nontrivial finite field extension $K(Y)\subset K(X)$; degree becomes a birational invariant under additional hypotheses.
\end{example}

\begin{example}[Advanced 4: Purely transcendental field]\label{advex:vi27-04}
The function field of $\mathbb A^n_k$ has transcendence basis $t_1,\dots,t_n$ over $k$.
\end{example}

\begin{example}[Advanced 5: Generic regularity]\label{advex:vi27-05}
Given $a/b\in\Frac(A)$ with $A$ a domain and $b\ne0$, the function is regular on $D(b)$, a nonempty dense open.
\end{example}

\begin{example}[Advanced 6: Intersection of local rings]\label{advex:vi27-06}
For a domain $A$, every $A_{\mathfrak p}$ embeds in $\Frac(A)$.  Their intersections inside the fraction field encode where a rational function is simultaneously regular.
\end{example}

\begin{example}[Advanced 7: Dominance from injectivity in the affine case]\label{advex:vi27-07}
For a map $A\to B$ of domains, $\Spec B\to\Spec A$ is dominant iff the kernel is zero; hence dominance is equivalent to an injection $A\hookrightarrow B$.
\end{example}

\begin{example}[Advanced 8: Generic point under dominance]\label{advex:vi27-08}
If $f:X\to Y$ is dominant between integral schemes, continuity gives $\overline{\{f(\eta_X)\}}=Y$, so $f(\eta_X)=\eta_Y$.
\end{example}

\begin{example}[Advanced 9: Non-dominant map and absent field map]\label{advex:vi27-09}
A morphism landing in a proper closed subset need not yield $K(Y)\to K(X)$ because the generic point of $Y$ is not hit.
\end{example}

\begin{example}[Advanced 10: Integral scheme over a field]\label{advex:vi27-10}
If $X$ is integral of finite type over $k$, the structure map gives an embedding $k\hookrightarrow K(X)$, making $K(X)$ a finitely generated field extension of $k$.
\end{example}

\begin{example}[Advanced 11: Transcendence degree survives affine shrinking]\label{advex:vi27-11}
Since $A_f$ and $A$ have the same fraction field for $f\ne0$, their associated function field has the same transcendence degree.
\end{example}

\begin{example}[Advanced 12: Reduced but not geometrically integral]\label{advex:vi27-12}
An integral $k$-scheme can become reducible after a field extension.  Geometric integrality is a stronger base-change condition and is not assumed here.
\end{example}

\begin{example}[Advanced 13: Integral does not mean normal]\label{advex:vi27-13}
$k[t^2,t^3]$ is a domain, so its spectrum is integral, but it is not normal.  Integrality is therefore strictly weaker than normality.
\end{example}

\begin{example}[Advanced 14: Rational functions form a field]\label{advex:vi27-14}
Nonzero germs at the generic point are invertible because $\OO_{X,\eta}$ is the localization of a domain at $(0)$.
\end{example}
