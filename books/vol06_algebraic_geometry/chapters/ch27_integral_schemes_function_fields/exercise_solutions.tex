% VI27 detailed exercise solutions -- VI/24+ proof-depth standard.

% VI27-DETAILED-EXERCISE-01
\paragraph{Solution 1.}
Let \(U\subseteq X\) be a nonempty open subset of an integral scheme. Since \(X\) is irreducible,
any two nonempty open subsets of \(X\) meet. If \(\overline U\ne X\), then
\[
V:=X\setminus\overline U
\]
would be a nonempty open subset disjoint from \(U\), contradiction. Hence \(\overline U=X\).

% VI27-DETAILED-EXERCISE-02
\paragraph{Solution 2.}
Let \(U\subseteq X\) be nonempty and open, with \(X\) integral. Reducedness is local and is preserved
by restriction to an open subscheme, so \(U\) is reduced. A nonempty open subspace of an
irreducible space is irreducible. Therefore \(U\) is reduced, irreducible, and nonempty, hence
integral.

% VI27-DETAILED-EXERCISE-03
\paragraph{Solution 3.}
If \(A\) is a domain, \((0)\) is prime. Its closure in \(\Spec A\) is
\[
V((0))=\Spec A,
\]
so \((0)\) is the generic point.

% VI27-DETAILED-EXERCISE-04
\paragraph{Solution 4.}
For \(X=\Spec k[x,y]\), the generic point is \((0)\). Therefore
\[
\mathcal O_{X,(0)}=k[x,y]_{(0)}.
\]
Localizing at \((0)\) inverts every nonzero polynomial, hence
\[
k[x,y]_{(0)}=\Frac(k[x,y])=k(x,y).
\]

% VI27-DETAILED-EXERCISE-05
\paragraph{Solution 5.}
Since \(\mathbb A^2_k=\Spec k[x,y]\) and \(k[x,y]\) is a domain,
\[
K(\mathbb A^2_k)=\Frac k[x,y]=k(x,y).
\]

% VI27-DETAILED-EXERCISE-06
\paragraph{Solution 6.}
For \(X=V(y-x^3)\),
\[
k[x,y]/(y-x^3)\cong k[x]
\]
by substituting \(y=x^3\). Hence \(X\) is integral and
\[
K(X)\cong k(x).
\]

% VI27-DETAILED-EXERCISE-07
\paragraph{Solution 7.}
Let \(A\) be a domain and \(0\ne f\in A\). Since \(A_f\subseteq\Frac(A)\),
\(\Frac(A_f)\subseteq\Frac(A)\). Conversely \(A\subseteq A_f\), so every fraction of elements of
\(A\) lies in \(\Frac(A_f)\). Thus
\[
\Frac(A_f)=\Frac(A).
\]

% VI27-DETAILED-EXERCISE-08
\paragraph{Solution 8.}
Put \(A=k[x,y]/(xy)\). The classes of \(x\) and \(y\) are nonzero but multiply to zero, so \(A\)
is not a domain. It is reduced because
\[
(xy)=(x)\cap(y)
\]
is an intersection of prime ideals and hence radical. Thus \(\Spec A\) is reduced but reducible,
so not integral.

% VI27-DETAILED-EXERCISE-09
\paragraph{Solution 9.}
For \(A=k[\varepsilon]/(\varepsilon^2)\), the spectrum has one point, so it is irreducible.
But \(\varepsilon\ne0\) and \(\varepsilon^2=0\), so \(A\) is not reduced. Hence the scheme is
irreducible but not integral.

% VI27-DETAILED-EXERCISE-10
\paragraph{Solution 10.}
The scheme \(\Spec k\amalg\Spec k\) has integral components, but it is the union of two disjoint
nonempty open-and-closed subsets. Hence it is reducible and therefore not integral.

% VI27-DETAILED-EXERCISE-11
\paragraph{Solution 11.}
Let \(U\subseteq X\) be nonempty and \(\eta\) the generic point. Consider
\[
\rho_U:\mathcal O_X(U)\to K(X)=\mathcal O_{X,\eta}.
\]
If \(s\) maps to zero, then \(s\) vanishes on some neighborhood \(V\) of \(\eta\).
Take a nonempty affine \(W=\Spec A\subseteq U\). Then \(V\cap W\ne\varnothing\), so it contains
a principal open \(D(f)\) with \(f\ne0\). If \(s|_W\) corresponds to \(a\in A\), then
\(a/1=0\) in \(A_f\), hence \(f^na=0\) for some \(n\). Since \(A\) is a domain, \(a=0\).
Thus \(s=0\) on every affine chart, and \(\rho_U\) is injective.

% VI27-DETAILED-EXERCISE-12
\paragraph{Solution 12.}
If \(s,t\in\mathcal O_X(U)\) agree on a nonempty open \(V\subseteq U\), then \(\eta\in V\), so
their generic germs agree. Injectivity of
\[
\mathcal O_X(U)\hookrightarrow K(X)
\]
gives \(s=t\).

% VI27-DETAILED-EXERCISE-13
\paragraph{Solution 13.}
For \(r=a/b\in\Frac(A)\) with \(b\ne0\), the denominator \(b\) is invertible on
\(D(b)\). Hence
\[
r=a\cdot b^{-1}\in A_b=\mathcal O_X(D(b)).
\]
Thus \(r\) is regular on the nonempty dense open \(D(b)\).

% VI27-DETAILED-EXERCISE-14
\paragraph{Solution 14.}
If \(r\in K(X)\) is regular at \(x\), then by the definition of a stalk it is represented by a
section on some neighborhood \(U_x\) of \(x\). The same section shows regularity at every point of
\(U_x\). Therefore the regular locus is the union of such neighborhoods and is open. It is
nonempty by Solution~13.

% VI27-DETAILED-EXERCISE-15
\paragraph{Solution 15.}
For a domain \(A\) and \(\mathfrak p\in\Spec A\), define
\[
A_{\mathfrak p}\to\Frac(A),\qquad a/s\mapsto a/s.
\]
If two localization fractions agree, a nonzero multiplier outside \(\mathfrak p\) annihilates their
cross-difference; the domain condition forces the cross-difference itself to vanish. Hence the map
is well defined and injective.

% VI27-DETAILED-EXERCISE-16
\paragraph{Solution 16.}
For dominant \(f:X\to Y\),
\[
f(X)=f(\overline{\{\eta_X\}})\subseteq\overline{\{f(\eta_X)\}}.
\]
Taking closures and using \(\overline{f(X)}=Y\) gives
\[
\overline{\{f(\eta_X)\}}=Y.
\]
Thus \(f(\eta_X)\) is the generic point of \(Y\), so \(f(\eta_X)=\eta_Y\).

% VI27-DETAILED-EXERCISE-17
\paragraph{Solution 17.}
At the generic points, a dominant morphism induces
\[
\mathcal O_{Y,\eta_Y}\to\mathcal O_{X,\eta_X}.
\]
These are the fields \(K(Y)\) and \(K(X)\). A unital map of fields has zero kernel, so
\[
K(Y)\hookrightarrow K(X).
\]

% VI27-DETAILED-EXERCISE-18
\paragraph{Solution 18.}
For \(\varphi:A\to B\), the closure of the image of \(\Spec B\to\Spec A\) is
\[
V(\ker\varphi).
\]
Thus dominance is equivalent to \(\ker\varphi\subseteq\sqrt{(0)}\). Since \(A\) is a domain,
\(\sqrt{(0)}=(0)\). Therefore dominance is equivalent to injectivity of \(A\to B\).

% VI27-DETAILED-EXERCISE-19
\paragraph{Solution 19.}
The map \(u\mapsto t^3\) defines an injection \(k[u]\hookrightarrow k[t]\), hence a dominant map
\(\mathbb A^1_t\to\mathbb A^1_u\). Passing to fraction fields gives
\[
k(u)\hookrightarrow k(t),\qquad u\mapsto t^3.
\]

% VI27-DETAILED-EXERCISE-20
\paragraph{Solution 20.}
A proper closed immersion \(Z\hookrightarrow X\) has closed image \(|Z|\subsetneq|X|\). Its image
cannot be dense, so the morphism is not dominant. Equivalently, a proper closed subset of an
integral scheme cannot contain the generic point.

% VI27-DETAILED-EXERCISE-21
\paragraph{Solution 21.}
The coordinate ring
\[
k[x,y]/(y-x^2)\cong k[x].
\]
Therefore its function field is
\[
k(x).
\]

% VI27-DETAILED-EXERCISE-22
\paragraph{Solution 22.}
We have \(k[t^2,t^3]\subseteq k[t]\subseteq k(t)\). Inside the fraction field of
\(k[t^2,t^3]\),
\[
t=\frac{t^3}{t^2}.
\]
Hence that fraction field contains \(k[t]\) and therefore \(k(t)\). Thus
\[
\Frac(k[t^2,t^3])=k(t).
\]

% VI27-DETAILED-EXERCISE-23
\paragraph{Solution 23.}
Choose nonempty affine \(U=\Spec A\subseteq X\). Since \(X\) is finite type over \(k\),
\[
A=k[a_1,\ldots,a_n].
\]
Since \(A\) is a domain,
\[
K(X)=\Frac(A)=k(a_1,\ldots,a_n),
\]
so \(K(X)/k\) is a finitely generated field extension.

% VI27-DETAILED-EXERCISE-24
\paragraph{Solution 24.}
A nonempty open \(U\subseteq X\) has the same generic point \(\eta\), hence
\[
K(U)=\mathcal O_{U,\eta}=\mathcal O_{X,\eta}=K(X).
\]
Therefore
\[
\operatorname{trdeg}_kK(U)=\operatorname{trdeg}_kK(X).
\]

% VI27-DETAILED-EXERCISE-25
\paragraph{Solution 25.}
\(K(X)\) is a field. Therefore every \(0\ne r\in K(X)\) has an inverse. If \(r=a/b\) on an
affine chart, then \(a\ne0\) and
\[
r^{-1}=b/a.
\]

% VI27-DETAILED-EXERCISE-26
\paragraph{Solution 26.}
Take
\[
X=\Spec k[t^2,t^3].
\]
The coordinate ring is a domain, so \(X\) is integral. By Solution~22,
\[
K(X)=k(t).
\]
The element \(t\in k(t)\) is integral over \(k[t^2,t^3]\), since it satisfies
\[
T^2-t^2=0,
\]
but \(t\notin k[t^2,t^3]\). Hence the coordinate ring is not integrally closed, so \(X\) is not
normal.
