\noindent\textbf{Exercise 1.} If $U,V$ are nonempty opens with $V$ the complement of the closure of $U$, irreducibility forces $V=\varnothing$.\par

\noindent\textbf{Exercise 2.} Nonempty opens of an irreducible space are irreducible, and reducedness is local.\par

\noindent\textbf{Exercise 3.} For a domain the zero ideal is prime and its closure is all of the spectrum.\par

\noindent\textbf{Exercise 4.} The generic point is $(0)$, so the stalk is $k[x,y]_{(0)}=k(x,y)$.\par

\noindent\textbf{Exercise 5.} $K(\mathbb A^2_k)=\Frac k[x,y]=k(x,y)$.\par

\noindent\textbf{Exercise 6.} $k[x,y]/(y-x^3)\cong k[x]$, hence the function field is $k(x)$.\par

\noindent\textbf{Exercise 7.} Both are the smallest field containing $A$, because $f$ is already invertible in $\Frac(A)$.\par

\noindent\textbf{Exercise 8.} The ring has zero divisors, hence is not a domain; it is reduced because $(xy)$ is radical.\par

\noindent\textbf{Exercise 9.} The spectrum has one point, hence is irreducible, but $\varepsilon$ is nonzero nilpotent.\par

\noindent\textbf{Exercise 10.} $\Spec k\amalg\Spec k$ is reducible despite each component being integral.\par

\noindent\textbf{Exercise 11.} A section with zero generic germ vanishes on a nonempty open; its nonvanishing locus, if nonempty, would intersect that open.\par

\noindent\textbf{Exercise 12.} Their difference has zero generic germ and hence is zero by injectivity.\par

\noindent\textbf{Exercise 13.} On $D(b)$, $b$ is invertible, so $a/b$ is regular.\par

\noindent\textbf{Exercise 14.} Regularity is local, so the union of all representative opens is open.\par

\noindent\textbf{Exercise 15.} Map $a/s$ to the same fraction in $\Frac(A)$; the map is injective because $A$ is a domain.\par

\noindent\textbf{Exercise 16.} The closure of $f(\eta_X)$ contains the dense image of $X$, so it is all of $Y$.\par

\noindent\textbf{Exercise 17.} The generic stalk map is a unital homomorphism of fields, hence injective.\par

\noindent\textbf{Exercise 18.} The image closure is $V(\ker(A\to B))$; in a domain it is all of $\Spec A$ iff the kernel is zero.\par

\noindent\textbf{Exercise 19.} The induced embedding is $k(u)\hookrightarrow k(t)$, $u\mapsto t^3$.\par

\noindent\textbf{Exercise 20.} A proper closed subset cannot contain the generic point of an integral scheme.\par

\noindent\textbf{Exercise 21.} The coordinate ring is isomorphic to $k[x]$, so its fraction field is $k(x)$.\par

\noindent\textbf{Exercise 22.} Clearly $k[t^2,t^3]\subset k[t]$; also $t=t^3/t^2$ in the fraction field, giving equality $k(t)$.\par

\noindent\textbf{Exercise 23.} On $U=\Spec A$, write $A=k[a_1,\dots,a_n]$; then $K(X)=k(a_1,\dots,a_n)$.\par

\noindent\textbf{Exercise 24.} A nonempty open has the same generic point and hence the same generic stalk.\par

\noindent\textbf{Exercise 25.} $K(X)$ is a field, so every nonzero element has a multiplicative inverse.\par

\noindent\textbf{Exercise 26.} $X=\Spec k[t^2,t^3]$ is integral but not normal; $K(X)=k(t)$.\par
