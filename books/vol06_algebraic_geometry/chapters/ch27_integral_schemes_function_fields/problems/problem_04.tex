% VI27-DETAILED-PROBLEM-04
\begin{problem}[Legacy AG8-P2c: the function-field slice of the variety dimension problem]\label{prob:vi27-p04}
Let \(X\) be an integral finite-type \(k\)-scheme and \(U=\Spec A\subseteq X\) nonempty affine. Show that \(K(X)=\Frac(A)\) is a finitely generated field extension of \(k\) and that \(\operatorname{trdeg}_kK(X)\) is independent of \(U\). State how this interfaces with the later equality \(\dim X=\operatorname{trdeg}_kK(X)\).
\par\smallskip\noindent\textit{Provenance.} Function-field/transcendence-degree slice of split legacy AG8-P2; dimension and codimension slices remain reserved for VI/30--VI/32.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Because \(X\) is finite type over \(k\), write
\[
A=k[a_1,\ldots,a_n].
\]
Since \(A\) is a domain,
\[
K(X)=\Frac(A)=k(a_1,\ldots,a_n),
\]
so \(K(X)/k\) is finitely generated as a field extension. If \(V=\Spec B\) is another nonempty affine chart,
then
\[
\Frac(A)\cong\mathcal O_{X,\eta}\cong\Frac(B)
\]
canonically as \(k\)-fields. Hence their transcendence degrees over \(k\) agree. VI/27 therefore establishes
\(\operatorname{trdeg}_kK(X)\) as a chart-independent invariant. The further equality
\[
\dim X=\operatorname{trdeg}_kK(X)
\]
is the dimension theorem and remains reserved for VI/30.
\end{solution}
\fi
