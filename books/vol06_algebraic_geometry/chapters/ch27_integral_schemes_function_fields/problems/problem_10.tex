\begin{problem}[Local rings inside the function field]\label{prob:vi27-p10}
For \(x\in X\) integral, construct the canonical injection \(\mathcal O_{X,x}\hookrightarrow K(X)\).
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Choose affine \(U=\Spec A\) containing \(x\leftrightarrow\mathfrak p\). Then
\[
\mathcal O_{X,x}=A_{\mathfrak p},
\qquad
K(X)=\Frac(A).
\]
The localization map \(A_{\mathfrak p}\to\Frac(A)\), \(a/s\mapsto a/s\), is injective because \(A\) is a
domain. This is independent of the chart because both descriptions factor through the intrinsic generic stalk
\(\mathcal O_{X,\eta}\). Hence every local ring canonically embeds into \(K(X)\).
\end{solution}
\fi
