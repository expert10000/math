\begin{problem}[Dominance sends generic to generic]\label{prob:vi27-p11}
Let $f:X\to Y$ be dominant between integral schemes.  Show $f(\eta_X)=\eta_Y$.
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}
\begin{solution}
Continuity gives $f(\overline{\{\eta_X\}})\subseteq\overline{\{f(\eta_X)\}}$.  Since the left side is $f(X)$ and its closure is $Y$ by dominance, $\overline{\{f(\eta_X)\}}=Y$.  Uniqueness of the generic point gives the claim.
\end{solution}
