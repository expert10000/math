\begin{problem}[Function field of an irreducible hypersurface]\label{prob:vi27-p15}
Let \(F\in k[x_1,\dots,x_n]\) generate a prime ideal. Compute the function field of \(X=V(F)\).
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Put
\[
A=k[x_1,\ldots,x_n]/(F).
\]
The hypothesis that \((F)\) is prime means \(A\) is a domain, hence \(X\) is integral. Therefore
\[
K(X)=\Frac(A)
=
\Frac\!\left(k[x_1,\ldots,x_n]/(F)\right).
\]
If the equation eliminates a coordinate, this field can simplify. For instance \(F=y-x^2\) gives \(A\cong k[x]\)
and \(K(X)\cong k(x)\).
\end{solution}
\fi
