\begin{problem}[Function field of an irreducible hypersurface]\label{prob:vi27-p15}
Let $F\in k[x_1,\dots,x_n]$ generate a prime ideal.  Compute the function field of $X=V(F)$.
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}
\begin{solution}
$X=\Spec A$ with $A=k[x_1,\dots,x_n]/(F)$ a domain.  Therefore $K(X)=\Frac(A)$.
\end{solution}
