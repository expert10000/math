\begin{problem}[Affine dominance criterion]\label{prob:vi27-p13}
For a homomorphism of domains \(A\to B\), show \(\Spec B\to\Spec A\) is dominant iff \(A\to B\) is injective.
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
For any ring map \(\varphi:A\to B\),
\[
\overline{\operatorname{im}(\Spec B\to\Spec A)}=V(\ker\varphi).
\]
Thus dominance is equivalent to \(V(\ker\varphi)=\Spec A\), i.e.
\[
\ker\varphi\subseteq\sqrt{(0)}.
\]
Since \(A\) is a domain, \(\sqrt{(0)}=(0)\). Hence the map on spectra is dominant iff \(\ker\varphi=0\), which is
exactly injectivity of \(A\to B\).
\end{solution}
\fi
