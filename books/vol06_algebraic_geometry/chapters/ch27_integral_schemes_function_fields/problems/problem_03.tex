% VI27-DETAILED-PROBLEM-03
\begin{problem}[Legacy AG6-P7b: function field and generic stalk]\label{prob:vi27-p03}
Let \(X\) be integral with generic point \(\eta\). Prove that \(\mathcal O_{X,\eta}\) is a field and that for every nonempty affine open \(U=\Spec A\subseteq X\) one has \(\mathcal O_{X,\eta}\cong\Frac(A)\).
\par\smallskip\noindent\textit{Provenance.} Function-field half of split legacy AG6-P7; the generic-point half was already merged into VI/09.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Every nonempty affine open \(U=\Spec A\) contains \(\eta\). Since \(X\) is integral, \(A\) is a domain, and
\(\eta\) corresponds to \((0)\subset A\). Therefore
\[
\mathcal O_{X,\eta}=\mathcal O_{U,\eta}=A_{(0)}.
\]
The localization \(A_{(0)}\) inverts every nonzero element of the domain \(A\), so
\[
A_{(0)}=\Frac(A),
\]
which is a field. If \(V=\Spec B\) is another nonempty affine chart, then both \(\Frac(A)\) and \(\Frac(B)\)
identify with the same intrinsic stalk \(\mathcal O_{X,\eta}\), yielding the canonical chart-independent
function field.
\end{solution}
\fi
