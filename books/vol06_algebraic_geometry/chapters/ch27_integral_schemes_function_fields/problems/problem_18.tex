\begin{problem}[The field map from a power map]\label{prob:vi27-p18}
For \(f:\mathbb A^1_k\to\mathbb A^1_k\) induced by \(k[u]\to k[t]\), \(u\mapsto t^m\), compute the induced map on function fields.
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The ring map
\[
k[u]\to k[t],\qquad u\mapsto t^m
\]
is injective, since \(P(t^m)\ne0\) for every nonzero \(P(u)\). Hence \(f\) is dominant. Passing to fraction fields
gives
\[
f^*:k(u)\hookrightarrow k(t),\qquad u\mapsto t^m.
\]
Thus
\[
f^*\!\left(\frac{P(u)}{Q(u)}\right)=\frac{P(t^m)}{Q(t^m)}.
\]
\end{solution}
\fi
