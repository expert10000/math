% VI27-DETAILED-PROBLEM-20
\begin{problem}[Why normalization needs \(K(X)\)]\label{prob:vi27-p20}
Explain why the normalization of an integral affine scheme naturally lives inside its function field.
\par\smallskip\noindent\textit{Provenance.} Canonical bridge to VI/28.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
For \(X=\Spec A\) integral,
\[
K(X)=\Frac(A).
\]
The elements that should be added in normalization are precisely the elements \(z\in K(X)\) satisfying a monic
equation
\[
z^n+a_{n-1}z^{n-1}+\cdots+a_0=0,\qquad a_i\in A.
\]
Thus the integral closure
\[
\overline A=\{z\in K(X):z\text{ is integral over }A\}
\]
is naturally a subring of the same function field containing \(A\). For the cusp \(A=k[t^2,t^3]\), the missing
element \(t\) already lies in \(K(X)=k(t)\) and is integral over \(A\). VI/28 proves that these affine integral
closures glue to the normalization; VI/27 supplies the common ambient field.
\end{solution}
\fi
