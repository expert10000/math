% VI27-DETAILED-PROBLEM-05
\begin{problem}[Affine criterion for integrality]\label{prob:vi27-p05}
Prove that a nonempty scheme is integral iff it admits an affine open cover by spectra of domains and is irreducible.
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
If \(X\) is integral, every nonempty affine open is integral, hence has a domain as coordinate ring, and \(X\)
is irreducible. Conversely, suppose \(X\) is irreducible and has an affine cover \(\Spec A_i\) with every \(A_i\)
a domain. Domains are reduced, so each chart is reduced. Reducedness is local, hence \(X\) is reduced. Together
with nonemptiness and irreducibility, \(X\) is integral. The irreducibility hypothesis is essential:
\(\Spec k\amalg\Spec k\) has an affine cover by domains but is reducible.
\end{solution}
\fi
