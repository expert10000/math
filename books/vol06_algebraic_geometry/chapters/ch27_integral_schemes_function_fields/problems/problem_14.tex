\begin{problem}[Function field of affine space]\label{prob:vi27-p14}
Compute \(K(\mathbb A^n_k)\).
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The coordinate ring is the domain
\[
k[t_1,\ldots,t_n].
\]
The generic point is the zero ideal, so
\[
K(\mathbb A^n_k)
=
k[t_1,\ldots,t_n]_{(0)}
=
\Frac(k[t_1,\ldots,t_n])
=
k(t_1,\ldots,t_n).
\]
\end{solution}
\fi
