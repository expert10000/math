\begin{problem}[Shrinking preserves transcendence degree]\label{prob:vi27-p17}
Let \(X\) be integral finite type over \(k\) and \(U\subseteq X\) nonempty open. Show \(\operatorname{trdeg}_kK(U)=\operatorname{trdeg}_kK(X)\).
\par\smallskip\noindent\textit{Provenance.} Canonical VI/27 dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The nonempty open \(U\) has the same generic point \(\eta\) as \(X\). Hence
\[
K(U)=\mathcal O_{U,\eta}=\mathcal O_{X,\eta}=K(X)
\]
canonically as \(k\)-fields. Therefore their transcendence degrees over \(k\) are equal. No dimension theorem is
needed.
\end{solution}
\fi
