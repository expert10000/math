\begin{example}[Affine line]\label{ex:vi27-01}
For $X=\Spec k[t]$, the zero ideal is prime, $X$ is integral, and $K(X)=k(t)$.
\end{example}

\begin{example}[Affine space]\label{ex:vi27-02}
$\mathbb A^n_k=\Spec k[t_1,\dots,t_n]$ is integral and has $K(X)=k(t_1,\dots,t_n)$.
\end{example}

\begin{example}[Parabola]\label{ex:vi27-03}
$k[x,y]/(y-x^2)\cong k[x]$, so the parabola is integral and its function field is $k(x)$.
\end{example}

\begin{example}[Irreducible hypersurface]\label{ex:vi27-04}
If $F\in k[x_1,\dots,x_n]$ generates a prime ideal, then $\Spec k[x_1,\dots,x_n]/(F)$ is integral.
\end{example}

\begin{example}[Two axes]\label{ex:vi27-05}
$\Spec k[x,y]/(xy)$ is reduced when $k$ is a field, but it is reducible and therefore not integral.
\end{example}

\begin{example}[Doubled point]\label{ex:vi27-06}
$\Spec k[\varepsilon]/(\varepsilon^2)$ is irreducible but nonreduced, hence not integral.
\end{example}

\begin{example}[Disjoint integral pieces]\label{ex:vi27-07}
$\Spec k\amalg\Spec k$ has field-valued local rings but is reducible; local domains alone do not imply integrality.
\end{example}

\begin{example}[A dense principal open]\label{ex:vi27-08}
For integral $X=\Spec A$ and $0\ne f\in A$, $D(f)$ is nonempty, dense, integral, and has the same function field $\Frac(A)$.
\end{example}

\begin{example}[A rational function]\label{ex:vi27-09}
On $\mathbb A^1_k$, $1/t\in k(t)$ is regular exactly on $D(t)$, not at the closed point $(t)$.
\end{example}

\begin{example}[Agreement on a dense open]\label{ex:vi27-10}
If $f,g\in A$ for a domain $A$ and their restrictions agree on some nonempty $D(h)$, then $h^N(f-g)=0$ for some $N$; hence $f=g$.
\end{example}

\begin{example}[Localization does not change fractions]\label{ex:vi27-11}
For a domain $A$ and $0\ne f\in A$, $\Frac(A_f)=\Frac(A)$ canonically.
\end{example}

\begin{example}[Generic stalk]\label{ex:vi27-12}
At the generic point $(0)$ of $\Spec A$ for a domain $A$, the local ring is $A_{(0)}=\Frac(A)$.
\end{example}

\begin{example}[Dominant polynomial map]\label{ex:vi27-13}
The map $\mathbb A^1_k\to\mathbb A^1_k$, $t\mapsto t^m$, is dominant and induces $k(u)\hookrightarrow k(t)$ by $u\mapsto t^m$.
\end{example}

\begin{example}[Closed immersion is usually not dominant]\label{ex:vi27-14}
A proper closed subscheme of an integral scheme does not contain the generic point, so its inclusion is not dominant.
\end{example}

\begin{example}[Function field after shrinking]\label{ex:vi27-15}
Removing finitely many closed points from an integral curve leaves the function field unchanged.
\end{example}

\begin{example}[Integral finite-type curve]\label{ex:vi27-16}
If $A=k[x,y]/(F)$ is a finitely generated domain, then $K(\Spec A)=\Frac(A)$ and its transcendence degree over $k$ is chart-independent.
\end{example}

\begin{example}[Finite extension]\label{ex:vi27-17}
For $X=\Spec k[t]$ and $Y=\Spec k[t^m]$, the dominant finite map gives $K(Y)=k(t^m)\subset k(t)=K(X)$.
\end{example}

\begin{example}[Generic point belongs to every nonempty open]\label{ex:vi27-18}
If a nonempty open $U$ missed $\eta$, then its complement would be a proper closed set containing $\eta$, contradicting density of $\eta$.
\end{example}

\begin{example}[Irreducibility of a nonempty open]\label{ex:vi27-19}
A nonempty open subset of an irreducible space is irreducible; together with reducedness this makes every nonempty open subscheme of an integral scheme integral.
\end{example}

\begin{example}[Same rational function, different formulas]\label{ex:vi27-20}
On overlapping affine charts, two fractions represent the same element of $K(X)$ precisely when their germs agree at the generic point.
\end{example}
