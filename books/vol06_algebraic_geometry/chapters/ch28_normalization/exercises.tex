\begin{exercise}[VI.28.E1: Normal domain]\label{exer:vi28-01}
State the definition of a normal domain.
\end{exercise}

\begin{exercise}[VI.28.E2: Affine normality]\label{exer:vi28-02}
Show $\Spec A$ is normal when $A$ is integrally closed.
\end{exercise}

\begin{exercise}[VI.28.E3: Field]\label{exer:vi28-03}
Show every field is normal.
\end{exercise}

\begin{exercise}[VI.28.E4: Localization]\label{exer:vi28-04}
Show a localization of a normal domain is normal.
\end{exercise}

\begin{exercise}[VI.28.E5: Fraction fields]\label{exer:vi28-05}
Show $\Frac(\overline A)=\Frac(A)$.
\end{exercise}

\begin{exercise}[VI.28.E6: Integral morphism]\label{exer:vi28-06}
Why is $\Spec\overline A\to\Spec A$ integral?
\end{exercise}

\begin{exercise}[VI.28.E7: Surjectivity]\label{exer:vi28-07}
Why is the affine normalization morphism surjective?
\end{exercise}

\begin{exercise}[VI.28.E8: Normal fixed point]\label{exer:vi28-08}
Show a normal affine scheme equals its normalization.
\end{exercise}

\begin{exercise}[VI.28.E9: Cusp element]\label{exer:vi28-09}
For $A=k[t^2,t^3]$, prove $t$ is integral over $A$.
\end{exercise}

\begin{exercise}[VI.28.E10: Cusp closure]\label{exer:vi28-10}
Show $k[t]$ is the integral closure of $k[t^2,t^3]$ in $k(t)$.
\end{exercise}

\begin{exercise}[VI.28.E11: Principal open]\label{exer:vi28-11}
Compute the normalization of $D(f)\subseteq\Spec A$ from $\overline A$.
\end{exercise}

\begin{exercise}[VI.28.E12: Open restriction]\label{exer:vi28-12}
Explain why normalization restricts to nonempty opens.
\end{exercise}

\begin{exercise}[VI.28.E13: Generic point]\label{exer:vi28-13}
Show $X$ and $X^\nu$ have corresponding generic points.
\end{exercise}

\begin{exercise}[VI.28.E14: Function field]\label{exer:vi28-14}
Show $K(X^\nu)=K(X)$.
\end{exercise}

\begin{exercise}[VI.28.E15: Idempotence]\label{exer:vi28-15}
Show normalization is idempotent.
\end{exercise}

\begin{exercise}[VI.28.E16: UFD]\label{exer:vi28-16}
Explain why a UFD is normal.
\end{exercise}

\begin{exercise}[VI.28.E17: Polynomial ring]\label{exer:vi28-17}
Deduce that $\mathbb A^n_k$ is normal.
\end{exercise}

\begin{exercise}[VI.28.E18: Quadratic integer]\label{exer:vi28-18}
Show $(1+\sqrt5)/2$ is integral over $\mathbb Z[\sqrt5]$.
\end{exercise}

\begin{exercise}[VI.28.E19: Node preimages]\label{exer:vi28-19}
Under $x=t^2-1$, $y=t(t^2-1)$, find the preimages of $(0,0)$.
\end{exercise}

\begin{exercise}[VI.28.E20: Reduction comparison]\label{exer:vi28-20}
Give a reduced nonnormal affine scheme.
\end{exercise}

\begin{exercise}[VI.28.E21: Finite type]\label{exer:vi28-21}
State when normalization is finite for a finitely generated $k$-domain.
\end{exercise}

\begin{exercise}[VI.28.E22: Dominant factorization]\label{exer:vi28-22}
State the universal property of normalization.
\end{exercise}

\begin{exercise}[VI.28.E23: Local ring upstairs]\label{exer:vi28-23}
Describe $\OO_{X^\nu,y}$ on an affine normalization.
\end{exercise}

\begin{exercise}[VI.28.E24: Base-change warning]\label{exer:vi28-24}
Why need normality not survive inseparable scalar extension?
\end{exercise}

\begin{exercise}[VI.28.E25: Birationality]\label{exer:vi28-25}
Why is normalization birational?
\end{exercise}

\begin{exercise}[VI.28.E26: Boundary]\label{exer:vi28-26}
Name one topic deliberately deferred beyond VI/28.
\end{exercise}
