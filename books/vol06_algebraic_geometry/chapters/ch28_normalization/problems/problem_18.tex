% VI28-DETAILED-PROBLEM-18
\begin{problem}[VI.28.P18: Normality after field extension: warning]\label{prob:vi28-18}
Explain why normality over \(k\) need not survive arbitrary field extension in positive characteristic.

\textbf{Provenance.} Canonical base-change warning; systematic base change was VI/24.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
k=\mathbb F_p(u),\qquad L=k(t),\qquad t^p=u.
\]
The extension \(L/k\) is finite and purely inseparable. Since \(L\) is a field,
\[
X=\Spec L
\]
is normal.

Base change along \(k\to L\) gives
\[
X_L=\Spec(L\otimes_kL).
\]
Using \(L\cong k[T]/(T^p-u)\),
\[
L\otimes_kL
\cong
L[T]/(T^p-u).
\]
But in \(L\),
\[
T^p-u=T^p-t^p=(T-t)^p.
\]
Therefore
\[
L\otimes_kL
\cong
L[\varepsilon]/(\varepsilon^p),
\qquad
\varepsilon=T-t.
\]
This ring has a nonzero nilpotent, so it is not reduced and hence cannot be normal. Thus normality need not survive purely inseparable scalar extension.
\end{solution}
\fi
