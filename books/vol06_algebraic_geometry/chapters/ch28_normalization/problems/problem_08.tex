% VI28-DETAILED-PROBLEM-08
\begin{problem}[VI.28.P8: Local rings upstairs]\label{prob:vi28-08}
Let \(y\in X^\nu\) lie over \(x\in X\). Explain how \(\mathcal O_{X^\nu,y}\) sits inside \(K(X)\) and why it is integral over the image of \(\mathcal O_{X,x}\).

\textbf{Provenance.} Canonical dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Choose an affine neighborhood \(U=\Spec A\) of \(x\). Above it,
\[
\nu^{-1}(U)=\Spec\overline A.
\]
Let \(x\leftrightarrow\mathfrak p\subseteq A\) and
\(y\leftrightarrow\mathfrak q\subseteq\overline A\), with
\(\mathfrak q\cap A=\mathfrak p\). Then
\[
\mathcal O_{X,x}=A_{\mathfrak p},
\qquad
\mathcal O_{X^\nu,y}=(\overline A)_{\mathfrak q}.
\]
Since \(\overline A\subseteq K(X)\), its localization embeds in the same field:
\[
(\overline A)_{\mathfrak q}\subseteq K(X).
\]

The extension \(A\subseteq\overline A\) is integral, and localization preserves integrality. Localizing first at \(A\setminus\mathfrak p\) and then at the prime above it yields an integral local homomorphism
\[
A_{\mathfrak p}\to(\overline A)_{\mathfrak q}.
\]
Thus the local ring upstairs is an integrally closed local domain inside the common function field and is integral over the downstairs local ring.
\end{solution}
\fi
