% VI28-DETAILED-PROBLEM-07
\begin{problem}[VI.28.P7: Restriction to a nonempty open]\label{prob:vi28-07}
For nonempty open \(U\subseteq X\), prove that \(\nu^{-1}(U)\to U\) is the normalization of \(U\).

\textbf{Provenance.} Canonical dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
It suffices to verify the claim on a distinguished affine basis. Suppose
\[
X=\Spec A,\qquad U=D(f)=\Spec A_f.
\]
The inverse image under normalization is
\[
\nu^{-1}(D(f))
=
D(f)\subseteq\Spec\overline A
=
\Spec(\overline A)_f.
\]
But integral closure commutes with localization:
\[
\overline{A_f}=(\overline A)_f.
\]
Therefore this inverse image is exactly
\[
\Spec\overline{A_f},
\]
the normalization of \(D(f)\).

For a general nonempty open \(U\), cover it by distinguished opens inside affine charts. The preceding identifications agree on further distinguished refinements because they are all localization maps. They glue to identify \(\nu^{-1}(U)\to U\) with the normalization of \(U\).
\end{solution}
\fi
