% VI28-DETAILED-PROBLEM-12
\begin{problem}[VI.28.P12: UFDs give normal schemes]\label{prob:vi28-12}
Show that if \(A\) is a UFD then \(\Spec A\) is normal.

\textbf{Provenance.} Canonical bridge to standard algebra examples.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
We prove that a UFD \(A\) is integrally closed. Let
\[
z=\frac ab\in\Frac(A)
\]
be in lowest terms and integral over \(A\). Then
\[
z^n+c_{n-1}z^{n-1}+\cdots+c_0=0
\]
for some \(c_i\in A\). Multiplying by \(b^n\) gives
\[
a^n=-b(c_{n-1}a^{n-1}+c_{n-2}a^{n-2}b+\cdots+c_0b^{n-1}),
\]
so \(b\mid a^n\). Since \(\gcd(a,b)=1\), unique factorization forces \(b\) to be a unit. Hence \(z\in A\).

Thus \(A\) is integrally closed in \(\Frac(A)\), so it is a normal domain and \(\Spec A\) is a normal scheme.
\end{solution}
\fi
