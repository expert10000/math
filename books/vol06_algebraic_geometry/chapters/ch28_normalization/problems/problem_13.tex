\begin{problem}[VI.28.P13: A quadratic arithmetic normalization]\label{prob:vi28-13}
Compute the integral closure of \(\mathbb Z[\sqrt5]\) in \(\mathbb Q(\sqrt5)\).

\textbf{Provenance.} Canonical arithmetic example; algebraic integral-closure theory belongs to Volume V.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Set
\[
\alpha=\frac{1+\sqrt5}{2}.
\]
Then
\[
\alpha^2-\alpha-1=0,
\]
so \(\alpha\) is integral over \(\mathbb Z[\sqrt5]\). Thus
\[
\mathbb Z[\sqrt5]\subseteq\mathbb Z[\alpha]\subseteq\mathbb Q(\sqrt5)
\]
is an integral extension.

The standard quadratic-integer theorem from Volume V says that for squarefree \(d\equiv1\pmod4\), the full ring of integers of \(\mathbb Q(\sqrt d)\) is
\[
\mathbb Z\!\left[\frac{1+\sqrt d}{2}\right].
\]
Since \(5\equiv1\pmod4\),
\[
\mathcal O_{\mathbb Q(\sqrt5)}
=
\mathbb Z[\alpha].
\]
The ring of integers of a number field is integrally closed in its fraction field. Therefore
\[
\boxed{
\overline{\mathbb Z[\sqrt5]}^{\,\mathbb Q(\sqrt5)}
=
\mathbb Z\!\left[\frac{1+\sqrt5}{2}\right].
}
\]
So the normalization adds exactly the missing half-integral generator \(\alpha\).
\end{solution}
\fi
