\begin{problem}[VI.28.P9: Same function field]\label{prob:vi28-09}
Prove globally that \(K(X^\nu)\cong K(X)\) canonically.

\textbf{Provenance.} Canonical dossier.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let \(U=\Spec A\subseteq X\) be any nonempty affine open. Its inverse image in the normalization is
\[
U^\nu=\Spec\overline A.
\]
The generic point of \(U^\nu\) maps to the generic point of \(U\), and
\[
K(U^\nu)=\Frac(\overline A)=\Frac(A)=K(U).
\]
Since \(U\) and \(X\) have the same generic point, \(K(U)=K(X)\); similarly
\(K(U^\nu)=K(X^\nu)\). Hence
\[
K(X^\nu)\cong K(X).
\]
The identification is canonical because every step identifies both fields with the corresponding generic stalk, and the affine identifications commute with localization on overlaps.
\end{solution}
\fi
