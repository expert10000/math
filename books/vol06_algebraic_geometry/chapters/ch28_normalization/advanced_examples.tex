\begin{example}[Advanced 1: Semigroup ring]\label{advex:vi28-01}
For $A=k[t^m,t^n]$ with $\gcd(m,n)=1$, the fraction field is $k(t)$ and the normalization is $k[t]$.
\end{example}

\begin{example}[Advanced 2: Branch separation]\label{advex:vi28-02}
Normalization can be finite and birational without being injective on points; a node is the basic curve example.
\end{example}

\begin{example}[Advanced 3: Unibranch cusp]\label{advex:vi28-03}
The cusp normalization is bijective on geometric points in characteristic zero but is still not an isomorphism because the local rings differ at the cusp.
\end{example}

\begin{example}[Advanced 4: Integral closure after localization]\label{advex:vi28-04}
For a domain $A$ and multiplicative set $S$, normalization inside $\Frac(A)$ satisfies $\overline{S^{-1}A}=S^{-1}\overline A$.
\end{example}

\begin{example}[Advanced 5: Universal factorization]\label{advex:vi28-05}
A dominant map from a normal integral affine scheme $\Spec B$ to $\Spec A$ extends uniquely from $A$ to $\overline A$ inside the induced embedding of fraction fields.
\end{example}

\begin{example}[Advanced 6: Normalization is idempotent]\label{advex:vi28-06}
Since $X^\nu$ is normal, normalizing it again gives a canonical isomorphism $(X^\nu)^\nu\cong X^\nu$.
\end{example}

\begin{example}[Advanced 7: Birational invariance]\label{advex:vi28-07}
$X$ and $X^\nu$ have canonically isomorphic function fields, so normalization stays in the same birational class.
\end{example}

\begin{example}[Advanced 8: Finite type over a field]\label{advex:vi28-08}
For a finitely generated $k$-domain, finiteness of integral closure makes normalization proper over affine charts because finite morphisms are affine and proper.
\end{example}

\begin{example}[Advanced 9: Positive-characteristic warning]\label{advex:vi28-09}
A normal $k$-scheme need not remain normal after an arbitrary inseparable field extension; geometric normality is a stronger condition.
\end{example}

\begin{example}[Advanced 10: Base-change warning]\label{advex:vi28-10}
Because normality may fail after field extension, forming $X^\nu$ and then base-changing need not equal normalizing the base change.
\end{example}

\begin{example}[Advanced 11: Normal versus regular]\label{advex:vi28-11}
Regular schemes are normal under the standard Noetherian hypotheses, but normal schemes can still have singularities in dimension at least two.
\end{example}

\begin{example}[Advanced 12: Codimension-one preview]\label{advex:vi28-12}
Later divisor theory detects normality using height-one local data together with a depth condition; VI/28 does not use that criterion yet.
\end{example}

\begin{example}[Advanced 13: Normalization of an order]\label{advex:vi28-13}
The inclusion $\mathbb Z[\sqrt5]\subset\mathbb Z[(1+\sqrt5)/2]$ is finite and birational on spectra, illustrating arithmetic normalization.
\end{example}

\begin{example}[Advanced 14: Nonfinite warning]\label{advex:vi28-14}
For arbitrary non-Noetherian domains, the integral closure in the fraction field need not be finite as a module, although the normalization morphism remains integral.
\end{example}
