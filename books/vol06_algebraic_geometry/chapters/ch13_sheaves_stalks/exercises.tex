\begin{exercise}[Locality gives uniqueness]
Prove that in the definition of a sheaf, once locality is assumed, a gluing (if it exists) is unique.
\end{exercise}

\begin{exercise}[The empty set]
For a sheaf of abelian groups, prove that $\mathcal F(\varnothing)=0$. For a sheaf of sets, prove that $\mathcal F(\varnothing)$ is a singleton.
\end{exercise}

\begin{exercise}[Continuous functions]
Prove directly that $U\mapsto C^0(U,\RR)$ is a sheaf of rings.
\end{exercise}

\begin{exercise}[Smooth functions]
Prove directly that smooth real-valued functions on a smooth manifold form a sheaf of rings.
\end{exercise}

\begin{exercise}[Locally constant functions]
Let $A$ be an abelian group. Prove that locally constant $A$-valued functions form a sheaf.
\end{exercise}

\begin{exercise}[Naive constant presheaf]
Give an explicit disconnected open set on which the naive constant presheaf with value $A$ fails gluing when $A$ has at least two elements.
\end{exercise}

\begin{exercise}[Bounded functions]
Verify carefully that bounded continuous functions form a presheaf on $\RR$ but not a sheaf.
\end{exercise}

\begin{exercise}[Germ equivalence]
Prove that the relation defining equality of germs at a point is an equivalence relation.
\end{exercise}

\begin{exercise}[Same value, different germ]
Find two smooth functions on $\RR$ with the same value at $0$ but different germs at $0$.
\end{exercise}

\begin{exercise}[Same germ, different global sections]
Find two smooth functions on $\RR$ with the same germ at $0$ but which are not equal globally.
\end{exercise}

\begin{exercise}[Algebra on stalks]
For a presheaf of rings, define addition and multiplication of germs and prove that they are well defined.
\end{exercise}

\begin{exercise}[Constant-sheaf stalk]
Prove that $(\underline A)_x\cong A$ for every point $x$.
\end{exercise}

\begin{exercise}[Skyscraper stalks]
On a $T_1$ space, compute every stalk of the skyscraper sheaf $\mathcal S_x(A)$.
\end{exercise}

\begin{exercise}[Morphism on stalks]
Given a morphism of presheaves $\varphi:\mathcal F\to\mathcal G$, prove that $\varphi_x(s_x)=(\varphi(s))_x$ is well defined.
\end{exercise}

\begin{exercise}[Functoriality]
Prove $(\psi\circ\varphi)_x=\psi_x\circ\varphi_x$ and $(\id_\mathcal F)_x=\id_{\mathcal F_x}$.
\end{exercise}

\begin{exercise}[Equality from germs]
Let $\mathcal F$ be a sheaf. Prove that two sections over $U$ are equal iff all their germs agree.
\end{exercise}

\begin{exercise}[Zero section from germs]
For a sheaf of abelian groups, prove $s=0$ iff $s_x=0$ for every $x$.
\end{exercise}

\begin{exercise}[Morphisms determined by stalks]
Prove that two morphisms of sheaves are equal if all induced stalk maps are equal.
\end{exercise}

\begin{exercise}[Discrete space]
Let $X$ be discrete. Prove $\mathcal F(U)\cong\prod_{x\in U}\mathcal F_x$ for every sheaf of abelian groups.
\end{exercise}

\begin{exercise}[Two-point discrete classification]
Classify sheaves of abelian groups on a two-point discrete space in terms of two abelian groups.
\end{exercise}

\begin{exercise}[Sierpinski stalks]
For $X=\{\eta,x\}$ with opens $\varnothing,\{\eta\},X$, write a sheaf as a map $A\to B$ and compute its two stalks.
\end{exercise}

\begin{exercise}[Support]
Show that for a sheaf of abelian groups, the set $\{x:s_x=0\}$ is open. Deduce the usual formula for support.
\end{exercise}

\begin{exercise}[Restriction and stalks]
Let $j:U\hookrightarrow X$ be open and $x\in U$. Prove $(\mathcal F|_U)_x\cong\mathcal F_x$.
\end{exercise}

\begin{exercise}[Product sheaves]
For sheaves $\mathcal F,\mathcal G$ of abelian groups, show that the objectwise product is a sheaf and $(\mathcal F\times\mathcal G)_x\cong\mathcal F_x\times\mathcal G_x$.
\end{exercise}

\begin{exercise}[Basis locality]
Let $\mathcal B$ be a basis. Show that to prove two sheaf sections over $U$ are equal, it suffices to prove equality on a basis cover of $U$.
\end{exercise}

\begin{exercise}[Local solutions]
Let $L$ be a local linear differential operator. Prove that $U\mapsto\ker(L:C^\infty(U)\to C^\infty(U))$ is a sheaf.
\end{exercise}
