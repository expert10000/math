\begin{example}[Continuous real-valued functions]
For each open $U\subseteq X$, let $\mathcal C^0(U)=C^0(U,\RR)$.  Restrictions are ordinary
function restrictions.  If continuous functions agree on overlaps, the pointwise glued
function is continuous because continuity is local.  Hence $\mathcal C^0$ is a sheaf of
rings.
\end{example}

\begin{example}[Smooth functions]
On a smooth manifold $M$, $U\mapsto C^\infty(U)$ is a sheaf.  Compatible smooth functions
glue as functions, and smoothness can be checked in a neighborhood of each point.
\end{example}

\begin{example}[Holomorphic functions]
On an open subset $X\subseteq\CC$, let $\OO(U)$ be the holomorphic functions on $U$.
Compatible holomorphic functions glue uniquely, and holomorphicity is local.  The stalk
$\OO_x$ is the ring of holomorphic germs at $x$.
\end{example}

\begin{example}[Locally constant functions]
Let $A$ be an abelian group.  The constant sheaf $\underline A$ assigns to $U$ the locally
constant maps $U\to A$.  If $U$ has connected components $U_\alpha$, then
$\underline A(U)\cong\prod_\alpha A$.
\end{example}

\begin{example}[The naive constant presheaf fails gluing]
Let $U=U_1\sqcup U_2$ with both pieces nonempty.  In the naive constant presheaf choose
$a$ on $U_1$ and $b\neq a$ on $U_2$.  The overlap is empty, so compatibility is automatic,
but no single element over $U$ restricts to both.  This is the standard reason the constant
sheaf must use locally constant functions.
\end{example}

\begin{example}[Bounded functions fail gluing]
On $\RR$, define
\[
\mathcal B(U)=\{f:U\to\RR : f\text{ is continuous and bounded}\}.
\]
This is a presheaf.  On $U_n=(-n,n)$, the sections $x\mapsto x$ are compatible and individually bounded.  Their only possible gluing on $\RR$ is unbounded, so $\mathcal B$ is not a sheaf.
\end{example}

\begin{example}[Two sections with the same value but different germs]
In $C^0_\RR$, let $f(t)=t$ and $g(t)=0$.  Then $f(0)=g(0)$, but $f_0\neq g_0$ because they
do not agree on any neighborhood of $0$.
\end{example}

\begin{example}[Two different global sections with the same germ]
Let $f(t)=0$ for all $t$ and let $g(t)=0$ on $(-1,1)$ but $g(t)=t^2-1$ outside a slightly
larger neighborhood, smoothed continuously across the transition.  Then $f_0=g_0$ although
$f\neq g$ globally.  The stalk ignores behavior far from $0$.
\end{example}

\begin{example}[Germs of continuous functions]
An element of $(C^0_X)_x$ is represented by a continuous function on some neighborhood of
$x$.  Two representatives are equal exactly when they agree after shrinking.  Addition and
multiplication are performed on a common smaller neighborhood.
\end{example}

\begin{example}[Germs of smooth functions]
For $M=\RR$ and $x=0$, the germs of $e^t$, $1+t+t^2/2+\cdots$, and any actual smooth
function agreeing with $e^t$ near $0$ define the same element of $C^\infty_{M,0}$.  A Taylor
series alone does not determine a smooth germ, however: distinct smooth germs can have the
same Taylor series.
\end{example}

\begin{example}[Holomorphic germs are rigid]
For holomorphic functions, the identity theorem gives extra rigidity.  If two holomorphic
germs agree on a set with an accumulation point in a connected representative neighborhood,
the germs agree.  This contrasts with smooth germs.
\end{example}

\begin{example}[A germ represented on many neighborhoods]
The germ at $0$ of $1/(1-t)$ can be represented on any interval $(-r,r)$ with $0<r<1$.
There is no preferred representative; the stalk identifies all compatible restrictions.
\end{example}

\begin{example}[Stalk of the constant sheaf]
For every $x\in X$, $(\underline A)_x\cong A$.  A locally constant function is constant on
some neighborhood of $x$, so its germ is determined by one element of $A$, and every element
occurs.
\end{example}

\begin{example}[Skyscraper at a point of a Hausdorff space]
Let $\mathcal S_x(A)$ be the skyscraper sheaf.  At $x$ its stalk is $A$.  If $y\neq x$ and
$X$ is Hausdorff (indeed $T_1$ suffices), choose a neighborhood of $y$ avoiding $x$; on that
neighborhood the sheaf is zero, so the stalk at $y$ is zero.
\end{example}

\begin{example}[Sections on a two-point discrete space]
Let $X=\{p,q\}$ be discrete and let a sheaf have stalks $A$ at $p$ and $B$ at $q$.  Then
\[
\mathcal F(X)\cong A\times B,
\qquad
\mathcal F(\{p\})\cong A,
\qquad
\mathcal F(\{q\})\cong B.
\]
The sheaf condition reconstructs global sections from the two independent point sections.
\end{example}

\begin{example}[A sheaf of maps to a space]
Fix a topological space $Y$.  The assignment $U\mapsto C^0(U,Y)$ is a sheaf of sets.  Local
continuous maps agreeing on overlaps glue to a unique continuous map.
\end{example}

\begin{example}[Sections of a covering map]
For a covering map $p:E\to X$, assign to $U$ the continuous sections $s:U\to E$ with
$p\circ s=\id_U$.  Compatible local sections glue as maps, so this is a sheaf of sets.
\end{example}

\begin{example}[Differential forms]
On a smooth manifold, $U\mapsto\Omega^k(U)$ is a sheaf of vector spaces.  The exterior
derivative commutes with restriction and therefore gives a morphism of sheaves
$d:\Omega^k\to\Omega^{k+1}$.
\end{example}

\begin{example}[A sheaf morphism and its stalk map]
The derivative $D:C^\infty_\RR\to C^\infty_\RR$ is compatible with restriction.  At each
$x$, it induces $D_x$ sending the germ of $f$ to the germ of $f'$.  The definition does not
depend on which neighborhood represents the germ.
\end{example}

\begin{example}[Support of a smooth section]
Let $f\in C^\infty(\RR)$ be a bump function positive on $(-1,1)$ and zero outside
$[-1,1]$.  The germ $f_x$ is zero for $|x|>1$ and nonzero for $|x|<1$.  Hence
$\operatorname{supp}(f)=[-1,1]$.
\end{example}
