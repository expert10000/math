\begin{problem}[VI.29.P5: Dimension of the affine-line coordinate ring]\label{prob:vi29-05}
Show \(\dim k[x]=1\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The ring \(k[x]\) is a PID.  Its prime ideals are \((0)\) and ideals \((f)\) generated by irreducible
polynomials.  Since
\[
k[x]/(f)
\]
is a field, every nonzero prime is maximal.  Therefore
\[
(0)\subsetneq(f)
\]
is a chain of length \(1\), and every nontrivial prime chain has this form.  Thus
\[
\boxed{\dim k[x]=1}.
\]
\end{solution}
\fi
