\begin{problem}[VI.29.P3: Dimension of a field]\label{prob:vi29-03}
Prove that a field has Krull dimension \(0\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
A field \(k\) has only one prime ideal, namely \((0)\).  Hence there are no two distinct prime ideals
\(\mathfrak p\subsetneq\mathfrak q\).  All prime chains therefore have length \(0\).  Since the ring is nonzero,
\[
\boxed{\dim k=0}.
\]
\end{solution}
\fi
