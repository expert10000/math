% VI29-DETAILED-PROBLEM-15
\begin{problem}[VI.29.P15: Integral extensions preserve dimension]\label{prob:vi29-15}
Assuming going up and incomparability, prove that an integral extension \(A\subseteq B\) has \(\dim A=\dim B\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
First prove
\[
\dim B\ge\dim A.
\]
Given a chain
\[
\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_n
\]
in \(A\), lying over gives a prime \(\mathfrak q_0\subseteq B\) over \(\mathfrak p_0\).  Applying going up
successively produces
\[
\mathfrak q_0\subsetneq\mathfrak q_1\subsetneq\cdots\subsetneq\mathfrak q_n
\]
with \(\mathfrak q_i\cap A=\mathfrak p_i\).  Hence every chain length downstairs occurs upstairs.

For the reverse inequality, take a strict chain
\[
\mathfrak q_0\subsetneq\cdots\subsetneq\mathfrak q_n
\]
in \(B\).  Contract to
\[
\mathfrak p_i=\mathfrak q_i\cap A.
\]
The contractions form a weakly increasing chain.  If \(\mathfrak p_i=\mathfrak p_{i+1}\), then
\(\mathfrak q_i\subsetneq\mathfrak q_{i+1}\) would be two comparable primes lying over the same prime, contradicting
incomparability for integral extensions.  Therefore all inclusions remain strict.  Thus every chain upstairs
contracts to one of the same length downstairs, so \(\dim B\le\dim A\).

Hence
\[
\boxed{\dim B=\dim A}.
\]
\end{solution}
\fi
