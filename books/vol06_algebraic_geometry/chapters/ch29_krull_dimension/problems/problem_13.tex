\begin{problem}[VI.29.P13: A dimension-preserving localization]\label{prob:vi29-13}
Show \(\dim k[x,y]_x=2\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Here \(k[x,y]_x\) means localization at the multiplicative set
\[
S=\{1,x,x^2,\ldots\}.
\]
Since localization cannot increase dimension,
\[
\dim k[x,y]_x\le\dim k[x,y]=2.
\]

For the lower bound, consider
\[
(0)\subsetneq(y)\subsetneq(y,x-1).
\]
Each ideal is prime: the successive quotients are \(k[x,y]\), \(k[x]\), and \(k\).  None meets \(S\): neither
\((y)\) nor \((y,x-1)\) contains \(x\), because modulo \((y,x-1)\) the element \(x\) becomes \(1\).  Hence the
chain survives localization:
\[
(0)\subsetneq(y)k[x,y]_x\subsetneq(y,x-1)k[x,y]_x.
\]
This has length \(2\), so
\[
\boxed{\dim k[x,y]_x=2}.
\]
\end{solution}
\fi
