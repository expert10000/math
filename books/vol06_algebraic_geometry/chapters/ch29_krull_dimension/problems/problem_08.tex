\begin{problem}[VI.29.P8: Quotient chains]\label{prob:vi29-08}
Show that \(\dim(A/I)\) is the supremum of chain lengths among primes of \(A\) containing \(I\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The quotient map \(A\to A/I\) induces the bijection
\[
\Spec(A/I)\longleftrightarrow V(I)\subseteq\Spec A,
\qquad
\mathfrak p/I\longleftrightarrow\mathfrak p.
\]
It preserves strict inclusions.  Thus a chain
\[
\mathfrak q_0\subsetneq\cdots\subsetneq\mathfrak q_n
\]
in \(A/I\) corresponds to
\[
I\subseteq\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_n
\]
in \(A\), all primes containing \(I\), with the same length.  Taking suprema proves the claim and immediately gives
\[
\dim(A/I)\le\dim A.
\]
\end{solution}
\fi
