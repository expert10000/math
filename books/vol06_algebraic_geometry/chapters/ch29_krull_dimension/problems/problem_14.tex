\begin{problem}[VI.29.P14: Infinite Krull dimension]\label{prob:vi29-14}
Show \(k[x_1,x_2,\ldots]\) has infinite Krull dimension.

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
A=k[x_1,x_2,\ldots].
\]
For each \(n\ge1\), define
\[
\mathfrak p_r=(x_1,\ldots,x_r),\qquad 1\le r\le n.
\]
Then
\[
A/\mathfrak p_r\cong k[x_{r+1},x_{r+2},\ldots]
\]
is a domain, so \(\mathfrak p_r\) is prime.  Moreover
\[
(0)\subsetneq\mathfrak p_1\subsetneq\cdots\subsetneq\mathfrak p_n
\]
has length \(n\).  Since such a chain exists for every \(n\), the set of possible finite chain lengths is
unbounded.  Therefore
\[
\boxed{\dim A=\infty}.
\]
No single finite chain has ``infinite length''; the infinite value is the supremum over all finite lengths.
\end{solution}
\fi
