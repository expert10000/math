\begin{problem}[VI.29.P14: Infinite Krull dimension]\label{prob:vi29-14}
Show $k[x_1,x_2,\ldots]$ has infinite Krull dimension.

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}
\begin{solution}
For every $n$, the coordinate ideals $(0)\subsetneq(x_1)\subsetneq\cdots\subsetneq(x_1,\ldots,x_n)$ are prime and form a chain of length $n$.
\end{solution}
