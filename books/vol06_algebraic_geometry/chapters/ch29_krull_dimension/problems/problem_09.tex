\begin{problem}[VI.29.P9: Localization chains]\label{prob:vi29-09}
Show that $\dim S^{-1}A\le\dim A$.

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}
\begin{solution}
Prime ideals of $S^{-1}A$ correspond to primes of $A$ disjoint from $S$, with strict inclusions preserved.  Thus every localized chain comes from a chain in $A$.
\end{solution}
