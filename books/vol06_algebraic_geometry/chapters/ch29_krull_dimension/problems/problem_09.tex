\begin{problem}[VI.29.P9: Localization chains]\label{prob:vi29-09}
Show that \(\dim S^{-1}A\le\dim A\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Primes of \(S^{-1}A\) correspond to primes \(\mathfrak p\subseteq A\) with
\[
\mathfrak p\cap S=\varnothing.
\]
Under contraction and extension, inclusions are preserved and reflected.  Hence every strict chain
\[
\mathfrak q_0\subsetneq\cdots\subsetneq\mathfrak q_n
\]
of primes in \(S^{-1}A\) contracts to a strict chain of the same length in \(A\).  Therefore the set of chain
lengths available after localization is a subset of the chain lengths available before localization, so
\[
\boxed{\dim S^{-1}A\le\dim A}.
\]
\end{solution}
\fi
