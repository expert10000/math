\begin{problem}[VI.29.P10: Dimension and reduction]\label{prob:vi29-10}
Prove \(\dim A=\dim A_{\mathrm{red}}\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let \(N=\sqrt{(0)}\).  Every prime ideal of \(A\) contains \(N\), so quotienting by \(N\) gives an
order-preserving bijection
\[
\Spec A\longrightarrow\Spec(A/N),
\qquad
\mathfrak p\longmapsto\mathfrak p/N.
\]
Its inverse is contraction under \(A\to A/N\).  Strict inclusions are preserved in both directions.  Therefore
prime-chain lengths are identical in \(A\) and \(A_{\mathrm{red}}\), giving
\[
\boxed{\dim A=\dim A_{\mathrm{red}}}.
\]
\end{solution}
\fi
