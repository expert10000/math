\begin{problem}[VI.29.P6: Coordinate chain in a polynomial ring]\label{prob:vi29-06}
Construct a prime chain of length $n$ in $k[x_1,\ldots,x_n]$.

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}
\begin{solution}
Use $(0)\subsetneq(x_1)\subsetneq(x_1,x_2)\subsetneq\cdots\subsetneq(x_1,\ldots,x_n)$.  Each quotient is a polynomial ring over $k$, hence each ideal is prime.
\end{solution}
