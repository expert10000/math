\begin{problem}[VI.29.P6: Coordinate chain in a polynomial ring]\label{prob:vi29-06}
Construct a prime chain of length \(n\) in \(k[x_1,\ldots,x_n]\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Consider
\[
(0)
\subsetneq(x_1)
\subsetneq(x_1,x_2)
\subsetneq\cdots
\subsetneq(x_1,\ldots,x_n).
\]
For
\[
\mathfrak p_r=(x_1,\ldots,x_r),
\]
the quotient is
\[
k[x_1,\ldots,x_n]/\mathfrak p_r
\cong
k[x_{r+1},\ldots,x_n],
\]
a domain.  Hence every \(\mathfrak p_r\) is prime.  The inclusions are strict because \(x_{r+1}\notin\mathfrak p_r\).
There are \(n\) strict inclusions, so this is a prime chain of length \(n\).
\end{solution}
\fi
