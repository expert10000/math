\begin{problem}[VI.29.P7: Height equals localized dimension]\label{prob:vi29-07}
Prove \(\operatorname{ht}(\mathfrak p)=\dim A_{\mathfrak p}\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Prime ideals of \(A_{\mathfrak p}\) correspond bijectively to prime ideals
\[
\mathfrak q\subseteq\mathfrak p
\]
of \(A\), via
\[
\mathfrak q\longmapsto\mathfrak q A_{\mathfrak p}.
\]
The correspondence preserves and reflects strict inclusions.  Therefore prime chains in \(A_{\mathfrak p}\) are
exactly prime chains of \(A\) terminating below \(\mathfrak p\).  Taking suprema of lengths gives
\[
\boxed{\dim A_{\mathfrak p}=\operatorname{ht}(\mathfrak p)}.
\]
\end{solution}
\fi
