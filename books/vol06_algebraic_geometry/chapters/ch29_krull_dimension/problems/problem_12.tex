\begin{problem}[VI.29.P12: A dimension drop by localization]\label{prob:vi29-12}
Give a localization of a one-dimensional domain that has dimension $0$.

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}
\begin{solution}
For $A=k[x]$, localize at $S=A\setminus\{0\}$.  Then $S^{-1}A=k(x)$ is a field, so the dimension drops from $1$ to $0$.
\end{solution}
