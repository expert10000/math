\begin{problem}[VI.29.P12: A dimension drop by localization]\label{prob:vi29-12}
Give a localization of a one-dimensional domain that has dimension \(0\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Take
\[
A=k[x],
\]
which has dimension \(1\).  Let
\[
S=A\setminus\{0\}.
\]
Since \(A\) is a domain,
\[
S^{-1}A=\Frac(A)=k(x),
\]
a field.  Fields have Krull dimension \(0\).  Thus localization can strictly lower dimension:
\[
\dim k[x]=1,
\qquad
\dim k(x)=0.
\]
Geometrically, the localization retains only the generic point of \(\Spec k[x]\).
\end{solution}
\fi
