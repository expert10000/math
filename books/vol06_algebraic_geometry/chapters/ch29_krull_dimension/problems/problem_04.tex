\begin{problem}[VI.29.P4: Dimension of \(\mathbb Z\)]\label{prob:vi29-04}
Compute \(\dim\mathbb Z\).

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The prime ideals of \(\mathbb Z\) are \((0)\) and \((p)\) for rational primes \(p\).  Each \((p)\) is maximal.
Thus
\[
(0)\subsetneq(p)
\]
is a chain of length \(1\).  No strict prime ideal can lie above \((p)\), and there is no nonzero prime strictly
between \((0)\) and \((p)\).  Hence
\[
\boxed{\dim\mathbb Z=1}.
\]
\end{solution}
\fi
