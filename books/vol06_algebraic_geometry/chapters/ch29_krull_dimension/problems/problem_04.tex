\begin{problem}[VI.29.P4: Dimension of $\mathbb Z$]\label{prob:vi29-04}
Compute $\dim\mathbb Z$.

\textbf{Provenance.} Canonical VI/29 development.
\end{problem}
\begin{solution}
The primes are $(0)$ and $(p)$ for prime integers $p$.  Each $(p)$ is maximal, so $(0)\subsetneq(p)$ has length $1$ and no longer chain exists.
\end{solution}
