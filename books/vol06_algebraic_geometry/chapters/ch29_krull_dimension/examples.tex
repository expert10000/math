\begin{example}[Example 1: Field]\label{ex:vi29-01}
For a field $K$, the only prime is $(0)$, so $\dim K=0$.
\end{example}

\begin{example}[Example 2: Integers]\label{ex:vi29-02}
The chains $(0)\subsetneq(p)$ show $\dim\mathbb Z=1$; every nonzero prime is maximal.
\end{example}

\begin{example}[Example 3: Affine line]\label{ex:vi29-03}
For $k[x]$, $(0)\subsetneq(x-a)$ is a maximal chain, hence $\dim k[x]=1$.
\end{example}

\begin{example}[Example 4: Affine plane]\label{ex:vi29-04}
In $k[x,y]$, $(0)\subsetneq(x)\subsetneq(x,y)$ has length $2$.
\end{example}

\begin{example}[Example 5: Affine n-space]\label{ex:vi29-05}
A coordinate-prime chain has length $n$, and the polynomial dimension theorem gives $\dim k[x_1,\ldots,x_n]=n$.
\end{example}

\begin{example}[Example 6: A hypersurface quotient]\label{ex:vi29-06}
$k[x,y]/(x)$ is isomorphic to $k[y]$, so the quotient has dimension $1$.
\end{example}

\begin{example}[Example 7: Artinian ring]\label{ex:vi29-07}
Every prime in an Artinian ring is maximal; therefore every nonzero Artinian ring has Krull dimension $0$.
\end{example}

\begin{example}[Example 8: Dual numbers]\label{ex:vi29-08}
$k[\varepsilon]/(\varepsilon^2)$ has the same spectrum as $k$, hence dimension $0$.
\end{example}

\begin{example}[Example 9: Reducible curve]\label{ex:vi29-09}
$k[x,y]/(xy)$ has dimension $1$: its two irreducible components are affine lines.
\end{example}

\begin{example}[Example 10: Localization to a field]\label{ex:vi29-10}
Localizing $k[x]$ at all nonzero elements gives $k(x)$ and drops dimension from $1$ to $0$.
\end{example}

\begin{example}[Example 11: Localizing at a maximal ideal]\label{ex:vi29-11}
$k[x,y]_{(x,y)}$ has dimension $2$ because $(0)\subsetneq(x)\subsetneq(x,y)$ survives.
\end{example}

\begin{example}[Example 12: Product ring]\label{ex:vi29-12}
$\dim(k\times k[x])=\max\{0,1\}=1$.
\end{example}

\begin{example}[Example 13: Reduction]\label{ex:vi29-13}
$k[x,\varepsilon]/(\varepsilon^2)$ and $k[x]$ have the same prime chains, hence both have dimension $1$.
\end{example}

\begin{example}[Example 14: Closed point quotient]\label{ex:vi29-14}
$k[x,y]/(x,y)\cong k$ has dimension $0$.
\end{example}

\begin{example}[Example 15: Principal open]\label{ex:vi29-15}
$k[x,y]_x$ still has dimension $2$: for example $(0)\subsetneq(y)\subsetneq(y,x-1)$ survives when $x$ is inverted.
\end{example}

\begin{example}[Example 16: Infinite polynomial ring]\label{ex:vi29-16}
$k[x_1,x_2,\ldots]$ has infinite dimension from chains of coordinate ideals of arbitrary length.
\end{example}

\begin{example}[Example 17: Height one prime]\label{ex:vi29-17}
In $k[x,y]$, the prime $(x)$ has height $1$ and $\dim k[x,y]_{(x)}=1$.
\end{example}

\begin{example}[Example 18: Maximal ideal height]\label{ex:vi29-18}
In $k[x,y]$, $(x,y)$ has height $2$.
\end{example}

\begin{example}[Example 19: Integral extension]\label{ex:vi29-19}
$k[t^2,t^3]\subset k[t]$ is integral; both rings have dimension $1$.
\end{example}

\begin{example}[Example 20: Empty-dimensional warning]\label{ex:vi29-20}
Dimension conventions for the zero ring or empty spectrum vary; VI/29 avoids using those conventions in computations.
\end{example}
