\chapter{Krull Dimension}\label{ch:vi29-krull-dimension}

The first invariant in the dimension arc is entirely order-theoretic: how long can a strict chain of prime ideals be?
For an affine scheme this same number measures the longest chain of irreducible closed subsets.  VI/29 develops this
ring-theoretic and affine-topological core.  VI/30 will globalize dimension to arbitrary schemes and prove the
finite-type dimension formulas; VI/31 will isolate codimension, and VI/32 will study local dimension and tangent data.

\section{Prime chains and Krull dimension}
\begin{definition}[Krull dimension]\label{def:vi29-krull-dimension}
Let $A$ be a nonzero ring.  The \emph{Krull dimension} of $A$ is
\[
\dim A=\sup\{n:\mathfrak p_0\subsetneq\mathfrak p_1\subsetneq\cdots\subsetneq\mathfrak p_n
\text{ is a chain of prime ideals of }A\}.
\]
The length of the displayed chain is $n$, the number of strict inclusions, not the number $n+1$ of primes.
\end{definition}

A field has dimension $0$.  The ring $k[x]$ has dimension $1$, while $k[x,y]$ has dimension $2$.
The infinite polynomial ring $k[x_1,x_2,\ldots]$ has infinite dimension because
\[
(0)\subsetneq(x_1)\subsetneq(x_1,x_2)\subsetneq\cdots
\]
contains chains of arbitrary finite length.

\section{Irreducible closed subsets of an affine scheme}
Every irreducible closed subset of $\Spec A$ is $V(\mathfrak p)$ for a unique prime ideal $\mathfrak p$.  Inclusion is
reversed:
\[
\mathfrak p\subseteq\mathfrak q\quad\Longleftrightarrow\quad V(\mathfrak q)\subseteq V(\mathfrak p).
\]
Therefore a prime chain
\[
\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_n
\]
corresponds to the irreducible-closed chain
\[
V(\mathfrak p_n)\subsetneq\cdots\subsetneq V(\mathfrak p_0).
\]
Thus the topological dimension of $\Spec A$ equals $\dim A$.

\begin{proposition}[Affine chain dictionary]\label{prop:vi29-affine-chain-dictionary}
For every nonzero ring $A$, the supremum of lengths of strict chains of prime ideals equals the supremum of lengths
of strict chains of irreducible closed subsets of $\Spec A$.
\end{proposition}

\begin{example}[Dimension of a field]\label{ex:vi29-expand-01}
A field has only the prime ideal $(0)$, so there is no strict nontrivial chain of primes.  Its Krull dimension is $0$.
\end{example}

\section{Height of a prime}
\begin{definition}[Height]\label{def:vi29-height}
For $\mathfrak p\in\Spec A$, the \emph{height} of $\mathfrak p$ is
\[
\operatorname{ht}(\mathfrak p)=\sup\{n:\mathfrak p_0\subsetneq\cdots\subsetneq\mathfrak p_n=\mathfrak p\}.
\]
\end{definition}
Prime ideals below $\mathfrak p$ are exactly the primes of the localization $A_{\mathfrak p}$.  Hence
\begin{proposition}[Height as local Krull dimension]\label{prop:vi29-height-localization}
\[
\operatorname{ht}(\mathfrak p)=\dim A_{\mathfrak p}.
\]
\end{proposition}
This is a ring-theoretic statement.  The geometric interpretation of $\dim\OO_{X,x}$ as local dimension belongs to
VI/32.

\input{chapters/ch29_krull_dimension/figures/figure_03}

\section{Quotients and closed subsets}
Prime ideals of $A/I$ correspond to primes $\mathfrak p\supseteq I$ of $A$.  Consequently
\begin{proposition}[Dimension of a quotient]\label{prop:vi29-quotient-dimension}
\[
\dim(A/I)\leq\dim A.
\]
More precisely, $\dim(A/I)$ is the supremum of lengths of chains of primes of $A$ all containing $I$.
\end{proposition}
Geometrically $\Spec(A/I)$ is the closed subspace $V(I)\subseteq\Spec A$ with its scheme structure.

\input{chapters/ch29_krull_dimension/figures/figure_04}

\section{Localization and open subsets}
Prime ideals of $S^{-1}A$ correspond to primes of $A$ disjoint from $S$, with inclusions preserved.  Therefore
\begin{proposition}[Localization cannot increase dimension]\label{prop:vi29-localization-dimension}
\[
\dim S^{-1}A\leq\dim A.
\]
For a principal localization, $\dim A_f$ is computed from prime chains entirely inside $D(f)$.
\end{proposition}
The inequality can be strict: localizing $k[x]$ at the multiplicative set of all nonzero polynomials gives $k(x)$,
whose dimension is $0$.

\input{chapters/ch29_krull_dimension/figures/figure_05}

\begin{example}[Dimension of a polynomial line]\label{ex:vi29-expand-02}
In $k[x]$, the chain $(0)\subset(x-a)$ has length $1$, and no longer prime chain exists.  Thus $\dim k[x]=1$.
\end{example}
% END VOL06-EXPANSION VI29-example-02
\section{Nilpotents do not affect dimension}
The reduction map $A\to A_{\mathrm{red}}=A/\sqrt{(0)}$ induces a homeomorphism on spectra and a bijection on prime
ideals preserving inclusions.  Hence
\begin{proposition}[Dimension ignores nilpotents]\label{prop:vi29-reduction-dimension}
\[
\dim A=\dim A_{\mathrm{red}}.
\]
\end{proposition}
Dimension measures the specialization order of points; nilpotent thickening changes the structure sheaf, not this
order.

\input{chapters/ch29_krull_dimension/figures/figure_06}

\section{Products and disconnected spectra}
Since
\[
\Spec(A\times B)=\Spec A\sqcup\Spec B,
\]
no prime chain crosses from one component to the other.  Thus
\begin{proposition}[Dimension of a product]\label{prop:vi29-product-dimension}
\[
\dim(A\times B)=\max\{\dim A,\dim B\}.
\]
\end{proposition}
This simple example anticipates a warning for VI/30: global dimension is a supremum over components, so local or open
pieces need not all have the same dimension without irreducibility hypotheses.

\input{chapters/ch29_krull_dimension/figures/figure_07}

\section{Integral extensions}
Let $A\subseteq B$ be integral.  Going up lifts prime chains from $A$ to $B$, while incomparability prevents a strict
chain in $B$ from contracting two comparable primes to the same prime of $A$.  Therefore
\begin{theorem}[Dimension under integral extensions]\label{thm:vi29-integral-dimension}
If $A\subseteq B$ is integral, then
\[
\dim B=\dim A.
\]
\end{theorem}
In particular, normalization does not change Krull dimension.  This links VI/28 to the dimension arc without
reopening the normalization construction.

\input{chapters/ch29_krull_dimension/figures/figure_08}

\begin{example}[A two-dimensional chain]\label{ex:vi29-expand-03}
In $k[x,y]$, $(0)\subset(x)\subset(x,y)$ is a chain of length $2$.  The polynomial ring has Krull dimension $2$.
\end{example}
% END VOL06-EXPANSION VI29-example-03
\section{Polynomial rings and finite-type algebras}
For a field $k$,
\[
\dim k[x_1,\ldots,x_n]=n.
\]
The lower bound is the chain
\[
(0)\subsetneq(x_1)\subsetneq(x_1,x_2)\subsetneq\cdots\subsetneq(x_1,\ldots,x_n),
\]
and the upper bound is the classical dimension theorem for polynomial rings.  Quotients therefore give many
finite-dimensional affine schemes.  The stronger equality between dimension and transcendence degree for integral
finite-type $k$-algebras is used systematically in VI/30.

\input{chapters/ch29_krull_dimension/figures/figure_09}

\section{Boundaries of VI/29}
VI/29 defines and computes Krull dimension through prime chains and affine topology.  It deliberately does not yet
prove the global variety statements from AG8 Problem 2: equality with local dimension at closed points, additivity
with codimension, invariance on nonempty opens of an irreducible variety, or the full transcendence-degree formula.
Those are routed to VI/30--VI/32.  Likewise AG8 Problem 3 tests failures of those global statements for general
schemes; the VI/29 instance audit finds no Krull-dimension-only numbered subproblem there.

\section{Chapter synthesis}
Krull dimension records the depth of the specialization order.  For affine schemes, prime chains and chains of
irreducible closed subsets are the same data read in opposite directions.  Quotients restrict to closed subsets,
localizations restrict to opens, nilpotents leave the order unchanged, products take maxima, and integral extensions
preserve the dimension.  These mechanisms supply the algebraic engine for the global dimension theory of VI/30.

\end{exercise}

Why do prime chains measure dimension?
\end{exercise}

\end{hint}
\begin{solution}
They encode nested irreducible closed subsets.
\end{solution}

\end{solution}
