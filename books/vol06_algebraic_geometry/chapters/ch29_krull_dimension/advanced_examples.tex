\begin{example}[Advanced 1: Saturated prime chains]\label{advex:vi29-01}
A maximal-length chain need not be unique.  Dimension records only the supremal length, not the branching pattern of $\Spec A$.
\end{example}

\begin{example}[Advanced 2: Noetherian does not mean finite-dimensional]\label{advex:vi29-02}
There exist Noetherian rings of infinite Krull dimension; finite-dimensionality needs additional hypotheses such as finite type over a field.
\end{example}

\begin{example}[Advanced 3: Catenary preview]\label{advex:vi29-03}
Even when endpoint primes are fixed, different saturated chains can have different lengths in noncatenary rings.  Krull dimension alone does not detect this pathology.
\end{example}

\begin{example}[Advanced 4: Equidimensionality preview]\label{advex:vi29-04}
A reducible ring can have irreducible components of different dimensions; the ring dimension is the maximum component dimension.
\end{example}

\begin{example}[Advanced 5: Dimension after quotient]\label{advex:vi29-05}
The inequality $\dim A/I\le\dim A$ can be strict or an equality, depending on which chains remain above $I$.
\end{example}

\begin{example}[Advanced 6: Dimension after localization]\label{advex:vi29-06}
Localization selects primes disjoint from $S$; it may preserve dimension or lower it drastically.
\end{example}

\begin{example}[Advanced 7: Integral closure]\label{advex:vi29-07}
An integral domain and its normalization have equal Krull dimension even when the normalization separates branches geometrically.
\end{example}

\begin{example}[Advanced 8: Minimal primes]\label{advex:vi29-08}
Every prime chain begins above some minimal prime, so $\dim A$ is the supremum of the dimensions of the irreducible components $V(\mathfrak p_{\min})$.
\end{example}

\begin{example}[Advanced 9: Height versus coheight]\label{advex:vi29-09}
$\operatorname{ht}(\mathfrak p)$ measures chains below $\mathfrak p$, whereas $\dim(A/\mathfrak p)$ measures chains above it.
\end{example}

\begin{example}[Advanced 10: Dimension formula warning]\label{advex:vi29-10}
The equality $\operatorname{ht}(\mathfrak p)+\dim(A/\mathfrak p)=\dim A$ is not a formal consequence of the definition for arbitrary rings.
\end{example}

\begin{example}[Advanced 11: Valuation domains]\label{advex:vi29-11}
Prime ideals in a valuation domain are totally ordered; their order type makes valuation domains a rich source of prescribed Krull dimensions.
\end{example}

\begin{example}[Advanced 12: Polynomial extension]\label{advex:vi29-12}
For Noetherian $A$, one has $\dim A[x]=\dim A+1$ under the classical polynomial dimension theorem; VI/29 uses the field case explicitly.
\end{example}

\begin{example}[Advanced 13: Specialization order]\label{advex:vi29-13}
In $\Spec A$, $\mathfrak q$ is a specialization of $\mathfrak p$ exactly when $\mathfrak p\subseteq\mathfrak q$.
\end{example}

\begin{example}[Advanced 14: Topological invariance]\label{advex:vi29-14}
Because Krull dimension of an affine spectrum depends only on the specialization order, nilpotent thickenings leave it unchanged.
\end{example}
