\begin{problem}[Gluing on two principal opens]\label{prob:vi17-p10}
Let $U=D(f)\cup D(g)$.  Prove
\[\OO_X(U)\cong A_f\times_{A_{fg}}A_g.\]
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This is the sheaf axiom translated into explicit commutative algebra.

\paragraph{Useful ideas.}
Use the cover by two principal opens and identify the overlap with $D(fg)$.

\paragraph{Guided hints.}
\begin{enumerate}
\item A section produces a compatible pair by restriction.
\item A compatible pair glues uniquely by the sheaf axiom.
\end{enumerate}

% VI17_FULL_SOLUTION_REFINED
% VI17_FULL_SOLUTION_REFINED
\begin{solution}
Let $U=D(f)\cup D(g)$.  Restriction defines
\[
\Phi:\OO_X(U)\longrightarrow A_f\times A_g,
\qquad
s\longmapsto (s|_{D(f)},s|_{D(g)}).
\]
The two components of $\Phi(s)$ have the same restriction to
\[
D(f)\cap D(g)=D(fg),
\]
so their images in $A_{fg}$ agree.  Therefore $\Phi$ lands in the fiber product
\[
A_f\times_{A_{fg}}A_g.
\]

Conversely, take $(u,v)$ in the fiber product.  Regard $u$ and $v$ as sections on $D(f)$ and $D(g)$ using the principal-open theorem.  Their equality in $A_{fg}$ says exactly that the two sections agree on the overlap.  The sheaf gluing axiom therefore produces a section $s\in\OO_X(U)$ restricting to $u$ and $v$.  Locality gives uniqueness.

The restriction and gluing constructions are inverse ring homomorphisms, hence
\[
\OO_X(U)\cong A_f\times_{A_{fg}}A_g.
\]
This is the sheaf equalizer written entirely in localization algebra.
\end{solution}



\paragraph{What happened mathematically.}
The abstract sheaf equalizer becomes a concrete localization equalizer.

\paragraph{Extensions.}
Generalize to a finite principal-open cover and write the full equalizer diagram.
