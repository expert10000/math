\begin{example}[Local fraction representatives need not be global]
A section over a general open $U$ need not admit one fraction $a/b$ valid everywhere.  The definition requires only that every point have a neighborhood on which one fraction works.  Compatible local fractions then glue by the sheaf axiom.
\end{example}

\begin{example}[The punctured affine plane]
Let $A=k[x,y]$ and
\[
U=D(x)\cup D(y)=\mathbb A^2_k\setminus\{(x,y)\}.
\]
Then
\[
\OO_X(U)=A_x\times_{A_{xy}}A_y.
\]
Inside $k(x,y)$ one has $A_x\cap A_y=A$, because a denominator that is simultaneously a power of $x$ and a power of $y$ must cancel in the UFD $k[x,y]$.  Hence
\[
\OO_X(U)\cong k[x,y].
\]
\end{example}

\begin{example}[Removing the crossing point separates branches]
Let $A=k[x,y]/(xy)$ and $U=D(x)\cup D(y)$.  Since
\[
D(x)\cap D(y)=D(xy)=\varnothing,
\]
a section on $U$ is an arbitrary pair
\[
(s_x,s_y)\in A_x\times A_y.
\]
The two branches can therefore carry independent functions once the intersection point has been removed.
\end{example}

\begin{example}[Generic stalks of the two components]
For $A=k[x,y]/(xy)$, the minimal primes are $(x)$ and $(y)$.  Localizing at $(x)$ makes $y$ invertible and forces $x=0$, hence
\[
A_{(x)}\cong k(y).
\]
Similarly $A_{(y)}\cong k(x)$.  Each irreducible component has its own function field encoded in a generic stalk.
\end{example}

\begin{example}[The reduced quotient changes the sheaf]
For $A=k[\varepsilon]/(\varepsilon^2)$,
\[
A_{\mathrm{red}}=k.
\]
The underlying spectra are homeomorphic, but on the unique open nonempty set the section rings are $A$ and $k$.  The quotient map kills precisely the nilpotent infinitesimal direction.
\end{example}

\begin{example}[Clopen pieces and idempotents]
If $e^2=e$, then
\[
D(e)\cap D(1-e)=\varnothing,
\qquad
D(e)\cup D(1-e)=X.
\]
The sheaf therefore decomposes into independent section theories on two clopen pieces.  Algebraically this reflects the product decomposition
\[
A\cong A_e\times A_{1-e}.
\]
\end{example}

\begin{example}[A germ represented on a principal neighborhood]
Given $\alpha\in A_{\mfp}$, choose a representation $a/s$ with $s\notin\mfp$.  Then $\mfp\in D(s)$ and
\[
\frac as\in A_s=\OO_X(D(s))
\]
represents the germ $\alpha$.  Thus every stalk element is represented by a regular section on a principal neighborhood.
\end{example}

\begin{example}[Equality of germs]
Two fractions $a/s$ and $b/t$ define the same germ in $A_{\mfp}$ iff there exists $u\notin\mfp$ with
\[
u(at-bs)=0.
\]
Equivalently, after restricting to $D(ust)$ near $\mfp$, the corresponding regular sections agree.
\end{example}

\begin{example}[The value field varies under specialization]
In $\Spec\mathbb Z$, the generic point has value field $\mathbb Q$, while the closed point $(p)$ has value field $\mathbb F_p$.  A global integer $n$ therefore has generic value $n\in\mathbb Q$ and special value $n\bmod p\in\mathbb F_p$.
\end{example}

\begin{example}[A section can vanish locally without being zero globally]
In $A=k[x,y]/(xy)$, the global section $x$ has zero germ at the generic point $(x)$ of the $y$-axis, but it is nonzero at the generic point $(y)$ of the $x$-axis.  The structure sheaf records componentwise behavior invisible to a single scalar evaluation.
\end{example}

\begin{example}[Principal covers and the sheaf equalizer]
If $D(f_i)$ cover $X$, then regular sections satisfy
\[
A\cong\operatorname{Eq}\left(
\prod_i A_{f_i}\rightrightarrows\prod_{i,j}A_{f_if_j}
\right).
\]
For a finite cover this is the algebraic form of gluing local fractions into a global element of $A$.
\end{example}

\begin{example}[Local rings detect the closed point]
In $A_{\mfp}$ the nonunits are exactly the fractions whose numerator lies in $\mfp$.  They form the unique maximal ideal $\mfp A_{\mfp}$.  Hence the stalk remembers which prime of $A$ is the center of the localization.
\end{example}

\begin{example}[Residue field as a fraction field]
For a prime $\mfp$, localization and quotient give
\[
A_{\mfp}/\mfp A_{\mfp}
\cong (A/\mfp)_{(0)}
\cong\Frac(A/\mfp).
\]
This gives a uniform formula valid for generic and closed points alike.
\end{example}

\end{example}
