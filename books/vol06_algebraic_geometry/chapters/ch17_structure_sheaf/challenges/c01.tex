\begin{problem}[Challenge: Finite distinguished-cover equalizer]\label{chal:vi17-c01}
Let $D(f_1),\dots,D(f_n)$ cover $\Spec A$.  Prove
\[
A\longrightarrow\prod_i A_{f_i}\rightrightarrows\prod_{i,j}A_{f_if_j}
\]
is an equalizer diagram.  Give a direct algebraic proof as well as the sheaf-theoretic proof.
\end{problem}
