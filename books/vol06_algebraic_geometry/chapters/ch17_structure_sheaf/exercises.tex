\begin{exercise}[Global sections]\label{ex:vi17-01}
Prove directly from $D(1)=\Spec A$ that $\OO_X(X)\cong A$.
\end{exercise}

\begin{exercise}[A basic-open section]\label{ex:vi17-02}
For $A=k[x]$, compute $\OO_X(D(x^3))$ and compare it with $\OO_X(D(x))$.
\end{exercise}

\begin{exercise}[Restriction]\label{ex:vi17-03}
For $A=k[x,y]$, write explicitly the restriction map $A_x\to A_{xy}$.
\end{exercise}

\begin{exercise}[Stalk from principal neighborhoods]\label{ex:vi17-04}
Show that $\varinjlim_{f\notin\mfp}A_f\cong A_{\mfp}$.
\end{exercise}

\begin{exercise}[Unique maximal ideal]\label{ex:vi17-05}
Prove that $A_{\mfp}$ is local with maximal ideal $\mfp A_{\mfp}$.
\end{exercise}

\begin{exercise}[Residue field]\label{ex:vi17-06}
Prove $\kappa(\mfp)\cong\Frac(A/\mfp)$.
\end{exercise}

\begin{exercise}[Generic point]\label{ex:vi17-07}
If $A$ is a domain, compute $\OO_{X,(0)}$.
\end{exercise}

\begin{exercise}[Closed point on the affine line]\label{ex:vi17-08}
Compute the stalk and residue field at $(x-a)\in\Spec k[x]$.
\end{exercise}

\begin{exercise}[Arithmetic stalk]\label{ex:vi17-09}
Compute $\OO_{\Spec\mathbb Z,(p)}$ and its residue field.
\end{exercise}

\begin{exercise}[Arithmetic principal open]\label{ex:vi17-10}
Compute $\OO_{\Spec\mathbb Z}(D(30))$.
\end{exercise}

\begin{exercise}[Nilpotent principal open]\label{ex:vi17-11}
Show that if $f$ is nilpotent then $D(f)=\varnothing$ and $A_f=0$.
\end{exercise}

\begin{exercise}[Unit locus]\label{ex:vi17-12}
Show that the unit locus of $a\in A=\OO_X(X)$ is $D(a)$.
\end{exercise}

\begin{exercise}[Local zero criterion]\label{ex:vi17-13}
Show $a/1=0$ in $A_{\mfp}$ iff some $s\notin\mfp$ satisfies $sa=0$.
\end{exercise}

\begin{exercise}[One-point nonreduced scheme]\label{ex:vi17-14}
Compare the structure sheaves of $\Spec k$ and $\Spec k[\varepsilon]/(\varepsilon^2)$.
\end{exercise}

\begin{exercise}[Product ring]\label{ex:vi17-15}
Describe $\OO_{\Spec(k\times k)}$ on all open subsets.
\end{exercise}

\begin{exercise}[Equalizer on two opens]\label{ex:vi17-16}
Prove $\OO_X(D(f)\cup D(g))\cong A_f\times_{A_{fg}}A_g$.
\end{exercise}

\begin{exercise}[Empty intersection]\label{ex:vi17-17}
For $A=k[x,y]/(xy)$, compute sections on $D(x)\cup D(y)$.
\end{exercise}

\begin{exercise}[Local representative]\label{ex:vi17-18}
Given $\alpha\in A_{\mfp}$, find a principal neighborhood on which $\alpha$ is represented by a section.
\end{exercise}

\begin{exercise}[Equality of germs]\label{ex:vi17-19}
Give an algebraic criterion for $a/s$ and $b/t$ to have the same germ at $\mfp$.
\end{exercise}

\begin{exercise}[Values at points]\label{ex:vi17-20}
For $A=\mathbb Z$ and global section $n$, compute its value at $(0)$ and at $(p)$.
\end{exercise}

\begin{exercise}[Irreducible component stalk]\label{ex:vi17-21}
For $A=k[x,y]/(xy)$ compute the stalks at the minimal primes $(x)$ and $(y)$.
\end{exercise}

\begin{exercise}[Reduced quotient]\label{ex:vi17-22}
Describe the induced map on stalks from $A\to A_{\mathrm{red}}$.
\end{exercise}

\begin{exercise}[Principal-cover gluing]\label{ex:vi17-23}
Suppose $D(f)\cup D(g)=X$.  Show compatible elements of $A_f$ and $A_g$ glue to a unique element of $A$.
\end{exercise}

\begin{exercise}[Locality of regularity]\label{ex:vi17-24}
Show that if a family of germs is locally represented by fractions, then its restriction to any open subset has the same property.
\end{exercise}

\begin{exercise}[Sheaf locality]\label{ex:vi17-25}
Prove two sections of $\OO_X(U)$ are equal iff all of their germs are equal.
\end{exercise}

\begin{exercise}[Bridge to affine schemes]\label{ex:vi17-26}
State precisely which extra assertion is needed to pass from the structure-sheaf formula on $D(f)$ to the legacy statement $(D(f),\OO_X|_{D(f)})\cong\Spec A_f$.
\end{exercise}
