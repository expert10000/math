\begin{example}[A one-point affine spectrum]
For $A=k$ a field, $X=\Spec k$ has one point $(0)$ and only two opens.  Since
$X=D(1)$,
\[
\OO_X(X)=k.
\]
The unique stalk is also $k$, so the structure sheaf adds no extra localization in this case.
\end{example}

\begin{example}[The punctured affine line]
Let $A=k[x]$.  On
\[
D(x)=\mathbb A^1_k\setminus V(x)
\]
we obtain
\[
\OO_X(D(x))=k[x]_x=k[x,x^{-1}].
\]
Thus $1/x$ is regular on $D(x)$ although it is not a global polynomial on $X$.
\end{example}

\begin{example}[Deleting one closed point]
For $A=k[x]$ and $a\in k$,
\[
\OO_X(D(x-a))=k[x,(x-a)^{-1}].
\]
The denominator $x-a$ is permitted precisely because it does not vanish on the chosen open.
\end{example}

\begin{example}[Inverting a product]
For $A=k[x,y]$,
\[
\OO_X(D(xy))=A_{xy}.
\]
Since $x^{-1}=y/(xy)$ and $y^{-1}=x/(xy)$ in $A_{xy}$,
\[
A_{xy}=k[x,y,x^{-1},y^{-1}].
\]
\end{example}

\begin{example}[An arithmetic principal open]
For $X=\Spec\mathbb Z$,
\[
\OO_X(D(6))=\mathbb Z[1/6].
\]
The primes $(2)$ and $(3)$ have been removed, exactly the primes at which $6$ fails to be a unit.
\end{example}

\begin{example}[The local ring at a prime integer]
At $(p)\in\Spec\mathbb Z$,
\[
\OO_{X,(p)}=\mathbb Z_{(p)}
=\left\{\frac ab:b\not\equiv0\pmod p\right\}.
\]
Its maximal ideal is $p\mathbb Z_{(p)}$ and its residue field is $\mathbb F_p$.
\end{example}

\begin{example}[The generic arithmetic stalk]
At $(0)\in\Spec\mathbb Z$, every nonzero integer is inverted, hence
\[
\OO_{X,(0)}=\mathbb Q.
\]
The generic point therefore sees rational functions rather than modular functions.
\end{example}

\begin{example}[A closed point of the affine line]
For $X=\Spec k[x]$ and $\mfm=(x-a)$,
\[
\OO_{X,\mfm}=k[x]_{(x-a)},\qquad
\kappa(\mfm)=k[x]_{(x-a)}/(x-a)\cong k.
\]
Fractions are allowed exactly when their denominators are nonzero at $a$.
\end{example}

\begin{example}[The generic point of $\mathbb A^1$]
If $A=k[x]$, then
\[
\OO_{X,(0)}=k(x).
\]
Thus the same sheaf contains both local polynomial rings at closed points and the rational-function field at the generic point.
\end{example}

\begin{example}[A closed point with a larger residue field]
Let $A=\mathbb R[x]$ and $\mfm=(x^2+1)$.  Then
\[
\OO_{X,\mfm}=\mathbb R[x]_{(x^2+1)},
\qquad
\kappa(\mfm)\cong\mathbb R[x]/(x^2+1)\cong\mathbb C.
\]
So values of regular functions need not lie in the base field $\mathbb R$.
\end{example}

\begin{example}[A nonreduced point]
For $A=k[\varepsilon]/(\varepsilon^2)$, the spectrum has one point, but
\[
\OO_X(X)=A
\]
contains the nonzero nilpotent section $\varepsilon$.  The structure sheaf sees infinitesimal information that the topology misses.
\end{example}

\begin{example}[A product ring]
For $A=k\times k$, $\Spec A$ is the disjoint union of two points.  The idempotents
$e_1=(1,0)$ and $e_2=(0,1)$ cut out the two clopen pieces, and the restrictions of the structure sheaf to them have section ring $k$.
\end{example}

\begin{example}[The crossing $xy=0$]
Let $A=k[x,y]/(xy)$ and let $\mfm=(x,y)$.  The stalk
\[
\OO_{X,\mfm}=A_{(x,y)}
\]
remembers both branches and the relation $xy=0$.  Its residue field is $k$.
\end{example}

\begin{example}[Restriction means one more localization]
For $A=k[x,y]$,
\[
D(xy)\subseteq D(x).
\]
The restriction
\[
\OO_X(D(x))=A_x\longrightarrow\OO_X(D(xy))=A_{xy}
\]
is obtained by additionally inverting $y$.
\end{example}

\begin{example}[A rational expression regular on its natural domain]
The fraction
\[
\frac{x+y}{y}
\]
defines a section on $D(y)\subseteq\Spec k[x,y]$ because $y$ is invertible there.  Its value at a prime $\mfp\in D(y)$ is the residue of $(x+y)/y$ in $\kappa(\mfp)$.
\end{example}

\begin{example}[Global sections are the original ring]
Because $X=D(1)$,
\[
\OO_X(X)=A_1\cong A.
\]
Thus the global regular section represented by $a\in A$ is the family $a/1\in A_{\mfp}$ for all primes $\mfp$.
\end{example}

\begin{example}[A principal open can be empty]
If $f$ is nilpotent, then $D(f)=\varnothing$.  Localizing at $f$ forces a nilpotent element to be a unit, hence forces $1=0$; therefore
\[
A_f=0=\OO_X(\varnothing).
\]
\end{example}

\begin{example}[The unit locus of a section]
For $a\in A$, the germ $a/1\in A_{\mfp}$ is invertible iff $a\notin\mfp$.  Consequently the set of points where the global section $a$ is a unit is exactly $D(a)$.
\end{example}

\begin{example}[Local vanishing in a ring with zero divisors]
Let $A=k[x,y]/(xy)$ and consider $a=x$.  At the prime $(x)$ the element $y\notin(x)$ and $yx=0$, so
\[
x/1=0\quad\text{in }A_{(x)}.
\]
Thus local vanishing can occur because an annihilator becomes invertible.
\end{example}

\begin{example}[A section determined by compatible principal pieces]
Let $U=D(f)\cup D(g)$.  If $s_f\in A_f$ and $s_g\in A_g$ have the same image in $A_{fg}$, the sheaf axiom produces a unique
\[
s\in\OO_X(U)
\]
restricting to them.  This is the equalizer description of sections on $U$.
\end{example}
