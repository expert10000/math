\begin{solution}[Exercise VI.17.1]
Since $X=D(1)$, the principal-open theorem gives $\OO_X(X)=A_1$.  Localizing at $1$ changes nothing, so $A_1\cong A$ canonically.
\end{solution}

\begin{solution}[Exercise VI.17.2]
The principal opens $D(x^3)$ and $D(x)$ are equal.  Moreover $k[x]_{x^3}=k[x]_x=k[x,x^{-1}]$, because making $x^3$ invertible makes $x$ invertible and conversely.
\end{solution}

\begin{solution}[Exercise VI.17.3]
The restriction is the canonical localization map $A_x\to(A_x)_y\cong A_{xy}$, $a/x^m\mapsto a/x^m$.  In the target one may further multiply denominators by powers of $y$.
\end{solution}

\begin{solution}[Exercise VI.17.4]
Map each $A_f$ with $f\notin\mfp$ to $A_{\mfp}$.  These maps are compatible.  Every $a/s\in A_{\mfp}$ comes from $A_s$, and the localization equality criterion gives injectivity of the induced colimit map.
\end{solution}

\begin{solution}[Exercise VI.17.5]
If $a/s\in A_{\mfp}$ has $a\notin\mfp$, then $a$ is inverted and $a/s$ is a unit.  If $a\in\mfp$, every representative lies in $\mfp A_{\mfp}$ and cannot be a unit.  Thus the nonunits form the unique maximal ideal $\mfp A_{\mfp}$.
\end{solution}

\begin{solution}[Exercise VI.17.6]
Quotienting gives $A_{\mfp}/\mfp A_{\mfp}\cong (A/\mfp)_{(0)}$.  Since $A/\mfp$ is a domain, localization at its zero prime is its fraction field.
\end{solution}

\begin{solution}[Exercise VI.17.7]
For a domain, the complement of $(0)$ is the set of nonzero elements, so $A_{(0)}=\Frac(A)$.  By the stalk theorem this equals $\OO_{X,(0)}$.
\end{solution}

\begin{solution}[Exercise VI.17.8]
The stalk is $k[x]_{(x-a)}$.  Its maximal ideal is generated by $x-a$, and quotienting by that ideal evaluates at $a$, giving residue field $k$.
\end{solution}

\begin{solution}[Exercise VI.17.9]
One obtains $\mathbb Z_{(p)}=\{a/b:p\nmid b\}$.  Modulo its maximal ideal $p\mathbb Z_{(p)}$ the denominator remains invertible and the quotient is $\mathbb F_p$.
\end{solution}

\begin{solution}[Exercise VI.17.10]
The answer is $\mathbb Z[1/30]=\mathbb Z[1/2,1/3,1/5]$.  These are exactly the rational numbers whose denominator has no prime factors other than $2,3,5$ after cancellation.
\end{solution}

\begin{solution}[Exercise VI.17.11]
Every prime contains every nilpotent, so $D(f)=\varnothing$.  In $A_f$, if $f^N=0$ and $f$ is a unit, then $1=f^{-N}f^N=0$, hence $A_f$ is the zero ring.
\end{solution}

\begin{solution}[Exercise VI.17.12]
In the local ring $A_{\mfp}$ the germ $a/1$ is a unit iff $a\notin\mfp$.  Therefore the unit locus is $\{\mfp:a\notin\mfp\}=D(a)$.
\end{solution}

\begin{solution}[Exercise VI.17.13]
By the localization zero criterion, $a/1=0$ in $A_{\mfp}$ iff there exists $s\in A\setminus\mfp$ such that $sa=0$ in $A$.
\end{solution}

\begin{solution}[Exercise VI.17.14]
Both spectra have one point.  On the unique nonempty open, however, the section rings are $k$ and $k[\varepsilon]/(\varepsilon^2)$.  The second has a nonzero nilpotent section, so the ringed spaces are different.
\end{solution}

\begin{solution}[Exercise VI.17.15]
There are two clopen points $U_1=D((1,0))$ and $U_2=D((0,1))$.  The section rings are $k$ on each singleton, $k\times k$ on their union, and $0$ on the empty set, with restrictions the coordinate projections.
\end{solution}

\begin{solution}[Exercise VI.17.16]
A section on the union restricts to a pair in $A_f\times A_g$ with equal images on the overlap $D(fg)$.  Conversely compatible sections glue uniquely.  Hence $\OO_X(D(f)\cup D(g))\cong A_f\times_{A_{fg}}A_g$.
\end{solution}

\begin{solution}[Exercise VI.17.17]
Because $xy=0$ in $A$, $D(xy)=\varnothing$.  Thus $D(x)$ and $D(y)$ are disjoint and $\OO_X(D(x)\cup D(y))\cong A_x\times A_y$.
\end{solution}

\begin{solution}[Exercise VI.17.18]
Write $\alpha=a/s$ with $s\notin\mfp$.  Then $\mfp\in D(s)$ and $a/s\in A_s=\OO_X(D(s))$ has germ $\alpha$ at $\mfp$.
\end{solution}

\begin{solution}[Exercise VI.17.19]
The germs are equal iff $a/s=b/t$ in $A_{\mfp}$.  By the equality criterion for localization, this holds iff some $u\notin\mfp$ satisfies $u(at-bs)=0$.
\end{solution}

\begin{solution}[Exercise VI.17.20]
At $(0)$ the value of $n$ lies in $\mathbb Q$ and is simply the rational integer $n$.  At $(p)$ the residue map sends it to $n\bmod p\in\mathbb F_p$.
\end{solution}

\begin{solution}[Exercise VI.17.21]
At $(x)$ the element $y$ is invertible, so $xy=0$ implies $x=0$ and $A_{(x)}\cong k[y]_{(0)}=k(y)$.  Symmetrically $A_{(y)}\cong k(x)$.
\end{solution}

\begin{solution}[Exercise VI.17.22]
Let $N=\Nil(A)$.  For the corresponding prime $\overline{\mfp}=\mfp/N$, one has $(A/N)_{\overline{\mfp}}\cong A_{\mfp}/N_{\mfp}$.  Thus the reduced structure sheaf is obtained stalkwise by killing localized nilpotents.
\end{solution}

\begin{solution}[Exercise VI.17.23]
Compatibility gives a unique section $s\in\OO_X(X)$ by the sheaf axiom.  Since $\OO_X(X)\cong A$, the section corresponds to a unique $a\in A$, which maps to the prescribed localized elements.
\end{solution}

\begin{solution}[Exercise VI.17.24]
If $s$ is locally $a/b$ on $V\subseteq U$, then on $V\cap W$ it is still represented by the same fraction for any open $W\subseteq U$.  Hence the restriction remains locally representable.
\end{solution}

\begin{solution}[Exercise VI.17.25]
If sections are equal, their germs are equal.  Conversely, equal germs imply that around every point the sections agree after shrinking; these neighborhoods cover $U$, and the locality axiom forces global equality.
\end{solution}

\begin{solution}[Exercise VI.17.26]
One must identify the underlying space $\Spec A_f$ with $D(f)$, then identify the structure sheaf of $A_f$ with $\OO_X|_{D(f)}$ on a basis, compatibly with restrictions; the stalk maps must be local-ring isomorphisms.  That is the full VI/18 statement.
\end{solution}
