% Canonical exercises for VI/06.

\begin{exercise}\label{ex:vi06-prime-quotient}
Prove directly that an ideal \(\mathfrak p\subsetneq A\) is prime if and only if
\(A/\mathfrak p\) is an integral domain.
\end{exercise}

\begin{exercise}\label{ex:vi06-maximal-prime}
Prove that every maximal ideal is prime.  Give an example of a prime ideal that is not
maximal.
\end{exercise}

\begin{exercise}\label{ex:vi06-zero-ideal}
Show that \((0)\) is prime in \(A\) if and only if \(A\) is an integral domain.
Apply this to \(\mathbb Z\), \(k[t]\), and \(k[x,y]/(xy)\).
\end{exercise}

\begin{exercise}\label{ex:vi06-point-ideal}
For \(a=(a_1,\dots,a_n)\in k^n\), compute the kernel of evaluation at \(a\) and prove
that it is maximal.
\end{exercise}

\begin{exercise}\label{ex:vi06-k-t-primes}
Classify the prime ideals of \(k[t]\).  Specialize your answer when \(k\) is algebraically
closed.
\end{exercise}

\begin{exercise}\label{ex:vi06-real-line}
List the prime ideals of \(\mathbb R[t]\) by type.  Explain why \((t^2+1)\) is prime and
maximal although it is not the ideal of a real coordinate point.
\end{exercise}

\begin{exercise}\label{ex:vi06-chain-plane}
Verify that
\[
(0)\subset(x)\subset(x,y)
\]
is a chain of prime ideals in \(k[x,y]\).  Compute the corresponding algebraic loci and
check that their containments reverse.
\end{exercise}

\begin{exercise}\label{ex:vi06-order-reversal}
Assume \(k\) is algebraically closed.  For prime ideals \(\mathfrak p,\mathfrak q\), prove
\[
\mathfrak p\subseteq\mathfrak q
\quad\Longleftrightarrow\quad
V(\mathfrak q)\subseteq V(\mathfrak p).
\]
\end{exercise}

\begin{exercise}\label{ex:vi06-xy-primes}
Let \(\mathfrak p\subset k[x,y]\) be prime and suppose \((xy)\subseteq\mathfrak p\).
Prove that \(x\in\mathfrak p\) or \(y\in\mathfrak p\).  Interpret this geometrically.
\end{exercise}

\begin{exercise}\label{ex:vi06-quotient-correspondence}
Prove the prime correspondence between primes of \(A/I\) and primes of \(A\) containing
\(I\), including preservation of inclusions.
\end{exercise}

\begin{exercise}\label{ex:vi06-irred-prime}
Assume \(k\) is algebraically closed.  Prove that an algebraic subset
\(Y\subseteq\mathbb A_k^n\) is irreducible if and only if \(I(Y)\) is prime.
\end{exercise}

\begin{exercise}\label{ex:vi06-maximal-points}
Assume \(k\) algebraically closed and use the weak Nullstellensatz to show that the maximal
ideals of \(k[x,y]\) are exactly \((x-a,y-b)\).
\end{exercise}

\begin{exercise}\label{ex:vi06-Z-six}
Determine all prime ideals of \(\mathbb Z\) that contain \((6)\), \((30)\), and \((0)\).
What arithmetic information is the containment relation recording?
\end{exercise}

\begin{exercise}\label{ex:vi06-radical-not-prime}
Show that \((xy)\subset k[x,y]\) is radical but not prime.  Compare
\(k[x,y]/(xy)\) with a domain quotient.
\end{exercise}

\begin{exercise}\label{ex:vi06-minimal-primes-cross}
In \(A=k[x,y]/(xy)\), prove that \((\bar x)\) and \((\bar y)\) are the minimal prime
ideals.  Relate them to the irreducible components of the crossing axes.
\end{exercise}

\begin{exercise}\label{ex:vi06-preimage-prime}
Let \(\varphi:A\to B\) be a unital ring homomorphism and \(\mathfrak q\subset B\) prime.
Prove that \(\varphi^{-1}(\mathfrak q)\) is prime.
\end{exercise}

\begin{exercise}\label{ex:vi06-products-ideals}
Let \(\mathfrak p\) be prime and let \(I,J\subseteq A\) be ideals with
\(IJ\subseteq\mathfrak p\).  Prove that \(I\subseteq\mathfrak p\) or
\(J\subseteq\mathfrak p\).
\end{exercise}

\begin{exercise}\label{ex:vi06-complex-splitting}
Show that \((t^2+1)\) is prime in \(\mathbb R[t]\) but that its extension to
\(\mathbb C[t]\) is not prime.  Factor the polynomial and identify the two resulting
maximal ideals.
\end{exercise}
