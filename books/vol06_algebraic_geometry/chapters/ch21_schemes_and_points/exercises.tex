\end{exercise}
\begin{exercise}\label{exer:vi21-e02}
Compute the image of $(2,t)$ and $(0)$ in $\Spec\mathbb Z[t]$ under the absolute structure morphism.
\end{exercise}
\begin{exercise}\label{exer:vi21-e03}
For $x\in X$, prove directly that $\Spec\kappa(x)\to X$ is unchanged after replacing an affine neighbourhood by a smaller principal neighbourhood.
\end{exercise}
\begin{exercise}\label{exer:vi21-e04}
Classify morphisms $\Spec\mathbb C\to\Spec\mathbb R[t]$ whose image is $(t^2+1)$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e05}
Describe the specialization order on $\Spec k[t]$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e06}
Show that a closed point can have a nontrivial residue-field extension of the base field.
\end{exercise}
\begin{exercise}\label{exer:vi21-e07}
For $A\to B$ and $\mathfrak q\in\Spec B$, write the induced map $\kappa(\mathfrak q\cap A)\to\kappa(\mathfrak q)$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e08}
Prove that an isomorphism of schemes induces isomorphisms on all residue fields.
\end{exercise}
\begin{exercise}\label{exer:vi21-e09}
For a domain $A$, identify the residue field at the generic point.
\end{exercise}
\begin{exercise}\label{exer:vi21-e10}
Check that $\Spec A_{\mathfrak p}\to\Spec A$ sends its closed point to $\mathfrak p$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e11}
Identify all points in the image of $\Spec A_{\mathfrak p}\to\Spec A$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e12}
Prove that the image of a specialization chain under a scheme morphism is again a specialization chain, possibly with repetitions.
\end{exercise}
\begin{exercise}\label{exer:vi21-e13}
Show that $X(k)$ for $X=\mathbb A^n_k$ is naturally $k^n$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e14}
For $X=\Spec k[x,y]/(xy)$, describe the canonical residue-field morphisms at the two generic points of the irreducible components.
\end{exercise}
\begin{exercise}\label{exer:vi21-e15}
Explain why $\Spec k[\varepsilon]/(\varepsilon^2)$ has the same point set as $\Spec k$ but different stalk data.
\end{exercise}
\begin{exercise}\label{exer:vi21-e16}
Prove that $\pi_X$ is compatible with restriction to an open subscheme.
\end{exercise}
\begin{exercise}\label{exer:vi21-e17}
If every point of $X$ has residue characteristic $p$, determine the image of the underlying point map $X\to\Spec\mathbb Z$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e18}
Give an example of two distinct $K$-valued points with the same image point.
\end{exercise}
\begin{exercise}\label{exer:vi21-e19}
Show that a $k$-rational point $x$ has a $k$-algebra homomorphism $\kappa(x)\to k$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e20}
When $k$ is algebraically closed, prove that a closed point of an affine finite-type $k$-scheme has residue field $k$ only if you may invoke the appropriate Nullstellensatz; record the dependency rather than proving it here.
\end{exercise}
\begin{exercise}\label{exer:vi21-e21}
Compute the residue characteristic of the point $(3,t^2+1)\in\Spec\mathbb Z[t]$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e22}
For $X\to Y\to Z$, draw the stalk and residue-field diagrams at a point $x\in X$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e23}
Prove that the canonical map $\Spec\kappa(x)\to X$ factors through every open neighbourhood of $x$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e24}
For $A\to K$ with $K$ a field, prove that the induced local map $A_{\mathfrak p}\to K$ exists for $\mathfrak p=\ker(A\to K)$.
\end{exercise}
\begin{exercise}\label{exer:vi21-e25}
Explain why the final-object property of $\Spec\mathbb Z$ is stronger than the affine statement alone.
\end{exercise}
\begin{exercise}\label{exer:vi21-e26}
Summarize in one diagram the data attached to $x\in X$: stalk, maximal ideal, residue field, and point morphism.
\end{exercise}
