\chapter{Schemes and Their Points}\label{ch:vi21-schemes-points}

A scheme is simultaneously a topological space, a sheaf of rings, and a system of local
rings.  Its points therefore carry more information than locations alone.  Each point has a
local ring, a residue field, a specialization position, and functorial behavior under scheme
morphisms.  This chapter makes that pointwise language explicit and places every scheme over
the absolute arithmetic base $\Spec\mathbb Z$.

\section{The absolute structure morphism}
The universal property proved in VI/19 gives, for every scheme $X$,
\[
 \Hom_{\Sch}(X,\Spec\mathbb Z)
 \cong \Hom_{\Rings}(\mathbb Z,\Gamma(X,\OO_X)).
\]
There is exactly one unital map $\mathbb Z\to R$ for every ring $R$.  Hence:

\begin{theorem}[Absolute structure morphism]\label{thm:vi21-absolute-structure}
For every scheme $X$ there is a unique morphism
\[
 \pi_X:X\longrightarrow \Spec\mathbb Z.
\]
Thus $\Spec\mathbb Z$ is final in $\Sch$.
\end{theorem}

\begin{remark}
The theorem is the scheme-theoretic reflection of the fact that $\mathbb Z$ is initial among
unital rings.  It is the content of legacy AG3 Problem 13, reconstructed in full as VI.21.P1.
\end{remark}

\section{What a point of a scheme carries}
Let $x\in X$.  The stalk $\OO_{X,x}$ is a local ring with maximal ideal
$\mathfrak m_x$, and the residue field is
\[
 \kappa(x)=\OO_{X,x}/\mathfrak m_x.
\]
The quotient map $\OO_{X,x}\to\kappa(x)$ packages evaluation at $x$ in the local language.

\begin{proposition}[Canonical residue-field point]\label{prop:vi21-residue-point}
Every point $x\in X$ determines a canonical morphism
\[
 i_x:\Spec\kappa(x)\longrightarrow X
\]
whose unique source point maps to $x$.
\end{proposition}
\begin{proof}
Choose an affine neighbourhood $U=\Spec A$ of $x$, corresponding to a prime
$\mathfrak p\subset A$.  The field map $A\to A_{\mathfrak p}\to\kappa(\mathfrak p)$ induces
$\Spec\kappa(\mathfrak p)\to\Spec A=U\to X$.  Independence of $U$ follows because on a smaller
affine neighbourhood the maps are induced by the same local homomorphism
$\OO_{X,x}\to\kappa(x)$.
\end{proof}

\section{Field-valued points}
The canonical point morphism has a useful converse.

\begin{theorem}[Field-valued points]\label{thm:vi21-field-valued-points}
Let $K$ be a field.  To give a morphism
\[
 u:\Spec K\longrightarrow X
\]
is equivalent to giving a pair $(x,\iota)$ consisting of a point $x\in X$ and a field
homomorphism
\[
 \iota:\kappa(x)\longrightarrow K.
\]
\end{theorem}
\begin{proof}
A morphism $u$ has a unique image point $x$.  The local stalk map
$\OO_{X,x}\to K$ kills $\mathfrak m_x$, hence factors uniquely through $\kappa(x)$.  Conversely,
compose the canonical map $\Spec K\to\Spec\kappa(x)$ induced by $\iota$ with $i_x$.
The constructions are inverse.
\end{proof}

\section{Morphisms on points, stalks, and residue fields}
A scheme morphism $f:X\to Y$ gives three compatible layers of data:
\[
 x\longmapsto f(x),\qquad
 f_x^\#:\OO_{Y,f(x)}\longrightarrow\OO_{X,x},\qquad
 \kappa(f(x))\longrightarrow\kappa(x).
\]
The stalk homomorphism is local, so it descends to the indicated residue-field map.

\begin{proposition}[Residue-field functoriality]\label{prop:vi21-residue-functoriality}
For morphisms $X\xrightarrow{f}Y\xrightarrow{g}Z$ and $x\in X$, the residue-field map for
$g\circ f$ is the composite
\[
 \kappa(g(f(x)))\longrightarrow\kappa(f(x))\longrightarrow\kappa(x).
\]
\end{proposition}

\section{Specialization and generalization}
A point $x$ is a \emph{specialization} of $y$ if $x\in\overline{\{y\}}$; equivalently $y$ is a
generalization of $x$.  In an affine scheme $\Spec A$ this is exactly reverse inclusion of prime
ideals:
\[
 \mathfrak p\leadsto\mathfrak q
 \quad\Longleftrightarrow\quad
 \mathfrak p\subseteq\mathfrak q,
\]
where $\mathfrak q$ is a specialization of $\mathfrak p$.

\begin{proposition}[Morphisms preserve specialization]\label{prop:vi21-specialization}
If $x$ is a specialization of $y$ in $X$ and $f:X\to Y$ is a scheme morphism, then $f(x)$ is a
specialization of $f(y)$.
\end{proposition}
\begin{proof}
Continuity gives
$f(x)\in f(\overline{\{y\}})\subseteq\overline{\{f(y)\}}$.
\end{proof}

\section{The arithmetic image of a point}
The structure morphism detects the characteristic of residue fields.

\begin{proposition}[Residue characteristic]\label{prop:vi21-residue-characteristic}
Let $x\in X$.  Under $\pi_X:X\to\Spec\mathbb Z$,
\[
 \pi_X(x)=\ker(\mathbb Z\to\kappa(x)).
\]
Hence $\pi_X(x)=(0)$ when $\operatorname{char}\kappa(x)=0$, and
$\pi_X(x)=(p)$ when $\operatorname{char}\kappa(x)=p>0$.
\end{proposition}

\section{Affine formulas}
For $X=\Spec A$ and $x=\mathfrak p$, one has
\[
 \OO_{X,x}=A_{\mathfrak p},\qquad
 \kappa(x)=\operatorname{Frac}(A/\mathfrak p)
\]
when $A/\mathfrak p$ is a domain, with the second expression understood as its fraction field.
If $A$ is a $k$-algebra, then $k$-rational points of $X$ are the morphisms
$\Spec k\to X$ over $\Spec k$, equivalently the $k$-algebra maps $A\to k$.

\section{Schemes over a base}\label{sec:vi21-schemes-over-base}
The absolute structure morphism is automatic, but much of algebraic geometry is relative to a chosen base.

\begin{definition}[$S$-scheme]
Let $S$ be a scheme.  An \emph{$S$-scheme} is a scheme $X$ equipped with a specified morphism
\[
p_X:X\longrightarrow S.
\]
A morphism of $S$-schemes $f:X\to Y$ is a scheme morphism satisfying
\[
p_Y\circ f=p_X.
\]
Equivalently, the category of $S$-schemes is the slice category $\mathbf{Sch}/S$.
\end{definition}

When $S=\operatorname{Spec}k$ for a field $k$, one speaks of a \emph{$k$-scheme}.  If $K/k$ is a field extension, the inclusion $k\hookrightarrow K$ gives $\operatorname{Spec}K\to\operatorname{Spec}k$, and one defines
\[
\boxed{X(K)=\operatorname{Hom}_{\operatorname{Spec}k}(\operatorname{Spec}K,X).}
\]
In particular,
\[
X(k)=\operatorname{Hom}_{\operatorname{Spec}k}(\operatorname{Spec}k,X)
\]
is the set of $k$-rational points.

This relative notation is stricter than the absolute set $\operatorname{Hom}(\operatorname{Spec}K,X)$: the triangle over $\operatorname{Spec}k$ must commute.  For an affine $k$-scheme $X=\operatorname{Spec}A$,
\[
X(K)\cong\operatorname{Hom}_{k\text{-alg}}(A,K).
\]
The relative viewpoint is the natural language for fiber products, base change, fibers, and geometric points in the following chapters.
\section{Local schemes around a point}
The local ring itself has a geometric spectrum.  There is a canonical morphism
\[
 \lambda_x:\Spec\OO_{X,x}\longrightarrow X
\]
whose closed point maps to $x$.  Its points encode generalizations of $x$ visible in sufficiently
small affine neighbourhoods.  This construction is local and should not be confused with the
closed/open subscheme formalism deferred to VI/22.

\section{Why the local condition matters}
A morphism of schemes is a morphism of \emph{locally} ringed spaces: for each $x$, the map
$f_x^\#$ must carry the maximal ideal of $\OO_{Y,f(x)}$ into the maximal ideal of $\OO_{X,x}$,
or equivalently pull the latter maximal ideal back to the former.  This is precisely what makes
residue-field maps functorial and makes the image point algebraically correct.

\section{Chapter synthesis}
The pointwise data of a scheme are organized by the diagram
\[
 \text{point }x
 \rightsquigarrow
 \OO_{X,x}
 \rightsquigarrow
 \kappa(x)
 \rightsquigarrow
 \Spec\kappa(x)\to X,
\]
while every morphism transports this structure by local maps.  The absolute structure morphism
then places every point over a unique prime of $\mathbb Z$, recording its residue characteristic.
