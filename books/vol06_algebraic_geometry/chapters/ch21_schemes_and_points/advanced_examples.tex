\begin{example}[Field-valued point classification]\label{ex:vi21-ae01}
For any field $K$, $X(K)=\Hom(\Spec K,X)$ decomposes as pairs $(x,\kappa(x)\to K)$.  This separates the topological image from the residue-field embedding.
\end{example}
\begin{example}[Geometric points]\label{ex:vi21-ae02}
When $K$ is algebraically closed, a morphism $\Spec K\to X$ is often called a geometric point.  Its image need not be closed without additional finiteness hypotheses.
\end{example}
\begin{example}[The local scheme at a point]\label{ex:vi21-ae03}
$\Spec\OO_{X,x}\to X$ packages all sufficiently local affine information around $x$; its closed point maps to $x$ and its other points represent generalizations.
\end{example}
\begin{example}[Specialization order and prime inclusion]\label{ex:vi21-ae04}
On $\Spec A$, the specialization preorder is exactly inclusion of primes.  This order is functorial under every scheme morphism.
\end{example}
\begin{example}[Residue-characteristic stratification of points]\label{ex:vi21-ae05}
The point set of any scheme is partitioned by the primes of $\mathbb Z$ according to residue characteristic, with characteristic zero over $(0)$ and characteristic $p$ over $(p)$.
\end{example}
\begin{example}[Factorization through $\Spec\mathbb F_p$]\label{ex:vi21-ae06}
The absolute structure morphism factors through $\Spec\mathbb F_p$ exactly when the global section $p\cdot1$ is zero; equivalently it is zero in every stalk.
\end{example}
\begin{example}[Affine points after field extension]\label{ex:vi21-ae07}
For a $k$-algebra $A$ and field extension $K/k$, affine points are described by
\[
\Hom_k(\Spec K,\Spec A)\cong\Hom_{k\text{-alg}}(A,K).
\]
\end{example}
\begin{example}[Residue fields and composition]\label{ex:vi21-ae08}
For $X\to Y\to Z$, the induced residue-field embeddings compose contravariantly along the image chain.
\end{example}
\begin{example}[Underlying topology forgets nilpotents]\label{ex:vi21-ae09}
Reduction $X_{\mathrm{red}}\to X$ is a homeomorphism on point sets while changing local rings.  This is a canonical reminder that scheme points are enriched by sheaf data.
\end{example}
\begin{example}[The generic point as a field-valued point]\label{ex:vi21-ae10}
For an integral scheme with generic point $\eta$, the canonical $\Spec\kappa(\eta)\to X$ is the generic-point morphism and $\kappa(\eta)$ is the function field when that terminology is introduced later.
\end{example}
\begin{example}[Arithmetic movement under specialization]\label{ex:vi21-ae11}
If $x$ specializes to $y$, then $\pi_X(x)$ specializes to $\pi_X(y)$ in $\Spec\mathbb Z$.  Along a specialization $x\leadsto y$, residue characteristic may change $0\to p$, but never $p\to0$.  In particular, a positive-characteristic point cannot specialize to a characteristic-zero point.
\end{example}
\begin{example}[Point detection by affine neighbourhoods]\label{ex:vi21-ae12}
The image of $\Spec K\to X$ can be studied in any affine neighbourhood of its image, reducing point questions to kernels of ring maps into fields.
\end{example}
\begin{example}[Locality of image primes]\label{ex:vi21-ae13}
Given $A\to K$ to a field, the image point in $\Spec A$ is the kernel.  Localizing at that kernel produces the unique local map $A_{\mathfrak p}\to K$.
\end{example}
\begin{example}[Absolute versus relative viewpoints]\label{ex:vi21-ae14}
Every scheme is automatically over $\Spec\mathbb Z$.  Choosing another base $S$ adds structure; it does not replace the absolute structure morphism, which is obtained by composition $X\to S\to\Spec\mathbb Z$.
\end{example}
