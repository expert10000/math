% VI21-EXERCISE-SOLUTION: E01
\paragraph{Solution to VI.21.E1.}
\begin{solution}
Let $X=\Spec\mathbb F_p[t]$.  The absolute structure morphism is induced by
\[
\mathbb Z\longrightarrow \mathbb F_p[t],\qquad n\longmapsto \overline n.
\]
Let $\mathfrak q\in\Spec\mathbb F_p[t]$.  Its image is the contraction
$\mathfrak q\cap\mathbb Z$.  The kernel of the displayed ring map is $(p)$, so $(p)$ is
contained in every contraction.  If $n\notin(p)$, then $\overline n\in\mathbb F_p^\times$ is
a unit of $\mathbb F_p[t]$, and a proper prime ideal cannot contain a unit.  Hence no larger
ideal of $\mathbb Z$ occurs and
\[
\pi_X(\mathfrak q)=(p)
\]
for every point $\mathfrak q$.  Thus the whole nonempty spectrum lies over the characteristic-$p$
point of $\Spec\mathbb Z$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E02
\paragraph{Solution to VI.21.E2.}
\begin{solution}
For $X=\Spec\mathbb Z[t]$, the absolute structure morphism comes from the inclusion
$\mathbb Z\hookrightarrow\mathbb Z[t]$, so its point map is contraction.  Therefore
\[
(2,t)\cap\mathbb Z=(2).
\]
For the generic prime $(0)\subset\mathbb Z[t]$, contraction is
\[
(0)\cap\mathbb Z=(0),
\]
because the inclusion of $\mathbb Z$ in the domain $\mathbb Z[t]$ is injective.  Hence the first
point has residue characteristic $2$, whereas the generic point lies over the characteristic-zero
generic point of $\Spec\mathbb Z$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E03
\paragraph{Solution to VI.21.E3.}
\begin{solution}
Choose an affine neighbourhood $U=\Spec A$ of $x$, with $x$ corresponding to
$\mathfrak p\subset A$.  The canonical residue-field morphism is induced by
\[
A\longrightarrow A_{\mathfrak p}\longrightarrow\kappa(\mathfrak p).
\]
Now let $D(f)\subseteq U$ be a smaller principal neighbourhood of $x$.  Then $f\notin\mathfrak p$,
so the image of $f$ in $\kappa(\mathfrak p)$ is nonzero and hence invertible.  By the universal
property of localization, the same map factors uniquely as
\[
A\longrightarrow A_f\longrightarrow\kappa(\mathfrak p).
\]
Applying $\Spec$ shows that the morphism $\Spec\kappa(x)\to D(f)\to U$ is exactly the original
morphism $\Spec\kappa(x)\to U$.  Composing with $U\hookrightarrow X$ proves independence under
shrinking to a principal neighbourhood.
\end{solution}

% VI21-EXERCISE-SOLUTION: E04
\paragraph{Solution to VI.21.E4.}
\begin{solution}
A morphism $u:\Spec\mathbb C\to\Spec\mathbb R[t]$ with image
$\mathfrak p=(t^2+1)$ is, by the field-valued point theorem, equivalent to a field homomorphism
\[
\kappa(\mathfrak p)=\mathbb R[t]/(t^2+1)\cong\mathbb C\longrightarrow\mathbb C.
\]
The induced map fixes $\mathbb R$ (equivalently, it comes from a ring map with kernel exactly
$(t^2+1)$), so it is an $\mathbb R$-embedding of $\mathbb C$ into itself.  Such an embedding is
determined by the image of $i$, which must satisfy $z^2+1=0$.  Thus $i$ can map only to $i$ or
$-i$.  Consequently there are exactly two morphisms with that image point, corresponding to the
identity and complex conjugation.
\end{solution}

% VI21-EXERCISE-SOLUTION: E05
\paragraph{Solution to VI.21.E5.}
\begin{solution}
Since $k[t]$ is a principal ideal domain, its prime ideals are $(0)$ and the ideals
$(f)$ generated by irreducible polynomials $f$.  The closure formula
\[
\overline{\{\mathfrak p\}}=V(\mathfrak p)
\]
shows that $\mathfrak q$ is a specialization of $\mathfrak p$ exactly when
$\mathfrak p\subseteq\mathfrak q$.  Thus $(0)$ is the generic point and specializes to every
closed point $(f)$.  A nonzero prime $(f)$ is maximal, hence has no proper specialization other
than itself.  Distinct closed points are incomparable in the specialization order.
\end{solution}

% VI21-EXERCISE-SOLUTION: E06
\paragraph{Solution to VI.21.E6.}
\begin{solution}
Take $X=\mathbb A^1_{\mathbb R}=\Spec\mathbb R[t]$ and the maximal ideal
\[
\mathfrak m=(t^2+1).
\]
It is maximal because the quotient
\[
\mathbb R[t]/(t^2+1)\cong\mathbb C
\]
is a field.  Hence $\mathfrak m$ is a closed point, but its residue field is
$\kappa(\mathfrak m)\cong\mathbb C$, a nontrivial extension of the base field $\mathbb R$.
Therefore closed points need not be rational over the chosen base field.
\end{solution}

% VI21-EXERCISE-SOLUTION: E07
\paragraph{Solution to VI.21.E7.}
\begin{solution}
Let $\varphi:A\to B$, let $\mathfrak q\in\Spec B$, and put
$\mathfrak p=\varphi^{-1}(\mathfrak q)$.  Since $s\notin\mathfrak p$ implies
$\varphi(s)\notin\mathfrak q$, localization gives a local homomorphism
\[
A_{\mathfrak p}\longrightarrow B_{\mathfrak q},\qquad
\frac a s\longmapsto\frac{\varphi(a)}{\varphi(s)}.
\]
It carries $\mathfrak pA_{\mathfrak p}$ into $\mathfrak qB_{\mathfrak q}$ and therefore descends
to
\[
\kappa(\mathfrak p)\longrightarrow\kappa(\mathfrak q),\qquad
\overline{a/s}\longmapsto\overline{\varphi(a)/\varphi(s)}.
\]
This is the residue-field map attached to the affine scheme morphism $\Spec B\to\Spec A$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E08
\paragraph{Solution to VI.21.E8.}
\begin{solution}
Let $f:X\to Y$ be an isomorphism with inverse $g$.  For $x\in X$, functoriality gives
field maps
\[
\kappa(f(x))\xrightarrow{f_x}\kappa(x),\qquad
\kappa(x)\xrightarrow{g_{f(x)}}\kappa(f(x)).
\]
The residue-field map attached to $g\circ f=\operatorname{id}_X$ is the identity of
$\kappa(x)$, and the map attached to $f\circ g=\operatorname{id}_Y$ is the identity of
$\kappa(f(x))$.  Therefore the two displayed field maps are inverse isomorphisms.  Thus scheme
isomorphisms preserve residue fields point by point.
\end{solution}

% VI21-EXERCISE-SOLUTION: E09
\paragraph{Solution to VI.21.E9.}
\begin{solution}
If $A$ is a domain, the generic point of $\Spec A$ is $(0)$.  Localization at this prime
inverts every nonzero element, hence
\[
A_{(0)}=\operatorname{Frac}(A).
\]
The maximal ideal of this local ring is $(0)A_{(0)}=0$, so its residue field is
\[
\kappa((0))=A_{(0)}/0=\operatorname{Frac}(A).
\]
Thus the generic point naturally carries the function field of the affine integral scheme.
\end{solution}

% VI21-EXERCISE-SOLUTION: E10
\paragraph{Solution to VI.21.E10.}
\begin{solution}
The points of $\Spec A_{\mathfrak p}$ correspond to primes of $A$ contained in
$\mathfrak p$.  Its unique closed point is the maximal ideal
$\mathfrak pA_{\mathfrak p}$.  Under the localization morphism
\[
\Spec A_{\mathfrak p}\longrightarrow\Spec A,
\]
a prime is sent to its contraction.  The contraction of $\mathfrak pA_{\mathfrak p}$ is
$\mathfrak p$, so the closed point of the local scheme maps exactly to the chosen point
$\mathfrak p$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E11
\paragraph{Solution to VI.21.E11.}
\begin{solution}
Prime ideals of $A_{\mathfrak p}$ are in bijection with prime ideals
$\mathfrak q\subseteq A$ disjoint from $A\setminus\mathfrak p$, equivalently with primes
$\mathfrak q\subseteq\mathfrak p$.  Under the canonical map to $\Spec A$, these primes map by
contraction to the same $\mathfrak q$.  Hence the image is
\[
\{\mathfrak q\in\Spec A:\mathfrak q\subseteq\mathfrak p\}.
\]
Since $\mathfrak p$ is a specialization of $\mathfrak q$ precisely when
$\mathfrak q\subseteq\mathfrak p$, this is exactly the set of generalizations of
$\mathfrak p$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E12
\paragraph{Solution to VI.21.E12.}
\begin{solution}
Write a specialization chain as
\[
x_0\leadsto x_1\leadsto\cdots\leadsto x_r,
\qquad x_{i+1}\in\overline{\{x_i\}}.
\]
For a scheme morphism $f:X\to Y$, continuity gives
\[
f(x_{i+1})\in f(\overline{\{x_i\}})\subseteq
\overline{\{f(x_i)\}}.
\]
Thus each $f(x_{i+1})$ is a specialization of $f(x_i)$.  It may happen that
$f(x_i)=f(x_{i+1})$, so repetitions are allowed, but the order relation can never be reversed.
\end{solution}

% VI21-EXERCISE-SOLUTION: E13
\paragraph{Solution to VI.21.E13.}
\begin{solution}
For $X=\mathbb A^n_k=\Spec k[t_1,\dots,t_n]$, a $k$-rational point is a morphism over
$\Spec k$ from $\Spec k$ to $X$.  By affine anti-equivalence in the slice over $k$, such morphisms
are the same as $k$-algebra homomorphisms
\[
k[t_1,\dots,t_n]\longrightarrow k.
\]
A homomorphism is uniquely determined by the $n$ elements $a_i$ which are the images of $t_i$,
and every tuple $(a_1,\dots,a_n)\in k^n$ defines the evaluation map
$p\mapsto p(a_1,\dots,a_n)$.  Hence $X(k)\cong k^n$ naturally.
\end{solution}

% VI21-EXERCISE-SOLUTION: E14
\paragraph{Solution to VI.21.E14.}
\begin{solution}
Put $R=k[x,y]/(xy)$.  The two irreducible components have generic primes
$\mathfrak p_x=(x)$ and $\mathfrak p_y=(y)$ in $R$.  Since
\[
R/\mathfrak p_x\cong k[y],\qquad R/\mathfrak p_y\cong k[x],
\]
their residue fields are
\[
\kappa(\mathfrak p_x)=k(y),\qquad \kappa(\mathfrak p_y)=k(x).
\]
The canonical point morphisms are induced by the composites
\[
R\to R_{\mathfrak p_x}\to k(y),\qquad
R\to R_{\mathfrak p_y}\to k(x).
\]
Thus $\Spec k(y)$ and $\Spec k(x)$ map to the two generic points and encode the function fields
of the corresponding coordinate axes.
\end{solution}

% VI21-EXERCISE-SOLUTION: E15
\paragraph{Solution to VI.21.E15.}
\begin{solution}
Both $k$ and $A=k[\varepsilon]/(\varepsilon^2)$ have exactly one prime ideal: $(0)$ for
$k$ and $(\varepsilon)$ for $A$.  Hence both spectra are one-point topological spaces.  But the
unique stalk of an affine one-point scheme is its global ring.  For $\Spec k$ the stalk is the
field $k$, while for $\Spec A$ it is $A$, which contains the nonzero nilpotent $\varepsilon$.
Therefore the locally ringed spaces, and hence the schemes, are not isomorphic even though their
underlying point sets and topologies agree.
\end{solution}

% VI21-EXERCISE-SOLUTION: E16
\paragraph{Solution to VI.21.E16.}
\begin{solution}
Let $j:U\hookrightarrow X$ be an open subscheme.  The composite
\[
U\xrightarrow{j}X\xrightarrow{\pi_X}\Spec\mathbb Z
\]
is a morphism from $U$ to $\Spec\mathbb Z$.  By finality of $\Spec\mathbb Z$, there is exactly one
such morphism, namely $\pi_U$.  Hence
\[
\pi_U=\pi_X\circ j.
\]
On an affine open $V=\Spec A\subseteq U$ this says the same thing algebraically: both morphisms
are induced by the unique unital map $\mathbb Z\to A$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E17
\paragraph{Solution to VI.21.E17.}
\begin{solution}
For every $x\in X$, the point formula for the structure morphism is
\[
\pi_X(x)=\ker(\mathbb Z\to\kappa(x)).
\]
If every residue field has characteristic $p$, this kernel is $(p)$ for every $x$.  Therefore
\[
\pi_X(|X|)\subseteq\{(p)\}.
\]
If $X$ is nonempty, the image is exactly the singleton $\{(p)\}$; for the empty scheme the image
is empty.  This is a statement about the underlying point map only and does not by itself imply
factorization through $\Spec\mathbb F_p$.
\end{solution}

\paragraph{Solution to VI.21.E18.}
\begin{solution}
Use the point $x=(t^2+1)$ of $X=\Spec\mathbb R[t]$ and take $K=\mathbb C$.  Its residue
field is $\kappa(x)\cong\mathbb C$.  By the field-valued point theorem, morphisms
$\Spec\mathbb C\to X$ with image $x$ correspond to $\mathbb R$-embeddings
$\mathbb C\to\mathbb C$.  The identity and complex conjugation are distinct such embeddings.
Hence they determine two distinct $K$-valued points with the same underlying image point $x$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E19
\paragraph{Solution to VI.21.E19.}
\begin{solution}
A $k$-rational point is a morphism
\[
u:\Spec k\longrightarrow X
\]
over $\Spec k$.  Let $x$ be its image.  The field-valued point theorem gives a field map
\[
\iota:\kappa(x)\longrightarrow k.
\]
Because $u$ is a morphism over $k$, the composite
\[
k\longrightarrow\kappa(x)\xrightarrow{\iota}k
\]
is the identity.  Thus $\iota$ is a $k$-algebra homomorphism.  Since field homomorphisms are
injective and its image contains $k$, this in fact identifies $\kappa(x)$ with $k$ whenever the
structural map $k\to\kappa(x)$ is the usual field inclusion.
\end{solution}

% VI21-EXERCISE-SOLUTION: E20
\paragraph{Solution to VI.21.E20.}
\begin{solution}
Let $X=\Spec A$ be of finite type over an algebraically closed field $k$, and let $x$ be a
closed point corresponding to a maximal ideal $\mathfrak m\subset A$.  The desired conclusion
\[
\kappa(x)=A/\mathfrak m\cong k
\]
is precisely a consequence of the weak Nullstellensatz: every residue field of a maximal ideal in
a finitely generated $k$-algebra is a finite algebraic extension of $k$, and an algebraically closed
field has no nontrivial finite algebraic extensions.  This chapter records that dependency rather
than reproving the Nullstellensatz; no circular point-theoretic argument is being used.
\end{solution}

\paragraph{Solution to VI.21.E21.}
\begin{solution}
Let $\mathfrak m=(3,t^2+1)\subset\mathbb Z[t]$.  Its contraction to $\mathbb Z$ is
$\mathfrak m\cap\mathbb Z=(3)$, so the absolute structure morphism sends this point to $(3)$.  In
fact
\[
\kappa(\mathfrak m)=\mathbb Z[t]/(3,t^2+1)
\cong\mathbb F_3[t]/(t^2+1).
\]
The polynomial $t^2+1$ has no root in $\mathbb F_3$, so this quotient is the field $\mathbb F_9$.
Its characteristic is therefore $3$, agreeing with the contraction computation.
\end{solution}

% VI21-EXERCISE-SOLUTION: E22
\paragraph{Solution to VI.21.E22.}
\begin{solution}
For morphisms $X\xrightarrow{f}Y\xrightarrow{g}Z$ and $x\in X$, the stalk diagram is
\[
\OO_{Z,g(f(x))}\xrightarrow{g_{f(x)}^\#}
\OO_{Y,f(x)}\xrightarrow{f_x^\#}\OO_{X,x}.
\]
Each arrow is local.  Passing to quotients by the corresponding maximal ideals produces
\[
\kappa(g(f(x)))\longrightarrow\kappa(f(x))\longrightarrow\kappa(x).
\]
The residue-field map for $g\circ f$ is the composite because the stalk map for $g\circ f$ is the
composite of the two stalk maps.  Thus the two diagrams are functorially compatible.
\end{solution}

% VI21-EXERCISE-SOLUTION: E23
\paragraph{Solution to VI.21.E23.}
\begin{solution}
Let $i_x:\Spec\kappa(x)\to X$ be the canonical point morphism and let $U\subseteq X$ be
an open neighbourhood of $x$.  The source has one point, and its image is $x\in U$, so
$i_x^{-1}(U)=\Spec\kappa(x)$.  Restricting the sheaf morphism therefore gives a morphism
\[
\Spec\kappa(x)\longrightarrow U
\]
whose composite with the open immersion $U\hookrightarrow X$ is $i_x$.  Uniqueness follows because
an open immersion is a monomorphism, or directly because both factorizations induce the same local
map $\OO_{X,x}\to\kappa(x)$.
\end{solution}

% VI21-EXERCISE-SOLUTION: E24
\paragraph{Solution to VI.21.E24.}
\begin{solution}
Let $\alpha:A\to K$ be a unital ring map to a field and put
$\mathfrak p=\ker\alpha$.  If $s\notin\mathfrak p$, then $\alpha(s)\neq0$ in $K$, hence
$\alpha(s)$ is invertible.  Therefore $\alpha$ sends every element of the multiplicative set
$A\setminus\mathfrak p$ to a unit.  By the universal property of localization there is a unique map
\[
A_{\mathfrak p}\longrightarrow K,\qquad
\frac a s\longmapsto\alpha(a)\alpha(s)^{-1}.
\]
Its inverse image of the maximal ideal $(0)$ of $K$ is $\mathfrak pA_{\mathfrak p}$, so it is a
local homomorphism.
\end{solution}

% VI21-EXERCISE-SOLUTION: E25
\paragraph{Solution to VI.21.E25.}
\begin{solution}
The affine statement says that every affine scheme $U=\Spec A$ has a unique morphism to
$\Spec\mathbb Z$, induced by the unique map $\mathbb Z\to A$.  Finality is a statement about
\emph{every} scheme $X$, which may require many affine charts.  One must know that the affine maps
agree on overlaps and glue uniquely.  Equivalently, VI/19's universal property is already stated
for an arbitrary locally ringed source, so it gives
\[
\operatorname{Hom}(X,\Spec\mathbb Z)
\cong\operatorname{Hom}(\mathbb Z,\Gamma(X,\OO_X))
\]
directly.  Thus finality is genuinely global, not merely a pointwise repetition of the affine
case.
\end{solution}

% VI21-EXERCISE-SOLUTION: E26
\paragraph{Solution to VI.21.E26.}
\begin{solution}
The data attached to $x\in X$ fit into the diagram
\[
\OO_{X,x}\twoheadrightarrow
\kappa(x)=\OO_{X,x}/\mathfrak m_x,
\qquad
\ker(\OO_{X,x}\to\kappa(x))=\mathfrak m_x,
\]
and contravariance turns the quotient/local evaluation map into the canonical point morphism
\[
\Spec\kappa(x)\xrightarrow{i_x}X,
\qquad *\longmapsto x.
\]
Thus the topological point $x$, its local ring, maximal ideal, residue field, and field-valued point
morphism are different faces of the same local scheme-theoretic datum.
\end{solution}
