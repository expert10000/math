\subsection*{VI.21.P12: The generic point morphism}
\begin{problem}\label{prob:vi21-p12}
Let $A$ be a domain.  Describe the canonical morphism attached to the generic point $(0)\in\Spec A$.
\par\textbf{Solution.} Its residue field is $\operatorname{Frac}(A)$, so the point morphism is induced
by the inclusion $A\hookrightarrow\operatorname{Frac}(A)$.  Its image is $(0)$.
\end{problem}
