\subsection*{VI.21.P5: Residue fields under a scheme morphism}
\begin{problem}\label{prob:vi21-p05}
Let $f:X\to Y$ and $x\in X$.  Construct canonically
$\kappa(f(x))\to\kappa(x)$.
\par\textbf{Solution.} The local homomorphism $f_x^\#:\OO_{Y,f(x)}\to\OO_{X,x}$ sends the maximal
ideal of the source into the maximal ideal of the target and therefore descends to the quotient fields.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
