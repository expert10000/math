\subsection*{VI.21.P19: Naturality of the structure morphism}
\begin{problem}\label{prob:vi21-p19}
For every $f:X\to Y$, prove the square with $\pi_X,\pi_Y$ and $\Spec\mathbb Z$ commutes.
\par\textbf{Solution.} Both $\pi_X$ and $\pi_Y\circ f$ are morphisms from $X$ to the final object
$\Spec\mathbb Z$.  By uniqueness they coincide.
\end{problem}
