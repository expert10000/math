\subsection*{VI.21.P4: Classify morphisms from a field spectrum}
\begin{problem}\label{prob:vi21-p04}
Prove that morphisms $\Spec K\to X$ are in bijection with pairs $(x,\kappa(x)\to K)$.
\par\textbf{Solution.} A morphism has one image point $x$.  Its stalk map $\OO_{X,x}\to K$ is local,
so its kernel contains $\mathfrak m_x$ and factors uniquely through $\kappa(x)$.  Conversely a field
map $\kappa(x)\to K$ induces $\Spec K\to\Spec\kappa(x)\to X$.
\end{problem}
