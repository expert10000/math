\subsection*{VI.21.P8: Morphisms preserve specialization}
\begin{problem}\label{prob:vi21-p08}
If $x\in\overline{\{y\}}$ and $f:X\to Y$, prove
$f(x)\in\overline{\{f(y)\}}$.
\par\textbf{Solution.} This is immediate from continuity:
$f(\overline{\{y\}})\subseteq\overline{f(\{y\})}$.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
