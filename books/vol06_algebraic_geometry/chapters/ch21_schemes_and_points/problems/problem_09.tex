\subsection*{VI.21.P9: Residue characteristic and the absolute base}
\begin{problem}\label{prob:vi21-p09}
For $x\in X$, prove
\[
\pi_X(x)=\ker(\mathbb Z\to\kappa(x)),
\]
and deduce the characteristic-zero/characteristic-$p$ alternatives.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
The absolute structure morphism is a geometric portrait of residue characteristic across the whole scheme.

\paragraph{Useful ideas before starting.}
Translate point statements through local rings and residue fields; when the target is affine, use the VI/19 universal property and then verify the pointwise interpretation.

\paragraph{Guided hints.}
\begin{enumerate}
\item Identify the relevant canonical ring or stalk homomorphism.
\item Track maximal ideals or prime contractions to identify the image point.
\item Use locality to pass to residue fields when needed.
\item Check independence, uniqueness, or functoriality explicitly rather than only on points.
\end{enumerate}

\begin{solution}
Choose an affine neighbourhood $U=\Spec A$ of $x$, with $x$ corresponding to
$\mathfrak p$.  By the affine formula, $\pi_X(x)=\mathfrak p\cap\mathbb Z$.  The canonical map
$A\to\kappa(\mathfrak p)$ has kernel containing $\mathfrak p$ and, after restricting to
$\mathbb Z$, has kernel exactly $\mathfrak p\cap\mathbb Z$.  Thus
\[
\pi_X(x)=\ker(\mathbb Z\to\kappa(x)).
\]
A unital map from $\mathbb Z$ to a field has kernel either $(0)$, when the field has characteristic
zero, or $(p)$ for a unique prime $p$, when its characteristic is $p$.  Therefore
\[
\pi_X(x)=
\begin{cases}
(0),&\operatorname{char}\kappa(x)=0,\\
(p),&\operatorname{char}\kappa(x)=p>0.
\end{cases}
\]
\end{solution}

\paragraph{Extensions.}
Analyze the image for $\Spec\mathbb Z$, $\Spec\mathbb Z[t]$, and characteristic-$p$ schemes.
