\subsection*{VI.21.P9: Residue characteristic and the absolute base}
\begin{problem}\label{prob:vi21-p09}
Show that $\pi_X(x)$ equals $(0)$ in characteristic zero and $(p)$ in characteristic $p$.
\par\textbf{Solution.} The image point is the kernel of $\mathbb Z\to\kappa(x)$.  Kernels of maps
from $\mathbb Z$ to fields are $(0)$ or $(p)$, and they record the field characteristic.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
