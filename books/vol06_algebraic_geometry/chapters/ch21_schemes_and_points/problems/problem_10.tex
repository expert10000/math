\subsection*{VI.21.P10: $k$-rational points of an affine $k$-scheme}
\begin{problem}\label{prob:vi21-p10}
Let $X=\Spec A$ for a $k$-algebra $A$.  Prove
$X(k)=\Hom_k(\Spec k,X)\cong\Hom_{k\text{-alg}}(A,k)$.
\par\textbf{Solution.} Apply the affine anti-equivalence in the slice over $\Spec k$: commutation with
the structure maps is exactly the condition that the ring homomorphism restricts to the identity on $k$.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
