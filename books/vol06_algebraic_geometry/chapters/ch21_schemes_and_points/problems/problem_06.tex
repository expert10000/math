\subsection*{VI.21.P6: Composition on residue fields}
\begin{problem}\label{prob:vi21-p06}
For $X\xrightarrow fY\xrightarrow gZ$, prove that the residue-field map for $g\circ f$ is the composite of the two residue-field maps.
\par\textbf{Solution.} Stalk morphisms compose by functoriality of morphisms of locally ringed spaces.
Passing to quotients by maximal ideals preserves the composite.
\end{problem}
