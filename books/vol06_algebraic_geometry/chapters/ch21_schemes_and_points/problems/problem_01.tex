\subsection*{VI.21.P1: AG3 Problem 13: $\Spec\mathbb Z$ is final}
\begin{problem}\label{prob:vi21-p01}
Let $X$ be a scheme.  Prove that there exists a unique morphism $X\to\Spec\mathbb Z$, and identify it as the structure morphism.
\par\textbf{Solution.} By VI/19,
\[
\Hom_{\Sch}(X,\Spec\mathbb Z)\cong\Hom_{\Rings}(\mathbb Z,\Gamma(X,\OO_X)).
\]
Every unital ring admits exactly one homomorphism from $\mathbb Z$, namely $n\mapsto n1$.
Thus the left-hand Hom-set is a singleton.  This proves existence and uniqueness for every scheme,
not merely every affine scheme.  Naturality is automatic: if $f:X\to Y$, both $\pi_Y\circ f$ and
$\pi_X$ are morphisms $X\to\Spec\mathbb Z$, hence are equal.
\end{problem}
