\subsection*{VI.21.P13: The local scheme at a point}
\begin{problem}\label{prob:vi21-p13}
Construct the canonical morphism $\Spec\OO_{X,x}\to X$ and show that its closed point maps to $x$.
\par\textbf{Solution.} Choose $x\in U=\Spec A$ with prime $\mathfrak p$.  Then
$\OO_{X,x}\cong A_{\mathfrak p}$ and localization induces
$\Spec A_{\mathfrak p}\to\Spec A\to X$.  The maximal ideal $\mathfrak pA_{\mathfrak p}$ contracts
to $\mathfrak p$.  Independence of the affine neighbourhood follows from localization compatibility.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
