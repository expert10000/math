\subsection*{VI.21.P20: Recovering the image point from a field map}
\begin{problem}\label{prob:vi21-p20}
Let $U=\Spec A\subseteq X$ and let $u:\Spec K\to X$ have image in $U$.  Show that its image point corresponds to the kernel of the induced map $A\to K$.
\par\textbf{Solution.} Restrict $u$ to $U$ and use the affine morphism dictionary.  Spectrum maps
send the unique prime $(0)\subset K$ to its contraction, which is precisely $\ker(A\to K)$.
\end{problem}
