\subsection*{VI.21.P3: The canonical morphism of a point}
\begin{problem}\label{prob:vi21-p03}
For $x\in X$, construct canonically $i_x:\Spec\kappa(x)\to X$ with image $x$.
\par\textbf{Solution.} On an affine neighbourhood $U=\Spec A$ with $x\leftrightarrow\mathfrak p$,
use $A\to A_{\mathfrak p}\to\kappa(\mathfrak p)$.  The induced map has image $\mathfrak p$.
On smaller affine neighbourhoods these maps agree because they arise from the same local map
$\OO_{X,x}\to\kappa(x)$; hence the construction is independent of $U$.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
