\subsection*{VI.21.P2: The affine formula for the structure morphism}
\begin{problem}\label{prob:vi21-p02}
Let $X=\Spec A$.  Show that $\pi_X$ is induced by $\mathbb Z\to A$ and that
$\pi_X(\mathfrak p)=\mathfrak p\cap\mathbb Z$.
\par\textbf{Solution.} The affine anti-equivalence sends the unique ring map $\mathbb Z\to A$ to
$\Spec A\to\Spec\mathbb Z$.  On primes, spectrum maps act by contraction, giving the formula.
\end{problem}
