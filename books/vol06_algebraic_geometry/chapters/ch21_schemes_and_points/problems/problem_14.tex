\subsection*{VI.21.P14: Generalizations visible in the local scheme}
\begin{problem}\label{prob:vi21-p14}
In the affine situation $x=\mathfrak p\in\Spec A$, identify the image of $\Spec A_{\mathfrak p}\to\Spec A$.
\par\textbf{Solution.} Primes of $A_{\mathfrak p}$ correspond to primes $\mathfrak q\subseteq\mathfrak p$.
These are precisely the generalizations of $\mathfrak p$.
\end{problem}

\paragraph{Provenance.} Canonical VI/21 derivative/application; not an additional legacy migration.
