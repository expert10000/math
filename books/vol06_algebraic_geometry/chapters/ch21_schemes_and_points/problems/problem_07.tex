\subsection*{VI.21.P7: Specialization in an affine spectrum}
\begin{problem}\label{prob:vi21-p07}
Show that for primes $\mathfrak p,\mathfrak q\subset A$,
$\mathfrak q\in\overline{\{\mathfrak p\}}$ iff $\mathfrak p\subseteq\mathfrak q$.
\par\textbf{Solution.} The closure of $\{\mathfrak p\}$ is $V(\mathfrak p)$, consisting exactly of
primes containing $\mathfrak p$.
\end{problem}
