% VI32-DETAILED-CHALLENGE-01
\subsection*{Challenge VI.32.C1 solution}
\begin{solution}
For a plane curve
\[
C=V(f)\subseteq\mathbb A^2_k,
\]
the hypersurface Jacobian criterion says that, at a \(k\)-rational point and in the usual characteristic range,
a singular point must satisfy
\[
f=f_x=f_y=0.
\]
Thus the singular locus is obtained by solving these three equations simultaneously.  We work out three examples.

\textbf{The cusp.}
Let
\[
f=y^2-x^3
\]
and assume \(\operatorname{char}k\ne2,3\). Then
\[
f_x=-3x^2,
\qquad
f_y=2y.
\]
The equations \(f_x=f_y=0\) give
\[
x=0,\qquad y=0.
\]
The point \((0,0)\) lies on \(C\), so
\[
\boxed{\operatorname{Sing}(C)=\{(0,0)\}}.
\]

\textbf{The node.}
Let
\[
g=y^2-x^2(x+1)
=
y^2-x^3-x^2
\]
and assume \(\operatorname{char}k\ne2\), with the familiar node picture also avoiding exceptional derivative
behavior. We have
\[
g_x=-3x^2-2x=-x(3x+2),
\qquad
g_y=2y.
\]
Thus a singular point has \(y=0\) and
\[
x(3x+2)=0.
\]
Together with \(g=0\), if \(y=0\) then
\[
x^2(x+1)=0.
\]
The common solution in the usual characteristic range is \(x=0\). Hence
\[
\boxed{\operatorname{Sing}(N)=\{(0,0)\}}.
\]
At that point the degree-two initial form
\[
y^2-x^2=(y-x)(y+x)
\]
shows the two tangent-cone branches.

\textbf{A regular parabola.}
Take
\[
h=y-x^2.
\]
Then
\[
h_x=-2x,
\qquad
h_y=1.
\]
Since \(h_y\) never vanishes,
\[
h=h_x=h_y=0
\]
has no solution. Therefore
\[
\boxed{\operatorname{Sing}(V(y-x^2))=\varnothing}.
\]

This computation also shows why characteristic hypotheses matter.  If a derivative coefficient vanishes in the
ground field, the Jacobian equations can change.  One must recompute in that characteristic rather than import the
characteristic-zero answer unchanged.
\end{solution}

% VI32-DETAILED-CHALLENGE-02
\subsection*{Challenge VI.32.C2 solution}
\begin{solution}
Let
\[
X=\Spec A
\]
and let \(x\in X(k)\) be a \(k\)-rational point.  The point corresponds to a \(k\)-algebra homomorphism
\[
\operatorname{ev}_x:A\to k.
\]
A lift of \(x\) to the dual numbers is a \(k\)-algebra map
\[
\widetilde x:A\to k[\varepsilon]/(\varepsilon^2)
\]
whose reduction modulo \(\varepsilon\) is \(\operatorname{ev}_x\).

Every such map has a unique expression
\[
\widetilde x(a)
=
\operatorname{ev}_x(a)+\varepsilon D(a)
\]
for a \(k\)-linear function \(D:A\to k\).  The map is multiplicative exactly when
\[
\widetilde x(ab)=\widetilde x(a)\widetilde x(b).
\]
Expanding both sides gives
\[
\operatorname{ev}_x(ab)+\varepsilon D(ab)
=
\operatorname{ev}_x(a)\operatorname{ev}_x(b)
+
\varepsilon\bigl(
\operatorname{ev}_x(a)D(b)+\operatorname{ev}_x(b)D(a)
\bigr).
\]
Thus multiplicativity is equivalent to the Leibniz rule
\[
\boxed{
D(ab)
=
a(x)D(b)+b(x)D(a).
}
\]
Also \(D(c)=0\) for \(c\in k\).  Hence the lifts are exactly the \(k\)-derivations
\[
\operatorname{Der}_k(A,k_x),
\]
where \(k_x=k\) is regarded as an \(A\)-module through evaluation at \(x\).

Now let
\[
\mathfrak m=\ker(\operatorname{ev}_x).
\]
For \(a,b\in\mathfrak m\),
\[
D(ab)=a(x)D(b)+b(x)D(a)=0.
\]
Therefore \(D\) kills \(\mathfrak m^2\), and its restriction to \(\mathfrak m\) descends to a linear functional
\[
\mathfrak m/\mathfrak m^2\to k.
\]
Conversely, a \(k\)-linear functional on \(\mathfrak m/\mathfrak m^2\) uniquely reconstructs a derivation by writing
\[
A=k\oplus\mathfrak m
\]
as vector spaces through the rational point and applying the functional to the \(\mathfrak m\)-part.

Thus
\[
\operatorname{Der}_k(A,k)
\cong
\operatorname{Hom}_k(\mathfrak m/\mathfrak m^2,k)
=
T_xX.
\]
Passing from ring maps to morphisms reverses arrows, so we obtain the intrinsic identification
\[
\boxed{
T_xX
\cong
\left\{
\Spec k[\varepsilon]/(\varepsilon^2)\to X
\text{ reducing to }x
\right\}.
}
\]
This is why the Zariski tangent space records first-order infinitesimal motions.
\end{solution}

% VI32-DETAILED-CHALLENGE-03
\subsection*{Challenge VI.32.C3 solution}
\begin{solution}
Let \(A\) be Noetherian and let
\[
\mathfrak q\subseteq\mathfrak p
\]
be prime ideals.  Suppose \(A_{\mathfrak p}\) is regular.  Inside the local ring \(A_{\mathfrak p}\), the ideal
\[
\mathfrak qA_{\mathfrak p}
\]
is prime, and localizing once more gives
\[
(A_{\mathfrak p})_{\mathfrak qA_{\mathfrak p}}
\cong
A_{\mathfrak q}.
\]
A standard theorem of regular local algebra says that every localization of a regular local ring at a prime is
again regular. Hence
\[
\boxed{
A_{\mathfrak p}\text{ regular and }\mathfrak q\subseteq\mathfrak p
\Longrightarrow
A_{\mathfrak q}\text{ regular}.
}
\]

Geometrically, if \(x\in\Spec A\) is regular, then every generalization of \(x\) is regular. Thus
\[
\operatorname{Reg}(\Spec A)
\]
is stable under generalization.

One must distinguish this from openness.  For an arbitrary Noetherian ring, stability under generalization does
\emph{not} by itself prove that the regular locus is open, and the regular locus of an arbitrary Noetherian scheme
need not be open without further hypotheses.  Openness holds in the standard algebraic-geometric settings used
later, for example for schemes of finite type over a field (more generally under suitable excellence/J-2
hypotheses).

Therefore the safe conclusions at the VI/32 level are:
\[
\boxed{\text{regularity localizes}}
\]
and
\[
\boxed{\text{the regular locus is generalization-stable}},
\]
while openness is recorded as a stronger theorem requiring additional structural hypotheses.
\end{solution}

% VI32-DETAILED-CHALLENGE-04
\subsection*{Challenge VI.32.C4 solution}
\begin{solution}
Consider the quadric cone
\[
X=V(xy-z^2)\subseteq\mathbb A^3_k
\]
with coordinate ring
\[
A=k[x,y,z]/(xy-z^2).
\]
It is a two-dimensional domain: the polynomial \(xy-z^2\) is irreducible, so the hypersurface has dimension \(2\).

At the origin, the gradient is
\[
(y,x,-2z),
\]
which vanishes at \((0,0,0)\) in the usual characteristic range; even in characteristic \(2\), the first two
entries vanish and the equation has no linear term.  Thus the cotangent space at the vertex has basis represented
by \(x,y,z\), and
\[
\dim_kT_0X=3.
\]
But
\[
\dim\mathcal O_{X,0}=2.
\]
Therefore
\[
\boxed{0\text{ is singular on }X}.
\]

Now explain normality.  There is an injective parametrization
\[
A
\cong
k[s^2,st,t^2]
\subseteq
k[s,t],
\]
through
\[
x=s^2,\qquad z=st,\qquad y=t^2.
\]
The exponent semigroup is generated by
\[
(2,0),\quad(1,1),\quad(0,2).
\]
Its group is
\[
G=\{(a,b)\in\mathbb Z^2:a+b\text{ even}\},
\]
and the semigroup itself is
\[
S=G\cap\mathbb N^2.
\]
Hence \(S\) is saturated in \(G\): if \(u\in G\) and \(nu\in S\) for some \(n>0\), then the coordinates of \(u\)
are already nonnegative, so \(u\in S\).  By the standard theorem that the semigroup algebra of a saturated affine
semigroup is normal,
\[
k[S]=k[s^2,st,t^2]
\]
is integrally closed. Thus
\[
\boxed{X\text{ is normal}.}
\]

We have therefore produced the basic warning
\[
\boxed{\text{normal}\not\Longrightarrow\text{regular}}
\]
in dimension \(2\).  The vertex is normal but singular.  This does not contradict the one-dimensional situation,
where a Noetherian normal local domain is regular; dimension two is the first place where the notions separate in
this way.
\end{solution}

% VI32-DETAILED-CHALLENGE-05
\subsection*{Challenge VI.32.C5 solution}
\begin{solution}
The five notions should be separated carefully.

\begin{itemize}
\item \textbf{Regular:}
\(\mathcal O_{X,x}\) is a regular local ring:
\[
\dim\mathcal O_{X,x}
=
\dim_{\kappa(x)}\mathfrak m_x/\mathfrak m_x^2.
\]
\item \textbf{Jacobian-nonsingular:}
for a suitable affine finite-type presentation, the Jacobian has the rank predicted by the local codimension.
At a rational hypersurface point this means that the gradient is nonzero.
\item \textbf{Smooth over \(k\):}
the structure morphism \(X\to\Spec k\) is smooth at \(x\).  This is a relative notion.
\item \textbf{Geometrically regular:}
regularity persists after the relevant field extensions of the base or residue field.
\item \textbf{Normal:}
the local ring is a normal domain, hence integrally closed in its fraction field in the integral case.
\end{itemize}

The robust implications are as follows.

For a scheme locally of finite type over a field,
\[
\boxed{\text{smooth over }k\Longrightarrow\text{regular}.}
\]
More precisely, smoothness is characterized by geometric regularity of the relevant fibers in this finite-type
setting.  Thus smoothness is stronger than ordinary regularity over an imperfect field.

If \(k\) is perfect and \(X\) is locally of finite type over \(k\), regularity is already geometrically regular,
so
\[
\boxed{
k\text{ perfect, finite type}
\quad\Longrightarrow\quad
\text{regular}\iff\text{smooth over }k
}
\]
for the standard variety/scheme setting.

At rational points of affine schemes where the Jacobian criterion applies, the expected-rank Jacobian condition
is another concrete presentation of the same regular/smooth behavior.  For a hypersurface,
\[
\nabla f(a)\ne0
\]
is exactly the regularity test at the rational point \(a\).

Regular local rings are normal domains, so
\[
\boxed{\text{regular}\Longrightarrow\text{normal}}
\]
in the Noetherian local setting.  The converse is false: the quadric cone
\[
xy-z^2=0
\]
is normal but singular at its vertex.

Over an imperfect field, regular need not imply smooth.  A standard source is a finite purely inseparable field
extension \(L/k\): the scheme \(\Spec L\) is a regular zero-dimensional scheme, but the morphism
\[
\Spec L\to\Spec k
\]
is not smooth when \(L/k\) is nontrivial purely inseparable, because smooth zero-dimensional field extensions are
separable.

Thus the practical hierarchy is
\[
\text{smooth}
\Longrightarrow
\text{geometrically regular}
\Longrightarrow
\text{regular}
\Longrightarrow
\text{normal},
\]
with the first equivalence interpreted under the finite-type hypotheses appropriate to smooth morphisms, and with
regular \(\Rightarrow\) smooth recovered over perfect fields.  The systematic theory of smoothness, geometric
regularity, and differentials is deliberately deferred beyond VI/32.
\end{solution}
