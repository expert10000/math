% VI32-DETAILED-PROBLEM-06
\begin{problem}[VI.32.P6: Parabola is regular]\label{prob:vi32-06}
At the origin of
\[
V(y-x^2)\subset\mathbb A^2_k,
\]
compute local dimension and tangent dimension and prove regularity.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
A=k[x,y]/(y-x^2).
\]
The substitution \(y=x^2\) gives
\[
A\cong k[x],
\]
so the curve is isomorphic to \(\mathbb A^1_k\). At the origin, the corresponding local ring is
\[
k[x]_{(x)},
\]
hence
\[
\dim\mathcal O_{X,0}=1.
\]

The defining equation is
\[
f=y-x^2,
\]
with gradient
\[
\nabla f=(-2x,1).
\]
At the origin,
\[
\nabla f(0,0)=(0,1),
\]
which has rank \(1\) in every characteristic. Therefore
\[
T_0X
=
\ker\bigl((0,1):k^2\to k\bigr)
=
\{(a,0):a\in k\},
\]
and
\[
\dim_kT_0X=1.
\]
Thus
\[
\dim\mathcal O_{X,0}
=
\dim_kT_0X
=
1.
\]
Hence the local ring is regular:
\[
\boxed{0\in V(y-x^2)\text{ is a regular point}.}
\]
\end{solution}
\fi
