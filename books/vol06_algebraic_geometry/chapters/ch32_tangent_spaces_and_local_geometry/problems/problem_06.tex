\begin{problem}[VI.32.P6: Parabola is regular]\label{prob:vi32-06}
At the origin of $V(y-x^2)\subset\mathbb A^2_k$, compute local dimension and tangent dimension and prove regularity.

\textbf{Provenance.} Canonical VI/32 synthesis.
\begin{solution}
The hypersurface has local dimension $1$. The Jacobian row is $(-2x,1)$, which at the origin has rank $1$ in every characteristic because the second entry is $1$. Thus the tangent dimension is $2-1=1$, equal to local dimension.
\end{solution}
\end{problem}
