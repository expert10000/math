\begin{problem}[VI.32.P12: Jacobian tangent formula]\label{prob:vi32-12}
Derive the tangent equations $J(a)v=0$ for $X=V(f_1,\ldots,f_r)\subset\mathbb A^n$ at a rational point $a$.

\textbf{Provenance.} Canonical VI/32 synthesis.
\begin{solution}
Substitute $a+\varepsilon v$ into each polynomial modulo $\varepsilon^2$. Taylor expansion to first order gives $f_i(a+\varepsilon v)=f_i(a)+\varepsilon\sum_j(\partial f_i/\partial x_j)(a)v_j$. Since $f_i(a)=0$, the dual-number point lies on $X$ exactly when every linear coefficient vanishes.
\end{solution}
\end{problem}
