% VI32-DETAILED-PROBLEM-12
\begin{problem}[VI.32.P12: Jacobian tangent formula]\label{prob:vi32-12}
Derive the tangent equations
\[
J(a)v=0
\]
for
\[
X=V(f_1,\ldots,f_r)\subset\mathbb A^n
\]
at a rational point \(a\).

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
A tangent vector at a \(k\)-rational point \(a=(a_1,\ldots,a_n)\) can be represented by a dual-number point
\[
a+\varepsilon v
=
(a_1+\varepsilon v_1,\ldots,a_n+\varepsilon v_n),
\qquad
\varepsilon^2=0.
\]
Such a vector lies tangent to \(X\) exactly when every defining equation still vanishes modulo \(\varepsilon^2\).

For each polynomial \(f_i\), expand to first order:
\[
f_i(a+\varepsilon v)
=
f_i(a)
+
\varepsilon
\sum_{j=1}^n
\frac{\partial f_i}{\partial x_j}(a)v_j.
\]
All terms containing at least two factors of \(\varepsilon\) vanish. Since \(a\in X(k)\),
\[
f_i(a)=0.
\]
Therefore the lifting condition is
\[
\sum_{j=1}^n
\frac{\partial f_i}{\partial x_j}(a)v_j
=
0
\qquad
(1\le i\le r).
\]
Collecting these equations gives
\[
\boxed{J(a)v=0}.
\]
Hence
\[
T_aX\cong\ker(J(a):k^n\to k^r),
\]
and rank-nullity gives
\[
\boxed{\dim_kT_aX=n-\operatorname{rank}J(a)}.
\]
\end{solution}
\fi
