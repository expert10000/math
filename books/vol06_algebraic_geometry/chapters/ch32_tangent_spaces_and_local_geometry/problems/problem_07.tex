% VI32-DETAILED-PROBLEM-07
\begin{problem}[VI.32.P7: Cusp singularity]\label{prob:vi32-07}
Assume \(\operatorname{char}k\ne2,3\). Prove the origin of
\[
y^2=x^3
\]
is singular.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Set
\[
A=k[x,y]/(y^2-x^3).
\]
This is a one-dimensional domain: for example VI/27 identified it with
\[
k[t^2,t^3]\subset k[t].
\]
Hence at the cusp point
\[
\mathfrak m=(x,y),
\]
the local ring has dimension \(1\).

Now
\[
f=y^2-x^3
\]
has gradient
\[
\nabla f=(-3x^2,2y).
\]
At the origin,
\[
\nabla f(0,0)=(0,0).
\]
Therefore the Jacobian has rank \(0\), so
\[
T_0C=k^2
\]
and
\[
\dim_kT_0C=2.
\]
Since
\[
\dim\mathcal O_{C,0}=1<2=\dim_kT_0C,
\]
the regular-local equality fails. Thus
\[
\boxed{0\text{ is singular on the cusp}.}
\]

Equivalently, in the local ring the classes of \(x\) and \(y\) are independent modulo \(\mathfrak m^2\), because
the relation \(y^2-x^3\) has no linear part. Hence the embedding dimension is \(2\).
\end{solution}
\fi
