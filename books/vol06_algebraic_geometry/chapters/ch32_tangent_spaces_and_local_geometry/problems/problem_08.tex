% VI32-DETAILED-PROBLEM-08
\begin{problem}[VI.32.P8: Node singularity]\label{prob:vi32-08}
Assume \(\operatorname{char}k\ne2\). Prove the origin of
\[
y^2=x^2(x+1)
\]
is singular and identify the two tangent-cone directions.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
f=y^2-x^2(x+1)=y^2-x^3-x^2.
\]
The curve is one-dimensional, so the local dimension at the origin is \(1\).

Its first derivatives are
\[
f_x=-3x^2-2x,
\qquad
f_y=2y.
\]
Both vanish at \((0,0)\), hence
\[
\dim_kT_0N=2.
\]
Since \(2>1\), the origin is singular.

To distinguish the node from the cusp, keep the lowest nonzero homogeneous part of \(f\):
\[
f_2=y^2-x^2.
\]
When \(\operatorname{char}k\ne2\),
\[
f_2=(y-x)(y+x).
\]
Thus the tangent cone consists of the two distinct lines
\[
\boxed{y=x\quad\text{and}\quad y=-x}.
\]
The Zariski tangent space itself is all of \(k^2\), so tangent-space dimension detects singularity but does not by
itself distinguish the node from the cusp; the first nonzero homogeneous term supplies the finer information.
\end{solution}
\fi
