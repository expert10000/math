% VI32-DETAILED-PROBLEM-19
\begin{problem}[VI.32.P19: Tangent dimensions in a product]\label{prob:vi32-19}
Compute the tangent space at the origin of
\[
\mathbb A^m_k\times\mathbb A^n_k
\]
and relate its dimension to the factors.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The product is canonically
\[
\mathbb A^{m+n}_k
=
\Spec k[x_1,\ldots,x_m,y_1,\ldots,y_n].
\]
At the origin, the maximal ideal is
\[
\mathfrak m=(x_1,\ldots,x_m,y_1,\ldots,y_n).
\]
Modulo \(\mathfrak m^2\), the coordinate classes form a basis. Thus
\[
\mathfrak m/\mathfrak m^2
\cong
k^m\oplus k^n.
\]
More intrinsically, this is the direct sum of the cotangent spaces of the two factors at their origins.

Dualizing,
\[
T_{(0,0)}
(\mathbb A^m\times\mathbb A^n)
\cong
T_0\mathbb A^m\oplus T_0\mathbb A^n
\cong
k^m\oplus k^n.
\]
Therefore
\[
\boxed{
\dim_kT_{(0,0)}
=
m+n
=
\dim_kT_0\mathbb A^m+\dim_kT_0\mathbb A^n.
}
\]
This is the simplest manifestation of the product behavior of tangent spaces.
\end{solution}
\fi
