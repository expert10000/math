\begin{problem}[VI.32.P14: Non-rational point over \(\mathbb R\)]\label{prob:vi32-14}
For
\[
x=(t^2+1)\in\Spec\mathbb R[t],
\]
compute \(\kappa(x)\), local dimension, and embedding dimension.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The polynomial \(t^2+1\) is irreducible over \(\mathbb R\), so
\[
\mathfrak m=(t^2+1)
\]
is maximal. The residue field is
\[
\kappa(x)
=
\mathbb R[t]/(t^2+1)
\cong
\mathbb C.
\]

The ring \(\mathbb R[t]\) is a PID of Krull dimension \(1\). Since \(\mathfrak m\) is a nonzero prime,
\[
\operatorname{ht}(\mathfrak m)=1.
\]
Therefore
\[
\dim\mathcal O_{X,x}
=
\dim\mathbb R[t]_{\mathfrak m}
=
1.
\]

The localized ring is a one-dimensional local PID, and its maximal ideal is generated by the class of \(t^2+1\).
Hence
\[
\mathfrak m_x/\mathfrak m_x^2
\]
is one-dimensional over the residue field \(\kappa(x)\cong\mathbb C\). Thus
\[
\boxed{
\kappa(x)\cong\mathbb C,\qquad
\dim\mathcal O_{X,x}=1,\qquad
\operatorname{edim}_xX=1.
}
\]
The point is regular, even though it is not \(\mathbb R\)-rational.
\end{solution}
\fi
