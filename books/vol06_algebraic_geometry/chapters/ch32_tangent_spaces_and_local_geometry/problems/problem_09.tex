\begin{problem}[VI.32.P9: Crossing axes]\label{prob:vi32-09}
Analyze regularity at the origin of $V(xy)\subset\mathbb A^2_k$.

\textbf{Provenance.} Canonical VI/32 synthesis.
\begin{solution}
The local dimension is $1$. The Jacobian row $(y,x)$ vanishes at the origin, so tangent dimension is $2$. Hence the origin is singular.
\end{solution}
\end{problem}
