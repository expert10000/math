\begin{problem}[VI.32.P2: AG8 Problem 3(a): failure for a general scheme]\label{prob:vi32-02}
Let $X=\Spec k\sqcup\mathbb A^1_k$ and let $x$ be the point of the first component. Show that $x$ is closed but $\dim\OO_{X,x}\ne\dim X$.

\textbf{Provenance.} Legacy AG8 Problem 3(a) counterexample requirement; canonical destination VI/32.
\begin{solution}
The point component is open and closed, so $x$ is closed in $X$. Its local ring is $k$, of dimension $0$. The other component has dimension $1$, so $\dim X=1$. Hence the equality fails.
\end{solution}
\end{problem}
