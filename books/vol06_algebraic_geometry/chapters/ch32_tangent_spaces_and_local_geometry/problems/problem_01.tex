\begin{problem}[VI.32.P1: AG8 local dimension at closed points]\label{prob:vi32-01}
Let $X$ be a variety over $k$ and let $x\in X$ be closed. Prove $\dim\OO_{X,x}=\dim X$.

\textbf{Provenance.} Legacy AG8 Problem 2 local-dimension slice; canonical destination VI/32.
\begin{solution}
Choose an affine neighborhood $\Spec A$ with maximal ideal $\mathfrak m$ for $x$. The finite-type domain dimension formula gives $\operatorname{ht}(\mathfrak m)+\operatorname{trdeg}_k\kappa(x)=\dim A$. Zariski's lemma makes $\kappa(x)/k$ algebraic, so the second term is zero. Thus $\dim\OO_{X,x}=\dim A_{\mathfrak m}=\operatorname{ht}(\mathfrak m)=\dim A=\dim X$.
\end{solution}
\end{problem}
