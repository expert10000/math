\begin{problem}[VI.32.P4: Cotangent space of affine space]\label{prob:vi32-04}
Compute $\mathfrak m/\mathfrak m^2$ at the origin of $\mathbb A^n_k$.

\textbf{Provenance.} Canonical VI/32 synthesis.
\begin{solution}
Here $\mathfrak m=(x_1,\ldots,x_n)$ in the local ring. Modulo $\mathfrak m^2$, the classes of $x_1,\ldots,x_n$ span and are linearly independent. Hence the cotangent space has dimension $n$.
\end{solution}
\end{problem}
