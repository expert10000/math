% VI32-DETAILED-PROBLEM-11
\begin{problem}[VI.32.P11: Nilpotent thickened line]\label{prob:vi32-11}
At the origin of
\[
\Spec k[x,\varepsilon]/(\varepsilon^2),
\]
compute local and embedding dimensions.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
A=k[x,\varepsilon]/(\varepsilon^2)
\]
and let
\[
\mathfrak m=(x,\varepsilon).
\]
The nilradical is \((\varepsilon)\), and
\[
A_{\mathrm{red}}\cong k[x].
\]
Reduction preserves prime chains, so
\[
\dim A_{\mathfrak m}
=
\dim k[x]_{(x)}
=
1.
\]
Thus the local dimension is \(1\).

For the embedding dimension, note that
\[
\mathfrak m^2=(x^2,x\varepsilon,\varepsilon^2)
=(x^2,x\varepsilon).
\]
The classes of \(x\) and \(\varepsilon\) survive modulo \(\mathfrak m^2\). They are linearly independent over the
residue field \(k\): a relation
\[
ax+b\varepsilon\in\mathfrak m^2
\]
would force the degree-one part to vanish, so \(a=b=0\). Hence
\[
\dim_k\mathfrak m/\mathfrak m^2=2.
\]
Therefore
\[
\boxed{
\dim\mathcal O_{X,0}=1,\qquad
\operatorname{edim}_0X=2.
}
\]
The thickened line is singular at the origin, even though its reduction is the regular affine line.
\end{solution}
\fi
