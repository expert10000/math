\begin{problem}[VI.32.P3: Affine local dimension equals height]\label{prob:vi32-03}
Let $x=\mathfrak p\in\Spec A$. Prove $\dim\OO_{X,x}=\operatorname{ht}(\mathfrak p)$.

\textbf{Provenance.} Canonical VI/32 synthesis.
\begin{solution}
The local ring is $A_{\mathfrak p}$. Its prime ideals correspond inclusion-preservingly to primes of $A$ contained in $\mathfrak p$. Thus chains in $A_{\mathfrak p}$ are exactly chains ending below $\mathfrak p$, and their supremal length is the height.
\end{solution}
\end{problem}
