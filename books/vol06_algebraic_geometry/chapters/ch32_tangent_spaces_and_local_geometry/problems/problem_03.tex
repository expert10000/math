% VI32-DETAILED-PROBLEM-03
\begin{problem}[VI.32.P3: Affine local dimension equals height]\label{prob:vi32-03}
Let \(x=\mathfrak p\in\Spec A\). Prove
\[
\dim\mathcal O_{X,x}=\operatorname{ht}(\mathfrak p).
\]

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The local ring at \(x=\mathfrak p\) is
\[
\mathcal O_{X,x}=A_{\mathfrak p}.
\]
Prime ideals of \(A_{\mathfrak p}\) are in inclusion-preserving bijection with prime ideals
\[
\mathfrak q\subseteq\mathfrak p
\]
of \(A\), through
\[
\mathfrak q\longmapsto\mathfrak qA_{\mathfrak p}.
\]
Thus a prime chain
\[
\mathfrak q_0A_{\mathfrak p}
\subsetneq\cdots\subsetneq
\mathfrak q_nA_{\mathfrak p}
\]
in the local ring corresponds exactly to
\[
\mathfrak q_0\subsetneq\cdots\subsetneq\mathfrak q_n\subseteq\mathfrak p
\]
in \(A\). Every chain can be extended, if necessary, by \(\mathfrak p\) at the top without decreasing its length.
Therefore the supremum of chain lengths in \(A_{\mathfrak p}\) is precisely the supremum of prime-chain lengths
ending at \(\mathfrak p\), which is the height:
\[
\boxed{
\dim\mathcal O_{X,x}
=
\dim A_{\mathfrak p}
=
\operatorname{ht}(\mathfrak p).
}
\]

This identity is the pointwise version of the VI/29 height theorem and is the basic algebraic input for every local
dimension computation in VI/32.
\end{solution}
\fi
