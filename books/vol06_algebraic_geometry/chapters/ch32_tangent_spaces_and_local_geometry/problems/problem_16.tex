% VI32-DETAILED-PROBLEM-16
\begin{problem}[VI.32.P16: Cusp multiple lenses]\label{prob:vi32-16}
For
\[
A=k[t^2,t^3],
\]
compute the local dimension and embedding dimension at \((t^2,t^3)\) and explain the normalization comparison.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
\mathfrak m=(t^2,t^3)\subset A.
\]
The ring \(A\) is a one-dimensional domain with fraction field \(k(t)\). Hence the nonzero maximal ideal
\(\mathfrak m\) has height \(1\), and
\[
\dim A_{\mathfrak m}=1.
\]

Now
\[
\mathfrak m^2=(t^4,t^5,t^6).
\]
The classes of \(t^2\) and \(t^3\) both survive in
\[
\mathfrak m/\mathfrak m^2.
\]
They are linearly independent over \(k\): a relation
\[
a t^2+b t^3\in(t^4,t^5,t^6)
\]
forces \(a=b=0\), since the left side has valuation \(2\) or \(3\) unless it vanishes. Therefore
\[
\dim_k\mathfrak m/\mathfrak m^2=2.
\]
Thus the cusp local ring satisfies
\[
\boxed{\dim A_{\mathfrak m}=1,\qquad \operatorname{edim}=2},
\]
so it is singular.

VI/28 computed the normalization
\[
A\subset k[t].
\]
Above the cusp point, the normalized local ring is
\[
k[t]_{(t)}.
\]
It still has dimension \(1\), but its maximal ideal is generated by \(t\), so its embedding dimension is \(1\).
Thus normalization preserves the one-dimensional prime-chain geometry while replacing this singular local ring by
a regular one.
\end{solution}
\fi
