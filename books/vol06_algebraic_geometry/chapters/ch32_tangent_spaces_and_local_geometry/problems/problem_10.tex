\begin{problem}[VI.32.P10: Dual numbers are singular]\label{prob:vi32-10}
Prove the unique point of $\Spec k[\varepsilon]/(\varepsilon^2)$ is not regular.

\textbf{Provenance.} Canonical VI/32 synthesis.
\begin{solution}
The ring is Artinian, so its Krull dimension is $0$. Its maximal ideal $(\varepsilon)$ has square zero, so $\mathfrak m/\mathfrak m^2\cong k$ has dimension $1$. The dimensions do not agree.
\end{solution}
\end{problem}
