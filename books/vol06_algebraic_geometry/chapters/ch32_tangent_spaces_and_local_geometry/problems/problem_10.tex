% VI32-DETAILED-PROBLEM-10
\begin{problem}[VI.32.P10: Dual numbers are singular]\label{prob:vi32-10}
Prove the unique point of
\[
\Spec k[\varepsilon]/(\varepsilon^2)
\]
is not regular.

\textbf{Provenance.} Canonical VI/32 synthesis.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
R=k[\varepsilon]/(\varepsilon^2).
\]
Its unique prime and maximal ideal is
\[
\mathfrak m=(\varepsilon).
\]
Since every prime is maximal, \(R\) is Artinian and
\[
\dim R=0.
\]

But
\[
\mathfrak m^2=(\varepsilon^2)=0,
\]
so
\[
\mathfrak m/\mathfrak m^2
=
(\varepsilon)
\cong k.
\]
Hence the embedding dimension is
\[
\operatorname{edim}R=1.
\]
Therefore
\[
\dim R=0<1=\operatorname{edim}R.
\]
By the regular-local criterion,
\[
\boxed{R\text{ is not regular}.}
\]

This example is especially useful because the underlying topological space is a single point, exactly as for
\(\Spec k\). The singularity is entirely infinitesimal: nilpotent structure enlarges the cotangent space without
changing prime chains or Krull dimension.
\end{solution}
\fi
