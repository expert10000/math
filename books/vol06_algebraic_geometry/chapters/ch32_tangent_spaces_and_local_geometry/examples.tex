\begin{example}[A field point]\label{ex:vi32-01}
For $X=\Spec k$, the unique local ring is $k$, so local dimension and embedding dimension are both $0$. The point is regular.
\end{example}

\begin{example}[Affine line at a closed point]\label{ex:vi32-02}
At $a\in\mathbb A^1_k(k)$, $\OO_{X,a}=k[t]_{(t-a)}$ has dimension $1$ and maximal ideal generated by one element. Hence the point is regular.
\end{example}

\begin{example}[Affine plane at the origin]\label{ex:vi32-03}
$k[x,y]_{(x,y)}$ has dimension $2$, while $(x,y)/(x,y)^2$ has basis given by the classes of $x,y$. Thus the embedding dimension is $2$.
\end{example}

\begin{example}[Generic point of an integral scheme]\label{ex:vi32-04}
At the generic point $\eta$ of an integral scheme, $\OO_{X,\eta}=K(X)$ is a field. Hence the local-ring dimension is $0$.
\end{example}

\begin{example}[A height-one point in the plane]\label{ex:vi32-05}
For $\mathfrak p=(x)\subset k[x,y]$, the local ring $k[x,y]_{(x)}$ has dimension $1$.
\end{example}

\begin{example}[Parabola at the origin]\label{ex:vi32-06}
For $y-x^2=0$, the Jacobian row at the origin is $(0,1)$, so the tangent space is one-dimensional. The curve has local dimension $1$, hence the origin is regular.
\end{example}

\begin{example}[Cusp at the origin]\label{ex:vi32-07}
For $y^2-x^3=0$, both partial derivatives vanish at $(0,0)$. The tangent space is $k^2$ while the local dimension is $1$, so the cusp point is singular.
\end{example}

\begin{example}[Node at the origin]\label{ex:vi32-08}
For $y^2-x^2(x+1)=0$ in characteristic not $2$, the origin has tangent dimension $2$ and local dimension $1$, hence is singular.
\end{example}

\begin{example}[Crossing axes]\label{ex:vi32-09}
For $xy=0$, the origin has tangent space all of $k^2$ but local dimension $1$. The singularity comes from two components meeting.
\end{example}

\begin{example}[Dual numbers]\label{ex:vi32-10}
$X=\Spec k[\varepsilon]/(\varepsilon^2)$ has one point and local dimension $0$, but $\mathfrak m/\mathfrak m^2$ is one-dimensional. Thus the point is singular despite the one-point topology.
\end{example}

\begin{example}[A smooth hypersurface point]\label{ex:vi32-11}
On $V(x_1)\subset\mathbb A^n_k$, the gradient is $(1,0,\ldots,0)$ everywhere, so the tangent space has dimension $n-1$, equal to the local dimension.
\end{example}

\begin{example}[A singular quadric cone point]\label{ex:vi32-12}
For $xy-z^2=0$ in $\mathbb A^3$, the gradient vanishes at the origin. The local dimension is $2$ while the tangent space has dimension $3$.
\end{example}

\begin{example}[Non-rational closed point]\label{ex:vi32-13}
For $X=\Spec\mathbb R[t]$ and $x=(t^2+1)$, the residue field is $\mathbb C$. The local ring has dimension $1$ and $\mathfrak m/\mathfrak m^2$ is one-dimensional over $\mathbb C$.
\end{example}

\begin{example}[Closed point on a variety]\label{ex:vi32-14}
If $X$ is a $d$-dimensional variety and $x$ is closed, then $\dim\OO_{X,x}=d$, even when $\kappa(x)\ne k$.
\end{example}

\begin{example}[Disconnected failure]\label{ex:vi32-15}
For $X=\Spec k\sqcup\mathbb A^2_k$, the point component has local dimension $0$ while the global scheme has dimension $2$.
\end{example}

\begin{example}[Product affine space]\label{ex:vi32-16}
At the origin of $\mathbb A^m_k\times\mathbb A^n_k\cong\mathbb A^{m+n}_k$, both local and embedding dimensions equal $m+n$.
\end{example}

\begin{example}[Coordinate subspace]\label{ex:vi32-17}
At a closed point of $V(x_1,\ldots,x_r)\subset\mathbb A^n$, the local and embedding dimensions are $n-r$.
\end{example}

\begin{example}[A redundant equation]\label{ex:vi32-18}
$V(x,xy)\subset\mathbb A^2$ equals the line $V(x)$. At a closed point its local dimension and tangent dimension are both $1$ despite the two displayed equations.
\end{example}

\begin{example}[Nilpotent thickened line]\label{ex:vi32-19}
For $A=k[x,\varepsilon]/(\varepsilon^2)$ at $(x,\varepsilon)$, the local dimension is $1$ but the cotangent space has two independent classes. The thickening is singular.
\end{example}

\begin{example}[Regularity after localization]\label{ex:vi32-20}
If a Noetherian ring $A$ has $A_{\mathfrak p}$ regular, then the scheme $\Spec A$ is regular at the point $\mathfrak p$ by definition.
\end{example}
