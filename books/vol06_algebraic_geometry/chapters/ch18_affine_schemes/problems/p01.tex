\begin{problem}[Legacy AG3 Problem 11: The basic open subscheme $D(f)$]\label{prob:vi18-p01}
Let $A$ be a ring, $f\in A$, and $X=\Spec A$.  Prove, as locally ringed spaces,
\[
\boxed{(D(f),\OO_X|_{D(f)})\cong\Spec A_f.}
\]
Your proof must identify the underlying topological spaces, compare the structure sheaves on a basis, verify compatibility of restriction maps, and identify the local rings on stalks.
\end{problem}

\ifdefined\FullProblemDossiers
\paragraph{Why this problem matters.}
This is the geometric form of localization. It upgrades the topological fact that primes of the localization correspond to primes avoiding the inverted element to an isomorphism of affine schemes.

\paragraph{Useful ideas before starting.}
Use extension-contraction of primes under localization. On $\Spec A_f$ use basic opens $D(g/1)$; on $D(f)$ these correspond to $D(g)\cap D(f)=D(fg)$. Compare $(A_f)_{g/1}$ with $A_{fg}$.

\paragraph{Guided hints.}
\begin{enumerate}
\item Define $\Phi:\Spec A_f\to D(f)$ by contraction $\mfq\mapsto\mfq\cap A$.
\item Show the inverse sends $\mfp\in D(f)$ to $\mfp A_f$.
\item Track $D(g/1)$ under $\Phi$.
\item Use $(A_f)_{g/1}\cong A_{fg}$ to compare the sheaves on this basis.
\item At $\mfp\in D(f)$ identify $(A_f)_{\mfp A_f}\cong A_{\mfp}$ and note that the isomorphism is local.
\end{enumerate}

\begin{solution}
Prime ideals of $A_f$ are in bijection with primes of $A$ disjoint from the multiplicative set $\{1,f,f^2,\dots\}$. Hence
\[
\Phi:\Spec A_f\longrightarrow D(f),\qquad \mfq\longmapsto\mfq\cap A
\]
is a bijection with inverse $\mfp\mapsto\mfp A_f$.

If $g\in A$, then every basic open of $\Spec A_f$ may be written $D(g/1)$ because $f$ is a unit in $A_f$. Under $\Phi$,
\[
D(g/1)\longmapsto D(g)\cap D(f)=D(fg).
\]
Thus $\Phi$ is a homeomorphism because it identifies bases of open sets.

Now compare structure sheaves. On the open $D(g/1)\subseteq\Spec A_f$,
\[
\OO_{\Spec A_f}(D(g/1))=(A_f)_{g/1}.
\]
On the corresponding open $D(fg)\subseteq D(f)$,
\[
(\OO_X|_{D(f)})(D(fg))=\OO_X(D(fg))=A_{fg}.
\]
Localization is associative, and the canonical map
\[
(A_f)_{g/1}\longrightarrow A_{fg},\qquad
\frac{a/f^m}{(g/1)^n}\longmapsto\frac{a}{f^mg^n}
\]
is an isomorphism. These isomorphisms commute with further localization, hence with restriction maps on the chosen basis. Therefore they determine an isomorphism of sheaves
\[
\OO_{\Spec A_f}\cong \Phi^{-1}(\OO_X|_{D(f)}).
\]

Finally, if $\mfp\in D(f)$ corresponds to $\mfp A_f$, then
\[
\OO_{\Spec A_f,\mfp A_f}=(A_f)_{\mfp A_f}\cong A_{\mfp}=\OO_{X,\mfp}.
\]
This is an isomorphism of local rings; the maximal ideals correspond. Hence the homeomorphism together with the sheaf isomorphism is an isomorphism of locally ringed spaces. Therefore
\[
\boxed{(D(f),\OO_X|_{D(f)})\cong\Spec A_f.}
\]
\end{solution}

\paragraph{What happened mathematically.}
Localization is not merely analogous to restricting to a distinguished open: it gives the exact coordinate ring, topology, structure sheaf, and local rings of that open chart.

\paragraph{Extensions.}
Use the result to identify $D(f_1\cdots f_r)$ and to compare iterated distinguished-open restrictions.
