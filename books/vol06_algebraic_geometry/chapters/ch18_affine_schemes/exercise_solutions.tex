\begin{solution}\textbf{VI/18 Exercise 1.}
$\Spec k=\{(0)\}$, its unique stalk is $k$, and its residue field is $k$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 2.}
The generic point is $(0)$ and the closed points are $(t-a)$.  The local rings are
$k(t)$ at $(0)$ and $k[t]_{(t-a)}$ at $(t-a)$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 3.}
An isomorphism of locally ringed spaces induces
$\Gamma(\Spec A,\OO)\cong\Gamma(\Spec B,\OO)$, hence $A\cong B$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 4.}
Localization theory gives a bijection between primes of $S^{-1}A$ and primes of $A$
disjoint from $S$.  For powers of $f$, disjointness is exactly $f\notin\mfp$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 5.}
Under $\mfq\leftrightarrow\mfp=\mfq\cap A$, one has
$g/1\notin\mfq$ iff $g\notin\mfp$, while automatically $f\notin\mfp$.  Thus the image is
$D(f)\cap D(g)=D(fg)$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 6.}
The map
\[
\frac{a/f^m}{(g/1)^n}\mapsto\frac{a}{f^mg^n}
\]
is induced by the universal property.  Conversely $A\to(A_f)_{g/1}$ makes both $f$ and
$g$ units, hence factors uniquely through $A_{fg}$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 7.}
Both localizations invert precisely the images of elements of $A\setminus\mfp$.
The canonical isomorphism carries the unique maximal ideal
$\mfp(A_f)_{\mfp A_f}$ to $\mfp A_{\mfp}$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 8.}
If $f$ is a unit, no prime contains it.  Conversely, if no prime contains $f$ and $(f)$
were proper, a maximal ideal containing $(f)$ would contradict this.  Thus $f$ is a unit,
and localization at $f$ is $A$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 9.}
$D(f)=\varnothing$ iff $f$ lies in every prime, i.e. in $\sqrt{(0)}$, so $f$ is nilpotent.
If $f^n=0$ but $f$ is invertible in $A_f$, then $1=f^{-n}f^n=0$, so $A_f=0$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 10.}
\[
\Aone_k\setminus V(t)=D(t)\cong\Spec k[t,t^{-1}],
\]
whose global section ring is $k[t,t^{-1}]$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 11.}
The generic point $(0)$ remains, and among closed points $(p)$ exactly those with
$p\nmid30$ remain.  The global section ring is $\mathbb Z[1/30]$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 12.}
For any open neighborhood $U$ of $\mfp$, the distinguished-open basis supplies
$\mfp\in D(f)\subseteq U$.  By the basic-open affine theorem, $D(f)\cong\Spec A_f$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 13.}
If $\{D(f_i)\}$ covers $\Spec A$, the $f_i$ generate $A$; otherwise a maximal ideal
containing all of them is uncovered.  A finite expression $1=\sum_{j=1}^r a_jf_{i_j}$
then yields a finite subcover.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 14.}
Every prime of $A\times B$ is $\mfp\times B$ or $A\times\mfq$.  The two families are the
distinguished opens defined by $(1,0)$ and $(0,1)$, giving
$\Spec(A\times B)\cong\Spec A\sqcup\Spec B$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 15.}
Both spectra have one point, but their global section rings are
$k[\varepsilon]/(\varepsilon^2)$ and $k$.  Since an affine-scheme isomorphism induces an
isomorphism on global sections, they are not isomorphic.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 16.}
In $A_x$, $x$ is a unit and $xy=0$, so $y=0$ and
$A_x\cong k[x,x^{-1}]$.  Similarly $A_y\cong k[y,y^{-1}]$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 17.}
Successive restrictions correspond to
\[
A\to A_x\to(A_x)_y\to((A_x)_y)_z\cong A_{xyz}.
\]
Thus the final open is $D(xyz)$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 18.}
A global function is a Laurent polynomial lying in both $k[t]$ and $k[t^{-1}]$.
Their intersection in $k[t,t^{-1}]$ is $k$, so $\Gamma(\mathbb P^1_k,\OO)=k$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 19.}
Equal radicals give $f^m\in(g)$ and $g^n\in(f)$ for some $m,n$.  Thus $g$ is a unit in
$A_f$ and $f$ is a unit in $A_g$.  The universal property supplies mutually inverse
$A$-algebra maps $A_f\leftrightarrows A_g$.
\end{solution}

\begin{solution}\textbf{VI/18 Exercise 20.}
VI/19 proves the natural bijection
\[
\operatorname{Hom}_{\mathrm{Sch}}(\Spec B,\Spec A)
\cong
\operatorname{Hom}_{\mathrm{Ring}}(A,B),
\]
which upgrades object-level recovery to the anti-equivalence.
\end{solution}
