\paragraph{Solution to VI.18.E1 (Definition check).}
\begin{solution}
An affine scheme associated with a commutative ring $A$ is the locally ringed space
\[
X=\Spec A=(\Spec A,\mathcal O_{\Spec A}).
\]
Thus the data consist of the prime spectrum with its Zariski topology together with the structure sheaf.  At a point $\mathfrak p\in\Spec A$, the stalk is obtained by taking the direct limit over neighborhoods of $\mathfrak p$.  Since the distinguished opens $D(f)$ with $f\notin\mathfrak p$ form a neighborhood basis and
\(\mathcal O_X(D(f))=A_f\), one obtains
\[
\mathcal O_{X,\mathfrak p}\cong\varinjlim_{f\notin\mathfrak p}A_f\cong A_{\mathfrak p}.
\]
This is a local ring with maximal ideal $\mathfrak pA_{\mathfrak p}$, which is why $\Spec A$ with its structure sheaf is a locally ringed space.
\end{solution}

\paragraph{Solution to VI.18.E2 (Global sections).}
\begin{solution}
The whole spectrum is the distinguished open $D(1)$.  The principal-open formula therefore gives
\[
\Gamma(\Spec A,\mathcal O)=\mathcal O_{\Spec A}(D(1))=A_1.
\]
Localizing at $1$ does not change the ring: the canonical map $A\to A_1$, $a\mapsto a/1$, is an isomorphism. Hence
\[
\boxed{\Gamma(\Spec A,\mathcal O)\cong A.}
\]
The isomorphism is canonical and sends $a\in A$ to the global regular function whose germ at every prime is $a/1$.
\end{solution}

\paragraph{Solution to VI.18.E3 (Affine reconstruction).}
\begin{solution}
Suppose $X$ is affine. Choose an isomorphism of locally ringed spaces
\(\phi:X\xrightarrow{\sim}\Spec A\).  Isomorphic locally ringed spaces have isomorphic rings of global sections, so
\[
\Gamma(X,\mathcal O_X)\cong\Gamma(\Spec A,\mathcal O_{\Spec A})\cong A.
\]
Taking spectra gives
\[
\Spec\Gamma(X,\mathcal O_X)\cong\Spec A\cong X.
\]
Although the displayed proof uses a presentation $X\cong\Spec A$, the resulting ring $\Gamma(X,\mathcal O_X)$ is intrinsic to $X$; hence reconstruction does not depend on the chosen affine presentation.
\end{solution}

\paragraph{Solution to VI.18.E4 (Field spectrum).}
\begin{solution}
A field $k$ has exactly one prime ideal, namely $(0)$. Thus $\Spec k$ is a one-point topological space. Its only nonempty open set is the whole space, and
\[
\Gamma(\Spec k,\mathcal O)=k.
\]
The unique stalk is $k_{(0)}=k$, whose maximal ideal is $(0)$. Hence $\Spec k$ is the one-point reduced affine scheme with residue field $k$.
\end{solution}

\paragraph{Solution to VI.18.E5 (Empty affine scheme).}
\begin{solution}
Prime ideals are, by definition, proper ideals. The zero ring has only one ideal, namely $(0)=(1)$, so it has no proper ideals and hence no prime ideals. Therefore
\[
\Spec 0=\varnothing.
\]
Conversely, if a unital ring $A$ is nonzero, then $A$ has a maximal ideal (by the usual maximal-ideal existence theorem), and every maximal ideal is prime. Thus a nonzero ring has nonempty spectrum. Consequently $\Spec A=\varnothing$ exactly when $A=0$.
\end{solution}

\paragraph{Solution to VI.18.E6 (Affine line).}
\begin{solution}
Let $A=k[t]$ with $k$ algebraically closed. Since $A$ is a domain, $(0)$ is a prime ideal and is the generic point of $\Spec k[t]$: its closure is all of the spectrum.  The closed points are the maximal ideals. By the weak Nullstellensatz every maximal ideal of $k[t]$ has the form
\[
(t-a),\qquad a\in k.
\]
Thus the affine line as a scheme contains the classical points $a\in k$ together with the additional generic point $(0)$.
\end{solution}

\paragraph{Solution to VI.18.E7 (Arithmetic line).}
\begin{solution}
The prime ideals of $\mathbb Z$ are $(0)$ and $(p)$ for rational primes $p$. The generic point is $(0)$ and the closed points are the maximal ideals $(p)$.  At the generic point,
\[
\kappa((0))=\operatorname{Frac}(\mathbb Z)=\mathbb Q.
\]
At a closed point $(p)$,
\[
\kappa((p))=\operatorname{Frac}(\mathbb Z/(p))=\mathbb F_p.
\]
Thus one affine scheme contains a rational generic residue field and a different finite residue field at each closed point.
\end{solution}

\paragraph{Solution to VI.18.E8 (Principal open).}
\begin{solution}
For $A=k[x,y]$, the fundamental localization theorem identifies
\[
D(x)\cong\Spec A_x.
\]
Localizing at $x$ makes $x$ invertible, so
\[
A_x=k[x,y,x^{-1}].
\]
Therefore
\[
\boxed{D(x)\cong\Spec k[x,y,x^{-1}].}
\]
Under this identification a prime of the localized ring corresponds to a prime of $k[x,y]$ that does not contain $x$.
\end{solution}

\paragraph{Solution to VI.18.E9 (Intersection).}
\begin{solution}
A prime $\mathfrak p$ lies in $D(f)\cap D(g)$ exactly when $f\notin\mathfrak p$ and $g\notin\mathfrak p$. Since $\mathfrak p$ is prime, this is equivalent to $fg\notin\mathfrak p$. Hence
\[
D(f)\cap D(g)=D(fg).
\]
Applying the distinguished-open localization theorem gives, as locally ringed spaces,
\[
D(f)\cap D(g)=D(fg)\cong\Spec A_{fg}.
\]
The sheaf identification follows from the canonical isomorphism $(A_f)_{g/1}\cong A_{fg}$.
\end{solution}

\paragraph{Solution to VI.18.E10 (Unit principal open).}
\begin{solution}
If $u\in A^\times$, then no proper ideal can contain $u$: if an ideal contains a unit, it contains $1$ and equals $A$. In particular no prime ideal contains $u$. Thus every prime lies in $D(u)$ and
\[
D(u)=\Spec A.
\]
Algebraically, localizing at a unit changes nothing. The map $A\to A_u$ is an isomorphism because $u^{-1}$ already exists in $A$. This agrees with the geometric statement that the corresponding distinguished open is the whole affine scheme.
\end{solution}

\paragraph{Solution to VI.18.E11 (Nilpotent principal open).}
\begin{solution}
Suppose $f^N=0$. Every prime ideal contains every nilpotent element, so every prime contains $f$. Therefore
\[
D(f)=\varnothing.
\]
In $A_f$, however, $f$ is invertible. Multiplying $f^N=0$ by $f^{-N}$ gives $1=0$, so $A_f$ is the zero ring. Thus
\[
D(f)\cong\Spec A_f=\Spec0=\varnothing,
\]
as predicted by the localization theorem.
\end{solution}

\paragraph{Solution to VI.18.E12 (Finite cover criterion).}
\begin{solution}
The distinguished opens $D(f_1),\dots,D(f_r)$ cover $\Spec A$ iff their complement is empty. Using $\Spec A\setminus D(f_i)=V(f_i)$ and intersections of closed sets,
\[
\Spec A\setminus\bigcup_iD(f_i)=V(f_1,\dots,f_r).
\]
Hence the cover condition is $V(I)=\varnothing$ for $I=(f_1,\dots,f_r)$. If $I$ were proper, it would be contained in a maximal ideal, producing a point of $V(I)$. Therefore $V(I)=\varnothing$ iff $I=A$. Equivalently, there exist $a_i\in A$ such that
\[
1=a_1f_1+\cdots+a_rf_r.
\]
\end{solution}

\paragraph{Solution to VI.18.E13 (Quasi-compactness).}
\begin{solution}
Let $\{U_\lambda\}$ be an arbitrary open cover of $X=\Spec A$. Because distinguished opens form a basis, refine it by distinguished opens $D(f_i)$, each contained in some $U_{\lambda(i)}$. The refined family still covers $X$, so the ideal generated by all the $f_i$ is $A$. Thus
\[
1=a_1f_{i_1}+\cdots+a_rf_{i_r}
\]
for finitely many of them. By the finite cover criterion, $D(f_{i_1}),\dots,D(f_{i_r})$ cover $X$. The corresponding finitely many original opens $U_{\lambda(i_j)}$ therefore cover $X$. Hence every affine scheme is quasi-compact.
\end{solution}

\paragraph{Solution to VI.18.E14 (Product ring).}
\begin{solution}
Let $R=A\times B$ and put $e_1=(1,0)$ and $e_2=(0,1)$. Since $e_1e_2=0$ and $e_1+e_2=1$, a prime ideal of $R$ contains exactly one of $e_1,e_2$. If it contains $e_2$, it has the form $\mathfrak p\times B$ for a unique $\mathfrak p\in\Spec A$; if it contains $e_1$, it has the form $A\times\mathfrak q$ for a unique $\mathfrak q\in\Spec B$.

The distinguished opens $D(e_1)$ and $D(e_2)$ are disjoint, clopen, and cover $\Spec R$. Moreover
\[
R_{e_1}\cong A,\qquad R_{e_2}\cong B.
\]
Hence the localization theorem identifies these two pieces, including their structure sheaves, with $\Spec A$ and $\Spec B$. Thus
\[
\boxed{\Spec(A\times B)\cong\Spec A\sqcup\Spec B}
\]
as affine schemes.
\end{solution}

\paragraph{Solution to VI.18.E15 (Local ring geometry).}
\begin{solution}
The closed points of $\Spec A$ are exactly the maximal ideals of $A$. If $A$ is local, it has one maximal ideal $\mathfrak m$, so $\Spec A$ has exactly one closed point. Conversely, if $\Spec A$ has exactly one closed point, then $A$ has exactly one maximal ideal. By definition this means $A$ is a local ring.

This does not say that $\Spec A$ has only one point: nonmaximal prime ideals may still occur. For example, a local domain of positive dimension has its closed point together with a generic point and possibly intermediate primes.
\end{solution}

\paragraph{Solution to VI.18.E16 (Dual numbers).}
\begin{solution}
The field $k$ has one prime ideal. In $B=k[\varepsilon]/(\varepsilon^2)$, every prime contains the nilpotent element $\varepsilon$, and the quotient $B/(\varepsilon)\cong k$ is a field; hence $(\varepsilon)$ is the unique prime. Thus $\Spec k$ and $\Spec B$ are homeomorphic one-point spaces.

They are not isomorphic as affine schemes. Global sections recover the coordinate rings:
\[
\Gamma(\Spec k,\mathcal O)=k,
\qquad
\Gamma(\Spec B,\mathcal O)=B.
\]
The ring $B$ has the nonzero nilpotent $\varepsilon$, whereas $k$ is reduced, so the rings are not isomorphic. Therefore the structure sheaves distinguish the two schemes despite identical topology.
\end{solution}

\paragraph{Solution to VI.18.E17 (Laurent line).}
\begin{solution}
The distinguished-open theorem gives
\[
D(t)\subseteq\Spec k[t]\cong\Spec k[t]_t.
\]
Inverting $t$ allows both positive and negative powers of $t$, so every element of $k[t]_t$ is a Laurent polynomial and
\[
k[t]_t=k[t,t^{-1}].
\]
Consequently
\[
\boxed{D(t)\cong\Spec k[t,t^{-1}]=\mathbb G_m.}
\]
\end{solution}

\paragraph{Solution to VI.18.E18 (Arithmetic localization).}
\begin{solution}
By localization,
\[
D(30)\cong\Spec\mathbb Z[1/30].
\]
A closed point $(p)$ lies in $D(30)$ iff $30\notin(p)$, equivalently iff $p\nmid30$. Thus the closed points $(2),(3),(5)$ are removed, while every $(p)$ with $p\notin\{2,3,5\}$ remains. The generic point $(0)$ also remains because $30\neq0$.

The coordinate ring may be written $\mathbb Z[1/2,1/3,1/5]$, making explicit exactly which primes have been inverted.
\end{solution}

\paragraph{Solution to VI.18.E19 (Iterated localization).}
\begin{solution}
First localize $A$ at powers of $f$, obtaining $A_f$. Then localize at $g/1$. An element of $(A_f)_{g/1}$ can be represented as
\[
\frac{a/f^m}{(g/1)^n},
\]
which maps naturally to $a/(f^mg^n)$ in $A_{fg}$. This gives a ring homomorphism
\[
(A_f)_{g/1}\longrightarrow A_{fg}.
\]
Conversely, in $(A_f)_{g/1}$ both $f$ and $g$ are units, so $fg$ is a unit. The universal property of localization therefore gives a map $A_{fg}\to(A_f)_{g/1}$. The two maps are inverse on the image of $A$ and hence inverse everywhere. Thus
\[
(A_f)_{g/1}\cong A_{fg}.
\]
\end{solution}

\paragraph{Solution to VI.18.E20 (Residue field).}
\begin{solution}
The polynomial $t^2+1$ is irreducible over $\mathbb R$, so $(t^2+1)$ is a maximal ideal of $\mathbb R[t]$. The residue field at this point is
\[
\kappa((t^2+1))
\cong \mathbb R[t]/(t^2+1).
\]
Sending the class of $t$ to $i$ identifies this quotient with $\mathbb C$. Hence
\[
\boxed{\kappa((t^2+1))\cong\mathbb C.}
\]
This is an example of a closed point whose residue field is a nontrivial extension of the ground field.
\end{solution}

\paragraph{Solution to VI.18.E21 (Classical affine set).}
\begin{solution}
Let
\[
A=k[x,y]/(y-x^2).
\]
The map $k[x,y]\to k[x]$ given by $x\mapsto x$, $y\mapsto x^2$ is surjective with kernel $(y-x^2)$, so $A\cong k[x]$. Therefore $A$ is a domain and its generic point is the zero prime.

If $k$ is algebraically closed, the maximal ideals of $k[x]$ are $(x-a)$, $a\in k$. Under the isomorphism with $A$, these correspond to
\[
(x-a,\,y-a^2)/(y-x^2).
\]
Thus the closed points are exactly the classical points $(a,a^2)$ on the parabola, while the scheme also contains the generic point of the whole irreducible curve.
\end{solution}

\paragraph{Solution to VI.18.E22 (One-point criterion).}
\begin{solution}
By definition, $\Spec A$ has one point exactly when $A$ has exactly one prime ideal. If that prime is $\mathfrak p$, then it contains the nilradical, while the nilradical is the intersection of all prime ideals; hence $\sqrt{(0)}=\mathfrak p$. Every maximal ideal is prime, so $\mathfrak p$ is also the unique maximal ideal.

Equivalently, one may say that $\sqrt{(0)}$ is a maximal ideal. Indeed, if the nilradical is maximal, every prime contains it, hence must equal it. Thus
\[
\Spec A\text{ has one point}\iff \sqrt{(0)}\text{ is maximal}.
\]
\end{solution}

\paragraph{Solution to VI.18.E23 (Principal neighborhoods).}
\begin{solution}
Let $\mathfrak p\in\Spec A$. The distinguished opens $D(f)$ with $f\notin\mathfrak p$ form a neighborhood basis at $\mathfrak p$: every open neighborhood contains some such distinguished open. By the localization theorem each of these neighborhoods is affine,
\[
D(f)\cong\Spec A_f.
\]
Therefore every point of an affine scheme has a basis consisting of affine open neighborhoods.
\end{solution}

\paragraph{Solution to VI.18.E24 (Recover an isomorphism invariant).}
\begin{solution}
An isomorphism of affine schemes is in particular an isomorphism of locally ringed spaces, so it induces an isomorphism on global section rings. Thus
\[
X\cong Y\quad\Longrightarrow\quad
\Gamma(X,\mathcal O_X)\cong\Gamma(Y,\mathcal O_Y).
\]
Consequently, if the two global rings are not isomorphic, then the affine schemes cannot be isomorphic. For presentations $X\cong\Spec A$ and $Y\cong\Spec B$, this becomes the simple obstruction $A\not\cong B\Rightarrow X\not\cong Y$.
\end{solution}

\paragraph{Solution to VI.18.E25 (Projective-line global functions).}
\begin{solution}
Cover $\mathbb P^1_k$ by the standard affine charts
\[
U_0\cong\Spec k[t],\qquad U_1\cong\Spec k[u],
\]
with $u=t^{-1}$ on the overlap. A global regular function is therefore a pair
\[
f(t)\in k[t],\qquad g(u)\in k[u]
\]
whose restrictions agree. Inside $k[t,t^{-1}]$ this means
\[
f(t)=g(t^{-1}).
\]
The polynomial $f(t)$ contains only nonnegative powers of $t$, while $g(t^{-1})$ contains only nonpositive powers. Equality is possible only for the common degree-zero part. Hence both are the same constant and
\[
\boxed{\Gamma(\mathbb P^1_k,\mathcal O)=k.}
\]
\end{solution}

\paragraph{Solution to VI.18.E26 (Boundary to VI/19).}
\begin{solution}
A continuous map of the underlying spectra is not enough to define a morphism of schemes. A morphism of schemes is a morphism of locally ringed spaces. Thus, in addition to a continuous map $f:X\to Y$, one needs a morphism of sheaves of rings
\[
f^\#:\mathcal O_Y\longrightarrow f_*\mathcal O_X
\]
compatible with restrictions. For every $x\in X$, this induces a homomorphism on stalks
\[
f_x^\#:\mathcal O_{Y,f(x)}\longrightarrow\mathcal O_{X,x},
\]
and the locally ringed-space condition requires this homomorphism to be local, i.e. to send the maximal ideal of the source into the maximal ideal of the target. VI/19 proves that for affine schemes these data are equivalent to a contravariant ring homomorphism.
\end{solution}
