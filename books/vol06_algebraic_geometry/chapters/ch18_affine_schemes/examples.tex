\begin{example}[A doubled Laurent line]
Let $A=k[t,\varepsilon]/(\varepsilon^2)$ and consider $D(t)$.  Then
\[
D(t)\cong\Spec k[t,t^{-1},\varepsilon]/(\varepsilon^2).
\]
The nilpotent direction survives localization even though the underlying open point set is
the same as for the reduced Laurent line.
\end{example}

\begin{example}[Arithmetic localization]
For $X=\Spec\mathbb Z$ and $n\ne0$,
\[
D(n)\cong\Spec\mathbb Z[1/n].
\]
The closed points $(p)$ with $p\mid n$ are removed, while the generic point remains.
\end{example}

\begin{example}[Two successive restrictions]
For $X=\Spec k[x,y]$, first restrict to $D(x)$ and then inside it to $D(y/1)$.
The resulting affine scheme is
\[
\Spec(k[x,y]_x)_{y/1}
\cong\Spec k[x,y]_{xy}
\cong D(xy).
\]
\end{example}

\begin{example}[A clopen affine decomposition]
Let $A=k\times k[t]$.  Then
\[
\Spec A\cong\Spec k\;\sqcup\;\Spec k[t].
\]
Both pieces are distinguished opens cut out by idempotents and hence affine.
\end{example}

\begin{example}[A principal open determined only by the radical]
If $\sqrt{(f)}=\sqrt{(g)}$, then $D(f)=D(g)$.  Each element becomes a unit in the
localization at the other, so the universal property gives a canonical $A$-algebra
isomorphism
\[
A_f\cong A_g.
\]
Thus the same principal open carries the same affine scheme independent of the chosen
equation defining its complement.
\end{example}

\begin{example}[Principal open on crossing axes]
For $A=k[x,y]/(xy)$, localizing at $x$ makes $x$ a unit, hence the relation $xy=0$
forces $y=0$.  Therefore
\[
A_x\cong k[x,x^{-1}],
\qquad
D(x)\cong\Spec k[x,x^{-1}].
\]
Similarly $D(y)\cong\Spec k[y,y^{-1}]$.
\end{example}
