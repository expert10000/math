\begin{example}[The spectrum of a field]
For a field $k$, $\Spec k$ has one point and
\[
\Gamma(\Spec k,\OO)=k.
\]
Its unique local ring and residue field are both $k$.
\end{example}

\begin{example}[The affine line]
The affine line over $k$ is
\[
\A^1_k=\Spec k[t].
\]
It has generic point $(0)$, closed points from maximal ideals, and structure sheaf
$\OO(D(f))=k[t]_f$.
\end{example}

\begin{example}[Affine $n$-space]
\[
\A^n_k=\Spec k[x_1,\dots,x_n].
\]
When $k$ is algebraically closed, its closed points are the maximal ideals
$(x_1-a_1,\dots,x_n-a_n)$, while nonmaximal primes encode irreducible subvarieties.
\end{example}

\begin{example}[The arithmetic affine line]
$X=\Spec\mathbb Z$ is affine.  Its generic point has local ring $\mathbb Q$, and the closed
point $(p)$ has local ring $\mathbb Z_{(p)}$ and residue field $\mathbb F_p$.
\end{example}

\begin{example}[A nonreduced affine point]
For $A=k[\varepsilon]/(\varepsilon^2)$, $\Spec A$ has one point, but
$\Gamma(X,\OO_X)=A$ contains the nonzero nilpotent $\varepsilon$.
\end{example}

\begin{example}[Two crossing affine lines]
\[
X=\Spec k[x,y]/(xy)
\]
is affine and reducible.  Its minimal primes $(x)$ and $(y)$ are the generic points of the
two components.
\end{example}

\begin{example}[A disconnected affine scheme]
For $A=k\times k$,
\[
\Spec A\cong\Spec k\sqcup\Spec k.
\]
The two points are both open and closed, and the global ring is $k\times k$.
\end{example}

\begin{example}[The empty affine scheme]
For the zero ring,
\[
\Spec 0=\varnothing.
\]
This is the empty affine scheme.
\end{example}

\begin{example}[A punctured affine line is affine]
Inside $\A^1_k$,
\[
D(t)\cong\Spec k[t,t^{-1}].
\]
Thus removing the origin from the affine line gives the affine multiplicative group $\Gm$.
\end{example}

\begin{example}[Removing two coordinate axes]
In $\A^2_k=\Spec k[x,y]$,
\[
D(xy)\cong\Spec k[x,y,x^{-1},y^{-1}].
\]
Both $x$ and $y$ become units.
\end{example}

\begin{example}[An arithmetic distinguished open]
\[
D(6)\subseteq\Spec\mathbb Z
\]
is affine with coordinate ring $\mathbb Z[1/6]$.  The points $(2)$ and $(3)$ are removed.
\end{example}

\begin{example}[A unit gives the whole affine scheme]
If $u\in A^\times$, then $D(u)=\Spec A$ and $A_u\cong A$.  The localization theorem gives
no change of affine scheme.
\end{example}

\begin{example}[A nilpotent gives the empty affine scheme]
If $f$ is nilpotent, then $D(f)=\varnothing$ and $A_f=0$, so the localization theorem reads
\[
\varnothing\cong\Spec 0.
\]
\end{example}

\begin{example}[Iterated localization]
For $f,g\in A$,
\[
D(g/1)\subseteq\Spec A_f
\]
corresponds to $D(fg)\subseteq\Spec A$, and
\[
(A_f)_{g/1}\cong A_{fg}.
\]
\end{example}

\begin{example}[An affine local scheme]
For the local ring $A=k[x]_{(x)}$, the affine scheme $\Spec A$ has unique closed point
$(x)A$, but it also has the generic point $(0)$.
\end{example}

\begin{example}[Recovering a ring from its affine scheme]
If $X=\Spec k[x,y]/(x^2-y^3)$, then
\[
\Gamma(X,\OO_X)=k[x,y]/(x^2-y^3).
\]
Thus the coordinate ring is an invariant of the affine scheme.
\end{example}

\begin{example}[A quotient ring still gives an affine scheme]
For any ideal $I\subseteq A$, the ring $A/I$ defines the affine scheme $\Spec(A/I)$.
The interpretation as a closed subscheme of $\Spec A$ is postponed to VI/22.
\end{example}

\begin{example}[Affine scheme attached to a classical conic]
The classical curve $x^2+y^2=1$ has coordinate ring
\[
A=k[x,y]/(x^2+y^2-1),
\]
and its affine-scheme enhancement is $\Spec A$.
\end{example}

\begin{example}[Closed points need not have residue field $k$]
For $X=\Spec\mathbb R[t]$, the maximal ideal $(t^2+1)$ defines a closed point with
\[
\kappa(t^2+1)\cong\mathbb C.
\]
The affine scheme naturally remembers this extension field.
\end{example}

\begin{example}[A principal affine neighborhood]
Let $\mfp\in\Spec A$ and choose $f\notin\mfp$.  Then
\[
\mfp\in D(f)\cong\Spec A_f.
\]
Thus every point of an affine scheme has arbitrarily small affine neighborhoods from the
distinguished-open basis.
\end{example}
