\begin{example}[Advanced: Product decompositions]
For a finite product $A=A_1\times\cdots\times A_r$,
\[
\Spec A\cong\coprod_{i=1}^r\Spec A_i.
\]
Each component is open and closed, and the structure sheaf restricts to the corresponding
factor.  Affine disconnectedness is therefore encoded by idempotents.
\end{example}

\begin{example}[Advanced: One-point spectra with different infinitesimal thickness]
For every $n\ge2$, the ring $k[\varepsilon]/(\varepsilon^n)$ has a one-point spectrum, but
its affine scheme is distinguished by its global ring and nilpotent filtration.
\end{example}

\begin{example}[Advanced: Distinguished opens are stable under intersection]
Using $D(f)\cap D(g)=D(fg)$,
\[
\Spec A_f\cap\Spec A_g\quad\text{inside }\Spec A
\]
is represented by $\Spec A_{fg}$.  This makes principal affine charts exceptionally easy
to compare.
\end{example}

\begin{example}[Advanced: A finite principal cover]
If $f_1,\dots,f_r$ generate the unit ideal, then
\[
\Spec A=D(f_1)\cup\cdots\cup D(f_r).
\]
For $A=k[x]$ one has $D(x)\cup D(1-x)=\A^1_k$ because $x+(1-x)=1$.
\end{example}

\begin{example}[Advanced: Why affine schemes are quasi-compact]
Even if an affine scheme has many points, an arbitrary open cover can be refined to
principal opens and then reduced to finitely many by the unit-ideal criterion.  This is an
algebraic compactness phenomenon, not Hausdorff compactness.
\end{example}

\begin{example}[Advanced: A local ring with many points]
Let $A=k[x,y]_{(x,y)}$.  The closed point is $(x,y)A$, but the primes $(0)$, $(x)$, $(y)$,
and many height-one primes also remain.  Local means one closed point, not one point total.
\end{example}

\begin{example}[Advanced: Generic point of an irreducible affine scheme]
If $A$ is a domain, $(0)$ is a point of $\Spec A$ whose closure is all of $\Spec A$.  Its
local ring is $\Frac(A)$.  The broader integral-scheme theory is reserved for VI/27.
\end{example}

\begin{example}[Advanced: Affineness survives localization]
Any iterated localization $S^{-1}A$ defines an affine scheme.  If $S$ is generated by one
element this is a distinguished open; for general $S$ it is the inverse limit of the
corresponding principal-open localizations at the ring level, while its spectrum consists of
primes disjoint from $S$.
\end{example}

\begin{example}[Advanced: Topology does not determine affineness data]
$\Spec k$ and $\Spec k[\varepsilon]/(\varepsilon^2)$ are homeomorphic, but the global
section rings differ.  Any test of affine-scheme isomorphism that ignores the sheaf therefore
fails immediately.
\end{example}

\begin{example}[Advanced: The canonical affine reconstruction]
If $X$ is known to be affine, choose an isomorphism $X\cong\Spec A$.  Then
$\Gamma(X,\OO_X)\cong A$, so composing with the induced spectrum isomorphism gives
\[
X\cong\Spec\Gamma(X,\OO_X).
\]
No arbitrary choice of coordinate ring survives in the final expression.
\end{example}

\begin{example}[Advanced: The projective line has only constant global functions]
Cover $\mathbb P^1_k$ by $U_0\cong\Spec k[t]$ and $U_1\cong\Spec k[u]$, with
$u=t^{-1}$ on the overlap.  A global regular function is a pair
\[
f(t)\in k[t],\qquad g(u)\in k[u]
\]
that agree in $k[t,t^{-1}]$.  Hence $f$ belongs to both $k[t]$ and $k[t^{-1}]$, so $f\in k$.
Thus $\Gamma(\mathbb P^1_k,\OO)=k$.
\end{example}

\begin{example}[Advanced: Non-affineness of $\mathbb P^1$]
If $\mathbb P^1_k$ were affine, affine reconstruction would give
\[
\mathbb P^1_k\cong\Spec\Gamma(\mathbb P^1_k,\OO)=\Spec k,
\]
a one-point scheme.  Since $\mathbb P^1_k$ has many points, this is impossible.
\end{example}

\begin{example}[Advanced: The punctured affine plane]
Let $U=\A^2_k\setminus\{(x,y)\} = D(x)\cup D(y)$.  The sheaf equalizer gives
\[
\Gamma(U,\OO)=k[x,y]_x\cap k[x,y]_y=k[x,y]
\]
inside $k(x,y)$.  This is the starting calculation for the classical fact that $U$ is not
affine; a complete proof is posed as a Challenge because it uses the canonical map to
$\Spec\Gamma(U,\OO)$.
\end{example}

\begin{example}[Advanced: Affine charts as a basis for later scheme theory]
The theorem $D(f)\cong\Spec A_f$ shows that an affine scheme already carries many overlapping
affine models.  VI/20 will reverse this perspective: compatible affine models can be glued to
construct spaces that need not themselves be affine.
\end{example}
