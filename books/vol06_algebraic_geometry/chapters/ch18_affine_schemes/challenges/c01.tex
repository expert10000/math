\begin{problem}[Challenge: The punctured affine plane]\label{chal:vi18-c01}
Let $U=\mathbb A^2_k\setminus\{(0,0)\}=D(x)\cup D(y)$.  Prove
\[
\Gamma(U,\OO_U)=k[x,y]
\]
by an explicit intersection calculation.  Then, using the canonical map to
$\Spec\Gamma(U,\OO_U)$, prove that $U$ is not affine.
\end{problem}
