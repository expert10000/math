\begin{exercise}[A field]\label{ex:vi18-01}
Describe $\Spec k$ as an affine scheme, including its unique stalk and residue field.
\end{exercise}

\begin{exercise}[Affine line]\label{ex:vi18-02}
For algebraically closed $k$, list the generic and closed points of $\Aone_k$ and compute
the local rings at $(0)$ and $(t-a)$.
\end{exercise}

\begin{exercise}[Global recovery]\label{ex:vi18-03}
Prove that an isomorphism $\Spec A\cong\Spec B$ of locally ringed spaces induces $A\cong B$.
\end{exercise}

\begin{exercise}[Prime correspondence]\label{ex:vi18-04}
Prove directly that primes of $A_f$ correspond bijectively to primes of $A$ avoiding $f$.
\end{exercise}

\begin{exercise}[Basic opens under localization]\label{ex:vi18-05}
Show that $D(g/1)\subseteq\Spec A_f$ corresponds to $D(fg)\subseteq\Spec A$.
\end{exercise}

\begin{exercise}[Associativity of localization]\label{ex:vi18-06}
Construct explicitly the isomorphism $(A_f)_{g/1}\cong A_{fg}$ and its inverse.
\end{exercise}

\begin{exercise}[Stalk comparison]\label{ex:vi18-07}
For $\mfp\in D(f)$, prove $(A_f)_{\mfp A_f}\cong A_{\mfp}$ as local rings.
\end{exercise}

\begin{exercise}[Unit principal open]\label{ex:vi18-08}
Show $D(f)=\Spec A$ iff $f$ is a unit, and compare $A_f$ with $A$.
\end{exercise}

\begin{exercise}[Nilpotent principal open]\label{ex:vi18-09}
Show $D(f)=\varnothing$ iff $f$ is nilpotent, and explain why $A_f$ is then the zero ring.
\end{exercise}

\begin{exercise}[Laurent line]\label{ex:vi18-10}
Identify $\Aone_k\setminus V(t)$ as an affine scheme and compute its global sections.
\end{exercise}

\begin{exercise}[Arithmetic open]\label{ex:vi18-11}
Describe the points and global sections of $D(30)\subseteq\Spec\mathbb Z$.
\end{exercise}

\begin{exercise}[Affine basis]\label{ex:vi18-12}
Prove that every point of an affine scheme has a basis of affine open neighborhoods.
\end{exercise}

\begin{exercise}[Quasi-compactness]\label{ex:vi18-13}
Give a complete proof that every distinguished-open cover of $\Spec A$ has a finite subcover.
\end{exercise}

\begin{exercise}[Product ring]\label{ex:vi18-14}
Describe $\Spec(A\times B)$ and identify the two clopen affine pieces as distinguished opens.
\end{exercise}

\begin{exercise}[Dual numbers]\label{ex:vi18-15}
Explain why $\Spec k[\varepsilon]/(\varepsilon^2)$ and $\Spec k$ are homeomorphic but not
isomorphic as affine schemes.
\end{exercise}

\begin{exercise}[Crossing axes]\label{ex:vi18-16}
For $A=k[x,y]/(xy)$ compute $D(x)$ and $D(y)$ as affine schemes.
\end{exercise}

\begin{exercise}[Repeated localization]\label{ex:vi18-17}
For $A=k[x,y,z]$, identify the result of restricting successively to $D(x)$, then $D(y)$,
then $D(z)$.
\end{exercise}

\begin{exercise}[Projective global functions]\label{ex:vi18-18}
Using the two standard charts of $\Pone_k$, reprove that its global regular functions are
exactly $k$.
\end{exercise}

\begin{exercise}[Equal principal opens]\label{ex:vi18-19}
Assume $\sqrt{(f)}=\sqrt{(g)}$.  Prove $D(f)=D(g)$ and construct a canonical
$A$-algebra isomorphism $A_f\cong A_g$.
\end{exercise}

\begin{exercise}[Boundary to VI/19]\label{ex:vi18-20}
State the additional theorem about morphisms needed to upgrade global-section recovery to
an anti-equivalence between rings and affine schemes.
\end{exercise}
