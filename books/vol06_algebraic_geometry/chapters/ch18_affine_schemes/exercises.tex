\begin{exercise}[Definition check]\label{ex:vi18-01}
State the data of the affine scheme $\Spec A$ and identify its stalk at $\mfp$.
\end{exercise}

\begin{exercise}[Global sections]\label{ex:vi18-02}
Prove $\Gamma(\Spec A,\OO)=A$.
\end{exercise}

\begin{exercise}[Affine reconstruction]\label{ex:vi18-03}
If $X$ is affine, show $X\cong\Spec\Gamma(X,\OO_X)$.
\end{exercise}

\begin{exercise}[Field spectrum]\label{ex:vi18-04}
Describe $\Spec k$ as an affine scheme for a field $k$.
\end{exercise}

\begin{exercise}[Empty affine scheme]\label{ex:vi18-05}
Show $\Spec 0=\varnothing$.
\end{exercise}

\begin{exercise}[Affine line]\label{ex:vi18-06}
List the generic and closed points of $\Spec k[t]$ when $k$ is algebraically closed.
\end{exercise}

\begin{exercise}[Arithmetic line]\label{ex:vi18-07}
Describe the points and residue fields of $\Spec\mathbb Z$.
\end{exercise}

\begin{exercise}[Principal open]\label{ex:vi18-08}
For $A=k[x,y]$, identify $D(x)$ with an affine spectrum.
\end{exercise}

\begin{exercise}[Intersection]\label{ex:vi18-09}
Show $D(f)\cap D(g)\cong\Spec A_{fg}$.
\end{exercise}

\begin{exercise}[Unit principal open]\label{ex:vi18-10}
Show $D(u)=\Spec A$ for $u\in A^\times$.
\end{exercise}

\begin{exercise}[Nilpotent principal open]\label{ex:vi18-11}
Show $D(f)=\varnothing$ for nilpotent $f$ and identify $A_f$.
\end{exercise}

\begin{exercise}[Finite cover criterion]\label{ex:vi18-12}
Prove that the distinguished opens $D(f_1),\dots,D(f_r)$ cover $\Spec A$ exactly when the ideal $(f_1,\dots,f_r)$ is $A$.
\end{exercise}

\begin{exercise}[Quasi-compactness]\label{ex:vi18-13}
Deduce that every affine scheme is quasi-compact.
\end{exercise}

\begin{exercise}[Product ring]\label{ex:vi18-14}
Describe $\Spec(A\times B)$ and its two open-and-closed pieces.
\end{exercise}

\begin{exercise}[Local ring geometry]\label{ex:vi18-15}
Prove $A$ is local iff $\Spec A$ has a unique closed point.
\end{exercise}

\begin{exercise}[Dual numbers]\label{ex:vi18-16}
Compare $\Spec k$ and $\Spec k[\varepsilon]/(\varepsilon^2)$ as spaces and as affine schemes.
\end{exercise}

\begin{exercise}[Laurent line]\label{ex:vi18-17}
Show $D(t)\subseteq\Spec k[t]$ is $\Spec k[t,t^{-1}]$.
\end{exercise}

\begin{exercise}[Arithmetic localization]\label{ex:vi18-18}
Identify $D(30)\subseteq\Spec\mathbb Z$ with a spectrum and list the closed points removed.
\end{exercise}

\begin{exercise}[Iterated localization]\label{ex:vi18-19}
Prove $(A_f)_{g/1}\cong A_{fg}$.
\end{exercise}

\begin{exercise}[Residue field]\label{ex:vi18-20}
Compute the residue field at $(t^2+1)\in\Spec\mathbb R[t]$.
\end{exercise}

\begin{exercise}[Classical affine set]\label{ex:vi18-21}
For $A=k[x,y]/(y-x^2)$, describe the closed points and generic point of $\Spec A$.
\end{exercise}

\begin{exercise}[One-point criterion]\label{ex:vi18-22}
Give a necessary and sufficient condition on the prime ideals of $A$ for $\Spec A$ to have one point.
\end{exercise}

\begin{exercise}[Principal neighborhoods]\label{ex:vi18-23}
Show every point of an affine scheme has a basis of affine neighborhoods.
\end{exercise}

\begin{exercise}[Recover an isomorphism invariant]\label{ex:vi18-24}
Explain why nonisomorphic global rings imply nonisomorphic affine schemes.
\end{exercise}

\begin{exercise}[Projective-line global functions]\label{ex:vi18-25}
Using the two standard affine charts, prove $\Gamma(\mathbb P^1_k,\OO)=k$.
\end{exercise}

\begin{exercise}[Boundary to VI/19]\label{ex:vi18-26}
State what extra data beyond a continuous map of spectra are required for a morphism of affine schemes, without proving the VI/19 universal property.
\end{exercise}
