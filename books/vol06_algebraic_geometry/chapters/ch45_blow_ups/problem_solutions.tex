% VI45-DETAILED-PROBLEM-01
\subsection*{Problem VI.45.1}
\begin{solution}
For \(I=(x,y)\), the Rees algebra is \(k[x,y][xT,yT]\).  Introduce degree-one variables \(S,U\) mapping to \(xT,yT\).  The syzygy \(yS-xU=0\) gives
\[
k[x,y][IT]\cong k[x,y,S,U]/(yS-xU).
\]
Taking Proj in \([S:U]\) yields the incidence equation \(ys=xu\), equivalently \(xt=ys\) after renaming.
\end{solution}

% VI45-DETAILED-PROBLEM-02
\subsection*{Problem VI.45.2}
\begin{solution}
The exceptional fiber is
\[
\operatorname{Proj}\bigoplus \mathfrak m_p^n/\mathfrak m_p^{n+1}
\cong\operatorname{Proj}\operatorname{Sym}(\mathfrak m_p/\mathfrak m_p^2)
=\mathbb P(T_pX).
\]
For a smooth surface this is \(\mathbb P^1\), intrinsically the projective tangent-direction space.
\end{solution}

% VI45-DETAILED-PROBLEM-03
\subsection*{Problem VI.45.3}
\begin{solution}
Substitution \(y=xu\) gives \(x^2(u^2-1-x)\).  The strict transform is \(u^2=1+x\), meeting \(E:x=0\) at \(u=\pm1\), the two tangent directions.  Since \(2u\neq0\) there, the strict transform is smooth and the branches are separated.
\end{solution}

% VI45-DETAILED-PROBLEM-04
\subsection*{Problem VI.45.4}
\begin{solution}
Two distinct lines through \(p\) have disjoint strict transforms of class \(H-E\), so \((H-E)^2=0\).  Since \(H^2=1\) and \(H\cdot E=0\), it follows that \(E^2=-1\).
\end{solution}

% VI45-DETAILED-PROBLEM-05
\subsection*{Problem VI.45.5}
\begin{solution}
For \(\widetilde C=dH-mE\),
\[
\widetilde C^2=d^2-m^2,\qquad
K_S\cdot\widetilde C=-3d+m.
\]
Adjunction gives \(2p_a(\widetilde C)-2=d^2-m^2-3d+m\), exhibiting the usual \(m(m-1)/2\) genus correction for an ordinary point.
\end{solution}

% VI45-DETAILED-PROBLEM-06
\subsection*{Problem VI.45.6}
\begin{solution}
For \(D=2H-E_1-E_2-E_3\), \(D^2=1\), \(D\cdot E_i=1\), and
\[
D\cdot(H-E_j-E_k)=0.
\]
Thus it has exactly the numerical behavior of the pullback of a target line: square one, degree one on each exceptional curve, and contraction of the opposite line transforms.
\end{solution}

% VI45-DETAILED-PROBLEM-07
\subsection*{Problem VI.45.7}
\begin{solution}
Near \(p_1\), the two blow-up charts give regular formulas \([ut:t:1]\) and \([vs:1:s]\).  On the exceptional divisor these cover \([0:t:1]\) and \([0:1:s]\), together the entire target line \(X=0\).  Hence no point of \(E_1\) remains indeterminate.
\end{solution}

% VI45-DETAILED-PROBLEM-08
\subsection*{Problem VI.45.8}
\begin{solution}
A general target line pulls back to \(D=2H-E_1-E_2-E_3\).  Therefore for \(C=dH-\sum m_iE_i\),
\[
\deg\sigma_*(C)=D\cdot C=2d-m_1-m_2-m_3.
\]
This is the degree formula as an intersection product.
\end{solution}

% VI45-DETAILED-PROBLEM-09
\subsection*{Problem VI.45.9}
\begin{solution}
In the basis \((H,E_1,E_2,E_3)\), the Cremona action has columns given by the four basis images.  Direct multiplication gives \(M^2=I\), and \(M^T\operatorname{diag}(1,-1,-1,-1)M\) equals the same diagonal form, so it is an involutive isometry.
\end{solution}

% VI45-DETAILED-PROBLEM-10
\subsection*{Problem VI.45.10}
\begin{solution}
The common resolution is
\[
\mathbb P^2\xleftarrow{\pi}S\xrightarrow{\widetilde\sigma}\mathbb P^2.
\]
The left morphism contracts \(E_1,E_2,E_3\); the right contracts the three strict transforms of coordinate lines.  All six are \((-1)\)-curves, but they form two different triples.
\end{solution}

% VI45-DETAILED-PROBLEM-11
\subsection*{Problem VI.45.11}
\begin{solution}
A residual base point \(q\in E\) is a fixed tangent direction of the transformed pencil.  Blow up \(q\); the new exceptional curve records second-order directions through that infinitely near point.  Repeat until the transformed system is base-point free.
\end{solution}

% VI45-DETAILED-PROBLEM-12
\subsection*{Problem VI.45.12}
\begin{solution}
On the Rees chart where \(f_j\) is selected, all ratios \(f_i/f_j\) are regular.  Hence \([f_0:\cdots:f_r]\) becomes a regular projective-coordinate map there.  These chart maps agree on overlaps and glue to the resolved morphism.
\end{solution}
