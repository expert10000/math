% VI45-DETAILED-EXERCISE-01
\subsection*{Exercise VI.45.1}
\begin{solution}
On \(s\neq0\), put \(u=t/s\); then \(xt=ys\) becomes \(y=xu\), so the chart is \(\operatorname{Spec}k[x,u]\cong\mathbb A^2\).  On \(t\neq0\), put \(v=s/t\); then \(x=yv\), giving \(\operatorname{Spec}k[v,y]\cong\mathbb A^2\).  On the overlap \(uv=1\), so the two affine planes glue to the blow-up.
\end{solution}

% VI45-DETAILED-EXERCISE-02
\subsection*{Exercise VI.45.2}
\begin{solution}
The exceptional fiber is obtained by setting \(x=y=0\).  The incidence equation becomes \(0=0\), leaving the full homogeneous direction coordinate \([s:t]\).  Hence \(E=\{0\}\times\mathbb P^1\cong\mathbb P^1\).
\end{solution}

% VI45-DETAILED-EXERCISE-03
\subsection*{Exercise VI.45.3}
\begin{solution}
In the chart \(y=xu\), the line equation becomes \(x(u-a)=0\).  The component \(x=0\) is exceptional; after removing it, the strict transform is \(u=a\).  It meets \(E\) at \((0,a)\), the point encoding slope \(a\).
\end{solution}

% VI45-DETAILED-EXERCISE-04
\subsection*{Exercise VI.45.4}
\begin{solution}
Away from the center the blow-up is an isomorphism, so the strict transforms can meet only above the former intersection point.  On \(E\), different lines meet different direction points.  Therefore distinct lines through the center have disjoint strict transforms.
\end{solution}

% VI45-DETAILED-EXERCISE-05
\subsection*{Exercise VI.45.5}
\begin{solution}
Substitution gives \(y^2-x^3=x^2(u^2-x)\).  Thus the total transform contains \(2E\), while the strict transform is \(u^2=x\).  Its gradient \((-1,2u)\) never vanishes, so it is smooth in this chart.
\end{solution}

% VI45-DETAILED-EXERCISE-06
\subsection*{Exercise VI.45.6}
\begin{solution}
If the multiplicity is \(m\), the strict transform meets \(E\) in the degree-\(m\) tangent-cone equation \(f_m(1,u)=0\).  Its zero divisor on \(E\simeq\mathbb P^1\) has total length \(m\), so \(\widetilde C\cdot E=m\).
\end{solution}

% VI45-DETAILED-EXERCISE-07
\subsection*{Exercise VI.45.7}
\begin{solution}
The blow-up is \(\operatorname{Proj}_X\bigoplus_{n\ge0}\mathcal I^n\).  For a coherent ideal on a Noetherian scheme, the Rees algebra is finitely generated, and relative Proj of a finitely generated graded algebra is projective over the base.  Hence the blow-up morphism is projective.
\end{solution}

% VI45-DETAILED-EXERCISE-08
\subsection*{Exercise VI.45.8}
\begin{solution}
If \(\mathcal I\) is invertible, the identity \(X\to X\) satisfies the blow-up universal property and factors through \(\pi:\operatorname{Bl}_{\mathcal I}X\to X\).  Uniqueness applied again to \(\pi\) shows the two maps are inverse, so blowing up an invertible ideal is an isomorphism.
\end{solution}

% VI45-DETAILED-EXERCISE-09
\subsection*{Exercise VI.45.9}
\begin{solution}
A conic has \(d=2\), and simple passage through the center gives \(m=1\).  Thus \([\widetilde C]=2H-E\).  Its square is \(4-1=3\).
\end{solution}

% VI45-DETAILED-EXERCISE-10
\subsection*{Exercise VI.45.10}
\begin{solution}
The strict transform has class \(H-E\), so \((H-E)^2=1-1=0\).  Geometrically these curves are fibers of the pencil of lines through the blown-up point.
\end{solution}

% VI45-DETAILED-EXERCISE-11
\subsection*{Exercise VI.45.11}
\begin{solution}
Using \(K_S=-3H+E\), \(H^2=1\), \(H\cdot E=0\), \(E^2=-1\),
\[
K_S^2=9-1=8.
\]
Thus one point blow-up decreases \(K^2\) by one.
\end{solution}

% VI45-DETAILED-EXERCISE-12
\subsection*{Exercise VI.45.12}
\begin{solution}
For three distinct point blow-ups,
\[
K_S=-3H+E_1+E_2+E_3,
\]
so \(K_S^2=9-1-1-1=6\).
\end{solution}

% VI45-DETAILED-EXERCISE-13
\subsection*{Exercise VI.45.13}
\begin{solution}
\((H-E_2-E_3)^2=1-1-1=-1\).  This is the strict transform of the line through \(p_2,p_3\), one of the curves contracted by the resolved Cremona morphism.
\end{solution}

% VI45-DETAILED-EXERCISE-14
\subsection*{Exercise VI.45.14}
\begin{solution}
For \(D=2H-E_1-E_2-E_3\),
\[
D^2=4-1-1-1=1.
\]
This matches the self-intersection of a target line.
\end{solution}

% VI45-DETAILED-EXERCISE-15
\subsection*{Exercise VI.45.15}
\begin{solution}
\[
D\cdot E_1=(-E_1)\cdot E_1=1.
\]
Hence the resolved map has degree one on \(E_1\), consistent with \(E_1\) mapping isomorphically to a target coordinate line.
\end{solution}

% VI45-DETAILED-EXERCISE-16
\subsection*{Exercise VI.45.16}
\begin{solution}
\[
D\cdot(H-E_2-E_3)=2-1-1=0.
\]
A base-point-free divisor has degree zero on a contracted curve, so the strict transform of \(x=0\) is contracted.
\end{solution}

% VI45-DETAILED-EXERCISE-17
\subsection*{Exercise VI.45.17}
\begin{solution}
Applying the basis formulas a second time to \(2H-E_1-E_2-E_3\) gives
\[
2(2H-\sum E_i)-(H-E_2-E_3)-(H-E_1-E_3)-(H-E_1-E_2)=H.
\]
Thus the action squares to the identity on \(H\).
\end{solution}

% VI45-DETAILED-EXERCISE-18
\subsection*{Exercise VI.45.18}
\begin{solution}
For \((d;m_1,m_2,m_3)=(3;1,1,1)\),
\[
d'=6-3=3,\qquad m_i'=3-1-1=1.
\]
So the numerical type is fixed.
\end{solution}

% VI45-DETAILED-EXERCISE-19
\subsection*{Exercise VI.45.19}
\begin{solution}
For \((4;2,1,1)\),
\[
d'=8-4=4,\quad m_1'=4-1-1=2,\quad m_2'=m_3'=4-2-1=1.
\]
Thus the full type is again fixed.
\end{solution}

% VI45-DETAILED-EXERCISE-20
\subsection*{Exercise VI.45.20}
\begin{solution}
A point of \(E\) records a first-order tangent direction through the original center.  It is a genuine point only on the blown-up surface and can itself become a later blow-up center, so it is naturally called infinitely near to the original point.
\end{solution}

% VI45-DETAILED-EXERCISE-21
\subsection*{Exercise VI.45.21}
\begin{solution}
Each pair of coordinate lines met at one of the three vertices.  After blowing up that vertex, the two strict transforms meet different points of the exceptional divisor because their tangent directions differ.  Since they had no other intersections, the three strict transforms become pairwise disjoint.
\end{solution}

% VI45-DETAILED-EXERCISE-22
\subsection*{Exercise VI.45.22}
\begin{solution}
For \([\widetilde C]=dH-mE\),
\[
\widetilde C^2=d^2H^2+m^2E^2=d^2-m^2.
\]
A blow-up therefore lowers self-intersection by \(m^2\).
\end{solution}

% VI45-DETAILED-EXERCISE-23
\subsection*{Exercise VI.45.23}
\begin{solution}
An effective Cartier divisor has an invertible ideal sheaf.  By the universal-property argument for invertible centers, its blow-up is isomorphic to the original scheme.
\end{solution}

% VI45-DETAILED-EXERCISE-24
\subsection*{Exercise VI.45.24}
\begin{solution}
In the chart \(v=ut\),
\[
[uv:v:u]=[u^2t:ut:u]=[ut:t:1].
\]
On \(E_1:u=0\), the image is \([0:t:1]\), an affine chart of the target line \(X=0\); the second blow-up chart supplies the point at infinity.
\end{solution}
