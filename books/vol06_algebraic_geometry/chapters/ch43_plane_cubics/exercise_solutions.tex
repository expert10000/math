% VI43-DETAILED-EXERCISE-01
\subsection*{Exercise VI.43.1}
\begin{solution}
For \(C=V_+(F)\subset\mathbb P^2\), a geometric point \(P\) is singular exactly when the differential \(dF\) vanishes at \(P\).  In homogeneous coordinates this is
\[
F(P)=F_x(P)=F_y(P)=F_z(P)=0.
\]
Thus \(C\) is smooth precisely when no geometric projective point satisfies these equations.  Keeping \(F(P)=0\) explicit avoids misuse of Euler's identity in characteristic \(3\).
\end{solution}

% VI43-DETAILED-EXERCISE-02
\subsection*{Exercise VI.43.2}
\begin{solution}
The partial derivatives are \(3x^2,3y^2,3z^2\).  Since \(3\neq0\), simultaneous vanishing forces \(x=y=z=0\), which is not a projective point.  Therefore the Fermat cubic is smooth.  This argument also shows why characteristic \(3\) is exceptional: there the cubic becomes \((x+y+z)^3\).
\end{solution}

% VI43-DETAILED-EXERCISE-03
\subsection*{Exercise VI.43.3}
\begin{solution}
Over an algebraically closed field, write a reducible cubic as a union of positive-degree components.  Any two distinct components meet by Bézout.  At an intersection the local equation factors into at least two nonunits, so the gradient vanishes in the normal-crossing tangent directions and the union is singular.  A repeated component is nonreduced and is likewise singular.
\end{solution}

% VI43-DETAILED-EXERCISE-04
\subsection*{Exercise VI.43.4}
\begin{solution}
Bézout's theorem gives
\[
\sum_{P\in L\cap C} I_P(L,C)=\deg L\cdot\deg C=1\cdot3=3
\]
provided \(L\) is not a component of \(C\).  The formula counts tangent contacts and non-rational intersection points with their scheme-theoretic multiplicities.
\end{solution}

% VI43-DETAILED-EXERCISE-05
\subsection*{Exercise VI.43.5}
\begin{solution}
Let \(L=(\ell=0)\) and \(M=(m=0)\).  On \(E\), zeros of \(\ell/m\) are exactly the intersection divisor \(L\cdot E\), while poles are \(M\cdot E\).  Hence
\[
\operatorname{div}_E(\ell/m)=(L\cdot E)-(M\cdot E),
\]
which proves all line sections are linearly equivalent.
\end{solution}

% VI43-DETAILED-EXERCISE-06
\subsection*{Exercise VI.43.6}
\begin{solution}
A general line meets a smooth cubic transversely in three geometric points.  Therefore its divisor has degree three.  Since every line section represents \(H\), \(\deg H=3\) and \(\deg\mathcal O_E(1)=3\).
\end{solution}

% VI43-DETAILED-EXERCISE-07
\subsection*{Exercise VI.43.7}
\begin{solution}
At a flex \(O\), the tangent line has intersection divisor \(3O\).  But every line section represents the hyperplane divisor \(H\).  Thus \(H\sim3O\), and applying the divisor-line-bundle correspondence yields
\[
\mathcal O_E(1)\simeq\mathcal O_E(3O).
\]
\end{solution}

% VI43-DETAILED-EXERCISE-08
\subsection*{Exercise VI.43.8}
\begin{solution}
A line through \(P,Q,R\) gives \(P+Q+R\sim H\), and the flex tangent gives \(H\sim3O\).  Subtracting \(3O\),
\[
(P-O)+(Q-O)+(R-O)\sim0.
\]
Therefore the three corresponding elements of \(\mathrm{Pic}^0(E)\) sum to zero.
\end{solution}

% VI43-DETAILED-EXERCISE-09
\subsection*{Exercise VI.43.9}
\begin{solution}
If \(\mathcal O_E(P-O)\simeq\mathcal O_E(Q-O)\), then \(P-O\sim Q-O\), hence \(P\sim Q\).  Degree-one rigidity says two effective degree-one divisors on a smooth cubic can be linearly equivalent only when they are equal.  Thus \(P=Q\), proving injectivity.
\end{solution}

% VI43-DETAILED-EXERCISE-10
\subsection*{Exercise VI.43.10}
\begin{solution}
Let the first secant/tangent through \(P,Q\) have residual point \(R\).  Then \(P+Q+R\sim H\).  Let the line through \(R,O\) have residual point \(S\); then \(R+O+S\sim H\).  Subtracting,
\[
P+Q\sim O+S.
\]
Degree-one rigidity makes \(S\) unique, so this construction defines \(P\oplus Q\).
\end{solution}

% VI43-DETAILED-EXERCISE-11
\subsection*{Exercise VI.43.11}
\begin{solution}
The Abel--Picard map satisfies \(\alpha_O(O)=[O-O]=0\).  Therefore
\[
\alpha_O(P\oplus O)=\alpha_O(P)+0=\alpha_O(P).
\]
Injectivity of \(\alpha_O\) gives \(P\oplus O=P\).  The same argument gives \(O\oplus P=P\).
\end{solution}

% VI43-DETAILED-EXERCISE-12
\subsection*{Exercise VI.43.12}
\begin{solution}
By construction
\[
\alpha_O(P\oplus Q)=\alpha_O(P)+\alpha_O(Q).
\]
Tensor product in \(\mathrm{Pic}^0(E)\) is commutative, so the right-hand side is unchanged after swapping \(P,Q\).  Injectivity of \(\alpha_O\) therefore gives \(P\oplus Q=Q\oplus P\), with no appeal to a picture.
\end{solution}

% VI43-DETAILED-EXERCISE-13
\subsection*{Exercise VI.43.13}
\begin{solution}
Both parenthesizations map under \(\alpha_O\) to
\[
\alpha_O(P)+\alpha_O(Q)+\alpha_O(R).
\]
Associativity in the Picard group implies the two images agree, and injectivity then gives
\[
(P\oplus Q)\oplus R=P\oplus(Q\oplus R).
\]
This pinpoints the role of the Picard group in the geometric law.
\end{solution}

% VI43-DETAILED-EXERCISE-14
\subsection*{Exercise VI.43.14}
\begin{solution}
For a flex origin, the line through \(P\) and \(O\) has divisor \(P+O+R\sim3O\), so \(P+R\sim2O\).  Hence
\[
[R-O]=-[P-O]
\]
in \(\mathrm{Pic}^0(E)\), and \(R\) is the inverse of \(P\).
\end{solution}

% VI43-DETAILED-EXERCISE-15
\subsection*{Exercise VI.43.15}
\begin{solution}
On \(y^2=x^3+Ax+B\), the vertical line \(x=x(P)\) meets the cubic at \(P=(x,y)\), \((x,-y)\), and the point at infinity \(O\).  Collinearity therefore gives
\[
P+(x,-y)+O=O
\]
in the group law, so \(-P=(x,-y)\).
\end{solution}

% VI43-DETAILED-EXERCISE-16
\subsection*{Exercise VI.43.16}
\begin{solution}
Write the secant through \(P_1,P_2\) as \(y=m(x-x_1)+y_1\) and substitute into \(y^2=x^3+Ax+B\).  The resulting monic cubic in \(x\) has roots \(x_1,x_2,x_R\) and \(x^2\)-coefficient \(-m^2\).  Vieta gives
\[
x_1+x_2+x_R=m^2.
\]
Reflection preserves the \(x\)-coordinate, hence \(x(P_1+P_2)=m^2-x_1-x_2\).
\end{solution}

% VI43-DETAILED-EXERCISE-17
\subsection*{Exercise VI.43.17}
\begin{solution}
Differentiate \(y^2=x^3+Ax+B\):
\[
2y\,\frac{dy}{dx}=3x^2+A.
\]
At \(P=(x_1,y_1)\) with \(y_1\neq0\), the tangent slope is
\[
m=\frac{3x_1^2+A}{2y_1}.
\]
This is the limiting secant slope used in the doubling formula.
\end{solution}

% VI43-DETAILED-EXERCISE-18
\subsection*{Exercise VI.43.18}
\begin{solution}
A point has order dividing two exactly when \(P=-P\).  In short Weierstrass form \(-P=(x,-y)\), so \(P=-P\) iff \(y=-y\).  In characteristic different from two this is equivalent to \(y=0\).  Each root of \(x^3+Ax+B\) therefore gives a nontrivial two-torsion point.
\end{solution}

% VI43-DETAILED-EXERCISE-19
\subsection*{Exercise VI.43.19}
\begin{solution}
If \(P\) is a flex, its tangent section is \(3P\), while the tangent at the chosen flex origin gives \(H\sim3O\).  Since all line sections represent \(H\),
\[
3P\sim3O.
\]
Thus \(3[P-O]=0\) in \(\mathrm{Pic}^0(E)\), i.e. \(3P=O\).
\end{solution}

% VI43-DETAILED-EXERCISE-20
\subsection*{Exercise VI.43.20}
\begin{solution}
A singular affine point satisfies
\[
2y=0,\qquad 3x^2+A=0,\qquad y^2-x^3-Ax-B=0.
\]
Under the characteristic hypotheses, \(y=0\), so \(x\) is a common root of \(x^3+Ax+B\) and its derivative.  This occurs exactly when the cubic polynomial has a repeated root, equivalently when its discriminant \(-4A^3-27B^2\) vanishes.
\end{solution}

% VI43-DETAILED-EXERCISE-21
\subsection*{Exercise VI.43.21}
\begin{solution}
The secant or tangent line is defined over \(k\).  Restricting the cubic equation to that line gives a homogeneous cubic on \(\mathbb P^1_k\).  Two intersection roots, counted with multiplicity, are already \(k\)-rational, so after dividing by their linear factors the remaining factor has degree one over \(k\).  Its zero is the residual \(k\)-rational point.
\end{solution}

% VI43-DETAILED-EXERCISE-22
\subsection*{Exercise VI.43.22}
\begin{solution}
Normalization separates algebraic branches.  A node has two distinct tangent directions and hence two branches; these become two distinct points on the normalization.  A cusp is unibranch, so there is only one point above it.  This branch count is stable under completion and gives the stated normalization behavior.
\end{solution}

% VI43-DETAILED-EXERCISE-23
\subsection*{Exercise VI.43.23}
\begin{solution}
Choose an automorphism of \(\mathbb P^1\) carrying the two points to \(0\) and \(\infty\).  Their complement is
\[
\mathbb P^1\setminus\{0,\infty\}=\operatorname{Spec}k[t,t^{-1}]=\mathbb G_m.
\]
This is why the smooth locus of a nodal cubic carries a multiplicative group parameter.
\end{solution}

% VI43-DETAILED-EXERCISE-24
\subsection*{Exercise VI.43.24}
\begin{solution}
Move the single removed point to \(\infty\) by a projective automorphism.  Then
\[
\mathbb P^1\setminus\{\infty\}=\operatorname{Spec}k[t]=\mathbb A^1.
\]
With the parameter inherited from normalization, the generalized chord law becomes the additive group law of \(\mathbb G_a\).
\end{solution}
