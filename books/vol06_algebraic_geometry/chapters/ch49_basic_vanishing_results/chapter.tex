\chapter{Basic Vanishing Results}
\label{ch:vi49-basic-vanishing-results}

VI/46 introduced flabby sheaves, VI/47 made cohomology computable by \v{C}ech complexes, and VI/48 showed how exact sequences transport cohomological information.  This final chapter of Volume VI turns those tools into vanishing theorems.  The central geometric fact is that quasi-coherent sheaves on affine schemes have no higher cohomology.

\[
\boxed{
X=\operatorname{Spec}A,\ \mathcal F\ \text{quasi-coherent}
\quad\Longrightarrow\quad
H^q(X,\mathcal F)=0\ (q>0)
}
\]

Once affine vanishing is known, affine covers become computational engines.  On separated schemes their finite intersections remain affine, so the \v{C}ech groups of an affine cover compute actual sheaf cohomology.  The line-bundle calculation on $\PP^1$ from VI/47 therefore becomes a genuine sheaf-cohomology theorem rather than merely a fixed-cover calculation.

\section*{Standing conventions}
Cohomology means sheaf cohomology on the Zariski topology.  Unless stated otherwise, schemes are schemes over a fixed base and quasi-coherent sheaves are $\mathcal O_X$-modules.  We use the results on flabby resolutions, \v{C}ech comparison, and long exact sequences established in VI/46--VI/48.

\section*{Learning goals}
By the end of the chapter, the reader should be able to:
\begin{itemize}
\item recognize flabby and injective sheaves as acyclic objects;
\item prove exactness of the localization \v{C}ech complex for a principal affine cover;
\item state and use affine vanishing for quasi-coherent sheaves;
\item identify affine covers of separated schemes as Leray covers;
\item derive cohomological bounds from the number of affine opens in a cover;
\item compute the complete cohomology of $\mathcal O_{\PP^1}(n)$;
\item propagate vanishing through long exact sequences;
\item use divisor sequences and dimension shifting;
\item distinguish basic affine vanishing from deeper projective vanishing theorems.
\end{itemize}

\section{Acyclic sheaves revisited}
\begin{definition}[Acyclic sheaf]\label{def:vi49-acyclic}
A sheaf $\mathcal F$ on $X$ is called \emph{$\Gamma$-acyclic} if
\[
H^q(X,\mathcal F)=0\qquad(q>0).
\]
Acyclicity depends on the space on which the sheaf is considered; when necessary we say ``acyclic on $X$.'' 
\end{definition}

\input{chapters/ch49_basic_vanishing_results/figures/figure_02}
\begin{theorem}[Flabby vanishing]\label{thm:vi49-flabby-vanish}
Every flabby sheaf is $\Gamma$-acyclic. More generally, the restriction of a flabby sheaf to any open subset is flabby and hence acyclic.
\begin{proof}
VI/46 constructed flabby resolutions and proved that global sections applied to a flabby sheaf already have no positive derived cohomology. Restriction preserves the surjectivity of restriction maps, so the same argument applies on every open subset.
\end{proof}
\end{theorem}

\begin{corollary}[Injective vanishing]\label{cor:vi49-injective-vanish}
Every injective sheaf of abelian groups is acyclic.
\begin{proof}
Injective sheaves are flabby by VI/46, so the previous theorem applies.
\end{proof}
\end{corollary}

\begin{example}[Skyscraper sheaves]\label{ex:vi49-skyscraper-vanish}
A skyscraper sheaf $i_{x,*}A$ is flabby, hence
\[
H^q(X,i_{x,*}A)=0\qquad(q>0).
\]
Finite direct sums of skyscraper sheaves are therefore acyclic as well.
\end{example}

\begin{definition}[Affine quasi-coherent situation]\label{def:vi49-affine-qc}
Let $X=\operatorname{Spec}A$ and let $M$ be an $A$-module. We write $\widetilde M$ for the associated quasi-coherent sheaf. On a principal open $D(f)$,
\[
\Gamma(D(f),\widetilde M)\cong M_f.
\]
\end{definition}

\begin{example}[Flabby vanishing]\label{ex:vi49-expand-02}
A flabby sheaf has vanishing higher cohomology on any space.  Thus a flabby resolution computes the cohomology of an arbitrary sheaf from global sections of an acyclic complex.
\end{example}

\section{Affine vanishing}
\input{chapters/ch49_basic_vanishing_results/figures/figure_03}
\begin{proposition}[Exact localization Čech complex]\label{prop:vi49-localization-cech}
Suppose $f_1,\ldots,f_r\in A$ generate the unit ideal. For every $A$-module $M$, the augmented localization complex
\[
0\to M\to\prod_i M_{f_i}\to\prod_{i<j}M_{f_if_j}\to\cdots\to M_{f_1\cdots f_r}\to0
\]
with the alternating Čech differential is exact.
\begin{proof}
Exactness may be checked after localizing at each $f_i$, because the principal opens $D(f_i)$ cover $\operatorname{Spec}A$. After localizing at $f_i$, one member of the induced cover is the whole space, and the resulting augmented Čech complex admits the standard contracting homotopy obtained by inserting the distinguished index $i$. Hence every localized homology module vanishes. Since the $D(f_i)$ cover, the original homology modules vanish as well.
\end{proof}
\end{proposition}

\input{chapters/ch49_basic_vanishing_results/figures/figure_04}
\begin{theorem}[Affine vanishing theorem]\label{thm:vi49-affine-vanishing}
Let $X=\operatorname{Spec}A$ be affine and let $\mathcal F$ be quasi-coherent. Then
\[
\boxed{H^q(X,\mathcal F)=0\qquad(q>0).}
\]
\begin{proof}
Write $\mathcal F\cong\widetilde M$. A finite principal cover of $X$ gives the exact localization Čech complex of the preceding proposition. The standard Čech-to-derived comparison on the principal affine basis identifies this exact complex with the cohomology of $\widetilde M$. Thus all positive cohomology groups vanish.
\end{proof}
\end{theorem}

\begin{corollary}[Principal opens are acyclic]\label{cor:vi49-basic-open}
If $X=\operatorname{Spec}A$, $f\in A$, and $\mathcal F$ is quasi-coherent, then
\[
H^q(D(f),\mathcal F|_{D(f)})=0\qquad(q>0).
\]
\begin{proof}
The principal open $D(f)$ is affine and the restriction of a quasi-coherent sheaf is quasi-coherent.
\end{proof}
\end{corollary}

\begin{proposition}[Čech exactness on a principal cover]\label{prop:vi49-affine-cech}
Let $X=\operatorname{Spec}A$ and let $\mathfrak U=\{D(f_i)\}$ be a finite principal cover. For every quasi-coherent $\mathcal F$, the augmented Čech complex of $\mathfrak U$ is exact in positive degree.
\begin{proof}
If $\mathcal F=\widetilde M$, its Čech terms are precisely the localizations $M_{f_{i_0}\cdots f_{i_p}}$, so this is the localization proposition.
\end{proof}
\end{proposition}

\begin{definition}[Acyclic cover]\label{def:vi49-leray}
An open cover $\mathfrak U=\{U_i\}$ is \emph{$\mathcal F$-acyclic} (or Leray for $\mathcal F$) if every nonempty finite intersection satisfies
\[
H^q(U_{i_0\cdots i_p},\mathcal F)=0\qquad(q>0).
\]
\end{definition}

\section{Affine covers and Leray comparison}
\begin{theorem}[Affine Leray covers]\label{thm:vi49-affine-leray}
Let $X$ be a separated scheme and let $\mathfrak U=\{U_i\}$ be an affine open cover. For every quasi-coherent sheaf $\mathcal F$, the cover is Leray, and hence
\[
\check H^q(\mathfrak U,\mathcal F)\cong H^q(X,\mathcal F).
\]
\begin{proof}
On a separated scheme, every finite intersection of affine opens is affine. The restriction of $\mathcal F$ to such an intersection is quasi-coherent, so affine vanishing applies. VI/47's Leray comparison theorem now identifies Čech and sheaf cohomology.
\end{proof}
\end{theorem}

\begin{proposition}[Why separatedness matters]\label{prop:vi49-separated-intersections}
If $X$ is separated and $U,V\subseteq X$ are affine opens, then $U\cap V$ is affine.
\begin{proof}
The intersection is the inverse image of $U\times V$ under the diagonal $\Delta_X:X\to X\times X$. For a separated scheme the diagonal is a closed immersion, hence affine; therefore its inverse image over the affine scheme $U\times V$ is affine.
\end{proof}
\end{proposition}

\input{chapters/ch49_basic_vanishing_results/figures/figure_05}
\begin{corollary}[Cover-length vanishing bound]\label{cor:vi49-cover-length}
Let $X$ be separated and covered by $r$ affine opens. For every quasi-coherent sheaf $\mathcal F$ one has
\[
H^q(X,\mathcal F)=0\qquad(q\ge r).
\]
\begin{proof}
The affine cover is Leray. Its Čech complex has no terms in degrees $q\ge r$, because there are no strictly increasing $(q+1)$-tuples of indices.
\end{proof}
\end{corollary}

\begin{corollary}[Two-affine vanishing]\label{cor:vi49-two-affine}
If a separated scheme is the union of two affine opens, then every quasi-coherent sheaf $\mathcal F$ satisfies
\[
H^q(X,\mathcal F)=0\qquad(q\ge2).
\]
\end{corollary}

\begin{proposition}[Global sections of $\mathcal O_{\PP^1}(n)$]\label{prop:vi49-p1-h0}
For the projective line over a field $k$,
\[
\dim_k H^0(\PP^1,\mathcal O(n))=
\begin{cases}
n+1,&n\ge0,\\
0,&n<0.
\end{cases}
\]
For $n\ge0$, the homogeneous monomials $x_0^{n-j}x_1^j$, $0\le j\le n$, form a basis.
\begin{proof}
Use the standard affine cover $U_0\cup U_1$ and write a section as compatible polynomials in $t$ and $u=t^{-1}$. Compatibility forces a Laurent polynomial to have exponents between $0$ and $n$. Such exponents exist exactly when $n\ge0$.
\end{proof}
\end{proposition}

\section{The complete cohomology of line bundles on \texorpdfstring{$\PP^1$}{P1}}
\input{chapters/ch49_basic_vanishing_results/figures/figure_06}
\begin{theorem}[First cohomology of $\mathcal O_{\PP^1}(n)$]\label{thm:vi49-p1-h1}
For every integer $n$,
\[
H^1(\PP^1,\mathcal O(n))\cong
\frac{k[t,t^{-1}]}{k[t]+t^n k[t^{-1}]}.
\]
Consequently,
\[
\dim_k H^1(\PP^1,\mathcal O(n))=
\begin{cases}
0,&n\ge-1,\\
-n-1,&n\le-2.
\end{cases}
\]
\begin{proof}
The standard cover consists of two affine opens, and their intersection is affine, so the cover is Leray by affine vanishing. The Čech computation from VI/47 therefore computes actual sheaf cohomology. The quotient leaves precisely the Laurent monomials $t^{n+1},\ldots,t^{-1}$ when $n\le-2$.
\end{proof}
\end{theorem}

\begin{corollary}[Higher cohomology on $\PP^1$]\label{cor:vi49-p1-higher}
For every integer $n$,
\[
H^q(\PP^1,\mathcal O(n))=0\qquad(q\ge2).
\]
\begin{proof}
The projective line is separated and covered by two affine opens, so the two-affine vanishing result applies.
\end{proof}
\end{corollary}

\begin{proposition}[Euler characteristic on $\PP^1$]\label{prop:vi49-p1-euler}
For every integer $n$,
\[
\chi(\PP^1,\mathcal O(n))
=\dim H^0-\dim H^1
=n+1.
\]
\begin{proof}
Combine the two preceding formulas in the cases $n\ge0$, $n=-1$, and $n\le-2$.
\end{proof}
\end{proposition}

\begin{proposition}[Acyclic kernel and middle term]\label{prop:vi49-acyclic-quotient}
Suppose
\[
0\to\mathcal F'\to\mathcal F\to\mathcal F''\to0
\]
is exact. If $\mathcal F'$ and $\mathcal F$ are acyclic, then $\mathcal F''$ is acyclic and
\[
\Gamma(X,\mathcal F)\twoheadrightarrow\Gamma(X,\mathcal F'').
\]
\begin{proof}
Read the statement directly from the long exact cohomology sequence of VI/48.
\end{proof}
\end{proposition}

\section{Vanishing and exact sequences}
\input{chapters/ch49_basic_vanishing_results/figures/figure_07}
\begin{proposition}[Recovering acyclicity of a kernel]\label{prop:vi49-acyclic-subobject}
In the same short exact sequence, assume $\mathcal F$ and $\mathcal F''$ are acyclic and that
\[
\Gamma(X,\mathcal F)\to\Gamma(X,\mathcal F'')
\]
is surjective. Then $\mathcal F'$ is acyclic.
\begin{proof}
Surjectivity forces $H^1(X,\mathcal F')=0$. For $q\ge2$, exactness places $H^q(X,\mathcal F')$ between two zero groups coming from $\mathcal F''$ and $\mathcal F$.
\end{proof}
\end{proposition}

\begin{proposition}[Dimension shifting]\label{prop:vi49-dimension-shift}
If
\[
0\to\mathcal F'\to\mathcal I\to\mathcal F''\to0
\]
is exact and $\mathcal I$ is acyclic, then for every $q\ge1$ there are natural isomorphisms
\[
H^q(X,\mathcal F'')\cong H^{q+1}(X,\mathcal F').
\]
\begin{proof}
The relevant portion of the long exact sequence has zero middle terms $H^q(X,\mathcal I)$ and $H^{q+1}(X,\mathcal I)$.
\end{proof}
\end{proposition}

\begin{proposition}[Vanishing in the divisor sequence]\label{prop:vi49-divisor-consequence}
Let $D$ be an effective Cartier divisor on $X$ and $\mathcal L$ a line bundle. Assume
\[
H^q(D,\mathcal L|_D)=0\qquad(q>0),
\]
for example when $D$ is a finite scheme over a field. Then the exact sequence
\[
0\to\mathcal L(-D)\to\mathcal L\to i_*(\mathcal L|_D)\to0
\]
gives isomorphisms
\[
H^q(X,\mathcal L(-D))\xrightarrow{\sim}H^q(X,\mathcal L)
\qquad(q\ge2),
\]
and the low-degree exact sequence controls the difference between $H^0$ and $H^1$.
\begin{proof}
Closed-immersion pushforward preserves cohomology, so the higher cohomology of the quotient term vanishes. Apply the long exact sequence.
\end{proof}
\end{proposition}

\begin{corollary}[Point recurrence on $\PP^1$]\label{cor:vi49-p1-point}
For a $k$-rational point $p\in\PP^1$,
\[
0\to\mathcal O(n-1)\to\mathcal O(n)\to k_p\to0
\]
relates the cohomology of adjacent twists. Since $k_p$ is acyclic, this sequence recursively recovers the formulas for $H^0$ and $H^1$ of $\mathcal O(n)$.
\end{corollary}

\begin{remark}[What is not proved here: Serre vanishing]\label{remark:vi49-serre-preview}
A deeper projective theorem says that for a coherent sheaf $\mathcal F$ on a projective scheme and a sufficiently positive twist, $H^q(X,\mathcal F(n))=0$ for $q>0$ and $n\gg0$. That theorem requires substantially more projective machinery than the basic affine, Leray, and exact-sequence vanishing principles developed here.
\end{remark}

\section{Synthesis and limits of the basic theory}
\input{chapters/ch49_basic_vanishing_results/figures/figure_08}
\begin{theorem}[Part IX synthesis]\label{thm:vi49-part-ix-synthesis}
Let $X$ be a separated scheme with a finite affine open cover $\mathfrak U$ of size $r$, and let $\mathcal F$ be quasi-coherent. Then
\[
H^q(X,\mathcal F)\cong\check H^q(\mathfrak U,\mathcal F)
\]
for all $q$, and
\[
H^q(X,\mathcal F)=0\qquad(q\ge r).
\]
In particular, affine schemes have no positive quasi-coherent cohomology, while two-affine schemes have no cohomology in degrees at least two.
\begin{proof}
Finite intersections of members of $\mathfrak U$ are affine by separatedness. Affine vanishing makes the cover Leray, so VI/47 identifies Čech and sheaf cohomology. The Čech complex has length at most $r-1$.
\end{proof}
\end{theorem}

\begin{remark}[Closing Volume VI]\label{remark:vi49-volume-close}
The route from affine algebraic sets to schemes, divisors, line bundles, blow-ups, and sheaf cohomology is now complete. The final lesson is structural: local algebra supplies the sections, sheaves organize their compatibility, exact sequences measure obstructions, and vanishing theorems identify the situations in which those obstructions disappear.
\end{remark}
