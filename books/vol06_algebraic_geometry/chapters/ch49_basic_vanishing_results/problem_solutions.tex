% VI49-DETAILED-PROBLEM-01
\subsection*{Problem VI.49.1}
\begin{solution}
Let \(H^q\) be a homology module of the augmented localization Čech complex.  After localizing at \(f_i\), the induced cover contains the whole localized affine scheme as one member.  Inserting that distinguished index gives an explicit contracting homotopy, so \((H^q)_{f_i}=0\) for every \(i\).  Since the principal opens \(D(f_i)\) cover \(\operatorname{Spec}A\), the localization criterion forces \(H^q=0\).  Thus the augmented complex is exact.
\end{solution}

% VI49-DETAILED-PROBLEM-02
\subsection*{Problem VI.49.2}
\begin{solution}
Because \(X\) is separated, \(U\cap V\) is affine.  Hence the two-affine cover is Leray for quasi-coherent \(\mathcal F\).  Its Čech complex is
\[
\Gamma(U,\mathcal F)\oplus\Gamma(V,\mathcal F)\to\Gamma(U\cap V,\mathcal F),
\]
so
\[
H^1(X,\mathcal F)\cong
\frac{\Gamma(U\cap V,\mathcal F)}
{\Gamma(U,\mathcal F)|-\Gamma(V,\mathcal F)|}.
\]
\end{solution}

% VI49-DETAILED-PROBLEM-03
\subsection*{Problem VI.49.3}
\begin{solution}
The standard two-chart cover is Leray.  For \(\mathcal O(-3)\), there are no global sections, while
\[
H^1\simeq
k[t,t^{-1}]/(k[t]+t^{-3}k[t^{-1}])
=\langle t^{-2},t^{-1}\rangle_k.
\]
Thus \(H^0=0\), \(H^1\simeq k^2\), and two-affine vanishing gives \(H^q=0\) for \(q\ge2\).
\end{solution}

% VI49-DETAILED-PROBLEM-04
\subsection*{Problem VI.49.4}
\begin{solution}
For \(\mathcal O(3)\), global sections are the four cubic monomials
\[
x_0^3,\ x_0^2x_1,\ x_0x_1^2,\ x_1^3.
\]
The Laurent quotient for \(H^1\) vanishes because \(3\ge-1\), and the two-affine bound kills every \(H^q\) for \(q\ge2\).  Hence only \(H^0\simeq k^4\) survives.
\end{solution}

% VI49-DETAILED-PROBLEM-05
\subsection*{Problem VI.49.5}
\begin{solution}
The point sequence is
\[
0\to\mathcal O(n-1)\to\mathcal O(n)\to k_p\to0.
\]
Euler characteristic is additive on short exact sequences and \(\chi(k_p)=1\), so
\[
\chi(\mathcal O(n))=\chi(\mathcal O(n-1))+1.
\]
From \(\chi(\mathcal O)=1\), induction gives \(\chi(\mathcal O(n))=n+1\).
\end{solution}

% VI49-DETAILED-PROBLEM-06
\subsection*{Problem VI.49.6}
\begin{solution}
A four-member affine cover of a separated scheme is Leray.  Its Čech complex has degrees \(0,1,2,3\) and no term in degree \(4\) or above.  Consequently
\[
H^q(X,\mathcal F)=0\qquad(q\ge4)
\]
for every quasi-coherent \(\mathcal F\).  No stronger bound follows from cover length alone.
\end{solution}

% VI49-DETAILED-PROBLEM-07
\subsection*{Problem VI.49.7}
\begin{solution}
Flabbiness makes \(\mathcal I\) acyclic.  The long exact sequence contains
\[
0=H^1(\mathcal I)\to H^1(\mathcal F'')\xrightarrow{\sim}
H^2(\mathcal F')\to H^2(\mathcal I)=0.
\]
Thus the connecting homomorphism is an isomorphism, the basic dimension-shifting phenomenon.
\end{solution}

% VI49-DETAILED-PROBLEM-08
\subsection*{Problem VI.49.8}
\begin{solution}
Since \(D\) is finite, \(\mathcal L|_D\) has finite support and is acyclic.  The divisor sequence
\[
0\to\mathcal L(-D)\to\mathcal L\to\mathcal L|_D\to0
\]
therefore gives
\[
H^q(X,\mathcal L(-D))\cong H^q(X,\mathcal L)\qquad(q\ge2).
\]
On a projective curve these groups are zero once a two-affine Leray cover is available.
\end{solution}

% VI49-DETAILED-PROBLEM-09
\subsection*{Problem VI.49.9}
\begin{solution}
For a short exact sequence of quasi-coherent sheaves on affine \(X\), affine vanishing gives \(H^1(X,\mathcal F')=0\).  The beginning of the long exact sequence therefore ends
\[
\Gamma(X,\mathcal F)\twoheadrightarrow\Gamma(X,\mathcal F''),
\]
so global sections are exact.  This is the sheaf-cohomological formulation of exactness of the module functor on affine schemes.
\end{solution}

% VI49-DETAILED-PROBLEM-10
\subsection*{Problem VI.49.10}
\begin{solution}
For \(X=\operatorname{Spec}A\),
\[
D(f)=\operatorname{Spec}A_f,\qquad
\widetilde M|_{D(f)}\simeq\widetilde{M_f}.
\]
Therefore
\[
H^0(D(f),\widetilde M)=M_f,\qquad
H^q(D(f),\widetilde M)=0\ (q>0).
\]
Thus both degree-zero localization and higher vanishing are compatible with restriction to principal opens.
\end{solution}

% VI49-DETAILED-PROBLEM-11
\subsection*{Problem VI.49.11}
\begin{solution}
The standard cover of \(\mathbb P^1\) consists of two affine lines and their affine intersection \(\operatorname{Spec}k[t,t^{-1}]\).  Since \(\mathcal O(n)\) is quasi-coherent, affine vanishing makes all finite intersections acyclic.  Hence the cover is Leray, and the fixed-cover Čech groups computed in VI/47 are canonically the sheaf-cohomology groups.
\end{solution}

% VI49-DETAILED-PROBLEM-12
\subsection*{Problem VI.49.12}
\begin{solution}
The three mechanisms are: (1) flabby vanishing on arbitrary spaces for flabby sheaves; (2) affine vanishing for quasi-coherent sheaves on affine schemes; and (3) cover-length vanishing for quasi-coherent sheaves on separated schemes with finite affine covers.  Long exact sequences then transport these vanishings to kernels, quotients, twists, and divisor sequences when the required neighboring groups are zero.
\end{solution}
