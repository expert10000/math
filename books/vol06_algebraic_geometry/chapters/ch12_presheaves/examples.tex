\begin{example}[Continuous real-valued functions]
For a topological space $X$, let
\[
\mathcal C^0(U)=\{f:U\to\RR\mid f\text{ continuous}\}.
\]
Restriction is literal restriction of functions.  The identity and composition axioms are
immediate.
\end{example}

\begin{example}[Smooth functions]
If $M$ is a smooth manifold, set $\mathcal C^\infty(U)$ equal to the smooth real-valued
functions on $U$.  Smoothness is preserved by restriction, so this is a presheaf of rings.
\end{example}

\begin{example}[Holomorphic functions]
On a complex manifold or open subset $X\subseteq\CC^n$, the assignment
\[
U\longmapsto\mathcal O(U)
\]
of holomorphic functions is a presheaf of $\CC$-algebras.
\end{example}

\begin{example}[Locally constant functions]
For an abelian group $A$, let $\mathcal L_A(U)$ be the group of locally constant maps
$U\to A$.  Restriction preserves local constancy.
\end{example}

\begin{example}[Naive constant presheaf]
For nonempty $U$, put $\underline A^{pre}(U)=A$, with identity restrictions.  Put
$\underline A^{pre}(\varnothing)=0$.  This is a presheaf, but on a disconnected open it
cannot glue two different local constants to one global element.
\end{example}

\begin{example}[Power-set presheaf]
Define
\[
\mathcal P(U)=\{S:S\subseteq U\}.
\]
For $V\subseteq U$, restrict by $S\mapsto S\cap V$.  Then
$(S\cap V)\cap W=S\cap W$, so the presheaf axiom is visible as associativity of intersection.
\end{example}

\begin{example}[All functions with values in a set]
For a set $A$, define $\mathcal F(U)=A^U$, the set of all functions $U\to A$.  Restriction is
precomposition with the inclusion $V\hookrightarrow U$.
\end{example}

\begin{example}[Vector-valued continuous functions]
Fix a real vector space $V$.  Continuous maps $U\to V$ form a presheaf of real vector spaces.
This is the section presheaf of the trivial bundle $X\times V\to X$.
\end{example}

\begin{example}[Differential forms]
On a smooth manifold $M$, the assignment $U\mapsto\Omega^p(U)$ is a presheaf of real vector
spaces.  Pulling a form back along the inclusion $V\hookrightarrow U$ is exactly restriction.
\end{example}

\begin{example}[Harmonic functions]
For an open subset $X\subseteq\RR^n$, let
\[
\mathcal H(U)=\{u\in C^2(U):\Delta u=0\}.
\]
The differential equation is local and survives restriction, so $\mathcal H$ is a presheaf.
\end{example}

\begin{example}[Solutions of a first-order ODE]
On $X=\RR$, let $\mathcal S(U)$ be the smooth functions satisfying $u'=u$ on $U$.  Restriction
preserves the equation.  On each connected component the solutions are $ce^x$.
\end{example}

\begin{example}[Polynomial functions restricted from an ambient ring]
Fix $X=\RR$.  Let $\mathcal P_{\rm amb}(U)$ be the set of restrictions to $U$ of polynomials
in $\RR[x]$.  Restriction is well defined and gives a presheaf of rings.
\end{example}

\begin{example}[Rational functions without poles on the open]
On an open $U\subseteq\RR$, let $\mathcal R(U)$ be rational functions $p/q$ whose denominator
has no zero on $U$.  Shrinking $U$ can only remove possible poles, so restriction is defined.
\end{example}

\begin{example}[Bounded continuous functions]
Let $\mathcal B(U)$ be bounded continuous functions $U\to\RR$.  Restriction preserves
boundedness, so $\mathcal B$ is a presheaf.  It will later provide a clean example of a
presheaf that fails gluing on an infinite cover.
\end{example}

\begin{example}[Functions vanishing at a fixed point]
Fix $x_0\in X$.  Define
\[
\mathcal I_{x_0}(U)=
\begin{cases}
\{f\in\mathcal C^0(U):f(x_0)=0\},&x_0\in U,\\
\mathcal C^0(U),&x_0\notin U.
\end{cases}
\]
Restriction gives a subpresheaf of $\mathcal C^0$.
\end{example}

\begin{example}[Skyscraper-type presheaf]
Fix $x\in X$ and an abelian group $A$.  Set
\[
\mathcal K_x(U)=
\begin{cases}
A,&x\in U,\\
0,&x\notin U.
\end{cases}
\]
Restrictions are identity when both opens contain $x$ and zero otherwise.
\end{example}

\begin{example}[Product presheaf]
If $\mathcal F,\mathcal G$ are presheaves of abelian groups, then
\[
U\longmapsto\mathcal F(U)\times\mathcal G(U)
\]
with componentwise restriction is again a presheaf.
\end{example}

\begin{example}[Kernel presheaf]
For a morphism $\varphi:\mathcal F\to\mathcal G$, define
\[
(\ker\varphi)(U)=\ker(\varphi_U).
\]
Naturality ensures that restrictions send kernels to kernels.
\end{example}
