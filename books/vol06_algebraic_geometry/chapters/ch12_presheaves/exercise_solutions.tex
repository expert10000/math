\paragraph{VI.12.E01 --- Restriction axioms.}
For every inclusion $V\subseteq U$ there is a restriction $\rho^U_V$.  The identity inclusion gives $\rho^U_U=\id$, and the composite $W\hookrightarrow V\hookrightarrow U$ gives $\rho^V_W\rho^U_V=\rho^U_W$.  These are exactly the identity and composition axioms for a functor from $\operatorname{Open}(X)^{op}$.

\paragraph{VI.12.E02 --- Continuous functions.}
If $V\subseteq U$, send $f$ to $f|_V$.  Continuity survives restriction.  Restricting to the same set changes nothing, and $(f|_V)|_W=f|_W$.  Pointwise addition and multiplication commute with restriction.

\paragraph{VI.12.E03 --- Power sets.}
Set $\rho^U_V(S)=S\cap V$.  Then $\rho^U_U(S)=S$, and for $W\subseteq V\subseteq U$, $(S\cap V)\cap W=S\cap W$.

\paragraph{VI.12.E04 --- Naive constant presheaf.}
Put $\underline A^{pre}(U)=A$ for $U\ne\varnothing$ and $0$ for $U=\varnothing$.  Restrictions between nonempty opens are identities and all restrictions to the empty set are the unique map to $0$.

\paragraph{VI.12.E05 --- Locally constant functions.}
If $f:U\to A$ is locally constant and $V\subseteq U$, intersect a neighborhood on which $f$ is constant with $V$.  Thus $f|_V$ is locally constant.

\paragraph{VI.12.E06 --- Bounded functions.}
If $|f|\le M$ on $U$, the same bound works on every $V\subseteq U$.  Ordinary restrictions are functorial.

\paragraph{VI.12.E07 --- Compact support failure.}
Take a continuous bump $f$ with support exactly $[0,1]$ and $f>0$ on $(0,1)$.  On $V=(0,1)$ its support relative to $V$ is all of $V$, which is not compact.

\paragraph{VI.12.E08 --- Morphisms.}
For each $V\subseteq U$, naturality is $\varphi_V\circ\rho^U_V=\rho^U_V\circ\varphi_U$.

\paragraph{VI.12.E09 --- Kernel presheaf.}
If $s\in\ker\varphi_U$, then $\varphi_V(s|_V)=\varphi_U(s)|_V=0$.  Hence restrictions preserve kernels.

\paragraph{VI.12.E10 --- Image presheaf.}
If $t=\varphi_U(s)$, then $t|_V=\varphi_V(s|_V)$, so restrictions preserve objectwise images.

\paragraph{VI.12.E11 --- Quotient presheaf.}
Define $[s]|_V=[s|_V]$.  If $s-s'\in\mathcal H(U)$, then $(s-s')|_V\in\mathcal H(V)$, so the map is well defined.

\paragraph{VI.12.E12 --- Restriction to an open.}
For open $V\subseteq U$, set $(\mathcal F|_U)(V)=\mathcal F(V)$ with the same restriction maps.  Repeating the construction inside $V$ gives the same presheaf.

\paragraph{VI.12.E13 --- Direct image.}
If $W\subseteq V$, then $f^{-1}W\subseteq f^{-1}V$.  Use the restriction map of $\mathcal F$.  Identity and composition follow from inverse-image functoriality.

\paragraph{VI.12.E14 --- Composition of direct images.}
For open $W\subseteq Z$, both sides evaluate to $\mathcal F(f^{-1}(g^{-1}W))$ with identical restrictions.

\paragraph{VI.12.E15 --- Sierpinski classification.}
With normalized value $0$ on $\varnothing$, the data are $A=\mathcal F(X)$, $B=\mathcal F(\{\eta\})$, and one homomorphism $A\to B$.

\paragraph{VI.12.E16 --- Three-open chain.}
Choose $A_0,A_1,A_2$ and maps $A_0\to A_1\to A_2$; the direct restriction $A_0\to A_2$ must be their composite.

\paragraph{VI.12.E17 --- Skyscraper-type presheaf.}
If $x\in V\subseteq U$, use $\id_A$; if $x\notin V$, the target is $0$.  All composites are forced.

\paragraph{VI.12.E18 --- ODE solutions.}
Restriction preserves $u'=u$.  On each connected component every solution is $ce^x$, so a disconnected open carries one constant per component.

\paragraph{VI.12.E19 --- Singular cochains.}
The inclusion $V\hookrightarrow U$ gives $C_q(V)\to C_q(U)$; precomposition gives $C^q(U;A)\to C^q(V;A)$, and compositions reverse correctly.

\paragraph{VI.12.E20 --- Principal-open localization.}
Localizing $A_f$ further at $g$ gives $(A_f)_g\cong A_{fg}$.  A further localization at $h$ agrees with direct localization at $fgh$.

\paragraph{VI.12.E21 --- Module localization.}
The universal property of module localization gives $M_f\to M_{fg}$, and transitivity identifies $(M_f)_g$ with $M_{fg}$.

\paragraph{VI.12.E22 --- Arithmetic principal opens.}
The contravariant algebra maps are $\ZZ[1/2]\to\ZZ[1/6]\to\ZZ[1/30]$.

\paragraph{VI.12.E23 --- Presheaf but not sheaf.}
Each $s_n(x)=x$ is bounded on $(-n,n)$ and compatible.  A global glue must equal $x$ everywhere, which is unbounded.

\paragraph{VI.12.E24 --- Basis warning.}
Choose a finite space whose proper basis covers $X$ but omits $X$.  Define identical values and restrictions on the basis, then choose nonisomorphic groups at $X$ with maps to the same basis values.
