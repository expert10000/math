\begin{example}[Sierpiński space as one homomorphism]
Let $X=\{\eta,s\}$ with opens $\varnothing,\{\eta\},X$.  A normalized presheaf of abelian
groups is exactly a homomorphism
\[
A=\mathcal F(X)\longrightarrow B=\mathcal F(\{\eta\}).
\]
Thus the abstract definition reduces to a single algebraic arrow.
\end{example}

\begin{example}[A three-step finite topology]
Suppose the opens form a chain
\[
\varnothing\subset U_2\subset U_1\subset X.
\]
A normalized presheaf is a diagram
\[
\mathcal F(X)\longrightarrow\mathcal F(U_1)\longrightarrow\mathcal F(U_2),
\]
with the direct restriction $X\to U_2$ forced to be the composite.
\end{example}

\begin{example}[Bounded functions: presheaf but not sheaf]
Cover $\RR$ by $U_n=(-n,n)$ and take $s_n(x)=x$.  Each $s_n$ is bounded on $U_n$ and the
family is compatible.  The only possible global glue is $s(x)=x$, which is unbounded on
$\RR$.  Hence bounded continuous functions form a presheaf but fail the gluing axiom.

\end{example}

\begin{example}[Compact support can fail restriction]
Let $U=\RR$ and choose a continuous function supported on $[0,1]$ and nonzero on $(0,1)$.
Restrict to $V=(0,1)$.  Its support inside $V$ is all of $(0,1)$, which is not compact.
Therefore ``compactly supported continuous functions'' is not a presheaf under ordinary
restriction.
\end{example}

\begin{example}[Singular cochains]
For an abelian group $A$, define
\[
\mathcal C^q(U;A)=\Hom(C_q(U),A).
\]
An inclusion $V\subseteq U$ induces $C_q(V)\to C_q(U)$, hence by precomposition a restriction
\[
\mathcal C^q(U;A)\longrightarrow\mathcal C^q(V;A).
\]
This is a presheaf of abelian groups.
\end{example}

\begin{example}[Direct image]
For $f:X\to Y$ and continuous functions $\mathcal C^0_X$, the presheaf
\[
(f_*\mathcal C^0_X)(V)=\mathcal C^0(f^{-1}V)
\]
records functions on inverse images of opens in $Y$.
\end{example}

\begin{example}[Localization on principal opens]
For $X=\Spec A$, define on the basis of principal opens
\[
\mathcal O^{pre}(D(f))=A_f.
\]
For $D(fg)\subseteq D(f)$, restriction is $A_f\to A_{fg}$.  The functoriality identity is
the transitivity of localization.
\end{example}

\begin{example}[Module localization]
If $M$ is an $A$-module, the analogous basis assignment
\[
\widetilde M^{pre}(D(f))=M_f
\]
has restrictions $M_f\to M_{fg}$.  This is the prototype for quasi-coherent sheaves.
\end{example}

\begin{example}[Arithmetic localization]
On $X=\Spec\ZZ$, the principal-open prototype assigns
\[
D(n)\longmapsto \ZZ[1/n].
\]
If $D(mn)\subseteq D(n)$, the restriction map is
\[
\ZZ[1/n]\longrightarrow\ZZ[1/mn].
\]
\end{example}

\begin{example}[Objectwise image versus local image]
Given a morphism of presheaves $\varphi:\mathcal F\to\mathcal G$, the assignment
\[
U\mapsto\operatorname{im}(\varphi_U)
\]
is always a presheaf.  Later, for sheaves, this objectwise image may be too small because a
section can be locally in the image without being globally in the image.
\end{example}

\begin{example}[Representable presheaf on opens]
Fix an open $U_0\subseteq X$.  Define
\[
h_{U_0}(V)=\Hom_{\operatorname{Open}(X)}(V,U_0).
\]
Because $\operatorname{Open}(X)$ is a poset, this is a singleton when $V\subseteq U_0$ and
empty otherwise.  It is a presheaf of sets and is the simplest entry point to the Yoneda
viewpoint.
\end{example}

\begin{example}[Agreement on a basis need not determine a presheaf]
Let $X$ have a basis $\mathcal B$ that does not contain $X$ itself.  One may define two
presheaves that agree on every $B\in\mathcal B$ and on all restrictions among basis opens,
but assign different objects to $X$ with restriction maps landing in the same basis data.
Without a locality axiom, the basis data do not force the global value.
\end{example}
