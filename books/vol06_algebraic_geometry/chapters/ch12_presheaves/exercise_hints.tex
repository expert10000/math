\paragraph{VI.12.E01.} Translate inclusions into arrows in the opposite category.

\paragraph{VI.12.E02.} Restriction of a continuous function is continuous; ring operations are pointwise.

\paragraph{VI.12.E03.} Use $(S\cap V)\cap W=S\cap W$.

\paragraph{VI.12.E04.} Use $0$ on the empty set so there is a unique restriction to $\varnothing$.

\paragraph{VI.12.E05.} Local constancy is preserved after shrinking the domain.

\paragraph{VI.12.E06.} A restriction of a bounded function remains bounded by the same bound.

\paragraph{VI.12.E07.} Choose a bump function positive on $(0,1)$ and zero outside $[0,1]$.

\paragraph{VI.12.E08.} Compare $\varphi_V\circ\rho^U_V$ with $\rho^U_V\circ\varphi_U$.

\paragraph{VI.12.E09.} Naturality sends an element killed by $\varphi_U$ to one killed by $\varphi_V$.

\paragraph{VI.12.E10.} Restrict a representative $\varphi_U(s)$.

\paragraph{VI.12.E11.} Use $[s]\mapsto[s|_V]$ and check well-definedness.

\paragraph{VI.12.E12.} The open subsets of $U$ are already open in $X$ because $U$ is open.

\paragraph{VI.12.E13.} Inverse images preserve inclusions and identities.

\paragraph{VI.12.E14.} Use $(g\circ f)^{-1}(W)=f^{-1}(g^{-1}(W))$.

\paragraph{VI.12.E15.} The only nontrivial inclusion of nonempty opens is $\{\eta\}\subseteq X$.

\paragraph{VI.12.E16.} Composition forces the long restriction to equal the composite of the two short ones.

\paragraph{VI.12.E17.} Check separately whether $x$ belongs to the source and target opens.

\paragraph{VI.12.E18.} On each connected component, every solution is $ce^x$.

\paragraph{VI.12.E19.} An inclusion induces chains forward and therefore cochains backward by precomposition.

\paragraph{VI.12.E20.} Invert $g$ after $f$ is already inverted.

\paragraph{VI.12.E21.} Use the universal property of module localization.

\paragraph{VI.12.E22.} Write the rings as $\ZZ[1/2]\to\ZZ[1/6]\to\ZZ[1/30]$.

\paragraph{VI.12.E23.} The only possible glue is $x\mapsto x$, which is not bounded on $\RR$.

\paragraph{VI.12.E24.} Let the basis omit the whole space, then alter only the object assigned to the whole space.
