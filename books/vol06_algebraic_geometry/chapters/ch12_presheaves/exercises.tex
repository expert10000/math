\begin{exercise}[VI.12.E01 --- Restriction axioms]
Write the presheaf axioms explicitly for a chain $W\subseteq V\subseteq U$ and explain why they are precisely functoriality on $\operatorname{Open}(X)^{op}$.
\end{exercise}

\begin{exercise}[VI.12.E02 --- Continuous functions]
Verify directly that $U\mapsto C^0(U,\RR)$ is a presheaf of rings.
\end{exercise}

\begin{exercise}[VI.12.E03 --- Power sets]
Verify that $U\mapsto\mathcal P(U)$ with $S\mapsto S\cap V$ is a presheaf of sets.
\end{exercise}

\begin{exercise}[VI.12.E04 --- Naive constant presheaf]
Define the naive constant presheaf of an abelian group $A$, including its value on $\varnothing$, and verify the axioms.
\end{exercise}

\begin{exercise}[VI.12.E05 --- Locally constant functions]
Show that locally constant $A$-valued functions form a presheaf.
\end{exercise}

\begin{exercise}[VI.12.E06 --- Bounded functions]
Show bounded continuous functions form a presheaf.
\end{exercise}

\begin{exercise}[VI.12.E07 --- Compact support failure]
Give an explicit continuous compactly supported function on $\RR$ whose restriction to $(0,1)$ is not compactly supported there.
\end{exercise}

\begin{exercise}[VI.12.E08 --- Morphisms]
For presheaves of rings, write the naturality square for a morphism $\varphi:\mathcal F\to\mathcal G$.
\end{exercise}

\begin{exercise}[VI.12.E09 --- Kernel presheaf]
Prove $U\mapsto\ker(\varphi_U)$ is a subpresheaf of $\mathcal F$.
\end{exercise}

\begin{exercise}[VI.12.E10 --- Image presheaf]
Prove $U\mapsto\operatorname{im}(\varphi_U)$ is a presheaf.
\end{exercise}

\begin{exercise}[VI.12.E11 --- Quotient presheaf]
If $\mathcal H\subseteq\mathcal F$ are presheaves of abelian groups, construct the quotient presheaf sectionwise.
\end{exercise}

\begin{exercise}[VI.12.E12 --- Restriction to an open]
For $j:U\hookrightarrow X$, prove $\mathcal F|_U$ is a presheaf and that $(\mathcal F|_U)|_V=\mathcal F|_V$ for $V\subseteq U$.
\end{exercise}

\begin{exercise}[VI.12.E13 --- Direct image]
For a continuous map $f:X\to Y$, verify that $f_*\mathcal F(V)=\mathcal F(f^{-1}V)$ defines a presheaf.
\end{exercise}

\begin{exercise}[VI.12.E14 --- Composition of direct images]
For $X\xrightarrow fY\xrightarrow gZ$, prove $(g\circ f)_*\mathcal F=g_*(f_*\mathcal F)$.
\end{exercise}

\begin{exercise}[VI.12.E15 --- Sierpinski classification]
Classify normalized presheaves of abelian groups on the Sierpiński space in terms of one group homomorphism.
\end{exercise}

\begin{exercise}[VI.12.E16 --- Three-open chain]
Describe presheaves on $\varnothing\subset U_2\subset U_1\subset X$ as composable homomorphisms.
\end{exercise}

\begin{exercise}[VI.12.E17 --- Skyscraper-type presheaf]
Verify the fixed-point assignment $\mathcal K_x(U)=A$ if $x\in U$ and $0$ otherwise is a presheaf.
\end{exercise}

\begin{exercise}[VI.12.E18 --- ODE solutions]
Show solutions of $u'=u$ form a presheaf and compute sections on a disconnected open subset of $\RR$.
\end{exercise}

\begin{exercise}[VI.12.E19 --- Singular cochains]
Show $U\mapsto C^q(U;A)$ is contravariant under inclusions.
\end{exercise}

\begin{exercise}[VI.12.E20 --- Principal-open localization]
For $D(fg)\subseteq D(f)$, construct the restriction $A_f\to A_{fg}$ and verify transitivity for $D(fgh)\subseteq D(fg)\subseteq D(f)$.
\end{exercise}

\begin{exercise}[VI.12.E21 --- Module localization]
Repeat the preceding exercise for an $A$-module $M$ and the maps $M_f\to M_{fg}$.
\end{exercise}

\begin{exercise}[VI.12.E22 --- Arithmetic principal opens]
On $\Spec\ZZ$, compute the maps attached to $D(30)\subseteq D(6)\subseteq D(2)$.
\end{exercise}

\begin{exercise}[VI.12.E23 --- Presheaf but not sheaf]
Use $U_n=(-n,n)$ and $s_n(x)=x$ to prove bounded continuous functions fail the gluing axiom on $\RR$.
\end{exercise}

\begin{exercise}[VI.12.E24 --- Basis warning]
Construct two presheaves on a finite topological space that agree on a proper basis but differ on the whole space.
\end{exercise}
