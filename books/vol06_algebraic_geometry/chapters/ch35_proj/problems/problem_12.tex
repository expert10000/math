% VI35-DETAILED-PROBLEM-12
\begin{problem}[VI.35.P12: Boundary to projective morphisms]\label{prob:vi35-12}
Explain why a graded map \(S\to T\) can fail to define a morphism \(\Proj T\to\Proj S\) everywhere, and formulate the obstruction in terms of irrelevant ideals.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
\varphi:S\to T
\]
be degree preserving. If
\[
\mathfrak q\in\Proj T,
\]
then its contraction
\[
\mathfrak p=\varphi^{-1}(\mathfrak q)
\]
is a homogeneous prime of \(S\). But it is a point of \(\Proj S\) only if
\[
S_+\not\subseteq\mathfrak p.
\]
Failure occurs exactly when
\[
\varphi(S_+)\subseteq\mathfrak q,
\]
equivalently when
\[
\varphi(S_+)T\subseteq\mathfrak q.
\]

Hence contraction is defined projectively precisely on the open
\[
\boxed{
U
=
\Proj T\setminus V_+\!\bigl(\varphi(S_+)T\bigr).
}
\]
If
\[
\sqrt{\varphi(S_+)T}\supseteq T_+,
\]
then no point of \(\Proj T\) contains \(\varphi(S_+)T\), so \(U=\Proj T\) and contraction is everywhere defined.
Without such a hypothesis one obtains only a morphism on the indicated open subset. VI/38 develops the systematic
functorial theory.
\end{solution}
\fi
