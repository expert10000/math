% VI35-DETAILED-PROBLEM-09
\begin{problem}[VI.35.P9: Homogeneous quotient topology]\label{prob:vi35-09}
For homogeneous \(I\), prove that the natural prime correspondence identifies
\[
\Proj(S/I)\longrightarrow \Proj S
\]
homeomorphically with \(V_+(I)\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Prime ideals of \(S/I\) correspond to prime ideals of \(S\) containing \(I\), and homogeneity is preserved under
this correspondence. The irrelevant ideal of the quotient is
\[
(S/I)_+=(S_++I)/I.
\]
Thus a homogeneous prime \(\bar{\mathfrak p}\subseteq S/I\) contains \((S/I)_+\) exactly when its inverse image
\(\mathfrak p\subseteq S\) contains \(S_+\). Therefore projectively relevant primes correspond exactly to
\[
\mathfrak p\in V_+(I).
\]
This gives a bijection
\[
\Proj(S/I)\xrightarrow{\sim}V_+(I).
\]

Closed sets in the source are defined by homogeneous ideals \(\bar J\subseteq S/I\), corresponding to homogeneous
ideals \(J\subseteq S\) containing \(I\). Under the bijection,
\[
V_+(\bar J)
\longmapsto
V_+(J)\cap V_+(I)=V_+(J).
\]
Thus closed sets correspond, and the bijection is a homeomorphism:
\[
\boxed{\Proj(S/I)\cong V_+(I)}.
\]
\end{solution}
\fi
