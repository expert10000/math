% VI35-DETAILED-PROBLEM-18
\begin{problem}[VI.35.P18: Saturation is invisible on every standard chart]\label{prob:vi35-18}
Let \(I\subseteq S\) be homogeneous and set
\[
I^{\mathrm{sat}}=I:S_+^\infty.
\]
Prove that for every homogeneous \(f\in S_+\),
\[
(I^{\mathrm{sat}})_f=I_f,
\]
and deduce equality of the corresponding \(V_+\)-closed sets.

\textbf{Provenance.} Canonical VI/35 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The inclusion
\[
I_f\subseteq(I^{\mathrm{sat}})_f
\]
is immediate. Conversely, let
\[
h\in I^{\mathrm{sat}}.
\]
Then for some \(N\),
\[
hS_+^N\subseteq I.
\]
Since \(f^N\in S_+^N\),
\[
hf^N\in I.
\]
After localizing at \(f\), this gives
\[
h/1\in I_f.
\]
Hence
\[
\boxed{(I^{\mathrm{sat}})_f=I_f}.
\]

Now let \(\mathfrak p\in\Proj S\). Choose homogeneous \(f\in S_+\setminus\mathfrak p\). On the chart \(D_+(f)\),
the ideals \(I\) and \(I^{\mathrm{sat}}\) have the same localization. Thus
\[
I\subseteq\mathfrak p
\iff
I^{\mathrm{sat}}\subseteq\mathfrak p.
\]
Therefore
\[
\boxed{V_+(I)=V_+(I^{\mathrm{sat}})}.
\]
Challenge C1 analyzes the stronger scheme-theoretic and converse saturation statements.
\end{solution}
\fi
