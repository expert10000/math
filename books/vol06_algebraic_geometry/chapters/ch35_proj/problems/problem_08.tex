% VI35-DETAILED-PROBLEM-08
\begin{problem}[VI.35.P8: Reducible and nonreduced Proj]\label{prob:vi35-08}
Compare \(\Proj k[x,y]/(xy)\) and \(\Proj k[x,y]/(y^2)\) as topological spaces and schemes.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
For
\[
S_1=k[x,y]/(xy),
\]
the projectively relevant homogeneous primes are \((x)\) and \((y)\). On \(D_+(x)\), invertibility of \(x\) forces
\(y=0\), so
\[
(S_1)_{(x)}\cong k.
\]
Similarly
\[
(S_1)_{(y)}\cong k.
\]
The overlap is empty because \(xy=0\). Therefore
\[
\boxed{\Proj S_1\cong\Spec k\sqcup\Spec k}
\]
is two reduced points.

For
\[
S_2=k[x,y]/(y^2),
\]
every prime contains the nilpotent element \(y\). The only projectively relevant homogeneous prime is \((y)\), so
the underlying space has one point. On the \(x\)-chart,
\[
t=y/x
\]
has degree zero and satisfies \(t^2=0\). Hence
\[
(S_2)_{(x)}\cong k[t]/(t^2).
\]
Thus
\[
\boxed{\Proj S_2\cong\Spec k[t]/(t^2)}
\]
is one nonreduced point. The comparison shows that Proj remembers scheme structure, not merely homogeneous-prime
incidence.
\end{solution}
\fi
