% VI35-DETAILED-PROBLEM-03
\begin{problem}[VI.35.P3: Structure-sheaf gluing]\label{prob:vi35-03}
Using VI/20, prove that the affine schemes \(\Spec S_{(f)}\) glue along the standard overlaps to a scheme structure on \(\Proj S\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The standard opens \(D_+(f)\) cover \(\Proj S\). By P2 each has an affine model
\[
D_+(f)\cong\Spec S_{(f)}.
\]

Consider two homogeneous elements \(f,g\). Choose positive powers \(f^a,g^b\) of equal degree. Then VI/34 gives
\[
(S_{(f)})_{g^b/f^a}
\cong
S_{(f^ag^b)}
\cong
(S_{(g)})_{f^a/g^b}.
\]
Thus the overlap \(D_+(f)\cap D_+(g)\) is identified from both sides with the same affine scheme
\[
\Spec S_{(f^ag^b)}.
\]
These overlap isomorphisms are induced by localization maps. Localization in stages is associative, so on triple
overlaps the two possible composites agree automatically.

Therefore the affine schemes \(\Spec S_{(f)}\), together with these overlap isomorphisms, satisfy the cocycle
conditions of the scheme-gluing theorem from VI/20. They glue to a scheme whose underlying topological space is
the previously defined \(\Proj S\). Its structure sheaf restricts on each standard chart to the affine structure
sheaf of \(\Spec S_{(f)}\).
\end{solution}
\fi
