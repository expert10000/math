% VI35-DETAILED-PROBLEM-14
\begin{problem}[VI.35.P14: Kernels and cokernels after projective sheafification]\label{prob:vi35-14}
Let \(u:M\to N\) be a degree-zero homomorphism of graded \(S\)-modules. Prove
\[
\widetilde{\ker u}\cong\ker(\widetilde u),
\qquad
\widetilde{\operatorname{coker}u}\cong\operatorname{coker}(\widetilde u).
\]

\textbf{Provenance.} Canonical VI/35 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The graded module sequences
\[
0\to\ker u\to M\xrightarrow{u}N
\]
and
\[
M\xrightarrow{u}N\to\operatorname{coker}u\to0
\]
are exact, and all maps have degree zero. By P6, projective sheafification is exact on such sequences. Hence
\[
0\to\widetilde{\ker u}\to\widetilde M\xrightarrow{\widetilde u}\widetilde N
\]
is exact, so
\[
\widetilde{\ker u}\cong\ker(\widetilde u).
\]
Similarly,
\[
\widetilde M\xrightarrow{\widetilde u}\widetilde N
\to
\widetilde{\operatorname{coker}u}\to0
\]
is exact, giving
\[
\widetilde{\operatorname{coker}u}
\cong
\operatorname{coker}(\widetilde u).
\]
Therefore
\[
\boxed{
\widetilde{\ker u}\cong\ker(\widetilde u),
\qquad
\widetilde{\operatorname{coker}u}\cong\operatorname{coker}(\widetilde u).
}
\]
The same argument identifies the sheaf image with the sheafification of the graded image.
\end{solution}
\fi
