% VI35-DETAILED-PROBLEM-13
\begin{problem}[VI.35.P13: Quotient charts]\label{prob:vi35-13}
Let \(I\subseteq S\) be homogeneous and let \(f\in S_+\) be homogeneous. Prove
\[
(S/I)_{(f)}
\cong
S_{(f)}/(I_f)_0.
\]
Interpret this as the chart-level compatibility between \(\Proj(S/I)\) and \(V_+(I)\).

\textbf{Provenance.} Canonical VI/35 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Localization commutes with quotient:
\[
(S/I)_f
\cong
S_f/I_f.
\]
Because \(I_f\) is a graded ideal of the graded localization \(S_f\), taking degree zero gives
\[
\bigl((S/I)_f\bigr)_0
\cong
(S_f/I_f)_0.
\]
For a graded quotient, the degree-zero piece is the quotient of degree-zero pieces:
\[
(S_f/I_f)_0
\cong
(S_f)_0/(I_f)_0.
\]
Therefore
\[
\boxed{
(S/I)_{(f)}
\cong
S_{(f)}/(I_f)_0.
}
\]

Under the affine chart theorem,
\[
D_+(\bar f)\subseteq\Proj(S/I)
\]
is therefore identified with
\[
\Spec\bigl(S_{(f)}/(I_f)_0\bigr),
\]
while
\[
D_+(f)\subseteq\Proj S
\]
is \(\Spec S_{(f)}\). Thus the quotient chart is cut out inside the ambient chart by the localized degree-zero
ideal \((I_f)_0\). VI/37 develops the global closed-subscheme interpretation; VI/35 records only this chart-level
compatibility.
\end{solution}
\fi
