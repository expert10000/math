% VI35-DETAILED-PROBLEM-04
\begin{problem}[VI.35.P4: Stalk computation]\label{prob:vi35-04}
Prove \(\OO_{\Proj S,\mathfrak p}\cong S_{(\mathfrak p)}\) and identify its maximal ideal and residue field.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Choose a homogeneous \(f\notin\mathfrak p\). Under the affine chart isomorphism
\[
D_+(f)\cong\Spec S_{(f)},
\]
the point \(\mathfrak p\) corresponds to
\[
\mathfrak q=(\mathfrak pS_f)_0.
\]
Hence
\[
\OO_{\Proj S,\mathfrak p}
\cong
(S_{(f)})_{\mathfrak q}.
\]

An element of this local ring is represented by a quotient of degree-zero homogeneous fractions. Clearing the
nested denominators gives a balanced homogeneous fraction
\[
\frac ab
\]
with \(a,b\) homogeneous of equal degree and \(b\notin\mathfrak p\). Conversely every such fraction appears in
some standard neighborhood. Thus
\[
\boxed{\OO_{\Proj S,\mathfrak p}\cong S_{(\mathfrak p)}}.
\]

Its nonunits are exactly the fractions whose numerator lies in \(\mathfrak p\):
\[
\boxed{
\mathfrak m_{\mathfrak p}
=
\left\{\frac ab:\ a\in\mathfrak p,\ b\notin\mathfrak p,\ \deg a=\deg b\right\}.
}
\]
Indeed, if \(a\notin\mathfrak p\), then \(b/a\) is an inverse. Therefore the residue field is
\[
\boxed{
\kappa(\mathfrak p)
=
S_{(\mathfrak p)}/\mathfrak m_{\mathfrak p}.
}
\]
\end{solution}
\fi
