% VI35-DETAILED-PROBLEM-01
\begin{problem}[VI.35.P1: The standard-open basis]\label{prob:vi35-01}
Prove from the definitions that the sets \(D_+(f)\) form a basis and give an explicit basis refinement inside an arbitrary \(V_+(I)^c\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
U=V_+(I)^c
\]
and let \(\mathfrak p\in U\). Since \(\mathfrak p\notin V_+(I)\), we have
\[
I\not\subseteq\mathfrak p.
\]
Because \(I\) is homogeneous, it is generated by its homogeneous elements, so there exists a homogeneous
\[
f\in I\setminus\mathfrak p.
\]
Then
\[
\mathfrak p\in D_+(f).
\]
Moreover, if \(\mathfrak q\in D_+(f)\), then \(f\notin\mathfrak q\). Since \(f\in I\), the ideal \(I\) cannot be
contained in \(\mathfrak q\). Hence
\[
D_+(f)\subseteq V_+(I)^c.
\]
This explicitly refines any projective open neighborhood of \(\mathfrak p\) by a standard open.

Finally, standard opens are stable under finite intersections:
\[
D_+(f)\cap D_+(g)=D_+(fg).
\]
Therefore the family of \(D_+(f)\), for homogeneous positive-degree \(f\), is a basis for the topology on
\(\Proj S\).
\end{solution}
\fi
