% VI35-DETAILED-PROBLEM-10
\begin{problem}[VI.35.P10: Veronese invariance on charts]\label{prob:vi35-10}
Give a chartwise proof that \(\Proj S\cong\Proj S^{(d)}\) in the standard setting.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
S^{(d)}=\bigoplus_{n\ge0}S_{nd}.
\]
If \(f\in S_{md}\), then \(f\) has degree \(m\) in the regraded Veronese ring. A degree-zero fraction in
\((S^{(d)})_f\) has the form
\[
a/f^r
\]
with
\[
a\in S_{rmd},
\]
which is exactly the degree-zero condition in \(S_f\). Therefore
\[
\boxed{
(S^{(d)})_{(f)}\cong S_{(f)}.
}
\]

Because \(S\) is standard graded, the opens indexed by homogeneous elements whose degrees are divisible by \(d\)
cover \(\Proj S\): if \(g\) is any positive-degree homogeneous element, then
\[
D_+(g)=D_+(g^d),
\qquad
g^d\in S^{(d)}.
\]
Thus both Proj schemes have matching affine covers with identical chart rings.

On overlaps the identifications are induced by the identity on balanced fractions, hence commute with localization.
They satisfy the cocycle condition and glue to a scheme isomorphism
\[
\boxed{\Proj S\cong\Proj S^{(d)}}.
\]
\end{solution}
\fi
