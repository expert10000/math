% VI35-DETAILED-PROBLEM-07
\begin{problem}[VI.35.P7: A two-chart prototype]\label{prob:vi35-07}
For \(S=k[x,y]\), compute both standard charts and their overlap, including the transition \(t=y/x\leftrightarrow t^{-1}=x/y\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
On
\[
D_+(x)
\]
the degree-zero chart ring is
\[
S_{(x)}=k[y/x].
\]
Writing
\[
t=y/x,
\]
this chart is \(\Spec k[t]\).

Similarly
\[
S_{(y)}=k[x/y].
\]
Writing
\[
u=x/y,
\]
the second chart is \(\Spec k[u]\).

The overlap is
\[
D_+(x)\cap D_+(y)=D_+(xy).
\]
On the first chart this means \(t\neq0\), so the overlap ring is
\[
k[t,t^{-1}].
\]
On the second it is
\[
k[u,u^{-1}].
\]
The transition is
\[
u=\frac xy=\left(\frac yx\right)^{-1}=t^{-1}.
\]
Thus the gluing is
\[
\boxed{
\Spec k[t]\ \cup_{\Spec k[t,t^{-1}]}\ \Spec k[u],
\qquad u=t^{-1}.
}
\]
VI/36 names and develops this scheme as \(\mathbb P^1_k\).
\end{solution}
\fi
