% VI35-DETAILED-PROBLEM-06
\begin{problem}[VI.35.P6: Exactness of the tilde construction]\label{prob:vi35-06}
Prove that a short exact sequence of graded modules with degree-zero maps remains exact after applying \(M\mapsto\widetilde M\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let
\[
0\to M'\to M\to M''\to0
\]
be exact with degree-zero maps. For any homogeneous \(f\), ordinary localization is exact:
\[
0\to M'_f\to M_f\to M''_f\to0.
\]
The localized maps are still degree zero. Exactness of graded modules is degreewise, so taking degree zero gives
\[
0\to M'_{(f)}\to M_{(f)}\to M''_{(f)}\to0.
\]

On the affine chart \(D_+(f)\), VI/33 says the corresponding affine tilde functor is exact. Hence
\[
0\to\widetilde{M'}|_{D_+(f)}
\to\widetilde M|_{D_+(f)}
\to\widetilde{M''}|_{D_+(f)}
\to0
\]
is exact for every standard chart. Since the standard opens cover \(\Proj S\), equivalently since the sequence is
exact on every stalk,
\[
\boxed{
0\to\widetilde{M'}\to\widetilde M\to\widetilde{M''}\to0
}
\]
is exact globally.
\end{solution}
\fi
