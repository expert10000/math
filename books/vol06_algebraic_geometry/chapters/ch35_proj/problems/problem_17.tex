% VI35-DETAILED-PROBLEM-17
\begin{problem}[VI.35.P17: Weighted chart rings for weights \(1,2,3\)]\label{prob:vi35-17}
Let
\[
S=k[x,y,z],
\qquad
\deg x=1,\quad \deg y=2,\quad \deg z=3.
\]
Compute \(S_{(x)},S_{(y)},S_{(z)}\).

\textbf{Provenance.} Canonical VI/35 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
On the \(x\)-chart, the degree-zero generators are
\[
a=\frac y{x^2},
\qquad
b=\frac z{x^3}.
\]
Every balanced monomial is a polynomial in \(a,b\), so
\[
\boxed{S_{(x)}\cong k[a,b]}.
\]

On the \(y\)-chart, a numerator over \(y^r\) has weighted degree \(2r\). The semigroup of pairs \((i,k)\) for
monomials \(x^iz^k\) satisfying
\[
i+3k\equiv0\pmod2
\]
is generated by
\[
(2,0),\quad(1,1),\quad(0,2).
\]
Set
\[
u=\frac{x^2}{y},
\qquad
v=\frac{xz}{y^2},
\qquad
w=\frac{z^2}{y^3}.
\]
They satisfy
\[
uw=v^2,
\]
and generate every degree-zero monomial. Hence
\[
\boxed{
S_{(y)}
\cong
k[u,v,w]/(uw-v^2).
}
\]

On the \(z\)-chart, the condition is
\[
i+2j\equiv0\pmod3.
\]
The semigroup is generated by
\[
(3,0),\quad(1,1),\quad(0,3).
\]
Set
\[
r=\frac{x^3}{z},
\qquad
s=\frac{xy}{z},
\qquad
t=\frac{y^3}{z^2}.
\]
Then
\[
rt=s^3,
\]
and these generate the degree-zero ring. Therefore
\[
\boxed{
S_{(z)}
\cong
k[r,s,t]/(rt-s^3).
}
\]
Weighted Proj can thus have singular affine charts even when the graded ring itself is polynomial.
\end{solution}
\fi
