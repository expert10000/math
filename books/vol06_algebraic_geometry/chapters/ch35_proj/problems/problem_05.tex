% VI35-DETAILED-PROBLEM-05
\begin{problem}[VI.35.P5: Graded modules become quasi-coherent sheaves]\label{prob:vi35-05}
Construct \(\widetilde M\) from the modules \(M_{(f)}\) and prove quasi-coherence.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
For every standard open
\[
D_+(f)\cong\Spec S_{(f)},
\]
VI/34 supplies the \(S_{(f)}\)-module
\[
M_{(f)}=(M_f)_0.
\]
By VI/33, this determines a quasi-coherent sheaf
\[
\widetilde{M_{(f)}}^{\,\mathrm{aff}}
\]
on the affine chart.

On an overlap \(D_+(f)\cap D_+(g)\), choose equal-degree powers \(f^a,g^b\). VI/34 gives the canonical module
localization identity
\[
(M_{(f)})_{g^b/f^a}
\cong
M_{(f^ag^b)}
\cong
(M_{(g)})_{f^a/g^b}.
\]
Therefore the affine module sheaves agree on the overlap. The identifications satisfy the cocycle condition because
they come from localization in stages.

The affine sheaves thus glue to a sheaf \(\widetilde M\) on \(\Proj S\). Since quasi-coherence is affine-local and
each restriction is quasi-coherent,
\[
\boxed{\widetilde M\text{ is quasi-coherent}.}
\]
By construction,
\[
\boxed{\Gamma(D_+(f),\widetilde M)=M_{(f)}}.
\]
\end{solution}
\fi
