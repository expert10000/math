% VI35-DETAILED-PROBLEM-11
\begin{problem}[VI.35.P11: Why the Spec sheaf restriction is wrong]\label{prob:vi35-11}
Give a concrete example showing that \(\OO_{\Spec S}(D(f))=S_f\) is too large for projective geometry, while \(S_{(f)}\) has the correct scaling-invariant functions.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Take
\[
S=k[x,y],
\qquad f=x.
\]
The ordinary affine localization is
\[
S_x=k[x,x^{-1},y].
\]
It contains \(x\), \(x^{-1}\), \(y\), and homogeneous fractions of every integer degree. For example, under common
scaling
\[
(x,y)\mapsto(\lambda x,\lambda y),
\]
the function \(x\) changes by the factor \(\lambda\), while \(x^{-1}\) changes by \(\lambda^{-1}\). Such elements
cannot define functions depending only on the projective direction.

The degree-zero subring is generated by
\[
t=y/x.
\]
Indeed every balanced fraction \(F/x^n\), with \(F\) homogeneous of degree \(n\), is a polynomial in \(y/x\).
Thus
\[
\boxed{S_{(x)}=k[y/x]}.
\]
The ratio \(y/x\) is unchanged by common scaling, so this is exactly the correct affine chart ring. Therefore the
projective structure sheaf cannot be the restriction of the affine structure sheaf from \(\Spec S\).
\end{solution}
\fi
