% VI35-DETAILED-PROBLEM-02
\begin{problem}[VI.35.P2: The affine chart correspondence]\label{prob:vi35-02}
Construct explicitly the prime correspondence in \(D_+(f)\cong\Spec S_{(f)}\) and verify that it respects basic opens.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Let \(f\) be homogeneous of positive degree. If
\[
\mathfrak p\in D_+(f),
\]
then \(f\notin\mathfrak p\), so localization gives the homogeneous prime
\[
\mathfrak pS_f\subseteq S_f.
\]
Its degree-zero part
\[
\mathfrak q=(\mathfrak pS_f)_0
\]
is a prime ideal of
\[
S_{(f)}=(S_f)_0.
\]

Conversely, let \(\mathfrak q\subseteq S_{(f)}\) be prime. Inside the graded ring \(S_f\), define a homogeneous
ideal degreewise by
\[
Q_n=\{u\in(S_f)_n:\ u f^r\in \mathfrak q
\text{ after choosing }r\text{ with }\deg(uf^r)=0\},
\]
or equivalently take the unique homogeneous prime of \(S_f\) whose degree-zero part is \(\mathfrak q\). Contract it
to \(S\):
\[
\mathfrak p=Q\cap S.
\]
Then \(\mathfrak p\) is homogeneous and prime, and \(f\notin\mathfrak p\) because \(f\) is invertible in \(S_f\).
Thus \(\mathfrak p\in D_+(f)\). The two operations are inverse.

To compare topologies, let \(g\) be homogeneous. Choose positive integers \(a,b\) such that
\[
a\deg f=b\deg g.
\]
Then
\[
u=\frac{g^b}{f^a}\in S_{(f)}.
\]
For the corresponding primes,
\[
g\notin\mathfrak p
\iff
u\notin\mathfrak q.
\]
Hence the projective standard open inside \(D_+(f)\) corresponds to the ordinary distinguished open
\[
D(u)\subseteq\Spec S_{(f)}.
\]
Thus the bijection is a homeomorphism and respects the standard affine bases.
\end{solution}
\fi
