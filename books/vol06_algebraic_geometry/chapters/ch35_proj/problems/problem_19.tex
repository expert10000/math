% VI35-DETAILED-PROBLEM-19
\begin{problem}[VI.35.P19: Local dimension of the polynomial Proj charts]\label{prob:vi35-19}
Let
\[
S=k[x_0,\ldots,x_n]
\]
with standard grading. Show every standard chart \(D_+(x_i)\) is an integral \(n\)-dimensional affine scheme.

\textbf{Provenance.} Canonical VI/35 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The VI/34 chart calculation gives
\[
S_{(x_i)}
\cong
k\!\left[
\frac{x_0}{x_i},\ldots,
\widehat{\frac{x_i}{x_i}},
\ldots,\frac{x_n}{x_i}
\right]
\cong
k[t_1,\ldots,t_n].
\]
Therefore
\[
D_+(x_i)
\cong
\Spec k[t_1,\ldots,t_n].
\]
The polynomial ring is a domain, so the chart is integral. Its Krull dimension is
\[
n.
\]
Hence
\[
\boxed{D_+(x_i)\text{ is integral of dimension }n}.
\]
This proves the local affine dimension statement needed before VI/36 develops projective \(n\)-space globally.
\end{solution}
\fi
