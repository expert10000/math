% VI35 detailed exercise solutions -- VI/24+ proof-depth standard.

% VI35-DETAILED-EXERCISE-01
\paragraph{Solution VI.35.E1.}
For the standard grading on
\[
S=k[x,y,z],
\]
the irrelevant ideal is
\[
S_+=(x,y,z).
\]
A point of \(\Proj S\) is a homogeneous prime \(\mathfrak p\) satisfying
\[
S_+\not\subseteq\mathfrak p.
\]
But the maximal ideal \((x,y,z)\) contains \(S_+\) exactly. Therefore
\[
\boxed{(x,y,z)\notin\Proj S}.
\]
Geometrically this is the affine-cone vertex, which projectivization discards.

% VI35-DETAILED-EXERCISE-02
\paragraph{Solution VI.35.E2.}
The ideal
\[
(x)\subset k[x,y]
\]
is homogeneous. Its quotient is
\[
k[x,y]/(x)\cong k[y],
\]
which is a domain, so \((x)\) is prime. The irrelevant ideal is
\[
S_+=(x,y).
\]
Since \(y\notin(x)\), we have \(S_+\not\subseteq(x)\). Hence
\[
\boxed{(x)\in\Proj k[x,y]}.
\]

% VI35-DETAILED-EXERCISE-03
\paragraph{Solution VI.35.E3.}
A point \(\mathfrak p\in\Proj S\) lies in
\[
V_+(I)\cup V_+(J)
\]
exactly when \(I\subseteq\mathfrak p\) or \(J\subseteq\mathfrak p\). Since \(\mathfrak p\) is prime, this is
equivalent to
\[
IJ\subseteq\mathfrak p.
\]
Therefore
\[
\boxed{V_+(I)\cup V_+(J)=V_+(IJ)}.
\]

% VI35-DETAILED-EXERCISE-04
\paragraph{Solution VI.35.E4.}
A point \(\mathfrak p\) lies in every \(V_+(I_\alpha)\) if and only if
\[
I_\alpha\subseteq\mathfrak p
\]
for every \(\alpha\). This is equivalent to the containment of the ideal generated by all the \(I_\alpha\):
\[
\sum_\alpha I_\alpha\subseteq\mathfrak p.
\]
Hence
\[
\boxed{
\bigcap_\alpha V_+(I_\alpha)
=
V_+\!\left(\sum_\alpha I_\alpha\right).
}
\]

% VI35-DETAILED-EXERCISE-05
\paragraph{Solution VI.35.E5.}
For a prime ideal \(\mathfrak p\),
\[
fg\notin\mathfrak p
\iff
f\notin\mathfrak p\text{ and }g\notin\mathfrak p.
\]
Therefore
\[
\mathfrak p\in D_+(fg)
\iff
\mathfrak p\in D_+(f)\cap D_+(g).
\]
Thus
\[
\boxed{D_+(f)\cap D_+(g)=D_+(fg)}.
\]

% VI35-DETAILED-EXERCISE-06
\paragraph{Solution VI.35.E6.}
Let
\[
S_+=(f_0,\ldots,f_r)
\]
with each \(f_i\) homogeneous. If a point \(\mathfrak p\in\Proj S\) lay outside every \(D_+(f_i)\), then
\[
f_i\in\mathfrak p
\]
for all \(i\). Hence \(S_+\subseteq\mathfrak p\), contradicting the definition of \(\Proj S\). Therefore every
point lies in at least one standard open:
\[
\boxed{\Proj S=\bigcup_iD_+(f_i)}.
\]

% VI35-DETAILED-EXERCISE-07
\paragraph{Solution VI.35.E7.}
For \(S=k[x]\) with \(\deg x=1\), the homogeneous prime ideals are \((0)\) and \((x)\). Since
\[
S_+=(x),
\]
the prime \((x)\) is excluded, while \((0)\) remains. Thus \(\Proj S\) has one point.

The unique standard open is
\[
D_+(x)=\Proj S.
\]
Its chart ring is
\[
S_{(x)}=(k[x]_x)_0=k,
\]
because a degree-zero fraction is \(a x^n/x^n=a\). Hence
\[
\boxed{\Proj k[x]\cong\Spec k}
\]
as schemes, with structure sheaf the usual structure sheaf of the one-point affine scheme.

% VI35-DETAILED-EXERCISE-08
\paragraph{Solution VI.35.E8.}
Let
\[
S=k[\varepsilon]/(\varepsilon^3),
\qquad
\deg\varepsilon=1.
\]
The nilpotent element \(\varepsilon\) lies in every prime ideal. Thus every prime contains
\[
S_+=(\varepsilon).
\]
No homogeneous prime survives the Proj condition, so
\[
\boxed{\Proj S=\varnothing}.
\]

% VI35-DETAILED-EXERCISE-09
\paragraph{Solution VI.35.E9.}
In
\[
S=k[x,y,z]
\]
with standard grading, invert \(x\). The degree-zero generators are
\[
u=\frac yx,
\qquad
v=\frac zx.
\]
Every degree-zero fraction \(F/x^d\), with \(F\) homogeneous of degree \(d\), becomes
\[
F(1,u,v).
\]
Therefore
\[
\boxed{S_{(x)}\cong k[u,v]\cong k[y/x,z/x]}.
\]

% VI35-DETAILED-EXERCISE-10
\paragraph{Solution VI.35.E10.}
For \(S=k[x,y]\),
\[
D_+(x)\cap D_+(y)=D_+(xy).
\]
On the \(x\)-chart set
\[
t=\frac yx.
\]
The overlap is obtained by requiring \(y\neq0\), hence \(t\neq0\). Therefore its ring is
\[
k[t]_t=k[t,t^{-1}].
\]
Equivalently, on the \(y\)-chart the coordinate \(u=x/y\) satisfies \(u=t^{-1}\). Thus
\[
\boxed{\Gamma(D_+(x)\cap D_+(y),\OO)=k[t,t^{-1}]}.
\]

% VI35-DETAILED-EXERCISE-11
\paragraph{Solution VI.35.E11.}
Homogeneous prime ideals of \(S/I\) correspond to homogeneous primes \(\mathfrak p\subseteq S\) containing \(I\).
Under this correspondence, the irrelevant ideal of the quotient is
\[
(S/I)_+=(S_++I)/I.
\]
A quotient prime contains \((S/I)_+\) exactly when its inverse image contains \(S_+\). Thus projectively relevant
primes correspond precisely to points
\[
\mathfrak p\in\Proj S
\]
with \(I\subseteq\mathfrak p\). Therefore
\[
\boxed{\Proj(S/I)\cong V_+(I)}
\]
as sets, and the same prime correspondence is the topological one used later.

% VI35-DETAILED-EXERCISE-12
\paragraph{Solution VI.35.E12.}
The homogeneous local ring
\[
S_{(\mathfrak p)}
\]
is obtained by inverting homogeneous elements outside \(\mathfrak p\) and then taking degree zero. Hence its
elements are represented by fractions
\[
\frac ab
\]
with \(a,b\) homogeneous,
\[
\deg a=\deg b,
\qquad
b\notin\mathfrak p.
\]
Two fractions are identified by the ordinary localization relation. Thus
\[
\boxed{
S_{(\mathfrak p)}
=
\left\{\frac ab:\ a,b\text{ homogeneous of equal degree},\ b\notin\mathfrak p\right\}/\sim.
}
\]

% VI35-DETAILED-EXERCISE-13
\paragraph{Solution VI.35.E13.}
Let
\[
\mathfrak m
=
\left\{\frac ab\in S_{(\mathfrak p)}:\ a\in\mathfrak p\right\}.
\]
It is an ideal because \(\mathfrak p\) is an ideal and denominators lie outside \(\mathfrak p\).

Now take \(a/b\notin\mathfrak m\). Then \(a\notin\mathfrak p\), so
\[
\frac ba
\]
is a valid degree-zero fraction in \(S_{(\mathfrak p)}\), and
\[
\frac ab\frac ba=1.
\]
Thus every element outside \(\mathfrak m\) is a unit. Therefore \(\mathfrak m\) is the unique maximal ideal:
\[
\boxed{
\mathfrak m_{\mathfrak p}
=
\left\{\frac ab:\ a\in\mathfrak p\right\}.
}
\]

% VI35-DETAILED-EXERCISE-14
\paragraph{Solution VI.35.E14.}
By definition of the sheaf associated to a graded module,
\[
\widetilde M|_{D_+(f)}
\]
is the affine quasi-coherent sheaf associated with the \(S_{(f)}\)-module
\[
M_{(f)}=(M_f)_0.
\]
Therefore
\[
\boxed{\Gamma(D_+(f),\widetilde M)=M_{(f)}=(M_f)_0}.
\]

% VI35-DETAILED-EXERCISE-15
\paragraph{Solution VI.35.E15.}
Choose a homogeneous \(f\notin\mathfrak p\). Inside the affine chart
\[
D_+(f)\cong\Spec S_{(f)},
\]
the sheaf \(\widetilde M\) is the affine module sheaf associated with \(M_{(f)}\). Its stalk at the prime
corresponding to \(\mathfrak p\) is the ordinary localization of \(M_{(f)}\).

Unwinding that localization permits exactly homogeneous fractions
\[
\frac mb
\]
with numerator and denominator of matching degree and \(b\notin\mathfrak p\). This is the homogeneous
degree-zero localization \(M_{(\mathfrak p)}\). Hence
\[
\boxed{\widetilde M_{\mathfrak p}\cong M_{(\mathfrak p)}}.
\]

% VI35-DETAILED-EXERCISE-16
\paragraph{Solution VI.35.E16.}
For the graded module \(M=S\), the local module on a standard chart is
\[
S_{(f)}.
\]
Thus
\[
\Gamma(D_+(f),\widetilde S)=S_{(f)}.
\]
But the structure sheaf of \(\Proj S\) was defined by the same affine chart formula:
\[
\Gamma(D_+(f),\OO_{\Proj S})=S_{(f)}.
\]
The restriction maps on overlaps are the same homogeneous localization maps. Since the standard opens form a
basis,
\[
\boxed{\widetilde S\cong\OO_{\Proj S}}.
\]

% VI35-DETAILED-EXERCISE-17
\paragraph{Solution VI.35.E17.}
Let
\[
0\to M'\to M\to M''\to0
\]
be exact with degree-zero maps. VI/34 proves that for every homogeneous \(f\),
\[
0\to M'_{(f)}\to M_{(f)}\to M''_{(f)}\to0
\]
is exact. On the affine chart \(D_+(f)\), VI/33 turns this into an exact sequence of quasi-coherent module sheaves.
Therefore the associated sheaf sequence is exact on a basis of affine opens, equivalently on all stalks:
\[
\boxed{
0\to\widetilde{M'}\to\widetilde M\to\widetilde{M''}\to0.
}
\]

% VI35-DETAILED-EXERCISE-18
\paragraph{Solution VI.35.E18.}
By definition,
\[
\OO(d)=\widetilde{S(d)}.
\]
Hence on a standard open
\[
D_+(f)
\]
its sections are
\[
\boxed{
\Gamma(D_+(f),\OO(d))
=
S(d)_{(f)}
=
(S_f)_d.
}
\]
The last equality is the shift-localization identity proved in VI/34.

% VI35-DETAILED-EXERCISE-19
\paragraph{Solution VI.35.E19.}
Let
\[
S=k[x,y]/(xy)
\]
with standard grading. The homogeneous primes that avoid the irrelevant ideal are
\[
(x)
\quad\text{and}\quad
(y).
\]
The prime \((x,y)\) is excluded because it contains \(S_+\).

On \(D_+(x)\), the relation \(xy=0\) and invertibility of \(x\) force \(y=0\), so
\[
S_{(x)}\cong k.
\]
Similarly
\[
S_{(y)}\cong k.
\]
Their intersection is empty because
\[
D_+(xy)=\varnothing
\]
in the quotient. Therefore
\[
\boxed{
\Proj k[x,y]/(xy)\cong\Spec k\sqcup\Spec k.
}
\]

% VI35-DETAILED-EXERCISE-20
\paragraph{Solution VI.35.E20.}
Let
\[
S=k[x,y]/(y^2).
\]
On the \(x\)-chart put
\[
t=\frac yx.
\]
This has degree zero. Dividing the relation \(y^2=0\) by \(x^2\) gives
\[
t^2=0.
\]
Every degree-zero fraction is a polynomial in \(t\), so
\[
\boxed{
S_{(x)}\cong k[t]/(t^2).
}
\]
Thus the chart has one underlying point but nontrivial nilpotent structure.

% VI35-DETAILED-EXERCISE-21
\paragraph{Solution VI.35.E21.}
For
\[
S=k[x,y],
\]
the \(x\)-chart has ring
\[
S_{(x)}=k[y/x].
\]
The second Veronese is
\[
S^{(2)}=k[x^2,xy,y^2].
\]
On its \(x^2\)-chart, a degree-zero generator is
\[
\frac{xy}{x^2}=\frac yx,
\]
and
\[
\frac{y^2}{x^2}=\left(\frac yx\right)^2.
\]
Thus
\[
\boxed{
(S^{(2)})_{(x^2)}
=
k[y/x]
=
S_{(x)}.
}
\]
This is the local chart form of Veronese invariance.

% VI35-DETAILED-EXERCISE-22
\paragraph{Solution VI.35.E22.}
In standard \(k[x,y]\),
\[
\deg x=1,
\qquad
\deg y^2=2.
\]
Thus
\[
x+y^2
\]
is not homogeneous. Under common scaling,
\[
x+y^2
\longmapsto
\lambda x+\lambda^2y^2,
\]
which is not a single scalar multiple of the original expression. Therefore its vanishing is not naturally a
condition on scalar-equivalence classes. Hence it is not a valid homogeneous projective equation.

% VI35-DETAILED-EXERCISE-23
\paragraph{Solution VI.35.E23.}
For a homogeneous ideal \(I\),
\[
V_+(I)=\Proj S\cap V_{\Spec S}(I),
\]
so every \(V_+(I)\) is closed in the subspace topology.

Conversely, let
\[
C=\Proj S\cap V(J)
\]
be a subspace-closed set for an arbitrary ideal \(J\subseteq S\). Let \(J^{\mathrm h}\) be the ideal generated by
all homogeneous components of all elements of \(J\). If \(\mathfrak p\) is homogeneous, then
\[
J\subseteq\mathfrak p
\iff
J^{\mathrm h}\subseteq\mathfrak p
\]
by the homogeneous component criterion. Therefore
\[
C=V_+(J^{\mathrm h}).
\]
Thus the \(V_+\)-topology is exactly the subspace topology inherited from \(\Spec S\).

% VI35-DETAILED-EXERCISE-24
\paragraph{Solution VI.35.E24.}
For
\[
S=k[x_0,\ldots,x_n]
\]
with standard grading, the standard opens
\[
D_+(x_i)
\]
cover \(\Proj S\). Their coordinate rings are
\[
S_{(x_i)}
\cong
k\!\left[
\frac{x_0}{x_i},\ldots,
\widehat{\frac{x_i}{x_i}},
\ldots,\frac{x_n}{x_i}
\right]
\cong
k[t_1,\ldots,t_n].
\]
Hence every standard chart is affine \(n\)-space. VI/36 develops their global interpretation as projective space.

% VI35-DETAILED-EXERCISE-25
\paragraph{Solution VI.35.E25.}
Assume
\[
S_+^NM=0.
\]
Let \(f\in S_+\) be homogeneous. Then
\[
f^N\in S_+^N
\]
and therefore
\[
f^NM=0.
\]
After localizing at \(f\), the element \(f^N\) becomes a unit, so
\[
M_f=0.
\]
Consequently
\[
M_{(f)}=(M_f)_0=0
\]
on every standard open. Since standard opens cover \(\Proj S\),
\[
\boxed{\widetilde M=0}.
\]
Thus module information supported purely in the irrelevant direction disappears under projective sheafification.

% VI35-DETAILED-EXERCISE-26
\paragraph{Solution VI.35.E26.}
Let
\[
\varphi:S\to T
\]
be degree preserving and let \(\mathfrak q\in\Proj T\). Its contraction
\[
\varphi^{-1}(\mathfrak q)
\]
is a homogeneous prime of \(S\). It defines a point of \(\Proj S\) exactly when it does not contain \(S_+\).
Equivalently, there exists a homogeneous \(f\in S_+\) with
\[
f\notin\varphi^{-1}(\mathfrak q),
\]
that is,
\[
\varphi(f)\notin\mathfrak q.
\]
Hence the allowable points are precisely
\[
\bigcup_{f\in S_+\text{ homogeneous}}D_+(\varphi(f)).
\]
A point lies outside this union exactly when it contains the ideal generated by all \(\varphi(f)\), namely
\[
\varphi(S_+)T.
\]
Therefore
\[
\boxed{
U
=
\Proj T\setminus V_+\!\bigl(\varphi(S_+)T\bigr).
}
\]
