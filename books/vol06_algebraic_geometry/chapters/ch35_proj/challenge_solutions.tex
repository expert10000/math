% VI35-DETAILED-CHALLENGE-01
\subsection*{Challenge VI.35.C1 solution}
\begin{solution}
Let
\[
I^{\mathrm{sat}}
=
I:S_+^\infty
=
\bigcup_{m\ge0}(I:S_+^m).
\]
VI/34 already shows this is homogeneous.  We first compare projective closed sets.

Take
\[
\mathfrak p\in\Proj S.
\]
Choose a homogeneous element
\[
f\in S_+\setminus\mathfrak p.
\]
VI/34 gives
\[
(I^{\mathrm{sat}})_f=I_f.
\]
Therefore
\[
I\subseteq\mathfrak p
\iff
I_f\subseteq\mathfrak pS_f
\iff
(I^{\mathrm{sat}})_f\subseteq\mathfrak pS_f
\iff
I^{\mathrm{sat}}\subseteq\mathfrak p.
\]
Hence
\[
\boxed{V_+(I)=V_+(I^{\mathrm{sat}})}.
\]

Now compare the projective quotient schemes chartwise.  On a standard chart,
\[
D_+(f)\cap\Proj(S/I)
\cong
\Spec (S/I)_{(f)}.
\]
By the quotient-chart formula,
\[
(S/I)_{(f)}
\cong
S_{(f)}/(I_f)_0.
\]
Since
\[
I_f=(I^{\mathrm{sat}})_f,
\]
the same ring is obtained from \(I^{\mathrm{sat}}\):
\[
(S/I)_{(f)}
\cong
(S/I^{\mathrm{sat}})_{(f)}.
\]
The identifications commute with overlap localization, so the quotient Proj schemes are canonically isomorphic:
\[
\boxed{
\Proj(S/I)\cong\Proj(S/I^{\mathrm{sat}})
}
\]
through the identity on the common standard charts.

Thus replacing one homogeneous ideal by its saturation does not alter the closed projective subscheme produced by
the Proj quotient construction.  No extra Noetherian argument is needed for this forward statement beyond the
homogeneous saturation/localization formulas.

A stronger converse requires finiteness hypotheses: if two homogeneous ideals define the same closed projective
subscheme, when can one recover equality of their saturations?  In the standard finitely generated Noetherian
setting, equality of the ideal sheaves on a finite standard cover allows denominators to be cleared uniformly, and
one obtains
\[
I^{\mathrm{sat}}=J^{\mathrm{sat}}.
\]
This is the familiar reason saturated homogeneous ideals become the correct global algebraic representatives of
closed projective subschemes.  VI/37 develops that correspondence systematically.
\end{solution}

% VI35-DETAILED-CHALLENGE-02
\subsection*{Challenge VI.35.C2 solution}
\begin{solution}
For every \(n\in\mathbb Z\), the graded piece \(S_n\) maps naturally into chart sections of
\[
\OO(n)=\widetilde{S(n)}.
\]
Indeed a homogeneous element \(s\in S_n\) gives, on \(D_+(f)\), the degree-zero localized element represented by
\(s\) in
\[
S(n)_{(f)}=(S_f)_n.
\]
These local sections are compatible, producing
\[
\alpha_n:S_n\longrightarrow\Gamma(\Proj S,\OO(n)).
\]

There is no unconditional theorem saying \(\alpha_n\) is always an isomorphism.

\textbf{Failure for negative degree.}
Take
\[
S=k[x],
\qquad
\deg x=1.
\]
Then
\[
\Proj S\cong\Spec k
\]
is one point.  On its only chart,
\[
\OO(n)(\Proj S)
=
S(n)_{(x)}
=
(S_x)_n.
\]
Since \(x\) is invertible in \(S_x=k[x,x^{-1}]\),
\[
(S_x)_n=kx^n
\cong k
\]
for every \(n\in\mathbb Z\).  Thus
\[
\Gamma(\Proj S,\OO(-1))\cong k,
\]
whereas
\[
S_{-1}=0.
\]
Therefore
\[
\alpha_{-1}:0\to k
\]
is not surjective.

\textbf{Failure of injectivity even in positive degree.}
Take
\[
S=k[x,y]/(x^2,xy)
\]
with standard grading.  The class of \(x\) is nonzero in \(S_1\), but
\[
S_+x=0
\]
because both \(x^2\) and \(xy\) vanish.  Hence the graded submodule \(kx\subseteq S\) is irrelevant torsion and
sheafifies to zero.  Consequently the global section defined by \(x\) in \(\OO(1)\) is zero:
\[
\alpha_1(x)=0.
\]
Thus
\[
S_1\longrightarrow\Gamma(\Proj S,\OO(1))
\]
need not be injective.

Under stronger hypotheses, especially for polynomial rings and later sufficiently positive twists of suitable
finitely generated graded modules, powerful reconstruction theorems recover graded pieces from global sections.
But those results require more than the definition of Proj itself.  VI/35 therefore correctly treats
\[
S_n\to\Gamma(\Proj S,\OO(n))
\]
as a natural map, not as an unconditional isomorphism.
\end{solution}

% VI35-DETAILED-CHALLENGE-03
\subsection*{Challenge VI.35.C3 solution}
\begin{solution}
Let
\[
S=k[x,y,z],
\qquad
\deg x=1,\quad \deg y=2,\quad \deg z=3.
\]

\textbf{The \(x\)-chart.}
Because \(x\) has degree \(1\), the basic degree-zero generators are
\[
a=\frac y{x^2},
\qquad
b=\frac z{x^3}.
\]
Every balanced monomial fraction is a polynomial in \(a,b\). Hence
\[
\boxed{S_{(x)}\cong k[a,b]}.
\]

\textbf{The \(y\)-chart.}
A degree-zero monomial over a power of \(y\) is controlled by pairs \((i,k)\) with
\[
i+3k\equiv0\pmod2.
\]
This semigroup is generated by
\[
(2,0),\qquad(1,1),\qquad(0,2).
\]
Define
\[
u=\frac{x^2}{y},
\qquad
v=\frac{xz}{y^2},
\qquad
w=\frac{z^2}{y^3}.
\]
Then
\[
uw=v^2,
\]
and these generators produce every degree-zero monomial. Therefore
\[
\boxed{
S_{(y)}
\cong
k[u,v,w]/(uw-v^2).
}
\]

\textbf{The \(z\)-chart.}
Now the degree condition is
\[
i+2j\equiv0\pmod3.
\]
The semigroup is generated by
\[
(3,0),\qquad(1,1),\qquad(0,3).
\]
Set
\[
r=\frac{x^3}{z},
\qquad
s=\frac{xy}{z},
\qquad
t=\frac{y^3}{z^2}.
\]
They satisfy
\[
rt=s^3.
\]
Thus
\[
\boxed{
S_{(z)}
\cong
k[r,s,t]/(rt-s^3).
}
\]

Only the weight-one chart is ordinary affine plane.  The weight-two and weight-three charts are singular affine
surface charts.  This is the first concrete indication that weighted projective geometry is locally different from
ordinary projective space even though both arise from the same Proj machine.
\end{solution}

% VI35-DETAILED-CHALLENGE-04
\subsection*{Challenge VI.35.C4 solution}
\begin{solution}
Let
\[
\varphi:S\to T
\]
be degree preserving.  For
\[
\mathfrak q\in\Proj T,
\]
the contraction
\[
\varphi^{-1}(\mathfrak q)
\]
is a homogeneous prime of \(S\).  It defines a point of \(\Proj S\) exactly when
\[
S_+\not\subseteq\varphi^{-1}(\mathfrak q).
\]
Equivalently, there is some homogeneous \(f\in S_+\) with
\[
\varphi(f)\notin\mathfrak q.
\]
Therefore the largest possible point-set domain for contraction is
\[
\boxed{
U
=
\bigcup_{f\in S_+\text{ homogeneous}}
D_+(\varphi(f))
=
\Proj T\setminus V_+\!\bigl(\varphi(S_+)T\bigr).
}
\]

On the open \(D_+(\varphi(f))\subseteq U\), the map \(\varphi\) induces
\[
S_f\longrightarrow T_{\varphi(f)}.
\]
Because \(\varphi\) preserves degrees, degree zero maps to degree zero:
\[
S_{(f)}
\longrightarrow
T_{(\varphi(f))}.
\]
Contravariance of \(\Spec\) yields
\[
D_+(\varphi(f))
\longrightarrow
D_+(f).
\]
Its underlying map sends \(\mathfrak q\) to \(\varphi^{-1}(\mathfrak q)\).

If \(f\) and \(g\) are both allowed at a point, the two chart morphisms agree on the overlap: after passing to
equal-degree powers, both are induced by the same localized map
\[
S_{(f^ag^b)}
\longrightarrow
T_{(\varphi(f)^a\varphi(g)^b)}.
\]
Hence the chart morphisms glue to
\[
\boxed{U\longrightarrow\Proj S}.
\]

Finally, if \(\mathfrak q\notin U\), then
\[
\varphi(S_+)T\subseteq\mathfrak q,
\]
and therefore
\[
S_+\subseteq\varphi^{-1}(\mathfrak q).
\]
The contracted prime is not projectively relevant.  Thus no contraction-defined extension can include such a
point, proving maximality.

In particular the construction is everywhere defined precisely when
\[
V_+\!\bigl(\varphi(S_+)T\bigr)=\varnothing,
\]
for example when
\[
\sqrt{\varphi(S_+)T}\supseteq T_+.
\]
\end{solution}

% VI35-DETAILED-CHALLENGE-05
\subsection*{Challenge VI.35.C5 solution}
\begin{solution}
Let
\[
S=k[x_0,\ldots,x_n]
\]
with standard grading.  VI/35 gives the standard cover
\[
\Proj S=\bigcup_{i=0}^nD_+(x_i).
\]

On \(D_+(x_i)\),
\[
S_{(x_i)}
\cong
k\!\left[
\frac{x_0}{x_i},
\ldots,
\widehat{\frac{x_i}{x_i}},
\ldots,
\frac{x_n}{x_i}
\right]
\cong
k[t_1,\ldots,t_n].
\]
Hence every chart is affine \(n\)-space.

On the overlap \(D_+(x_i)\cap D_+(x_j)\), the coordinate
\[
\frac{x_j}{x_i}
\]
is invertible.  The overlap ring from the \(i\)-chart is obtained by localizing at this ratio.  From the \(j\)-chart
one localizes at \(x_i/x_j\).  The transition sends
\[
\frac{x_i}{x_j}
=
\left(\frac{x_j}{x_i}\right)^{-1},
\]
and for \(r\neq i,j\),
\[
\frac{x_r}{x_j}
=
\frac{x_r/x_i}{x_j/x_i}.
\]
These are the complete affine transition formulas.

Now compute \(k\)-valued points from the atlas.  A \(k\)-point in \(D_+(x_i)\) is specified by values
\[
a_j/a_i\in k
\qquad(j\neq i),
\]
which one may normalize by choosing \(a_i=1\).  On an overlap, changing from the \(i\)-normalization to the
\(j\)-normalization divides all coordinates by the same nonzero scalar \(a_j/a_i\).

Thus the chart-gluing equivalence relation identifies nonzero tuples
\[
(a_0,\ldots,a_n)\in k^{n+1}\setminus\{0\}
\]
that differ by multiplication by an element of \(k^\times\).  Equivalently, the \(k\)-points are the
one-dimensional \(k\)-subspaces of \(k^{n+1}\).

This reconstruction uses only the general Proj machine and its affine transitions.  VI/36 begins from this result
and develops homogeneous coordinates, projective subspaces, and the global geometry systematically.
\end{solution}
