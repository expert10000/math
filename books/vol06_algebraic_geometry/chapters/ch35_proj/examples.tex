\begin{example}[Proj of a field in degree zero]\label{ex:vi35-01}
If \(S=k\) is concentrated in degree zero, then \(S_+=0\). Every prime contains \(S_+\), so \(\Proj S=\varnothing\).
\end{example}

\begin{example}[One standard generator]\label{ex:vi35-02}
For \(S=k[x]\), \(\Proj S=\{(0)\}\) and \(D_+(x)=\Proj S\cong\Spec k\).
\end{example}

\begin{example}[Two-variable chart]\label{ex:vi35-03}
For \(S=k[x,y]\), \(S_{(x)}=k[y/x]\), so \(D_+(x)\cong\mathbb A^1_k\).  Similarly \(D_+(y)\cong\mathbb A^1_k\).
\end{example}

\begin{example}[The overlap]\label{ex:vi35-04}
On \(D_+(x)\cap D_+(y)\), the coordinate \(t=y/x\) is invertible.  Hence the overlap is \(\Spec k[t,t^{-1}]\).
\end{example}

\begin{example}[A projectively empty nilpotent positive part]\label{ex:vi35-05}
Let \(S=k[\varepsilon]\) with \(\deg\varepsilon=1\) and \(\varepsilon^2=0\).  Every prime contains \(\varepsilon\), hence contains \(S_+\), so \(\Proj S=\varnothing\).
\end{example}

\begin{example}[Reducible Proj]\label{ex:vi35-06}
For \(S=k[x,y]/(xy)\), the projectively relevant primes are \((x)\) and \((y)\).  Each standard chart is one reduced point.
\end{example}

\begin{example}[Nonreduced Proj]\label{ex:vi35-07}
For \(S=k[x,y]/(y^2)\), the chart \(D_+(x)\) has ring
\[
k[t]/(t^2),\qquad t=y/x.
\]
The topology is one point but the local ring is nonreduced.
\end{example}

\begin{example}[A homogeneous quotient]\label{ex:vi35-08}
For homogeneous \(I\subset S\), points of \(\Proj(S/I)\) correspond exactly to points of \(V_+(I)\subseteq\Proj S\).
\end{example}

\begin{example}[A nonhomogeneous equation warning]\label{ex:vi35-09}
In \(k[x,y]\), the polynomial \(x+y^2\) mixes degrees. Its vanishing is not invariant under simultaneous scalar rescaling, so it is not a projective equation for the standard grading.
\end{example}

\begin{example}[Stalk on a chart]\label{ex:vi35-10}
If \(\mathfrak p\in D_+(f)\), then \(\OO_{\Proj S,\mathfrak p}\) is the localization of \(S_{(f)}\) at the prime corresponding to \(\mathfrak p\), hence equals \(S_{(\mathfrak p)}\).
\end{example}

\begin{example}[Structure sheaf from the module S]\label{ex:vi35-11}
For the graded module \(M=S\), one has \(M_{(f)}=S_{(f)}\). Thus \(\widetilde S\cong\OO_{\Proj S}\).
\end{example}

\begin{example}[A quotient module]\label{ex:vi35-12}
For homogeneous \(I\subseteq S\), \(\widetilde{S/I}\) restricts on \(D_+(f)\) to the affine sheaf associated to \((S/I)_{(f)}\).
\end{example}

\begin{example}[Shifted module]\label{ex:vi35-13}
The shift \(S(d)\) gives \(\OO_{\Proj S}(d)=\widetilde{S(d)}\).  On \(D_+(f)\) its sections are \(S(d)_{(f)}\).
\end{example}

\begin{example}[Exactness chart by chart]\label{ex:vi35-14}
If \(0\to M'\to M\to M''\to0\) is exact in graded modules, then on every \(D_+(f)\) the degree-zero localized sequence is exact, so the associated sheaf sequence is exact.
\end{example}

\begin{example}[Veronese chart]\label{ex:vi35-15}
For \(S=k[x,y]\) and \(S^{(2)}=k[x^2,xy,y^2]\), the chart indexed by \(x^2\) has degree-zero ring \(k[y/x]\), the same affine chart as \(D_+(x)\) for \(S\).
\end{example}

\begin{example}[Global-section warning]\label{ex:vi35-16}
The recipe \(S\mapsto\Proj S\) discards and reorganizes graded information.  One should not expect to recover \(S\) simply as \(\Gamma(\Proj S,\OO)\); later projective theory instead studies all twists and their sections.
\end{example}
