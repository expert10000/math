\subsection*{Challenge 4: Immersions are locally affine quotients of localizations}
\begin{problem}\label{chal:vi22-04}
Show that an immersion into an affine scheme is locally represented by a surjection $A_f\twoheadrightarrow B$.
\par\textbf{Solution sketch.} Factor through an open subscheme.  Around each source point choose a principal open $D(f)$ inside that open image.  The closed immersion into $D(f)=\Spec A_f$ is affine-locally induced by a quotient of $A_f$.
\end{problem}
