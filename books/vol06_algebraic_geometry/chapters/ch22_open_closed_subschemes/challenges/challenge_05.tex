\subsection*{Challenge 5: Detecting scheme-theoretic vanishing}
\begin{problem}\label{chal:vi22-05}
Let $I\subseteq A$ and $f:\Spec B\to\Spec A$.  Compare the conditions that $f$ maps all points into $V(I)$ and that $f$ factors through $\Spec(A/I)$.  Give an example showing the first is weaker.
\par\textbf{Solution sketch.} Pointwise containment means the image of $I$ lies in every prime of $B$, hence in the nilradical $\sqrt{(0)}_B$.  Scheme-theoretic factorization requires the image of $I$ to be zero.  Take $A=k[t]$, $B=k[\varepsilon]/(\varepsilon^2)$, and $t\mapsto\varepsilon$: the unique point maps to $V(t)$, but the map does not factor through $k[t]/(t)$.
\end{problem}
