\subsection*{Challenge 5: Detecting scheme-theoretic vanishing}
\begin{problem}\label{chal:vi22-05}
Let $I\subseteq A$ and $f:\Spec B\to\Spec A$.  Compare the conditions that $f$ maps all points into $V(I)$ and that $f$ factors through $\Spec(A/I)$.  Give an example showing the first is weaker.
\end{problem}
