\subsection*{Challenge 3: Closed structures with fixed support}
\begin{problem}\label{chal:vi22-03}
For $X=\Spec A$ and a closed subset $V(J)$, show closed subscheme structures with this support correspond to ideals $I$ with $\sqrt I=\sqrt J$.
\par\textbf{Solution sketch.} The support of $\Spec(A/I)$ is $V(I)$, and $V(I)=V(J)$ iff $\sqrt I=\sqrt J$.
\end{problem}
