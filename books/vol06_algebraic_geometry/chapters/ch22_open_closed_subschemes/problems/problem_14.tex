\subsection*{VI.22.P14: Scheme-theoretic union}
\begin{problem}\label{prob:vi22-p14}
Show that the smallest affine closed subscheme containing both $V(I)$ and $V(J)$ is defined by $I\cap J$.
\par\textbf{Solution.} A closed subscheme $V(K)$ contains $V(I)$ scheme-theoretically exactly when $K\subseteq I$, and similarly for $J$.  Thus the largest ideal $K$ contained in both is $I\cap J$, producing the smallest closed subscheme containing both.
\end{problem}
