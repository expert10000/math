\subsection*{VI.22.P5: Restriction of a closed subscheme to a principal open}
\begin{problem}\label{prob:vi22-p05}
For $Z=\Spec(A/I)\hookrightarrow X=\Spec A$ and $U=D(f)$, prove $Z\cap U\cong\Spec(A_f/IA_f)$.
\end{problem}

\ifdefined\FullProblemDossiers
% VI22-FULL-P05
\paragraph{Why this problem matters.}
This is the compatibility that allows affine closed subschemes to glue across open charts.

\paragraph{Useful ideas before starting.}
Pull the principal open back to $\Spec(A/I)$ and identify its coordinate ring by localization followed by quotient.

\paragraph{Guided hints.}
\begin{enumerate}
\item The inverse image of $D(f)$ is $D(\bar f)$.
\item Use the principal-open theorem on $\Spec(A/I)$.
\item Prove $(A/I)_{\bar f}\cong A_f/IA_f$.
\end{enumerate}

\begin{solution}
Let $\bar f$ be the image of $f$ in $A/I$.  Under the closed immersion
$i:\Spec(A/I)\to\Spec A$, a prime $\mathfrak p/I$ lies over $D(f)$ exactly when
$f\notin\mathfrak p$, equivalently $\bar f\notin\mathfrak p/I$.  Hence
\[
 i^{-1}(D(f))=D(\bar f)\subseteq\Spec(A/I).
\]
By the principal-open theorem,
\[
 D(\bar f)\cong\Spec((A/I)_{\bar f}).
\]
There is a canonical ring isomorphism
\[
 (A/I)_{\bar f}\xrightarrow{\sim}A_f/IA_f,
\]
obtained by sending $(a+I)/\bar f^{\,n}$ to the class of $a/f^n$.  It is well defined and
invertible by the universal properties of quotient and localization.  Therefore
\[
 Z\cap D(f)\cong\Spec(A_f/IA_f).
\]
In particular, restricting a closed subscheme to an open affine chart simply localizes its
defining ideal.
\end{solution}

\paragraph{What happened mathematically.}
Open restriction and quotient commute: localizing the ambient ring localizes the equations defining the closed subscheme.

\paragraph{Extensions.}
Use the formula to derive the criterion for $Z\cap D(f)=\varnothing$.
