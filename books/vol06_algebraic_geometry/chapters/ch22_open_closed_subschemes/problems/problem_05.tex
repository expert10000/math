\subsection*{VI.22.P5: Restriction of a closed subscheme to a principal open}
\begin{problem}\label{prob:vi22-p05}
For $Z=\Spec(A/I)\hookrightarrow X=\Spec A$ and $U=D(f)$, prove $Z\cap U\cong\Spec(A_f/IA_f)$.
\par\textbf{Solution.} The inverse image of $D(f)$ in $\Spec(A/I)$ is $D(\bar f)$, whose ring is $(A/I)_{\bar f}$.  Localization commutes with quotient, giving $(A/I)_{\bar f}\cong A_f/IA_f$.
\end{problem}
