\subsection*{VI.22.P20: Open immersion and local rings}
\begin{problem}\label{prob:vi22-p20}
For $j:U\hookrightarrow X$ open and $u\in U$, prove $\OO_{X,u}\to\OO_{U,u}$ is an isomorphism.
\par\textbf{Solution.} The neighbourhood systems of $u$ in $U$ and in $X$ are cofinal after intersecting with $U$.  Therefore the direct limits defining the two stalks are canonically isomorphic; equivalently the structure sheaf on $U$ is the restriction $\OO_X|_U$.
\end{problem}
