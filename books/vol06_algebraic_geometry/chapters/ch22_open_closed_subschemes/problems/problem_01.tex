\subsection*{VI.22.P1: The canonical open subscheme}
\begin{problem}\label{prob:vi22-p01}
Let $U\subseteq X$ be open.  Prove that $(U,\OO_X|_U)$ is a scheme and that $U\hookrightarrow X$ is an open immersion.
\par\textbf{Solution.} Every $u\in U$ has an affine neighbourhood $V=\Spec A$ in $X$.  Since $U\cap V$ is open in $V$, it is covered by principal opens $D(f)$, each affine by VI/18.  Hence $(U,\OO_X|_U)$ is locally affine.  The inclusion is a homeomorphism onto the open subset $U$ and the sheaf map is the identity identification with the restriction.
\end{problem}
