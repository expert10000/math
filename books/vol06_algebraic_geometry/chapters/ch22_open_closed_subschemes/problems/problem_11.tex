\subsection*{VI.22.P11: Empty restriction criterion}
\begin{problem}\label{prob:vi22-p11}
Let $Z=V(I)\subset\Spec A$.  Prove $Z\cap D(f)=\varnothing$ iff $IA_f=A_f$, equivalently iff $f^n\in I$ for some $n\ge0$.
\end{problem}

\ifdefined\FullProblemDossiers
% VI22-FULL-P11
\paragraph{Why this problem matters.}
This translates geometric emptiness into localization algebra.

\paragraph{Useful ideas before starting.}
Use the restricted closed-subscheme ring $A_f/IA_f$, then clear denominators.

\paragraph{Guided hints.}
\begin{enumerate}
\item An affine scheme is empty exactly when its coordinate ring is the zero ring.
\item $A_f/IA_f=0$ iff $1\in IA_f$.
\item Clear a common power of $f$ from an expression for $1$.
\end{enumerate}

\begin{solution}
By restriction to the principal open,
\[
 Z\cap D(f)\cong\Spec(A_f/IA_f).
\]
This affine scheme is empty exactly when $A_f/IA_f$ is the zero ring, equivalently
\[
 IA_f=A_f
 \iff 1\in IA_f.
\]
Suppose $1\in IA_f$.  Then there are $i_r\in I$, $a_r\in A$, and integers $n_r\ge0$ such
that
\[
 1=\sum_r \frac{a_r i_r}{f^{n_r}}.
\]
Choose $N\ge n_r$ for all $r$ and multiply by $f^N$:
\[
 f^N=\sum_r a_r i_r f^{N-n_r}\in I.
\]
Conversely, if $f^N\in I$, then $f^N/1$ is a unit of $A_f$ lying in $IA_f$.  Multiplying by
its inverse shows $1\in IA_f$.  Hence all three conditions are equivalent.
\end{solution}

\paragraph{What happened mathematically.}
A closed subscheme misses $D(f)$ precisely when some power of $f$ is forced to vanish on it.

\paragraph{Extensions.}
Express the same condition as $D(f)\subseteq\Spec A\setminus V(I)$ and compare with radicals.
