\subsection*{VI.22.P13: Scheme-theoretic intersection}
\begin{problem}\label{prob:vi22-p13}
Let $Z_1=V(I)$ and $Z_2=V(J)$ be affine closed subschemes of $\Spec A$.  Show the closed subscheme imposing both sets of equations is $V(I+J)$.
\end{problem}

\ifdefined\FullProblemDossiers
% VI22-FULL-P13
\paragraph{Why this problem matters.}
The sum of ideals is the scheme-theoretic operation of imposing two systems of equations simultaneously.

\paragraph{Useful ideas before starting.}
Characterize the desired object by factorization of test morphisms, or equivalently by a quotient that kills both ideals.

\paragraph{Guided hints.}
\begin{enumerate}
\item A ring map kills both $I$ and $J$ iff it kills $I+J$.
\item Use the quotient $A/(I+J)$.
\item Compare its support with $V(I)\cap V(J)$.
\end{enumerate}

\begin{solution}
Let $T=\Spec B$ and let $f:T\to\Spec A$ correspond to
$\alpha:A\to B$.  The morphism factors through $Z_1=\Spec(A/I)$ exactly when
$I\subseteq\ker\alpha$, and it factors through $Z_2=\Spec(A/J)$ exactly when
$J\subseteq\ker\alpha$.  Both conditions hold exactly when
\[
 I+J\subseteq\ker\alpha.
\]
By the quotient universal property, this is equivalent to a unique factorization through
\[
 \Spec(A/(I+J)).
\]
Thus $A/(I+J)$ represents simultaneous scheme-theoretic vanishing of the equations in $I$ and
$J$.  Its support is
\[
 V(I+J)=V(I)\cap V(J),
\]
so it has the expected underlying intersection as well.
\end{solution}

\paragraph{What happened mathematically.}
Scheme-theoretic intersection is controlled by adding ideals because all equations from both sides must vanish.

\paragraph{Extensions.}
Compute the intersection of the two coordinate axes in $\mathbb A^2_k$.
