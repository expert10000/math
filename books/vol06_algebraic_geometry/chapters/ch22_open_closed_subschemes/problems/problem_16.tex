\subsection*{VI.22.P16: Closed immersions compose}
\begin{problem}\label{prob:vi22-p16}
Prove the composite of two closed immersions is a closed immersion.
\par\textbf{Solution.} This is affine-local on the target.  On affine charts the two morphisms are induced by surjective ring maps.  Their composite is surjective, so the affine-local criterion applies.
\end{problem}
