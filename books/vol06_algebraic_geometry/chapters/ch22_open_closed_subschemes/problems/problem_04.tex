\subsection*{VI.22.P4: Same support, different closed subschemes}
\begin{problem}\label{prob:vi22-p04}
Show that $(x)$ and $(x^2)$ in $k[x]$ define the same closed subset but nonisomorphic closed subschemes of $\mathbb A^1_k$.
\par\textbf{Solution.} Both radicals equal $(x)$, hence both supports are $V(x)$.  The quotient by $(x)$ is $k$, reduced, whereas $k[x]/(x^2)$ has the nonzero nilpotent class of $x$.  Therefore the schemes are not isomorphic.
\end{problem}
