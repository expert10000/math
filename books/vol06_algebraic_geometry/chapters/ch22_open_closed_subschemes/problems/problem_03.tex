\subsection*{VI.22.P3: Affine closed subschemes from ideals}
\begin{problem}\label{prob:vi22-p03}
Let $I\subseteq A$.  Prove that $\Spec(A/I)\to\Spec A$ is a homeomorphism onto $V(I)$ and compute the stalks.
\par\textbf{Solution.} Prime ideals of $A/I$ correspond to primes of $A$ containing $I$ by inverse image.  Closed sets correspond under this bijection, giving the homeomorphism onto $V(I)$.  At $\mathfrak p\supseteq I$, localization commutes with quotient: $(A/I)_{\mathfrak p/I}\cong A_{\mathfrak p}/IA_{\mathfrak p}$.
\end{problem}
