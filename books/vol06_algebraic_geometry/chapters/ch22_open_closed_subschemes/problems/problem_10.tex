\subsection*{VI.22.P10: Universal property of reduction}
\begin{problem}\label{prob:vi22-p10}
Let $T$ be reduced.  Prove every morphism $T\to X$ factors uniquely through $X_{\rm red}$.
\par\textbf{Solution.} Work on affine opens.  A ring map $A\to B$ with $B$ reduced kills every nilpotent of $A$, hence kills $\sqrt{(0)}$ and factors uniquely through $A/\sqrt{(0)}$.  The local factorizations agree on overlaps by uniqueness and glue.
\end{problem}
