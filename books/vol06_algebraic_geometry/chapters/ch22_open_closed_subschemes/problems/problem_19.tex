\subsection*{VI.22.P19: Closed immersion and residue fields}
\begin{problem}\label{prob:vi22-p19}
For $i:Z\hookrightarrow X$ closed and $z\in Z$, prove the induced map $\kappa(i(z))\to\kappa(z)$ is an isomorphism.
\par\textbf{Solution.} Affine-locally write $Z=\Spec(A/I)$ and $z=\mathfrak p/I$.  The local ring is $A_\mathfrak p/IA_\mathfrak p$ with maximal ideal $\mathfrak pA_\mathfrak p/IA_\mathfrak p$.  Quotienting gives the same field as $A_\mathfrak p/\mathfrak pA_\mathfrak p$.
\end{problem}
