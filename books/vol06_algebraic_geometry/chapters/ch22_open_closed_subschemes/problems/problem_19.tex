\subsection*{VI.22.P19: Closed immersion and residue fields}
\begin{problem}\label{prob:vi22-p19}
For $i:Z\hookrightarrow X$ closed and $z\in Z$, prove the induced map $\kappa(i(z))\to\kappa(z)$ is an isomorphism.
\end{problem}

\ifdefined\FullProblemDossiers
% VI22-FULL-P19
\paragraph{Why this problem matters.}
A closed immersion changes infinitesimal neighborhoods but not the field attached to a point.

\paragraph{Useful ideas before starting.}
Reduce to the affine quotient model and compute both residue fields from the localized quotient.

\paragraph{Guided hints.}
\begin{enumerate}
\item Write $X=\Spec A$, $Z=\Spec(A/I)$ near the point.
\item Let $i(z)=\mathfrak p$ and $z=\mathfrak p/I$.
\item Compute the two local rings and quotient by their maximal ideals.
\end{enumerate}

\begin{solution}
The claim is local on $X$, so choose an affine neighbourhood
$X=\Spec A$ of $i(z)$ on which
\[
 Z=\Spec(A/I)
\]
and write $i(z)=\mathfrak p$ with $I\subseteq\mathfrak p$.  Then
$z$ corresponds to $\mathfrak p/I$.  The local ring of $Z$ at $z$ is
\[
 \OO_{Z,z}\cong A_{\mathfrak p}/IA_{\mathfrak p},
\]
whose maximal ideal is
\[
 \mathfrak pA_{\mathfrak p}/IA_{\mathfrak p}.
\]
Therefore
\[
 \kappa(z)
 \cong
 \frac{A_{\mathfrak p}/IA_{\mathfrak p}}
      {\mathfrak pA_{\mathfrak p}/IA_{\mathfrak p}}
 \cong
 \frac{A_{\mathfrak p}}{\mathfrak pA_{\mathfrak p}}
 =\kappa(i(z)).
\]
Under this identification, the residue-field map induced by the closed immersion is the
displayed canonical isomorphism.
\end{solution}

\paragraph{What happened mathematically.}
Closed immersions preserve residue fields even though they can change nilpotent and higher-order local structure.

\paragraph{Extensions.}
Compare with open immersions, which in fact preserve the entire local ring.
