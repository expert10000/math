\subsection*{VI.22.P8: Factorization through a closed subscheme}
\begin{problem}\label{prob:vi22-p08}
Let $A\to B$ define $f:\Spec B\to\Spec A$ and let $Z=\Spec(A/I)$.  Prove $f$ factors through $Z$ iff $I$ maps to zero in $B$.
\par\textbf{Solution.} A factorization corresponds contravariantly to a ring map $A/I\to B$ whose composite with $A\to A/I$ is the given map.  Such a map exists uniquely exactly when $I$ is contained in the kernel of $A\to B$.
\end{problem}
