\begin{example}[Two closed structures on one support]\label{ex:vi22-ae01}
For $I=(x)$ and $J=(x^3)$ in $k[x]$, the supports agree while the conormal nilpotent structure differs.
\end{example}

\begin{example}[Scheme-theoretic intersection in an affine chart]\label{ex:vi22-ae02}
Closed subschemes defined by $I$ and $J$ impose both sets of equations; locally their common closed subscheme is defined by $I+J$.
\end{example}

\begin{example}[Scheme-theoretic union in an affine chart]\label{ex:vi22-ae03}
The smallest closed subscheme containing both $V(I)$ and $V(J)$ is defined by $I\cap J$.
\end{example}

\begin{example}[Set-theoretic versus scheme-theoretic union]\label{ex:vi22-ae04}
$V(IJ)=V(I\cap J)$ topologically, although the quotient schemes defined by $IJ$ and $I\cap J$ can differ.
\end{example}

\begin{example}[A first infinitesimal neighbourhood]\label{ex:vi22-ae05}
If $Z=V(I)\subset X$ is affine, $V(I^2)$ has the same support but remembers first-order normal information.
\end{example}

\begin{example}[Closed immersion detected locally]\label{ex:vi22-ae06}
If a morphism becomes a quotient map on coordinate rings after restricting to an affine cover of the target, the pieces glue to a closed immersion.
\end{example}

\begin{example}[Open immersion detected locally]\label{ex:vi22-ae07}
If a morphism identifies inverse images of an open cover with open subschemes and the local maps agree, it is an open immersion.
\end{example}

\begin{example}[Immersion into an affine scheme]\label{ex:vi22-ae08}
A locally closed subscheme $Z\subset U\subset\Spec A$ is affine-locally a quotient of a localization $A_f$.
\end{example}

\begin{example}[The reduced induced structure]\label{ex:vi22-ae09}
Among closed subschemes with support $V(I)$, $\Spec(A/\sqrt I)$ is the unique reduced one.
\end{example}

\begin{example}[Nilpotents are invisible to points]\label{ex:vi22-ae10}
The canonical closed immersion $X_{\rm red}\hookrightarrow X$ is a homeomorphism on underlying spaces.
\end{example}

\begin{example}[Vanishing after localization]\label{ex:vi22-ae11}
The restriction of $V(I)$ to $D(f)$ is empty exactly when $IA_f=A_f$, equivalently some power of $f$ lies in $I$.
\end{example}

\begin{example}[Closed subscheme of a closed subscheme]\label{ex:vi22-ae12}
If $I\subset J\subset A$, then $\Spec(A/J)\hookrightarrow\Spec(A/I)\hookrightarrow\Spec A$ is a chain of closed immersions.
\end{example}

\begin{example}[Composition of open immersions]\label{ex:vi22-ae13}
The composite of two open immersions is an open immersion because restrictions of restrictions are restrictions.
\end{example}

\begin{example}[Composition of closed immersions]\label{ex:vi22-ae14}
Affine-locally it is induced by a composite of surjective ring maps, hence is again a closed immersion.
\end{example}
