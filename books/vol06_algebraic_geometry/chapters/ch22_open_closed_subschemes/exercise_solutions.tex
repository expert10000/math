\paragraph{Exercise VI.22.1. Open affine subsets of an open subscheme.}
% VI22-EXERCISE-FULL-01
\begin{solution}
Let $U\subseteq X$ be open and let $V\subseteq U$ be an affine open subscheme of $U$.
Because the topology on $U$ is the subspace topology, $V$ is open in $U$ and hence open in
$X$.  Moreover
\[
 \OO_U=\OO_X|_U,
 \qquad
 \OO_V=\OO_U|_V=(\OO_X|_U)|_V\cong\OO_X|_V.
\]
Thus the inclusion $V\hookrightarrow X$ is the composite of the open immersions
$V\hookrightarrow U$ and $U\hookrightarrow X$, and therefore is an open immersion.  In
particular, the affine scheme structure already carried by $V$ is exactly the canonical open
subscheme structure obtained by restricting $\OO_X$.
\end{solution}

\paragraph{Exercise VI.22.2. The closed subscheme of $\Spec\mathbb Z$ defined by $(n)$.}
% VI22-EXERCISE-FULL-02
\begin{solution}
The quotient map $\mathbb Z\twoheadrightarrow\mathbb Z/(n)$ defines the closed immersion
\[
 \Spec(\mathbb Z/n\mathbb Z)\hookrightarrow\Spec\mathbb Z.
\]
Its support is
\[
 V(n)=\{\mathfrak p\in\Spec\mathbb Z:n\in\mathfrak p\}.
\]
If $n\neq0$, the prime ideals containing $(n)$ are precisely $(p)$ for primes $p$ dividing
$n$.  If $n=\pm1$, the quotient ring is the zero ring and the closed subscheme is empty.
If $n=0$, the quotient is $\mathbb Z$ and the closed subscheme is the whole spectrum.
Notice that for $n$ divisible by $p^r$ with $r>1$, the scheme remembers the higher power:
$\Spec(\mathbb Z/p^r\mathbb Z)$ has the same one-point support as
$\Spec\mathbb F_p$ but a thicker nonreduced structure when $r>1$.
\end{solution}

\paragraph{Exercise VI.22.3. The thickenings $(x)$, $(x^2)$, and $(x^3)$.}
% VI22-EXERCISE-FULL-03
\begin{solution}
All three ideals have radical $(x)$, so all three closed subschemes have the same support,
namely the origin of $\mathbb A^1_k$.  Their coordinate rings are
\[
 k[x]/(x)\cong k,\qquad
 k[x]/(x^2),\qquad
 k[x]/(x^3).
\]
The first is reduced.  In the second, the class $\bar x$ is nonzero with $\bar x^2=0$.
In the third, $\bar x^2\neq0$ but $\bar x^3=0$.  Thus the nilpotency order distinguishes
the three quotient rings, hence the three affine closed subschemes.  They are successively
thicker infinitesimal neighborhoods of the same topological point.
\end{solution}

\paragraph{Exercise VI.22.4. The local ring of the origin subscheme.}
% VI22-EXERCISE-FULL-04
\begin{solution}
Let $A=k[x,y]$ and $I=(x,y)$.  Then
\[
 V(I)=\Spec(A/I)=\Spec k,
\]
so it has a unique point.  Its local ring is therefore the localization of $k$ at its zero
prime, which is just $k$.  Equivalently, if $\mathfrak m=(x,y)$ is the corresponding point
of $\Spec A$, the closed-subscheme stalk formula gives
\[
 \OO_{V(I),0}
 \cong A_{\mathfrak m}/IA_{\mathfrak m}
 =k[x,y]_{(x,y)}/(x,y)k[x,y]_{(x,y)}
 \cong k.
\]
\end{solution}

\paragraph{Exercise VI.22.5. Why $V(I)=V(\sqrt I)$.}
% VI22-EXERCISE-FULL-05
\begin{solution}
For a prime ideal $\mathfrak p$,
\[
 I\subseteq\mathfrak p\quad\Longrightarrow\quad \sqrt I\subseteq\sqrt{\mathfrak p}
 =\mathfrak p,
\]
because prime ideals are radical.  The converse is immediate from $I\subseteq\sqrt I$.
Hence
\[
 \mathfrak p\in V(I)
 \iff I\subseteq\mathfrak p
 \iff \sqrt I\subseteq\mathfrak p
 \iff \mathfrak p\in V(\sqrt I).
\]
Thus the support of a closed subscheme only sees the radical of its defining ideal, whereas
its structure sheaf sees the full ideal.
\end{solution}

\paragraph{Exercise VI.22.6. Same radical, different quotient rings.}
% VI22-EXERCISE-FULL-06
\begin{solution}
Take $A=k[x]$, $I=(x)$ and $J=(x^2)$.  Then
\[
 \sqrt I=(x)=\sqrt J.
\]
However
\[
 A/I\cong k
\]
is reduced, whereas
\[
 A/J\cong k[x]/(x^2)
\]
contains the nonzero nilpotent class of $x$.  A ring isomorphism preserves nilpotent
elements, so these quotient rings cannot be isomorphic.  Consequently they define different
closed subscheme structures on the same closed subset $V(x)$.
\end{solution}

\paragraph{Exercise VI.22.7. Computing $V(x)\cap D(y)$ scheme-theoretically.}
% VI22-EXERCISE-FULL-07
\begin{solution}
Let $A=k[x,y]$ and $I=(x)$.  Restricting the closed subscheme
$\Spec(A/I)\hookrightarrow\Spec A$ to $D(y)$ gives
\[
 \Spec(A_y/IA_y).
\]
Now
\[
 A_y/IA_y
 \cong k[x,y,y^{-1}]/(x)
 \cong k[y,y^{-1}].
\]
Therefore
\[
 V(x)\cap D(y)\cong\Spec k[y,y^{-1}],
\]
the punctured $y$-axis.  This computation uses both operations: first localize to the open
chart and then quotient by the localized ideal.
\end{solution}

\paragraph{Exercise VI.22.8. When $V(I)\cap D(f)$ is empty.}
% VI22-EXERCISE-FULL-08
\begin{solution}
Scheme-theoretically,
\[
 V(I)\cap D(f)\cong\Spec(A_f/IA_f).
\]
This scheme is empty exactly when its coordinate ring is the zero ring, equivalently when
$1\in IA_f$.  If $1\in IA_f$, write
\[
 1=\sum_{r=1}^m \frac{a_r}{f^{n_r}}\,i_r
 \qquad (a_r\in A,\ i_r\in I).
\]
Multiplying by a sufficiently high power $f^N$ gives $f^N\in I$.  Conversely, if
$f^N\in I$, then $f^N/1$ is a unit in $A_f$ lying in $IA_f$, so $1\in IA_f$.  Hence
\[
 V(I)\cap D(f)=\varnothing
 \iff IA_f=A_f
 \iff f^N\in I\text{ for some }N\ge0.
\]
\end{solution}

\paragraph{Exercise VI.22.9. Quotients of local rings.}
% VI22-EXERCISE-FULL-09
\begin{solution}
Let $(R,\mathfrak m)$ be local and let $I\subsetneq R$.  Every maximal ideal of $R/I$
has the form $\mathfrak n/I$ for a maximal ideal $\mathfrak n$ of $R$ containing $I$.
Since $R$ has only one maximal ideal, necessarily $\mathfrak n=\mathfrak m$.  Thus
$\mathfrak m/I$ is the unique maximal ideal of $R/I$, so $R/I$ is local.  Equivalently,
an element $\bar r\in R/I$ outside $\mathfrak m/I$ has a representative
$r\notin\mathfrak m$, hence $r$ is a unit in $R$, and its inverse descends to an inverse
of $\bar r$.
\end{solution}

\paragraph{Exercise VI.22.10. Residue fields under a surjection.}
% VI22-EXERCISE-FULL-10
\begin{solution}
Let $\varphi:A\twoheadrightarrow B=A/I$ and let $\mathfrak q\in\Spec B$ correspond to
$\mathfrak p=\varphi^{-1}(\mathfrak q)\supseteq I$.  Localizing gives a surjection
\[
 A_{\mathfrak p}\twoheadrightarrow B_{\mathfrak q}
 \cong A_{\mathfrak p}/IA_{\mathfrak p}.
\]
Its maximal ideals correspond:
\[
 \mathfrak qB_{\mathfrak q}
 \cong \mathfrak pA_{\mathfrak p}/IA_{\mathfrak p}.
\]
Hence
\[
 \kappa(\mathfrak q)
 =\frac{B_{\mathfrak q}}{\mathfrak qB_{\mathfrak q}}
 \cong
 \frac{A_{\mathfrak p}/IA_{\mathfrak p}}
      {\mathfrak pA_{\mathfrak p}/IA_{\mathfrak p}}
 \cong
 \frac{A_{\mathfrak p}}{\mathfrak pA_{\mathfrak p}}
 =\kappa(\mathfrak p).
\]
Thus a closed immersion does not change the residue field at a corresponding point.
\end{solution}

\paragraph{Exercise VI.22.11. Reduction of a reduced scheme.}
% VI22-EXERCISE-FULL-11
\begin{solution}
If $X$ is reduced, then for every affine open $U=\Spec A\subseteq X$ the ring $A$ is
reduced, so its nilradical is zero.  Therefore
\[
 U_{\mathrm{red}}
 =\Spec(A/\sqrt{(0)})
 =\Spec A
 =U.
\]
The identifications are compatible on overlaps, so the global reduction morphism
$X_{\mathrm{red}}\to X$ is an isomorphism.  Thus reduction acts as the identity on already
reduced schemes.
\end{solution}

\paragraph{Exercise VI.22.12. Reduction of $k[x,y]/(x^2y^3)$.}
% VI22-EXERCISE-FULL-12
\begin{solution}
Let
\[
 A=k[x,y]/(x^2y^3).
\]
The nilradical of $A$ is the radical of the ideal $(x^2y^3)$ modulo that ideal.  In the UFD
$k[x,y]$,
\[
 \sqrt{(x^2y^3)}=(xy),
\]
because a prime containing $x^2y^3$ contains $x$ or $y$, and $(xy)$ is radical since
$(xy)=(x)\cap(y)$.  Therefore
\[
 A_{\mathrm{red}}
 \cong k[x,y]/(xy),
\]
and
\[
 (\Spec A)_{\mathrm{red}}
 \cong \Spec k[x,y]/(xy).
\]
The underlying topological space is unchanged: it is the union of the two coordinate axes.
\end{solution}

\paragraph{Exercise VI.22.13. Idempotence of reduction.}
% VI22-EXERCISE-FULL-13
\begin{solution}
By construction $X_{\mathrm{red}}$ is reduced.  Applying Exercise VI.22.11 to the reduced
scheme $X_{\mathrm{red}}$ therefore gives
\[
 (X_{\mathrm{red}})_{\mathrm{red}}\cong X_{\mathrm{red}}.
\]
Affine-locally this is the elementary identity
\[
 \left(A/\sqrt{(0)_A}\right)_{\mathrm{red}}
 =
 \left(A/\sqrt{(0)_A}\right)/0
 =
 A/\sqrt{(0)_A}.
\]
The isomorphisms are canonical and glue.
\end{solution}

\paragraph{Exercise VI.22.14. Maps from reduced affine schemes kill nilpotents.}
% VI22-EXERCISE-FULL-14
\begin{solution}
Let $T=\Spec B$ with $B$ reduced, and let a morphism
$T\to\Spec A$ correspond to a ring map $\varphi:A\to B$.  If $a\in\sqrt{(0)_A}$, then
$a^n=0$ for some $n$.  Hence
\[
 \varphi(a)^n=\varphi(a^n)=0.
\]
Since $B$ is reduced, it has no nonzero nilpotents, so $\varphi(a)=0$.  Thus
$\sqrt{(0)_A}\subseteq\ker\varphi$, and the map factors uniquely through
\[
 A/\sqrt{(0)_A}\longrightarrow B.
\]
Contravariantly this is the affine form of the universal property of $X_{\mathrm{red}}$.
\end{solution}

\paragraph{Exercise VI.22.15. Topological intersection of closed sets.}
% VI22-EXERCISE-FULL-15
\begin{solution}
For a prime $\mathfrak p$,
\[
 \mathfrak p\in V(I+J)
 \iff I+J\subseteq\mathfrak p
 \iff I\subseteq\mathfrak p\text{ and }J\subseteq\mathfrak p.
\]
Therefore
\[
 V(I+J)=V(I)\cap V(J).
\]
Scheme-theoretically, this same ideal $I+J$ is also the correct ideal for the simultaneous
vanishing condition: a morphism into $\Spec A$ factors through both closed subschemes exactly
when its ring map kills both $I$ and $J$, equivalently $I+J$.
\end{solution}

\paragraph{Exercise VI.22.16. Topological union of closed sets.}
% VI22-EXERCISE-FULL-16
\begin{solution}
For a prime $\mathfrak p$,
\[
 I\cap J\subseteq\mathfrak p
\]
implies $IJ\subseteq I\cap J\subseteq\mathfrak p$.  Since $\mathfrak p$ is prime,
$IJ\subseteq\mathfrak p$ implies $I\subseteq\mathfrak p$ or $J\subseteq\mathfrak p$.
The converse is immediate.  Hence
\[
 V(I\cap J)=V(I)\cup V(J).
\]
One may equally use $IJ$ topologically because
$\sqrt{I\cap J}=\sqrt{IJ}=\sqrt I\cap\sqrt J$.  The scheme-theoretic union, however, is
defined by $I\cap J$, not merely by an arbitrary ideal with the same radical.
\end{solution}

\paragraph{Exercise VI.22.17. Scheme-theoretic intersection of the axes.}
% VI22-EXERCISE-FULL-17
\begin{solution}
In $A=k[x,y]$, the $y$-axis is defined by $I=(x)$ and the $x$-axis by $J=(y)$.
The scheme-theoretic intersection is defined by the sum:
\[
 I+J=(x,y).
\]
Therefore
\[
 V(x)\cap_{\mathrm{sch}}V(y)
 =\Spec k[x,y]/(x,y)
 \cong\Spec k.
\]
Thus the intersection is the reduced origin.  This is stronger than merely observing that the
two underlying closed sets meet in one point: the quotient computation determines its scheme
structure.
\end{solution}

\paragraph{Exercise VI.22.18. Scheme-theoretic union of the axes.}
% VI22-EXERCISE-FULL-18
\begin{solution}
The scheme-theoretic union is defined by the intersection of the two ideals:
\[
 (x)\cap(y)=(xy)
\]
in the UFD $k[x,y]$.  Hence
\[
 V(x)\cup_{\mathrm{sch}}V(y)
 =\Spec k[x,y]/(xy).
\]
Its support is the union of the two coordinate axes.  Since $(xy)$ is radical, this particular
union is reduced.
\end{solution}

\paragraph{Exercise VI.22.19. An immersion with nonclosed image.}
% VI22-EXERCISE-FULL-19
\begin{solution}
Take the principal open immersion
\[
 j:D(t)\hookrightarrow\mathbb A^1_k=\Spec k[t].
\]
Every open immersion is an immersion: factor it as the identity closed immersion
$D(t)\hookrightarrow D(t)$ followed by $j$.  Its image is $D(t)$, the complement of the
origin.  This subset is open and dense, hence not closed when $\mathbb A^1_k$ has more than one
point.  Thus an immersion need only have locally closed image, not closed image.
\end{solution}

\paragraph{Exercise VI.22.20. Open immersions are injective on points.}
% VI22-EXERCISE-FULL-20
\begin{solution}
By definition an open immersion $j:U\to X$ identifies $U$ isomorphically with an open
subscheme of $X$.  In particular, the underlying continuous map
\[
 |U|\longrightarrow|X|
\]
is a homeomorphism onto its image.  Every homeomorphism onto a subspace is injective, so $j$
is injective on points.  No affine hypothesis is needed.
\end{solution}

\paragraph{Exercise VI.22.21. Closed immersions are injective on points.}
% VI22-EXERCISE-FULL-21
\begin{solution}
A closed immersion $i:Z\to X$ identifies $Z$ with a closed subscheme whose underlying space is
a closed subset of $|X|$.  Hence the underlying map
\[
 |Z|\longrightarrow|X|
\]
is a homeomorphism onto that closed subset and is therefore injective.  Affine-locally this is
the familiar bijection from primes of $A/I$ to primes of $A$ containing $I$.
\end{solution}

\paragraph{Exercise VI.22.22. Simultaneously open and closed immersions.}
% VI22-EXERCISE-FULL-22
\begin{solution}
Suppose $f:Z\to X$ is both an open and a closed immersion.  As an open immersion it identifies
$Z$ with an open subset $W\subseteq X$ and identifies $\OO_Z$ with $\OO_X|_W$.  As a closed
immersion, the same image $W$ is closed in $X$.  Thus $W$ is open and closed.  Because the open
immersion already fixes the scheme structure on $W$ uniquely as the restricted structure sheaf,
$f$ identifies $Z$ with the canonical open-and-closed subscheme $(W,\OO_X|_W)$.
\end{solution}

\paragraph{Exercise VI.22.23. A closed subscheme inside another closed subscheme.}
% VI22-EXERCISE-FULL-23
\begin{solution}
Let $I\subseteq J\subseteq A$.  The quotient map
\[
 A/I\twoheadrightarrow A/J
\]
has kernel
\[
 J/I\subseteq A/I.
\]
Therefore
\[
 \Spec(A/J)\hookrightarrow\Spec(A/I)
\]
is the closed subscheme defined by $J/I$.  Its support consists exactly of the primes
of $A/I$ corresponding to primes of $A$ that contain $J$.
\end{solution}

\paragraph{Exercise VI.22.24. Localization commutes with quotient.}
% VI22-EXERCISE-FULL-24
\begin{solution}
Let $\bar f$ denote the image of $f$ in $A/I$.  The quotient map $A\to A/I$ sends $f$ to
$\bar f$, so after localization it induces
\[
 \Phi:A_f\longrightarrow(A/I)_{\bar f}.
\]
The ideal $IA_f$ lies in $\ker\Phi$, hence $\Phi$ descends to
\[
 A_f/IA_f\longrightarrow(A/I)_{\bar f}.
\]
Conversely, the map $A\to A_f/IA_f$ kills $I$ and sends $f$ to a unit, so by the universal
properties of quotient and localization it induces
\[
 (A/I)_{\bar f}\longrightarrow A_f/IA_f.
\]
The two maps are inverse because both send $a/f^n$ to the same fraction.  Thus
\[
 (A/I)_{\bar f}\cong A_f/IA_f.
\]
\end{solution}

\paragraph{Exercise VI.22.25. Canonical open complement.}
% VI22-EXERCISE-FULL-25
\begin{solution}
Let $F\subseteq|X|$ be closed and set $U=X\setminus F$.  The subset $U$ is determined purely
topologically and is open.  Every open subset of a scheme has the canonical open subscheme
structure
\[
 (U,\OO_X|_U),
\]
and this structure is uniquely characterized by the requirement that $U\hookrightarrow X$ be
an open immersion.  By contrast, a closed subset $F$ can carry many closed subscheme structures,
because different ideal sheaves may have the same radical and therefore the same support.
Thus no choice on the closed side affects the canonical scheme structure of the open complement.
\end{solution}

\paragraph{Exercise VI.22.26. A nilpotent thickening with the same topology.}
% VI22-EXERCISE-FULL-26
\begin{solution}
Take
\[
 X=\Spec k[\varepsilon]/(\varepsilon^2)
\]
and its reduction
\[
 X_{\mathrm{red}}=\Spec k.
\]
The ring of dual numbers has one prime ideal, $(\varepsilon)$, so $|X|$ is a one-point space,
just like $|\Spec k|$.  The reduction map is therefore a homeomorphism on underlying spaces.
It is not an isomorphism of schemes because the global ring of $X$ contains the nonzero
nilpotent $\varepsilon$, whereas $k$ is reduced.  Thus $X$ is a genuine nilpotent thickening
of its reduced point.
\end{solution}
