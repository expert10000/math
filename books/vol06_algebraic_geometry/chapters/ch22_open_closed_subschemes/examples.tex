\begin{example}[A principal open]\label{ex:vi22-e01}
In $X=\Spec k[x,y]$, the open subscheme $D(x)$ is affine, with
\[
D(x)\cong\Spec k[x,y,x^{-1}].
\]
\end{example}

\begin{example}[The complement of the origin in the affine line]\label{ex:vi22-e02}
$D(t)\subset\Spec k[t]$ is the multiplicative group $\Spec k[t,t^{-1}]$.
\end{example}

\begin{example}[A reduced closed point]\label{ex:vi22-e03}
The ideal $(x)\subset k[x]$ defines the closed subscheme $\Spec k$ at the origin.
\end{example}

\begin{example}[A doubled closed point]\label{ex:vi22-e04}
The ideal $(x^2)$ defines a nonreduced closed subscheme supported at the same origin.
\end{example}

\begin{example}[A coordinate axis]\label{ex:vi22-e05}
In $\mathbb A^2_k=\Spec k[x,y]$, the ideal $(x)$ defines the $y$-axis $\Spec k[y]$.
\end{example}

\begin{example}[The union of coordinate axes]\label{ex:vi22-e06}
The ideal $(xy)$ defines $V(xy)=V(x)\cup V(y)$ with coordinate ring $k[x,y]/(xy)$.
\end{example}

\begin{example}[A thickened axis]\label{ex:vi22-e07}
$(x^2)\subset k[x,y]$ defines an infinitesimal thickening of the $y$-axis.
\end{example}

\begin{example}[Restriction of a closed point]\label{ex:vi22-e08}
If $Z=V(x)\subset\Spec k[x]$ and $U=D(x)$, then $Z\cap U=\varnothing$ because $(k[x]/(x))_x=0$.
\end{example}

\begin{example}[Restriction of an axis]\label{ex:vi22-e09}
For $Z=V(x)\subset\mathbb A^2$ and $U=D(y)$, $Z\cap U=\Spec k[y,y^{-1}]$.
\end{example}

\begin{example}[A quotient-induced closed immersion]\label{ex:vi22-e10}
A surjection $A\twoheadrightarrow B$ always induces a closed immersion $\Spec B\hookrightarrow\Spec A$.
\end{example}

\begin{example}[An open immersion from localization]\label{ex:vi22-e11}
$A\to A_f$ induces the open immersion $D(f)\hookrightarrow\Spec A$.
\end{example}

\begin{example}[Same support, different subschemes]\label{ex:vi22-e12}
$(x)$ and $(x^n)$ have the same radical for $n\ge2$ but different quotient rings.
\end{example}

\begin{example}[Reduction of a doubled point]\label{ex:vi22-e13}
$(\Spec k[\varepsilon]/(\varepsilon^2))_{\rm red}=\Spec k$.
\end{example}

\begin{example}[Reduction of a reducible affine scheme]\label{ex:vi22-e14}
$k[x,y]/(x^2y)$ reduces to $k[x,y]/(xy)$ because $\sqrt{(x^2y)}=(xy)$.
\end{example}

\begin{example}[A locally closed point]\label{ex:vi22-e15}
A closed point in an open subscheme is a locally closed subscheme of the ambient scheme.
\end{example}

\begin{example}[A punctured closed curve]\label{ex:vi22-e16}
If $C=V(y-x^2)\subset\mathbb A^2$ and $U=D(x)$, then $C\cap U$ is closed in $U$ but generally not closed in all of $\mathbb A^2$.
\end{example}

\begin{example}[Vanishing criterion]\label{ex:vi22-e17}
A map $\Spec B\to\Spec A$ factors through $V(I)$ exactly when the ring map kills $I$.
\end{example}

\begin{example}[Closed immersion and residue fields]\label{ex:vi22-e18}
For $\Spec(A/I)\hookrightarrow\Spec A$, the residue field at $\mathfrak p/I$ is canonically the residue field at $\mathfrak p$.
\end{example}

\begin{example}[Open immersion and stalks]\label{ex:vi22-e19}
For $j:U\hookrightarrow X$ and $u\in U$, the stalk map $\OO_{X,u}\to\OO_{U,u}$ is an isomorphism.
\end{example}

\begin{example}[Closed immersion and local rings]\label{ex:vi22-e20}
At $z\in Z\hookrightarrow X$, the local ring $\OO_{Z,z}$ is a quotient of $\OO_{X,z}$.
\end{example}
