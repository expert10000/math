\chapter{Open and Closed Subschemes}\label{ch:vi22-open-closed-subschemes}

Subschemes are the scheme-theoretic way to isolate a region or impose equations.  Open
subschemes are obtained by restriction.  Closed subschemes require more information than a
closed subset: an ideal of functions records not only where the subscheme lies, but also its
possible nilpotent and infinitesimal structure.  This chapter develops the local affine
construction, the global gluing principle, and the language of open, closed, and locally closed
immersions without consuming the later theories of fiber products or quasi-coherent sheaves.

\section{Open subschemes}
Let $X$ be a scheme and $U\subseteq X$ open.  The pair
\[
 (U,\OO_X|_U)
\]
is again a scheme.  Its inclusion
\[
 j:U\hookrightarrow X
\]
is an \emph{open immersion}: it is a homeomorphism onto an open subset and identifies the
structure sheaf of $U$ with the restricted structure sheaf of $X$.

\begin{proposition}[Uniqueness of the open subscheme structure]\label{prop:vi22-open-unique}
For a fixed open subset $U\subseteq X$, there is a unique scheme structure on $U$ for which the
inclusion is an open immersion, up to the unique isomorphism compatible with the inclusion.
\end{proposition}

\section{Principal opens revisited}
If $X=\Spec A$ and $f\in A$, then VI/18 gives
\[
 D(f)\cong \Spec A_f.
\]
Thus the canonical open subscheme on a principal open is simultaneously the localization
construction.  For $g\in A$,
\[
 D(f)\cap D(g)=D(fg),\qquad (A_f)_{g/1}\cong A_{fg}.
\]
This chapter uses that theorem; it does not re-migrate it.

\section{Closed subschemes in the affine case}
Let $A$ be a ring and $I\subseteq A$ an ideal.  The quotient map
\[
 A\twoheadrightarrow A/I
\]
induces
\[
 i:\Spec(A/I)\longrightarrow \Spec A.
\]
Its image is the closed subset
\[
 V(I)=\{\mathfrak p\in\Spec A:I\subseteq\mathfrak p\}.
\]
On points, $\mathfrak q\subseteq A/I$ is sent to its inverse image in $A$.

\begin{theorem}[Affine closed subscheme]\label{thm:vi22-affine-closed}
The morphism $\Spec(A/I)\to\Spec A$ is a homeomorphism onto $V(I)$.  For
$\mathfrak p\supseteq I$, the induced local ring is
\[
 \OO_{\Spec(A/I),\mathfrak p/I}\cong A_{\mathfrak p}/IA_{\mathfrak p}.
\]
\end{theorem}

\section{A closed subset is not a closed subscheme}
The topological set $V(I)$ depends only on $\sqrt I$, but the closed subscheme
$\Spec(A/I)$ depends on $I$ itself.  Hence
\[
 V(I)=V(\sqrt I)
\]
while
\[
 \Spec(A/I)\quad\text{and}\quad\Spec(A/\sqrt I)
\]
need not be isomorphic.  For example, $(x)$ and $(x^2)$ define the same closed point of
$\mathbb A^1_k$ but different closed subschemes.

\begin{definition}[Reduced induced closed subscheme]\label{def:vi22-reduced-closed}
For an affine closed set $V(I)\subseteq\Spec A$, its reduced induced closed subscheme is
$\Spec(A/\sqrt I)$.
\end{definition}

\section{Closed subschemes on a general scheme}
A closed subscheme $Z\hookrightarrow X$ is locally affine in the following precise sense.  For
an affine open cover $X=\bigcup U_i$ with $U_i=\Spec A_i$, each inverse image
$Z\cap U_i$ is of the form
\[
 \Spec(A_i/I_i)\hookrightarrow\Spec A_i,
\]
and the ideals agree after restriction to overlaps.  These compatible local quotients glue by
VI/20.  Equivalently, one may package the compatible ideals into an ideal sheaf
$\mathcal I\subseteq\OO_X$ and write
\[
 \OO_Z\cong (\OO_X/\mathcal I)|_Z.
\]
The full theory of quasi-coherent ideal sheaves is deferred to VI/33; only the local affine
compatibility needed for closed subschemes is used here.

\section{Closed immersions}
A morphism $i:Z\to X$ is a \emph{closed immersion} if it identifies $Z$ with a closed
subscheme of $X$.

\begin{theorem}[Affine-local criterion]\label{thm:vi22-closed-affine-local}
A morphism $i:Z\to X$ is a closed immersion if and only if $X$ has an affine open cover
$U_\alpha=\Spec A_\alpha$ such that each $i^{-1}(U_\alpha)$ is affine, say
$\Spec B_\alpha$, and each induced map
\[
 A_\alpha\longrightarrow B_\alpha
\]
is surjective.
\end{theorem}

\section{Restriction of closed subschemes to opens}
If $Z\hookrightarrow X$ is closed and $U\subseteq X$ is open, then
\[
 Z\cap U\hookrightarrow U
\]
is a closed subscheme.  In an affine chart $X=\Spec A$, $Z=\Spec(A/I)$ and
$U=D(f)$, one obtains
\[
 Z\cap D(f)\cong \Spec((A/I)_{\bar f})
 \cong \Spec(A_f/IA_f).
\]
This compatibility is the local mechanism by which closed subschemes glue.

\section{Open, closed, and locally closed immersions}
An \emph{immersion} is a morphism that factors as a closed immersion followed by an open
immersion.  Equivalently, it identifies the source with a closed subscheme of some open
subscheme of the target.  Its image is therefore locally closed as a topological subset.

\begin{remark}
This chapter treats the basic immersion formalism only.  Projective closed embeddings and their
homogeneous ideals are reserved for VI/38.
\end{remark}

\section{Nilpotent thickenings and the reduced subscheme}
For any affine $X=\Spec A$, the quotient by the nilradical gives
\[
 X_{\mathrm{red}}=\Spec(A/\sqrt{(0)})\hookrightarrow X.
\]
It has the same underlying topological space as $X$ but no nilpotents.  These affine reductions
are compatible on overlaps, so they define a canonical closed subscheme for every scheme.

\begin{proposition}[Universal property of reduction]\label{prop:vi22-reduction-universal}
If $T$ is reduced, every morphism $T\to X$ factors uniquely through
$X_{\mathrm{red}}\hookrightarrow X$.
\end{proposition}

\input{chapters/ch22_open_closed_subschemes/figures/figure_09}

\section{Equations and factorization}
For $X=\Spec A$, $Z=\Spec(A/I)$, and an affine scheme $T=\Spec B$, a morphism
$T\to X$ corresponding to $A\to B$ factors through $Z$ exactly when every element of $I$
maps to zero in $B$.  Thus closed subschemes represent the condition that specified functions
vanish scheme-theoretically, not merely pointwise.

\section{Chapter synthesis}
Open subschemes restrict the structure sheaf; closed subschemes quotient it by compatible local
ideals.  The distinction between a closed subset and a closed subscheme is the distinction between
radical information and full ideal-theoretic information.  Immersions combine the two operations,
and nilpotent thickenings show why scheme structure is strictly richer than topology.
