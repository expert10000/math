\begin{exercise}\label{exr:vi22-01}
For $U\subseteq X$ open, show every affine open of $U$ is also an open subscheme of $X$.
\end{exercise}

\begin{exercise}\label{exr:vi22-02}
Compute the closed subscheme of $\Spec\mathbb Z$ defined by $(n)$.
\end{exercise}

\begin{exercise}\label{exr:vi22-03}
Compare the closed subschemes of $\mathbb A^1_k$ defined by $(x)$, $(x^2)$, and $(x^3)$.
\end{exercise}

\begin{exercise}\label{exr:vi22-04}
For $I=(x,y)\subset k[x,y]$, compute $\OO_{V(I),0}$.
\end{exercise}

\begin{exercise}\label{exr:vi22-05}
Prove $V(I)=V(\sqrt I)$ as subsets of $\Spec A$.
\end{exercise}

\begin{exercise}\label{exr:vi22-06}
Give two ideals with the same radical but nonisomorphic quotient rings.
\end{exercise}

\begin{exercise}\label{exr:vi22-07}
Compute $V(x)\cap D(y)$ scheme-theoretically in $\Spec k[x,y]$.
\end{exercise}

\begin{exercise}\label{exr:vi22-08}
For $Z=V(I)$, prove $Z\cap D(f)$ is empty iff some power of $f$ belongs to $I$.
\end{exercise}

\begin{exercise}\label{exr:vi22-09}
Prove a quotient of a local ring by a proper ideal is local.
\end{exercise}

\begin{exercise}\label{exr:vi22-10}
Show a surjective ring map induces isomorphisms on residue fields at corresponding primes.
\end{exercise}

\begin{exercise}\label{exr:vi22-11}
Prove the reduction of a reduced scheme is itself.
\end{exercise}

\begin{exercise}\label{exr:vi22-12}
Compute the reduction of $\Spec k[x,y]/(x^2y^3)$.
\end{exercise}

\begin{exercise}\label{exr:vi22-13}
Show $(X_{
m red})_{
m red}=X_{
m red}$.
\end{exercise}

\begin{exercise}\label{exr:vi22-14}
Prove every map from a reduced affine scheme to $\Spec A$ kills the nilradical of $A$.
\end{exercise}

\begin{exercise}\label{exr:vi22-15}
For ideals $I,J$, compare $V(I+J)$ with $V(I)\cap V(J)$ topologically.
\end{exercise}

\begin{exercise}\label{exr:vi22-16}
For ideals $I,J$, compare $V(I\cap J)$ with $V(I)\cup V(J)$ topologically.
\end{exercise}

\begin{exercise}\label{exr:vi22-17}
Compute the scheme-theoretic intersection of the axes in $\mathbb A^2_k$.
\end{exercise}

\begin{exercise}\label{exr:vi22-18}
Compute the scheme-theoretic union of the axes in $\mathbb A^2_k$.
\end{exercise}

\begin{exercise}\label{exr:vi22-19}
Give an immersion whose image is not closed in the ambient scheme.
\end{exercise}

\begin{exercise}\label{exr:vi22-20}
Prove an open immersion is injective on points.
\end{exercise}

\begin{exercise}\label{exr:vi22-21}
Prove a closed immersion is injective on points.
\end{exercise}

\begin{exercise}\label{exr:vi22-22}
Show a morphism that is both open and closed immersion identifies the source with an open-and-closed subscheme.
\end{exercise}

\begin{exercise}\label{exr:vi22-23}
For $I\subset J$, identify the ideal of $\Spec(A/J)$ inside $\Spec(A/I)$.
\end{exercise}

\begin{exercise}\label{exr:vi22-24}
Check that localization commutes with quotient: $(A/I)_f\cong A_f/IA_f$.
\end{exercise}

\begin{exercise}\label{exr:vi22-25}
Explain why the open complement of a closed subset has a canonical open subscheme structure even when the closed subset has many closed subscheme structures.
\end{exercise}

\begin{exercise}\label{exr:vi22-26}
Give an example of a nilpotent thickening with unchanged underlying topological space.
\end{exercise}
