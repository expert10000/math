% VI36 detailed exercise solutions -- VI/24+ proof-depth standard.

% VI36-DETAILED-EXERCISE-01
\paragraph{Solution VI.36.E1.}
Let \(X=K^{n+1}\setminus\{0\}\). Define \(a\sim b\) when \(b=\lambda a\) for some
\(\lambda\in K^\times\). Reflexivity holds with \(\lambda=1\). If \(b=\lambda a\), then
\[
a=\lambda^{-1}b,
\]
so symmetry holds. If \(b=\lambda a\) and \(c=\mu b\), then
\[
c=(\mu\lambda)a
\]
with \(\mu\lambda\in K^\times\), proving transitivity. Hence \(\sim\) is an equivalence relation.

% VI36-DETAILED-EXERCISE-02
\paragraph{Solution VI.36.E2.}
A point \([a_0:a_1:a_2:a_3]\) lies in \(U_i=D_+(x_i)\) exactly when \(a_i\neq0\). For
\[
[2:0:3:0]
\]
the zeroth and second coordinates are nonzero, while the first and third vanish. Therefore the point lies exactly in
\[
\boxed{U_0\text{ and }U_2}.
\]

% VI36-DETAILED-EXERCISE-03
\paragraph{Solution VI.36.E3.}
On
\[
U_0=D_+(x_0)
\]
the degree-zero chart ring is
\[
k[x_1/x_0,\ldots,x_n/x_0].
\]
Thus on \(K\)-valued points the chart map is
\[
[a_0:\cdots:a_n]
\longmapsto
\left(\frac{a_1}{a_0},\ldots,\frac{a_n}{a_0}\right),
\qquad a_0\neq0.
\]
Its inverse is
\[
(t_1,\ldots,t_n)\longmapsto[1:t_1:\cdots:t_n].
\]
These maps are inverse because projective rescaling by \(a_0^{-1}\) normalizes the zeroth coordinate to \(1\).

% VI36-DETAILED-EXERCISE-04
\paragraph{Solution VI.36.E4.}
On \(U_0\) write
\[
u=\frac{x_1}{x_0},
\qquad
v=\frac{x_2}{x_0}.
\]
The overlap with \(U_1\) is the principal open \(u\neq0\). In \(U_1\)-coordinates,
\[
r=\frac{x_0}{x_1}=\frac1u,
\qquad
s=\frac{x_2}{x_1}
=\frac{x_2/x_0}{x_1/x_0}
=\frac vu.
\]
Hence
\[
\boxed{(u,v)\longmapsto(1/u,v/u)}.
\]
The inverse is \((r,s)\mapsto(1/r,s/r)\).

% VI36-DETAILED-EXERCISE-05
\paragraph{Solution VI.36.E5.}
For \(\mathbb P^1\),
\[
U_0\cong\Spec k[t],
\qquad
t=x_1/x_0,
\]
and
\[
U_1\cong\Spec k[u],
\qquad
u=x_0/x_1.
\]
The overlap requires both \(x_0\) and \(x_1\) nonzero, so in the first chart \(t\) is invertible:
\[
U_0\cap U_1\cong\Spec k[t,t^{-1}]=\mathbb G_m.
\]
Since
\[
u=\frac{x_0}{x_1}=t^{-1},
\]
the gluing map is inversion.

% VI36-DETAILED-EXERCISE-06
\paragraph{Solution VI.36.E6.}
Let \(S=k[x_0]\) with the standard grading. The homogeneous primes are \((0)\) and \((x_0)\), while
\[
S_+=(x_0).
\]
Proj excludes primes containing \(S_+\), so only \((0)\) remains. Moreover
\[
D_+(x_0)=\Proj S
\]
and
\[
S_{(x_0)}=(k[x_0]_{x_0})_0=k.
\]
Therefore
\[
\boxed{\mathbb P^0_k\cong\Spec k}.
\]

% VI36-DETAILED-EXERCISE-07
\paragraph{Solution VI.36.E7.}
The complement of \(U_0=D_+(x_0)\) is
\[
V_+(x_0).
\]
A field-valued point there has the form
\[
[0:a_1:\cdots:a_n]
\]
with not all \(a_i\) zero. Removing the forced zero coordinate gives
\[
[a_1:\cdots:a_n]\in\mathbb P^{n-1}.
\]
The same identification follows from
\[
k[x_0,\ldots,x_n]/(x_0)\cong k[x_1,\ldots,x_n].
\]
Thus the underlying projective locus is naturally \(\mathbb P^{n-1}\).

% VI36-DETAILED-EXERCISE-08
\paragraph{Solution VI.36.E8.}
Each standard chart
\[
U_i\cong\mathbb A^n_k
\]
has dimension \(n\). Since \(U_0\) is a nonempty open subset of \(\mathbb P^n_k\),
\[
\dim\mathbb P^n_k\ge n.
\]
Conversely every point has a neighborhood isomorphic to an open subset of \(\mathbb A^n_k\), so the local
dimension cannot exceed \(n\). By the scheme-dimension results of VI/30,
\[
\boxed{\dim\mathbb P^n_k=n}.
\]

% VI36-DETAILED-EXERCISE-09
\paragraph{Solution VI.36.E9.}
The equations
\[
x_0=x_1=x_2=0
\]
leave the homogeneous coordinates
\[
[x_3:\cdots:x_n].
\]
Not all of them may vanish. Hence the locus is canonically
\[
\boxed{\mathbb P^{n-3}_k}.
\]

% VI36-DETAILED-EXERCISE-10
\paragraph{Solution VI.36.E10.}
Represent the two points by
\[
e_0=(1,0,0),
\qquad
e_1=(0,1,0).
\]
Their vector-space span consists of
\[
\lambda e_0+\mu e_1=(\lambda,\mu,0).
\]
Projectivizing gives
\[
\boxed{
[\lambda:\mu]\longmapsto[\lambda:\mu:0].
}
\]
Thus the line through the two points is the linear projective locus \(x_2=0\).

% VI36-DETAILED-EXERCISE-11
\paragraph{Solution VI.36.E11.}
On \(U_0\cong\Spec\mathbb R[t]\), the closed point is the maximal ideal
\[
(t^2+1).
\]
Its residue field is
\[
\kappa(x)
=
\mathbb R[t]/(t^2+1).
\]
Sending the class of \(t\) to \(i\) identifies this field with \(\mathbb C\). Hence
\[
\boxed{\kappa(x)\cong\mathbb C}.
\]

% VI36-DETAILED-EXERCISE-12
\paragraph{Solution VI.36.E12.}
The two standard charts of \(\mathbb P^1_k\) are
\[
\Spec k[t]
\quad\text{and}\quad
\Spec k[u],
\]
glued on
\[
\Spec k[t,t^{-1}]
\]
by \(u=t^{-1}\). Base-changing to \(K\) tensors each coordinate ring with \(K\):
\[
k[t]\otimes_kK\cong K[t],
\qquad
k[t,t^{-1}]\otimes_kK\cong K[t,t^{-1}].
\]
The transition remains \(u=t^{-1}\). These are exactly the standard charts and gluing of \(\mathbb P^1_K\), so
\[
\boxed{\mathbb P^1_k\times_kK\cong\mathbb P^1_K}.
\]

% VI36-DETAILED-EXERCISE-13
\paragraph{Solution VI.36.E13.}
By definition
\[
\mathcal O(1)=\widetilde{S(1)}.
\]
On \(U_i\),
\[
\Gamma(U_i,\mathcal O(1))
=
S(1)_{(x_i)}
=
(S_{x_i})_1.
\]
Every degree-one element of \(S_{x_i}\) can be written uniquely as
\[
a\,x_i,
\qquad
a\in(S_{x_i})_0=S_{(x_i)}.
\]
Thus \(x_i\) freely generates the local module:
\[
\boxed{\mathcal O(1)|_{U_i}\cong\mathcal O_{U_i}\cdot x_i}.
\]

% VI36-DETAILED-EXERCISE-14
\paragraph{Solution VI.36.E14.}
On \(U_i\), use the frame
\[
e_i=x_i^d.
\]
On \(U_i\cap U_j\),
\[
x_i^d
=
\left(\frac{x_i}{x_j}\right)^d x_j^d.
\]
Therefore
\[
\boxed{
e_i
=
\left(\frac{x_i}{x_j}\right)^d e_j.
}
\]
The transition factor is a unit because both \(x_i\) and \(x_j\) are nonzero on the overlap.

% VI36-DETAILED-EXERCISE-15
\paragraph{Solution VI.36.E15.}
The global-section theorem gives
\[
\Gamma(\mathbb P^n,\mathcal O(1))
\cong
k[x_0,\ldots,x_n]_1.
\]
The degree-one piece has basis
\[
x_0,\ldots,x_n.
\]
Hence
\[
\boxed{
\dim_k\Gamma(\mathbb P^n,\mathcal O(1))=n+1.
}
\]

% VI36-DETAILED-EXERCISE-16
\paragraph{Solution VI.36.E16.}
Again,
\[
\Gamma(\mathbb P^n,\mathcal O(2))
\cong
k[x_0,\ldots,x_n]_2.
\]
A basis is given by degree-two monomials
\[
x_ix_j,\qquad 0\le i\le j\le n.
\]
Their number is the number of multisets of size \(2\) chosen from \(n+1\) variables:
\[
\boxed{
\dim_k\Gamma(\mathbb P^n,\mathcal O(2))
=
\binom{n+2}{2}.
}
\]

% VI36-DETAILED-EXERCISE-17
\paragraph{Solution VI.36.E17.}
Let
\[
F=x_0x_2-x_1^2.
\]
On \(U_0\), divide by \(x_0^2\):
\[
\frac{F}{x_0^2}
=
\frac{x_2}{x_0}
-
\left(\frac{x_1}{x_0}\right)^2.
\]
On \(U_1\), divide by \(x_1^2\):
\[
\frac{F}{x_1^2}
=
\frac{x_0}{x_1}\frac{x_2}{x_1}-1.
\]
On \(U_2\), divide by \(x_2^2\):
\[
\frac{F}{x_2^2}
=
\frac{x_0}{x_2}
-
\left(\frac{x_1}{x_2}\right)^2.
\]
These are the three dehomogenized chart equations.

% VI36-DETAILED-EXERCISE-18
\paragraph{Solution VI.36.E18.}
Use affine coordinates
\[
x=x_1/x_0,
\qquad
y=x_2/x_0.
\]
The equation
\[
y-x^2=0
\]
has total degree \(2\). Clearing denominators gives
\[
\boxed{x_0x_2-x_1^2=0}.
\]
At infinity \(x_0=0\), so the equation becomes
\[
-x_1^2=0.
\]
Hence \(x_1=0\), while \(x_2\neq0\). The unique projective point at infinity is
\[
\boxed{[0:0:1]}.
\]

% VI36-DETAILED-EXERCISE-19
\paragraph{Solution VI.36.E19.}
The degree-\(d\) Veronese map lists all degree-\(d\) monomials. Replacing a representative
\[
a=(a_0,\ldots,a_n)
\]
by \(\lambda a\) multiplies every degree-\(d\) monomial by the same scalar \(\lambda^d\). Therefore the resulting
target homogeneous coordinates differ by one common nonzero scalar. Hence the projective point is unchanged.

% VI36-DETAILED-EXERCISE-20
\paragraph{Solution VI.36.E20.}
For
\[
[X:Y:Z]=[s^2:st:t^2],
\]
we have
\[
XZ-Y^2
=
s^2t^2-(st)^2
=
0.
\]
Thus every quadratic Veronese point lies on the conic
\[
\boxed{XZ-Y^2=0}.
\]

% VI36-DETAILED-EXERCISE-21
\paragraph{Solution VI.36.E21.}
The Segre coordinates are \(x_i y_j\). If
\[
x_i\mapsto\lambda x_i,
\qquad
y_j\mapsto\mu y_j,
\]
then every product becomes
\[
x_i y_j\mapsto(\lambda\mu)x_i y_j.
\]
All target coordinates are multiplied by the same nonzero scalar \(\lambda\mu\), so the projective image is
unchanged.

% VI36-DETAILED-EXERCISE-22
\paragraph{Solution VI.36.E22.}
For
\[
([s:t],[u:v])
\]
the Segre coordinates are
\[
[su:sv:tu:tv].
\]
Writing them as \(z_{00},z_{01},z_{10},z_{11}\), the \(2\times2\) minor is
\[
z_{00}z_{11}-z_{01}z_{10}.
\]
Substitution gives
\[
(su)(tv)-(sv)(tu)=stuv-stuv=0.
\]
Hence the image satisfies the rank-one relation.

% VI36-DETAILED-EXERCISE-23
\paragraph{Solution VI.36.E23.}
Set \(d=0\) in the global-section theorem:
\[
\Gamma(\mathbb P^n,\mathcal O)
\cong
k[x_0,\ldots,x_n]_0.
\]
The degree-zero part of the standard polynomial ring consists exactly of constants, so
\[
\boxed{\Gamma(\mathbb P^n,\mathcal O)=k}.
\]

% VI36-DETAILED-EXERCISE-24
\paragraph{Solution VI.36.E24.}
For an affine scheme \(X=\Spec A\), the affine reconstruction theorem identifies
\[
X\cong\Spec\Gamma(X,\mathcal O_X).
\]
There is no such reconstruction statement for arbitrary non-affine schemes. Although
\[
\Gamma(\mathbb P^n,\mathcal O)=k,
\]
for \(n>0\) we have
\[
\dim\mathbb P^n=n>0,
\qquad
\dim\Spec k=0.
\]
Therefore the two schemes cannot be isomorphic.

% VI36-DETAILED-EXERCISE-25
\paragraph{Solution VI.36.E25.}
Let \(A\in\operatorname{GL}_{n+1}(k)\). If \(v\neq0\), then \(Av\neq0\). If \(v'=\lambda v\), then
\[
Av'=\lambda Av,
\]
so the rule
\[
[v]\longmapsto[Av]
\]
is independent of the chosen representative. Its inverse is induced by \(A^{-1}\), hence it is an automorphism.

If \(A=\lambda I\) is scalar, then
\[
[Av]=[\lambda v]=[v]
\]
for every projective point. Thus scalar matrices act trivially, and the action factors through
\[
\operatorname{PGL}_{n+1}(k)
=
\operatorname{GL}_{n+1}(k)/k^\times.
\]

% VI36-DETAILED-EXERCISE-26
\paragraph{Solution VI.36.E26.}
A point of \(\mathbb P^n(\mathbb F_q)\) is a one-dimensional subspace of \(\mathbb F_q^{n+1}\). There are
\[
q^{n+1}-1
\]
nonzero vectors. Each one-dimensional subspace contains exactly
\[
q-1
\]
nonzero vectors. Distinct one-dimensional subspaces partition the nonzero vectors. Therefore
\[
\#\mathbb P^n(\mathbb F_q)
=
\frac{q^{n+1}-1}{q-1}.
\]
Using the geometric-series identity,
\[
\boxed{
\#\mathbb P^n(\mathbb F_q)
=
1+q+\cdots+q^n.
}
\]
