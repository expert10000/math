% VI36-DETAILED-PROBLEM-15
\begin{problem}[VI.36.P15: Spans and intersections]\label{prob:vi36-15}
Let \(A,B\subseteq k^{n+1}\) be nonzero vector subspaces. Write
\[
L=\mathbb P(A),\qquad M=\mathbb P(B)\subseteq\mathbb P^n_k.
\]
Express their projective span and intersection in terms of \(A+B\) and \(A\cap B\). Derive the dimension formula
when \(A\cap B\neq0\).

\textbf{Provenance.} Canonical VI/36 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
The smallest vector subspace containing both \(A\) and \(B\) is \(A+B\). Therefore the smallest projective
linear subspace containing both projectivizations is
\[
\boxed{
\langle\mathbb P(A),\mathbb P(B)\rangle
=
\mathbb P(A+B).
}
\]

A projective point belongs to both \(\mathbb P(A)\) and \(\mathbb P(B)\) precisely when the corresponding
one-dimensional vector subspace lies in \(A\cap B\). Hence, if \(A\cap B\neq0\),
\[
\boxed{
\mathbb P(A)\cap\mathbb P(B)
=
\mathbb P(A\cap B).
}
\]
If \(A\cap B=0\), the projective intersection is empty.

Write
\[
r=\dim\mathbb P(A)=\dim A-1,
\qquad
s=\dim\mathbb P(B)=\dim B-1.
\]
Then
\[
\dim(A+B)
=
\dim A+\dim B-\dim(A\cap B).
\]
After subtracting \(1\),
\[
\dim\mathbb P(A+B)
=
r+s+1-\dim\mathbb P(A\cap B).
\]
Equivalently,
\[
\boxed{
\dim\mathbb P(A\cap B)
=
r+s-\dim\mathbb P(A+B).
}
\]
Since \(\mathbb P(A+B)\subseteq\mathbb P^n\),
\[
\dim(\mathbb P(A)\cap\mathbb P(B))
\ge r+s-n
\]
whenever the intersection is nonempty.
\end{solution}
\fi
