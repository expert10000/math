% VI36-DETAILED-PROBLEM-19
\begin{problem}[VI.36.P19: Rational normal curve on a chart]\label{prob:vi36-19}
For the degree-\(d\) Veronese map
\[
\nu_d:\mathbb P^1\to\mathbb P^d,
\qquad
[s:t]\mapsto[s^d:s^{d-1}t:\cdots:t^d],
\]
compute the map on \(s\neq0\) and prove it is injective on field-valued points.

\textbf{Provenance.} Canonical VI/36 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
On the chart \(s\neq0\), write
\[
[s:t]=[1:u].
\]
Then
\[
\boxed{
\nu_d([1:u])
=
[1:u:u^2:\cdots:u^d].
}
\]
In the target chart where the first coordinate is nonzero, the first affine coordinate after normalization is
precisely \(u\). Thus two source points in this chart with the same image must have the same \(u\).

The remaining source point is
\[
[0:1],
\]
whose image is
\[
[0:\cdots:0:1].
\]
This target point is not in the first target chart, so it cannot equal the image of a point with \(s\neq0\).
Therefore \(\nu_d\) is injective on field-valued points.

VI/38 will prove the stronger scheme-theoretic statement that this coordinate morphism is a closed embedding.
\end{solution}
\fi
