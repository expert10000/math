% VI36-DETAILED-PROBLEM-02
\begin{problem}[VI.36.P2: Three-chart atlas for \(\mathbb P^2\)]\label{prob:vi36-02}
Write all three transition maps among the standard affine charts of \(\mathbb P^2\) and verify the cocycle identity on the triple overlap.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Use
\[
U_0:(u,v)=\left(\frac{x_1}{x_0},\frac{x_2}{x_0}\right),
\]
\[
U_1:(r,s)=\left(\frac{x_0}{x_1},\frac{x_2}{x_1}\right),
\]
and
\[
U_2:(p,q)=\left(\frac{x_0}{x_2},\frac{x_1}{x_2}\right).
\]

On \(U_0\cap U_1\), \(u\neq0\) and
\[
(r,s)=\left(\frac1u,\frac vu\right).
\]
On \(U_1\cap U_2\), \(s\neq0\) and
\[
(p,q)=\left(\frac rs,\frac1s\right).
\]
On \(U_0\cap U_2\), \(v\neq0\) and
\[
(p,q)=\left(\frac1v,\frac uv\right).
\]

Now compose the first two maps:
\[
(u,v)
\longmapsto
\left(\frac1u,\frac vu\right)
\longmapsto
\left(
\frac{(1/u)}{(v/u)},
\frac1{v/u}
\right)
=
\left(\frac1v,\frac uv\right),
\]
which is exactly the direct \(U_0\to U_2\) transition. Hence the cocycle identity holds on
\[
U_0\cap U_1\cap U_2.
\]
\end{solution}
\fi
