% VI36-DETAILED-PROBLEM-20
\begin{problem}[VI.36.P20: The main Segre affine chart]\label{prob:vi36-20}
For the Segre map
\[
\sigma:\mathbb P^m\times\mathbb P^n\to\mathbb P^{(m+1)(n+1)-1},
\]
compute the source chart \(x_0\neq0,\ y_0\neq0\) and show its image inside \(z_{00}\neq0\) is isomorphic to
\(\mathbb A^{m+n}\).

\textbf{Provenance.} Canonical VI/36 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Normalize the source coordinates:
\[
[x_0:\cdots:x_m]
=
[1:a_1:\cdots:a_m],
\]
\[
[y_0:\cdots:y_n]
=
[1:b_1:\cdots:b_n].
\]
Thus the source chart is
\[
\mathbb A^m\times\mathbb A^n
\cong
\mathbb A^{m+n}.
\]

The Segre coordinates are
\[
z_{00}=1,
\qquad
z_{i0}=a_i,
\qquad
z_{0j}=b_j,
\qquad
z_{ij}=a_i b_j.
\]
Hence in the target chart \(z_{00}\neq0\), all coordinates \(z_{ij}\) with \(i,j>0\) are determined by the
\(m+n\) free coordinates
\[
z_{i0}=a_i,
\qquad
z_{0j}=b_j.
\]
Conversely those \(m+n\) coordinates recover the source point uniquely. Therefore the chart image is the graph of
the equations
\[
z_{ij}=z_{i0}z_{0j},
\]
and is isomorphic to
\[
\boxed{\mathbb A^{m+n}}.
\]
This is the affine-chart form of the rank-one matrix description; the global closed-embedding theorem remains for VI/38.
\end{solution}
\fi
