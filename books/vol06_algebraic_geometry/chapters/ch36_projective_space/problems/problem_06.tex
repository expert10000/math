% VI36-DETAILED-PROBLEM-06
\begin{problem}[VI.36.P6: Base change of a non-rational point]\label{prob:vi36-06}
Let \(x\) be the closed point of \(\mathbb P^1_k\) associated with a separable irreducible polynomial \(f\) of degree \(d\). Describe what happens after base change to a splitting field \(K\).
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
On the affine chart, the point is
\[
x=\Spec k[t]/(f).
\]
After base change to \(K\), its fiber is
\[
x_K
=
\Spec\bigl(K\otimes_k k[t]/(f)\bigr)
\cong
\Spec K[t]/(f).
\]
Because \(K\) is a splitting field and \(f\) is separable,
\[
f(t)=c\prod_{i=1}^d(t-a_i)
\]
with distinct \(a_i\in K\). The ideals \((t-a_i)\) are pairwise comaximal, so the Chinese remainder theorem gives
\[
K[t]/(f)
\cong
\prod_{i=1}^dK.
\]
Therefore
\[
\boxed{
x_K\cong\coprod_{i=1}^d\Spec K.
}
\]
The points are the \(K\)-rational projective points
\[
\boxed{[1:a_i]}.
\]
Thus one degree-\(d\) closed point splits into \(d\) geometric points after passing to a splitting field.
\end{solution}
\fi
