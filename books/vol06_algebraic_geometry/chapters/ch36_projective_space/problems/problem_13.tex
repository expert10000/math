% VI36-DETAILED-PROBLEM-13
\begin{problem}[VI.36.P13: The \(\operatorname{PGL}\)-action]\label{prob:vi36-13}
Show that \(\operatorname{GL}_{n+1}(k)\) acts on \(\mathbb P^n_k(k)\), identify the kernel, and deduce an action of
\(\operatorname{PGL}_{n+1}(k)\).

\textbf{Provenance.} Canonical VI/36 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
For \(A\in\operatorname{GL}_{n+1}(k)\), define
\[
A\cdot[v]=[Av].
\]
If \(v'=\lambda v\), then
\[
Av'=\lambda Av,
\]
so the projective class is independent of the representative. The identity matrix acts trivially, and
\[
A\cdot(B\cdot[v])=[ABv],
\]
so this is a group action.

Every scalar matrix \(\lambda I\) acts trivially. Conversely, suppose \(A\) acts trivially on every point. Then each
basis vector \(e_i\) is an eigenvector:
\[
Ae_i=\lambda_i e_i.
\]
Now the point \([e_i+e_j]\) is fixed, so
\[
[\lambda_i e_i+\lambda_j e_j]
=
[e_i+e_j].
\]
Hence \(\lambda_i=\lambda_j\) for all \(i,j\). Thus \(A=\lambda I\). Therefore the kernel is exactly
\[
k^\times I.
\]
The action factors faithfully through
\[
\boxed{
\operatorname{PGL}_{n+1}(k)
=
\operatorname{GL}_{n+1}(k)/k^\times.
}
\]
\end{solution}
\fi
