\begin{problem}[VI.36.P17: Transition cocycle for \(\mathcal O(d)\)]\label{prob:vi36-17}
Using the standard frames \(e_i=x_i^d\) of \(\mathcal O(d)\) on \(U_i\), prove the transition factors satisfy the cocycle
identity on triple overlaps.

\textbf{Provenance.} Canonical VI/36 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
On \(U_i\cap U_j\),
\[
e_i
=
g_{ij}e_j,
\qquad
g_{ij}
=
\left(\frac{x_i}{x_j}\right)^d.
\]
These are units on the overlap. Clearly
\[
g_{ii}=1
\]
and
\[
g_{ji}=g_{ij}^{-1}.
\]

On \(U_i\cap U_j\cap U_k\),
\[
g_{ij}g_{jk}
=
\left(\frac{x_i}{x_j}\right)^d
\left(\frac{x_j}{x_k}\right)^d
=
\left(\frac{x_i}{x_k}\right)^d
=
g_{ik}.
\]
Therefore
\[
\boxed{
g_{ij}g_{jk}=g_{ik}.
}
\]
This is the Čech-style cocycle identity that permits the local free rank-one modules to glue. It is the explicit
projective-space realization of the abstract shift construction from VI/35.
\end{solution}
\fi
