\begin{problem}[VI.36.P14: Counting finite-field points]\label{prob:vi36-14}
For \(k=\mathbb F_q\), compute \(\#\mathbb P^n(k)\).

\textbf{Provenance.} Canonical VI/36 extension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Projective points are one-dimensional subspaces of
\[
\mathbb F_q^{n+1}.
\]
There are
\[
q^{n+1}-1
\]
nonzero vectors. Each projective line through the origin contains exactly
\[
q-1
\]
nonzero vectors. These sets partition the nonzero vectors. Therefore
\[
\#\mathbb P^n(\mathbb F_q)
=
\frac{q^{n+1}-1}{q-1}.
\]
Expanding the geometric series gives
\[
\boxed{
\#\mathbb P^n(\mathbb F_q)
=
1+q+\cdots+q^n.
}
\]
For \(n=1\), this recovers the familiar \(q+1\) points of the projective line over \(\mathbb F_q\).
\end{solution}
\fi
