\begin{problem}[VI.36.P4: Dimension and codimension of linear spaces]\label{prob:vi36-04}
Let \(L\subset\mathbb P^n\) be the projective linear locus cut out by \(r\) independent linear forms. Compute its dimension and codimension.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Choose a linear coordinate change sending the \(r\) independent equations to
\[
x_0=\cdots=x_{r-1}=0.
\]
The surviving coordinates are
\[
[x_r:\cdots:x_n],
\]
so
\[
L\cong\mathbb P^{n-r}.
\]
By the dimension theorem for projective space,
\[
\dim L=n-r.
\]
Since
\[
\dim\mathbb P^n=n,
\]
the codimension is
\[
n-(n-r)=r.
\]
Hence
\[
\boxed{
\dim L=n-r,
\qquad
\operatorname{codim}(L,\mathbb P^n)=r.
}
\]
\end{solution}
\fi
