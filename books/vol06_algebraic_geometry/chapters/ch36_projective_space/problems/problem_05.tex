% VI36-DETAILED-PROBLEM-05
\begin{problem}[VI.36.P5: A closed point over a nonclosed field]\label{prob:vi36-05}
For an irreducible polynomial \(f(t)\in k[t]\) of degree \(d>1\), construct the corresponding closed point of \(\mathbb P^1_k\) in the chart \(U_0\) and compute its residue field.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Identify
\[
U_0\cong\Spec k[t],
\qquad
t=x_1/x_0.
\]
Since \(f(t)\) is irreducible, the ideal
\[
(f)\subset k[t]
\]
is maximal. Hence it determines a closed point
\[
x\in U_0\subseteq\mathbb P^1_k.
\]
Its residue field is
\[
\boxed{
\kappa(x)=k[t]/(f),
}
\]
which has degree
\[
[\kappa(x):k]=d.
\]

If
\[
f(t)=a_dt^d+\cdots+a_0,
\]
its degree-\(d\) homogenization is
\[
F(x_0,x_1)
=
a_dx_1^d+a_{d-1}x_1^{d-1}x_0+\cdots+a_0x_0^d.
\]
On \(U_0\), dividing by \(x_0^d\) recovers \(f(x_1/x_0)\). Thus the same closed point is visible in homogeneous
coordinates through \(F\), without yet invoking the full VI/37 projective-subscheme theory.
\end{solution}
\fi
