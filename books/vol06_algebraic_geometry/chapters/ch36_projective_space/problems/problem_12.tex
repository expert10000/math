% VI36-DETAILED-PROBLEM-12
\begin{problem}[VI.36.P12: Why projective space is not affine]\label{prob:vi36-12}
For \(n>0\), use global sections and dimension to prove that \(\mathbb P^n_k\) is not affine.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Suppose
\[
\mathbb P^n_k
\]
were affine. For every affine scheme \(X\), affine reconstruction gives
\[
X\cong\Spec\Gamma(X,\mathcal O_X).
\]
The global-section theorem gives
\[
\Gamma(\mathbb P^n_k,\mathcal O)=k.
\]
Hence affineness would force
\[
\mathbb P^n_k\cong\Spec k.
\]
But
\[
\dim\mathbb P^n_k=n>0,
\]
while
\[
\dim\Spec k=0.
\]
This contradiction proves
\[
\boxed{\mathbb P^n_k\text{ is not affine for }n>0}.
\]
\end{solution}
\fi
