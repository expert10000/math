\begin{problem}[VI.36.P11: Projective completion of a hyperbola]\label{prob:vi36-11}
Homogenize \(xy-1=0\) in \(\mathbb A^2\) and determine its set-theoretic points at infinity.
\end{problem}

\ifdefined\FullProblemDossiers
\begin{solution}
Again let
\[
x=x_1/x_0,
\qquad
y=x_2/x_0.
\]
Homogenizing to degree \(2\) gives
\[
\boxed{x_1x_2-x_0^2=0}.
\]
At infinity \(x_0=0\), so
\[
x_1x_2=0.
\]
Because \((x_1,x_2)\neq(0,0)\), exactly two possibilities remain:
\[
\boxed{[0:1:0]}
\qquad\text{and}\qquad
\boxed{[0:0:1]}.
\]
Thus the affine hyperbola acquires two projective points at infinity.
\end{solution}
\fi
