\begin{exercise}[Homogeneous scaling]\label{exr:vi36-01}
Prove that the relation \((a_0,\ldots,a_n)\sim(\lambda a_0,\ldots,\lambda a_n)\) for \(\lambda\in K^\times\) is an equivalence relation on \(K^{n+1}\setminus\{0\}\).
\end{exercise}

\begin{exercise}[Chart membership]\label{exr:vi36-02}
Determine exactly which standard charts contain \([2:0:3:0]\in\mathbb P^3_k\), assuming \(2,3\ne0\) in \(k\).
\end{exercise}

\begin{exercise}[The \(U_0\) chart]\label{exr:vi36-03}
Write the explicit isomorphism \(U_0\to\mathbb A^n_k\) and its inverse on \(K\)-valued points.
\end{exercise}

\begin{exercise}[Transition in \(\mathbb P^2\)]\label{exr:vi36-04}
Write the change of coordinates from \(U_0\) coordinates \((u,v)=(x_1/x_0,x_2/x_0)\) to \(U_1\) coordinates on \(u\ne0\).
\end{exercise}

\begin{exercise}[The overlap of the two \(\mathbb P^1\) charts]\label{exr:vi36-05}
Show that \(U_0\cap U_1\cong\mathbb G_m\) and that the gluing map is inversion.
\end{exercise}

\begin{exercise}[Projective zero-space]\label{exr:vi36-06}
Prove directly from Proj that \(\mathbb P^0_k\cong\Spec k\).
\end{exercise}

\begin{exercise}[Hyperplane at infinity]\label{exr:vi36-07}
Show that \(\mathbb P^n\setminus U_0\) has underlying projective locus \(\mathbb P^{n-1}\).
\end{exercise}

\begin{exercise}[Dimension]\label{exr:vi36-08}
Use the standard affine cover to prove \(\dim\mathbb P^n_k=n\).
\end{exercise}

\begin{exercise}[A linear subspace]\label{exr:vi36-09}
Identify the projective locus \(x_0=x_1=x_2=0\) in \(\mathbb P^n\) when \(n\ge3\).
\end{exercise}

\begin{exercise}[Line through two points]\label{exr:vi36-10}
Find a parametrization of the projective line through \([1:0:0]\) and \([0:1:0]\) in \(\mathbb P^2\).
\end{exercise}

\begin{exercise}[Closed non-rational point]\label{exr:vi36-11}
Compute the residue field of the closed point of \(\mathbb P^1_{\mathbb R}\) given on \(U_0\) by \(t^2+1\).
\end{exercise}

\begin{exercise}[Base extension]\label{exr:vi36-12}
Verify the base-change theorem for \(\mathbb P^1\) by gluing the two base-changed affine charts.
\end{exercise}

\begin{exercise}[Trivializing \(\mathcal O(1)\)]\label{exr:vi36-13}
Show that \(x_i\) gives a local generator of \(\mathcal O(1)\) on \(U_i\).
\end{exercise}

\begin{exercise}[Transition for \(\mathcal O(d)\)]\label{exr:vi36-14}
Derive the transition factor between the generators \(x_i^d\) and \(x_j^d\) on \(U_i\cap U_j\).
\end{exercise}

\begin{exercise}[Global linear sections]\label{exr:vi36-15}
Compute \(\dim_k\Gamma(\mathbb P^n,\mathcal O(1))\).
\end{exercise}

\begin{exercise}[Global quadratic sections]\label{exr:vi36-16}
Compute \(\dim_k\Gamma(\mathbb P^n,\mathcal O(2))\).
\end{exercise}

\begin{exercise}[Dehomogenization]\label{exr:vi36-17}
Dehomogenize \(x_0x_2-x_1^2\) on each of the three standard charts of \(\mathbb P^2\).
\end{exercise}

\begin{exercise}[Homogenization]\label{exr:vi36-18}
Homogenize \(y-x^2\in k[x,y]\) to a degree-two polynomial in \(k[x_0,x_1,x_2]\), and determine its set-theoretic points at infinity.
\end{exercise}

\begin{exercise}[Veronese well-definedness]\label{exr:vi36-19}
Prove that the degree-\(d\) Veronese coordinate formula is independent of the representative of homogeneous coordinates.
\end{exercise}

\begin{exercise}[Quadratic Veronese equation]\label{exr:vi36-20}
Verify directly that \([s^2:st:t^2]\) satisfies \(XZ-Y^2=0\).
\end{exercise}

\begin{exercise}[Segre well-definedness]\label{exr:vi36-21}
Show that independent rescaling in the two factors changes all Segre coordinates by one common scalar.
\end{exercise}

\begin{exercise}[Segre relation]\label{exr:vi36-22}
Verify the \(2\times2\) minor relation for the Segre map \(\mathbb P^1\times\mathbb P^1\to\mathbb P^3\).
\end{exercise}

\begin{exercise}[Global functions]\label{exr:vi36-23}
Use the theorem on \(\mathcal O(d)\) to recover \(\Gamma(\mathbb P^n,\mathcal O)=k\).
\end{exercise}

\begin{exercise}[Affine versus projective]\label{exr:vi36-24}
Explain why \(\Gamma(\mathbb P^n,\mathcal O)=k\) does not imply \(\mathbb P^n\cong\Spec k\) for \(n>0\).
\end{exercise}
