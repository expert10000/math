\chapter{Associated Primes}
\label{ch:v17}

Associated primes identify prime annihilators of individual module elements. They refine support by detecting the primes that occur as embedded or minimal algebraic components.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item identify associated primes from annihilators of module elements;
\item compute associated primes in basic quotient and primary-type examples;
\item prove that associated primes lie inside the support;
\item use zero divisors on a module to detect unions of associated primes under Noetherian hypotheses;
\item analyze behavior of associated primes under localization;
\item distinguish minimal primes of support from embedded associated primes in representative examples;
\end{itemize}

\section{Definition}
A prime $\mathfrak p$ is associated to $M$ if $\mathfrak p=\operatorname{Ann}(m)$ for some $m\in M$.

\section{Cyclic submodules}
Associated primes arise exactly from embedded copies of $A/\mathfrak p$ inside $M$.

\end{example}

\section{Core structural results}

\begin{theorem}[Associated prime existence]\label{thm:v17-01}
If $M\ne0$ is finitely generated over a Noetherian ring, then $M$ has an associated prime.
\begin{proof}
Choose an annihilator maximal among annihilators of nonzero elements. If $ab$ lies in it and $b$ does not, the annihilator of $bm$ contains the chosen annihilator and also $a$; maximality forces $a$ into the original annihilator, proving primality.
\end{proof}
\end{theorem}

\begin{theorem}[Associated implies support]\label{thm:v17-02}
If $\mathfrak p\in\operatorname{Ass}M$, then $\mathfrak p\in\operatorname{Supp}M$.
\begin{proof}
An embedding $A/\mathfrak p\hookrightarrow M$ localizes to a nonzero embedding at $\mathfrak p$, since $(A/\mathfrak p)_{\mathfrak p}$ is nonzero.
\end{proof}
\end{theorem}

\begin{theorem}[Minimal support primes are associated]\label{thm:v17-03}
For a finitely generated module over a Noetherian ring, every prime minimal in $\operatorname{Supp}M$ lies in $\operatorname{Ass}M$.
\begin{proof}
Localize at the minimal prime. The localized support has only the maximal prime, so an associated prime of the nonzero localized module must be that maximal prime; contract back.
\end{proof}
\end{theorem}
