\chapter{Prime and Maximal Ideals}
\label{ch:v02}

Prime and maximal ideals encode the irreducible quotient structures of a commutative ring. Prime ideals detect domains, maximal ideals detect fields, and maximal ideals are automatically prime.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item test whether an ideal is prime or maximal using quotient-domain and quotient-field criteria;
\item compute prime and maximal ideals in standard rings such as integers, polynomial rings, and products;
\item prove that maximal ideals are prime and identify hypotheses needed for converses;
\item analyze inverse images of prime ideals under ring homomorphisms;
\item distinguish irreducible elements from prime ideals and prime elements in examples;
\item construct chains of prime ideals that reveal basic dimension-theoretic behavior;
\end{itemize}

\section{Prime ideals}
A proper ideal $\mathfrak p$ is prime when $ab\in\mathfrak p$ implies $a\in\mathfrak p$ or $b\in\mathfrak p$.

\section{Maximal ideals}
A proper ideal $\mathfrak m$ is maximal when no proper ideal lies strictly between $\mathfrak m$ and the whole ring.

\begin{example}[Prime but not maximal]\label{ex:v02-expand-01}
The zero ideal in $\mathbb Z$ is prime because $\mathbb Z$ is a domain, but it is not maximal since $(0)\subsetneq(2)\subsetneq\mathbb Z$.
\end{example}

\section{Quotient criteria}
Prime ideals correspond to domain quotients; maximal ideals correspond to field quotients.

\section{Existence of maximal ideals}
Every proper ideal lies in a maximal ideal, using Zorn's lemma.

\section{Contraction}
Inverse images of prime ideals under ring maps are prime when the inverse image remains proper.

\begin{example}[Maximal evaluation ideal]\label{ex:v02-expand-02}
In $k[x,y]$, evaluation at $(a,b)$ has kernel $(x-a,y-b)$ and image $k$. The quotient is a field, so the kernel is maximal and therefore prime.
\end{example}

\section{Core structural results}

\begin{theorem}[Prime quotient criterion]\label{thm:v02-01}
A proper ideal $\mathfrak p$ is prime iff $A/\mathfrak p$ is an integral domain.
\begin{proof}
The zero-divisor condition in the quotient is exactly the statement that $ab\in\mathfrak p$ forces one factor into $\mathfrak p$.
\end{proof}
\end{theorem}

\begin{theorem}[Maximal quotient criterion]\label{thm:v02-02}
A proper ideal $\mathfrak m$ is maximal iff $A/\mathfrak m$ is a field.
\begin{proof}
Ideals in $A/\mathfrak m$ correspond to ideals of $A$ containing $\mathfrak m$. A quotient is a field exactly when its only ideals are zero and the whole ring.
\end{proof}
\end{theorem}

\begin{theorem}[Maximal implies prime]\label{thm:v02-03}
Every maximal ideal in a commutative ring with identity is prime.
\begin{proof}
The quotient by a maximal ideal is a field, hence a domain; apply the prime quotient criterion.
\end{proof}
\end{theorem}
