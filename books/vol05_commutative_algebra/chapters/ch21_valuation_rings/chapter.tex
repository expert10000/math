\chapter{Valuation Rings}
\label{ch:v21}

Valuation rings impose a total divisibility order on a field. They are integrally closed local domains and provide a powerful way to test integrality and dominate local rings.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify the valuation-ring criterion that for each fraction either it or its inverse lies in the ring;
\item construct valuation rings from valuations on fraction fields;
\item identify prime ideals and units in elementary valuation rings;
\item compare divisibility of principal ideals through the valuation ordering;
\item use valuation rings to test integral closure in fraction fields where the chapter establishes the criterion;
\item distinguish discrete valuation rings from general valuation rings by value-group behavior;
\end{itemize}

\section{Valuations}
A valuation assigns ordered-group values compatible with products and satisfying a triangle inequality for sums.

\section{Valuation rings}
A subring $V$ of a field $K$ is a valuation ring when for every nonzero $x\in K$, either $x$ or $x^{-1}$ lies in $V$.

\begin{example}[The $p$-adic valuation]\label{ex:v21-expand-01}
For a prime $p$, write a nonzero rational number as $p^nu/v$ with $p\nmid uv$. Then $v_p$ equals $n$. The valuation ring consists of rationals with nonnegative $p$-adic valuation.
\end{example}

A subring $V$ of a field $K$ is a valuation ring exactly when for every nonzero $x\in K$, either $x\in V$ or $x^{-1}\in V$. This total comparability forces ideals of $V$ to be linearly ordered.
\end{example}

\section{Prime ideals}
Prime ideals of a valuation ring are linearly ordered.

\section{Integral closure criterion}
Integral closure of a domain can be recovered as an intersection of valuation rings containing it.

\section{Core structural results}

\begin{theorem}[Valuation rings are local]\label{thm:v21-01}
If $V$ is a valuation ring of a field, its nonunits form an ideal and hence the unique maximal ideal.
\begin{proof}
For nonunits $a,b$, compare $a/b$ or $b/a$ when both are nonzero; one divides the other, so their sum is a multiple of one nonunit. Multiplication by arbitrary ring elements preserves nonunits.
\end{proof}
\end{theorem}

\begin{theorem}[Principal ideals are totally ordered]\label{thm:v21-02}
For nonzero $a,b\in V$, either $(a)\subseteq(b)$ or $(b)\subseteq(a)$.
\begin{proof}
Apply the valuation-ring condition to $a/b$: if $a/b\in V$, then $a\in(b)$; otherwise $b/a\in V$, so $b\in(a)$.
\end{proof}
\end{theorem}

\begin{theorem}[Valuation rings are integrally closed]\label{thm:v21-03}
A valuation ring $V$ is integrally closed in its fraction field.
\begin{proof}
If $x$ is integral over $V$ but $x\notin V$, then $x^{-1}\in V$ and is a nonunit. Divide a monic equation for $x$ by the highest power of $x$ to express $1$ as an element of the maximal ideal generated by powers of $x^{-1}$, contradiction.
\end{proof}
\end{theorem}
