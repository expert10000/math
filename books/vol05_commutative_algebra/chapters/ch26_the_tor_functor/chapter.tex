\chapter{The Tor Functor}
\label{ch:v26}

The Tor functor is the homology left behind when tensor product fails to be exact. It is computed from projective resolutions and detects torsion, non-flatness, and derived intersections.

The reader should be able to state the definitions precisely, prove the core structural results, move between global and localized formulations, and audit finiteness and exactness hypotheses.

\section{Definition}
Choose a projective resolution $P_\bullet\to M$ and set $\operatorname{Tor}_n^A(M,N)=H_n(P_\bullet\otimes_A N)$.

\section{Independence of resolution}
Comparison and homotopy theorems make Tor well-defined up to canonical isomorphism.

\section{Degree zero}
$\operatorname{Tor}_0^A(M,N)\cong M\otimes_A N$.

\section{Long exact sequence}
Short exact sequences in either variable yield long exact Tor sequences.

\section{Projective vanishing}
Higher Tor with a projective module vanishes.

\section{Flatness criterion}
A module is flat iff $\operatorname{Tor}_1^A(-,F)=0$ for all modules.

\section{Symmetry}
Over a commutative ring, Tor is symmetric in its two module arguments.

\section{Cyclic calculation}
For integers or quotient modules, short resolutions compute Tor explicitly.

\section{Dimension shifting}
Syzygies reduce higher Tor computations to lower degrees.

\section{Core structural results}

\begin{theorem}[Well-definedness of Tor]\label{thm:v26-01}
The modules $H_n(P_\bullet\otimes N)$ are independent of the chosen projective resolution of $M$ up to canonical isomorphism.
\begin{proof}
Comparison maps between projective resolutions are unique up to chain homotopy. Tensor preserves chain homotopies, and chain-homotopic maps induce the same homology maps, yielding canonical isomorphisms.
\end{proof}
\end{theorem}

\begin{theorem}[Projective vanishing]\label{thm:v26-02}
If $P$ is projective, then $\operatorname{Tor}_n^A(P,N)=0$ for $n>0$.
\begin{proof}
Use the resolution concentrated in degree zero at $P$. The tensor complex has no positive-degree terms, so its positive homology vanishes.
\end{proof}
\end{theorem}

\begin{theorem}[Flatness via Tor]\label{thm:v26-03}
An $A$-module $F$ is flat iff $\operatorname{Tor}_1^A(A/I,F)=0$ for every finitely generated ideal $I$ under the standard finite-ideal criterion, and in particular iff $\operatorname{Tor}_1^A(M,F)=0$ for all $M$.
\begin{proof}
Apply Tor to $0\to I\to A\to A/I\to0$. Vanishing of the connecting kernel is exactly injectivity of $I\otimes F\to F$; the general criterion follows from the flatness tests and long exact sequences.
\end{proof}
\end{theorem}
