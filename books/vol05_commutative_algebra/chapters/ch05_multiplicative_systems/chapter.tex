\chapter{Multiplicative Systems}
\label{ch:v05}

Multiplicative systems specify which elements are to become invertible. They are the input data for localization and control which ideals, primes, and module elements survive.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a subset is multiplicatively closed and determine when it contains zero;
\item construct saturation sets and identify elements annihilated after localization;
\item compare localization at powers of one element with localization at a general multiplicative system;
\item identify universal properties satisfied by maps inverting a prescribed multiplicative set;
\item compute equality conditions between fractions using the localization equivalence relation;
\item choose multiplicative systems adapted to primes, elements, or geometric open sets;
\end{itemize}

\section{Multiplicative subsets}
A multiplicative set $S$ contains $1$ and is closed under products; usually $0\notin S$ for nontrivial localization.

\section{Saturation}
The saturation of $S$ consists of elements that become invertible whenever all of $S$ does.

The set $S=\{1,2,4,8,\ldots\}$ is multiplicatively closed. Inverting it produces $\mathbb Z[1/2]$, where $2$ is a unit but $3$ need not be.
\end{example}

\section{Prime avoidance viewpoint}

\section{Geometric preview}

\section{Core structural results}

\begin{theorem}[Maximal disjoint ideal is prime]\label{thm:v05-01}
If $I$ is maximal among ideals disjoint from a multiplicative set $S$, then $I$ is prime.
\begin{proof}
If $ab\in I$ with $a,b\notin I$, then both $I+(a)$ and $I+(b)$ meet $S$. Multiplying corresponding elements of $S$ yields an element of $I+(ab)\subseteq I$, contradicting disjointness.
\end{proof}
\end{theorem}

\begin{theorem}[Complement of prime is multiplicative]\label{thm:v05-02}
If $\mathfrak p$ is prime, then $A\setminus\mathfrak p$ is multiplicatively closed.
\begin{proof}
If $a,b\notin\mathfrak p$ but $ab\in\mathfrak p$, primality would force one factor into $\mathfrak p$.
\end{proof}
\end{theorem}

\begin{theorem}[Powers form a multiplicative set]\label{thm:v05-03}
For any $f\in A$, $S=\{1,f,f^2,\dots\}$ is multiplicative.
\begin{proof}
The product $f^m f^n=f^{m+n}$ remains in the set and $1=f^0$ is included.
\end{proof}
\end{theorem}
