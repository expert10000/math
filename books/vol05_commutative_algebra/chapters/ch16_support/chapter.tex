\chapter{Support}
\label{ch:v16}

The support of a module records the primes where the module remains nonzero after localization. It turns module-theoretic information into a subset of the prime spectrum and behaves well under exact sequences.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute the support of a module from localizations at prime ideals;
\item identify support as the set of primes containing the annihilator for finitely generated modules under the chapter's hypotheses;
\item compute supports of quotients, direct sums, and localized modules;
\item relate support to vanishing of a module after localization;
\item prove basic closedness properties of support for finitely generated modules;
\item use support to detect where module-theoretic phenomena are nontrivial;
\end{itemize}

\section{Definition}
$\operatorname{Supp}M=\{\mathfrak p:M_{\mathfrak p}\ne0\}$.

\section{Annihilator}
The annihilator $\operatorname{Ann}(M)$ consists of scalars killing every element of $M$.

% BEGIN VOL05-EXPANSION V16-example-01
\begin{example}[Support of an integer module]\label{ex:v16-expand-01}
For $M=\mathbb Z/12\mathbb Z$, the localization $M_{(p)}$ is nonzero exactly for $p=2$ or $3$. Thus the support consists of the prime ideals $(2)$ and $(3)$ together with primes containing the annihilator $(12)$ in the spectrum description.
\end{example}
% END VOL05-EXPANSION V16-example-01
\section{Finite-module support}
For finitely generated $M$, $\operatorname{Supp}M=V(\operatorname{Ann}M)$.

\section{Cyclic modules}
$\operatorname{Supp}(A/I)=V(I)$.

% BEGIN VOL05-EXPANSION V16-example-02
\begin{example}[Cyclic module support]\label{ex:v16-expand-02}
For $A=k[x,y]$ and $M=A/(xy)$, one has $\operatorname{Supp}M=V(xy)=V(x)\cup V(y)$. The module detects the union of the two coordinate axes.
\end{example}
% END VOL05-EXPANSION V16-example-02
\section{Exact sequences}
Support of the middle term in a short exact sequence is the union of supports of the ends.

\section{Direct sums}
Support of a direct sum is the union of the supports.

\begin{example}[Support in an exact sequence]\label{ex:v16-expand-03}
From $0\to A/(x)\to A/(x^2)\to A/(x)\to0$, the support of the middle module equals the union of the supports of the ends, namely $V(x)$.
\end{example}

\section{Tensor products}
Support of a tensor product is contained in the intersection of supports and equals it under common finite-generation hypotheses over local rings.

\section{Localization}
Support localizes by restricting to primes contained in the chosen localization domain.

\section{Vanishing detection}
A module is zero iff all localizations at maximal ideals are zero.

\section{Geometric role}
Support marks the locus where a sheaf-like algebraic object is present.

\section{Core structural results}

\begin{theorem}[Support of a finite module]\label{thm:v16-01}
If $M$ is finitely generated, then $\operatorname{Supp}M=V(\operatorname{Ann}M)$.
\begin{proof}
If an annihilator element becomes a unit at $\mathfrak p$, the localized module vanishes. Conversely, if $M_{\mathfrak p}=0$, finitely many generators are each killed by denominators outside $\mathfrak p$; their product lies in $\operatorname{Ann}M$ but outside $\mathfrak p$.
\end{proof}
\end{theorem}

\begin{theorem}[Support in a short exact sequence]\label{thm:v16-02}
For $0\to M'\to M\to M''\to0$, $\operatorname{Supp}M=\operatorname{Supp}M'\cup\operatorname{Supp}M''$.
\begin{proof}
Localize the sequence. Exactness of localization shows the middle localization is zero exactly when both end localizations are zero.
\end{proof}
\end{theorem}

\begin{theorem}[Maximal-local detection]\label{thm:v16-03}
An $A$-module $M$ is zero iff $M_{\mathfrak m}=0$ for every maximal ideal $\mathfrak m$.
\begin{proof}
If $m\ne0$, its annihilator is proper and lies in some maximal ideal $\mathfrak m$. Then $m/1$ cannot vanish in $M_{\mathfrak m}$, otherwise an element outside $\mathfrak m$ would annihilate $m$.
\end{proof}
\end{theorem}
