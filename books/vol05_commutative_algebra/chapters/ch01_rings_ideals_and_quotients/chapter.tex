\chapter{Rings, Ideals and Quotients}
\label{ch:v01}

Commutative algebra begins with rings, ideals, and quotients. Ideals are exactly the additive subgroups that can serve as kernels of ring homomorphisms, and quotient rings convert congruence conditions into algebraic objects.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify that a subset is an ideal and compute ideals generated by explicit elements;
\item construct quotient rings and identify their elements and operations concretely;
\item compute kernels and images of ring homomorphisms and apply the first isomorphism theorem;
\item translate ideal containment in a quotient through the correspondence theorem;
\item compute sums and intersections of ideals in elementary rings;
\item distinguish ring, subring, ideal, and quotient constructions by examples and counterexamples;
\end{itemize}

\section{Commutative rings}
A commutative ring has associative addition and multiplication, a multiplicative identity, and commutative multiplication.

\section{Ideals}
An ideal is an additive subgroup stable under multiplication by arbitrary ring elements.

% BEGIN VOL05-EXPANSION V01-example-01
\begin{example}[Quotient arithmetic]\label{ex:v01-expand-01}
In $R=\mathbb Z/12\mathbb Z$, the ideal $(4)$ is $\{0,4,8\}$. The reduction map $R\to\mathbb Z/4\mathbb Z$ is surjective with kernel $(4)$, so $R/(4)\cong\mathbb Z/4\mathbb Z$.
\end{example}
% END VOL05-EXPANSION V01-example-01
\section{Generated ideals}
The ideal generated by elements $a_1,\dots,a_n$ consists of finite combinations $r_1a_1+\cdots+r_na_n$.

\section{Sums and intersections}
Sums give the smallest ideal containing two ideals, while intersections give their greatest common subideal.

\section{Quotient rings}
For an ideal $I\subset A$, the quotient $A/I$ has cosets as elements and multiplication induced from $A$.

% BEGIN VOL05-EXPANSION V01-example-02
\begin{example}[Polynomial quotient]\label{ex:v01-expand-02}
In $k[x,y]/(x,y^2)$ every class has a unique representative $a+by$. The class of $y$ is nonzero but satisfies $y^2=0$, so the quotient exhibits a nilpotent direction explicitly.
\end{example}
% END VOL05-EXPANSION V01-example-02
\section{Ring homomorphisms}
Kernels are ideals and images are subrings; quotienting by the kernel isolates the injective part of a map.

\section{Correspondence of ideals}
Ideals of $A/I$ correspond to ideals of $A$ containing $I$.

% BEGIN VOL05-EXPANSION V01-example-03
\begin{example}[Evaluation kernel]\label{ex:v01-expand-03}
Evaluation at $1$ gives $\varphi:k[x]\to k$. Polynomial division shows $\ker\varphi=(x-1)$, and constants show surjectivity. Hence $k[x]/(x-1)\cong k$.
\end{example}
% END VOL05-EXPANSION V01-example-03
\section{Examples}
Polynomial rings, integer residue rings, coordinate-style quotients, and product rings provide the main laboratory.

\section{Core structural results}

\begin{theorem}[First isomorphism theorem]\label{thm:v01-01}
For a ring homomorphism $\varphi:A\to B$, there is a natural isomorphism $A/\ker\varphi\cong\operatorname{im}\varphi$.
\begin{proof}
Send the class of $a$ to $\varphi(a)$. Equality of images is exactly equality modulo the kernel, so the map is well-defined and injective; surjectivity onto the image is immediate.
\end{proof}
\end{theorem}

\begin{theorem}[Correspondence theorem]\label{thm:v01-02}
Ideals of $A/I$ are in inclusion-preserving bijection with ideals $J\subset A$ satisfying $I\subseteq J$.
\begin{proof}
Take inverse image under the quotient map in one direction and quotient $J/I$ in the other. The two constructions are inverse by direct inspection.
\end{proof}
\end{theorem}

\begin{theorem}[Third isomorphism theorem]\label{thm:v01-03}
If $I\subseteq J\subseteq A$ are ideals, then $(A/I)/(J/I)\cong A/J$.
\begin{proof}
Apply the first isomorphism theorem to the homomorphism $A/I\to A/J$ sending $a+I$ to $a+J$; its kernel is $J/I$.
\end{proof}
\end{theorem}
