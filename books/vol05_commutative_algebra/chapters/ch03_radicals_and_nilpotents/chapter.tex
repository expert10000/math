\chapter{Radicals and Nilpotents}
\label{ch:v03}

Radicals and nilpotents measure how equations can vanish after taking powers. The radical of an ideal forgets multiplicity-like information and retains the underlying prime behavior.

\section{Nilpotent elements}
An element $a$ is nilpotent if $a^n=0$ for some positive integer $n$.

\section{Nilradical}
The nilradical $\sqrt{(0)}$ is the ideal of all nilpotent elements.

\section{Radical of an ideal}
$\sqrt I=\{a:a^n\in I\text{ for some }n\ge1\}$.

\section{Radical ideals}
An ideal is radical when $a^n\in I$ implies $a\in I$.

\section{Quotient interpretation}
$\sqrt I/I$ is the nilradical of $A/I$.

\section{Prime containment}
The radical of an ideal is the intersection of all prime ideals containing it.

\section{Reduced rings}
A ring is reduced when its nilradical is zero.

\section{Examples}
Powers of principal ideals, square-zero extensions, and polynomial quotients show how radicals discard nilpotent thickening.

\section{Core structural results}

\begin{theorem}[Radical is an ideal]\label{thm:v03-01}
For every ideal $I$, the set $\sqrt I$ is an ideal containing $I$.
\begin{proof}
Closure under multiplication is immediate. If $a^m,b^n\in I$, then every term in $(a+b)^{m+n}$ contains either at least $m$ copies of $a$ or at least $n$ copies of $b$, hence lies in $I$.
\end{proof}
\end{theorem}

\begin{theorem}[Quotient nilradical identity]\label{thm:v03-02}
The nilradical of $A/I$ is $\sqrt I/I$.
\begin{proof}
The class of $a$ is nilpotent modulo $I$ exactly when some power $a^n$ lies in $I$.
\end{proof}
\end{theorem}

\begin{theorem}[Radical as intersection of primes]\label{thm:v03-03}
$\sqrt I$ equals the intersection of all prime ideals containing $I$.
\begin{proof}
One inclusion follows because primes are radical. For the reverse, if $a\notin\sqrt I$, localize the quotient $A/I$ at powers of $a$ and choose a maximal ideal there; its contraction is a prime containing $I$ but avoiding $a$.
\end{proof}
\end{theorem}
