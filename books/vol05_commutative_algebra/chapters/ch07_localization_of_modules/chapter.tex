\chapter{Localization of Modules}
\label{ch:v07}

Localization of modules extends fraction methods from rings to linear algebra over rings. It is exact, commutes with finite algebraic constructions, and detects local behavior of modules.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct localized modules and simplify fractions of module elements;
\item determine when a localized module element vanishes;
\item prove that localization is an exact functor on modules;
\item compute localized kernels, images, quotients, and finite presentations;
\item test module identities locally at prime ideals and recover global consequences where valid;
\item distinguish module localization from tensoring with an unrelated algebra by using the canonical isomorphism with localized rings;
\end{itemize}

\section{Exactness}
Localization preserves exact sequences.

\begin{example}[Module vanishing]\label{ex:v07-expand-02}
For $M=R/(f)$, localization at powers of $f$ gives $M_f=0$: the relation $f\cdot1=0$ and invertibility of $f$ force $1=0$.
\end{example}

\section{Submodules and quotients}
Localization commutes with quotients and sends submodules to localized submodules.

\section{Vanishing criterion}
An element $m/1$ vanishes locally iff some denominator in $S$ annihilates $m$.

Localizing $0\to\mathbb Z\xrightarrow{\cdot2}\mathbb Z\to\mathbb Z/2\to0$ at powers of $2$ turns the first map into an isomorphism and the localized cokernel into zero.
\end{example}

\section{Core structural results}

\begin{theorem}[Tensor description of localization]\label{thm:v07-01}
There is a natural isomorphism $S^{-1}A\otimes_A M\cong S^{-1}M$ sending $(a/s)\otimes m$ to $am/s$.
\begin{proof}
The formula is balanced and induces a homomorphism. Its inverse sends $m/s$ to $(1/s)\otimes m$; the two composites are identities.
\end{proof}
\end{theorem}

\begin{theorem}[Exactness of localization]\label{thm:v07-02}
If $0\to M'\to M\to M''\to0$ is exact, then $0\to S^{-1}M'\to S^{-1}M\to S^{-1}M''\to0$ is exact.
\begin{proof}
Surjectivity is immediate on fractions. For injectivity or kernel equality, if a localized element maps to zero, some denominator annihilates the relevant numerator, allowing the numerator to be moved into the preceding module before localization.
\end{proof}
\end{theorem}

\begin{theorem}[Localization commutes with quotient]\label{thm:v07-03}
For $N\subset M$, $S^{-1}(M/N)\cong S^{-1}M/S^{-1}N$.
\begin{proof}
Apply exactness to $0\to N\to M\to M/N\to0$.
\end{proof}
\end{theorem}
