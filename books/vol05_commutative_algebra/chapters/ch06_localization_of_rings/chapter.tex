\chapter{Localization of Rings}
\label{ch:v06}

Localization of rings adjoins inverses to a chosen multiplicative system. It is governed by a universal property and induces a precise correspondence between primes of the localization and primes of the original ring disjoint from the denominators.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct localized rings and simplify explicit fractions;
\item apply the universal property of localization to extend ring homomorphisms uniquely;
\item compute prime ideals of a localization from primes disjoint from the multiplicative set;
\item determine when an element becomes zero or a unit after localization;
\item compare localization at a prime, maximal ideal, and single element;
\item prove that localization commutes with quotients in the standard compatible form;
\end{itemize}

\section{Canonical map}
The map $A\to S^{-1}A$ sends $a$ to $a/1$ and makes every element of $S$ invertible.

\section{Universal property}
Any ring map from $A$ sending $S$ to units factors uniquely through $S^{-1}A$.

\section{Zero criterion}
A fraction $a/s$ is zero iff some $t\in S$ annihilates $a$.

% BEGIN VOL05-EXPANSION V06-example-02
\begin{example}[Principal localization]\label{ex:v06-expand-02}
In $k[x,y]_x$, $x$ is a unit, so $(x,y)$ becomes the whole ring. The ideal $(y)$ remains proper and records the part visible where $x$ is invertible.
\end{example}
% END VOL05-EXPANSION V06-example-02
\section{Localized ideals}
Extension sends $I$ to $S^{-1}I$, while contraction returns ideals saturated with respect to $S$.

\section{Prime correspondence}
Prime ideals of $S^{-1}A$ correspond to prime ideals of $A$ disjoint from $S$.

% BEGIN VOL05-EXPANSION V06-example-03
\begin{example}[Prime localization]\label{ex:v06-expand-03}
The ring $\mathbb Z_{(5)}$ consists of $a/b$ with $5\nmid b$. Its unique maximal ideal consists of fractions whose numerator is divisible by $5$.
\end{example}
% END VOL05-EXPANSION V06-example-03
\section{Localization at an element}
$A_f$ is obtained by inverting powers of one element $f$.

\section{Localization at a prime}
$A_{\mathfrak p}$ inverts everything outside $\mathfrak p$ and is a local ring.

\section{Core structural results}

\begin{theorem}[Universal property of localization]\label{thm:v06-01}
If $\varphi:A\to B$ sends every $s\in S$ to a unit, there is a unique map $\widetilde\varphi:S^{-1}A\to B$ with $\widetilde\varphi(a/s)=\varphi(a)\varphi(s)^{-1}$.
\begin{proof}
The formula respects the localization equivalence relation and is forced by the requirement that denominators become inverses, giving existence and uniqueness.
\end{proof}
\end{theorem}

\begin{theorem}[Prime correspondence]\label{thm:v06-02}
Prime ideals of $S^{-1}A$ are in bijection with prime ideals $\mathfrak p\subset A$ satisfying $\mathfrak p\cap S=\varnothing$.
\begin{proof}
Contraction of a prime avoids units, hence avoids $S$. For a prime disjoint from $S$, its extension consists of fractions with numerator in the prime and is prime; extension and contraction are inverse.
\end{proof}
\end{theorem}

\begin{theorem}[Localization commutes with quotient]\label{thm:v06-03}
For an ideal $I$, $S^{-1}(A/I)\cong S^{-1}A/S^{-1}I$ after replacing $S$ by its image in $A/I$.
\begin{proof}
Both rings satisfy the same universal property: maps out correspond to maps from $A$ killing $I$ and inverting $S$.
\end{proof}
\end{theorem}
