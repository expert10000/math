\chapter{Open and Closed Sets}
\label{ch:ii04}

Open and closed sets encode local and limiting structure without coordinates. Interiors, closures, boundaries, and limit points become the language in which continuity and topology are most naturally expressed.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item decide whether a set is open, closed, both, or neither from the definitions;
\item compute interiors, closures, boundaries, and limit-point sets for explicit subsets of metric spaces;
\item prove that arbitrary unions of open sets and finite intersections of open sets are open;
\item use complements to convert statements about open sets into statements about closed sets;
\item construct neighborhood or sequence arguments showing that a point lies in a closure or boundary;
\item produce counterexamples showing why infinite intersections of open sets or infinite unions of closed sets need not preserve type;
\end{itemize}

\section{Open sets}
A set is open if every one of its points contains a ball lying entirely in the set.

\section{Closed sets}
A set is closed if its complement is open; in metric spaces this is equivalent to containing limits of convergent sequences from the set.

\section{Interior}
The interior is the largest open subset contained in a set.

\section{Closure}
The closure is the smallest closed set containing the set.

\section{Limit points}
A point is a limit point if every punctured neighborhood meets the set.

\section{Boundary}
Boundary points have every neighborhood meeting both the set and its complement.

\section{Set operations}
Arbitrary unions and finite intersections preserve openness; dual statements hold for closed sets.

\section{Relative topology and density}
Relative openness is obtained by intersecting ambient open sets with a subspace, while density means the closure fills the ambient space.

\section{Core structural results}

\begin{theorem}[Open-set operations]\label{thm:ii04-01}
Arbitrary unions and finite intersections of open sets are open.
\begin{proof}
For unions inherit a ball from one containing member; for finite intersections take the minimum of finitely many available radii.
\end{proof}
\end{theorem}

\begin{theorem}[Sequential closedness]\label{thm:ii04-02}
A metric subset is closed iff it contains every limit of every convergent sequence drawn from it.
\begin{proof}
Closedness prevents a convergent sequence from entering an open complement; conversely a point in the closure but outside the set yields points within $1/n$ converging to it.
\end{proof}
\end{theorem}

\begin{theorem}[Closure by balls]\label{thm:ii04-03}
$x\in\overline A$ iff every ball centered at $x$ meets $A$.
\begin{proof}
A ball missing $A$ places $x$ in the open complement of the closure; the converse is immediate from the same characterization.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Interval anatomy]\label{ex:ii04-worked-01}
Find the interior, closure, and boundary of $(0,1]$. They are $(0,1)$, $[0,1]$, and $\{0,1\}$.
\end{example}

\begin{example}[Open balls are open]\label{ex:ii04-worked-02}
Prove every open metric ball is open. For $x\in B(a,r)$ choose radius $r-d(a,x)>0$ and use the triangle inequality.
\end{example}

\begin{example}[Closed balls are closed]\label{ex:ii04-worked-03}
Prove every closed metric ball is closed. If $x_n$ lies in the ball and converges to $x$, continuity of $y\mapsto d(a,y)$ gives $d(a,x)\le r$.
\end{example}

\begin{example}[Sphere is closed]\label{ex:ii04-worked-04}
Show $\{x:\|x\|=1\}$ is closed in a normed space. The norm is continuous by the reverse triangle inequality, so the sphere is the inverse image of the closed singleton $\{1\}$.
\end{example}

\section{Solved dossiers}

\begin{problem}[Interval anatomy]\label{prob:ii04-dossier-01}
Find the interior, closure, and boundary of $(0,1]$.
\end{problem}
\begin{solution}
They are $(0,1)$, $[0,1]$, and $\{0,1\}$.
\end{solution}

\begin{problem}[Open balls are open]\label{prob:ii04-dossier-02}
Prove every open metric ball is open.
\end{problem}
\begin{solution}
For $x\in B(a,r)$ choose radius $r-d(a,x)>0$ and use the triangle inequality.
\end{solution}

\begin{problem}[Closed balls are closed]\label{prob:ii04-dossier-03}
Prove every closed metric ball is closed.
\end{problem}
\begin{solution}
If $x_n$ lies in the ball and converges to $x$, continuity of $y\mapsto d(a,y)$ gives $d(a,x)\le r$.
\end{solution}

\begin{problem}[Sphere is closed]\label{prob:ii04-dossier-04}
Show $\{x:\|x\|=1\}$ is closed in a normed space.
\end{problem}
\begin{solution}
The norm is continuous by the reverse triangle inequality, so the sphere is the inverse image of the closed singleton $\{1\}$.
\end{solution}

\begin{problem}[Boundary criterion]\label{prob:ii04-dossier-05}
Show $x\in\partial A$ iff every ball about $x$ meets both $A$ and its complement.
\end{problem}
\begin{solution}
Membership in the closure gives intersection with $A$; failure to be interior gives intersection with the complement.
\end{solution}

\begin{problem}[Closure of rationals]\label{prob:ii04-dossier-06}
Prove $\overline{\mathbb Q}=\mathbb R$.
\end{problem}
\begin{solution}
Rational density says every real ball contains a rational point.
\end{solution}

\begin{problem}[Interior of rationals]\label{prob:ii04-dossier-07}
Show $\mathbb Q^\circ=\varnothing$.
\end{problem}
\begin{solution}
Every interval around a rational contains irrational points.
\end{solution}

\begin{problem}[A clopen subspace piece]\label{prob:ii04-dossier-08}
In $Y=(-\infty,0)\cup(0,\infty)$, show $(0,\infty)$ is clopen relative to $Y$.
\end{problem}
\begin{solution}
It and its complement are intersections of $Y$ with ambient open half-lines.
\end{solution}

\begin{problem}[Closure of a finite union]\label{prob:ii04-dossier-09}
Prove $\overline{A\cup B}=\overline A\cup\overline B$.
\end{problem}
\begin{solution}
The right side is closed and contains the union; monotonicity of closure gives the reverse inclusion.
\end{solution}

\begin{problem}[Interior of an intersection]\label{prob:ii04-dossier-10}
Prove $(A\cap B)^\circ=A^\circ\cap B^\circ$.
\end{problem}
\begin{solution}
The right side is open and contained in the intersection; every open subset of the intersection lies in both interiors.
\end{solution}

\begin{problem}[Limit-point sequence]\label{prob:ii04-dossier-11}
If $x$ is a limit point of $A$ in a metric space, construct a sequence of points of $A\setminus\{x\}$ converging to $x$.
\end{problem}
\begin{solution}
Choose $x_n\in A\cap(B(x,1/n)\setminus\{x\})$; then $d(x_n,x)<1/n$.
\end{solution}

\begin{problem}[Topology synthesis]\label{prob:ii04-dossier-12}
Why can topology be specified by open sets without mentioning a metric?
\end{problem}
\begin{solution}
Convergence, closedness, continuity, compactness, connectedness, interior, closure, and boundary can all be defined from the family of open sets.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii04-01}
Is $(0,1)$ open in $\mathbb R$?
\end{exercise}
\begin{hint}
Choose a point-dependent radius.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii04-02}
Is $[0,1]$ closed?
\end{exercise}
\begin{hint}
Inspect the complement.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii04-03}
Find $\partial(0,1)$.
\end{exercise}
\begin{hint}
Use closure minus interior.
\end{hint}
\begin{solution}
It is $\{0,1\}$.
\end{solution}

\begin{exercise}\label{exr:ii04-04}
Find $\overline{\mathbb Z}$ in $\mathbb R$.
\end{exercise}
\begin{hint}
Is $\mathbb Z$ closed?
\end{hint}
\begin{solution}
It equals $\mathbb Z$.
\end{solution}

\begin{exercise}\label{exr:ii04-05}
Show $\varnothing$ and $X$ are clopen.
\end{exercise}
\begin{hint}
Use complements.
\end{hint}
\begin{solution}
Each is open and the complement of the other.
\end{solution}

\begin{exercise}\label{exr:ii04-06}
Are arbitrary intersections of open sets open?
\end{exercise}
\begin{hint}
Use shrinking intervals.
\end{hint}
\begin{solution}
No; $\cap_n(-1/n,1/n)=\{0\}$.
\end{solution}

\begin{exercise}\label{exr:ii04-07}
Find $[0,1]^\circ$.
\end{exercise}
\begin{hint}
Endpoints fail the ball condition.
\end{hint}
\begin{solution}
It is $(0,1)$.
\end{solution}

\begin{exercise}\label{exr:ii04-08}
Show $A^\circ\subset A\subset\overline A$.
\end{exercise}
\begin{hint}
Use the defining extremal properties.
\end{hint}
\begin{solution}
Both inclusions are immediate from the definitions.
\end{solution}
