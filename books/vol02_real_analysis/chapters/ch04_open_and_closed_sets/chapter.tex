\chapter{Open and Closed Sets}
\label{ch:ii04}

Open and closed sets encode local and limiting structure without coordinates. Interiors, closures, boundaries, and limit points become the language in which continuity and topology are most naturally expressed.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item decide whether a set is open, closed, both, or neither from the definitions;
\item compute interiors, closures, boundaries, and limit-point sets for explicit subsets of metric spaces;
\item prove that arbitrary unions of open sets and finite intersections of open sets are open;
\item use complements to convert statements about open sets into statements about closed sets;
\item construct neighborhood or sequence arguments showing that a point lies in a closure or boundary;
\item produce counterexamples showing why infinite intersections of open sets or infinite unions of closed sets need not preserve type;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Open sets}
A set is open if every one of its points contains a ball lying entirely in the set.

\section{Closed sets}
A set is closed if its complement is open; in metric spaces this is equivalent to containing limits of convergent sequences from the set.

\section{Interior}
The interior is the largest open subset contained in a set.


% BEGIN VOL02-EXPANSION II04-example-01
\begin{example}[Interior, closure, and boundary in one calculation]\label{ex:ii04-expand-01}
For
\[
A=[0,1)\cup\{2\}\subset\mathbb R,
\]
the interior is \((0,1)\): neither endpoint has a ball contained in \(A\), and
the isolated point \(2\) has none either. The closure is
\[
\overline A=[0,1]\cup\{2\},
\]
because \(1\) is approached from the left and \(2\) is already included.
Therefore
\[
\partial A=\overline A\setminus A^\circ=\{0,1,2\}.
\]
Checking each exceptional point against neighborhoods verifies the set
identities.
\end{example}
% END VOL02-EXPANSION II04-example-01
\section{Closure}
The closure is the smallest closed set containing the set.


% BEGIN VOL02-EXPANSION II04-example-02
\begin{example}[Closure by a sequence]\label{ex:ii04-expand-02}
Let \(A=(0,1)\). To show \(0\in\overline A\), take
\[
x_n=\frac1n\in A,\qquad x_n\to0.
\]
Likewise \(1-1/n\to1\), so both endpoints lie in the closure. No point outside
\([0,1]\) can lie in the closure because its positive distance from \([0,1]\)
produces a disjoint open ball. Hence \(\overline A=[0,1]\).

The sequence proves inclusion; the positive-distance ball proves exclusion.
\end{example}
% END VOL02-EXPANSION II04-example-02
\section{Limit points}
A point is a limit point if every punctured neighborhood meets the set.

\section{Boundary}
Boundary points have every neighborhood meeting both the set and its complement.

\section{Set operations}
Arbitrary unions and finite intersections preserve openness; dual statements hold for closed sets.

\section{Relative topology and density}
Relative openness is obtained by intersecting ambient open sets with a subspace, while density means the closure fills the ambient space.


% BEGIN VOL02-EXPANSION II04-example-03
\begin{example}[Relative openness can differ from ambient openness]\label{ex:ii04-expand-03}
Let \(Y=[0,2]\subset\mathbb R\) and \(A=[0,1)\subset Y\). The set \(A\) is not
open in \(\mathbb R\), but it is open in the subspace \(Y\) because
\[
A=(-1,1)\cap Y.
\]
At the endpoint \(0\), a relative neighborhood is allowed to be cut off by the
ambient subspace. Relative topology therefore records neighborhoods internal
to the space under study, not to the larger containing space.
\end{example}
% END VOL02-EXPANSION II04-example-03
\section{Core structural results}

\begin{theorem}[Open-set operations]\label{thm:ii04-01}
Arbitrary unions and finite intersections of open sets are open.
\begin{proof}
For unions inherit a ball from one containing member; for finite intersections take the minimum of finitely many available radii.
\end{proof}
\end{theorem}

\begin{theorem}[Sequential closedness]\label{thm:ii04-02}
A metric subset is closed iff it contains every limit of every convergent sequence drawn from it.
\begin{proof}
Closedness prevents a convergent sequence from entering an open complement; conversely a point in the closure but outside the set yields points within $1/n$ converging to it.
\end{proof}
\end{theorem}

\begin{theorem}[Closure by balls]\label{thm:ii04-03}
$x\in\overline A$ iff every ball centered at $x$ meets $A$.
\begin{proof}
A ball missing $A$ places $x$ in the open complement of the closure; the converse is immediate from the same characterization.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Interval anatomy]\label{ex:ii04-worked-01}
Find the interior, closure, and boundary of $(0,1]$. They are $(0,1)$, $[0,1]$, and $\{0,1\}$.
\end{example}

\begin{example}[Open balls are open]\label{ex:ii04-worked-02}
Prove every open metric ball is open. For $x\in B(a,r)$ choose radius $r-d(a,x)>0$ and use the triangle inequality.
\end{example}

\begin{example}[Closed balls are closed]\label{ex:ii04-worked-03}
Prove every closed metric ball is closed. If $x_n$ lies in the ball and converges to $x$, continuity of $y\mapsto d(a,y)$ gives $d(a,x)\le r$.
\end{example}

\begin{example}[Sphere is closed]\label{ex:ii04-worked-04}
Show $\{x:\|x\|=1\}$ is closed in a normed space. The norm is continuous by the reverse triangle inequality, so the sphere is the inverse image of the closed singleton $\{1\}$.
\end{example}

\section{Solved dossiers}
Each dossier is a substantial solved problem. Corpus mappings guide topic choice where explicit retained source blocks exist; otherwise the dossier is a fresh canonical completion.

\begin{problem}[Interval anatomy]\label{prob:ii04-dossier-01}
Find the interior, closure, and boundary of $(0,1]$.
\end{problem}
\begin{solution}
They are $(0,1)$, $[0,1]$, and $\{0,1\}$.
\end{solution}

\begin{problem}[Open balls are open]\label{prob:ii04-dossier-02}
Prove every open metric ball is open.
\end{problem}
\begin{solution}
For $x\in B(a,r)$ choose radius $r-d(a,x)>0$ and use the triangle inequality.
\end{solution}

\begin{problem}[Closed balls are closed]\label{prob:ii04-dossier-03}
Prove every closed metric ball is closed.
\end{problem}
\begin{solution}
If $x_n$ lies in the ball and converges to $x$, continuity of $y\mapsto d(a,y)$ gives $d(a,x)\le r$.
\end{solution}

\begin{problem}[Sphere is closed]\label{prob:ii04-dossier-04}
Show $\{x:\|x\|=1\}$ is closed in a normed space.
\end{problem}
\begin{solution}
The norm is continuous by the reverse triangle inequality, so the sphere is the inverse image of the closed singleton $\{1\}$.
\end{solution}

\begin{problem}[Boundary criterion]\label{prob:ii04-dossier-05}
Show $x\in\partial A$ iff every ball about $x$ meets both $A$ and its complement.
\end{problem}
\begin{solution}
Membership in the closure gives intersection with $A$; failure to be interior gives intersection with the complement.
\end{solution}

\begin{problem}[Closure of rationals]\label{prob:ii04-dossier-06}
Prove $\overline{\mathbb Q}=\mathbb R$.
\end{problem}
\begin{solution}
Rational density says every real ball contains a rational point.
\end{solution}

\begin{problem}[Interior of rationals]\label{prob:ii04-dossier-07}
Show $\mathbb Q^\circ=\varnothing$.
\end{problem}
\begin{solution}
Every interval around a rational contains irrational points.
\end{solution}

\begin{problem}[A clopen subspace piece]\label{prob:ii04-dossier-08}
In $Y=(-\infty,0)\cup(0,\infty)$, show $(0,\infty)$ is clopen relative to $Y$.
\end{problem}
\begin{solution}
It and its complement are intersections of $Y$ with ambient open half-lines.
\end{solution}

\begin{problem}[Closure of a finite union]\label{prob:ii04-dossier-09}
Prove $\overline{A\cup B}=\overline A\cup\overline B$.
\end{problem}
\begin{solution}
The right side is closed and contains the union; monotonicity of closure gives the reverse inclusion.
\end{solution}

\begin{problem}[Interior of an intersection]\label{prob:ii04-dossier-10}
Prove $(A\cap B)^\circ=A^\circ\cap B^\circ$.
\end{problem}
\begin{solution}
The right side is open and contained in the intersection; every open subset of the intersection lies in both interiors.
\end{solution}

\begin{problem}[Limit-point sequence]\label{prob:ii04-dossier-11}
If $x$ is a limit point of $A$ in a metric space, construct a sequence of points of $A\setminus\{x\}$ converging to $x$.
\end{problem}
\begin{solution}
Choose $x_n\in A\cap(B(x,1/n)\setminus\{x\})$; then $d(x_n,x)<1/n$.
\end{solution}

\begin{problem}[Topology synthesis]\label{prob:ii04-dossier-12}
Why can topology be specified by open sets without mentioning a metric?
\end{problem}
\begin{solution}
Convergence, closedness, continuity, compactness, connectedness, interior, closure, and boundary can all be defined from the family of open sets.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii04-01}
Is $(0,1)$ open in $\mathbb R$?
\end{exercise}
\begin{hint}
Choose a point-dependent radius. Use the metric-ball definition directly: for each point of the set, find a radius whose ball stays inside.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii04-02}
Is $[0,1]$ closed?
\end{exercise}
\begin{hint}
Inspect the complement. To prove closedness, work with the complement or show that every convergent sequence from the set has its limit in the set.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii04-03}
Find $\partial(0,1)$.
\end{exercise}
\begin{hint}
Use closure minus interior. A point is in the closure exactly when every ball around it meets the set; test this neighborhood condition.
\end{hint}
\begin{solution}
It is $\{0,1\}$.
\end{solution}

\begin{exercise}\label{exr:ii04-04}
Find $\overline{\mathbb Z}$ in $\mathbb R$.
\end{exercise}
\begin{hint}
Is $\mathbb Z$ closed? For a boundary point, show that every sufficiently small ball meets both the set and its complement.
\end{hint}
\begin{solution}
It equals $\mathbb Z$.
\end{solution}

\begin{exercise}\label{exr:ii04-05}
Show $\varnothing$ and $X$ are clopen.
\end{exercise}
\begin{hint}
Use complements. Arbitrary unions preserve openness because one containing open set already supplies the needed neighborhood.
\end{hint}
\begin{solution}
Each is open and the complement of the other.
\end{solution}

\begin{exercise}\label{exr:ii04-06}
Are arbitrary intersections of open sets open?
\end{exercise}
\begin{hint}
Use shrinking intervals. Finite intersections preserve openness by taking the minimum of finitely many available radii.
\end{hint}
\begin{solution}
No; $\cap_n(-1/n,1/n)=\{0\}$.
\end{solution}

\begin{exercise}\label{exr:ii04-07}
Find $[0,1]^\circ$.
\end{exercise}
\begin{hint}
Endpoints fail the ball condition. For an infinite-intersection counterexample, let the open sets shrink toward a boundary point.
\end{hint}
\begin{solution}
It is $(0,1)$.
\end{solution}

\begin{exercise}\label{exr:ii04-08}
Show $A^\circ\subset A\subset\overline A$.
\end{exercise}
\begin{hint}
Use the defining extremal properties. Compute interior, closure, and boundary separately; do not infer one from a sketch alone.
\end{hint}
\begin{solution}
Both inclusions are immediate from the definitions.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Interval anatomy]\label{exr:ii04-expand-01}
Find interior, closure, and boundary of \((0,2]\).
\end{exercise}
\begin{hint}
Test the two endpoints separately.
\end{hint}
\begin{solution}
The interior is \((0,2)\), closure \([0,2]\), boundary \(\{0,2\}\).
\end{solution}

\begin{exercise}[Finite set]\label{exr:ii04-expand-02}
Find the closure and interior of \(\{1,2,3\}\subset\mathbb R\).
\end{exercise}
\begin{hint}
Finite sets are closed and contain no intervals.
\end{hint}
\begin{solution}
The closure is the set itself; the interior is empty.
\end{solution}

\begin{exercise}[Rational closure]\label{exr:ii04-expand-03}
Find the closure of \(\mathbb Q\) in \(\mathbb R\).
\end{exercise}
\begin{hint}
Use density of rationals.
\end{hint}
\begin{solution}
\(\overline{\mathbb Q}=\mathbb R\).
\end{solution}

\begin{exercise}[Boundary of a ball]\label{exr:ii04-expand-04}
In \(\mathbb R^n\), identify the boundary of \(B(0,1)\).
\end{exercise}
\begin{hint}
Points with norm below one are interior; points with norm above one are exterior.
\end{hint}
\begin{solution}
The boundary is the unit sphere \(\{x:\|x\|=1\}\).
\end{solution}

\begin{exercise}[Relative topology]\label{exr:ii04-expand-05}
In \(Y=[0,1]\), is \([0,1/2)\) open relative to \(Y\)?
\end{exercise}
\begin{hint}
Write it as an intersection with an ambient open interval.
\end{hint}
\begin{solution}
Yes: \([0,1/2)=(-1,1/2)\cap Y\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Union of open sets]\label{exr:ii04-expand-06}
Prove arbitrary unions of open sets are open.
\end{exercise}
\begin{hint}
A point in the union lies in at least one member.
\end{hint}
\begin{solution}
Use an open ball contained in that member; it is then contained in the union.
\end{solution}

\begin{exercise}[Finite intersection]\label{exr:ii04-expand-07}
Prove a finite intersection of open sets is open.
\end{exercise}
\begin{hint}
Take the minimum of finitely many local radii.
\end{hint}
\begin{solution}
The minimum positive radius gives a ball lying in every member.
\end{solution}

\begin{exercise}[Sequential closedness]\label{exr:ii04-expand-08}
Prove a closed metric subset contains limits of its convergent sequences.
\end{exercise}
\begin{hint}
If the limit were outside, use an open complement neighborhood.
\end{hint}
\begin{solution}
Eventually the sequence would have to lie in the complement, contradicting that all terms lie in the set.
\end{solution}

\begin{exercise}[Closure minimality]\label{exr:ii04-expand-09}
Prove \(\overline A\) is the smallest closed set containing \(A\).
\end{exercise}
\begin{hint}
Show every closed superset contains all closure points.
\end{hint}
\begin{solution}
If closed \(F\supset A\) omitted \(x\in\overline A\), its open complement would give a neighborhood of \(x\) missing \(A\), contradiction.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Infinite intersection]\label{exr:ii04-expand-10}
Give open sets whose infinite intersection is not open.
\end{exercise}
\begin{hint}
Use shrinking intervals around \(0\).
\end{hint}
\begin{solution}
\(\bigcap_n(-1/n,1/n)=\{0\}\), not open in \(\mathbb R\).
\end{solution}

\begin{exercise}[Infinite union closed]\label{exr:ii04-expand-11}
Give closed sets whose infinite union is not closed.
\end{exercise}
\begin{hint}
Use \([1/n,1]\).
\end{hint}
\begin{solution}
\(\bigcup_n[1/n,1]=(0,1]\), which is not closed.
\end{solution}

\begin{exercise}[Neither open nor closed]\label{exr:ii04-expand-12}
Give a subset of \(\mathbb R\) that is neither open nor closed.
\end{exercise}
\begin{hint}
Use a half-open interval.
\end{hint}
\begin{solution}
\([0,1)\) is neither.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Feasible-set closure]\label{exr:ii04-expand-13}
Why is a closed feasible set useful when taking limits of approximating solutions?
\end{exercise}
\begin{hint}
Use sequential closedness.
\end{hint}
\begin{solution}
If feasible approximants converge, their limit remains feasible.
\end{solution}

\begin{exercise}[Dense sampling]\label{exr:ii04-expand-14}
What does density of a sample set \(D\subset X\) mean operationally?
\end{exercise}
\begin{hint}
Translate \(\overline D=X\) into balls.
\end{hint}
\begin{solution}
Every neighborhood of every target point contains a sample point, so arbitrary local approximation is possible.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Boundary identity]\label{exr:ii04-expand-15}
Prove \(\partial A=\overline A\cap\overline{X\setminus A}\).
\end{exercise}
\begin{hint}
A boundary neighborhood must meet both \(A\) and its complement.
\end{hint}
\begin{solution}
The two closure conditions are exactly the two required neighborhood-intersection conditions.
\end{solution}

\begin{exercise}[Closure of union]\label{exr:ii04-expand-16}
Prove \(\overline{A\cup B}=\overline A\cup\overline B\).
\end{exercise}
\begin{hint}
Use closedness/minimality or neighborhoods.
\end{hint}
\begin{solution}
The right side is closed and contains \(A\cup B\), giving one inclusion; monotonicity of closure gives the reverse inclusions.
\end{solution}
