\chapter{Infinite Series of Functions}
\label{ch:ii13}

Infinite series of functions are controlled most effectively by uniform comparison tests. Absolute and uniform convergence turn termwise algebra, integration, and sometimes differentiation into rigorous operations.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item test convergence of numerical and function series using comparison, ratio, root, or uniform criteria;
\item apply the Weierstrass M-test to prove uniform and absolute convergence of a function series;
\item justify termwise integration or differentiation of a series when the required uniform hypotheses hold;
\item determine the interval or region of convergence of a power series;
\item estimate tails of convergent series sufficiently to meet a prescribed error tolerance;
\item construct examples separating pointwise, uniform, absolute, and conditional convergence;
\end{itemize}

\section{Partial sums}
A function series is studied through its partial-sum sequence.

\section{Uniform convergence}
Uniform convergence of partial sums is the key global notion.

\section{Weierstrass M-test}
Uniform domination by a summable numerical series gives uniform absolute convergence.

\section{Power series}
Inside the radius of convergence, power series converge uniformly on compact subintervals.

\section{Algebra of series}
Uniformly convergent series can be added and scaled termwise.

\section{Integration}
Uniform convergence justifies termwise integration.

\section{Differentiation}
Uniform convergence of derivative series plus one-point convergence justifies termwise differentiation.

\section{Remainder bounds}
Majorant tails provide explicit uniform truncation error estimates.

\section{Core structural results}

\begin{theorem}[Weierstrass M-test]\label{thm:ii13-01}
If $|f_n(x)|\le M_n$ for all $x$ and $\sum M_n<\infty$, then $\sum f_n$ converges uniformly and absolutely.
\begin{proof}
The numerical tail bound $\sum_{n>N}M_n$ controls every function tail uniformly, so the partial sums are uniformly Cauchy.
\end{proof}
\end{theorem}

\begin{theorem}[Uniform termwise integration]\label{thm:ii13-02}
A uniformly convergent series of integrable functions may be integrated termwise.
\begin{proof}
Apply the uniform-limit integral theorem to partial sums.
\end{proof}
\end{theorem}

\begin{theorem}[Power-series local uniformity]\label{thm:ii13-03}
A power series converges uniformly on every closed subinterval strictly inside its radius of convergence.
\begin{proof}
Choose a larger interior radius and dominate coefficients times $r^n$ by a convergent geometric comparison from the root or ratio test.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[M-test with geometric majorant]\label{ex:ii13-worked-01}
Show $\sum x^n/n^2$ converges uniformly on $[-1,1]$. $|x^n/n^2|\le1/n^2$ and $\sum1/n^2$ converges.
\end{example}

\begin{example}[Uniform tail bound]\label{ex:ii13-worked-02}
If $|f_n|\le2^{-n}$, bound the tail after $N$. The sup tail is at most $\sum_{n>N}2^{-n}=2^{-N}$.
\end{example}

\begin{example}[Harmonic obstruction]\label{ex:ii13-worked-03}
Why does the M-test fail for $x^n/n$ on $[0,1]$? The natural majorant $1/n$ diverges; indeed at $x=1$ the series is harmonic.
\end{example}

\begin{example}[Power-series compact convergence]\label{ex:ii13-worked-04}
Why does $\sum x^n$ converge uniformly on $[-r,r]$ for $r<1$? $|x^n|\le r^n$ and the geometric series converges.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[M-test with geometric majorant]\label{prob:ii13-dossier-01}
Show $\sum x^n/n^2$ converges uniformly on $[-1,1]$.
\end{problem}
\begin{solution}
$|x^n/n^2|\le1/n^2$ and $\sum1/n^2$ converges.
\end{solution}

\begin{problem}[Uniform tail bound]\label{prob:ii13-dossier-02}
If $|f_n|\le2^{-n}$, bound the tail after $N$.
\end{problem}
\begin{solution}
The sup tail is at most $\sum_{n>N}2^{-n}=2^{-N}$.
\end{solution}

\begin{problem}[Harmonic obstruction]\label{prob:ii13-dossier-03}
Why does the M-test fail for $x^n/n$ on $[0,1]$?
\end{problem}
\begin{solution}
The natural majorant $1/n$ diverges; indeed at $x=1$ the series is harmonic.
\end{solution}

\begin{problem}[Power-series compact convergence]\label{prob:ii13-dossier-04}
Why does $\sum x^n$ converge uniformly on $[-r,r]$ for $r<1$?
\end{problem}
\begin{solution}
$|x^n|\le r^n$ and the geometric series converges.
\end{solution}

\begin{problem}[Termwise integral]\label{prob:ii13-dossier-05}
Integrate $\sum_{n\ge0}x^n$ on $[0,r]$, $r<1$.
\end{problem}
\begin{solution}
Uniform convergence gives $\sum r^{n+1}/(n+1)=-\log(1-r)$.
\end{solution}

\begin{problem}[A derivative series]\label{prob:ii13-dossier-06}
For $\sum x^n/n!$, justify termwise differentiation on bounded intervals.
\end{problem}
\begin{solution}
Derivative terms are $x^{n-1}/(n-1)!$, uniformly dominated on $[-R,R]$ by $R^{n-1}/(n-1)!$.
\end{solution}

\begin{problem}[Absolute versus uniform]\label{prob:ii13-dossier-07}
Does pointwise absolute convergence imply uniform convergence?
\end{problem}
\begin{solution}
No. Uniformity requires tail control independent of $x$; absolute convergence can deteriorate as $x$ varies.
\end{solution}

\begin{problem}[Uniform Cauchy criterion]\label{prob:ii13-dossier-08}
State it for a series.
\end{problem}
\begin{solution}
For every epsilon there is $N$ such that every finite tail $\sum_{n=p}^qf_n(x)$ has magnitude below epsilon for all $x$ whenever $q\ge p\ge N$.
\end{solution}

\begin{problem}[Remainder estimate]\label{prob:ii13-dossier-09}
If $|f_n|\le M_n$, how can one certify truncation accuracy?
\end{problem}
\begin{solution}
Bound the sup error by the numerical tail $\sum_{n>N}M_n$.
\end{solution}

\begin{problem}[Uniform convergence preserves continuity]\label{prob:ii13-dossier-10}
Why is a uniformly convergent series of continuous functions continuous?
\end{problem}
\begin{solution}
Its partial sums are continuous and converge uniformly, so the uniform-limit theorem applies.
\end{solution}

\begin{problem}[Local uniformity on noncompact domains]\label{prob:ii13-dossier-11}
What does it mean for a series to converge locally uniformly?
\end{problem}
\begin{solution}
It converges uniformly on every compact subset of the domain.
\end{solution}

\begin{problem}[Series synthesis]\label{prob:ii13-dossier-12}
Why are majorants central?
\end{problem}
\begin{solution}
They replace a difficult family of function tails by one numerical tail independent of the point, converting pointwise estimates into uniform convergence.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii13-01}
Apply M-test to $\sum x^n/2^n$ on $[-1,1]$.
\end{exercise}
\begin{hint}
Use $(1/2)^n$. For the M-test, dominate the absolute value of each term by a numerical sequence whose sum is known to converge.
\end{hint}
\begin{solution}
It converges uniformly.
\end{solution}

\begin{exercise}\label{exr:ii13-02}
Does $\sum x^n$ converge uniformly on $[0,1]$?
\end{exercise}
\begin{hint}
Check $x=1$. To find a power-series radius, apply the ratio or root test to the coefficient sequence and then inspect endpoints separately.
\end{hint}
\begin{solution}
No; it does not even converge there.
\end{solution}

\begin{exercise}\label{exr:ii13-03}
Uniform convergence of $\sum f_n$ means what for partial sums?
\end{exercise}
\begin{hint}
Use the definition. For termwise integration, uniform convergence on the interval is the key safe route in this chapter.
\end{hint}
\begin{solution}
Partial sums converge uniformly.
\end{solution}

\begin{exercise}\label{exr:ii13-04}
If $\sum M_n$ converges, what about its tails?
\end{exercise}
\begin{hint}
Cauchy criterion. For termwise differentiation, control the derivative series uniformly and anchor the original series at one point.
\end{hint}
\begin{solution}
They tend to zero.
\end{solution}

\begin{exercise}\label{exr:ii13-05}
Can a uniformly convergent series of continuous functions be discontinuous?
\end{exercise}
\begin{hint}
Estimate a series tail by comparison with a geometric or integral bound rather than summing many terms explicitly.
\end{hint}
\begin{solution}
Its sum is continuous.
\end{solution}

\begin{exercise}\label{exr:ii13-06}
What is a power-series radius?
\end{exercise}
\begin{hint}
Boundary between convergence and divergence. Absolute convergence implies ordinary convergence, but the converse can fail; test signs and absolute values separately.
\end{hint}
\begin{solution}
A number $R$ such that convergence holds for $|x-a|<R$ and divergence for $|x-a|>R$.
\end{solution}

\begin{exercise}\label{exr:ii13-07}
Can the M-test prove absolute convergence?
\end{exercise}
\begin{hint}
To disprove uniform convergence, locate points where the tail does not become uniformly small.
\end{hint}
\begin{solution}
It gives uniform absolute convergence.
\end{solution}

\begin{exercise}\label{exr:ii13-08}
How do you estimate a remainder under M-test?
\end{exercise}
\begin{hint}
Use majorant tail. Use the Cauchy criterion for series when no closed-form sum is available.
\end{hint}
\begin{solution}
By $\sum_{n>N}M_n$.
\end{solution}
