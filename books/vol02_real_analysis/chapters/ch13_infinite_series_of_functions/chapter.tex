\chapter{Infinite Series of Functions}
\label{ch:ii13}

Infinite series of functions are controlled most effectively by uniform comparison tests. Absolute and uniform convergence turn termwise algebra, integration, and sometimes differentiation into rigorous operations.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item test convergence of numerical and function series using comparison, ratio, root, or uniform criteria;
\item apply the Weierstrass M-test to prove uniform and absolute convergence of a function series;
\item justify termwise integration or differentiation of a series when the required uniform hypotheses hold;
\item determine the interval or region of convergence of a power series;
\item estimate tails of convergent series sufficiently to meet a prescribed error tolerance;
\item construct examples separating pointwise, uniform, absolute, and conditional convergence;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Partial sums}
A function series is studied through its partial-sum sequence.

\section{Uniform convergence}
Uniform convergence of partial sums is the key global notion.

\section{Weierstrass M-test}
Uniform domination by a summable numerical series gives uniform absolute convergence.


% BEGIN VOL02-EXPANSION II13-example-01
\begin{example}[M-test with a computable truncation guarantee]\label{ex:ii13-expand-01}
On \([-1,1]\), consider
\[
\sum_{n=1}^{\infty}\frac{x^n}{3^n}.
\]
Since \(|x^n/3^n|\le 3^{-n}\), the M-test gives uniform absolute convergence.
Moreover the tail after \(N\) satisfies
\[
\sup_{|x|\le1}\left|\sum_{n>N}\frac{x^n}{3^n}\right|
\le\sum_{n>N}3^{-n}
=\frac{1}{2\cdot3^N}.
\]
Thus \(N\) can be chosen directly from a requested uniform tolerance. The
majorant proves convergence and supplies an error certificate at the same time.
\end{example}
% END VOL02-EXPANSION II13-example-01
\section{Power series}
Inside the radius of convergence, power series converge uniformly on compact subintervals.


% BEGIN VOL02-EXPANSION II13-example-02
\begin{example}[Power-series convergence is locally uniform inside the radius]\label{ex:ii13-expand-02}
For the geometric power series \(\sum_{n\ge0}x^n\), fix \(0<r<1\). On
\([-r,r]\),
\[
|x^n|\le r^n,
\]
and \(\sum r^n\) converges. Hence the series converges uniformly there and
\[
\sum_{n=0}^{\infty}x^n=\frac1{1-x}.
\]
The same argument works on every compact subinterval strictly inside a general
power series' radius of convergence: the compact set stays a positive distance
from the boundary where convergence may deteriorate.
\end{example}
% END VOL02-EXPANSION II13-example-02

\section{Remainder bounds}
Majorant tails provide explicit uniform truncation error estimates.


% BEGIN VOL02-EXPANSION II13-example-03
\begin{example}[Turning a majorant tail into a stopping rule]\label{ex:ii13-expand-03}
Suppose \(|f_n(x)|\le2^{-n}\) for every \(x\in X\). Then the remainder after
the \(N\)-th term obeys
\[
\|R_N\|_\infty
\le\sum_{n=N+1}^{\infty}2^{-n}=2^{-N}.
\]
To guarantee uniform error below \(10^{-6}\), it is enough to take \(N=20\).
This is stronger than knowing only that the series converges: it converts the
proof into a practical truncation rule.
\end{example}

\section{Core structural results}

\begin{theorem}[Weierstrass M-test]\label{thm:ii13-01}
If $|f_n(x)|\le M_n$ for all $x$ and $\sum M_n<\infty$, then $\sum f_n$ converges uniformly and absolutely.
\begin{proof}
The numerical tail bound $\sum_{n>N}M_n$ controls every function tail uniformly, so the partial sums are uniformly Cauchy.
\end{proof}
\end{theorem}

\begin{theorem}[Uniform termwise integration]\label{thm:ii13-02}
A uniformly convergent series of integrable functions may be integrated termwise.
\begin{proof}
Apply the uniform-limit integral theorem to partial sums.
\end{proof}
\end{theorem}

\begin{theorem}[Power-series local uniformity]\label{thm:ii13-03}
A power series converges uniformly on every closed subinterval strictly inside its radius of convergence.
\begin{proof}
Choose a larger interior radius and dominate coefficients times $r^n$ by a convergent geometric comparison from the root or ratio test.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[M-test with geometric majorant]\label{ex:ii13-worked-01}
Show $\sum x^n/n^2$ converges uniformly on $[-1,1]$. $|x^n/n^2|\le1/n^2$ and $\sum1/n^2$ converges.
\end{example}

\begin{example}[Uniform tail bound]\label{ex:ii13-worked-02}
If $|f_n|\le2^{-n}$, bound the tail after $N$. The sup tail is at most $\sum_{n>N}2^{-n}=2^{-N}$.
\end{example}

\begin{example}[Harmonic obstruction]\label{ex:ii13-worked-03}
Why does the M-test fail for $x^n/n$ on $[0,1]$? The natural majorant $1/n$ diverges; indeed at $x=1$ the series is harmonic.
\end{example}

\begin{example}[Power-series compact convergence]\label{ex:ii13-worked-04}
Why does $\sum x^n$ converge uniformly on $[-r,r]$ for $r<1$? $|x^n|\le r^n$ and the geometric series converges.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[M-test with geometric majorant]\label{prob:ii13-dossier-01}
Show $\sum x^n/n^2$ converges uniformly on $[-1,1]$.
\end{problem}
\begin{solution}
$|x^n/n^2|\le1/n^2$ and $\sum1/n^2$ converges.
\end{solution}

\begin{problem}[Uniform tail bound]\label{prob:ii13-dossier-02}
If $|f_n|\le2^{-n}$, bound the tail after $N$.
\end{problem}
\begin{solution}
The sup tail is at most $\sum_{n>N}2^{-n}=2^{-N}$.
\end{solution}

\begin{problem}[Harmonic obstruction]\label{prob:ii13-dossier-03}
Why does the M-test fail for $x^n/n$ on $[0,1]$?
\end{problem}
\begin{solution}
The natural majorant $1/n$ diverges; indeed at $x=1$ the series is harmonic.
\end{solution}

\begin{problem}[Power-series compact convergence]\label{prob:ii13-dossier-04}
Why does $\sum x^n$ converge uniformly on $[-r,r]$ for $r<1$?
\end{problem}
\begin{solution}
$|x^n|\le r^n$ and the geometric series converges.
\end{solution}

\begin{problem}[Termwise integral]\label{prob:ii13-dossier-05}
Integrate $\sum_{n\ge0}x^n$ on $[0,r]$, $r<1$.
\end{problem}
\begin{solution}
Uniform convergence gives $\sum r^{n+1}/(n+1)=-\log(1-r)$.
\end{solution}

\begin{problem}[A derivative series]\label{prob:ii13-dossier-06}
For $\sum x^n/n!$, justify termwise differentiation on bounded intervals.
\end{problem}
\begin{solution}
Derivative terms are $x^{n-1}/(n-1)!$, uniformly dominated on $[-R,R]$ by $R^{n-1}/(n-1)!$.
\end{solution}

\begin{problem}[Absolute versus uniform]\label{prob:ii13-dossier-07}
Does pointwise absolute convergence imply uniform convergence?
\end{problem}
\begin{solution}
No. Uniformity requires tail control independent of $x$; absolute convergence can deteriorate as $x$ varies.
\end{solution}

\begin{problem}[Uniform Cauchy criterion]\label{prob:ii13-dossier-08}
State it for a series.
\end{problem}
\begin{solution}
For every epsilon there is $N$ such that every finite tail $\sum_{n=p}^qf_n(x)$ has magnitude below epsilon for all $x$ whenever $q\ge p\ge N$.
\end{solution}

\begin{problem}[Remainder estimate]\label{prob:ii13-dossier-09}
If $|f_n|\le M_n$, how can one certify truncation accuracy?
\end{problem}
\begin{solution}
Bound the sup error by the numerical tail $\sum_{n>N}M_n$.
\end{solution}

\begin{problem}[Uniform convergence preserves continuity]\label{prob:ii13-dossier-10}
Why is a uniformly convergent series of continuous functions continuous?
\end{problem}
\begin{solution}
Its partial sums are continuous and converge uniformly, so the uniform-limit theorem applies.
\end{solution}

\begin{problem}[Local uniformity on noncompact domains]\label{prob:ii13-dossier-11}
What does it mean for a series to converge locally uniformly?
\end{problem}
\begin{solution}
It converges uniformly on every compact subset of the domain.
\end{solution}

\begin{problem}[Series synthesis]\label{prob:ii13-dossier-12}
Why are majorants central?
\end{problem}
\begin{solution}
They replace a difficult family of function tails by one numerical tail independent of the point, converting pointwise estimates into uniform convergence.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii13-01}
Apply M-test to $\sum x^n/2^n$ on $[-1,1]$.
\end{exercise}
\begin{hint}
Use $(1/2)^n$. For the M-test, dominate the absolute value of each term by a numerical sequence whose sum is known to converge.
\end{hint}
\begin{solution}
It converges uniformly.
\end{solution}

\begin{exercise}\label{exr:ii13-02}
Does $\sum x^n$ converge uniformly on $[0,1]$?
\end{exercise}
\begin{hint}
Check $x=1$. To find a power-series radius, apply the ratio or root test to the coefficient sequence and then inspect endpoints separately.
\end{hint}
\begin{solution}
No; it does not even converge there.
\end{solution}

\begin{exercise}\label{exr:ii13-03}
Uniform convergence of $\sum f_n$ means what for partial sums?
\end{exercise}
\begin{hint}
Use the definition. For termwise integration, uniform convergence on the interval is the key safe route in this chapter.
\end{hint}
\begin{solution}
Partial sums converge uniformly.
\end{solution}

\begin{exercise}\label{exr:ii13-04}
If $\sum M_n$ converges, what about its tails?
\end{exercise}
\begin{hint}
Cauchy criterion. For termwise differentiation, control the derivative series uniformly and anchor the original series at one point.
\end{hint}
\begin{solution}
They tend to zero.
\end{solution}

\begin{exercise}\label{exr:ii13-05}
Can a uniformly convergent series of continuous functions be discontinuous?
\end{exercise}
\begin{hint}
Estimate a series tail by comparison with a geometric or integral bound rather than summing many terms explicitly.
\end{hint}
\begin{solution}
Its sum is continuous.
\end{solution}

\begin{exercise}\label{exr:ii13-06}
What is a power-series radius?
\end{exercise}
\begin{hint}
Boundary between convergence and divergence. Absolute convergence implies ordinary convergence, but the converse can fail; test signs and absolute values separately.
\end{hint}
\begin{solution}
A number $R$ such that convergence holds for $|x-a|<R$ and divergence for $|x-a|>R$.
\end{solution}

\begin{exercise}\label{exr:ii13-07}
Can the M-test prove absolute convergence?
\end{exercise}
\begin{hint}
To disprove uniform convergence, locate points where the tail does not become uniformly small.
\end{hint}
\begin{solution}
It gives uniform absolute convergence.
\end{solution}

\begin{exercise}\label{exr:ii13-08}
How do you estimate a remainder under M-test?
\end{exercise}
\begin{hint}
Use majorant tail. Use the Cauchy criterion for series when no closed-form sum is available.
\end{hint}
\begin{solution}
By $\sum_{n>N}M_n$.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Geometric function series]\label{exr:ii13-expand-01}
Find the uniform sum of \(\sum_{n\ge0}(x/2)^n\) on \([-1,1]\).
\end{exercise}
\begin{hint}
Use the geometric-series formula; \(|x/2|\le1/2\).
\end{hint}
\begin{solution}
The sum is \(1/(1-x/2)\), and convergence is uniform by the M-test.
\end{solution}

\begin{exercise}[Tail size]\label{exr:ii13-expand-02}
If \(|f_n|\le4^{-n}\), bound the tail after \(N\).
\end{exercise}
\begin{hint}
Sum the geometric majorant from \(N+1\) onward.
\end{hint}
\begin{solution}
The tail is at most \(\frac{4^{-N}}{3}\).
\end{solution}

\begin{exercise}[Power-series radius]\label{exr:ii13-expand-03}
Find the radius of convergence of \(\sum n x^n\).
\end{exercise}
\begin{hint}
Use the ratio test on \(n|x|^n\).
\end{hint}
\begin{solution}
The ratio tends to \(|x|\), so the radius is \(1\).
\end{solution}

\begin{exercise}[Termwise integral]\label{exr:ii13-expand-04}
Integrate \(\sum_{n\ge0}x^n\) termwise from \(0\) to \(r<1\).
\end{exercise}
\begin{hint}
Use uniform convergence on \([0,r]\).
\end{hint}
\begin{solution}
\(\sum_{n\ge0}r^{n+1}/(n+1)=-\log(1-r)\).
\end{solution}

\begin{exercise}[Uniform absolute convergence]\label{exr:ii13-expand-05}
Does \(\sum x^n/n^2\) converge uniformly on \([-1,1]\)?
\end{exercise}
\begin{hint}
Compare with \(\sum1/n^2\).
\end{hint}
\begin{solution}
Yes, absolutely and uniformly by the M-test.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[M-test proof]\label{exr:ii13-expand-06}
Prove the Weierstrass M-test from the uniform Cauchy criterion.
\end{exercise}
\begin{hint}
Bound every function tail by the corresponding numerical tail.
\end{hint}
\begin{solution}
For \(m>n\), \(\sup_x|\sum_{k=n+1}^m f_k(x)|\le\sum_{k=n+1}^m M_k\), which becomes arbitrarily small.
\end{solution}

\begin{exercise}[Uniform termwise integration]\label{exr:ii13-expand-07}
Prove a uniformly convergent series of Riemann-integrable functions can be integrated termwise.
\end{exercise}
\begin{hint}
Apply the uniform integral theorem to partial sums.
\end{hint}
\begin{solution}
The partial sums converge uniformly to the sum, and integration is continuous in sup norm.
\end{solution}

\begin{exercise}[Power-series compact uniformity]\label{exr:ii13-expand-08}
Prove \(\sum a_nx^n\) converges uniformly on every \([-r,r]\) with \(r<R\), where \(R\) is the radius.
\end{exercise}
\begin{hint}
Choose \(\rho\) with \(r<\rho<R\) and dominate \(|a_n|r^n\) by a convergent comparison based on \(\rho\).
\end{hint}
\begin{solution}
Convergence at \(\rho\) bounds \(|a_n|\rho^n\); multiplying by \((r/\rho)^n\) gives a geometric majorant.
\end{solution}

\begin{exercise}[Tail criterion]\label{exr:ii13-expand-09}
Show a uniformly absolutely convergent series has uniform tails tending to zero.
\end{exercise}
\begin{hint}
Use the numerical majorant tail.
\end{hint}
\begin{solution}
The sup norm of each tail is bounded by a scalar tail that tends to zero.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[M-test not necessary]\label{exr:ii13-expand-10}
Give a uniformly convergent series for which the obvious absolute M-test is unavailable.
\end{exercise}
\begin{hint}
Think of cancellation or conditional convergence with a common factor.
\end{hint}
\begin{solution}
For example a uniformly convergent scalar conditional series multiplied by a fixed bounded function can converge uniformly without an absolutely summable coefficient majorant.
\end{solution}

\begin{exercise}[Boundary failure]\label{exr:ii13-expand-11}
Why does \(\sum x^n\) fail to converge uniformly on \([0,1)\)?
\end{exercise}
\begin{hint}
The point producing the largest tail can move toward \(1\).
\end{hint}
\begin{solution}
The tail supremum over \([0,1)\) remains \(1\), so uniform convergence fails despite pointwise convergence.
\end{solution}

\begin{exercise}[Termwise differentiation warning]\label{exr:ii13-expand-12}
Why may a uniformly convergent series not be differentiated termwise?
\end{exercise}
\begin{hint}
Uniform convergence of the original series does not control derivative amplitudes.
\end{hint}
\begin{solution}
Derivative series can diverge or fail uniform convergence; extra derivative hypotheses are required.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Certified truncation]\label{exr:ii13-expand-13}
If a function series has majorant tail \(2^{-N}\), how should \(N\) be chosen for \(10^{-8}\) accuracy?
\end{exercise}
\begin{hint}
Solve \(2^{-N}<10^{-8}\).
\end{hint}
\begin{solution}
Any \(N\ge27\) works.
\end{solution}

\begin{exercise}[Spectral expansion viewpoint]\label{exr:ii13-expand-14}
Why are uniform tail bounds useful in numerical expansions?
\end{exercise}
\begin{hint}
They control worst-case error over the whole domain.
\end{hint}
\begin{solution}
They turn a formal infinite representation into a finite approximation with a rigorous global error guarantee.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Cauchy product]\label{exr:ii13-expand-15}
Under absolute uniform convergence of two function series, explain why their Cauchy product is well behaved.
\end{exercise}
\begin{hint}
Use absolute numerical majorants and the scalar Cauchy-product theorem.
\end{hint}
\begin{solution}
The product coefficients are dominated by the convolution of two summable majorants, so the product series converges uniformly and equals the pointwise product.
\end{solution}

\begin{exercise}[Differentiate a power series]\label{exr:ii13-expand-16}
Why may a power series be differentiated termwise strictly inside its radius of convergence?
\end{exercise}
\begin{hint}
On a smaller compact interval, dominate the derivative series using a slightly larger interior radius.
\end{hint}
\begin{solution}
The derivative series converges uniformly on compacta inside the radius; the derivative-limit theorem then justifies termwise differentiation.
\end{solution}
