\chapter{Connectedness and Path Connectedness}
\label{ch:ii07}

Connectedness excludes decompositions into separated open pieces, while path connectedness asks for explicit continuous curves between points. These notions capture global cohesion that compactness and completeness do not.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item decide whether an explicitly described set is connected or path connected;
\item construct explicit paths joining two points when path connectedness holds;
\item use separation arguments or continuous images to prove connectedness;
\item prove that intervals in \(\mathbb R\) are connected and apply the result to intermediate-value arguments;
\item construct examples showing that connectedness and path connectedness are not equivalent in complete generality;
\item identify connected components or path components in elementary examples;
\end{itemize}

\section{Connected spaces}
A space is connected if it admits no separation into disjoint nonempty open pieces.

\section{Clopen criterion}
Connected spaces have no nontrivial subset that is both open and closed.

\section{Connected subsets of the line}
A subset of $\mathbb R$ is connected exactly when it is an interval in the order-convex sense.

\section{Continuous images}
Continuous maps preserve connectedness.

\section{Intermediate values}
The intermediate-value theorem follows because continuous images of intervals are intervals.

\section{Path connectedness}
A path is a continuous map from $[0,1]$ joining two prescribed endpoints.

\section{Path connected implies connected}
A path cannot cross a separation without separating the interval parameter.

\section{Components}
Connected and path components are maximal subsets with the corresponding cohesion property.

\section{Core structural results}

\begin{theorem}[Continuous image theorem]\label{thm:ii07-01}
The continuous image of a connected space is connected.
\begin{proof}
A separation of the image pulls back to a separation of the domain.
\end{proof}
\end{theorem}

\begin{theorem}[Intervals are connected]\label{thm:ii07-02}
Every real interval is connected.
\begin{proof}
A hypothetical separation and a supremum argument at the interface produce a point that cannot consistently belong to either relatively open side.
\end{proof}
\end{theorem}

\begin{theorem}[Path connected implies connected]\label{thm:ii07-03}
Every path-connected space is connected.
\begin{proof}
A separation of the space would pull back along a path joining opposite sides to a separation of $[0,1]$.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Clopen criterion]\label{ex:ii07-worked-01}
Prove connectedness is equivalent to having only trivial clopen subsets. A nontrivial clopen set and its complement form a separation, and every separation makes each side clopen.
\end{example}

\begin{example}[Punctured line]\label{ex:ii07-worked-02}
Show $\mathbb R\setminus\{0\}$ is disconnected. The negative and positive half-lines are disjoint nonempty relatively open sets covering it.
\end{example}

\begin{example}[Connected subsets of the line]\label{ex:ii07-worked-03}
If $A\subset\mathbb R$ is connected and $x<z<y$ with $x,y\in A$, show $z\in A$. If $z\notin A$, split $A$ at $z$ into points below and above, creating a separation.
\end{example}

\begin{example}[Intermediate-value theorem]\label{ex:ii07-worked-04}
Derive it from connectedness. The continuous image of $[a,b]$ is connected in $\mathbb R$, hence an interval containing all values between endpoint values.
\end{example}

\begin{problem}[Clopen criterion]\label{prob:ii07-dossier-01}
Prove connectedness is equivalent to having only trivial clopen subsets.
\end{problem}
\begin{solution}
A nontrivial clopen set and its complement form a separation, and every separation makes each side clopen.
\end{solution}

\begin{problem}[Punctured line]\label{prob:ii07-dossier-02}
Show $\mathbb R\setminus\{0\}$ is disconnected.
\end{problem}
\begin{solution}
The negative and positive half-lines are disjoint nonempty relatively open sets covering it.
\end{solution}

\begin{problem}[Connected subsets of the line]\label{prob:ii07-dossier-03}
If $A\subset\mathbb R$ is connected and $x<z<y$ with $x,y\in A$, show $z\in A$.
\end{problem}
\begin{solution}
If $z\notin A$, split $A$ at $z$ into points below and above, creating a separation.
\end{solution}

\begin{problem}[Intermediate-value theorem]\label{prob:ii07-dossier-04}
Derive it from connectedness.
\end{problem}
\begin{solution}
The continuous image of $[a,b]$ is connected in $\mathbb R$, hence an interval containing all values between endpoint values.
\end{solution}

\begin{problem}[Convex implies path connected]\label{prob:ii07-dossier-05}
Prove every convex subset of a normed vector space is path connected.
\end{problem}
\begin{solution}
The straight-line path $(1-t)x+ty$ remains in the convex set.
\end{solution}

\begin{problem}[Path components are disjoint]\label{prob:ii07-dossier-06}
Show two path components are equal or disjoint.
\end{problem}
\begin{solution}
If they meet, concatenate paths through a common point to show their union is path connected, contradicting maximality unless they coincide.
\end{solution}

\begin{problem}[Union through a common point]\label{prob:ii07-dossier-07}
Show a union of connected sets with a common point is connected.
\end{problem}
\begin{solution}
Any separation would place the common point on one side; each connected member meeting that side must lie entirely there.
\end{solution}

\begin{problem}[Closure preserves connectedness]\label{prob:ii07-dossier-08}
Prove the closure of a connected set is connected.
\end{problem}
\begin{solution}
A separation of the closure would meet the original dense connected set on both sides and induce a separation of it.
\end{solution}

\begin{problem}[Continuous image of a path]\label{prob:ii07-dossier-09}
Show continuous maps preserve path connectedness.
\end{problem}
\begin{solution}
Compose the original path with the continuous map.
\end{solution}

\begin{problem}[Components are closed]\label{prob:ii07-dossier-10}
Prove every connected component is closed.
\end{problem}
\begin{solution}
The closure of a connected set is connected and contains the component; maximality forces equality.
\end{solution}

\begin{problem}[Products of path-connected spaces]\label{prob:ii07-dossier-11}
Show $X\times Y$ is path connected when both factors are.
\end{problem}
\begin{solution}
Pair paths coordinatewise: $t\mapsto(\alpha(t),\beta(t))$.
\end{solution}

\begin{problem}[Connectedness synthesis]\label{prob:ii07-dossier-12}
Contrast connectedness with compactness.
\end{problem}
\begin{solution}
Connectedness excludes a global split into separated regions; compactness excludes infinite escape by finite extraction. Neither implies the other in general.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii07-01}
Is $[0,1]\cup[2,3]$ connected?
\end{exercise}
\begin{hint}
Find a separation. To prove path connectedness, write an explicit continuous path \(\gamma:[0,1]\to X\) joining arbitrary endpoints.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii07-02}
Is $\mathbb R$ path connected?
\end{exercise}
\begin{hint}
Use straight lines. Path connectedness implies connectedness, so use a path construction if it is easier than a separation argument.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii07-03}
Are continuous images of connected sets connected?
\end{exercise}
\begin{hint}
Use the theorem. To prove disconnectedness, exhibit two disjoint nonempty relatively open sets whose union is the whole set.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii07-04}
Are connected subsets of $\mathbb R$ intervals?
\end{exercise}
\begin{hint}
Use an intermediate-point argument. Continuous images of connected sets are connected; map an interval onto the set when possible.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii07-05}
Show a singleton is connected.
\end{exercise}
\begin{hint}
There is no nontrivial split. An interval cannot be separated because an assumed gap would contradict the least-upper-bound property.
\end{hint}
\begin{solution}
Its only clopen subsets are trivial.
\end{solution}

\begin{exercise}\label{exr:ii07-06}
Is $\mathbb Q$ connected?
\end{exercise}
\begin{hint}
Cut at an irrational number. For intermediate-value arguments, apply connectedness to the continuous image and use that connected subsets of \(\mathbb R\) are intervals.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii07-07}
Does path connected imply connected?
\end{exercise}
\begin{hint}
Use interval connectedness. To identify components, find maximal connected subsets rather than merely visually separate clusters.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii07-08}
Does connected imply path connected?
\end{exercise}
\begin{hint}
Recall a classical plane counterexample. For a non-path-connected example, test whether every attempted path would force passage through a missing limit point.
\end{hint}
\begin{solution}
No; the topologist's sine curve is a standard counterexample.
\end{solution}
