\chapter{Chebyshev and Minimax Approximation}
\label{ch:ii22}

Chebyshev and minimax approximation focus on worst-case uniform error. Equioscillation and Chebyshev polynomials reveal why optimal approximants spread their error rather than concentrating it.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item formulate the minimax approximation problem in the sup norm;
\item compute simple best approximants by balancing extreme errors;
\item use Chebyshev polynomials to control the maximum size of monic polynomials on an interval;
\item compare interpolation error with best-uniform-approximation error;
\item identify equioscillation patterns in explicit low-degree examples;
\item derive error lower bounds that certify optimality of a candidate minimax approximant;
\end{itemize}

\section{Minimax problem}
Choose an approximant minimizing the maximum absolute error.

\section{Chebyshev polynomials}
$T_n(\cos\theta)=\cos(n\theta)$ oscillates between $\pm1$ with controlled leading coefficient.


% BEGIN VOL02-EXPANSION II22-example-01
\begin{example}[Chebyshev recurrence and bounded oscillation]\label{ex:ii22-expand-01}
From
\[
T_n(\cos\theta)=\cos(n\theta)
\]
and the cosine addition identity,
\[
T_{n+1}(x)=2xT_n(x)-T_{n-1}(x).
\]
Starting with \(T_0=1,T_1=x\), this gives
\[
T_2=2x^2-1,\qquad T_3=4x^3-3x.
\]
On \([-1,1]\), \(|T_n(x)|\le1\), and the extrema alternate between \(+1\)
and \(-1\) at \(x_k=\cos(k\pi/n)\). This controlled oscillation is exactly the
pattern used in minimax optimality.
\end{example}
% END VOL02-EXPANSION II22-example-01
\section{Extremal property}
Among monic degree-$n$ polynomials, a scaled Chebyshev polynomial minimizes sup norm on $[-1,1]$.

\section{Equioscillation}
Optimal errors alternate in sign at enough extremal points.


% BEGIN VOL02-EXPANSION II22-example-02
\begin{example}[Balancing error gives the best constant approximant]\label{ex:ii22-expand-02}
For a continuous function \(f\) with minimum \(m\) and maximum \(M\), approximate
it by a constant \(c\). The worst error is
\[
E(c)=\max(M-c,c-m).
\]
This is minimized when the two active errors are equal:
\[
M-c=c-m,
\]
so
\[
c_*=\frac{M+m}{2},\qquad E_*=\frac{M-m}{2}.
\]
The optimal error alternates with equal magnitude at points where \(f=M\) and
\(f=m\), the degree-zero case of equioscillation.
\end{example}
% END VOL02-EXPANSION II22-example-02
\section{Uniqueness}
Alternation forces uniqueness of the best approximant in classical polynomial spaces.

\section{Node placement}
Chebyshev extrema and zeros guide stable interpolation and quadrature.


% BEGIN VOL02-EXPANSION II22-example-03
\begin{example}[Chebyshev nodes reduce the dangerous interpolation factor]\label{ex:ii22-expand-03}
For degree-\(n\) interpolation, the remainder contains the node polynomial
\[
\omega_{n+1}(x)=\prod_{j=0}^n(x-x_j).
\]
Choosing nodes related to zeros of a Chebyshev polynomial makes the associated
monic polynomial have near-minimal sup norm on \([-1,1]\). The nodes cluster
near the endpoints, precisely where equally spaced high-degree interpolation
often develops large oscillations.

Thus Chebyshev placement is derived from a minimax property, not an arbitrary
nonuniform spacing rule.
\end{example}
% END VOL02-EXPANSION II22-example-03

\section{Core structural results}

\begin{theorem}[Chebyshev extremal property]\label{thm:ii22-01}
The monic polynomial $2^{1-n}T_n$ has minimal sup norm among all monic degree-$n$ polynomials on $[-1,1]$.
\begin{proof}
If another monic polynomial had smaller norm, their difference would have degree at most $n-1$ yet would change sign between the $n+1$ alternating extrema of the Chebyshev polynomial, forcing at least $n$ roots, contradiction.
\end{proof}
\end{theorem}

\begin{theorem}[Alternation criterion]\label{thm:ii22-02}
A best degree-at-most-$n$ uniform approximant to a continuous function is characterized by at least $n+2$ alternating extremal error points.
\begin{proof}
If alternation fails, one can perturb within the approximation space to reduce the maximum error; if alternation holds, any competitor difference has too many sign changes for its degree.
\end{proof}
\end{theorem}

\begin{theorem}[Uniqueness of minimax polynomial]\label{thm:ii22-03}
The best uniform polynomial approximant of prescribed degree is unique.
\begin{proof}
Two distinct best approximants averaged together would force a smaller maximum error unless their error extrema align; alternation then forces their difference to have too many zeros.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Compute low Chebyshev polynomials]\label{ex:ii22-worked-01}
Find $T_0,T_1,T_2,T_3$. From $T_n(\cos\theta)=\cos n\theta$: $1,x,2x^2-1,4x^3-3x$.
\end{example}

\begin{example}[Recurrence]\label{ex:ii22-worked-02}
Derive $T_{n+1}=2xT_n-T_{n-1}$. Use $\cos((n+1)\theta)=2\cos\theta\cos n\theta-\cos((n-1)\theta)$.
\end{example}

\begin{example}[Extrema]\label{ex:ii22-worked-03}
Where does $T_n$ attain $\pm1$? At $x_k=\cos(k\pi/n)$, $k=0,\dots,n$.
\end{example}

\begin{example}[Zeros]\label{ex:ii22-worked-04}
Where are the zeros of $T_n$? At $\cos((2k-1)\pi/(2n))$, $k=1,\dots,n$.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[Compute low Chebyshev polynomials]\label{prob:ii22-dossier-01}
Find $T_0,T_1,T_2,T_3$.
\end{problem}
\begin{solution}
From $T_n(\cos\theta)=\cos n\theta$: $1,x,2x^2-1,4x^3-3x$.
\end{solution}

\begin{problem}[Recurrence]\label{prob:ii22-dossier-02}
Derive $T_{n+1}=2xT_n-T_{n-1}$.
\end{problem}
\begin{solution}
Use $\cos((n+1)\theta)=2\cos\theta\cos n\theta-\cos((n-1)\theta)$.
\end{solution}

\begin{problem}[Extrema]\label{prob:ii22-dossier-03}
Where does $T_n$ attain $\pm1$?
\end{problem}
\begin{solution}
At $x_k=\cos(k\pi/n)$, $k=0,\dots,n$.
\end{solution}

\begin{problem}[Zeros]\label{prob:ii22-dossier-04}
Where are the zeros of $T_n$?
\end{problem}
\begin{solution}
At $\cos((2k-1)\pi/(2n))$, $k=1,\dots,n$.
\end{solution}

\begin{problem}[Monic scaling]\label{prob:ii22-dossier-05}
Why multiply by $2^{1-n}$?
\end{problem}
\begin{solution}
For $n\ge1$, the leading coefficient of $T_n$ is $2^{n-1}$, so scaling makes it monic.
\end{solution}

\begin{problem}[Worst-case interpretation]\label{prob:ii22-dossier-06}
What does minimax optimize?
\end{problem}
\begin{solution}
The largest absolute error over the entire interval.
\end{solution}

\begin{problem}[Alternating errors]\label{prob:ii22-dossier-07}
Why should an optimum touch both positive and negative error extremes?
\end{problem}
\begin{solution}
If all extreme errors had the same sign, a small vertical shift would reduce the maximum magnitude.
\end{solution}

\begin{problem}[Linear approximation intuition]\label{prob:ii22-dossier-08}
For degree one, how many alternating extremal points characterize the minimax fit?
\end{problem}
\begin{solution}
At least three.
\end{solution}

\begin{problem}[Affine rescaling]\label{prob:ii22-dossier-09}
How do Chebyshev points transfer to $[a,b]$?
\end{problem}
\begin{solution}
Map $x\in[-1,1]$ by $(a+b)/2+(b-a)x/2$.
\end{solution}

\begin{problem}[Interpolation connection]\label{prob:ii22-dossier-10}
Why do Chebyshev nodes reduce endpoint problems?
\end{problem}
\begin{solution}
They cluster near endpoints, reducing the maximum node-polynomial magnitude.
\end{solution}

\begin{problem}[Uniqueness mechanism]\label{prob:ii22-dossier-11}
How does root counting enter minimax uniqueness?
\end{problem}
\begin{solution}
Alternating error inequalities force the difference of two candidates to change sign too many times for its degree.
\end{solution}

\begin{problem}[Minimax synthesis]\label{prob:ii22-dossier-12}
How is minimax different from least squares?
\end{problem}
\begin{solution}
Minimax controls the maximum error; least squares minimizes an integrated squared error and may tolerate larger localized peaks.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii22-01}
What is $T_2$?
\end{exercise}
\begin{hint}
Use recurrence. Write the error function \(e=f-p\) and locate its largest positive and negative values before adjusting the approximant.
\end{hint}
\begin{solution}
$2x^2-1$.
\end{solution}

\begin{exercise}\label{exr:ii22-02}
What is $T_3$?
\end{exercise}
\begin{hint}
Use recurrence. For a constant best approximant, balance the maximum positive and negative errors.
\end{hint}
\begin{solution}
$4x^3-3x$.
\end{solution}

\begin{exercise}\label{exr:ii22-03}
What is $\|T_n\|_\infty$ on $[-1,1]$?
\end{exercise}
\begin{hint}
Cosine definition. Chebyshev polynomials are extremal because they oscillate with equal magnitude; scale them to the required leading coefficient.
\end{hint}
\begin{solution}
$1$.
\end{solution}

\begin{exercise}\label{exr:ii22-04}
How many extrema are used in alternation for degree $n$?
\end{exercise}
\begin{hint}
One more than dimension. To certify optimality, derive a lower bound on every competitor's maximum error and show the candidate attains it.
\end{hint}
\begin{solution}
At least $n+2$.
\end{solution}

\begin{exercise}\label{exr:ii22-05}
Are Chebyshev zeros equally spaced in $x$?
\end{exercise}
\begin{hint}
They are equally spaced in angle. Interpolation need not be minimax; compare its sup error with an equioscillating competitor.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii22-06}
Why scale $T_n$ to monic?
\end{exercise}
\begin{hint}
Compare monic polynomials. Normalize the interval to \([-1,1]\) before applying standard Chebyshev formulas.
\end{hint}
\begin{solution}
To formulate the extremal property.
\end{solution}

\begin{exercise}\label{exr:ii22-07}
What norm is minimized?
\end{exercise}
\begin{hint}
Sup norm. Count alternating extreme points in the error curve; insufficient alternation suggests the approximation can still be improved.
\end{hint}
\begin{solution}
Uniform norm.
\end{solution}

\begin{exercise}\label{exr:ii22-08}
Does minimax equal least squares?
\end{exercise}
\begin{hint}
Generally no. Use symmetry of the target function to reduce the possible form of a best approximating polynomial.
\end{hint}
\begin{solution}
No.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Recurrence proof]\label{exr:ii22-expand-06}
Derive the Chebyshev recurrence.
\end{exercise}
\begin{hint}
Use \(\cos((n+1)\theta)+\cos((n-1)\theta)=2\cos\theta\cos(n\theta)\).
\end{hint}
\begin{solution}
Substitute \(x=\cos\theta\) to obtain \(T_{n+1}=2xT_n-T_{n-1}\).
\end{solution}

\begin{exercise}[Extremal property idea]\label{exr:ii22-expand-07}
Explain the root-counting contradiction behind the monic Chebyshev extremal theorem.
\end{exercise}
\begin{hint}
Subtract a supposedly smaller-norm monic polynomial from the scaled Chebyshev polynomial.
\end{hint}
\begin{solution}
The leading terms cancel, leaving degree at most \(n-1\), but alternating extrema force at least \(n\) sign-change roots.
\end{solution}

\begin{exercise}[Uniqueness minimax]\label{exr:ii22-expand-08}
Why is the best polynomial approximant of fixed degree unique?
\end{exercise}
\begin{hint}
Use alternation or average two candidates.
\end{hint}
\begin{solution}
Alternation forces the difference of two best candidates to have too many zeros unless it is identically zero.
\end{solution}

\begin{exercise}[Affine scaling]\label{exr:ii22-expand-09}
Explain how Chebyshev minimax results on \([-1,1]\) transfer to \([a,b]\).
\end{exercise}
\begin{hint}
Use an affine bijection between intervals.
\end{hint}
\begin{solution}
Compose with \(t=(2x-a-b)/(b-a)\); polynomial degree is preserved and sup norms correspond.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Interpolation not minimax]\label{exr:ii22-expand-10}
Why need a Chebyshev-node interpolant not equal the best minimax polynomial?
\end{exercise}
\begin{hint}
Node interpolation imposes exact values, while minimax enforces optimal global error.
\end{hint}
\begin{solution}
Good nodes improve interpolation but do not make the optimization problems identical.
\end{solution}

\begin{exercise}[Average error not sup error]\label{exr:ii22-expand-11}
Why can a least-squares best fit differ from a minimax fit?
\end{exercise}
\begin{hint}
The objectives weight errors differently.
\end{hint}
\begin{solution}
Least squares minimizes integrated/summed squared error, whereas minimax minimizes the largest absolute error.
\end{solution}

\begin{exercise}[Too few alternating points]\label{exr:ii22-expand-12}
Why does an error pattern with fewer than \(n+2\) alternating extrema fail to certify degree-\(n\) optimality?
\end{exercise}
\begin{hint}
The alternation theorem requires enough sign constraints to block lower-degree perturbations.
\end{hint}
\begin{solution}
Without them, a perturbation may reduce the maximum error.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Worst-case approximation]\label{exr:ii22-expand-13}
When is minimax approximation preferable to least squares?
\end{exercise}
\begin{hint}
When a guaranteed maximum error matters.
\end{hint}
\begin{solution}
It optimizes worst-case deviation rather than average performance.
\end{solution}

\begin{exercise}[Stable node placement]\label{exr:ii22-expand-14}
Why do endpoint-clustered Chebyshev nodes help high-degree interpolation?
\end{exercise}
\begin{hint}
They reduce the sup size of the node polynomial and moderate endpoint amplification.
\end{hint}
\begin{solution}
This lowers a central worst-case factor in the interpolation error.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Best monic quadratic]\label{exr:ii22-expand-15}
Find the monic quadratic of smallest sup norm on \([-1,1]\).
\end{exercise}
\begin{hint}
Scale \(T_2=2x^2-1\) to be monic.
\end{hint}
\begin{solution}
\(x^2-\frac12\), with sup norm \(1/2\).
\end{solution}

\begin{exercise}[Alternation lower bound]\label{exr:ii22-expand-16}
If a degree-\(n\) candidate error alternates at \(n+2\) points with magnitude \(E\), explain why no competitor can have sup error below \(E\).
\end{exercise}
\begin{hint}
Compare the candidate with a hypothetical better competitor and track signs of their polynomial difference.
\end{hint}
\begin{solution}
The difference must change sign in each adjacent interval, giving at least \(n+1\) roots despite degree at most \(n\), contradiction.
\end{solution}
