\chapter{Trigonometric Series}
\label{ch:ii14}

Trigonometric series represent periodic functions in orthogonal sine and cosine modes. The first layer of Fourier analysis concerns coefficients, orthogonality, partial sums, and the distinction between formal expansions and genuine convergence.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute Fourier coefficients of elementary periodic functions;
\item use symmetry to predict vanishing sine or cosine coefficients before integration;
\item determine pointwise convergence values at continuity points and jump discontinuities under the stated hypotheses;
\item apply Parseval-type identities in concrete coefficient computations;
\item estimate partial sums or tails from coefficient information;
\item distinguish convergence of a Fourier series from convergence of an arbitrary trigonometric series;
\end{itemize}

\section{Periodic functions}
Periodicity identifies functions on the line with functions on a circle.

\section{Trigonometric system}
Sines and cosines form an orthogonal family on a full period.

\section{Fourier coefficients}
Coefficients are obtained by inner products with the trigonometric modes.


% BEGIN VOL02-EXPANSION II14-example-01
\begin{example}[Fourier coefficients of a simple odd function]\label{ex:ii14-expand-01}
Let \(f(x)=x\) on \([-\pi,\pi]\), extended periodically. Odd symmetry kills the
constant and cosine coefficients. For \(n\ge1\),
\[
b_n=\frac1\pi\int_{-\pi}^{\pi}x\sin(nx)\,dx
=\frac{2}{\pi}\int_0^\pi x\sin(nx)\,dx
=\frac{2(-1)^{n+1}}{n}.
\]
Thus the Fourier series is
\[
2\sum_{n\ge1}\frac{(-1)^{n+1}}{n}\sin(nx).
\]
The parity prediction provides a check before any integration is done.
\end{example}
% END VOL02-EXPANSION II14-example-01
\section{Partial sums}
Finite Fourier sums are trigonometric polynomials.

\section{Bessel inequality}
Orthogonality bounds the square-sum of coefficients by the $L^2$ energy.

\section{Dirichlet kernel}
Partial sums can be written as convolution with the Dirichlet kernel.


% BEGIN VOL02-EXPANSION II14-example-02
\begin{example}[Dirichlet kernel as the mechanism of a partial sum]\label{ex:ii14-expand-02}
The \(N\)-th Fourier partial sum can be written
\[
S_Nf(x)=\frac1{2\pi}\int_{-\pi}^{\pi}f(x-t)D_N(t)\,dt,
\]
where
\[
D_N(t)=1+2\sum_{n=1}^N\cos(nt)
=\frac{\sin((N+\tfrac12)t)}{\sin(t/2)}.
\]
This formula shows why convergence is not merely coefficient bookkeeping:
\(D_N\) concentrates near \(t=0\) but also oscillates strongly, so local
regularity and cancellation determine the limiting value.
\end{example}
% END VOL02-EXPANSION II14-example-02
\section{Pointwise convergence ideas}
Smoothness and one-sided limits influence pointwise convergence.

\section{Parseval preview}
When the system is complete, coefficient energy equals function energy.


% BEGIN VOL02-EXPANSION II14-example-03
\begin{example}[Parseval turns coefficient decay into an energy identity]\label{ex:ii14-expand-03}
For \(f(x)=\cos(3x)\) on \([-\pi,\pi]\), all Fourier coefficients vanish except
the \(n=3\) cosine coefficient, which equals \(1\) in the standard
normalization. Parseval therefore gives
\[
\frac1\pi\int_{-\pi}^{\pi}|f(x)|^2\,dx=1.
\]
Directly, the integral of \(\cos^2(3x)\) over a full period is \(\pi\), so the
same normalized energy is \(1\). The identity checks the coefficient
normalization and exhibits orthogonal energy decomposition.
\end{example}
% END VOL02-EXPANSION II14-example-03
\section{Core structural results}

\begin{theorem}[Orthogonality relations]\label{thm:ii14-01}
Distinct sine/cosine modes are orthogonal on $[-\pi,\pi]$.
\begin{proof}
Use product-to-sum identities; integrals of nonconstant integer-frequency sines and cosines over a full period vanish.
\end{proof}
\end{theorem}

\begin{theorem}[Bessel inequality]\label{thm:ii14-02}
For an orthonormal family, the sum of squared Fourier coefficients is bounded by the squared norm of the function.
\begin{proof}
Subtract the finite orthogonal projection and expand its nonnegative squared norm.
\end{proof}
\end{theorem}

\begin{theorem}[Dirichlet midpoint principle]\label{thm:ii14-03}
Under standard piecewise smooth hypotheses, Fourier sums converge at a jump to the midpoint of one-sided limits.
\begin{proof}
Rewrite the partial sum using the Dirichlet kernel and use oscillatory cancellation away from the origin plus local one-sided regularity.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Coefficient of a constant]\label{ex:ii14-worked-01}
Find the Fourier coefficients of $f(x)=1$. Only the constant coefficient is nonzero; all sine and positive cosine modes integrate to zero.
\end{example}

\begin{example}[Odd symmetry]\label{ex:ii14-worked-02}
What coefficients vanish for an odd function? The constant and cosine coefficients vanish; only sine coefficients may remain.
\end{example}

\begin{example}[Even symmetry]\label{ex:ii14-worked-03}
What coefficients vanish for an even function? All sine coefficients vanish.
\end{example}

\begin{example}[Orthogonality computation]\label{ex:ii14-worked-04}
Compute $\int_{-\pi}^{\pi}\cos(nx)\cos(mx)dx$ for $n\ne m$. Product-to-sum reduces it to integrals of nonconstant cosine modes, hence zero.
\end{example}

Find the Fourier coefficients of $f(x)=1$.
\end{problem}
\begin{solution}

\end{solution}

\begin{problem}[Odd symmetry]\label{prob:ii14-dossier-02}
What coefficients vanish for an odd function?
\end{problem}
\begin{solution}
The constant and cosine coefficients vanish; only sine coefficients may remain.
\end{solution}

\begin{problem}[Even symmetry]\label{prob:ii14-dossier-03}
What coefficients vanish for an even function?
\end{problem}
\begin{solution}
All sine coefficients vanish.
\end{solution}

\begin{problem}[Orthogonality computation]\label{prob:ii14-dossier-04}
Compute $\int_{-\pi}^{\pi}\cos(nx)\cos(mx)dx$ for $n\ne m$.
\end{problem}
\begin{solution}
Product-to-sum reduces it to integrals of nonconstant cosine modes, hence zero.
\end{solution}

\begin{problem}[Sawtooth structure]\label{prob:ii14-dossier-05}
Why does an odd sawtooth have a sine-only series?
\end{problem}
\begin{solution}
Odd symmetry kills the even cosine modes and constant term.
\end{solution}

\begin{problem}[Finite projection]\label{prob:ii14-dossier-06}
Why is the $N$th Fourier partial sum the best $L^2$ approximation among trigonometric polynomials of degree at most $N$?
\end{problem}
\begin{solution}
It is the orthogonal projection onto that finite-dimensional subspace.
\end{solution}

\begin{problem}[Bessel estimate]\label{prob:ii14-dossier-07}
If $\|f\|_2=1$, what can be said about Fourier coefficients?
\end{problem}
\begin{solution}
Their squared magnitudes have total finite partial sums bounded by $1$.
\end{solution}

\begin{problem}[Dirichlet kernel integral]\label{prob:ii14-dossier-08}
Why must the Dirichlet kernel have total mass $2\pi$ under the standard normalization?
\end{problem}
\begin{solution}
The partial-sum operator reproduces constants, so integrating the kernel against the constant function must return that constant.
\end{solution}

\begin{problem}[Jump value]\label{prob:ii14-dossier-09}
What Fourier value is expected at a jump from $L_-$ to $L_+$?
\end{problem}
\begin{solution}
The midpoint $(L_-+L_+)/2$ under Dirichlet hypotheses.
\end{solution}

\begin{problem}[Gibbs warning]\label{prob:ii14-dossier-10}
What does Gibbs phenomenon say qualitatively?
\end{problem}
\begin{solution}
Near jumps, partial sums exhibit persistent overshoot whose width shrinks but relative amplitude does not vanish.
\end{solution}

\begin{problem}[Parseval meaning]\label{prob:ii14-dossier-11}
Interpret Parseval's identity.
\end{problem}
\begin{solution}
It says the orthogonal mode coefficients preserve total quadratic energy.
\end{solution}

\begin{problem}[Trigonometric synthesis]\label{prob:ii14-dossier-12}
Why are Fourier coefficients canonical?
\end{problem}
\begin{solution}
Orthogonality isolates each mode by an inner product, making coefficients invariantly determined by projection rather than by solving a coupled system.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii14-01}
What modes survive for an even function?
\end{exercise}
\begin{hint}
Parity. Exploit parity first: even functions kill sine coefficients and odd functions kill cosine coefficients.
\end{hint}
\begin{solution}
Constant and cosines.
\end{solution}

\begin{exercise}\label{exr:ii14-02}
What modes survive for an odd function?
\end{exercise}
\begin{hint}
Parity. Compute Fourier coefficients from their defining integrals over one period and simplify symmetry before integrating.
\end{hint}
\begin{solution}
Sines.
\end{solution}

\begin{exercise}\label{exr:ii14-03}
Are $\sin nx$ and $\cos mx$ orthogonal?
\end{exercise}
\begin{hint}
Parity/product identities. At a jump, the standard convergence theorem gives the midpoint of the one-sided limits rather than either side value.
\end{hint}
\begin{solution}
Yes on a full symmetric period.
\end{solution}

\begin{exercise}\label{exr:ii14-04}
What is a trigonometric polynomial?
\end{exercise}
\begin{hint}
Finite linear combination. Parseval converts an \(L^2\) norm into a sum of squared coefficients; verify normalization constants before applying it.
\end{hint}
\begin{solution}
A finite combination of constant, sine, and cosine modes.
\end{solution}

\begin{exercise}\label{exr:ii14-05}
What is Bessel inequality about?
\end{exercise}
\begin{hint}
Coefficient energy. Integrate by parts to obtain coefficient decay when the function has additional smoothness.
\end{hint}
\begin{solution}
It bounds coefficient square-sums by function energy.
\end{solution}

\begin{exercise}\label{exr:ii14-06}
At a smooth point, what value is expected from Fourier convergence?
\end{exercise}
\begin{hint}
The function value. Compare a trigonometric polynomial with a partial Fourier sum by matching coefficients.
\end{hint}
\begin{solution}
The ordinary value.
\end{solution}

\begin{exercise}\label{exr:ii14-07}
At a jump, what value is expected?
\end{exercise}
\begin{hint}
Average of one-sided limits. Check the period and normalization convention before copying any coefficient formula.
\end{hint}
\begin{solution}
Their midpoint.
\end{solution}

\begin{exercise}\label{exr:ii14-08}
What operator is the Fourier partial sum?
\end{exercise}
\begin{hint}
Orthogonal projection in $L^2$. For tail estimates, use coefficient decay or an \(\ell^2\) identity rather than pointwise term estimates alone.
\end{hint}
\begin{solution}
Projection onto the first modes.
\end{solution}

\subsection*{Standard computations}

\begin{exercise}[Odd symmetry]\label{exr:ii14-expand-01}
For an odd \(2\pi\)-periodic integrable function, which Fourier coefficients vanish?
\end{exercise}
\begin{hint}
Check parity of the integrands.
\end{hint}
\begin{solution}
The constant and all cosine coefficients vanish.
\end{solution}

\begin{exercise}[Even symmetry]\label{exr:ii14-expand-02}
For an even function, which Fourier coefficients vanish?
\end{exercise}
\begin{hint}
Sine is odd.
\end{hint}
\begin{solution}
All sine coefficients vanish.
\end{solution}

\begin{exercise}[Orthogonality]\label{exr:ii14-expand-03}
Compute \(\int_{-\pi}^{\pi}\sin(nx)\sin(mx)\,dx\) for \(n\ne m\).
\end{exercise}
\begin{hint}
Use product-to-sum.
\end{hint}
\begin{solution}
The integral is \(0\).
\end{solution}

\begin{exercise}[Low Chebyshev-like mode]\label{exr:ii14-expand-04}
Find the Fourier series of \(f(x)=\cos(2x)\).
\end{exercise}
\begin{hint}
It is already one trigonometric basis mode.
\end{hint}
\begin{solution}
Only the \(n=2\) cosine coefficient is nonzero and equals \(1\).
\end{solution}

\begin{exercise}[Jump value]\label{exr:ii14-expand-05}
Under Dirichlet hypotheses, to what value does the Fourier series converge at a jump with one-sided limits \(2\) and \(6\)?
\end{exercise}
\begin{hint}
Use the midpoint principle.
\end{hint}
\begin{solution}
It converges to \(4\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Orthogonality proof]\label{exr:ii14-expand-06}
Prove distinct cosine modes are orthogonal on a full period.
\end{exercise}
\begin{hint}
Use \(\cos n\cos m=\frac12[\cos(n-m)+\cos(n+m)]\).
\end{hint}
\begin{solution}
Both nonconstant integer-frequency cosines integrate to zero.
\end{solution}

\begin{exercise}[Bessel inequality]\label{exr:ii14-expand-07}
Prove the finite Bessel inequality for an orthonormal family.
\end{exercise}
\begin{hint}
Subtract the orthogonal projection and expand the nonnegative squared norm.
\end{hint}
\begin{solution}
\(\|f-\sum_{k=1}^N\langle f,e_k\rangle e_k\|^2=\|f\|^2-\sum_{k=1}^N|\langle f,e_k\rangle|^2\ge0\).
\end{solution}

\begin{exercise}[Parity coefficients]\label{exr:ii14-expand-08}
Prove an odd function has zero cosine coefficients.
\end{exercise}
\begin{hint}
The product of odd \(f\) and even cosine is odd.
\end{hint}
\begin{solution}
The integral of an odd integrable function over \([-\pi,\pi]\) vanishes.
\end{solution}

\begin{exercise}[Dirichlet midpoint mechanism]\label{exr:ii14-expand-09}
Explain why a jump is represented by the average of one-sided limits under the standard theorem.
\end{exercise}
\begin{hint}
Symmetric kernel concentration samples both sides.
\end{hint}
\begin{solution}
The local symmetric contribution of the Dirichlet kernel averages the left and right limiting values; oscillatory far contributions cancel.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Formal coefficients insufficient]\label{exr:ii14-expand-10}
Why does writing down Fourier coefficients not by itself prove pointwise convergence to \(f\)?
\end{exercise}
\begin{hint}
Coefficient definition and convergence of partial sums are separate questions.
\end{hint}
\begin{solution}
An expansion may converge in another sense, diverge at points, or converge to midpoint values at jumps.
\end{solution}

\begin{exercise}[Uniform convergence warning]\label{exr:ii14-expand-11}
Why can a Fourier series of a discontinuous periodic function not converge uniformly to that function if partial sums are continuous?
\end{exercise}
\begin{hint}
Uniform limits of continuous functions are continuous.
\end{hint}
\begin{solution}
A discontinuous target rules out uniform convergence of continuous trigonometric partial sums.
\end{solution}

\begin{exercise}[Bessel not Parseval automatically]\label{exr:ii14-expand-12}
Why does Bessel inequality alone not imply equality?
\end{exercise}
\begin{hint}
Equality requires completeness of the orthonormal system for the function space.
\end{hint}
\begin{solution}
Bessel gives only an upper bound on captured coefficient energy; uncaptured orthogonal energy may remain.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Signal energy]\label{exr:ii14-expand-13}
How does Parseval help verify a computed Fourier spectrum?
\end{exercise}
\begin{hint}
Compare time-domain squared norm with coefficient energy.
\end{hint}
\begin{solution}
A mismatch signals missing modes or normalization errors.
\end{solution}

\begin{exercise}[Symmetry shortcut]\label{exr:ii14-expand-14}
Why should parity be checked before computing coefficients numerically?
\end{exercise}
\begin{hint}
It predicts exact zeros.
\end{hint}
\begin{solution}
Using symmetry cuts work and provides a strong diagnostic for numerical integration error.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Best finite Fourier approximation]\label{exr:ii14-expand-15}
Why is the \(N\)-mode Fourier partial sum the best \(L^2\) approximation from the span of those modes?
\end{exercise}
\begin{hint}
It is the orthogonal projection.
\end{hint}
\begin{solution}
Projection minimizes the norm of the residual by Pythagoras.
\end{solution}

\begin{exercise}[Coefficient decay from smoothness]\label{exr:ii14-expand-16}
Use integration by parts heuristically to explain why smoother periodic functions have faster Fourier coefficient decay.
\end{exercise}
\begin{hint}
Move derivatives from the function onto \(e^{-inx}\).
\end{hint}
\begin{solution}
Each integration by parts introduces a factor \(1/n\) when boundary terms are controlled, so additional smoothness yields additional powers of decay.
\end{solution}
