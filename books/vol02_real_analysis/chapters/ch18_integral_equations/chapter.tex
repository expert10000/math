\chapter{Integral Equations}
\label{ch:ii18}

Integral equations encode unknown functions under an integral operator. Their analytic treatment combines norm estimates, compactness ideas, and fixed-point principles.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item rewrite an integral equation as a fixed-point problem on a function space;
\item estimate the associated integral operator in a sup or other relevant norm;
\item verify conditions under which the integral operator is a contraction or maps a bounded set into itself;
\item solve elementary separable or explicitly integrable integral equations;
\item derive successive-approximation schemes and estimate their errors;
\item distinguish existence, uniqueness, and stability conclusions for an integral equation;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Fredholm equations}
The unknown appears under an integral over a fixed domain.

\section{Volterra equations}

\section{Integral operators}
Kernels define linear or nonlinear operators on function spaces.

\section{Sup-norm estimates}
Kernel bounds convert integral operators into norm inequalities.


% BEGIN VOL02-EXPANSION II18-example-01
\begin{example}[A sharp sup-norm kernel estimate]\label{ex:ii18-expand-01}
For
\[
(Kf)(x)=\int_0^1 xt\,f(t)\,dt,\qquad 0\le x\le1,
\]
we have
\[
|Kf(x)|\le x\|f\|_\infty\int_0^1t\,dt
=\frac{x}{2}\|f\|_\infty.
\]
Taking the supremum gives
\[
\|K\|_\infty\le\frac12.
\]
Equality is attained for \(f\equiv1\) at \(x=1\), so the norm is exactly
\(1/2\). The sharp estimate immediately shows \(I-\lambda K\) is invertible
by a Neumann series whenever \(|\lambda|<2\).
\end{example}
% END VOL02-EXPANSION II18-example-01
\section{Neumann series}
Small operator norm permits inversion of $I-K$ by a geometric series.


% BEGIN VOL02-EXPANSION II18-example-02
\begin{example}[Neumann series with an explicit operator error]\label{ex:ii18-expand-02}
If \(\|K\|=q<1\), then
\[
(I-K)^{-1}=I+K+K^2+\cdots.
\]
After truncating at \(N\),
\[
\left\|(I-K)^{-1}-\sum_{j=0}^N K^j\right\|
\le\sum_{j=N+1}^\infty q^j
=\frac{q^{N+1}}{1-q}.
\]
Thus the same geometric series proving invertibility supplies a certified
approximation of the inverse operator.
\end{example}
% END VOL02-EXPANSION II18-example-02
\section{Contraction methods}
Nonlinear integral equations can be solved by Banach's theorem under Lipschitz conditions.

Repeated substitution creates iterated kernel integrals.

\section{Connection to ODEs}
Initial-value ODEs are equivalent to Volterra integral equations.


% BEGIN VOL02-EXPANSION II18-example-03
\begin{example}[An ODE becomes a causal Volterra equation]\label{ex:ii18-expand-03}
The IVP
\[
y'(t)=t+y(t),\qquad y(0)=1
\]
is equivalent to
\[
y(t)=1+\int_0^t(s+y(s))\,ds.
\]
A Picard scheme starts with \(y_0(t)=1\) and defines
\[
y_{n+1}(t)=1+\int_0^t(s+y_n(s))\,ds.
\]
The first iterate is \(1+t+t^2/2\). The upper integration limit \(t\) makes the
operator causal: the value at time \(t\) depends only on earlier values.
\end{example}

\section{Core structural results}

\begin{theorem}[Neumann-series inversion]\label{thm:ii18-01}
If a bounded linear operator $K$ satisfies $\|K\|<1$, then $I-K$ is invertible with $(I-K)^{-1}=\sum_{n\ge0}K^n$.
\begin{proof}
The operator series converges absolutely in norm, and finite geometric identities pass to the limit.
\end{proof}
\end{theorem}

\begin{theorem}[Kernel sup bound]\label{thm:ii18-02}
For $(Kf)(x)=\int_a^b k(x,t)f(t)dt$, $\|K\|_\infty\le(b-a)\|k\|_\infty$.
\begin{proof}
Take absolute values, bound both kernel and function by sup norms, and integrate interval length.
\end{proof}
\end{theorem}

\begin{theorem}[Volterra contraction on short intervals]\label{thm:ii18-03}
If $F(t,y)$ is Lipschitz in $y$ with constant $L$, the integral map $Ty=y_0+\int_{t_0}^tF(s,y(s))ds$ is a contraction on an interval of length $h$ when $Lh<1$.
\begin{proof}
The output difference is at most $Lh$ times the input sup difference.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Constant kernel]\label{ex:ii18-worked-01}
Solve $f(x)=1+\lambda\int_0^1f(t)dt$. Let $c=\int_0^1f$. Then $f=1+\lambda c$ is constant and $c=1+\lambda c$, so for $\lambda\ne1$, $f=1/(1-\lambda)$.
\end{example}

\begin{example}[Norm of a simple kernel operator]\label{ex:ii18-worked-02}
For $Kf(x)=\int_0^1xtf(t)dt$, bound $\|K\|_\infty$. Since $|xt|\le1$, the general bound gives $\|K\|\le1$; the sharper bound is $1/2$ because $\sup_x\int_0^1xtdt=1/2$.
\end{example}

\begin{example}[Neumann approximation]\label{ex:ii18-worked-03}
If $\|K\|=q<1$, bound truncation after $N$ terms. The inverse-series tail has norm at most $q^{N+1}/(1-q)$.
\end{example}

\begin{example}[Volterra uniqueness]\label{ex:ii18-worked-04}
Why are Volterra equations naturally compatible with iteration? The value at time $t$ depends only on earlier times up to $t$, producing causal nested integrals and strong factorial-type bounds.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[Constant kernel]\label{prob:ii18-dossier-01}
Solve $f(x)=1+\lambda\int_0^1f(t)dt$.
\end{problem}
\begin{solution}
Let $c=\int_0^1f$. Then $f=1+\lambda c$ is constant and $c=1+\lambda c$, so for $\lambda\ne1$, $f=1/(1-\lambda)$.
\end{solution}

\begin{problem}[Norm of a simple kernel operator]\label{prob:ii18-dossier-02}
For $Kf(x)=\int_0^1xtf(t)dt$, bound $\|K\|_\infty$.
\end{problem}
\begin{solution}
Since $|xt|\le1$, the general bound gives $\|K\|\le1$; the sharper bound is $1/2$ because $\sup_x\int_0^1xtdt=1/2$.
\end{solution}

\begin{problem}[Neumann approximation]\label{prob:ii18-dossier-03}
If $\|K\|=q<1$, bound truncation after $N$ terms.
\end{problem}
\begin{solution}
The inverse-series tail has norm at most $q^{N+1}/(1-q)$.
\end{solution}

\begin{problem}[Volterra uniqueness]\label{prob:ii18-dossier-04}
Why are Volterra equations naturally compatible with iteration?
\end{problem}
\begin{solution}
The value at time $t$ depends only on earlier times up to $t$, producing causal nested integrals and strong factorial-type bounds.
\end{solution}

\begin{problem}[ODE equivalence]\label{prob:ii18-dossier-05}
Convert $y'=f(t,y)$, $y(t_0)=y_0$ to an integral equation.
\end{problem}
\begin{solution}
Integrate to obtain $y(t)=y_0+\int_{t_0}^tf(s,y(s))ds$.
\end{solution}

\begin{problem}[Linear resolvent idea]\label{prob:ii18-dossier-06}
What does $(I-\lambda K)^{-1}$ represent?
\end{problem}
\begin{solution}
It maps forcing terms to solutions of $f-\lambda Kf=g$ when the inverse exists.
\end{solution}

\begin{problem}[Short-interval contraction]\label{prob:ii18-dossier-07}
If Lipschitz constant is $L=3$, how small should interval length be for direct Banach contraction?
\end{problem}
\begin{solution}
Choose $h<1/3$.
\end{solution}

\begin{problem}[Iterated operator estimate]\label{prob:ii18-dossier-08}
If $\|K\|\le q$, what is $\|K^n\|$?
\end{problem}
\begin{solution}
At most $q^n$ by submultiplicativity.
\end{solution}

\begin{problem}[Kernel continuity]\label{prob:ii18-dossier-09}
Why does a continuous kernel on a compact square define a bounded sup-norm operator?
\end{problem}
\begin{solution}
The kernel is bounded by compactness, and the kernel sup bound applies.
\end{solution}

\begin{problem}[Nonlinear integral map]\label{prob:ii18-dossier-10}
How does a Lipschitz nonlinearity enter the estimate?
\end{problem}
\begin{solution}
Take the difference inside the integral and use the pointwise Lipschitz bound before integration.
\end{solution}

\begin{problem}[Fixed-point viewpoint]\label{prob:ii18-dossier-11}
What equation is actually solved by Banach?
\end{problem}
\begin{solution}
The integral equation is rewritten as $f=Tf$ and the solution is the unique fixed point of $T$.
\end{solution}

\begin{problem}[Integral-equation synthesis]\label{prob:ii18-dossier-12}
Why are integral equations often easier than differential equations?
\end{problem}
\begin{solution}
They encode initial conditions automatically and replace derivatives by averaging operators, which are better behaved in norm estimates.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii18-01}
What operator solves $f=g+Kf$?
\end{exercise}
\begin{hint}
Rearrange. Move every term except the unknown function to one side and define an operator \(Tu\) whose fixed points solve the equation.
\end{hint}
\begin{solution}
$(I-K)f=g$.
\end{solution}

\begin{exercise}\label{exr:ii18-02}
When does Neumann series converge?
\end{exercise}
\begin{hint}
Operator norm below one. Estimate \(\|Tu-Tv\|_\infty\) by pulling \(\|u-v\|_\infty\) outside the integral.
\end{hint}
\begin{solution}
If $\|K\|<1$.
\end{solution}

\begin{exercise}\label{exr:ii18-03}
What norm is convenient on $C[a,b]$?
\end{exercise}
\begin{hint}
Sup norm. Bound the integral kernel uniformly to obtain a contraction constant.
\end{hint}
\begin{solution}
$\|f\|_\infty$.
\end{solution}

\begin{exercise}\label{exr:ii18-04}
What does a Volterra upper limit look like?
\end{exercise}
\begin{hint}
Variable upper endpoint. Before iterating, check that the operator maps the selected closed ball or function class into itself.
\end{hint}
\begin{solution}
$\int_a^x$.
\end{solution}

\begin{exercise}\label{exr:ii18-05}
What does a Fredholm equation use?
\end{exercise}
\begin{hint}
Fixed interval. Picard iteration starts from any convenient initial function and repeatedly applies the integral operator.
\end{hint}
\begin{solution}
$\int_a^b$.
\end{solution}

\begin{exercise}\label{exr:ii18-06}
How do ODEs connect?
\end{exercise}
\begin{hint}
Integrate once. For an explicit kernel, differentiate the integral equation only when regularity justifies doing so.
\end{hint}
\begin{solution}
They become Volterra equations.
\end{solution}

\begin{exercise}\label{exr:ii18-07}
Why does short interval help?
\end{exercise}
\begin{hint}
Multiplies Lipschitz constant. Existence and uniqueness are separate conclusions; identify exactly which estimate supplies each one.
\end{hint}
\begin{solution}
It makes $Lh<1$.
\end{solution}

\begin{exercise}\label{exr:ii18-08}
Does $\|K\|<1$ imply uniqueness for $(I-K)f=g$?
\end{exercise}
\begin{hint}
Use a geometric-series bound on successive differences to estimate the error after finitely many iterations.
\end{hint}
\begin{solution}
Yes, since $I-K$ is invertible.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Constant kernel]\label{exr:ii18-expand-01}
Solve \(f(x)=2+\lambda\int_0^1f(t)\,dt\) for \(\lambda\ne1\).
\end{exercise}
\begin{hint}
The solution must be constant; call its integral \(c\).
\end{hint}
\begin{solution}
\(c=2+\lambda c\), so \(f\equiv2/(1-\lambda)\).
\end{solution}

\begin{exercise}[Kernel bound]\label{exr:ii18-expand-02}
For \(Kf(x)=\int_0^1 x^2t\,f(t)\,dt\), bound \(\|K\|_\infty\).
\end{exercise}
\begin{hint}
Integrate \(t\) and maximize \(x^2\).
\end{hint}
\begin{solution}
\(\|K\|\le1/2\), and equality occurs for \(f\equiv1,x=1\).
\end{solution}

\begin{exercise}[Neumann tail]\label{exr:ii18-expand-03}
If \(\|K\|=1/4\), bound the inverse-series tail after \(N=3\).
\end{exercise}
\begin{hint}
Use \(q^{N+1}/(1-q)\).
\end{hint}
\begin{solution}
The bound is \((1/4)^4/(3/4)=1/192\).
\end{solution}

\begin{exercise}[Volterra first iterate]\label{exr:ii18-expand-04}
For \(y=1+\int_0^t y(s)\,ds\), compute the first two Picard iterates from \(y_0=1\).
\end{exercise}
\begin{hint}
Substitute successively.
\end{hint}
\begin{solution}
\(y_1=1+t\), \(y_2=1+t+t^2/2\).
\end{solution}

\begin{exercise}[Contraction interval]\label{exr:ii18-expand-05}
If the state Lipschitz constant is \(L=5\), how short must an interval be to make the Volterra map contractive?
\end{exercise}
\begin{hint}
Require \(Lh<1\).
\end{hint}
\begin{solution}
Take \(h<1/5\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Kernel operator bound]\label{exr:ii18-expand-06}
Prove \(\|K\|_\infty\le(b-a)\|k\|_\infty\).
\end{exercise}
\begin{hint}
Take absolute values inside the integral.
\end{hint}
\begin{solution}
\(|Kf(x)|\le(b-a)\|k\|_\infty\|f\|_\infty\); take the sup.
\end{solution}

\begin{exercise}[Neumann inversion]\label{exr:ii18-expand-07}
Prove \((I-K)\sum_{j=0}^N K^j=I-K^{N+1}\).
\end{exercise}
\begin{hint}
Use the finite geometric identity.
\end{hint}
\begin{solution}
All intermediate powers cancel; norm convergence lets \(N\to\infty\) when \(\|K\|<1\).
\end{solution}

\begin{exercise}[Volterra contraction]\label{exr:ii18-expand-08}
Prove the Picard integral operator has contraction factor at most \(Lh\) on an interval of length \(h\).
\end{exercise}
\begin{hint}
Estimate the integral of the Lipschitz difference.
\end{hint}
\begin{solution}
\(\|Ty-Tz\|_\infty\le Lh\|y-z\|_\infty\).
\end{solution}

\begin{exercise}[ODE equivalence]\label{exr:ii18-expand-09}
Prove a \(C^1\) solution of the IVP is equivalent to its Volterra integral equation.
\end{exercise}
\begin{hint}
Integrate the ODE and differentiate the integral equation.
\end{hint}
\begin{solution}
The fundamental theorem of calculus gives both directions.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Neumann condition sufficient not necessary]\label{exr:ii18-expand-10}
Why does \(\|K\|<1\) give a sufficient but not necessary invertibility condition for \(I-K\)?
\end{exercise}
\begin{hint}
An operator may have norm above one while \(1\) is still outside its spectrum.
\end{hint}
\begin{solution}
The geometric-series proof fails when the norm is large, but invertibility can still hold by other arguments.
\end{solution}

\begin{exercise}[Kernel bound not sharp]\label{exr:ii18-expand-11}
Why can \((b-a)\|k\|_\infty\) overestimate the true operator norm?
\end{exercise}
\begin{hint}
It ignores cancellation and the integral profile in \(t\).
\end{hint}
\begin{solution}
A sharper bound often uses \(\sup_x\int|k(x,t)|dt\).
\end{solution}

\begin{exercise}[No uniqueness without Lipschitz]\label{exr:ii18-expand-12}
Why may a nonlinear integral equation have multiple solutions without a contraction condition?
\end{exercise}
\begin{hint}
Banach's uniqueness mechanism is lost.
\end{hint}
\begin{solution}
Continuity or self-map properties alone can imply existence without uniqueness.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Integral-equation solver]\label{exr:ii18-expand-13}
How does successive approximation turn an integral equation into an algorithm?
\end{exercise}
\begin{hint}
Iterate the associated operator.
\end{hint}
\begin{solution}
Each step evaluates known integrals from the previous approximation; contraction estimates certify convergence and error.
\end{solution}

\begin{exercise}[Resolvent stability]\label{exr:ii18-expand-14}
Why does small \(\|K\|\) make \(f-Kf=g\) stable?
\end{exercise}
\begin{hint}
Bound \(\|(I-K)^{-1}\|\).
\end{hint}
\begin{solution}
The Neumann series gives \(\|(I-K)^{-1}\|\le1/(1-\|K\|)\), controlling amplification of data errors.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Factorial Volterra bound]\label{exr:ii18-expand-15}
Explain why iterated Volterra operators can have \(1/n!\)-type bounds even when the one-step contraction estimate is not small globally.
\end{exercise}
\begin{hint}
Nested time integrals occupy an ordered simplex.
\end{hint}
\begin{solution}
The \(n\)-fold integral region has volume \(h^n/n!\), producing stronger decay for iterates.
\end{solution}

\begin{exercise}[Rank-one Fredholm operator]\label{exr:ii18-expand-16}
For \(Kf=\phi(x)\int\psi(t)f(t)dt\), reduce \(f-\lambda Kf=g\) to a scalar equation.
\end{exercise}
\begin{hint}
Let \(c=\int\psi f\).
\end{hint}
\begin{solution}
Then \(f=g+\lambda c\phi\), and \(c=\int\psi g+\lambda c\int\psi\phi\); solve this scalar equation when the denominator is nonzero.
\end{solution}
