\chapter{Numerical Quadrature}
\label{ch:ii24}

Numerical quadrature replaces an integral by a weighted finite sum of function values. Interpolation gives a systematic derivation, while error formulas explain exactness and the role of node placement.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item derive elementary quadrature rules by integrating an interpolating polynomial;
\item compute degrees of exactness for given quadrature formulas;
\item estimate quadrature errors from derivative bounds and remainder formulas;
\item compare midpoint, trapezoidal, and Simpson-type rules on explicit integrands;
\item construct composite rules and track how local error accumulates globally;
\item choose a mesh size that achieves a prescribed error tolerance from an available error estimate;
\end{itemize}

\section{Quadrature rules}
Approximate $\int_a^bf$ by $\sum w_if(x_i)$.

\section{Interpolatory quadrature}
Integrate an interpolation polynomial exactly to obtain weights.

\section{Newton--Cotes rules}
Equally spaced nodes yield midpoint, trapezoidal, Simpson, and higher rules.

\section{Degree of exactness}
A rule has degree $m$ if it integrates all polynomials through degree $m$ exactly.


% BEGIN VOL02-EXPANSION II24-example-01
\begin{example}[Degree of exactness by testing monomials]\label{ex:ii24-expand-01}
For the trapezoidal rule on \([0,1]\),
\[
Q(f)=\frac12(f(0)+f(1)).
\]
It is exact for \(1\) and \(x\):
\[
Q(1)=1=\int_0^1 1\,dx,\qquad
Q(x)=\frac12=\int_0^1x\,dx.
\]
But
\[
Q(x^2)=\frac12\ne\frac13.
\]
Therefore the degree of exactness is exactly \(1\). Testing the first failing
monomial is what certifies the degree rather than merely a lower bound.
\end{example}
% END VOL02-EXPANSION II24-example-01
\section{Error formulas}
Remainders depend on higher derivatives and powers of step size.

\section{Composite rules}
Apply a low-order rule repeatedly on a fine partition.


% BEGIN VOL02-EXPANSION II24-example-02
\begin{example}[Composite trapezoidal error scales quadratically in the mesh]\label{ex:ii24-expand-02}
For \(f\in C^2[a,b]\) and uniform step \(h=(b-a)/n\), summing the one-panel
trapezoidal errors gives
\[
|E_T|\le \frac{b-a}{12}h^2\|f''\|_\infty.
\]
Thus halving the mesh reduces the bound by a factor of four. To meet a target
tolerance, solve this inequality for \(h\) or \(n\). The global order comes
from local cubic error accumulated over \(O(1/h)\) panels.
\end{example}
% END VOL02-EXPANSION II24-example-02
\section{Gaussian quadrature}
Optimally chosen nodes increase polynomial exactness for a fixed number of samples.


% BEGIN VOL02-EXPANSION II24-example-03
\begin{example}[Two Gaussian nodes integrate cubics exactly]\label{ex:ii24-expand-03}
On \([-1,1]\), the two-node Gauss--Legendre rule uses
\[
x=\pm\frac1{\sqrt3},\qquad w_1=w_2=1.
\]
Symmetry makes odd monomials integrate exactly. For \(x^2\),
\[
Q(x^2)=2\cdot\frac13=\frac23=\int_{-1}^1x^2\,dx.
\]
Constants are also exact, so every polynomial through degree \(3\) is integrated
exactly. Two intelligently chosen samples therefore achieve higher exactness
than two equally spaced endpoint samples.
\end{example}
% END VOL02-EXPANSION II24-example-03

\section{Core structural results}

\begin{theorem}[Trapezoidal error]\label{thm:ii24-01}
For $f\in C^2[a,b]$, the one-panel trapezoidal error equals $-(b-a)^3f''(\xi)/12$ for some $\xi$.
\begin{proof}
Integrate the linear interpolation remainder or use a Peano-kernel argument.
\end{proof}
\end{theorem}

\begin{theorem}[Simpson exactness]\label{thm:ii24-02}
Simpson's rule is exact for polynomials of degree at most three.
\begin{proof}
Check the rule on the basis $1,x,x^2,x^3$ after affine reduction to a symmetric interval.
\end{proof}
\end{theorem}

\begin{theorem}[Gaussian exactness]\label{thm:ii24-03}
An $n$-node Gaussian rule for a positive weight is exact for polynomials through degree $2n-1$.
\begin{proof}
Orthogonal-polynomial nodes make the error polynomial divisible by the degree-$n$ orthogonal polynomial; division separates any degree-$2n-1$ polynomial into a vanishing part plus a lower-degree remainder integrated exactly by the weights.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Midpoint rule]\label{ex:ii24-worked-01}
Approximate $\int_0^1x^2dx$ by one midpoint sample. The midpoint value is $1/4$, so the one-panel approximation is $1/4$ versus exact $1/3$.
\end{example}

\begin{example}[Trapezoidal rule]\label{ex:ii24-worked-02}
Approximate the same integral. $(f(0)+f(1))/2=1/2$.
\end{example}

\begin{example}[Simpson rule]\label{ex:ii24-worked-03}
Apply Simpson to $x^2$ on $[0,1]$. $(1/6)(f(0)+4f(1/2)+f(1))=(1/6)(0+1+1)=1/3$, exact.
\end{example}

\begin{example}[Degree of exactness]\label{ex:ii24-worked-04}
Why is trapezoidal exact for linear functions? It integrates the linear interpolant, which equals the function itself for degree at most one.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[Midpoint rule]\label{prob:ii24-dossier-01}
Approximate $\int_0^1x^2dx$ by one midpoint sample.
\end{problem}
\begin{solution}
The midpoint value is $1/4$, so the one-panel approximation is $1/4$ versus exact $1/3$.
\end{solution}

\begin{problem}[Trapezoidal rule]\label{prob:ii24-dossier-02}
Approximate the same integral.
\end{problem}
\begin{solution}
$(f(0)+f(1))/2=1/2$.
\end{solution}

\begin{problem}[Simpson rule]\label{prob:ii24-dossier-03}
Apply Simpson to $x^2$ on $[0,1]$.
\end{problem}
\begin{solution}
$(1/6)(f(0)+4f(1/2)+f(1))=(1/6)(0+1+1)=1/3$, exact.
\end{solution}

\begin{problem}[Degree of exactness]\label{prob:ii24-dossier-04}
Why is trapezoidal exact for linear functions?
\end{problem}
\begin{solution}
It integrates the linear interpolant, which equals the function itself for degree at most one.
\end{solution}

\begin{problem}[Composite trapezoid]\label{prob:ii24-dossier-05}
What changes when the interval is subdivided?
\end{problem}
\begin{solution}
Apply trapezoidal rule on each subinterval and sum; the global error decreases like $h^2$ under smoothness.
\end{solution}

\begin{problem}[Composite Simpson]\label{prob:ii24-dossier-06}
What order is expected?
\end{problem}
\begin{solution}
For sufficiently smooth functions the global error behaves like $h^4$.
\end{solution}

\begin{problem}[Interpolatory weights]\label{prob:ii24-dossier-07}
How are weights obtained?
\end{problem}
\begin{solution}
Integrate each Lagrange cardinal polynomial over the interval.
\end{solution}

\begin{problem}[Positive weights]\label{prob:ii24-dossier-08}
Why are positive quadrature weights desirable?
\end{problem}
\begin{solution}
They improve stability and preserve positivity for nonnegative sampled data.
\end{solution}

\begin{problem}[Gaussian advantage]\label{prob:ii24-dossier-09}
Why can $n$ Gaussian nodes beat $n$ equally spaced interpolatory nodes?
\end{problem}
\begin{solution}
The nodes are optimized jointly with weights, yielding degree of exactness $2n-1$.
\end{solution}

\begin{problem}[Error and smoothness]\label{prob:ii24-dossier-10}
Why do high-order rules need higher derivatives?
\end{problem}
\begin{solution}
Their error formulas involve higher derivatives; without regularity, formal high order may not translate into actual accuracy.
\end{solution}

\begin{problem}[Adaptive idea]\label{prob:ii24-dossier-11}
How does adaptive quadrature choose where to sample more?
\end{problem}
\begin{solution}
Estimate local error on subintervals and refine where the estimate is largest.
\end{solution}

\begin{problem}[Quadrature synthesis]\label{prob:ii24-dossier-12}
How is quadrature connected to interpolation?
\end{problem}
\begin{solution}
Interpolatory quadrature integrates a polynomial surrogate exactly; the quadrature error is therefore the integral of the interpolation error.
\end{solution}

\begin{exercise}\label{exr:ii24-01}
Degree of trapezoidal exactness?
\end{exercise}
\begin{hint}
Linear. Derive the quadrature weights by integrating the interpolating basis polynomials, not by memorizing the rule.

\begin{solution}
$1$.
\end{solution}

\begin{exercise}\label{exr:ii24-02}
Degree of Simpson exactness?
\end{exercise}
\begin{hint}
Cubic. Test degree of exactness on monomials \(1,x,x^2,\ldots\) until the first failure.
\end{hint}
\begin{solution}

\end{solution}

\begin{exercise}\label{exr:ii24-03}
What does Gaussian $n$-point exactness reach?
\end{exercise}
\begin{hint}
$2n-1$. For an error estimate, insert the derivative bound into the interpolation or Peano-type remainder formula.
\end{hint}
\begin{solution}
Degree $2n-1$.
\end{solution}

\begin{exercise}\label{exr:ii24-04}
Why composite rules?
\end{exercise}
\begin{hint}
Reduce step size. Exploit symmetry of the rule and interval to eliminate odd terms when possible.
\end{hint}
\begin{solution}
To lower error on long intervals.
\end{solution}

\begin{exercise}\label{exr:ii24-05}
What are quadrature weights?
\end{exercise}
\begin{hint}
Integrated basis functions. For a composite rule, apply the local error bound on each subinterval and then sum the contributions.
\end{hint}
\begin{solution}
Coefficients multiplying samples.
\end{solution}

\begin{exercise}\label{exr:ii24-06}
Is Simpson exact for $x^4$?
\end{exercise}
\begin{hint}
Express the mesh width \(h\) in terms of the number of panels before solving the tolerance inequality.
\end{hint}

Not generally.
\end{solution}

\begin{exercise}\label{exr:ii24-07}
What smoothness controls trapezoidal error?
\end{exercise}
\begin{hint}
Second derivative. Compare rules at the same step size using their order and the size of the relevant derivative.
\end{hint}

A bound on $f''$.
\end{solution}

What is adaptive refinement based on?

\end{hint}
\begin{solution}

\end{solution}

Approximate \(\int_0^1x^2dx\) by one midpoint sample.

Average endpoint values.

Apply Simpson's rule to \(\int_0^1x^3dx\).

\begin{hint}
Use weights \(1,4,1\).

\(\frac16(0+4(1/8)+1)=1/4\), exact.

What is the degree of exactness of the midpoint rule on a symmetric interval?
\end{exercise}
\begin{hint}
Test \(1,x,x^2\).

It is \(1\): constants and linear functions are exact, quadratics are not.

\subsection*{Proofs}

\begin{exercise}[Trapezoid exactness]\label{exr:ii24-expand-06}
Prove trapezoidal quadrature is exact for all linear functions.
\end{exercise}
\begin{hint}
The linear interpolant equals the function.
\end{hint}
\begin{solution}
The rule integrates the endpoint interpolating line exactly.
\end{solution}

\begin{exercise}[Simpson exactness]\label{exr:ii24-expand-07}
Verify Simpson's rule on the basis \(1,x,x^2,x^3\) after scaling to \([-1,1]\).
\end{exercise}
\begin{hint}
Use symmetry for odd powers.
\end{hint}
\begin{solution}
Direct evaluation matches the exact moments for all four monomials.
\end{solution}

\begin{exercise}[Composite error scaling]\label{exr:ii24-expand-08}
Explain why a local \(O(h^3)\) panel error leads to global \(O(h^2)\) error over a fixed interval.
\end{exercise}
\begin{hint}
There are \(O(1/h)\) panels.
\end{hint}
\begin{solution}
\((1/h)\cdot h^3=O(h^2)\).
\end{solution}

\begin{exercise}[Gaussian degree]\label{exr:ii24-expand-09}
Explain why \(n\) Gaussian nodes can reach degree \(2n-1\).
\end{exercise}
\begin{hint}
Divide a polynomial by the degree-\(n\) orthogonal polynomial whose zeros are the nodes.
\end{hint}
\begin{solution}
The node-vanishing part integrates to zero by orthogonality, and the lower-degree remainder is matched by the weights.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[More nodes not automatically stable]\label{exr:ii24-expand-10}
Why can high-order equally spaced Newton--Cotes rules become unattractive?
\end{exercise}
\begin{hint}
Weights can grow and alternate, amplifying noise.
\end{hint}
\begin{solution}
Higher algebraic order can come with poor numerical stability.
\end{solution}

\begin{exercise}[Exactness not universal accuracy]\label{exr:ii24-expand-11}
Why does high polynomial degree of exactness not guarantee small error for every nonsmooth integrand?
\end{exercise}
\begin{hint}
Error formulas require smoothness/derivative control.
\end{hint}
\begin{solution}
A rule can be exact on many polynomials yet perform poorly on singular or highly oscillatory functions.
\end{solution}

\begin{exercise}[Single-panel limitation]\label{exr:ii24-expand-12}
Why can one Simpson panel be inaccurate on a long interval even though it is cubic-exact?
\end{exercise}
\begin{hint}
Noncubic remainder grows with interval length and fourth derivatives.
\end{hint}
\begin{solution}
Composite subdivision reduces the local scale and hence the remainder.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Mesh selection]\label{exr:ii24-expand-13}
How do you choose \(n\) from a composite trapezoid error bound?
\end{exercise}
\begin{hint}
Set the bound below tolerance and solve for \(h=(b-a)/n\).
\end{hint}
\begin{solution}
The derivative bound and tolerance produce an explicit lower bound on \(n\).
\end{solution}

\begin{exercise}[Adaptive quadrature]\label{exr:ii24-expand-14}
Why compare coarse and refined quadrature estimates locally?
\end{exercise}
\begin{hint}
The discrepancy is an empirical proxy for local truncation error.
\end{hint}
\begin{solution}
Subdivide where the estimates disagree most, concentrating samples where the integrand is hardest.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Moment equations]\label{exr:ii24-expand-15}
Derive symmetric two-node weights/nodes on \([-1,1]\) by requiring exactness for \(1\) and \(x^2\).
\end{exercise}
\begin{hint}
Let nodes be \(\pm a\) with equal weight \(w\).
\end{hint}
\begin{solution}
Constant exactness gives \(2w=2\Rightarrow w=1\); quadratic exactness gives \(2a^2=2/3\Rightarrow a=1/\sqrt3\).
\end{solution}

\begin{exercise}[Richardson idea]\label{exr:ii24-expand-16}
If an error behaves like \(Ch^p\), explain how two resolutions can eliminate the leading error term.
\end{exercise}
\begin{hint}
Combine \(Q_h\) and \(Q_{h/2}\) using the factor \(2^p\).
\end{hint}
\begin{solution}
\((2^pQ_{h/2}-Q_h)/(2^p-1)\) cancels the leading \(Ch^p\) term under the asymptotic model.
\end{solution}
