\chapter{Euclidean and Normed Spaces}
\label{ch:ii02}

Normed spaces add quantitative geometry to vector spaces. The norm axioms isolate exactly the estimates needed to speak about distance, convergence, boundedness, and continuity.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute norms and distances in Euclidean and normed spaces and verify the norm axioms in concrete examples;
\item compare two norms by proving explicit upper and lower inequalities between them;
\item describe open balls in several norms and explain how their geometry differs;
\item use the triangle and reverse-triangle inequalities to bound perturbations and distances;
\item decide whether a proposed formula defines a norm and identify the failed axiom when it does not;
\item translate coordinate estimates into norm estimates using finite-dimensional norm equivalence;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Euclidean norm}
The Euclidean norm is induced by the standard inner product and measures ordinary length.

\section{Norm axioms}
A norm is positive definite, absolutely homogeneous, and subadditive.

\section{Standard finite-dimensional norms}
The $1$-, $2$-, and infinity norms emphasize total magnitude, Euclidean length, and largest coordinate.


% BEGIN VOL02-EXPANSION II02-example-01
\begin{example}[Comparing three standard norms]\label{ex:ii02-expand-01}
For \(x=(1,-2,2)\),
\[
\|x\|_1=5,\qquad \|x\|_2=3,\qquad \|x\|_\infty=2.
\]
The general inequalities
\[
\|x\|_\infty\le\|x\|_2\le\sqrt3\,\|x\|_\infty
\]
become \(2\le3\le2\sqrt3\), which checks the computation. The example also
shows that equivalent norms need not assign the same numerical length; they
agree instead on the qualitative notions of convergence and openness.
\end{example}
% END VOL02-EXPANSION II02-example-01
\section{Balls and spheres}
Norm balls define neighborhoods and therefore the topology of a normed space.

\section{Operator norms}
The induced norm of a linear map measures its maximal stretching on the unit sphere.


% BEGIN VOL02-EXPANSION II02-example-02
\begin{example}[Computing an induced operator norm]\label{ex:ii02-expand-02}
Let \(T(x,y)=(2x,0)\) on Euclidean \(\mathbb R^2\). For a unit vector
\(v=(x,y)\),
\[
\|Tv\|_2=2|x|\le2.
\]
Equality occurs at \(v=e_1\), so
\[
\|T\|_{\mathrm{op}}=2.
\]
The upper estimate and an attaining unit vector are both needed to certify the
exact operator norm.
\end{example}
% END VOL02-EXPANSION II02-example-02
\section{Equivalent norms}
Comparable norms define the same convergence and open sets.


% BEGIN VOL02-EXPANSION II02-example-03
\begin{example}[Norm equivalence transfers convergence]\label{ex:ii02-expand-03}
Suppose on a finite-dimensional space
\[
c\|v\|_a\le\|v\|_b\le C\|v\|_a
\]
for positive constants \(c,C\). If \(v_n\to v\) in \(\|\cdot\|_a\), then
\[
\|v_n-v\|_b\le C\|v_n-v\|_a\to0.
\]
The reverse inequality gives the converse. Thus equivalent norms have exactly
the same convergent sequences, even though their balls may have very different
shapes.
\end{example}
% END VOL02-EXPANSION II02-example-03
\section{Convexity}
Every norm ball is convex by the triangle inequality.

\section{Finite versus infinite dimensions}
All norms are equivalent in finite dimensions, but this can fail in infinite-dimensional spaces.

\section{Core structural results}

\begin{theorem}[Cauchy--Schwarz]\label{thm:ii02-01}
For $x,y\in\mathbb R^n$, $|\langle x,y\rangle|\le\|x\|_2\|y\|_2$.
\begin{proof}
Apply nonnegativity to $\|x-ty\|_2^2$ and require its quadratic discriminant to be nonpositive.
\end{proof}
\end{theorem}

\begin{theorem}[Euclidean triangle inequality]\label{thm:ii02-02}
$\|x+y\|_2\le\|x\|_2+\|y\|_2$.
\begin{proof}
Expand the square and use Cauchy--Schwarz on the cross term.
\end{proof}
\end{theorem}

\begin{theorem}[Norm comparison]\label{thm:ii02-03}
On $\mathbb R^n$, $\|x\|_\infty\le\|x\|_2\le\sqrt n\|x\|_\infty$.
\begin{proof}
The largest coordinate square is bounded by the sum of squares, and each coordinate square is bounded by the largest one.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Compare $1$- and $2$-norms]\label{ex:ii02-worked-01}
Prove $\|x\|_2\le\|x\|_1\le\sqrt n\|x\|_2$. The first follows by expanding $(\sum|x_i|)^2$; the second is Cauchy--Schwarz against $(1,\dots,1)$.
\end{example}

\begin{example}[A seminorm, not a norm]\label{ex:ii02-worked-02}
Is $p(x,y)=|x-y|$ a norm on $\mathbb R^2$? No: $p(1,1)=0$ although $(1,1)\ne0$.
\end{example}

\begin{example}[Weighted Euclidean norm]\label{ex:ii02-worked-03}
Show $\sqrt{4x^2+y^2}$ is a norm. It is the Euclidean norm after the invertible linear change $(x,y)\mapsto(2x,y)$.
\end{example}

\begin{example}[Distance to a coordinate axis]\label{ex:ii02-worked-04}
Find the Euclidean distance from $(1,2)$ to the $x$-axis. Minimizing $(1-t)^2+4$ over $(t,0)$ gives $t=1$ and distance $2$.
\end{example}

\section{Solved dossiers}
Each dossier is a substantial solved problem. Corpus mappings guide topic choice where explicit retained source blocks exist; otherwise the dossier is a fresh canonical completion.

\begin{problem}[Compare $1$- and $2$-norms]\label{prob:ii02-dossier-01}
Prove $\|x\|_2\le\|x\|_1\le\sqrt n\|x\|_2$.
\end{problem}
\begin{solution}
The first follows by expanding $(\sum|x_i|)^2$; the second is Cauchy--Schwarz against $(1,\dots,1)$.
\end{solution}

\begin{problem}[A seminorm, not a norm]\label{prob:ii02-dossier-02}
Is $p(x,y)=|x-y|$ a norm on $\mathbb R^2$?
\end{problem}
\begin{solution}
No: $p(1,1)=0$ although $(1,1)\ne0$.
\end{solution}

\begin{problem}[Weighted Euclidean norm]\label{prob:ii02-dossier-03}
Show $\sqrt{4x^2+y^2}$ is a norm.
\end{problem}
\begin{solution}
It is the Euclidean norm after the invertible linear change $(x,y)\mapsto(2x,y)$.
\end{solution}

\begin{problem}[Distance to a coordinate axis]\label{prob:ii02-dossier-04}
Find the Euclidean distance from $(1,2)$ to the $x$-axis.
\end{problem}
\begin{solution}
Minimizing $(1-t)^2+4$ over $(t,0)$ gives $t=1$ and distance $2$.
\end{solution}

\begin{problem}[Projection operator norm]\label{prob:ii02-dossier-05}
For $P(x,y)=(x,0)$, find $\|P\|_2$.
\end{problem}
\begin{solution}
$\|P(x,y)\|\le\|(x,y)\|$ and equality holds at $(1,0)$, so $\|P\|=1$.
\end{solution}

\begin{problem}[Equivalent convergence]\label{prob:ii02-dossier-06}
Show $\|x_n-x\|_2\to0$ iff $\|x_n-x\|_\infty\to0$.
\end{problem}
\begin{solution}
Use $\|z\|_\infty\le\|z\|_2\le\sqrt n\|z\|_\infty$.
\end{solution}

\begin{problem}[Diameter of the unit sphere]\label{prob:ii02-dossier-07}
If $\|x\|=\|y\|=1$, show $\|x-y\|\le2$.
\end{problem}
\begin{solution}
Use the triangle inequality on $x+(-y)$.
\end{solution}

\begin{problem}[Reverse triangle inequality]\label{prob:ii02-dossier-08}
Prove $|\|x\|-\|y\||\le\|x-y\|$.
\end{problem}
\begin{solution}
Apply the triangle inequality to $x=(x-y)+y$ and then swap $x,y$.
\end{solution}

\begin{problem}[Norm of a Euclidean functional]\label{prob:ii02-dossier-09}
For $f(x)=a\cdot x$, compute $\|f\|$.
\end{problem}
\begin{solution}
Cauchy--Schwarz gives $\|f\|\le\|a\|_2$, with equality at $x=a/\|a\|_2$.
\end{solution}

\begin{problem}[Convexity of closed balls]\label{prob:ii02-dossier-10}
Prove every closed norm ball is convex.
\end{problem}
\begin{solution}
A convex combination of two radius-$r$ displacement vectors has norm at most the same convex combination of their norms, hence at most $r$.
\end{solution}

\begin{problem}[Infinite-dimensional warning]\label{prob:ii02-dossier-11}
Why is there no $C$ with $\|f\|_\infty\le C\|f\|_1$ for every continuous $f$ on $[0,1]$?
\end{problem}
\begin{solution}
Continuous spike functions can keep height $1$ while their support and integral shrink to zero.
\end{solution}

\begin{problem}[Equivalent-norm synthesis]\label{prob:ii02-dossier-12}
Why do two-sided norm bounds imply identical topology?
\end{problem}
\begin{solution}
Each ball for one norm contains a suitably scaled ball for the other, in both directions.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii02-01}
Compute all three standard norms of $(1,-1,2)$.
\end{exercise}
\begin{hint}
Use definitions. Write out the norm axioms one at a time; homogeneity and the triangle inequality are usually the decisive tests.
\end{hint}
\begin{solution}
They are $4,\sqrt6,2$.
\end{solution}

\begin{exercise}\label{exr:ii02-02}
Show $2|x|+3|y|$ is a norm.
\end{exercise}
\begin{hint}
Check the axioms. Compute the distance from the norm definition before trying to visualize the geometry.
\end{hint}
\begin{solution}
Positive weighted sums of absolute values satisfy the norm axioms.
\end{solution}

\begin{exercise}\label{exr:ii02-03}
Is $\max(|x|,2|y|)$ a norm?
\end{exercise}
\begin{hint}
View it as a weighted infinity norm. Describe the unit ball by solving the inequality \(\|x\|<1\) in coordinates.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii02-04}
Find the Euclidean distance from $(1,2)$ to $(-2,6)$.
\end{exercise}
\begin{hint}
Subtract first. Use the triangle inequality twice to obtain a reverse-triangle estimate for the difference of norms.
\end{hint}
\begin{solution}
The difference has length $5$.
\end{solution}

\begin{exercise}\label{exr:ii02-05}
Prove $\|0\|=0$.
\end{exercise}
\begin{hint}
Use homogeneity with scalar zero. In finite dimensions, compare each coordinate with the norm and then sum the coordinate bounds.
\end{hint}
\begin{solution}
$\|0x\|=0\|x\|=0$.
\end{solution}

\begin{exercise}\label{exr:ii02-06}
If $x_n\to x$, show $\|x_n\|\to\|x\|$.
\end{exercise}
\begin{hint}
Use reverse triangle inequality. To disprove a norm, test zero definiteness with a nonzero vector before attempting harder inequalities.
\end{hint}
\begin{solution}
The norm difference is bounded by $\|x_n-x\|$.
\end{solution}

\begin{exercise}\label{exr:ii02-07}
Find the operator norm of $T(x,y)=(3x,0)$.
\end{exercise}
\begin{hint}
Bound and attain. Normalize a nonzero vector to reduce a general estimate to the unit sphere.
\end{hint}
\begin{solution}
The norm is $3$.
\end{solution}

\begin{exercise}\label{exr:ii02-08}
Show the open unit ball is convex.
\end{exercise}
\begin{hint}
Use strict norm bounds. Track constants explicitly when proving norm equivalence; the direction of each inequality matters.
\end{hint}
\begin{solution}
A convex combination has norm below the corresponding convex combination of two numbers below $1$.
\end{solution}

\subsection*{Standard computations}

\begin{exercise}[Three norms]\label{exr:ii02-expand-01}
Compute \(\|(1,-2,3)\|_1,\|\,\cdot\,\|_2,\|\,\cdot\,\|_\infty\).
\end{exercise}
\begin{hint}
Use the definitions sum, square-root of squares, and maximum coordinate magnitude.
\end{hint}
\begin{solution}
The values are \(6,\sqrt{14},3\).
\end{solution}

\begin{exercise}[Weighted norm]\label{exr:ii02-expand-02}
Compute \(\sqrt{4x^2+y^2}\) at \((2,-3)\).
\end{exercise}
\begin{hint}
Substitute directly.
\end{hint}
\begin{solution}
The value is \(\sqrt{16+9}=5\).
\end{solution}

\begin{exercise}[Ball geometry]\label{exr:ii02-expand-03}
Describe the unit ball of \(\|\cdot\|_\infty\) in \(\mathbb R^2\).
\end{exercise}
\begin{hint}
Translate \(\max(|x|,|y|)<1\).
\end{hint}
\begin{solution}
It is the open square \((-1,1)\times(-1,1)\).
\end{solution}

\begin{exercise}[Operator stretch]\label{exr:ii02-expand-04}
For \(T=\operatorname{diag}(3,1)\), compute the Euclidean operator norm.
\end{exercise}
\begin{hint}
Bound \(\sqrt{9x^2+y^2}\) on \(x^2+y^2=1\).
\end{hint}
\begin{solution}
The maximum is \(3\), attained at \(e_1\).
\end{solution}

\begin{exercise}[Distance]\label{exr:ii02-expand-05}
Find the \(\ell^1\)-distance between \((1,2)\) and \((-1,5)\).
\end{exercise}
\begin{hint}
Apply the norm to their difference.
\end{hint}
\begin{solution}
\(|2|+|-3|=5\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Convexity of norm balls]\label{exr:ii02-expand-06}
Prove every norm ball is convex.
\end{exercise}
\begin{hint}
For \(0\le t\le1\), estimate the norm of \(t(x-a)+(1-t)(y-a)\).
\end{hint}
\begin{solution}
The triangle inequality gives a value below \(tr+(1-t)r=r\).
\end{solution}

\begin{exercise}[Reverse norm inequality]\label{exr:ii02-expand-07}
Prove \(|\|x\|-\|y\||\le\|x-y\|\).
\end{exercise}
\begin{hint}
Use \(x=(x-y)+y\) and swap.
\end{hint}
\begin{solution}
\(\|x\|-\|y\|\le\|x-y\|\) and the reversed inequality follow.
\end{solution}

\begin{exercise}[Norm metric]\label{exr:ii02-expand-08}
Prove \(d(x,y)=\|x-y\|\) is a metric.
\end{exercise}
\begin{hint}
Match each metric axiom to a norm axiom.
\end{hint}
\begin{solution}
Nonnegativity/definiteness, symmetry via \(\|-v\|=\|v\|\), and the triangle inequality follow directly.
\end{solution}

\begin{exercise}[Equivalent norms give same open sets]\label{exr:ii02-expand-09}
Prove comparable norms define the same topology.
\end{exercise}
\begin{hint}
Use the constants to put a ball for one norm inside a ball for the other.
\end{hint}
\begin{solution}
If \(c\|v\|_a\le\|v\|_b\le C\|v\|_a\), then \(B_a(x,r/C)\subset B_b(x,r)\) and conversely.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Seminorm]\label{exr:ii02-expand-10}
Why is \(p(x,y)=|x-y|\) not a norm?
\end{exercise}
\begin{hint}
Test positive definiteness.
\end{hint}
\begin{solution}
\(p(1,1)=0\) although \((1,1)\ne0\).
\end{solution}

\begin{exercise}[Wrong homogeneity]\label{exr:ii02-expand-11}
Why is \(p(x)=|x|^2\) not a norm on \(\mathbb R\)?
\end{exercise}
\begin{hint}
Test scaling by \(2\).
\end{hint}
\begin{solution}
\(p(2x)=4p(x)\), not \(2p(x)\).
\end{solution}

\begin{exercise}[Infinite-dimensional warning]\label{exr:ii02-expand-12}
Explain why finite-dimensional norm equivalence cannot simply be assumed on \(C[0,1]\).
\end{exercise}
\begin{hint}
Compare sup and integral-type norms on sharply peaked functions.
\end{hint}
\begin{solution}
One can build narrow peaks with fixed sup norm but arbitrarily small integral norm, preventing a uniform lower comparison constant.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Perturbation bound]\label{exr:ii02-expand-13}
If \(\|x-y\|\le10^{-4}\), how much can the norm change?
\end{exercise}
\begin{hint}
Use the reverse triangle inequality.
\end{hint}
\begin{solution}
At most \(10^{-4}\): \(|\|x\|-\|y\||\le10^{-4}\).
\end{solution}

\begin{exercise}[Condition estimate]\label{exr:ii02-expand-14}
If \(\|T\|=7\) and \(\|h\|\le0.01\), bound \(\|Th\|\).
\end{exercise}
\begin{hint}
Use the operator norm definition.
\end{hint}
\begin{solution}
\(\|Th\|\le0.07\).
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[One-dimensional norms]\label{exr:ii02-expand-15}
Prove every norm on \(\mathbb R\) has the form \(c|x|\) with \(c>0\).
\end{exercise}
\begin{hint}
Use homogeneity and evaluate the norm of \(1\).
\end{hint}
\begin{solution}
\(\|x\|=|x|\|\operatorname{sgn}(x)\|=|x|\|1\|\); set \(c=\|1\|>0\).
\end{solution}

\begin{exercise}[Sharp norm comparison]\label{exr:ii02-expand-16}
Show \(\|x\|_1\le\sqrt n\|x\|_2\) and identify an equality case.
\end{exercise}
\begin{hint}
Apply Cauchy--Schwarz to \((|x_i|)\) and \((1,\dots,1)\).
\end{hint}
\begin{solution}
The inequality follows; equality occurs when all \(|x_i|\) are equal, for example \(x=(1,\dots,1)\).
\end{solution}
