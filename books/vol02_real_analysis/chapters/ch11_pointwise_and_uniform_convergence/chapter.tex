\chapter{Pointwise and Uniform Convergence}
\label{ch:ii11}

Pointwise convergence controls each input separately; uniform convergence controls all inputs at one common scale. This distinction determines which global properties survive passage to limits.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item distinguish pointwise from uniform convergence using definitions and explicit examples;
\item prove uniform convergence by estimating \(\sup_x|f_n(x)-f(x)|\);
\item use uniform convergence to pass continuity from \(f_n\) to the limit;
\item construct examples where pointwise convergence does not preserve continuity or bounded error control;
\item apply the uniform Cauchy criterion to sequences of functions;
\item determine whether convergence is uniform on a whole domain or only on restricted subsets;
\end{itemize}

\section{Pointwise convergence}
For each fixed $x$, the numerical sequence $f_n(x)$ converges.

\section{Uniform convergence}
One index $N$ works for every $x$ simultaneously.

\section{Supremum norm}
Uniform convergence is exactly convergence in the sup norm when that norm is finite.

\section{Continuity under uniform limits}
Uniform limits of continuous functions are continuous.

\section{Cauchy criterion}
Uniform convergence is equivalent to a uniform Cauchy condition in complete codomains.

\section{Examples of failure}
Pointwise limits can lose continuity or exchange of limits.

\section{Uniform approximation}
Uniform error bounds quantify global approximation quality.

\section{Compact-domain tests}
Monotonicity and compactness can sometimes upgrade pointwise convergence to uniform convergence.

\section{Core structural results}

\begin{theorem}[Uniform limit theorem]\label{thm:ii11-01}
A uniform limit of continuous functions is continuous.
\begin{proof}
Choose one approximant uniformly close to the limit and use its continuity to bridge values of the limit at nearby points.
\end{proof}
\end{theorem}

\begin{theorem}[Uniform Cauchy criterion]\label{thm:ii11-02}
For complete codomain $Y$, a sequence $f_n:X\to Y$ converges uniformly iff for every $\varepsilon$ the tail is uniformly epsilon-Cauchy.
\begin{proof}
Uniform convergence implies uniform Cauchy by the triangle inequality. Conversely each pointwise tail is Cauchy and converges; the same uniform tail estimate survives in the limit.
\end{proof}
\end{theorem}

\begin{theorem}[Dini-type principle]\label{thm:ii11-03}
If continuous $f_n$ on compact $K$ decrease pointwise to continuous $f$, then convergence is uniform.
\begin{proof}
The open sets where $f_n-f<\varepsilon$ increase and cover compact $K$; a finite subcover reduces to one sufficiently large index.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Powers on $\lbrack{}0,1\rbrack{}$]\label{ex:ii11-worked-01}
Analyze $f_n(x)=x^n$ pointwise and uniformly. Pointwise the limit is $0$ on $[0,1)$ and $1$ at $1$, discontinuous. Uniform convergence fails because $\sup|f_n-f|=1$.
\end{example}

\begin{example}[Powers on $\lbrack{}0,a\rbrack{}$]\label{ex:ii11-worked-02}
For $0<a<1$, show $x^n\to0$ uniformly. $\sup_{[0,a]}x^n=a^n\to0$.
\end{example}

\begin{example}[Uniform implies pointwise]\label{ex:ii11-worked-03}
Prove the implication directly. A uniform epsilon index works in particular at any fixed point.
\end{example}

\begin{example}[Continuity of the uniform limit]\label{ex:ii11-worked-04}
Give the three-term epsilon estimate. $|f(x)-f(a)|\le|f-f_N|(x)+|f_N(x)-f_N(a)|+|f_N-f|(a)$, and choose the three terms below thirds of epsilon.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[Powers on $\lbrack{}0,1\rbrack{}$]\label{prob:ii11-dossier-01}
Analyze $f_n(x)=x^n$ pointwise and uniformly.
\end{problem}
\begin{solution}
Pointwise the limit is $0$ on $[0,1)$ and $1$ at $1$, discontinuous. Uniform convergence fails because $\sup|f_n-f|=1$.
\end{solution}

\begin{problem}[Powers on $\lbrack{}0,a\rbrack{}$]\label{prob:ii11-dossier-02}
For $0<a<1$, show $x^n\to0$ uniformly.
\end{problem}
\begin{solution}
$\sup_{[0,a]}x^n=a^n\to0$.
\end{solution}

\begin{problem}[Uniform implies pointwise]\label{prob:ii11-dossier-03}
Prove the implication directly.
\end{problem}
\begin{solution}
A uniform epsilon index works in particular at any fixed point.
\end{solution}

\begin{problem}[Continuity of the uniform limit]\label{prob:ii11-dossier-04}
Give the three-term epsilon estimate.
\end{problem}
\begin{solution}
$|f(x)-f(a)|\le|f-f_N|(x)+|f_N(x)-f_N(a)|+|f_N-f|(a)$, and choose the three terms below thirds of epsilon.
\end{solution}

\begin{problem}[Sup-norm reformulation]\label{prob:ii11-dossier-05}
Show $f_n\to f$ uniformly iff $\|f_n-f\|_\infty\to0$.
\end{problem}
\begin{solution}
This is just the definition of supremum norm and uniform epsilon control.
\end{solution}

\begin{problem}[Nonuniform pointwise convergence]\label{prob:ii11-dossier-06}
Analyze $f_n(x)=x/(1+nx)$ on $[0,1]$.
\end{problem}
\begin{solution}
For each fixed $x$, the limit is $0$. The maximum occurs near the endpoint and is $1/(n+1)$, so in fact convergence is uniform; this example illustrates the need to compute the supremum rather than guess.
\end{solution}

\begin{problem}[A genuine nonuniform sequence]\label{prob:ii11-dossier-07}
Analyze $f_n(x)=nx(1-x^2)^n$ on $[0,1]$.
\end{problem}
\begin{solution}
For fixed $x>0$ the exponential factor wins and the limit is zero, also at $0$. But maxima remain of order $\sqrt n$, so convergence is not uniform.
\end{solution}

\begin{problem}[Uniform Cauchy estimate]\label{prob:ii11-dossier-08}
If $\|f_{n+1}-f_n\|_\infty\le2^{-n}$, prove uniform convergence in $C(K)$.
\end{problem}
\begin{solution}
The tail is bounded by a geometric series, hence uniformly Cauchy; completeness of the sup-norm space gives uniform convergence.
\end{solution}

\begin{problem}[Uniform approximation preserves boundedness]\label{prob:ii11-dossier-09}
If $f_n$ are uniformly close to bounded $f_N$, show the limit is bounded.
\end{problem}
\begin{solution}
For one large $N$, $|f|\le|f-f_N|+|f_N|$, and the uniform error is globally bounded.
\end{solution}

\begin{problem}[Interchanging an outer limit]\label{prob:ii11-dossier-10}
If $x_m\to x$ and $f_n\to f$ uniformly with continuous $f_n$, explain why $f(x_m)\to f(x)$.
\end{problem}
\begin{solution}
Uniform convergence gives continuity of $f$, then ordinary continuity handles the sequence $x_m$.
\end{solution}

\begin{problem}[Dini mechanism]\label{prob:ii11-dossier-11}
Why is compactness decisive in Dini's theorem?
\end{problem}
\begin{solution}
Pointwise convergence gives one local index per point; compactness extracts finitely many local neighborhoods and hence one maximum index working globally.
\end{solution}

\begin{problem}[Convergence synthesis]\label{prob:ii11-dossier-12}
Why is uniform convergence the natural topology for preserving continuity?
\end{problem}
\begin{solution}
Continuity is a local statement needing one approximation to work at two nearby points simultaneously. Uniform error control supplies that common approximant everywhere.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii11-01}
Does uniform convergence imply pointwise?
\end{exercise}
\begin{hint}
Use definitions. Pointwise convergence fixes \(x\) first; uniform convergence requires one index that works simultaneously for every \(x\).
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii11-02}
Does pointwise imply uniform?
\end{exercise}
\begin{hint}
Use $x^n$ on $[0,1]$. Compute or estimate \(\sup_x|f_n(x)-f(x)|\); convergence of this supremum to zero is the direct uniform test.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii11-03}
If $\|f_n-f\|_\infty\le1/n$, what follows?
\end{exercise}
\begin{hint}
Take the bound to zero. To disprove uniform convergence, choose points \(x_n\) depending on \(n\) where the error stays bounded away from zero.
\end{hint}
\begin{solution}
Uniform convergence.
\end{solution}

\begin{exercise}\label{exr:ii11-04}
Can a uniform limit of continuous functions be discontinuous?
\end{exercise}
\begin{hint}
Use the uniform limit theorem. Uniform limits of continuous functions are continuous; if the proposed limit is discontinuous, uniform convergence is impossible.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii11-05}
Find the pointwise limit of $x/n$.
\end{exercise}
\begin{hint}
Fix $x$. Use the uniform Cauchy criterion when the candidate limit is difficult to identify explicitly.
\end{hint}
\begin{solution}
It is $0$.
\end{solution}

\begin{exercise}\label{exr:ii11-06}
Is $x/n\to0$ uniform on $[0,1]$?
\end{exercise}
\begin{hint}
Take the supremum. Restricting the domain can turn nonuniform convergence into uniform convergence; inspect where the bad points accumulate.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii11-07}
Is $x/n\to0$ uniform on $\mathbb R$?
\end{exercise}
\begin{hint}
Supremum is infinite. For monotone sequences of functions, do not infer uniform convergence without an additional theorem such as Dini under its hypotheses.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii11-08}
State the uniform Cauchy condition.
\end{exercise}
\begin{hint}
One index for all points and tail pairs. Keep endpoint behavior separate from interior behavior when the convergence deteriorates near a boundary.
\end{hint}
\begin{solution}
For every epsilon, $d(f_m(x),f_n(x))<\varepsilon$ for all $x$ and all large $m,n$.
\end{solution}
