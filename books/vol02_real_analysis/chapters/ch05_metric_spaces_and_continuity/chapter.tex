\chapter{Metric Spaces and Continuity}
\label{ch:ii05}

Metric spaces isolate the role of distance in analysis. Continuity becomes preservation of local closeness, with equivalent epsilon-delta, sequential, and open-set formulations.

\section*{Learning goals}
The reader should be able to use the chapter's definitions in proofs, recognize the decisive hypotheses, and move between concrete metric examples and invariant topological statements.

\section{Metrics}
A metric is nonnegative, definite, symmetric, and satisfies the triangle inequality.

\section{Metric examples}
Norm metrics, discrete metrics, and bounded transforms of metrics show how one set can carry different geometries.

\section{Metric convergence}
A sequence converges exactly when its distance to the limit tends to zero.

\section{Continuity at a point}
Output tolerances are controlled by sufficiently small input tolerances.

\section{Sequential continuity}
In metric spaces continuity is equivalent to preserving sequential limits.

\section{Open-set continuity}
Continuity is equivalent to inverse images of open sets being open.

\section{Uniform continuity}
One input scale must work simultaneously at every point.

\section{Lipschitz maps}
A Lipschitz estimate gives quantitative control and automatically implies uniform continuity.

\section{Core structural results}

\begin{theorem}[Sequential criterion]\label{thm:ii05-01}
A map between metric spaces is continuous iff it preserves limits of sequences.
\begin{proof}
Continuity transfers epsilon estimates to a convergent sequence. Failure of continuity creates points within $1/n$ whose images stay a fixed distance away.
\end{proof}
\end{theorem}

\begin{theorem}[Open-set criterion]\label{thm:ii05-02}
A map is continuous iff the inverse image of every open set is open.
\begin{proof}
At a point in a preimage, continuity pulls back a small target ball; conversely apply openness to balls around the image point.
\end{proof}
\end{theorem}

\begin{theorem}[Lipschitz implies uniform continuity]\label{thm:ii05-03}
If $d(f(x),f(y))\le Ld(x,y)$, then $f$ is uniformly continuous.
\begin{proof}
For $L>0$ choose $\delta=\varepsilon/L$; for $L=0$ the map is constant.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Discrete metric topology]\label{ex:ii05-worked-01}
Show every subset is open in the discrete metric. For any point $x$, the ball $B(x,1/2)$ is the singleton $\{x\}$.
\end{example}

\begin{example}[Norm gives a metric]\label{ex:ii05-worked-02}
Prove $d(x,y)=\|x-y\|$ is a metric. Definiteness and symmetry come from norm axioms, and the metric triangle inequality is the norm triangle inequality.
\end{example}

\begin{example}[Distance to a fixed point is Lipschitz]\label{ex:ii05-worked-03}
Show $x\mapsto d(x,a)$ is $1$-Lipschitz. The triangle inequality gives $|d(x,a)-d(y,a)|\le d(x,y)$.
\end{example}

\begin{example}[A sequential discontinuity witness]\label{ex:ii05-worked-04}
Define $f(0)=0$ and $f(x)=1/x$ otherwise. Show discontinuity at zero. Take $x_n=1/n\to0$; then $f(x_n)=n$ does not converge to $0$.
\end{example}

\begin{problem}[Discrete metric topology]\label{prob:ii05-dossier-01}
Show every subset is open in the discrete metric.
\end{problem}
\begin{solution}
For any point $x$, the ball $B(x,1/2)$ is the singleton $\{x\}$.
\end{solution}

\begin{problem}[Norm gives a metric]\label{prob:ii05-dossier-02}
Prove $d(x,y)=\|x-y\|$ is a metric.
\end{problem}
\begin{solution}
Definiteness and symmetry come from norm axioms, and the metric triangle inequality is the norm triangle inequality.
\end{solution}

\begin{problem}[Distance to a fixed point is Lipschitz]\label{prob:ii05-dossier-03}
Show $x\mapsto d(x,a)$ is $1$-Lipschitz.
\end{problem}
\begin{solution}
The triangle inequality gives $|d(x,a)-d(y,a)|\le d(x,y)$.
\end{solution}

\begin{problem}[A sequential discontinuity witness]\label{prob:ii05-dossier-04}
Define $f(0)=0$ and $f(x)=1/x$ otherwise. Show discontinuity at zero.
\end{problem}
\begin{solution}
Take $x_n=1/n\to0$; then $f(x_n)=n$ does not converge to $0$.
\end{solution}

\begin{problem}[Square is not uniformly continuous on $\mathbb R$]\label{prob:ii05-dossier-05}
Produce nearby points with separated squares.
\end{problem}
\begin{solution}
Take $x_n=n$, $y_n=n+1/n$; input distance tends to zero while square difference tends to $2$.
\end{solution}

\begin{problem}[Square on a bounded interval]\label{prob:ii05-dossier-06}
Show $x^2$ is uniformly continuous on $[-M,M]$.
\end{problem}
\begin{solution}
$|x^2-y^2|\le2M|x-y|$, so it is Lipschitz there.
\end{solution}

\begin{problem}[Composition]\label{prob:ii05-dossier-07}
Prove the composition of continuous maps is continuous.
\end{problem}
\begin{solution}
Inverse images compose: $(g\circ f)^{-1}(U)=f^{-1}(g^{-1}(U))$.
\end{solution}

\begin{problem}[A bounded equivalent metric]\label{prob:ii05-dossier-08}
Show $\rho=d/(1+d)$ has the same convergent sequences as $d$.
\end{problem}
\begin{solution}
$\rho\le d$, while for small $\rho$, $d=\rho/(1-\rho)\le2\rho$.
\end{solution}

\begin{problem}[Closed-set criterion]\label{prob:ii05-dossier-09}
Show continuous maps pull closed sets back to closed sets.
\end{problem}
\begin{solution}
Take complements and use the open-set criterion.
\end{solution}

\begin{problem}[Uniform continuity preserves Cauchy sequences]\label{prob:ii05-dossier-10}
Prove the claim.
\end{problem}
\begin{solution}
A uniform delta converts a Cauchy input tail into an epsilon-close image tail.
\end{solution}

\begin{problem}[Lipschitz diameter control]\label{prob:ii05-dossier-11}
Show $\operatorname{diam}f(A)\le L\operatorname{diam}A$.
\end{problem}
\begin{solution}
Apply the Lipschitz inequality to every pair and take the supremum.
\end{solution}

\begin{problem}[Continuity synthesis]\label{prob:ii05-dossier-12}
Why do inverse images appear in the topological definition?
\end{problem}
\begin{solution}
Inverse images preserve arbitrary unions and finite intersections exactly, matching the axioms for open sets.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii05-01}
Verify the triangle inequality for the discrete metric.
\end{exercise}
\begin{hint}
Consider whether endpoints agree.
\end{hint}
\begin{solution}
If endpoints differ, at least one leg has length $1$.
\end{solution}

\begin{exercise}\label{exr:ii05-02}
Is $f(x)=3x$ Lipschitz?
\end{exercise}
\begin{hint}
Compute differences.
\end{hint}
\begin{solution}
Yes, with constant $3$.
\end{solution}

\begin{exercise}\label{exr:ii05-03}
Show a constant map is uniformly continuous.
\end{exercise}
\begin{hint}
Output distance is zero.
\end{hint}
\begin{solution}
Any positive delta works.
\end{solution}

\begin{exercise}\label{exr:ii05-04}
Are sums of continuous real functions continuous?
\end{exercise}
\begin{hint}
Use limits or epsilon bounds.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii05-05}
Show $|x|$ is continuous.
\end{exercise}
\begin{hint}
Use reverse triangle inequality.
\end{hint}
\begin{solution}
It is $1$-Lipschitz.
\end{solution}

\begin{exercise}\label{exr:ii05-06}
Characterize convergence in the discrete metric.
\end{exercise}
\begin{hint}
Use epsilon $1/2$.
\end{hint}
\begin{solution}
A convergent sequence is eventually constant.
\end{solution}

\begin{exercise}\label{exr:ii05-07}
Does uniform continuity imply continuity?
\end{exercise}
\begin{hint}
Fix a point.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii05-08}
Give a continuous but not uniformly continuous map on $\mathbb R$.
\end{exercise}
\begin{hint}
Use quadratic growth.
\end{hint}
\begin{solution}
$x^2$ is a standard example.
\end{solution}
