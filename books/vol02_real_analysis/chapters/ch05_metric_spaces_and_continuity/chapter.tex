\chapter{Metric Spaces and Continuity}
\label{ch:ii05}

Metric spaces isolate the role of distance in analysis. Continuity becomes preservation of local closeness, with equivalent epsilon-delta, sequential, and open-set formulations.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item verify continuity at a point using \(\varepsilon\)-\(\delta\), sequential, and inverse-image criteria;
\item prove continuity of compositions and algebraic combinations from structural properties rather than coordinate expansion alone;
\item compute Lipschitz constants for explicit maps and use them to obtain continuity estimates;
\item distinguish continuity from uniform continuity by constructing or analyzing counterexamples;
\item identify which topological properties are preserved under continuous maps in the situations treated in the chapter;
\item translate metric estimates into neighborhood statements and back again;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Metrics}
A metric is nonnegative, definite, symmetric, and satisfies the triangle inequality.

\section{Metric examples}
Norm metrics, discrete metrics, and bounded transforms of metrics show how one set can carry different geometries.

\section{Metric convergence}
A sequence converges exactly when its distance to the limit tends to zero.

\section{Continuity at a point}
Output tolerances are controlled by sufficiently small input tolerances.


% BEGIN VOL02-EXPANSION II05-example-01
\begin{example}[Epsilon--delta continuity with an explicit local bound]\label{ex:ii05-expand-01}
For \(f(x)=x^2\), prove continuity at \(a\). If \(|x-a|<1\), then
\[
|x+a|\le |x-a|+2|a|<1+2|a|.
\]
Hence
\[
|x^2-a^2|=|x-a||x+a|
<(1+2|a|)|x-a|.
\]
Given \(\varepsilon>0\), choose
\[
\delta=\min\left(1,\frac{\varepsilon}{1+2|a|}\right).
\]
The preliminary bound on \(|x+a|\) is what makes the epsilon choice valid.
\end{example}
% END VOL02-EXPANSION II05-example-01
\section{Sequential continuity}
In metric spaces continuity is equivalent to preserving sequential limits.

\section{Open-set continuity}
Continuity is equivalent to inverse images of open sets being open.

\section{Uniform continuity}
One input scale must work simultaneously at every point.


% BEGIN VOL02-EXPANSION II05-example-02
\begin{example}[Continuity versus uniform continuity]\label{ex:ii05-expand-02}
The function \(f(x)=x^2\) is continuous on \(\mathbb R\) but not uniformly
continuous there. Take
\[
x_n=n,\qquad y_n=n+\frac1n.
\]
Then \(|x_n-y_n|=1/n\to0\), while
\[
|f(y_n)-f(x_n)|
=2+\frac1{n^2}\to2.
\]
Thus arbitrarily close inputs can still have outputs separated by a fixed
amount when the location drifts to infinity.
\end{example}
% END VOL02-EXPANSION II05-example-02
\section{Lipschitz maps}
A Lipschitz estimate gives quantitative control and automatically implies uniform continuity.


% BEGIN VOL02-EXPANSION II05-example-03
\begin{example}[A Lipschitz estimate from the reverse triangle inequality]\label{ex:ii05-expand-03}
Fix a nonempty set \(A\) in a metric space and define
\[
d_A(x)=\inf_{a\in A}d(x,a).
\]
For every \(a\in A\),
\[
d(x,a)\le d(x,y)+d(y,a).
\]
Taking infima gives \(d_A(x)\le d(x,y)+d_A(y)\). Swapping \(x,y\) yields
\[
|d_A(x)-d_A(y)|\le d(x,y).
\]
Therefore distance to a set is \(1\)-Lipschitz and hence uniformly continuous.
\end{example}
% END VOL02-EXPANSION II05-example-03
\section{Core structural results}

\begin{theorem}[Sequential criterion]\label{thm:ii05-01}
A map between metric spaces is continuous iff it preserves limits of sequences.
\begin{proof}
Continuity transfers epsilon estimates to a convergent sequence. Failure of continuity creates points within $1/n$ whose images stay a fixed distance away.
\end{proof}
\end{theorem}

\begin{theorem}[Open-set criterion]\label{thm:ii05-02}
A map is continuous iff the inverse image of every open set is open.
\begin{proof}
At a point in a preimage, continuity pulls back a small target ball; conversely apply openness to balls around the image point.
\end{proof}
\end{theorem}

\begin{theorem}[Lipschitz implies uniform continuity]\label{thm:ii05-03}
If $d(f(x),f(y))\le Ld(x,y)$, then $f$ is uniformly continuous.
\begin{proof}
For $L>0$ choose $\delta=\varepsilon/L$; for $L=0$ the map is constant.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Discrete metric topology]\label{ex:ii05-worked-01}
Show every subset is open in the discrete metric. For any point $x$, the ball $B(x,1/2)$ is the singleton $\{x\}$.
\end{example}

\begin{example}[Norm gives a metric]\label{ex:ii05-worked-02}
Prove $d(x,y)=\|x-y\|$ is a metric. Definiteness and symmetry come from norm axioms, and the metric triangle inequality is the norm triangle inequality.
\end{example}

\begin{example}[Distance to a fixed point is Lipschitz]\label{ex:ii05-worked-03}
Show $x\mapsto d(x,a)$ is $1$-Lipschitz. The triangle inequality gives $|d(x,a)-d(y,a)|\le d(x,y)$.
\end{example}

\begin{example}[A sequential discontinuity witness]\label{ex:ii05-worked-04}
Define $f(0)=0$ and $f(x)=1/x$ otherwise. Show discontinuity at zero. Take $x_n=1/n\to0$; then $f(x_n)=n$ does not converge to $0$.
\end{example}

\section{Solved dossiers}
Each dossier is a substantial solved problem. Corpus mappings guide topic choice where explicit retained source blocks exist; otherwise the dossier is a fresh canonical completion.

\begin{problem}[Discrete metric topology]\label{prob:ii05-dossier-01}
Show every subset is open in the discrete metric.
\end{problem}
\begin{solution}
For any point $x$, the ball $B(x,1/2)$ is the singleton $\{x\}$.
\end{solution}

\begin{problem}[Norm gives a metric]\label{prob:ii05-dossier-02}
Prove $d(x,y)=\|x-y\|$ is a metric.
\end{problem}
\begin{solution}
Definiteness and symmetry come from norm axioms, and the metric triangle inequality is the norm triangle inequality.
\end{solution}

\begin{problem}[Distance to a fixed point is Lipschitz]\label{prob:ii05-dossier-03}
Show $x\mapsto d(x,a)$ is $1$-Lipschitz.
\end{problem}
\begin{solution}
The triangle inequality gives $|d(x,a)-d(y,a)|\le d(x,y)$.
\end{solution}

\begin{problem}[A sequential discontinuity witness]\label{prob:ii05-dossier-04}
Define $f(0)=0$ and $f(x)=1/x$ otherwise. Show discontinuity at zero.
\end{problem}
\begin{solution}
Take $x_n=1/n\to0$; then $f(x_n)=n$ does not converge to $0$.
\end{solution}

\begin{problem}[Square is not uniformly continuous on $\mathbb R$]\label{prob:ii05-dossier-05}
Produce nearby points with separated squares.
\end{problem}
\begin{solution}
Take $x_n=n$, $y_n=n+1/n$; input distance tends to zero while square difference tends to $2$.
\end{solution}

\begin{problem}[Square on a bounded interval]\label{prob:ii05-dossier-06}
Show $x^2$ is uniformly continuous on $[-M,M]$.
\end{problem}
\begin{solution}
$|x^2-y^2|\le2M|x-y|$, so it is Lipschitz there.
\end{solution}

\begin{problem}[Composition]\label{prob:ii05-dossier-07}
Prove the composition of continuous maps is continuous.
\end{problem}
\begin{solution}
Inverse images compose: $(g\circ f)^{-1}(U)=f^{-1}(g^{-1}(U))$.
\end{solution}

\begin{problem}[A bounded equivalent metric]\label{prob:ii05-dossier-08}
Show $\rho=d/(1+d)$ has the same convergent sequences as $d$.
\end{problem}
\begin{solution}
$\rho\le d$, while for small $\rho$, $d=\rho/(1-\rho)\le2\rho$.
\end{solution}

\begin{problem}[Closed-set criterion]\label{prob:ii05-dossier-09}
Show continuous maps pull closed sets back to closed sets.
\end{problem}
\begin{solution}
Take complements and use the open-set criterion.
\end{solution}

\begin{problem}[Uniform continuity preserves Cauchy sequences]\label{prob:ii05-dossier-10}
Prove the claim.
\end{problem}
\begin{solution}
A uniform delta converts a Cauchy input tail into an epsilon-close image tail.
\end{solution}

\begin{problem}[Lipschitz diameter control]\label{prob:ii05-dossier-11}
Show $\operatorname{diam}f(A)\le L\operatorname{diam}A$.
\end{problem}
\begin{solution}
Apply the Lipschitz inequality to every pair and take the supremum.
\end{solution}

\begin{problem}[Continuity synthesis]\label{prob:ii05-dossier-12}
Why do inverse images appear in the topological definition?
\end{problem}
\begin{solution}
Inverse images preserve arbitrary unions and finite intersections exactly, matching the axioms for open sets.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii05-01}
Verify the triangle inequality for the discrete metric.
\end{exercise}
\begin{hint}
Consider whether endpoints agree. For \(\varepsilon\)-\(\delta\) continuity, isolate \(d(f(x),f(a))\) and bound it by a constant times \(d(x,a)\) when possible.
\end{hint}
\begin{solution}
If endpoints differ, at least one leg has length $1$.
\end{solution}

\begin{exercise}\label{exr:ii05-02}
Is $f(x)=3x$ Lipschitz?
\end{exercise}
\begin{hint}
Compute differences. For sequential continuity, start with an arbitrary sequence \(x_n\to a\) and push the convergence through the map.
\end{hint}
\begin{solution}
Yes, with constant $3$.
\end{solution}

\begin{exercise}\label{exr:ii05-03}
Show a constant map is uniformly continuous.
\end{exercise}
\begin{hint}
Output distance is zero. To prove uniform continuity, the chosen \(\delta\) must depend only on \(\varepsilon\), not on the base point.
\end{hint}
\begin{solution}
Any positive delta works.
\end{solution}

\begin{exercise}\label{exr:ii05-04}
Are sums of continuous real functions continuous?
\end{exercise}
\begin{hint}
Use limits or epsilon bounds. A Lipschitz estimate \(d(f(x),f(y))\le Ld(x,y)\) immediately gives a uniform-continuity modulus.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii05-05}
Show $|x|$ is continuous.
\end{exercise}
\begin{hint}
Use reverse triangle inequality. For composition, feed the outer map's \(\varepsilon\)-requirement into the inner map's continuity estimate.
\end{hint}
\begin{solution}
It is $1$-Lipschitz.
\end{solution}

\begin{exercise}\label{exr:ii05-06}
Characterize convergence in the discrete metric.
\end{exercise}
\begin{hint}
Use epsilon $1/2$. To disprove uniform continuity, construct pairs whose inputs become arbitrarily close while their outputs stay separated.
\end{hint}
\begin{solution}
A convergent sequence is eventually constant.
\end{solution}

\begin{exercise}\label{exr:ii05-07}
Does uniform continuity imply continuity?
\end{exercise}
\begin{hint}
Fix a point. Use inverse images of open sets when the topology is easier to understand than direct metric estimates.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii05-08}
Give a continuous but not uniformly continuous map on $\mathbb R$.
\end{exercise}
\begin{hint}
Use quadratic growth. On compact domains, combine continuity with compactness when a global estimate is needed.
\end{hint}
\begin{solution}
$x^2$ is a standard example.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Delta for a linear map]\label{exr:ii05-expand-01}
For \(f(x)=5x\), give \(\delta\) in terms of \(\varepsilon\).
\end{exercise}
\begin{hint}
Use \(|f(x)-f(a)|=5|x-a|\).
\end{hint}
\begin{solution}
Take \(\delta=\varepsilon/5\).
\end{solution}

\begin{exercise}[Lipschitz constant]\label{exr:ii05-expand-02}
Find a Lipschitz constant for \(f(x)=3x-7\) on \(\mathbb R\).
\end{exercise}
\begin{hint}
Subtract function values.
\end{hint}
\begin{solution}
\(|f(x)-f(y)|=3|x-y|\), so \(L=3\).
\end{solution}

\begin{exercise}[Discrete continuity]\label{exr:ii05-expand-03}
Why is every map from a discrete metric space continuous?
\end{exercise}
\begin{hint}
Use the singleton ball \(B(x,1/2)\).
\end{hint}
\begin{solution}
Every point has an open singleton neighborhood, so every inverse image of an open set is open.
\end{solution}

\begin{exercise}[Sequential test]\label{exr:ii05-expand-04}
Show \(f(x)=|x|\) is continuous using a metric estimate.
\end{exercise}
\begin{hint}
Use reverse triangle inequality.
\end{hint}
\begin{solution}
\(|\,|x|-|y|\,|\le|x-y|\), so \(f\) is \(1\)-Lipschitz.
\end{solution}

\begin{exercise}[Composition constant]\label{exr:ii05-expand-05}
If \(f\) is \(L\)-Lipschitz and \(g\) is \(M\)-Lipschitz, what constant works for \(g\circ f\)?
\end{exercise}
\begin{hint}
Apply the two estimates in succession.
\end{hint}
\begin{solution}
\(ML\) works.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Composition continuity]\label{exr:ii05-expand-06}
Prove a composition of continuous maps is continuous.
\end{exercise}
\begin{hint}
Use inverse images or nested epsilon choices.
\end{hint}
\begin{solution}
\((g\circ f)^{-1}(U)=f^{-1}(g^{-1}(U))\), and both inverse-image operations preserve openness.
\end{solution}

\begin{exercise}[Sequential criterion forward]\label{exr:ii05-expand-07}
Prove continuous maps preserve sequence limits.
\end{exercise}
\begin{hint}
Use the epsilon--delta condition after the sequence enters the delta-ball.
\end{hint}
\begin{solution}
For any epsilon choose delta from continuity; convergence provides \(N\) with inputs delta-close thereafter.
\end{solution}

\begin{exercise}[Lipschitz uniform continuity]\label{exr:ii05-expand-08}
Prove every Lipschitz map is uniformly continuous.
\end{exercise}
\begin{hint}
Choose one delta independent of the base point.
\end{hint}
\begin{solution}
For \(L>0\), take \(\delta=\varepsilon/L\); constant maps are immediate.
\end{solution}

\begin{exercise}[Uniformly continuous sends Cauchy to Cauchy]\label{exr:ii05-expand-09}
Prove a uniformly continuous map sends Cauchy sequences to Cauchy sequences.
\end{exercise}
\begin{hint}
Use one delta for all late pairs.
\end{hint}
\begin{solution}
Choose delta from uniform continuity and then a Cauchy tail whose pairwise distances are below delta.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Continuous not uniform]\label{exr:ii05-expand-10}
Show \(x^2\) is not uniformly continuous on \(\mathbb R\).
\end{exercise}
\begin{hint}
Use pairs \(n,n+1/n\).
\end{hint}
\begin{solution}
Input distances tend to zero while output differences tend to \(2\).
\end{solution}

\begin{exercise}[Pointwise delta dependence]\label{exr:ii05-expand-11}
Why does a proof with \(\delta=\varepsilon/(1+2|a|)\) establish continuity but not uniform continuity?
\end{exercise}
\begin{hint}
Observe dependence on the base point \(a\).
\end{hint}
\begin{solution}
Uniform continuity requires one delta working for all \(a\); this choice varies with \(a\).
\end{solution}

\begin{exercise}[Continuous inverse not automatic]\label{exr:ii05-expand-12}
Does a continuous bijection always have a continuous inverse?
\end{exercise}
\begin{hint}
Recall that extra hypotheses such as compact-to-Hausdorff are needed.
\end{hint}
\begin{solution}
No in general; continuity of the inverse is a separate topological property.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Error propagation]\label{exr:ii05-expand-13}
If \(f\) is \(12\)-Lipschitz and the input error is at most \(10^{-3}\), bound output error.
\end{exercise}
\begin{hint}
Multiply by the Lipschitz constant.
\end{hint}
\begin{solution}
The output error is at most \(0.012\).
\end{solution}

\begin{exercise}[Nearest-set distance]\label{exr:ii05-expand-14}
Why is \(d_A(x)\) stable under perturbing \(x\)?
\end{exercise}
\begin{hint}
Use its \(1\)-Lipschitz property.
\end{hint}
\begin{solution}
Moving \(x\) by at most \(\eta\) changes its distance to \(A\) by at most \(\eta\).
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Uniform continuity on compact sets]\label{exr:ii05-expand-15}
Sketch the contradiction proof that continuous \(f:K\to Y\) on compact metric \(K\) is uniformly continuous.
\end{exercise}
\begin{hint}
Assume pairs \(x_n,y_n\) get arbitrarily close while images stay epsilon-separated; extract a convergent subsequence.
\end{hint}
\begin{solution}
Compactness gives \(x_{n_k}\to x\); then \(y_{n_k}\to x\) too. Continuity sends both image subsequences to \(f(x)\), contradiction.
\end{solution}

\begin{exercise}[Equivalent metric]\label{exr:ii05-expand-16}
If \(\rho(x,y)=d(x,y)/(1+d(x,y))\), explain why \(d\)- and \(\rho\)-convergence are equivalent.
\end{exercise}
\begin{hint}
Study the scalar transform \(t\mapsto t/(1+t)\) near zero and invert it.
\end{hint}
\begin{solution}
\(\rho\to0\) iff \(d=\rho/(1-\rho)\to0\); thus they have the same convergent sequences.
\end{solution}
