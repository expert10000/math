\chapter{Elementary Existence Theory for ODEs}
\label{ch:ii19}

Elementary ODE existence theory is most transparent when an initial-value problem is rewritten as an integral fixed-point equation. Continuity gives existence mechanisms; Lipschitz control supplies uniqueness.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item convert a first-order ODE into an equivalent integral equation;
\item verify local Lipschitz conditions in the dependent variable for concrete right-hand sides;
\item apply Picard iteration to obtain successive approximations;
\item derive local existence and uniqueness from contraction estimates on a suitable rectangle or ball;
\item identify blow-up, nonuniqueness, or loss-of-Lipschitz examples where the standard theorem does not apply;
\item estimate how the solution changes under perturbations of initial data or the right-hand side in elementary settings;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Initial-value problems}
An ODE with initial data asks for a curve satisfying a differential rule and a point condition.

\section{Integral formulation}
Integrating converts $y'=f(t,y)$ into a Volterra equation.

\section{Picard iteration}
Successive substitution generates approximate solutions.


% BEGIN VOL02-EXPANSION II19-example-01
\begin{example}[Picard iterates reveal the exponential series]\label{ex:ii19-expand-01}
For
\[
y'=y,\qquad y(0)=1,
\]
the integral equation is
\[
y(t)=1+\int_0^t y(s)\,ds.
\]
Starting from \(y_0=1\),
\[
y_1=1+t,\qquad
y_2=1+t+\frac{t^2}{2},\qquad
y_3=1+t+\frac{t^2}{2}+\frac{t^3}{6}.
\]
The Picard iterates are successive Taylor polynomials for \(e^t\). The fixed
point is therefore \(e^t\), and the iteration exhibits existence constructively.
\end{example}
% END VOL02-EXPANSION II19-example-01
\section{Local existence}
Continuity and compactness can yield existence; Lipschitz hypotheses enable Banach's theorem.

\section{Uniqueness}
Lipschitz dependence on the state variable prevents branching solutions.

\section{Continuation}
Solutions extend while they remain in regions where the hypotheses stay controlled.

\section{Continuous dependence}
Small changes in initial data or vector field produce controlled changes in solutions.


% BEGIN VOL02-EXPANSION II19-example-02
\begin{example}[Gronwall quantifies dependence on initial data]\label{ex:ii19-expand-02}
Suppose two solutions of \(y'=f(t,y)\) satisfy a common state Lipschitz bound
\(L\) and initial values \(y_0,z_0\). Subtracting the integral equations gives
\[
|y(t)-z(t)|
\le |y_0-z_0|+L\int_0^t|y(s)-z(s)|\,ds.
\]
Gronwall yields
\[
|y(t)-z(t)|\le |y_0-z_0|e^{Lt}.
\]
Thus uniqueness is the zero-initial-separation case, while continuous
dependence is the quantitative version for nonzero perturbations.
\end{example}
% END VOL02-EXPANSION II19-example-02
\section{Autonomous systems}
Phase-space geometry reframes $y'=F(y)$ as a flow problem.


% BEGIN VOL02-EXPANSION II19-example-03
\begin{example}[Phase-line geometry for an autonomous equation]\label{ex:ii19-expand-03}
For
\[
y'=y(1-y),
\]
equilibria occur at \(y=0,1\). On \(0<y<1\), the derivative is positive, so
solutions move upward; for \(y>1\), the derivative is negative, so solutions
move downward. Hence \(y=1\) is attracting from both nearby sides, while
\(y=0\) is unstable to positive perturbations.

The autonomous formulation converts the differential equation into geometry of
the vector field on phase space.
\end{example}
% END VOL02-EXPANSION II19-example-03
\section{Core structural results}

\begin{theorem}[Picard--Lindelöf]\label{thm:ii19-01}
If $f$ is continuous and locally Lipschitz in $y$ near $(t_0,y_0)$, then the IVP $y'=f(t,y)$, $y(t_0)=y_0$ has a unique local solution.
\begin{proof}
On a sufficiently small rectangle, the integral operator maps a closed sup-norm ball into itself and is a contraction; Banach gives a unique fixed point.
\end{proof}
\end{theorem}

\begin{theorem}[Gronwall inequality]\label{thm:ii19-02}
If $u(t)\le A+B\int_0^tu(s)ds$ with $u\ge0$, then $u(t)\le Ae^{Bt}$.
\begin{proof}
Define the right-hand side majorant and differentiate it; multiplying by $e^{-Bt}$ yields a monotone quantity that bounds $u$.
\end{proof}
\end{theorem}

\begin{theorem}[Continuous dependence]\label{thm:ii19-03}
Under a Lipschitz bound $L$, two solutions satisfy an exponential stability estimate controlled by initial separation.
\begin{proof}
Subtract the integral equations, estimate by the initial difference plus $L$ times the integral of the difference, then apply Gronwall.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Linear scalar IVP]\label{ex:ii19-worked-01}
Solve $y'=ay$, $y(0)=y_0$. The solution is $y(t)=y_0e^{at}$.
\end{example}

\begin{example}[Integral reformulation]\label{ex:ii19-worked-02}
Rewrite $y'=t+y$, $y(0)=1$. $y(t)=1+\int_0^t(s+y(s))ds$.
\end{example}

\begin{example}[Picard first iterate]\label{ex:ii19-worked-03}
Starting from $y_0(t)=1$, compute one Picard iterate for the previous equation. $y_1(t)=1+\int_0^t(s+1)ds=1+t+t^2/2$.
\end{example}

\begin{example}[Uniqueness failure without Lipschitz]\label{ex:ii19-worked-04}
Why can $y'=\sqrt{|y|}$, $y(0)=0$ have nonunique solutions? The square-root field is not Lipschitz at zero; solutions may remain zero for a while and then depart along a translated quadratic branch.
\end{example}

\section{Solved dossiers}
The dossiers are longer solved problems. Legacy mappings guide topic selection where explicit retained source blocks exist; otherwise fresh canonical problems complete the chapter.

\begin{problem}[Linear scalar IVP]\label{prob:ii19-dossier-01}
Solve $y'=ay$, $y(0)=y_0$.
\end{problem}
\begin{solution}
The solution is $y(t)=y_0e^{at}$.
\end{solution}

\begin{problem}[Integral reformulation]\label{prob:ii19-dossier-02}
Rewrite $y'=t+y$, $y(0)=1$.
\end{problem}
\begin{solution}
$y(t)=1+\int_0^t(s+y(s))ds$.
\end{solution}

\begin{problem}[Picard first iterate]\label{prob:ii19-dossier-03}
Starting from $y_0(t)=1$, compute one Picard iterate for the previous equation.
\end{problem}
\begin{solution}
$y_1(t)=1+\int_0^t(s+1)ds=1+t+t^2/2$.
\end{solution}

\begin{problem}[Uniqueness failure without Lipschitz]\label{prob:ii19-dossier-04}
Why can $y'=\sqrt{|y|}$, $y(0)=0$ have nonunique solutions?
\end{problem}
\begin{solution}
The square-root field is not Lipschitz at zero; solutions may remain zero for a while and then depart along a translated quadratic branch.
\end{solution}

\begin{problem}[Local contraction scale]\label{prob:ii19-dossier-05}
If $|f(t,y)-f(t,z)|\le L|y-z|$, what interval condition gives direct contraction?
\end{problem}
\begin{solution}
Choose interval half-width or length $h$ with $Lh<1$.
\end{solution}

\begin{problem}[Gronwall stability]\label{prob:ii19-dossier-06}
If two solutions have initial gap $\delta$, bound their later gap.
\end{problem}
\begin{solution}
Under Lipschitz constant $L$, the gap is at most $\delta e^{L|t-t_0|}$.
\end{solution}

\begin{problem}[Blow-up and continuation]\label{prob:ii19-dossier-07}
Why does $y'=y^2$, $y(0)=1$ not extend for all positive time?
\end{problem}
\begin{solution}
The solution $y=1/(1-t)$ blows up at $t=1$, leaving every bounded region where local continuation could proceed.
\end{solution}

\begin{problem}[Autonomous equilibrium]\label{prob:ii19-dossier-08}
What is an equilibrium of $y'=F(y)$?
\end{problem}
\begin{solution}
A point $y_*$ with $F(y_*)=0$, giving the constant solution $y(t)=y_*$.
\end{solution}

\begin{problem}[Separable example]\label{prob:ii19-dossier-09}
Solve $y'=-2y$, $y(0)=3$.
\end{problem}
\begin{solution}
$y(t)=3e^{-2t}$.
\end{solution}

\begin{problem}[Parameter dependence]\label{prob:ii19-dossier-10}
Why does a Lipschitz theorem imply robustness?
\end{problem}
\begin{solution}
The same Gronwall estimate controlling uniqueness also bounds changes caused by perturbed initial data or forcing.
\end{solution}

\begin{problem}[Picard convergence]\label{prob:ii19-dossier-11}
What makes successive approximations converge?
\end{problem}
\begin{solution}
The integral operator is a contraction on a sufficiently small complete function-space ball.
\end{solution}

\begin{problem}[ODE synthesis]\label{prob:ii19-dossier-12}
Why is integration the natural foundation for existence theory?
\end{problem}
\begin{solution}
Differential equations ask for unknown derivatives, whereas the integral equation defines a self-map on a space of continuous functions where completeness and norm estimates can be applied.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii19-01}
Solve $y'=y$, $y(0)=2$.
\end{exercise}
\begin{hint}
Exponential. Integrate the ODE from the initial time to rewrite it as \(y(t)=y_0+\int f(s,y(s))\,ds\).
\end{hint}
\begin{solution}
$2e^t$.
\end{solution}

\begin{exercise}\label{exr:ii19-02}
What theorem gives local uniqueness?
\end{exercise}
\begin{hint}
Picard--Lindelöf. Estimate the derivative of \(f\) with respect to \(y\) on the chosen rectangle to obtain a local Lipschitz constant.
\end{hint}
\begin{solution}
Picard--Lindelöf.
\end{solution}

\begin{exercise}\label{exr:ii19-03}
What hypothesis controls uniqueness?
\end{exercise}
\begin{hint}
Lipschitz in state. Choose the time interval short enough that the Picard operator maps the function ball into itself and becomes a contraction.
\end{hint}
\begin{solution}
Local Lipschitz continuity in $y$.
\end{solution}

\begin{exercise}\label{exr:ii19-04}
What does Gronwall control?
\end{exercise}
\begin{hint}
Growth of differences. Compute the first few Picard iterates directly from the integral equation rather than differentiating them afterward.
\end{hint}
\begin{solution}
Exponential stability bounds.
\end{solution}

\begin{exercise}\label{exr:ii19-05}
Can solutions blow up in finite time?
\end{exercise}
\begin{hint}
If uniqueness fails, look for a right-hand side that is continuous but not locally Lipschitz in \(y\).
\end{hint}
\begin{solution}
Yes, e.g. $y'=y^2$.
\end{solution}

\begin{exercise}\label{exr:ii19-06}
What is Picard iteration applied to?
\end{exercise}
\begin{hint}
Integral map. To detect finite-time blow-up, solve or compare with a scalar equation whose growth can be integrated explicitly.
\end{hint}
\begin{solution}
The Volterra integral formulation.
\end{solution}

\begin{exercise}\label{exr:ii19-07}
What is an equilibrium?
\end{exercise}
\begin{hint}
Zero of vector field. For stability, subtract the two integral equations and estimate the difference before applying a Grönwall-type bound if available.
\end{hint}
\begin{solution}
A point $F(y_*)=0$.
\end{solution}

\begin{exercise}\label{exr:ii19-08}
Does continuity alone always imply uniqueness?
\end{exercise}
\begin{hint}
Keep the existence interval local unless you have an a priori bound preventing escape from the rectangle.
\end{hint}
\begin{solution}
No.
\end{solution}

\subsection*{Standard computations}

\begin{exercise}[Integral form]\label{exr:ii19-expand-01}
Rewrite \(y'=t-y,\ y(0)=2\) as an integral equation.
\end{exercise}
\begin{hint}
Integrate from \(0\) to \(t\).
\end{hint}
\begin{solution}
\(y(t)=2+\int_0^t(s-y(s))ds\).
\end{solution}

\begin{exercise}[Picard iterate]\label{exr:ii19-expand-02}
Starting from \(y_0=2\), compute one Picard iterate for the previous IVP.
\end{exercise}
\begin{hint}
Insert the constant guess.
\end{hint}
\begin{solution}
\(y_1(t)=2+t^2/2-2t\).
\end{solution}

\begin{exercise}[Lipschitz check]\label{exr:ii19-expand-03}
Is \(f(t,y)=t+y^3\) locally Lipschitz in \(y\)?
\end{exercise}
\begin{hint}
Bound \(f_y=3y^2\) on a bounded rectangle.
\end{hint}
\begin{solution}
Yes, locally; on \(|y|\le M\) a Lipschitz constant is \(3M^2\).
\end{solution}

\begin{exercise}[Gronwall bound]\label{exr:ii19-expand-04}
If initial separation is \(0.01\) and \(L=2\), bound separation at \(t=1\).
\end{exercise}
\begin{hint}
Use \(e^{Lt}\).
\end{hint}
\begin{solution}
At most \(0.01e^2\).
\end{solution}

\begin{exercise}[Equilibria]\label{exr:ii19-expand-05}
Find equilibria of \(y'=y(y-2)\).
\end{exercise}
\begin{hint}
Set the right side to zero.
\end{hint}
\begin{solution}
\(y=0\) and \(y=2\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Picard--Lindelöf mechanism]\label{exr:ii19-expand-06}
Explain how local Lipschitz control yields a contraction on a short time interval.
\end{exercise}
\begin{hint}
Use the Volterra integral operator and require \(Lh<1\).
\end{hint}
\begin{solution}
The sup difference of two outputs is bounded by \(Lh\) times the input sup difference.
\end{solution}

\begin{exercise}[Uniqueness via Gronwall]\label{exr:ii19-expand-07}
Prove two solutions with the same initial value coincide under a Lipschitz bound.
\end{exercise}
\begin{hint}
Apply Gronwall to their difference.
\end{hint}
\begin{solution}
The initial separation is zero, so the exponential bound is identically zero.
\end{solution}

\begin{exercise}[Integral/differential equivalence]\label{exr:ii19-expand-08}
Prove a continuous \(f\) makes the integral formulation differentiable and equivalent to the ODE.
\end{exercise}
\begin{hint}
Use the fundamental theorem of calculus.
\end{hint}
\begin{solution}
Differentiating the integral gives \(f(t,y(t))\); integrating the ODE recovers the integral equation.
\end{solution}

\begin{exercise}[Continuation principle]\label{exr:ii19-expand-09}
Explain why a local solution can often be extended while it remains inside a compact region where \(f\) and its Lipschitz constants stay controlled.
\end{exercise}
\begin{hint}
Restart the local theorem at a later point.
\end{hint}
\begin{solution}
Uniform local bounds allow repeated continuation steps; obstruction occurs only when the solution leaves every controlled region or time reaches a boundary.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Nonuniqueness]\label{exr:ii19-expand-10}
Give an IVP with continuous right side but nonunique solutions.
\end{exercise}
\begin{hint}
Use \(y'=\sqrt{|y|}, y(0)=0\).
\end{hint}
\begin{solution}
The zero solution and delayed quadratic departures show nonuniqueness; the field is not Lipschitz at \(0\).
\end{solution}

\begin{exercise}[Finite-time blow-up]\label{exr:ii19-expand-11}
Show \(y'=y^2,\ y(0)=1\) cannot be continued for all positive time.
\end{exercise}
\begin{hint}
Solve explicitly.
\end{hint}
\begin{solution}
\(y(t)=1/(1-t)\), which blows up at \(t=1\).
\end{solution}

\begin{exercise}[Continuity alone insufficient for uniqueness]\label{exr:ii19-expand-12}
Why does Peano-type existence not imply uniqueness?
\end{exercise}
\begin{hint}
Continuity lacks a state-difference estimate.
\end{hint}
\begin{solution}
Without Lipschitz or related control, solution branches can separate from the same initial point.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Error propagation]\label{exr:ii19-expand-13}
Why is the Gronwall estimate useful in numerical ODE analysis?
\end{exercise}
\begin{hint}
It converts local perturbation estimates into global-in-time stability bounds.
\end{hint}
\begin{solution}
It shows how discretization or data errors can grow at most exponentially under a Lipschitz field.
\end{solution}

\begin{exercise}[Population model]\label{exr:ii19-expand-14}
Interpret equilibria and stability for the logistic equation \(y'=ry(1-y/K)\).
\end{exercise}
\begin{hint}
Check the sign of the vector field around \(0\) and \(K\).
\end{hint}
\begin{solution}
\(0\) and \(K\) are equilibria; for \(r>0\), \(K\) is attracting for positive states and \(0\) is unstable.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Comparison estimate]\label{exr:ii19-expand-15}
If \(|f(t,y)-g(t,y)|\le\eta\) and both are \(L\)-Lipschitz in \(y\), derive a bound between solutions with the same initial value.
\end{exercise}
\begin{hint}
Subtract integral equations and apply Gronwall with forcing \(\eta t\).
\end{hint}
\begin{solution}
A standard bound is \(\frac{\eta}{L}(e^{Lt}-1)\) for \(L>0\), and \(\eta t\) for \(L=0\).
\end{solution}

\begin{exercise}[Invariant interval]\label{exr:ii19-expand-16}
For \(y'=y(1-y)\), explain why a solution starting in \([0,1]\) stays there.
\end{exercise}
\begin{hint}
Use uniqueness at the equilibrium boundary solutions.
\end{hint}
\begin{solution}
A solution cannot cross \(0\) or \(1\) without first meeting an equilibrium solution; uniqueness would then force it to coincide, preventing crossing.
\end{solution}
