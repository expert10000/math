\chapter{Pathological and Nowhere-Differentiable Functions}
\label{ch:ii15}

Pathological functions reveal the sharp limits of naive intuition. Continuity need not imply differentiability, pointwise limits can destroy regularity, and dense oscillation can coexist with precise global structure.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item analyze continuity and differentiability of classical pathological examples such as Takagi- or Weierstrass-type functions;
\item prove nowhere-differentiability or failure of regularity by isolating a scale-dependent oscillation;
\item distinguish uniform convergence of a defining series from differentiability of its sum;
\item estimate moduli of continuity for explicit pathological constructions;
\item construct sequences of scales that expose incompatible difference quotients;
\item explain which regularity theorem fails and which hypothesis is missing in a given counterexample;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{theorem}\;\longrightarrow\;\text{application}.}

\section{Everywhere continuous, nowhere differentiable}
Weierstrass-type series show continuity can coexist with complete failure of classical tangents.

\section{Oscillation across scales}
High-frequency components with slowly decaying amplitude create roughness at every scale.


% BEGIN VOL02-EXPANSION II15-example-01
\begin{example}[Scale competition in a Weierstrass-type series]\label{ex:ii15-expand-01}
For
\[
W(x)=\sum_{n\ge0}a^n\cos(b^n x),\qquad 0<a<1,
\]
uniform convergence follows from \(|a^n\cos(b^nx)|\le a^n\). But the formal
derivative has terms of size roughly
\[
a^n b^n=(ab)^n.
\]
When \(ab>1\), high-frequency derivative amplitudes grow rather than decay.
This does not alone prove nowhere differentiability, but it identifies the
scale conflict: continuity is controlled by \(a^n\), while slopes are
amplified by the frequency \(b^n\).
\end{example}
% END VOL02-EXPANSION II15-example-01
\section{Cantor-type functions}
Monotone continuous functions can have derivative zero on a large set yet still increase globally.


% BEGIN VOL02-EXPANSION II15-example-02
\begin{example}[The Cantor function separates derivative behavior from total change]\label{ex:ii15-expand-02}
The Cantor function is constant on every removed middle-third interval, so its
derivative is \(0\) at every point in the interiors of those intervals. Their
union is dense and open. Yet the function increases from \(0\) to \(1\) on
\([0,1]\).

Thus knowing the derivative on a dense set is not enough to reconstruct global
change when the derivative theorem's regularity hypotheses are absent. The
increase is concentrated on the Cantor set.
\end{example}
% END VOL02-EXPANSION II15-example-02

\section{Moduli of continuity}
Hölder estimates quantify regularity below differentiability.


% BEGIN VOL02-EXPANSION II15-example-03
\begin{example}[A Hölder modulus predicts possible slope blow-up]\label{ex:ii15-expand-03}
Suppose
\[
|f(x)-f(y)|\le C|x-y|^\alpha,\qquad 0<\alpha<1.
\]
Then a difference quotient is bounded only by
\[
\frac{|f(x)-f(y)|}{|x-y|}
\le C|x-y|^{\alpha-1},
\]
whose right side grows as \(y\to x\). Hölder continuity therefore guarantees
continuity but permits much rougher local behavior than Lipschitz continuity.
The exponent explicitly measures the available regularity scale.
\end{example}
% END VOL02-EXPANSION II15-example-03
\section{Lessons for hypotheses}

\section{Core structural results}

\begin{theorem}[Uniform convergence of a Weierstrass series]\label{thm:ii15-01}
If $\sum a^n\cos(b^nx)$ has $0<a<1$, then it converges uniformly and defines a continuous function.
\begin{proof}
Each term is bounded by $a^n$ and the geometric majorant converges; the uniform limit of continuous functions is continuous.
\end{proof}
\end{theorem}

\begin{theorem}[Cantor-function monotonicity]\label{thm:ii15-02}
The Cantor function is continuous, nondecreasing, and constant on every removed middle-third interval.
\begin{proof}
Its ternary/binary digit construction preserves order; uniform approximants are continuous monotone functions and stabilize on removed intervals.
\end{proof}
\end{theorem}

\begin{theorem}[Hölder roughness principle]\label{thm:ii15-03}
A Hölder estimate $|f(x)-f(y)|\le C|x-y|^\alpha$ with $0<\alpha<1$ permits much rougher behavior than Lipschitz regularity.
\begin{proof}
The quotient by $|x-y|$ may grow like $|x-y|^{\alpha-1}$, so differentiability is not forced.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Uniform continuity of Weierstrass partial sums]\label{ex:ii15-worked-01}
Why does the geometric amplitude guarantee a continuous limit? The M-test gives uniform convergence, and uniform limits preserve continuity.
\end{example}

\begin{example}[Derivative heuristic]\label{ex:ii15-worked-02}
Why can differentiating $a^n\cos(b^nx)$ termwise be dangerous when $ab>1$? Derivative amplitudes behave like $(ab)^n$, which grow rather than decay, signaling loss of derivative control.
\end{example}

\begin{example}[Cantor function and derivative zero]\label{ex:ii15-worked-03}
How can a nonconstant monotone function have derivative zero on a dense open set? It is constant on each complementary middle-third interval; the union of these intervals is dense and open, yet the function increases through the Cantor set.
\end{example}

\begin{example}[Pointwise discontinuous limit]\label{ex:ii15-worked-04}
Use $x^n$ on $[0,1]$ to exhibit a discontinuous pointwise limit. The limit is $0$ on $[0,1)$ and $1$ at $1$.
\end{example}

Why does the geometric amplitude guarantee a continuous limit?
\end{problem}
\begin{solution}
The M-test gives uniform convergence, and uniform limits preserve continuity.
\end{solution}

\begin{problem}[Derivative heuristic]\label{prob:ii15-dossier-02}

\end{problem}
\begin{solution}

\end{solution}

\begin{problem}[Cantor function and derivative zero]\label{prob:ii15-dossier-03}
How can a nonconstant monotone function have derivative zero on a dense open set?
\end{problem}
\begin{solution}
It is constant on each complementary middle-third interval; the union of these intervals is dense and open, yet the function increases through the Cantor set.
\end{solution}

\begin{problem}[Pointwise discontinuous limit]\label{prob:ii15-dossier-04}
Use $x^n$ on $[0,1]$ to exhibit a discontinuous pointwise limit.
\end{problem}
\begin{solution}
The limit is $0$ on $[0,1)$ and $1$ at $1$.
\end{solution}

\begin{problem}[Dirichlet function]\label{prob:ii15-dossier-05}
What is special about $1_{\mathbb Q}$?
\end{problem}
\begin{solution}
It is discontinuous at every real point because every interval contains both rationals and irrationals.
\end{solution}

\begin{problem}[Thomae contrast]\label{prob:ii15-dossier-06}
Describe why Thomae's function is discontinuous at rationals but continuous at irrationals.
\end{problem}
\begin{solution}
Large values occur only at rationals with small denominators, of which there are finitely many in a bounded interval; near an irrational one can avoid those finitely many.
\end{solution}

\begin{problem}[Hölder but not Lipschitz]\label{prob:ii15-dossier-07}
Show $f(x)=\sqrt{|x|}$ is Hölder-$1/2$ but not Lipschitz near zero.
\end{problem}
\begin{solution}
$|\sqrt{|x|}-\sqrt{|y|}|\le\sqrt{|x-y|}$, while $|f(x)-f(0)|/|x|=1/\sqrt{|x|}\to\infty$.
\end{solution}

\begin{problem}[Dense set of discontinuities]\label{prob:ii15-dossier-08}
Can a Riemann-integrable function have a dense set of discontinuities?
\end{problem}
\begin{solution}
Yes; Thomae's function is Riemann integrable and discontinuous at every rational, a dense set of measure zero.
\end{solution}

\begin{problem}[Nowhere differentiability lesson]\label{prob:ii15-dossier-09}
Why does continuity alone not control local slope?
\end{problem}
\begin{solution}
Continuity bounds function increments absolutely, but differentiability requires increments to be asymptotically linear relative to distance.
\end{solution}

\begin{problem}[Uniform-limit protection]\label{prob:ii15-dossier-10}
Which pathology does uniform convergence prevent?
\end{problem}
\begin{solution}
It prevents continuous approximants from converging to a discontinuous limit.
\end{solution}

\begin{problem}[Hypothesis testing]\label{prob:ii15-dossier-11}
How should a counterexample be used when reading a theorem?
\end{problem}
\begin{solution}
Remove one hypothesis at a time and see whether the conclusion fails; a sharp counterexample shows that hypothesis controls a real mechanism.
\end{solution}

\begin{problem}[Pathology synthesis]\label{prob:ii15-dossier-12}
Why are pathological examples central to analysis?
\end{problem}
\begin{solution}
They separate intuition based on smooth finite-dimensional examples from the exact logical content of definitions and theorems.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii15-01}
Is every continuous function differentiable somewhere?
\end{exercise}
\begin{hint}
Think of nowhere-differentiable examples. Isolate one scale of the construction where the oscillation is comparable to the increment under consideration.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii15-02}
Does uniform limit of continuous functions stay continuous?
\end{exercise}
\begin{hint}
For a nowhere-differentiability argument, choose increments adapted to the series frequencies so one term dominates the difference quotient.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii15-03}
Does pointwise limit?
\end{exercise}
\begin{hint}
Use $x^n$. Uniform convergence of the defining series proves continuity, not differentiability; analyze difference quotients separately.
\end{hint}
\begin{solution}
Not necessarily.
\end{solution}

\begin{exercise}\label{exr:ii15-04}
Can a monotone function be constant on dense open pieces yet nonconstant globally?
\end{exercise}
\begin{hint}
Cantor function. Split the series into low-frequency and high-frequency parts and estimate the two pieces differently.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii15-05}
Is $\sqrt{|x|}$ Lipschitz near zero?
\end{exercise}
\begin{hint}
Check quotient. Use the modulus of continuity to record how oscillation changes with scale.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii15-06}
Is it Hölder-$1/2$?
\end{exercise}
\begin{hint}
Use square-root inequality. To refute differentiability, produce two sequences of increments whose difference quotients cannot approach the same finite limit.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii15-07}
Is Dirichlet's function Riemann integrable?
\end{exercise}
\begin{hint}
Upper/lower sums. Check whether termwise differentiation would require a derivative series that actually converges.
\end{hint}
\begin{solution}
No.
\end{solution}

\begin{exercise}\label{exr:ii15-08}
Why study counterexamples?
\end{exercise}
\begin{hint}
Test hypotheses. Do not rely on a plot: convert visible oscillation into an explicit inequality at a chosen sequence of scales.
\end{hint}
\begin{solution}
They reveal necessity and boundaries of theorems.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[M-test continuity]\label{exr:ii15-expand-01}
Why does \(\sum a^n\cos(b^nx)\) define a continuous function for \(0<a<1\)?
\end{exercise}
\begin{hint}
Compare each term with \(a^n\).
\end{hint}
\begin{solution}
The M-test gives uniform convergence of continuous terms; the limit is continuous.
\end{solution}

\begin{exercise}[Derivative amplitude]\label{exr:ii15-expand-02}
What is the amplitude scale after formally differentiating \(a^n\cos(b^nx)\)?
\end{exercise}
\begin{hint}
Differentiate and inspect the coefficient.
\end{hint}
\begin{solution}
It is \(a^n b^n=(ab)^n\).
\end{solution}

\begin{exercise}[Cantor flat intervals]\label{exr:ii15-expand-03}
What is the derivative of the Cantor function inside a removed middle-third interval?
\end{exercise}
\begin{hint}
The function is locally constant there.
\end{hint}
\begin{solution}
The derivative is \(0\).
\end{solution}

\begin{exercise}[Hölder quotient]\label{exr:ii15-expand-04}
If \(|f(x)-f(y)|\le C|x-y|^{1/2}\), what bound follows for a difference quotient?
\end{exercise}
\begin{hint}
Divide by \(|x-y|\).
\end{hint}
\begin{solution}
It is at most \(C|x-y|^{-1/2}\).
\end{solution}

\begin{exercise}[Pointwise pathology]\label{exr:ii15-expand-05}
What is the pointwise limit of \(x^n\) on \([0,1]\)?
\end{exercise}
\begin{hint}
Separate \(x<1\) and \(x=1\).
\end{hint}
\begin{solution}
It is \(0\) on \([0,1)\) and \(1\) at \(1\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Uniform continuity of Weierstrass sum]\label{exr:ii15-expand-06}
Prove the uniformly convergent Weierstrass-type sum is continuous.
\end{exercise}
\begin{hint}
Use the uniform-limit theorem.
\end{hint}
\begin{solution}
The partial sums are continuous and converge uniformly by the M-test.
\end{solution}

\begin{exercise}[Dense open flat set]\label{exr:ii15-expand-07}
Explain why the union of removed middle-third intervals is dense in \([0,1]\).
\end{exercise}
\begin{hint}
Every surviving interval is subdivided at later stages.
\end{hint}
\begin{solution}
Any nonempty interval eventually contains a removed middle-third interval.
\end{solution}

\begin{exercise}[Hölder implies continuity]\label{exr:ii15-expand-08}
Prove a Hölder estimate with exponent \(\alpha>0\) implies uniform continuity.
\end{exercise}
\begin{hint}
Choose \(\delta=(\varepsilon/C)^{1/\alpha}\).
\end{hint}
\begin{solution}
Then \(|x-y|<\delta\) gives \(|f(x)-f(y)|<\varepsilon\).
\end{solution}

\begin{exercise}[Uniform limit pathology restriction]\label{exr:ii15-expand-09}
Prove a discontinuous function cannot be the uniform limit of continuous functions.
\end{exercise}
\begin{hint}
Use the uniform-limit theorem.
\end{hint}
\begin{solution}
Uniform limits preserve continuity.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Continuity not differentiability]\label{exr:ii15-expand-10}
Give a simple continuous function not differentiable at one point.
\end{exercise}
\begin{hint}
Use absolute value.
\end{hint}
\begin{solution}
\(|x|\) is continuous but not differentiable at \(0\).
\end{solution}

\begin{exercise}[Derivative zero densely not constant]\label{exr:ii15-expand-11}
Why does the Cantor function refute the naive claim that derivative zero on a dense set forces constancy?
\end{exercise}
\begin{hint}
Its derivative is zero on the dense open complement of the Cantor set but the function increases.
\end{hint}
\begin{solution}
The claim needs stronger derivative hypotheses; density alone is insufficient.
\end{solution}

\begin{exercise}[Uniform convergence not derivative convergence]\label{exr:ii15-expand-12}
Give a uniformly convergent sequence of differentiable functions whose derivatives do not converge uniformly.
\end{exercise}
\begin{hint}
Use \(\sin(nx)/n\).
\end{hint}
\begin{solution}
\(\sin(nx)/n\to0\) uniformly, but derivatives \(\cos(nx)\) fail to converge uniformly.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Rough-data model]\label{exr:ii15-expand-13}
Why are Hölder exponents useful for noisy or fractal signals?
\end{exercise}
\begin{hint}
They quantify how increments scale with distance.
\end{hint}
\begin{solution}
The exponent gives a measurable regularity rate even when classical derivatives do not exist.
\end{solution}

\begin{exercise}[Hypothesis stress test]\label{exr:ii15-expand-14}
What is the pedagogical role of pathological examples in analysis?
\end{exercise}
\begin{hint}
Identify the exact theorem hypothesis they violate.
\end{hint}
\begin{solution}
They distinguish genuinely necessary assumptions from convenient proof artifacts.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Takagi-scale heuristic]\label{exr:ii15-expand-15}
For a dyadic tent-function sum with amplitudes \(2^{-n}\), explain why continuity can coexist with persistent slope changes.
\end{exercise}
\begin{hint}
Each finer term has smaller height but higher frequency.
\end{hint}
\begin{solution}
Uniformly summable heights ensure continuity, while slopes of individual scales can remain order one, creating unresolved oscillation at every scale.
\end{solution}

\begin{exercise}[Baire viewpoint]\label{exr:ii15-expand-16}
Why can nowhere-differentiable behavior be considered abundant in \(C[0,1]\)?
\end{exercise}
\begin{hint}
Think of differentiability constraints as restrictive closed/nowhere-dense conditions in function space.
\end{hint}
\begin{solution}
Category arguments show the set of functions differentiable somewhere can be small in the Baire-category sense, so roughness is generic rather than exceptional.
\end{solution}
