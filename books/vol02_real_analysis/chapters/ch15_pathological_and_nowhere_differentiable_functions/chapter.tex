\chapter{Pathological and Nowhere-Differentiable Functions}
\label{ch:ii15}

Pathological functions reveal the sharp limits of naive intuition. Continuity need not imply differentiability, pointwise limits can destroy regularity, and dense oscillation can coexist with precise global structure.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item analyze continuity and differentiability of classical pathological examples such as Takagi- or Weierstrass-type functions;
\item prove nowhere-differentiability or failure of regularity by isolating a scale-dependent oscillation;
\item distinguish uniform convergence of a defining series from differentiability of its sum;
\item estimate moduli of continuity for explicit pathological constructions;
\item construct sequences of scales that expose incompatible difference quotients;
\item explain which regularity theorem fails and which hypothesis is missing in a given counterexample;
\end{itemize}

Limits can lose continuity when convergence is not uniform.

\section{Dense discontinuities}
Functions may be discontinuous on dense sets while retaining integrability properties.

\section{Core structural results}

\begin{theorem}[Uniform convergence of a Weierstrass series]\label{thm:ii15-01}
If $\sum a^n\cos(b^nx)$ has $0<a<1$, then it converges uniformly and defines a continuous function.
\begin{proof}
Each term is bounded by $a^n$ and the geometric majorant converges; the uniform limit of continuous functions is continuous.
\end{proof}
\end{theorem}

\begin{theorem}[Cantor-function monotonicity]\label{thm:ii15-02}
The Cantor function is continuous, nondecreasing, and constant on every removed middle-third interval.
\begin{proof}
Its ternary/binary digit construction preserves order; uniform approximants are continuous monotone functions and stabilize on removed intervals.
\end{proof}
\end{theorem}

\begin{theorem}[Hölder roughness principle]\label{thm:ii15-03}
A Hölder estimate $|f(x)-f(y)|\le C|x-y|^\alpha$ with $0<\alpha<1$ permits much rougher behavior than Lipschitz regularity.
\begin{proof}
The quotient by $|x-y|$ may grow like $|x-y|^{\alpha-1}$, so differentiability is not forced.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Uniform continuity of Weierstrass partial sums]\label{ex:ii15-worked-01}
Why does the geometric amplitude guarantee a continuous limit? The M-test gives uniform convergence, and uniform limits preserve continuity.
\end{example}

\begin{example}[Derivative heuristic]\label{ex:ii15-worked-02}
Why can differentiating $a^n\cos(b^nx)$ termwise be dangerous when $ab>1$? Derivative amplitudes behave like $(ab)^n$, which grow rather than decay, signaling loss of derivative control.
\end{example}

\begin{example}[Cantor function and derivative zero]\label{ex:ii15-worked-03}
How can a nonconstant monotone function have derivative zero on a dense open set? It is constant on each complementary middle-third interval; the union of these intervals is dense and open, yet the function increases through the Cantor set.
\end{example}

\begin{example}[Pointwise discontinuous limit]\label{ex:ii15-worked-04}
Use $x^n$ on $[0,1]$ to exhibit a discontinuous pointwise limit. The limit is $0$ on $[0,1)$ and $1$ at $1$.
\end{example}

\begin{problem}[Dirichlet function]\label{prob:ii15-dossier-05}
What is special about $1_{\mathbb Q}$?
\end{problem}
\begin{solution}
It is discontinuous at every real point because every interval contains both rationals and irrationals.
\end{solution}

\begin{problem}[Thomae contrast]\label{prob:ii15-dossier-06}
Describe why Thomae's function is discontinuous at rationals but continuous at irrationals.
\end{problem}
\begin{solution}
Large values occur only at rationals with small denominators, of which there are finitely many in a bounded interval; near an irrational one can avoid those finitely many.
\end{solution}

\begin{problem}[Hölder but not Lipschitz]\label{prob:ii15-dossier-07}
Show $f(x)=\sqrt{|x|}$ is Hölder-$1/2$ but not Lipschitz near zero.
\end{problem}
\begin{solution}
$|\sqrt{|x|}-\sqrt{|y|}|\le\sqrt{|x-y|}$, while $|f(x)-f(0)|/|x|=1/\sqrt{|x|}\to\infty$.
\end{solution}

\begin{problem}[Dense set of discontinuities]\label{prob:ii15-dossier-08}
Can a Riemann-integrable function have a dense set of discontinuities?
\end{problem}
\begin{solution}
Yes; Thomae's function is Riemann integrable and discontinuous at every rational, a dense set of measure zero.
\end{solution}

\begin{problem}[Nowhere differentiability lesson]\label{prob:ii15-dossier-09}
Why does continuity alone not control local slope?
\end{problem}
\begin{solution}
Continuity bounds function increments absolutely, but differentiability requires increments to be asymptotically linear relative to distance.
\end{solution}

\begin{problem}[Uniform-limit protection]\label{prob:ii15-dossier-10}
Which pathology does uniform convergence prevent?
\end{problem}
\begin{solution}
It prevents continuous approximants from converging to a discontinuous limit.
\end{solution}

\begin{problem}[Hypothesis testing]\label{prob:ii15-dossier-11}
How should a counterexample be used when reading a theorem?
\end{problem}
\begin{solution}
Remove one hypothesis at a time and see whether the conclusion fails; a sharp counterexample shows that hypothesis controls a real mechanism.
\end{solution}

\begin{problem}[Pathology synthesis]\label{prob:ii15-dossier-12}
Why are pathological examples central to analysis?
\end{problem}
\begin{solution}
They separate intuition based on smooth finite-dimensional examples from the exact logical content of definitions and theorems.
\end{solution}
