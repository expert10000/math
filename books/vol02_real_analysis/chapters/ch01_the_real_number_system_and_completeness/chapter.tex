\chapter{The Real Number System and Completeness}
\label{ch:ii01}

The real numbers are an ordered field completed by a least-upper-bound principle. Completeness converts approximation into existence and drives the convergence arguments used throughout analysis.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item compute suprema and infima of explicitly described subsets of \(\mathbb R\) and justify that the proposed bounds are sharp;
\item use the least-upper-bound property to prove existence statements rather than merely estimate candidate values;
\item apply the Archimedean property to produce integers or reciprocals satisfying a prescribed inequality;
\item prove density of rational or irrational numbers between two given reals by constructing the required element;
\item derive triangle and reverse-triangle inequalities in concrete absolute-value problems;
\item construct examples that distinguish maxima from suprema and expose the failure of completeness in \(\mathbb Q\);
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Ordered fields}
Order is compatible with addition and multiplication. Positivity, trichotomy, and cancellation permit algebraic manipulation of inequalities.

\section{Bounds and extrema}
Upper and lower bounds describe sets globally; maxima and minima are bounds attained by elements of the set.

\section{Suprema and infima}
The supremum is the least upper bound and the infimum the greatest lower bound. Neither need belong to the set.


% BEGIN VOL02-EXPANSION II01-example-01
\begin{example}[Proving a supremum rather than guessing it]\label{ex:ii01-expand-01}
Let
\[
A=\left\{\frac{n}{n+1}:n\in\mathbb N\right\}.
\]
Every element is less than \(1\), so \(1\) is an upper bound. To prove it is
the least upper bound, let \(\varepsilon>0\). Choose \(n>1/\varepsilon\). Then
\[
1-\frac{n}{n+1}=\frac1{n+1}<\varepsilon,
\]
so \(n/(n+1)>1-\varepsilon\). Hence no number below \(1\) can remain an upper
bound. Therefore \(\sup A=1\).

The two-part check---upper bound plus arbitrary approximation from below---is
the reusable supremum method.
\end{example}
% END VOL02-EXPANSION II01-example-01
\section{Completeness}
Every nonempty real set bounded above has a supremum. This property fails in the rational numbers.

\section{Archimedean property}
No real number dominates all positive integers; equivalently, reciprocals of positive integers can be made arbitrarily small.


% BEGIN VOL02-EXPANSION II01-example-02
\begin{example}[Archimedean choice with an explicit target]\label{ex:ii01-expand-02}
Suppose \(x<y\). We want an integer \(n\) for which \(1/n<y-x\). By the
Archimedean property choose
\[
n>\frac1{y-x}.
\]
Because both sides are positive, taking reciprocals reverses the inequality and
gives \(1/n<y-x\).

This apparently small move is the engine behind density proofs and many
epsilon arguments: a positive geometric gap can always be resolved by a
sufficiently fine reciprocal scale.
\end{example}
% END VOL02-EXPANSION II01-example-02
\section{Density}
Between distinct real numbers lie both rational and irrational numbers.

\section{Absolute value and inequalities}
Absolute value is the one-dimensional norm and satisfies triangle and reverse-triangle inequalities.

\section{Nested intervals and roots}
Completeness yields nested-interval principles and existence of positive square roots.


% BEGIN VOL02-EXPANSION II01-example-03
\begin{example}[Nested intervals force a unique point]\label{ex:ii01-expand-03}
Let \(I_n=[a_n,b_n]\) be nested nonempty closed intervals with
\(b_n-a_n\to0\). Completeness yields at least one common point \(x\). If \(y\)
were another, then for every \(n\)
\[
|x-y|\le b_n-a_n.
\]
Given \(\varepsilon>0\), choose \(N\) with \(b_N-a_N<\varepsilon\). Then
\(|x-y|<\varepsilon\) for every \(\varepsilon>0\), hence \(x=y\).

The length estimate is the verification that turns existence from completeness
into uniqueness.
\end{example}
% END VOL02-EXPANSION II01-example-03
\section{Core structural results}

\begin{theorem}[Archimedean property]\label{thm:ii01-01}
For every real $x$ there is $n\in\mathbb N$ with $n>x$.
\begin{proof}
Otherwise $\mathbb N$ would have a supremum $s$; then $s-1$ would fail to be an upper bound, producing $n>s-1$ and hence $n+1>s$, contradiction.
\end{proof}
\end{theorem}

\begin{theorem}[Supremum approximation]\label{thm:ii01-02}
If $s=\sup A$ and $\varepsilon>0$, some $a\in A$ satisfies $s-\varepsilon<a\le s$.
\begin{proof}
If not, $s-\varepsilon$ would itself be an upper bound, contradicting the leastness of $s$.
\end{proof}
\end{theorem}

\begin{theorem}[Square-root existence]\label{thm:ii01-03}
For every $a>0$ there is a unique $r>0$ with $r^2=a$.
\begin{proof}
Take $r=\sup\{x\ge0:x^2\le a\}$. If $r^2<a$, a small increase remains admissible; if $r^2>a$, a small decrease remains an upper bound. Both contradict the definition of $r$.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Supremum of a quadratic set]\label{ex:ii01-worked-01}
Let $A=\{x\ge0:x^2<5\}$. Identify $\sup A$. The set is nonempty and bounded. Its supremum $s$ must satisfy $s^2=5$ by the square-root supremum argument, so $s=\sqrt5$.
\end{example}

\begin{example}[Infimum by reflection]\label{ex:ii01-worked-02}
Prove $\inf A=-\sup(-A)$ for nonempty $A$ bounded below. Lower bounds of $A$ correspond under sign change to upper bounds of $-A$; reversing the order turns the least upper bound into the greatest lower bound.
\end{example}

\begin{example}[No largest rational below an irrational]\label{ex:ii01-worked-03}
If $\alpha$ is irrational, show $\{q\in\mathbb Q:q<\alpha\}$ has no maximum. Given $q<\alpha$, density of $\mathbb Q$ gives a rational $r$ with $q<r<\alpha$.
\end{example}

\begin{example}[Reciprocals become small]\label{ex:ii01-worked-04}
Given $\varepsilon>0$, prove some $n$ satisfies $1/n<\varepsilon$. Choose $n>1/\varepsilon$ by the Archimedean property and take reciprocals.
\end{example}

\section{Solved dossiers}
Each dossier is a substantial solved problem. Corpus mappings guide topic choice where explicit retained source blocks exist; otherwise the dossier is a fresh canonical completion.

\begin{problem}[Supremum of a quadratic set]\label{prob:ii01-dossier-01}
Let $A=\{x\ge0:x^2<5\}$. Identify $\sup A$.
\end{problem}
\begin{solution}
The set is nonempty and bounded. Its supremum $s$ must satisfy $s^2=5$ by the square-root supremum argument, so $s=\sqrt5$.
\end{solution}

\begin{problem}[Infimum by reflection]\label{prob:ii01-dossier-02}
Prove $\inf A=-\sup(-A)$ for nonempty $A$ bounded below.
\end{problem}
\begin{solution}
Lower bounds of $A$ correspond under sign change to upper bounds of $-A$; reversing the order turns the least upper bound into the greatest lower bound.
\end{solution}

\begin{problem}[No largest rational below an irrational]\label{prob:ii01-dossier-03}
If $\alpha$ is irrational, show $\{q\in\mathbb Q:q<\alpha\}$ has no maximum.
\end{problem}
\begin{solution}
Given $q<\alpha$, density of $\mathbb Q$ gives a rational $r$ with $q<r<\alpha$.
\end{solution}

\begin{problem}[Reciprocals become small]\label{prob:ii01-dossier-04}
Given $\varepsilon>0$, prove some $n$ satisfies $1/n<\varepsilon$.
\end{problem}
\begin{solution}
Choose $n>1/\varepsilon$ by the Archimedean property and take reciprocals.
\end{solution}

\begin{problem}[Integer bracketing]\label{prob:ii01-dossier-05}
Show every $x\in\mathbb R$ satisfies $m\le x<m+1$ for some $m\in\mathbb Z$.
\end{problem}
\begin{solution}
Use the Archimedean property to restrict to finitely many candidate integers and choose the largest integer not exceeding $x$.
\end{solution}

\begin{problem}[Rational density]\label{prob:ii01-dossier-06}
Prove that $x<y$ implies there is rational $q$ with $x<q<y$.
\end{problem}
\begin{solution}
Choose $n$ with $1/n<y-x$ and then an integer $m$ with $nx<m\le nx+1$; $m/n$ lies between $x$ and $y$.
\end{solution}

\begin{problem}[Irrational density]\label{prob:ii01-dossier-07}
Prove there is an irrational number between any two reals.
\end{problem}
\begin{solution}
Choose rational $r$ in $(x-\sqrt2,y-\sqrt2)$ and use $r+\sqrt2$.
\end{solution}

\begin{problem}[Reverse triangle inequality]\label{prob:ii01-dossier-08}
Prove $||x|-|y||\le|x-y|$.
\end{problem}
\begin{solution}
Apply the triangle inequality to $x=(x-y)+y$ and then interchange $x,y$.
\end{solution}

\begin{problem}[Nested intervals are unique when lengths vanish]\label{prob:ii01-dossier-09}
If nested closed intervals have lengths tending to zero, show their intersection has at most one point.
\end{problem}
\begin{solution}
Two common points would have distance bounded by every interval length, hence by arbitrarily small positive numbers, so they coincide.
\end{solution}

\begin{problem}[Why $\mathbb Q$ is incomplete]\label{prob:ii01-dossier-10}
Explain why $\{q\in\mathbb Q:q^2<2,q\ge0\}$ has no rational supremum.
\end{problem}
\begin{solution}
Its real supremum is $\sqrt2$, and any rational candidate below or above can be improved using the order argument; $\sqrt2\notin\mathbb Q$.
\end{solution}

\begin{problem}[Geometric partial sums]\label{prob:ii01-dossier-11}
Find the supremum of $1+1/2+\cdots+1/2^n$.
\end{problem}
\begin{solution}
The sum equals $2-2^{-n}$, so $2$ is an upper bound approached arbitrarily closely and hence is the supremum.
\end{solution}

\begin{problem}[Completeness synthesis]\label{prob:ii01-dossier-12}
Explain how a supremum packages arbitrarily accurate lower approximations into one number.
\end{problem}
\begin{solution}
Completeness supplies $s=\sup A$, and supremum approximation places elements of $A$ within every positive distance below $s$.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii01-01}
Find $\sup\{n/(n+1):n\ge1\}$.
\end{exercise}
\begin{hint}
Compare with $1$. To prove a candidate supremum, show first that it is an upper bound and then that every smaller number fails to be one.
\end{hint}
\begin{solution}
The supremum is $1$.
\end{solution}

\begin{exercise}\label{exr:ii01-02}
Find $\inf\{1/n:n\ge1\}$.
\end{exercise}
\begin{hint}
Use Archimedean smallness. Use the Archimedean property to choose an integer large enough that its reciprocal lies below the prescribed tolerance.
\end{hint}
\begin{solution}
The infimum is $0$.
\end{solution}

\begin{exercise}\label{exr:ii01-03}
Solve $|2x-1|\le3$.
\end{exercise}
\begin{hint}
Use a double inequality. Rewrite the absolute-value inequality as a pair of ordinary inequalities before solving for the variable.
\end{hint}
\begin{solution}
The solution is $[-1,2]$.
\end{solution}

\begin{exercise}\label{exr:ii01-04}
Show $|x+y|\ge||x|-|y||$.
\end{exercise}
\begin{hint}
Apply reverse triangle inequality. Apply the reverse triangle inequality to compare two absolute values without expanding cases.
\end{hint}
\begin{solution}
It is the reverse triangle inequality with $x$ and $-y$.
\end{solution}

\begin{exercise}\label{exr:ii01-05}
Give a bounded rational set with no rational supremum.
\end{exercise}
\begin{hint}
Use $q^2<3$. For incompleteness, use a bounded rational set whose real least upper bound is irrational.
\end{hint}
\begin{solution}
$\{q\in\mathbb Q:q^2<3,q\ge0\}$ works.
\end{solution}

\begin{exercise}\label{exr:ii01-06}
Show every finite nonempty real set has a maximum.
\end{exercise}
\begin{hint}
Induct on the cardinality. Induct on the number of elements: adjoining one new element reduces the maximum question to comparing two numbers.
\end{hint}
\begin{solution}
Adjoin one element at a time and take the larger of it and the previous maximum.
\end{solution}

\begin{exercise}\label{exr:ii01-07}
If $A\subset B$, compare suprema.
\end{exercise}
\begin{hint}
Every upper bound of $B$ bounds $A$. If \(A\subseteq B\), start from an arbitrary upper bound of \(B\) and ask what it says about \(A\).
\end{hint}
\begin{solution}
When both exist, $\sup A\le\sup B$.
\end{solution}

\begin{exercise}\label{exr:ii01-08}
If $|x-y|<\varepsilon$ for every $\varepsilon>0$, show $x=y$.
\end{exercise}
\begin{hint}
Assume positive distance. If the distance were positive, choose \(\varepsilon\) strictly smaller than that distance and contradict the hypothesis.
\end{hint}
\begin{solution}
Take $\varepsilon=|x-y|/2$ to obtain a contradiction.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Supremum of reciprocals]\label{exr:ii01-expand-01}
Find \(\sup\{1-1/n:n\ge1\}\).
\end{exercise}
\begin{hint}
Show \(1\) is an upper bound and approximate it from below using large \(n\).
\end{hint}
\begin{solution}
The supremum is \(1\), because \(1-(1-1/n)=1/n\to0\).
\end{solution}

\begin{exercise}[Infimum of a quadratic set]\label{exr:ii01-expand-02}
Find \(\inf\{x^2+1:x\in\mathbb R\}\).
\end{exercise}
\begin{hint}
Use \(x^2\ge0\) and check whether the bound is attained.
\end{hint}
\begin{solution}
The infimum is \(1\), attained at \(x=0\).
\end{solution}

\begin{exercise}[Absolute-value interval]\label{exr:ii01-expand-03}
Solve \(|3x-2|<4\).
\end{exercise}
\begin{hint}
Rewrite as \(-4<3x-2<4\).
\end{hint}
\begin{solution}
\(-2/3<x<2\).
\end{solution}

\begin{exercise}[Archimedean integer]\label{exr:ii01-expand-04}
Find a sufficient condition on \(n\) guaranteeing \(1/n<10^{-6}\).
\end{exercise}
\begin{hint}
Take reciprocals of the positive inequality.
\end{hint}
\begin{solution}
Any integer \(n>10^6\) works.
\end{solution}

\begin{exercise}[Sharp bound]\label{exr:ii01-expand-05}
Find the maximum and supremum of \([0,1)\).
\end{exercise}
\begin{hint}
Distinguish attainment from least upper bound.
\end{hint}
\begin{solution}
The supremum is \(1\); there is no maximum.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Supremum monotonicity]\label{exr:ii01-expand-06}
If \(A\subseteq B\) and both are nonempty and bounded above, prove \(\sup A\le\sup B\).
\end{exercise}
\begin{hint}
Every upper bound of \(B\) is an upper bound of \(A\).
\end{hint}
\begin{solution}
\(\sup B\) bounds \(A\); by leastness of \(\sup A\), \(\sup A\le\sup B\).
\end{solution}

\begin{exercise}[Reverse triangle inequality]\label{exr:ii01-expand-07}
Prove \(||x|-|y||\le|x-y|\).
\end{exercise}
\begin{hint}
Apply the triangle inequality to \(x=(x-y)+y\) and then swap \(x,y\).
\end{hint}
\begin{solution}
\(|x|-|y|\le|x-y|\) and \(|y|-|x|\le|x-y|\); combine them.
\end{solution}

\begin{exercise}[Rational density]\label{exr:ii01-expand-08}
Prove there is a rational number between \(x<y\).
\end{exercise}
\begin{hint}
Choose \(n\) with \(1/n<y-x\), then trap an integer between \(nx\) and \(ny\).
\end{hint}
\begin{solution}
Choose \(m\in\mathbb Z\) with \(nx<m\le nx+1\); the width condition gives \(m<ny\), hence \(x<m/n<y\).
\end{solution}

\begin{exercise}[No positive infinitesimal]\label{exr:ii01-expand-09}
Prove that if \(a\ge0\) and \(a<1/n\) for every \(n\), then \(a=0\).
\end{exercise}
\begin{hint}
If \(a>0\), use Archimedean smallness with \(\varepsilon=a\).
\end{hint}
\begin{solution}
Choose \(n\) with \(1/n<a\), contradicting the hypothesis.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Rational incompleteness]\label{exr:ii01-expand-10}
Give a bounded nonempty subset of \(\mathbb Q\) with no supremum in \(\mathbb Q\).
\end{exercise}
\begin{hint}
Use rational \(q\ge0\) with \(q^2<2\).
\end{hint}
\begin{solution}
Its real supremum is \(\sqrt2\notin\mathbb Q\), so no rational supremum exists.
\end{solution}

\begin{exercise}[Supremum need not belong]\label{exr:ii01-expand-11}
Give a bounded real set whose supremum is not an element.
\end{exercise}
\begin{hint}
Use an open interval.
\end{hint}
\begin{solution}
\((0,1)\) has supremum \(1\notin(0,1)\).
\end{solution}

\begin{exercise}[Bounded need not have maximum]\label{exr:ii01-expand-12}
Why does completeness not imply every bounded set has a maximum?
\end{exercise}
\begin{hint}
Completeness gives a supremum, not attainment.
\end{hint}
\begin{solution}
The set \((0,1)\) is bounded and complete as a subset of \(\mathbb R\) only in the ambient supremum sense, but its supremum \(1\) is not in the set.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Error scale]\label{exr:ii01-expand-13}
To guarantee a discretization scale below \(0.002\), how large should \(n\) be if the scale is \(1/n\)?
\end{exercise}
\begin{hint}
Solve \(1/n<0.002\).
\end{hint}
\begin{solution}
Any \(n>500\) works.
\end{solution}

\begin{exercise}[Bracketing a root]\label{exr:ii01-expand-14}
Explain how completeness supports bisection once nested closed intervals containing a root have lengths tending to zero.
\end{exercise}
\begin{hint}
Use the nested-interval theorem and the vanishing-length uniqueness argument.
\end{hint}
\begin{solution}
The intervals have a common point, and shrinking lengths make it unique; continuity identifies that point as the root.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Supremum of a sum set]\label{exr:ii01-expand-15}
For nonempty bounded-above \(A,B\subset\mathbb R\), prove \(\sup(A+B)=\sup A+\sup B\).
\end{exercise}
\begin{hint}
First prove the sum of suprema is an upper bound; then approximate each supremum within \(\varepsilon/2\).
\end{hint}
\begin{solution}
Upper boundedness is immediate. Choose \(a>\sup A-\varepsilon/2\), \(b>\sup B-\varepsilon/2\); then \(a+b>\sup A+\sup B-\varepsilon\).
\end{solution}

\begin{exercise}[Square-root construction]\label{exr:ii01-expand-16}
Let \(S=\{x\ge0:x^2<3\}\). Explain the contradiction if \(s=\sup S\) satisfies \(s^2<3\).
\end{exercise}
\begin{hint}
Choose a sufficiently small positive \(h\) so \((s+h)^2<3\).
\end{hint}
\begin{solution}
Since \(3-s^2>0\), choose \(h>0\) with \(2sh+h^2<3-s^2\). Then \(s+h\in S\), contradicting that \(s\) is an upper bound.
\end{solution}
