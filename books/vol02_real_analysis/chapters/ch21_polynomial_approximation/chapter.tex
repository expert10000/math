\chapter{Polynomial Approximation}
\label{ch:ii21}

Polynomial approximation asks not for exact matching at finitely many points but for small error on an entire compact interval. The Weierstrass theorem says continuous functions admit arbitrarily accurate uniform polynomial approximations.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct explicit polynomial approximants to continuous functions in the settings developed in the chapter;
\item apply a polynomial approximation theorem only after checking compactness and continuity hypotheses;
\item estimate approximation error using modulus-of-continuity or constructive bounds when available;
\item distinguish interpolation from uniform approximation;
\item prove density statements by reducing them to the appropriate approximation theorem;
\item construct examples showing why smoothness can improve approximation rates without being necessary for mere density;
\end{itemize}

\section{Core structural results}

\begin{theorem}[Weierstrass approximation theorem]\label{thm:ii21-01}
Every continuous real function on $[a,b]$ can be uniformly approximated by polynomials.
\begin{proof}
After rescaling to $[0,1]$, Bernstein polynomials converge uniformly; uniform continuity controls nearby samples and binomial concentration controls distant samples.
\end{proof}
\end{theorem}

\begin{theorem}[Bernstein reproduction]\label{thm:ii21-02}
Bernstein operators reproduce constants and the identity function.
\begin{proof}
Binomial identities give $B_n1=1$ and $B_nx=x$.
\end{proof}
\end{theorem}

\begin{theorem}[Density consequence]\label{thm:ii21-03}
For every $f\in C[a,b]$ and $\varepsilon>0$, some polynomial $p$ satisfies $\|f-p\|_\infty<\varepsilon$.
\begin{proof}
This is exactly the statement that the polynomial subspace is dense in the sup-norm space.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Approximate absolute value]\label{ex:ii21-worked-01}
Why can $|x|$ on $[-1,1]$ be uniformly approximated by polynomials despite not being differentiable at zero? Weierstrass requires continuity only, not differentiability.
\end{example}

\begin{example}[Bernstein constant]\label{ex:ii21-worked-02}
Show $B_n1=1$. The Bernstein basis sums to $(x+(1-x))^n=1$.
\end{example}

\begin{example}[Bernstein identity]\label{ex:ii21-worked-03}
Show $B_nx=x$. The weighted sum of $k/n$ against binomial probabilities equals the mean of a Binomial$(n,x)$ variable divided by $n$, namely $x$.
\end{example}

\begin{example}[Uniform versus pointwise approximation]\label{ex:ii21-worked-04}
Why is sup norm stronger? It bounds the worst error at every point simultaneously rather than allowing the bad point to vary with the approximant.
\end{example}

\begin{problem}[Density interpretation]\label{prob:ii21-dossier-05}
What does polynomial density mean geometrically?
\end{problem}
\begin{solution}
Every open sup-norm ball around every continuous function contains a polynomial.
\end{solution}

\begin{problem}[No exact representation required]\label{prob:ii21-dossier-06}
Does Weierstrass say every continuous function is a polynomial?
\end{problem}
\begin{solution}
No; it says polynomials can approach arbitrarily closely in sup norm.
\end{solution}

\begin{problem}[Approximation of $e^x$]\label{prob:ii21-dossier-07}
Give one natural polynomial approximant.
\end{problem}
\begin{solution}
Taylor polynomials provide explicit approximants; on a compact interval their remainders can be bounded uniformly.
\end{solution}

\begin{problem}[Approximation and integration]\label{prob:ii21-dossier-08}
If $p_n\to f$ uniformly, what happens to integrals?
\end{problem}
\begin{solution}
The integrals converge because integration is continuous in the sup norm.
\end{solution}

\begin{problem}[Approximation and roots]\label{prob:ii21-dossier-09}
Does a uniformly close polynomial necessarily preserve every zero?
\end{problem}
\begin{solution}
No; zeros can move or disappear unless extra sign or transversality information is available.
\end{solution}

\begin{problem}[Smoothness and rate]\label{prob:ii21-dossier-10}
Why should smoother functions be easier to approximate rapidly?
\end{problem}
\begin{solution}
Higher regularity controls oscillation and derivative growth, enabling sharper remainder and constructive approximation estimates.
\end{solution}

\begin{problem}[Positive operators]\label{prob:ii21-dossier-11}
Why are Bernstein polynomials stable?
\end{problem}
\begin{solution}
They are convex combinations of sampled values, so they do not amplify the sup norm and preserve positivity.
\end{solution}

\begin{problem}[Approximation synthesis]\label{prob:ii21-dossier-12}
How does this differ from interpolation?
\end{problem}
\begin{solution}
Interpolation enforces exact agreement at selected points; approximation minimizes or controls global error across the whole interval.
\end{solution}
