\chapter{Polynomial Approximation}
\label{ch:ii21}

Polynomial approximation asks not for exact matching at finitely many points but for small error on an entire compact interval. The Weierstrass theorem says continuous functions admit arbitrarily accurate uniform polynomial approximations.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item construct explicit polynomial approximants to continuous functions in the settings developed in the chapter;
\item apply a polynomial approximation theorem only after checking compactness and continuity hypotheses;
\item estimate approximation error using modulus-of-continuity or constructive bounds when available;
\item distinguish interpolation from uniform approximation;
\item prove density statements by reducing them to the appropriate approximation theorem;
\item construct examples showing why smoothness can improve approximation rates without being necessary for mere density;
\end{itemize}

\section{Uniform approximation}
Approximation quality is measured by the sup norm.

\section{Weierstrass theorem}
Polynomials are dense in $C[a,b]$.

\section{Bernstein polynomials}
On $[0,1]$, positive averaging polynomials give a constructive proof.


% BEGIN VOL02-EXPANSION II21-example-01
\begin{example}[Bernstein polynomials as probabilistic averaging]\label{ex:ii21-expand-01}
For \(f\in C[0,1]\),
\[
B_nf(x)=\sum_{k=0}^n f(k/n)\binom nk x^k(1-x)^{n-k}.
\]
The coefficients form a binomial probability distribution, so
\[
B_nf(x)=\mathbb E\,f(X_n/n),\qquad X_n\sim\mathrm{Binomial}(n,x).
\]
Constants and the identity are reproduced because probabilities sum to one and
\(\mathbb E(X_n/n)=x\). This averaging interpretation explains positivity and
why concentration of \(X_n/n\) near \(x\) drives convergence.
\end{example}
% END VOL02-EXPANSION II21-example-01
\section{Modulus of continuity}
Uniform continuity controls approximation error.


% BEGIN VOL02-EXPANSION II21-example-02
\begin{example}[Modulus of continuity separates near and far samples]\label{ex:ii21-expand-02}
Let
\[
\omega_f(\delta)=\sup_{|x-y|\le\delta}|f(x)-f(y)|.
\]
In Bernstein's proof, sample points \(k/n\) near \(x\) contribute at most
\(\omega_f(\delta)\), while far sample points are controlled by a probability
tail. Uniform continuity makes the near contribution small, and binomial
concentration makes the far contribution small uniformly in \(x\).

This decomposition turns qualitative continuity into a quantitative
approximation argument.
\end{example}
% END VOL02-EXPANSION II21-example-02

Extra smoothness improves quantitative convergence rates.


% BEGIN VOL02-EXPANSION II21-example-03
\begin{example}[Extra smoothness improves an approximation rate]\label{ex:ii21-expand-03}
For a Lipschitz function with constant \(L\),
\[
|f(u)-f(v)|\le L|u-v|.
\]
Using the Bernstein probabilistic form and Cauchy--Schwarz,
\[
|B_nf(x)-f(x)|
\le L\,\mathbb E|X_n/n-x|
\le L\sqrt{\frac{x(1-x)}{n}}
\le\frac{L}{2\sqrt n}.
\]
Thus smoothness stronger than continuity yields an explicit convergence rate.
\end{example}
% END VOL02-EXPANSION II21-example-03

\section{Core structural results}

\begin{theorem}[Weierstrass approximation theorem]\label{thm:ii21-01}
Every continuous real function on $[a,b]$ can be uniformly approximated by polynomials.
\begin{proof}
After rescaling to $[0,1]$, Bernstein polynomials converge uniformly; uniform continuity controls nearby samples and binomial concentration controls distant samples.
\end{proof}
\end{theorem}

\begin{theorem}[Bernstein reproduction]\label{thm:ii21-02}
Bernstein operators reproduce constants and the identity function.
\begin{proof}
Binomial identities give $B_n1=1$ and $B_nx=x$.
\end{proof}
\end{theorem}

\begin{theorem}[Density consequence]\label{thm:ii21-03}
For every $f\in C[a,b]$ and $\varepsilon>0$, some polynomial $p$ satisfies $\|f-p\|_\infty<\varepsilon$.
\begin{proof}
This is exactly the statement that the polynomial subspace is dense in the sup-norm space.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Approximate absolute value]\label{ex:ii21-worked-01}
Why can $|x|$ on $[-1,1]$ be uniformly approximated by polynomials despite not being differentiable at zero? Weierstrass requires continuity only, not differentiability.
\end{example}

\begin{example}[Bernstein constant]\label{ex:ii21-worked-02}
Show $B_n1=1$. The Bernstein basis sums to $(x+(1-x))^n=1$.
\end{example}

\begin{example}[Bernstein identity]\label{ex:ii21-worked-03}
Show $B_nx=x$. The weighted sum of $k/n$ against binomial probabilities equals the mean of a Binomial$(n,x)$ variable divided by $n$, namely $x$.
\end{example}

\begin{example}[Uniform versus pointwise approximation]\label{ex:ii21-worked-04}
Why is sup norm stronger? It bounds the worst error at every point simultaneously rather than allowing the bad point to vary with the approximant.
\end{example}

Why can $|x|$ on $[-1,1]$ be uniformly approximated by polynomials despite not being differentiable at zero?
\end{problem}
\begin{solution}
Weierstrass requires continuity only, not differentiability.
\end{solution}

\begin{problem}[Bernstein constant]\label{prob:ii21-dossier-02}
Show $B_n1=1$.
\end{problem}
\begin{solution}
The Bernstein basis sums to $(x+(1-x))^n=1$.
\end{solution}

\begin{problem}[Bernstein identity]\label{prob:ii21-dossier-03}
Show $B_nx=x$.
\end{problem}
\begin{solution}
The weighted sum of $k/n$ against binomial probabilities equals the mean of a Binomial$(n,x)$ variable divided by $n$, namely $x$.
\end{solution}

\begin{problem}[Uniform versus pointwise approximation]\label{prob:ii21-dossier-04}
Why is sup norm stronger?
\end{problem}
\begin{solution}
It bounds the worst error at every point simultaneously rather than allowing the bad point to vary with the approximant.
\end{solution}

\begin{problem}[Density interpretation]\label{prob:ii21-dossier-05}
What does polynomial density mean geometrically?
\end{problem}
\begin{solution}
Every open sup-norm ball around every continuous function contains a polynomial.
\end{solution}

\begin{problem}[No exact representation required]\label{prob:ii21-dossier-06}
Does Weierstrass say every continuous function is a polynomial?
\end{problem}
\begin{solution}
No; it says polynomials can approach arbitrarily closely in sup norm.
\end{solution}

\begin{problem}[Approximation of $e^x$]\label{prob:ii21-dossier-07}
Give one natural polynomial approximant.
\end{problem}
\begin{solution}
Taylor polynomials provide explicit approximants; on a compact interval their remainders can be bounded uniformly.
\end{solution}

\begin{problem}[Approximation and integration]\label{prob:ii21-dossier-08}
If $p_n\to f$ uniformly, what happens to integrals?
\end{problem}
\begin{solution}
The integrals converge because integration is continuous in the sup norm.
\end{solution}

\begin{problem}[Approximation and roots]\label{prob:ii21-dossier-09}
Does a uniformly close polynomial necessarily preserve every zero?
\end{problem}
\begin{solution}
No; zeros can move or disappear unless extra sign or transversality information is available.
\end{solution}

\begin{problem}[Smoothness and rate]\label{prob:ii21-dossier-10}
Why should smoother functions be easier to approximate rapidly?
\end{problem}
\begin{solution}
Higher regularity controls oscillation and derivative growth, enabling sharper remainder and constructive approximation estimates.
\end{solution}

\begin{problem}[Positive operators]\label{prob:ii21-dossier-11}
Why are Bernstein polynomials stable?
\end{problem}
\begin{solution}
They are convex combinations of sampled values, so they do not amplify the sup norm and preserve positivity.
\end{solution}

\begin{problem}[Approximation synthesis]\label{prob:ii21-dossier-12}
How does this differ from interpolation?
\end{problem}
\begin{solution}
Interpolation enforces exact agreement at selected points; approximation minimizes or controls global error across the whole interval.
\end{solution}

\section{Exercises with complete solutions}

\begin{exercise}\label{exr:ii21-01}
What norm does Weierstrass use?
\end{exercise}
\begin{hint}
Sup norm. Separate existence of a polynomial approximant from construction of a specific interpolant; these are different problems.
\end{hint}
\begin{solution}
Uniform norm.
\end{solution}

\begin{exercise}\label{exr:ii21-02}
Must $f$ be differentiable?
\end{exercise}
\begin{hint}
Check continuity on a compact interval before invoking the uniform polynomial approximation theorem.
\end{hint}
\begin{solution}
Continuity suffices.
\end{solution}

\begin{exercise}\label{exr:ii21-03}
What interval is standard for Bernstein polynomials?
\end{exercise}
\begin{hint}
$[0,1]$. Use the modulus of continuity to convert closeness of nearby arguments into a uniform function-value estimate.
\end{hint}
\begin{solution}
$[0,1]$.
\end{solution}

\begin{exercise}\label{exr:ii21-04}
Do Bernstein polynomials reproduce constants?
\end{exercise}
\begin{hint}
If a constructive approximant is given, estimate its error directly in the sup norm.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii21-05}
Do they reproduce $x$?
\end{exercise}
\begin{hint}
Density means every \(\varepsilon>0\) admits some polynomial approximant; do not confuse this with convergence of one fixed sequence unless specified.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\begin{exercise}\label{exr:ii21-06}
Does density mean equality?
\end{exercise}
\begin{hint}
Compare the approximating polynomial with the target on the whole interval, not only at selected nodes.
\end{hint}
\begin{solution}
Arbitrarily close approximation.
\end{solution}

\begin{exercise}\label{exr:ii21-07}
Does uniform approximation imply integral approximation?
\end{exercise}
\begin{hint}
Additional smoothness can improve rates, but uniform approximability of continuous functions does not require differentiability.
\end{hint}
\begin{solution}
Yes on compact intervals.
\end{solution}

\begin{exercise}\label{exr:ii21-08}
Can $|x|$ be polynomially approximated uniformly?
\end{exercise}
\begin{hint}
To disprove an overstrong claim, choose a continuous function with limited smoothness and test the asserted rate rather than the existence theorem.
\end{hint}
\begin{solution}
Yes.
\end{solution}

\end{solution}

\begin{exercise}[Density statement]\label{exr:ii21-expand-05}
What does it mean that polynomials are dense in \(C[a,b]\)?
\end{exercise}
\begin{hint}
Use epsilon and sup norm.
\end{hint}

\end{solution}

\subsection*{Proofs}

\begin{exercise}[Bernstein positivity]\label{exr:ii21-expand-06}
Prove \(f\ge0\) implies \(B_nf\ge0\).
\end{exercise}
\begin{hint}
Each Bernstein basis weight is nonnegative.
\end{hint}
\begin{solution}
The polynomial is a nonnegative weighted sum of nonnegative sample values.
\end{solution}

\begin{exercise}[Constant reproduction]\label{exr:ii21-expand-07}
Prove Bernstein polynomials reproduce constants.
\end{exercise}
\begin{hint}
Sum the basis weights.
\end{hint}
\begin{solution}
\(\sum_k\binom nkx^k(1-x)^{n-k}=1\).
\end{solution}

\begin{exercise}[Identity reproduction]\label{exr:ii21-expand-08}
Prove \(B_nx=x\).
\end{exercise}
\begin{hint}
Use \(k\binom nk=n\binom{n-1}{k-1}\).
\end{hint}
\begin{solution}
The weighted sum becomes \(x(x+1-x)^{n-1}=x\).
\end{solution}

\begin{exercise}[Density consequence]\label{exr:ii21-expand-09}
Explain why uniform Bernstein convergence proves Weierstrass on \([0,1]\).
\end{exercise}
\begin{hint}
Each \(B_nf\) is a polynomial.
\end{hint}
\begin{solution}
For every epsilon, some Bernstein polynomial lies within epsilon in sup norm.
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Interpolation vs approximation]\label{exr:ii21-expand-10}
Why is a best or good uniform approximant not required to match \(f\) at any selected nodes?
\end{exercise}
\begin{hint}
Uniform approximation optimizes global error rather than exact data constraints.
\end{hint}
\begin{solution}
Matching nodes is an interpolation requirement, not an approximation requirement.
\end{solution}

\begin{exercise}[Continuity necessary for uniform polynomial limit]\label{exr:ii21-expand-11}
Why can a discontinuous function not be uniformly approximated arbitrarily well by polynomials?
\end{exercise}
\begin{hint}
A uniform limit of continuous functions is continuous.
\end{hint}
\begin{solution}
If errors tended to zero uniformly, the polynomial sequence would converge uniformly to the discontinuous target, impossible.
\end{solution}

\begin{exercise}[Smoothness not necessary]\label{exr:ii21-expand-12}
Why is differentiability not required for Weierstrass approximation?
\end{exercise}
\begin{hint}
The theorem assumes continuity only.
\end{hint}
\begin{solution}
Nondifferentiable continuous functions such as \(|x|\) still admit arbitrary uniform polynomial approximation.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Function compression]\label{exr:ii21-expand-13}
Why are polynomial approximants useful computationally?
\end{exercise}
\begin{hint}
They replace a function by finitely many coefficients and support fast evaluation/differentiation/integration.
\end{hint}
\begin{solution}
A certified sup error makes that finite representation globally reliable.
\end{solution}

\begin{exercise}[Shape preservation]\label{exr:ii21-expand-14}
What shape property does positivity of Bernstein weights help preserve?
\end{exercise}
\begin{hint}
Positive averaging cannot create negative values from nonnegative data.
\end{hint}
\begin{solution}
At minimum it preserves nonnegativity; related averaging arguments can preserve monotonicity/convexity under suitable conditions.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Lipschitz Bernstein rate]\label{exr:ii21-expand-15}
Derive \(\|B_nf-f\|_\infty\le L/(2\sqrt n)\) for \(L\)-Lipschitz \(f\).
\end{exercise}
\begin{hint}
Use the binomial variance.
\end{hint}
\begin{solution}
\(\mathbb E|X_n/n-x|\le\sqrt{x(1-x)/n}\le1/(2\sqrt n)\); multiply by \(L\).
\end{solution}

\begin{exercise}[Rescale interval]\label{exr:ii21-expand-16}
Explain how Weierstrass on \([0,1]\) implies it on any \([a,b]\).
\end{exercise}
\begin{hint}
Use the affine change \(x=a+(b-a)t\).
\end{hint}
\begin{solution}
Compose \(f\) with the affine map, approximate in \(t\), then substitute \(t=(x-a)/(b-a)\); a polynomial remains a polynomial.
\end{solution}
