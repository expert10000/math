\chapter{Sequences and Cauchy Sequences}
\label{ch:ii03}

Sequences encode limiting behavior through quantifiers. Cauchy sequences compare late terms only with each other, and completeness is exactly the principle that such internal coherence produces a limit.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item determine whether a sequence converges and prove the claimed limit with an explicit \(\varepsilon\)-\(N\) argument;
\item test whether a sequence is Cauchy and use completeness to infer convergence when appropriate;
\item construct subsequences with prescribed limiting behavior;
\item apply monotone convergence and squeeze arguments to concrete sequences;
\item distinguish convergence, boundedness, Cauchy behavior, and existence of convergent subsequences by counterexample;
\item estimate convergence rates sufficiently sharply to choose an \(N\) for a prescribed tolerance;
\end{itemize}

\section{Convergence}
A sequence converges when all sufficiently late terms lie in every prescribed neighborhood of a candidate limit.

\section{Uniqueness}
Metric separation makes limits unique.

\section{Boundedness}
Every convergent sequence is bounded, but boundedness alone does not force convergence.

\section{Subsequences}
Subsequences preserve order and are the basic tool for detecting hidden limiting behavior.

\section{Cauchy sequences}
A sequence is Cauchy when sufficiently late terms are mutually close.

\section{Completeness}
A metric space is complete if every Cauchy sequence converges in the space.

\section{Monotone real sequences}
Bounded monotone real sequences converge by the supremum property.

\section{Limsup and liminf}
Tail suprema and infima capture asymptotic oscillation.

\section{Core structural results}

\begin{theorem}[Convergent implies Cauchy]\label{thm:ii03-01}
Every convergent sequence in a metric space is Cauchy.
\begin{proof}
Choose a tail within $\varepsilon/2$ of the limit and apply the triangle inequality.
\end{proof}
\end{theorem}

\begin{theorem}[Monotone convergence]\label{thm:ii03-02}
Every increasing real sequence bounded above converges to the supremum of its range.
\begin{proof}
Supremum approximation supplies one late term within $\varepsilon$ below the supremum; monotonicity traps every later term.
\end{proof}
\end{theorem}

\begin{theorem}[Cauchy boundedness]\label{thm:ii03-03}
Every Cauchy sequence is bounded.
\begin{proof}
A tail lies in a unit ball about one tail term; combine this with the finite initial segment.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Direct proof for $1/n$]\label{ex:ii03-worked-01}
Prove $1/n\to0$. Given $\varepsilon>0$, choose $N>1/\varepsilon$.
\end{example}

\begin{example}[Uniqueness of limits]\label{ex:ii03-worked-02}
Prove a metric sequence cannot converge to two distinct points. The triangle inequality would make the distance between the two proposed limits smaller than itself.
\end{example}

\begin{example}[Convergent implies bounded]\label{ex:ii03-worked-03}
Prove it without invoking a theorem name. Take a tail in a unit ball around the limit and enlarge the ball to contain the finite initial segment.
\end{example}

\begin{example}[Subsequence inheritance]\label{ex:ii03-worked-04}
Show every subsequence of a convergent sequence has the same limit. Subsequence indices eventually exceed any original convergence index.
\end{example}

\begin{problem}[A Cauchy sequence in $\mathbb Q$ without rational limit]\label{prob:ii03-dossier-05}
Use rational truncations of $\sqrt2$.
\end{problem}
\begin{solution}
They converge in $\mathbb R$ and are therefore Cauchy, but a rational limit would equal $\sqrt2$ by uniqueness.
\end{solution}

\begin{problem}[Geometric-series Cauchy criterion]\label{prob:ii03-dossier-06}
Show partial sums of $\sum r^k$ are Cauchy for $|r|<1$.
\end{problem}
\begin{solution}
Bound every tail by $|r|^{n+1}/(1-|r|)$.
\end{solution}

\begin{problem}[Recursive monotone convergence]\label{prob:ii03-dossier-07}
For $x_{n+1}=\sqrt{2+x_n}$, $x_1=1$, prove $x_n\to2$.
\end{problem}
\begin{solution}
Inductively the sequence is increasing and bounded by $2$; its limit satisfies $L^2=L+2$ and positivity forces $L=2$.
\end{solution}

\begin{problem}[Bounded but divergent]\label{prob:ii03-dossier-08}
Analyze $(-1)^n$.
\end{problem}
\begin{solution}
Even and odd subsequences converge to different values, so the full sequence diverges.
\end{solution}

\begin{problem}[Limsup and liminf]\label{prob:ii03-dossier-09}
Find them for $(-1)^n+1/n$.
\end{problem}
\begin{solution}
Even and odd subsequences approach $1$ and $-1$, giving limsup $1$ and liminf $-1$.
\end{solution}

\begin{problem}[Cauchy plus a convergent subsequence]\label{prob:ii03-dossier-10}
Prove the whole sequence converges to the subsequential limit.
\end{problem}
\begin{solution}
Use one late subsequence term as a bridge between the Cauchy tail and the limit.
\end{solution}

\begin{problem}[Completeness of reals via tail bounds]\label{prob:ii03-dossier-11}
Explain how a Cauchy real sequence can be trapped by tail infima and suprema.
\end{problem}
\begin{solution}
Tail infima increase, tail suprema decrease, and their gap tends to zero; completeness gives a common limiting value.
\end{solution}

\begin{problem}[Sequence synthesis]\label{prob:ii03-dossier-12}
Distinguish convergence from the Cauchy property conceptually.
\end{problem}
\begin{solution}
Convergence references an external limit; Cauchy behavior is internal. Completeness connects the two.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii03-01}
Prove $3/n\to0$.
\end{exercise}
\begin{hint}
Choose $N>3/\varepsilon$. For an \(\varepsilon\)-\(N\) proof, solve the desired inequality for \(n\) before choosing \(N\).
\end{hint}
\begin{solution}
Then $n\ge N$ implies $3/n<\varepsilon$.
\end{solution}

\begin{exercise}\label{exr:ii03-02}
Does $(-1)^n/n$ converge?
\end{exercise}
\begin{hint}
Bound the absolute value. To show Cauchy behavior, estimate \(|a_n-a_m|\) uniformly for all \(m,n\) beyond the same index.
\end{hint}
\begin{solution}
Yes, to $0$.
\end{solution}

\begin{exercise}\label{exr:ii03-03}
Show every constant sequence is Cauchy.
\end{exercise}
\begin{hint}
Distances vanish. If monotonicity and boundedness are available, identify the bound and invoke monotone convergence rather than guessing from numerics.
\end{hint}
\begin{solution}
Every pairwise distance is zero.
\end{solution}

\begin{exercise}\label{exr:ii03-04}
If $x_n\to x$, prove $x_{n+1}\to x$.
\end{exercise}
\begin{hint}
It is a tail subsequence. Choose a subsequence by selecting indices where the sequence enters successively smaller neighborhoods of the proposed limit.
\end{hint}
\begin{solution}
The same epsilon estimate applies.
\end{solution}

\begin{exercise}\label{exr:ii03-05}
Find $\lim n/(n+1)$.
\end{exercise}
\begin{hint}
Rewrite the expression. Use the squeeze theorem only after producing upper and lower comparison sequences with the same limit.
\end{hint}
\begin{solution}
It is $1$.
\end{solution}

\begin{exercise}\label{exr:ii03-06}
Give a bounded divergent sequence.
\end{exercise}
\begin{hint}
Use alternating signs. A convergent sequence is bounded; use this as a quick necessary-condition test for a proposed example.
\end{hint}
\begin{solution}
$(-1)^n$ works.
\end{solution}

\begin{exercise}\label{exr:ii03-07}
What happens to an increasing bounded real sequence?
\end{exercise}
\begin{hint}
Use monotone convergence. To separate convergence from Cauchy behavior, pay attention to whether the ambient space is complete.
\end{hint}
\begin{solution}
It converges to the supremum of its range.
\end{solution}

\begin{exercise}\label{exr:ii03-08}
Show a Cauchy tail is bounded.
\end{exercise}
\begin{hint}
Use epsilon $1$. For an oscillating sequence, inspect even and odd subsequences before deciding whether a global limit exists.
\end{hint}
\begin{solution}
Eventually all terms lie in a unit ball around one fixed tail term.
\end{solution}
