\chapter{Sequences and Cauchy Sequences}
\label{ch:ii03}

Sequences encode limiting behavior through quantifiers. Cauchy sequences compare late terms only with each other, and completeness is exactly the principle that such internal coherence produces a limit.

\section*{Learning goals}
After this chapter, the reader should be able to:
\begin{itemize}
\item determine whether a sequence converges and prove the claimed limit with an explicit \(\varepsilon\)-\(N\) argument;
\item test whether a sequence is Cauchy and use completeness to infer convergence when appropriate;
\item construct subsequences with prescribed limiting behavior;
\item apply monotone convergence and squeeze arguments to concrete sequences;
\item distinguish convergence, boundedness, Cauchy behavior, and existence of convergent subsequences by counterexample;
\item estimate convergence rates sufficiently sharply to choose an \(N\) for a prescribed tolerance;
\end{itemize}
\section*{Conceptual roadmap}
\[
\boxed{\text{definition}\;\longrightarrow\;\text{estimate}\;\longrightarrow\;\text{structural theorem}\;\longrightarrow\;\text{application}.}
\]

\section{Convergence}
A sequence converges when all sufficiently late terms lie in every prescribed neighborhood of a candidate limit.


% BEGIN VOL02-EXPANSION II03-example-01
\begin{example}[An epsilon--N proof with a rate]\label{ex:ii03-expand-01}
For \(a_n=(3n+1)/(n+2)\), the candidate limit is \(3\). Indeed
\[
|a_n-3|=\frac5{n+2}.
\]
Given \(\varepsilon>0\), it is enough to choose
\[
N>\frac5\varepsilon-2.
\]
Then \(n\ge N\) implies \(|a_n-3|<\varepsilon\). The explicit estimate does
more than prove convergence: it tells how far into the sequence one must go
for any requested tolerance.
\end{example}
% END VOL02-EXPANSION II03-example-01
\section{Uniqueness}
Metric separation makes limits unique.

\section{Boundedness}
Every convergent sequence is bounded, but boundedness alone does not force convergence.

\section{Subsequences}
Subsequences preserve order and are the basic tool for detecting hidden limiting behavior.

\section{Cauchy sequences}
A sequence is Cauchy when sufficiently late terms are mutually close.


% BEGIN VOL02-EXPANSION II03-example-02
\begin{example}[Cauchy without knowing the limit first]\label{ex:ii03-expand-02}
Let
\[
s_n=\sum_{k=1}^n2^{-k}.
\]
For \(m>n\),
\[
|s_m-s_n|
\le\sum_{k=n+1}^\infty2^{-k}=2^{-n}.
\]
Given \(\varepsilon>0\), choose \(N\) with \(2^{-N}<\varepsilon\). Then
\(m,n\ge N\) makes the tail distance below \(\varepsilon\). Thus \((s_n)\) is
Cauchy without first using its closed form. Completeness then guarantees a
limit.
\end{example}
% END VOL02-EXPANSION II03-example-02
\section{Completeness}
A metric space is complete if every Cauchy sequence converges in the space.

\section{Monotone real sequences}
Bounded monotone real sequences converge by the supremum property.

\section{Limsup and liminf}
Tail suprema and infima capture asymptotic oscillation.


% BEGIN VOL02-EXPANSION II03-example-03
\begin{example}[Limsup detects persistent oscillation]\label{ex:ii03-expand-03}
For
\[
a_n=(-1)^n+\frac1n,
\]
the even subsequence tends to \(1\) and the odd subsequence tends to \(-1\).
Hence
\[
\limsup a_n=1,\qquad \liminf a_n=-1.
\]
Because these differ, the full sequence does not converge. Limsup and liminf
therefore quantify the two persistent asymptotic edges of the oscillation.
\end{example}
% END VOL02-EXPANSION II03-example-03
\section{Core structural results}

\begin{theorem}[Convergent implies Cauchy]\label{thm:ii03-01}
Every convergent sequence in a metric space is Cauchy.
\begin{proof}
Choose a tail within $\varepsilon/2$ of the limit and apply the triangle inequality.
\end{proof}
\end{theorem}

\begin{theorem}[Monotone convergence]\label{thm:ii03-02}
Every increasing real sequence bounded above converges to the supremum of its range.
\begin{proof}
Supremum approximation supplies one late term within $\varepsilon$ below the supremum; monotonicity traps every later term.
\end{proof}
\end{theorem}

\begin{theorem}[Cauchy boundedness]\label{thm:ii03-03}
Every Cauchy sequence is bounded.
\begin{proof}
A tail lies in a unit ball about one tail term; combine this with the finite initial segment.
\end{proof}
\end{theorem}

\section{Worked examples}

\begin{example}[Direct proof for $1/n$]\label{ex:ii03-worked-01}
Prove $1/n\to0$. Given $\varepsilon>0$, choose $N>1/\varepsilon$.
\end{example}

\begin{example}[Uniqueness of limits]\label{ex:ii03-worked-02}
Prove a metric sequence cannot converge to two distinct points. The triangle inequality would make the distance between the two proposed limits smaller than itself.
\end{example}

\begin{example}[Convergent implies bounded]\label{ex:ii03-worked-03}
Prove it without invoking a theorem name. Take a tail in a unit ball around the limit and enlarge the ball to contain the finite initial segment.
\end{example}

\begin{example}[Subsequence inheritance]\label{ex:ii03-worked-04}
Show every subsequence of a convergent sequence has the same limit. Subsequence indices eventually exceed any original convergence index.
\end{example}

\begin{problem}[Direct proof for $1/n$]\label{prob:ii03-dossier-01}
Prove $1/n\to0$.
\end{problem}
\begin{solution}
Given $\varepsilon>0$, choose $N>1/\varepsilon$.
\end{solution}

\begin{problem}[Uniqueness of limits]\label{prob:ii03-dossier-02}
Prove a metric sequence cannot converge to two distinct points.
\end{problem}
\begin{solution}

\end{solution}

\begin{problem}[Convergent implies bounded]\label{prob:ii03-dossier-03}
Prove it without invoking a theorem name.
\end{problem}
\begin{solution}
Take a tail in a unit ball around the limit and enlarge the ball to contain the finite initial segment.
\end{solution}

\begin{problem}[Subsequence inheritance]\label{prob:ii03-dossier-04}
Show every subsequence of a convergent sequence has the same limit.
\end{problem}
\begin{solution}
Subsequence indices eventually exceed any original convergence index.
\end{solution}

\begin{problem}[A Cauchy sequence in $\mathbb Q$ without rational limit]\label{prob:ii03-dossier-05}
Use rational truncations of $\sqrt2$.
\end{problem}
\begin{solution}
They converge in $\mathbb R$ and are therefore Cauchy, but a rational limit would equal $\sqrt2$ by uniqueness.
\end{solution}

\begin{problem}[Geometric-series Cauchy criterion]\label{prob:ii03-dossier-06}
Show partial sums of $\sum r^k$ are Cauchy for $|r|<1$.
\end{problem}
\begin{solution}
Bound every tail by $|r|^{n+1}/(1-|r|)$.
\end{solution}

\begin{problem}[Recursive monotone convergence]\label{prob:ii03-dossier-07}
For $x_{n+1}=\sqrt{2+x_n}$, $x_1=1$, prove $x_n\to2$.
\end{problem}
\begin{solution}
Inductively the sequence is increasing and bounded by $2$; its limit satisfies $L^2=L+2$ and positivity forces $L=2$.
\end{solution}

\begin{problem}[Bounded but divergent]\label{prob:ii03-dossier-08}
Analyze $(-1)^n$.
\end{problem}
\begin{solution}
Even and odd subsequences converge to different values, so the full sequence diverges.
\end{solution}

\begin{problem}[Limsup and liminf]\label{prob:ii03-dossier-09}
Find them for $(-1)^n+1/n$.
\end{problem}
\begin{solution}
Even and odd subsequences approach $1$ and $-1$, giving limsup $1$ and liminf $-1$.
\end{solution}

\begin{problem}[Cauchy plus a convergent subsequence]\label{prob:ii03-dossier-10}
Prove the whole sequence converges to the subsequential limit.
\end{problem}
\begin{solution}
Use one late subsequence term as a bridge between the Cauchy tail and the limit.
\end{solution}

\begin{problem}[Completeness of reals via tail bounds]\label{prob:ii03-dossier-11}
Explain how a Cauchy real sequence can be trapped by tail infima and suprema.
\end{problem}
\begin{solution}
Tail infima increase, tail suprema decrease, and their gap tends to zero; completeness gives a common limiting value.
\end{solution}

\begin{problem}[Sequence synthesis]\label{prob:ii03-dossier-12}
Distinguish convergence from the Cauchy property conceptually.
\end{problem}
\begin{solution}
Convergence references an external limit; Cauchy behavior is internal. Completeness connects the two.
\end{solution}

\section{Exercises with complete solutions}
The shorter exercise layer is kept separate from the dossiers.

\begin{exercise}\label{exr:ii03-01}
Prove $3/n\to0$.
\end{exercise}
\begin{hint}
Choose $N>3/\varepsilon$. For an \(\varepsilon\)-\(N\) proof, solve the desired inequality for \(n\) before choosing \(N\).
\end{hint}
\begin{solution}
Then $n\ge N$ implies $3/n<\varepsilon$.
\end{solution}

\begin{exercise}\label{exr:ii03-02}
Does $(-1)^n/n$ converge?
\end{exercise}
\begin{hint}
Bound the absolute value. To show Cauchy behavior, estimate \(|a_n-a_m|\) uniformly for all \(m,n\) beyond the same index.
\end{hint}
\begin{solution}
Yes, to $0$.
\end{solution}

\begin{exercise}\label{exr:ii03-03}
Show every constant sequence is Cauchy.
\end{exercise}
\begin{hint}
Distances vanish. If monotonicity and boundedness are available, identify the bound and invoke monotone convergence rather than guessing from numerics.
\end{hint}
\begin{solution}
Every pairwise distance is zero.
\end{solution}

\begin{exercise}\label{exr:ii03-04}
If $x_n\to x$, prove $x_{n+1}\to x$.
\end{exercise}
\begin{hint}
It is a tail subsequence. Choose a subsequence by selecting indices where the sequence enters successively smaller neighborhoods of the proposed limit.
\end{hint}
\begin{solution}
The same epsilon estimate applies.
\end{solution}

\begin{exercise}\label{exr:ii03-05}
Find $\lim n/(n+1)$.
\end{exercise}
\begin{hint}
Rewrite the expression. Use the squeeze theorem only after producing upper and lower comparison sequences with the same limit.
\end{hint}
\begin{solution}
It is $1$.
\end{solution}

\begin{exercise}\label{exr:ii03-06}
Give a bounded divergent sequence.
\end{exercise}
\begin{hint}
Use alternating signs. A convergent sequence is bounded; use this as a quick necessary-condition test for a proposed example.
\end{hint}
\begin{solution}
$(-1)^n$ works.
\end{solution}

\begin{exercise}\label{exr:ii03-07}
What happens to an increasing bounded real sequence?
\end{exercise}
\begin{hint}
Use monotone convergence. To separate convergence from Cauchy behavior, pay attention to whether the ambient space is complete.
\end{hint}
\begin{solution}
It converges to the supremum of its range.
\end{solution}

\begin{exercise}\label{exr:ii03-08}
Show a Cauchy tail is bounded.
\end{exercise}
\begin{hint}
Use epsilon $1$. For an oscillating sequence, inspect even and odd subsequences before deciding whether a global limit exists.
\end{hint}
\begin{solution}
Eventually all terms lie in a unit ball around one fixed tail term.
\end{solution}

\section{Graded supplementary exercises}
These exercises extend the preserved foundational set and are graded by mathematical role.

\subsection*{Standard computations}

\begin{exercise}[Rational sequence limit]\label{exr:ii03-expand-01}
Find the limit of \((2n-1)/(n+3)\).
\end{exercise}
\begin{hint}
Divide numerator and denominator by \(n\).
\end{hint}
\begin{solution}
The limit is \(2\).
\end{solution}

\begin{exercise}[Choose N]\label{exr:ii03-expand-02}
For \(a_n=1/n\), find an \(N\) ensuring \(|a_n|<10^{-4}\) for all \(n\ge N\).
\end{exercise}
\begin{hint}
Solve \(1/n<10^{-4}\).
\end{hint}
\begin{solution}
Any \(N>10^4\) works; for instance \(10001\).
\end{solution}

\begin{exercise}[Subsequence limits]\label{exr:ii03-expand-03}
Find subsequential limits of \((-1)^n\).
\end{exercise}
\begin{hint}
Separate even and odd indices.
\end{hint}
\begin{solution}
They are \(1\) and \(-1\).
\end{solution}

\begin{exercise}[Geometric Cauchy tail]\label{exr:ii03-expand-04}
Bound \(|s_m-s_n|\) for \(s_n=\sum_{k=0}^n3^{-k}\), \(m>n\).
\end{exercise}
\begin{hint}
Use an infinite geometric tail.
\end{hint}
\begin{solution}
\(|s_m-s_n|\le \sum_{k=n+1}^\infty3^{-k}=\frac12\,3^{-n}\).
\end{solution}

\begin{exercise}[Limsup]\label{exr:ii03-expand-05}
Compute limsup and liminf of \(a_n=\sin(\pi n/2)\).
\end{exercise}
\begin{hint}
List the periodic values.
\end{hint}
\begin{solution}
The limsup is \(1\), liminf is \(-1\).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Convergent implies bounded]\label{exr:ii03-expand-06}
Prove every convergent sequence is bounded.
\end{exercise}
\begin{hint}
Put the tail in a unit ball around the limit and absorb finitely many initial terms.
\end{hint}
\begin{solution}
The tail has bounded distance from the limit; the finite initial segment has a finite maximum norm.
\end{solution}

\begin{exercise}[Convergent implies Cauchy]\label{exr:ii03-expand-07}
Prove convergence implies the Cauchy property.
\end{exercise}
\begin{hint}
Put two late terms within \(\varepsilon/2\) of the limit.
\end{hint}
\begin{solution}
The triangle inequality gives \(d(x_m,x_n)<\varepsilon\).
\end{solution}

\begin{exercise}[Cauchy boundedness]\label{exr:ii03-expand-08}
Prove every Cauchy sequence is bounded.
\end{exercise}
\begin{hint}
Use the Cauchy condition with tolerance \(1\) for the tail.
\end{hint}
\begin{solution}
The tail lies in a unit ball around one tail term; finitely many earlier terms add only a finite bound.
\end{solution}

\begin{exercise}[Monotone convergence]\label{exr:ii03-expand-09}
Prove an increasing bounded-above real sequence converges.
\end{exercise}
\begin{hint}
Let \(s\) be the supremum of its range and use supremum approximation.
\end{hint}
\begin{solution}
Choose \(N\) with \(a_N>s-\varepsilon\); monotonicity traps \(s-\varepsilon<a_n\le s\) for \(n\ge N\).
\end{solution}

\subsection*{Counterexamples and hypothesis testing}

\begin{exercise}[Bounded not convergent]\label{exr:ii03-expand-10}
Give a bounded sequence that does not converge.
\end{exercise}
\begin{hint}
Use alternating signs.
\end{hint}
\begin{solution}
\((-1)^n\) is bounded but has two subsequential limits.
\end{solution}

\begin{exercise}[Cauchy in incomplete space]\label{exr:ii03-expand-11}
Give a Cauchy sequence in \(\mathbb Q\) that does not converge in \(\mathbb Q\).
\end{exercise}
\begin{hint}
Use rational decimal approximations to \(\sqrt2\).
\end{hint}
\begin{solution}
Such approximations are Cauchy in the usual metric but their real limit \(\sqrt2\) is not rational.
\end{solution}

\begin{exercise}[Subsequence convergence insufficient]\label{exr:ii03-expand-12}
Does one convergent subsequence force the full sequence to converge?
\end{exercise}
\begin{hint}
Use \((-1)^n\).
\end{hint}
\begin{solution}
No; the even subsequence converges to \(1\), but the full sequence does not converge.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Stopping criterion]\label{exr:ii03-expand-13}
If an iterative method has error bound \(\|x_n-x_*\|\le2^{-n}\), how large must \(n\) be for error below \(10^{-6}\)?
\end{exercise}
\begin{hint}
Solve \(2^{-n}<10^{-6}\).
\end{hint}
\begin{solution}
It is enough to take \(n\ge20\), since \(2^{-20}\approx9.54\times10^{-7}\).
\end{solution}

\begin{exercise}[Cauchy diagnostics]\label{exr:ii03-expand-14}
Why is a computable bound on \(\|x_m-x_n\|\) useful when the limit is unknown?
\end{exercise}
\begin{hint}
The Cauchy criterion uses only sequence terms.
\end{hint}
\begin{solution}
It can prove convergence in a complete space without knowing or being able to evaluate the limit in advance.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Cauchy subsequence criterion]\label{exr:ii03-expand-15}
Prove a Cauchy sequence with a convergent subsequence converges to the same limit.
\end{exercise}
\begin{hint}
Bridge a late sequence term to a late subsequence term.
\end{hint}
\begin{solution}
For large \(n\), choose a subsequence index \(n_k\ge n\) with both \(d(x_n,x_{n_k})\) and \(d(x_{n_k},x)\) below \(\varepsilon/2\).
\end{solution}

\begin{exercise}[Limsup convergence criterion]\label{exr:ii03-expand-16}
For a bounded real sequence, prove convergence to \(L\) iff \(\limsup a_n=\liminf a_n=L\).
\end{exercise}
\begin{hint}
Tail suprema and infima squeeze every late term.
\end{hint}
\begin{solution}
If both tail bounds converge to \(L\), then eventually \(L-\varepsilon<a_n<L+\varepsilon\). The converse is immediate from convergence.
\end{solution}
