\newcommand{\CompanionVIStableID}[1]{%
  \par
  % Reserve seven baseline rows for the source heading and its opening text.
  % This applies uniformly to all 45 existing ID labels and cannot change a
  % mathematical statement or its underlying TeX. Avoid a forced page break
  % for ordinary pages with enough room.
  \begingroup
    \dimen0=7\baselineskip
    \ifdim\pagegoal<\maxdimen
      \ifdim\dimexpr\pagegoal-\pagetotal\relax<\dimen0
        \newpage
      \fi
    \fi
  \endgroup
  \smallskip
  \noindent{\normalfont\footnotesize\sffamily\bfseries #1}\par\nobreak
  \vspace{-4pt}% visually binds the metadata to the following source heading
}
