\begin{exercise}
	Compute
	\[
	D(x)\cap D(y)
	\]
	inside \(\operatorname{Spec} k[x,y]\).
\end{exercise}

\subsection*{Canonical solution}
Standard opens satisfy $D(f)\cap D(g)=D(fg)$. Therefore in $\operatorname{Spec} k[x,y]$,
\[
D(x)\cap D(y)=D(xy)\cong\operatorname{Spec} k[x,y]_{xy}=\operatorname{Spec} k[x^{\pm1},y^{\pm1}].
\]
It is the locus where both coordinates $x$ and $y$ are invertible.
