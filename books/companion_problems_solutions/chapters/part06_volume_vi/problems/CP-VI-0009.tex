\begin{exercise}
	Show that
	\[
	D(\varepsilon)=\varnothing
	\]
	in
	\[
	\operatorname{Spec} k[\varepsilon]/(\varepsilon^2).
	\]
\end{exercise}

\subsection*{Canonical solution}
A prime ideal contains every nilpotent element. Since $\varepsilon^2=0$, every prime of $k[\varepsilon]/(\varepsilon^2)$ contains $\varepsilon$. Hence there is no prime at which $\varepsilon$ is nonvanishing, and therefore
\[
D(\varepsilon)=\varnothing.
\]
