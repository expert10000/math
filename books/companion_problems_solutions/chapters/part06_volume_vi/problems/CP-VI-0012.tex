\begin{exercise}
	Find all idempotents in
	\[
	k\times k.
	\]
\end{exercise}

\subsection*{Canonical solution}
An element $(a,b)\in k\times k$ is idempotent exactly when $a^2=a$ and $b^2=b$. Over a field the only idempotents are $0$ and $1$. Hence the four idempotents are
\[
(0,0),\quad(1,0),\quad(0,1),\quad(1,1).
\]
