\begin{exercise}
	Show that
	\[
	D(x)\cong \operatorname{Spec} k[x,x^{-1}].
	\]
\end{exercise}

\subsection*{Canonical solution}
For every ring $A$ and $f\in A$, the standard open $D(f)\subseteq\operatorname{Spec} A$ is canonically isomorphic to $\operatorname{Spec} A_f$. Taking $A=k[x]$ and $f=x$ gives
\[
D(x)\cong\operatorname{Spec}(k[x]_x)=\operatorname{Spec} k[x,x^{-1}].
\]
Under this isomorphism a prime $\mathfrak p$ with $x\notin\mathfrak p$ corresponds to its extension $\mathfrak p k[x]_x$.
