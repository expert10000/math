\begin{problem}[CP-II-0179]
\label{prob:cp-ii-0179}
\par\noindent\textbullet\quad Dyadic intervals: $I_{n,k}=[k2^{-n},(k+1)2^{-n})$.
\end{problem}

\begin{solution}
The extracted source contains only the definition
\[
I_{n,k}
=
[k2^{-n},(k+1)2^{-n}).
\]

For a fixed level \(n\), these are the standard dyadic intervals. On \([0,1)\), taking
\[
k=0,1,\ldots,2^n-1
\]
gives a disjoint partition
\[
[0,1)
=
\bigcup_{k=0}^{2^n-1}I_{n,k}.
\]
Each interval has length
\[
|I_{n,k}|=2^{-n},
\]
and every level-\((n+1)\) interval is one of the two children of a unique level-\(n\)
interval:
\[
I_{n,k}
=
I_{n+1,2k}
\cup
I_{n+1,2k+1}.
\]

Thus the source fragment records the usual nested dyadic partition structure:
\[
\boxed{
I_{n,k}=[k2^{-n},(k+1)2^{-n}).
}
\]
No further assertion to prove is included in this problem body.
\end{solution}

\begin{problem}[CP-II-0369]
\label{prob:cp-ii-0369}
\par\noindent\textbullet\quad If $|x|>2$, then the supports of $y\mapsto f(x-y)$ and $y\mapsto g(y)$
	do not overlap, so $(f*g)(x)=0$.
\end{problem}

\begin{solution}
The source fragment is the standard support argument for convolution. Assuming the support
situation implicit in the conclusion,
\[
\operatorname{supp}f\subset[-1,1],
\qquad
\operatorname{supp}g\subset[-1,1],
\]
if \(|x|>2\) and \(g(y)\neq0\), then \(|y|\le1\), so
\[
|x-y|
\ge
|x|-|y|
>
1.
\]
Hence \(f(x-y)=0\), and therefore
\[
(f*g)(x)=0.
\]
Thus
\[
\boxed{
\operatorname{supp}(f*g)
\subset
\operatorname{supp}f+\operatorname{supp}g
\subset[-2,2].
}
\]
The support hypotheses themselves are omitted from the Batch 15 fragment, so the conclusion
is stated conditional on the situation used by the source.
\end{solution}
