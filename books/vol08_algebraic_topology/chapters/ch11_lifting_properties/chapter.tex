\chapter{Lifting Properties}
\label{ch:viii11-lifting-properties}

Covering maps are powerful because maps into the base can often be lifted to
the covering space once an initial point is chosen.  Paths always lift,
homotopies of paths lift compatibly, and more general maps lift exactly when
their fundamental-group image lands inside the subgroup represented by the
cover.  This chapter develops those principles carefully.

\section{Lifts}

\begin{definition}[Lift]
\label{def:viii11-lift}
Let
\[
p:\widetilde X\to X
\]
be a covering and \(f:Y\to X\) a map.  A lift of \(f\) is a map
\[
\widetilde f:Y\to\widetilde X
\]
such that
\[
p\circ\widetilde f=f.
\]
\end{definition}

A lift therefore makes the triangle commute.

\section{Path lifting}

\begin{theorem}[Path lifting]
\label{thm:viii11-path-lifting}
Let
\[
\alpha:I\to X
\]
be a path and choose
\[
\tilde x_0\in p^{-1}(\alpha(0)).
\]
Then there exists a unique path
\[
\widetilde\alpha:I\to\widetilde X
\]
such that
\[
\widetilde\alpha(0)=\tilde x_0
\qquad\text{and}\qquad
p\circ\widetilde\alpha=\alpha.
\]
\end{theorem}

The proof subdivides \(I\) into finitely many intervals whose images lie in
evenly covered neighborhoods and continues sheet by sheet.

\section{Uniqueness of lifts}

\begin{theorem}[Uniqueness on connected domains]
\label{thm:viii11-unique-lifts}
Let \(Y\) be connected.  If
\[
\widetilde f_0,\widetilde f_1:Y\to\widetilde X
\]
are lifts of the same map \(f:Y\to X\) and agree at one point of \(Y\), then
\[
\widetilde f_0=\widetilde f_1.
\]
\end{theorem}

For locally path-connected applications, this is often proved by showing the
agreement set is both open and closed.

\section{Homotopy lifting}

\begin{theorem}[Homotopy lifting]
\label{thm:viii11-homotopy-lifting}
Let
\[
H:Y\times I\to X
\]
be a homotopy and suppose
\[
\widetilde H_0:Y\to\widetilde X
\]
lifts the initial map \(H(-,0)\).  Then there is a unique lift
\[
\widetilde H:Y\times I\to\widetilde X
\]
with
\[
\widetilde H(-,0)=\widetilde H_0.
\]
\end{theorem}

Path lifting is the case \(Y=\{*\}\).

\section{Endpoint behavior of lifted homotopies}

\begin{corollary}[Homotopic paths have matching lifted endpoints]
\label{cor:viii11-endpoints}
If paths \(\alpha,\beta:I\to X\) are homotopic rel endpoints and their lifts
begin at the same point, then their lifts end at the same point.
\end{corollary}

This makes the endpoint of a lifted loop depend only on its fundamental-group
class.

\section{Loop lifting criterion}

Let
\[
p:(\widetilde X,\tilde x_0)\to(X,x_0)
\]
be a pointed covering.

\begin{proposition}[Closed-lift criterion]
\label{prop:viii11-loop-criterion}
A loop \(\alpha\) at \(x_0\) lifts to a loop at \(\tilde x_0\) if and only if
\[
[\alpha]\in
p_*\pi_1(\widetilde X,\tilde x_0).
\]
\end{proposition}

\section{Injection on fundamental groups}

\begin{theorem}[Coverings inject on \(\pi_1\)]
\label{thm:viii11-pi1-injective}
The homomorphism
\[
p_*:\pi_1(\widetilde X,\tilde x_0)
\to
\pi_1(X,x_0)
\]
induced by a covering map is injective.
\end{theorem}

\begin{proof}
If a loop upstairs projects to a null-homotopic loop downstairs, lift the
null-homotopy beginning with the original loop.  The lifted final path is the
constant loop at \(\tilde x_0\), so the original loop is null-homotopic.
\end{proof}

\section{General lifting criterion}

\begin{theorem}[Lifting criterion]
\label{thm:viii11-general-criterion}
Let \(Y\) be path connected and locally path connected, choose \(y_0\in Y\),
and let
\[
f:(Y,y_0)\to(X,x_0)
\]
be a based map.  For a covering
\[
p:(\widetilde X,\tilde x_0)\to(X,x_0),
\]
a based lift
\[
\widetilde f:(Y,y_0)\to(\widetilde X,\tilde x_0)
\]
exists if and only if
\[
f_*\pi_1(Y,y_0)
\subset
p_*\pi_1(\widetilde X,\tilde x_0).
\]
\end{theorem}

This is one of the central structural results of covering-space theory.

\section{Construction of the lifted map}

Assume the subgroup inclusion in the lifting criterion.  For each \(y\in Y\),
choose a path \(\gamma\) from \(y_0\) to \(y\).  Lift
\[
f\circ\gamma
\]
starting at \(\tilde x_0\), and define
\[
\widetilde f(y)
\]
to be its endpoint.  The subgroup condition guarantees independence of the
chosen path.

\section{Why path independence works}

If \(\gamma_0,\gamma_1\) both run from \(y_0\) to \(y\), then
\[
\gamma_0*\overline{\gamma_1}
\]
is a loop.  Its image under \(f\) lies in the covering subgroup, so its lift
from \(\tilde x_0\) is closed.  Therefore the lifts of \(f\gamma_0\) and
\(f\gamma_1\) end at the same point.

\section{Lifting maps to the universal cover}

If \(p:\widetilde X\to X\) is universal, then
\[
p_*\pi_1(\widetilde X)=0.
\]
Hence a based map \(f:Y\to X\) lifts to \(\widetilde X\) exactly when
\[
f_*\pi_1(Y)=0.
\]

\begin{corollary}[Simply connected domains lift]
\label{cor:viii11-simply-connected-domain}
If \(Y\) is simply connected and satisfies the standard local hypotheses, then
every based map \(Y\to X\) lifts to every covering of \(X\) after a lift of the
basepoint is chosen.
\end{corollary}

\section{The logarithm as a lift}

For the exponential cover
\[
p:\mathbb R\to S^1,\qquad p(t)=e^{2\pi it},
\]
a lift of a map
\[
f:Y\to S^1
\]
is a continuous real-valued argument function
\[
\widetilde f:Y\to\mathbb R.
\]

Thus the lifting criterion becomes a precise obstruction to the existence of a
global continuous argument.

\section{Roots as lifting problems}

The map
\[
p_n:S^1\to S^1,\qquad p_n(z)=z^n,
\]
is a covering.  A continuous \(n\)th root of
\[
f:Y\to S^1
\]
is exactly a lift of \(f\) through \(p_n\).

\section{Winding obstruction}

For a loop
\[
f:S^1\to S^1
\]
of degree \(m\), a lift through
\[
z\mapsto z^n
\]
exists if and only if
\[
m\in n\mathbb Z.
\]
Equivalently,
\[
n\mid m.
\]

\section{Monodromy on a fiber}

Fix \(x_0\in X\).  A loop \(\alpha\) at \(x_0\) lifts uniquely from every
\[
\tilde x\in p^{-1}(x_0).
\]
Taking the endpoint defines a permutation of the fiber.  Homotopic loops rel
basepoint give the same permutation.

This produces an action
\[
\pi_1(X,x_0)\curvearrowright p^{-1}(x_0),
\]
called monodromy.

\section{Connectedness and transitivity}

For a connected covering of a path-connected locally path-connected base, the
monodromy action on the fiber is transitive.  Any two points in the fiber can
be joined upstairs by a path; its projection is a loop downstairs whose lift
moves the first fiber point to the second.

\section{Lifting and covering classification}

The subgroup
\[
H=p_*\pi_1(\widetilde X,\tilde x_0)
\]
controls which loops close upstairs and which maps lift.  This subgroup is the
algebraic fingerprint of the pointed connected covering.

\section{Bridge to deck transformations}

Deck transformations are self-homeomorphisms of the covering space that lie
over the identity of the base.  Lifting uniqueness makes them rigid: on a
connected cover, a deck transformation is completely determined by the image
of one point.  VIII/12 develops the resulting group action.

\section{Solved problem dossiers}

\begin{problem}[Lift a circle loop]\label{prob:viii11-01}
Lift \(\alpha(t)=e^{2\pi int}\) through \(\mathbb R\to S^1\) starting at \(0\).
\end{problem}
\begin{solution}
The unique lift is \(\widetilde\alpha(t)=nt\).
\end{solution}

\begin{problem}[Lift from another sheet]\label{prob:viii11-02}
Lift the same loop starting at \(k\in\mathbb Z\).
\end{problem}
\begin{solution}
The unique lift is \(\widetilde\alpha(t)=k+nt\), ending at \(k+n\).
\end{solution}

\begin{problem}[Closed lift test on the universal circle cover]\label{prob:viii11-03}
Which loops in \(S^1\) lift to loops in \(\mathbb R\)?
\end{problem}
\begin{solution}
Only degree-zero loops.  The covering subgroup is trivial, so the lift closes exactly for the zero class in \(\mathbb Z\).
\end{solution}

\begin{problem}[Finite circle cover criterion]\label{prob:viii11-04}
For \(p_n(z)=z^n\), which loop classes in \(\pi_1(S^1)=\mathbb Z\) lift to loops?
\end{problem}
\begin{solution}
The subgroup is \(n\mathbb Z\), so exactly multiples of \(n\) lift closed.
\end{solution}

\begin{problem}[Injection proof]\label{prob:viii11-05}
Prove \(p_*\) is injective.
\end{problem}
\begin{solution}
If an upstairs loop projects to a null-homotopic loop, lift a null-homotopy starting with the upstairs loop. The opposite edge is the constant lift, showing the original loop is null-homotopic.
\end{solution}

\begin{problem}[Uniqueness of map lifts]\label{prob:viii11-06}
Why do two lifts agreeing at one point agree everywhere on a connected domain?
\end{problem}
\begin{solution}
Locally, both lifts must lie in the same sheet and equal the inverse sheet map applied to \(f\), so the agreement set is open; its complement is also open. Connectedness forces global agreement.
\end{solution}

\begin{problem}[Homotopy endpoint invariant]\label{prob:viii11-07}
Why do homotopic based loops induce the same endpoint permutation of a fiber?
\end{problem}
\begin{solution}
Lift the based homotopy from a chosen fiber point. The boundary of the lifted square forces the two lifted loops to have the same endpoint.
\end{solution}

\begin{problem}[Continuous argument on an interval]\label{prob:viii11-08}
Why does every map \(f:I\to S^1\) admit a continuous real argument after choosing the initial value?
\end{problem}
\begin{solution}
The interval is simply connected, so the lifting criterion to \(\mathbb R\to S^1\) is automatic.
\end{solution}

\begin{problem}[No global argument on \(S^1\)]\label{prob:viii11-09}
Why does the identity map \(S^1\to S^1\) not lift to \(\mathbb R\)?
\end{problem}
\begin{solution}
Its induced map on \(\pi_1\) is the identity \(\mathbb Z\to\mathbb Z\), not contained in the trivial covering subgroup.
\end{solution}

\begin{problem}[Square root of a circle map]\label{prob:viii11-10}
When does \(f:S^1\to S^1\) admit a continuous square root?
\end{problem}
\begin{solution}
A lift through \(z\mapsto z^2\) exists iff \(\deg f\in2\mathbb Z\), so exactly when the degree is even.
\end{solution}

\begin{problem}[Cube root]\label{prob:viii11-11}
When does a degree-\(8\) circle map admit a continuous cube root?
\end{problem}
\begin{solution}
Never, because \(3\nmid8\).
\end{solution}

\begin{problem}[Fourth root]\label{prob:viii11-12}
When does a degree-\(12\) circle map admit a continuous fourth root?
\end{problem}
\begin{solution}
Yes, because \(4\mid12\).
\end{solution}

\begin{problem}[Lift from simply connected domain]\label{prob:viii11-13}
Let \(Y=D^2\) and \(f:D^2\to X\). Why does \(f\) lift to any cover after choosing the initial lift point?
\end{problem}
\begin{solution}
\(\pi_1(D^2)=0\), so the lifting subgroup criterion is automatically satisfied.
\end{solution}

\begin{problem}[Universal-cover lift]\label{prob:viii11-14}
Let \(f:Y\to X\) induce the trivial map on \(\pi_1\). What can be said about a lift to a universal cover?
\end{problem}
\begin{solution}
Under the standard path-connected/local hypotheses and a chosen basepoint lift, a unique based lift exists.
\end{solution}

\begin{problem}[Path-independence calculation]\label{prob:viii11-15}
Explain explicitly how two paths \(\gamma_0,\gamma_1\) from \(y_0\) to \(y\) are compared in the lifting criterion proof.
\end{problem}
\begin{solution}
Form the loop \(\gamma_0*\overline{\gamma_1}\). Its image class lies in the covering subgroup, so its lift closes. Hence the lifted endpoints of \(f\gamma_0\) and \(f\gamma_1\) coincide.
\end{solution}

\begin{problem}[Monodromy of \(z\mapsto z^n\)]\label{prob:viii11-16}
Describe the action of the generator of \(\pi_1(S^1)\) on the \(n\)-point fiber over \(1\).
\end{problem}
\begin{solution}
Label the roots by \(0,\dots,n-1\). Lifting one positive turn sends label \(k\) to \(k+1\pmod n\), an \(n\)-cycle.
\end{solution}

\begin{problem}[Transitivity]\label{prob:viii11-17}
Why is that monodromy action transitive?
\end{problem}
\begin{solution}
The generator acts by one \(n\)-cycle, so any fiber point can be reached from any other.
\end{solution}

\begin{problem}[Disconnected trivial cover monodromy]\label{prob:viii11-18}
Describe monodromy for the trivial cover \(X\times F\to X\).
\end{problem}
\begin{solution}
Every lifted loop stays in the same global sheet, so the action on \(F\) is trivial and therefore not transitive when \(|F|>1\).
\end{solution}

\begin{problem}[Connectedness test preview]\label{prob:viii11-19}
What does transitivity of monodromy suggest about a cover?
\end{problem}
\begin{solution}
For standard path-connected locally path-connected bases, transitive monodromy corresponds to connectedness of the covering space.
\end{solution}

\begin{problem}[Lift of identity]\label{prob:viii11-20}
What are lifts of \(\operatorname{id}_X:X\to X\) through \(p:\widetilde X\to X\)?
\end{problem}
\begin{solution}
They are sections \(s:X\to\widetilde X\) satisfying \(ps=\operatorname{id}\).
\end{solution}

\begin{problem}[Deck transformation as a lift]\label{prob:viii11-21}
Why is a deck transformation a lift of \(p:\widetilde X\to X\) through itself?
\end{problem}
\begin{solution}
If \(h\) is a deck transformation, \(ph=p\). Thus \(h:\widetilde X\to\widetilde X\) is a lift of the map \(p\) to the covering \(p\).
\end{solution}

\begin{problem}[Bridge to deck rigidity]\label{prob:viii11-22}
Why is a deck transformation on a connected cover determined by one point?
\end{problem}
\begin{solution}
Two deck transformations are lifts of the same map \(p\); if they agree at one point, uniqueness of lifts on the connected domain forces them to agree everywhere.
\end{solution}
