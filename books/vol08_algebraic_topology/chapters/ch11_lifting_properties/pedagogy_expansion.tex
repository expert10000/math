\section{Supplementary worked examples}
\label{sec:viii11-pedagogy}

\begin{example}[Lifting a circle path]\label{ex:viii11-ped-01}
For $p:\mathbb R\to S^1$, a path $\gamma(t)=e^{2\pi i\theta(t)}$ with chosen initial lift $\widetilde\gamma(0)=a$ has a unique continuous lift whose real coordinate tracks the angle continuously from $a$.
\end{example}

\begin{example}[Lift of a loop need not close]\label{ex:viii11-ped-02}
A loop on $S^1$ of winding number $k$ lifts to a path in $\mathbb R$ whose endpoints differ by $k$.  Closing upstairs is therefore equivalent to zero winding for the universal circle cover.
\end{example}

\begin{example}[Homotopy lifting]\label{ex:viii11-ped-03}
Once a lift of the initial time slice is chosen, a homotopy $H:X\times I\to B$ lifts uniquely through a covering under the standard hypotheses.  The lifted homotopy coherently lifts every path $t\mapsto H(x,t)$.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Initial lift]\label{exr:viii11-ped-01}
What extra datum is needed to lift a path $\gamma:I\to B$?
\end{exercise}
\begin{hint}
Choose a point in the fiber over $\gamma(0)$.
\end{hint}
\begin{solution}
Choose $\widetilde x_0\in E$ with $p(\widetilde x_0)=\gamma(0)$.
\end{solution}

\begin{exercise}[Endpoint of winding one]\label{exr:viii11-ped-02}
For $p(t)=e^{2\pi it}$, lift the loop $e^{2\pi it}$ starting at $0$. What is the endpoint?
\end{exercise}
\begin{hint}
Use the obvious real lift.
\end{hint}
\begin{solution}
The lift is $\widetilde\gamma(t)=t$, so it ends at $1$.
\end{solution}

\begin{exercise}[Endpoint of winding $k$]\label{exr:viii11-ped-03}
For the same cover, where does the lift of $e^{2\pi i kt}$ starting at $0$ end?
\end{exercise}
\begin{hint}
Lift the angle $kt$.
\end{hint}
\begin{solution}
It ends at $k$.
\end{solution}

\begin{exercise}[Constant path]\label{exr:viii11-ped-04}
Lift a constant path at $b$ starting at $e\in p^{-1}(b)$.
\end{exercise}
\begin{hint}
Uniqueness prevents movement between discrete sheets.
\end{hint}
\begin{solution}
The unique lift is the constant path at $e$.
\end{solution}

\begin{exercise}[Project lifted homotopy]\label{exr:viii11-ped-05}
What equation must a lifted homotopy $\widetilde H$ satisfy?
\end{exercise}
\begin{hint}
It lies over $H$.
\end{hint}
\begin{solution}
$p\circ\widetilde H=H$, together with the prescribed lift at the initial time (or other specified boundary data).
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Path-lift uniqueness]\label{exr:viii11-ped-06}
Prove two lifts of the same path with the same initial point agree.
\end{exercise}
\begin{hint}
The agreement set is open and closed in $I$ using evenly covered neighborhoods.
\end{hint}
\begin{solution}
Where the lifts agree, an evenly covered neighborhood forces them into the same sheet and hence to agree locally; the agreement set is nonempty, open, and closed, so connectedness of $I$ gives equality everywhere.
\end{solution}

\begin{exercise}[Homotopic loops endpoint]\label{exr:viii11-ped-07}
Prove endpoint-fixed homotopic loops have lifts with the same endpoint when started at the same lift.
\end{exercise}
\begin{hint}
Lift the homotopy and inspect the endpoint edge.
\end{hint}
\begin{solution}
Homotopy lifting gives a continuous path in the discrete fiber traced by lifted endpoints; it must be constant, so endpoints agree.
\end{solution}

\begin{exercise}[Lift criterion intuition]\label{exr:viii11-ped-08}
Explain why a global lift $f:X\to B$ is constrained by loops in $X$.
\end{exercise}
\begin{hint}
Every loop upstairs must project to a loop whose lift closes.
\end{hint}
\begin{solution}
If $f$ lifts, then $f_*$ sends $\pi_1(X)$ into the subgroup $p_*\pi_1(E)$; otherwise some loop would lift with mismatched endpoints.
\end{solution}

\begin{exercise}[Uniqueness of map lift]\label{exr:viii11-ped-09}
Prove two lifts of a map from a connected space that agree at one point agree everywhere under the covering hypotheses.
\end{exercise}
\begin{hint}
Use local sheet uniqueness and connectedness.
\end{hint}
\begin{solution}
The equalizer is nonempty, open, and closed because local lifts into an evenly covered neighborhood are determined by the sheet; connectedness forces global equality.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Any initial point works]\label{exr:viii11-ped-10}
Test: a path can be lifted from an arbitrary point of $E$, regardless of its projection.
\end{exercise}
\begin{hint}
The initial point must lie over the path start.
\end{hint}
\begin{solution}
False. Necessarily $p(\widetilde\gamma(0))=\gamma(0)$.
\end{solution}

\begin{exercise}[Loop lift always loop]\label{exr:viii11-ped-11}
Test: the lift of every loop is a loop.
\end{exercise}
\begin{hint}
Use a nonzero winding loop under $\mathbb R\to S^1$.
\end{hint}
\begin{solution}
False. Its endpoint can be a different point in the same fiber.
\end{solution}

\begin{exercise}[Two lifts can cross]\label{exr:viii11-ped-12}
Test: distinct lifts of the same path with the same starting point can later separate.
\end{exercise}
\begin{hint}
Use uniqueness.
\end{hint}
\begin{solution}
False. Path-lift uniqueness makes them identical.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Winding detector]\label{exr:viii11-ped-13}
How does path lifting numerically reveal winding number for loops on $S^1$?
\end{exercise}
\begin{hint}
Track the endpoint displacement in the universal cover.
\end{hint}
\begin{solution}
Lift the loop to $\mathbb R$ from $0$; the integer endpoint is exactly its winding number.
\end{solution}

\begin{exercise}[Continuation algorithm]\label{exr:viii11-ped-14}
Why can a lift be constructed incrementally along a compact interval?
\end{exercise}
\begin{hint}
Cover the base path by finitely many evenly covered neighborhoods.
\end{hint}
\begin{solution}
Compactness gives a finite subdivision on which the path stays in trivializing neighborhoods; choose the unique continuing sheet successively, producing the global lift.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Subgroup criterion]\label{exr:viii11-ped-15}
State the standard fundamental-group criterion for lifting a based map $f:(X,x_0)\to(B,b_0)$ to $(E,e_0)$.
\end{exercise}
\begin{hint}
Compare induced subgroups.
\end{hint}
\begin{solution}
For path-connected, locally path-connected spaces in the usual covering setting, a lift with $\widetilde f(x_0)=e_0$ exists iff $f_*\pi_1(X,x_0)\subset p_*\pi_1(E,e_0)$.
\end{solution}

\begin{exercise}[Homotopy-lift synthesis]\label{exr:viii11-ped-16}
Explain how path-lift uniqueness and homotopy lifting together prove homotopy invariance of the endpoint permutation of a loop.
\end{exercise}
\begin{hint}
Lift the entire endpoint-fixed homotopy.
\end{hint}
\begin{solution}
The endpoint of each lifted loop lies in a discrete fiber and varies continuously with homotopy time; it is therefore constant, so endpoint action depends only on the loop homotopy class.
\end{solution}
