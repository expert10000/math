\section{Supplementary worked examples}
\label{sec:viii28-pedagogy}

\begin{example}[Cohomology of a sphere]\label{ex:viii28-ped-01}
For $S^n$ with integral coefficients,
\[
H^k(S^n;\mathbb Z)\cong
\begin{cases}
\mathbb Z,&k=0,n,\\
0,&\text{otherwise}.
\end{cases}
\]
The degree-$n$ generator evaluates to $1$ on the oriented fundamental class.
\end{example}

\begin{example}[Cocycles as observables on cycles]\label{ex:viii28-ped-02}
A cochain $\varphi\in C^n(X;G)=\operatorname{Hom}(C_n(X),G)$ is a cocycle when it vanishes on boundaries. Hence a cohomology class gives a well-defined evaluation on homology classes, turning cochains into algebraic probes of cycles.
\end{example}

\begin{example}[Cohomology of the circle]\label{ex:viii28-ped-03}
A CW decomposition of $S^1$ with one $0$-cell and one $1$-cell has zero cellular coboundary. Thus $H^0(S^1;\mathbb Z)\cong\mathbb Z$ and $H^1(S^1;\mathbb Z)\cong\mathbb Z$, with the degree-one class measuring winding homologically.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Cochain group]\label{exr:viii28-ped-01}
Define $C^n(X;G)$ from the chain group $C_n(X)$.
\end{exercise}
\begin{hint}
Apply $\operatorname{Hom}(-,G)$.
\end{hint}
\begin{solution}
$C^n(X;G)=\operatorname{Hom}(C_n(X),G)$.
\end{solution}

\begin{exercise}[Coboundary]\label{exr:viii28-ped-02}
How is the coboundary $\delta:C^n\to C^{n+1}$ defined from the chain boundary?
\end{exercise}
\begin{hint}
Precompose with $\partial$.
\end{hint}
\begin{solution}
$(\delta\varphi)(c)=\varphi(\partial c)$.
\end{solution}

\begin{exercise}[Why delta squared vanishes]\label{exr:viii28-ped-03}
Show $\delta^2=0$.
\end{exercise}
\begin{hint}
Use $\partial^2=0$.
\end{hint}
\begin{solution}
$(\delta^2\varphi)(c)=\varphi(\partial^2c)=0$.
\end{solution}

\begin{exercise}[Point cohomology]\label{exr:viii28-ped-04}
Compute $H^*(\{*\};\mathbb Z)$.
\end{exercise}
\begin{hint}
There is one degree-zero cochain generator.
\end{hint}
\begin{solution}
$H^0\cong\mathbb Z$ and $H^n=0$ for $n>0$.
\end{solution}

\begin{exercise}[Circle cohomology]\label{exr:viii28-ped-05}
Compute $H^0$ and $H^1$ of $S^1$.
\end{exercise}
\begin{hint}
Use the one-cell-per-dimension cellular cochain complex.
\end{hint}
\begin{solution}
Both are $\mathbb Z$; higher cohomology vanishes.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Cocycle evaluates on homology]\label{exr:viii28-ped-06}
Prove a cocycle has the same value on homologous cycles.
\end{exercise}
\begin{hint}
The difference of homologous cycles is a boundary.
\end{hint}
\begin{solution}
If $z-z'=\partial c$ and $\delta\varphi=0$, then $\varphi(z)-\varphi(z')=\varphi(\partial c)=(\delta\varphi)(c)=0$.
\end{solution}

\begin{exercise}[Coboundaries evaluate trivially]\label{exr:viii28-ped-07}
Prove a coboundary evaluates to zero on every cycle.
\end{exercise}
\begin{hint}
Write $\varphi=\delta\psi$.
\end{hint}
\begin{solution}
For $\partial z=0$, $\varphi(z)=\psi(\partial z)=0$.
\end{solution}

\begin{exercise}[Contravariance]\label{exr:viii28-ped-08}
Show a map $f:X\to Y$ induces $f^*:H^n(Y;G)\to H^n(X;G)$.
\end{exercise}
\begin{hint}
Precompose cochains with $f_\#$.
\end{hint}
\begin{solution}
$f^*\varphi=\varphi\circ f_\#$ commutes with coboundary, so it descends to cohomology and reverses arrows.
\end{solution}

\begin{exercise}[Homotopy invariance]\label{exr:viii28-ped-09}
Explain why homotopic maps induce the same map on cohomology.
\end{exercise}
\begin{hint}
Dualize the chain-homotopy argument.
\end{hint}
\begin{solution}
A chain homotopy between the induced chain maps yields a cochain homotopy after applying $\operatorname{Hom}(-,G)$, so the induced cohomology maps agree.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Cohomology is covariant]\label{exr:viii28-ped-10}
Test: cohomology is covariant in the space variable.
\end{exercise}
\begin{hint}
Track the direction of pullback.
\end{hint}
\begin{solution}
False. A map $X\to Y$ induces $H^*(Y)\to H^*(X)$, so ordinary cohomology is contravariant.
\end{solution}

\begin{exercise}[Cocycles are functions on homology only]\label{exr:viii28-ped-11}
Test: every cochain defines a function on homology classes.
\end{exercise}
\begin{hint}
Boundaries must be annihilated.
\end{hint}
\begin{solution}
False. Only cocycles descend to homology because arbitrary cochains may take nonzero values on boundaries.
\end{solution}

\begin{exercise}[Same groups mean same ring]\label{exr:viii28-ped-12}
Test: knowing only the additive cohomology groups determines all cohomological structure.
\end{exercise}
\begin{hint}
Cup products add extra information.
\end{hint}
\begin{solution}
False. Spaces can have isomorphic graded groups but different cup-product rings.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Detecting a cycle]\label{exr:viii28-ped-13}
How can a cohomology class certify that a cycle is nontrivial?
\end{exercise}
\begin{hint}
Pair it with the cycle.
\end{hint}
\begin{solution}
If $[\alpha]\in H^n$ evaluates nontrivially on $[z]\in H_n$, then $[z]$ cannot be zero because cocycles vanish on boundaries.
\end{solution}

\begin{exercise}[Discrete data interpretation]\label{exr:viii28-ped-14}
Why are cochains natural for assigning measurements to simplices, cells, or edges?
\end{exercise}
\begin{hint}
They are linear functionals on chains.
\end{hint}
\begin{solution}
A cochain assigns values to generators and extends linearly, so it models quantities such as potentials, circulations, or fluxes while the coboundary encodes consistency.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Sphere dual generator]\label{exr:viii28-ped-15}
Choose generators $u\in H^n(S^n)$ and $[S^n]\in H_n(S^n)$. How may their signs be normalized?
\end{exercise}
\begin{hint}
Use the evaluation pairing.
\end{hint}
\begin{solution}
Choose orientations so that $\langle u,[S^n]\rangle=1$.
\end{solution}

\begin{exercise}[Cohomology synthesis]\label{exr:viii28-ped-16}
Explain the conceptual difference between homology classes and cohomology classes.
\end{exercise}
\begin{hint}
One represents cycles; the other probes them.
\end{hint}
\begin{solution}
Homology records cycles modulo boundaries, while cohomology records cocycles modulo coboundaries; the evaluation pairing makes cohomology an algebra of invariants acting on homological geometry.
\end{solution}
