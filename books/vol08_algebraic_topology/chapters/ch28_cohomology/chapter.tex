\chapter{Cohomology}
\label{ch:viii28-cohomology}

Homology is built from chains and boundaries.  Cohomology reverses the
direction: a cochain assigns algebraic data to chains, and the coboundary
operator is obtained by precomposing with the chain boundary.  The resulting
groups are contravariant in spaces and support additional structures that have
no direct homological analogue.

\section{Cochains}

Let \(C_\bullet\) be a chain complex and \(G\) an abelian group.

\begin{definition}[Cochain group]
\label{def:viii28-cochains}
The group of \(n\)-cochains with coefficients in \(G\) is
\[
C^n(C;G)=\operatorname{Hom}(C_n,G).
\]
\end{definition}

For a space \(X\),
\[
C^n(X;G)=\operatorname{Hom}(C_n^{\mathrm{sing}}(X),G).
\]

\section{Coboundary}

\begin{definition}[Coboundary operator]
\label{def:viii28-coboundary}
For \(\varphi\in C^n(C;G)\), define
\[
\delta\varphi=\varphi\circ\partial_{n+1}.
\]
Thus
\[
\delta:C^n\to C^{n+1}.
\]
\end{definition}

Since \(\partial^2=0\),
\[
\delta^2=0.
\]

\section{Cocycles and coboundaries}

\begin{definition}
\label{def:viii28-cocycles}
\[
Z^n(C;G)=\ker\delta,
\qquad
B^n(C;G)=\operatorname{im}\delta.
\]
Elements of \(Z^n\) are cocycles and elements of \(B^n\) are coboundaries.
\end{definition}

\section{Cohomology groups}

\begin{definition}[Cohomology]
\label{def:viii28-cohomology}
\[
H^n(C;G)=Z^n(C;G)/B^n(C;G).
\]
For a space \(X\),
\[
H^n(X;G)=H^n(C^*(X;G)).
\]
\end{definition}

\section{Contravariance}

A continuous map
\[
f:X\to Y
\]
induces
\[
f_\#:C_*(X)\to C_*(Y).
\]
Precomposition defines
\[
f^\#:C^n(Y;G)\to C^n(X;G),
\qquad
f^\#(\varphi)=\varphi\circ f_\#.
\]

\begin{theorem}[Functoriality]
\label{thm:viii28-functoriality}
The cochain maps \(f^\#\) induce
\[
f^*:H^n(Y;G)\to H^n(X;G),
\]
with
\[
(g\circ f)^*=f^*\circ g^*.
\]
\end{theorem}

\section{Degree zero}

For a path-connected space and coefficient group \(G\),
\[
H^0(X;G)\cong G.
\]
More generally, \(H^0\) is the group of locally constant \(G\)-valued functions
on path components.

\section{Spheres}

With integer coefficients,
\[
H^k(S^n;\mathbb Z)\cong
\begin{cases}
\mathbb Z,&k=0,n,\\
0,&\text{otherwise}.
\end{cases}
\]

\section{Cohomology of a point}

\[
H^0(*;G)\cong G,
\qquad
H^n(*;G)=0\quad(n>0).
\]

\section{Simplicial cohomology}

For a simplicial complex, cochains are homomorphisms from simplicial chain
groups to \(G\).  With chosen oriented simplices, a cochain can be regarded as
an assignment of coefficients to oriented simplices, compatible with sign.

Simplicial and singular cohomology agree on simplicial complexes.

\section{Cellular cohomology}

For a CW complex, dualizing the cellular chain complex gives the cellular
cochain complex
\[
C^n_{\rm cell}(X;G)
=
\operatorname{Hom}(C_n^{\rm cell}(X),G).
\]

For finite CW complexes, this often gives the fastest computation.

\section{Evaluation pairing}

A cochain evaluates on a chain.  Passing to homology and cohomology gives the
Kronecker pairing
\[
\langle\ ,\ \rangle:
H^n(X;G)\times H_n(X;\mathbb Z)\to G.
\]

If \([\varphi]\in H^n\) and \([z]\in H_n\), then
\[
\langle[\varphi],[z]\rangle=\varphi(z).
\]

This is well defined because cocycles vanish on boundaries and coboundaries
vanish on cycles.

\section{Relative cohomology}

For a pair \(A\subset X\),
\[
C^n(X,A;G)
=
\operatorname{Hom}(C_n(X,A),G).
\]

\begin{definition}[Relative cohomology]
\label{def:viii28-relative}
\[
H^n(X,A;G)=H^n(C^*(X,A;G)).
\]
\end{definition}

\section{Long exact sequence of a pair}

A pair has a natural long exact cohomology sequence
\[
\cdots
\to H^n(X,A;G)
\to H^n(X;G)
\to H^n(A;G)
\xrightarrow{\delta}
H^{n+1}(X,A;G)
\to\cdots.
\]

Notice the connecting map raises degree.

\section{Connecting homomorphism}

If a cocycle on \(A\) extends to a cochain on \(X\), its coboundary vanishes on
\(A\) and therefore determines a relative cocycle one degree higher.  This is
the cohomological connecting map.

\section{Annulus example}

Let \(A\) be one boundary circle of an annulus \(X=S^1\times I\).
Since the annulus deformation retracts onto \(A\), the relative cohomology
\[
H^*(X,A;G)
\]
vanishes.  The pair exact sequence reflects that the restriction
\[
H^1(X;G)\to H^1(A;G)
\]
is an isomorphism.

This is the canonical home of the mapped legacy annulus/connecting visual.

\section{Universal coefficient theorem for cohomology}

For integral homology and coefficient group \(G\), one has a natural short exact
sequence
\[
0\to
\operatorname{Ext}\bigl(H_{n-1}(X;\mathbb Z),G\bigr)
\to
H^n(X;G)
\to
\operatorname{Hom}\bigl(H_n(X;\mathbb Z),G\bigr)
\to0.
\]

The sequence splits noncanonically in general.

\section{Free homology case}

If \(H_{n-1}(X;\mathbb Z)\) is free, then the Ext term vanishes and
\[
H^n(X;G)
\cong
\operatorname{Hom}(H_n(X;\mathbb Z),G).
\]

\section{Cyclic Ext formula}

For positive \(m\),
\[
\operatorname{Ext}(\mathbb Z/m,G)\cong G/mG.
\]

For \(G=\mathbb Z\),
\[
\operatorname{Ext}(\mathbb Z/m,\mathbb Z)\cong\mathbb Z/m.
\]

Thus torsion in homology may appear one degree higher in integral cohomology.

\section{Projective-plane example}

Integral homology of \(\mathbb RP^2\) has
\[
H_1\cong\mathbb Z/2,
\qquad
H_2=0.
\]
The cohomology UCT gives
\[
H^2(\mathbb RP^2;\mathbb Z)\cong\mathbb Z/2.
\]

\section{Homotopy invariance}

Homotopic maps induce the same cohomology homomorphisms because their chain maps
are chain homotopic and precomposition converts that chain homotopy into a
cochain homotopy.

Thus homotopy-equivalent spaces have isomorphic cohomology groups.

\section{Reduced cohomology}

Reduced cohomology removes the degree-zero contribution of a basepoint, in
parallel with reduced homology.  For \(n>0\),
\[
\widetilde H^n(X;G)\cong H^n(X;G).
\]

\section{Why cohomology is richer}

As graded groups, cohomology often resembles homology.  But cohomology admits a
natural multiplication
\[
H^p(X;R)\times H^q(X;R)\to H^{p+q}(X;R),
\]
the cup product.  VIII/29 develops this ring structure.

\section{Solved problem dossiers}

\begin{problem}[Coboundary squared]\label{prob:viii28-01}
Prove \(\delta^2=0\).
\end{problem}
\begin{solution}
For a cochain \(\varphi\), \(\delta^2\varphi=\varphi\partial\partial=0\).
\end{solution}

\begin{problem}[Contravariance]\label{prob:viii28-02}
Verify \((g\circ f)^*=f^*g^*\).
\end{problem}
\begin{solution}
Precomposition gives \(\varphi\mapsto\varphi g_\# f_\#\), which is exactly \(f^\#(g^\#\varphi)\).
\end{solution}

\begin{problem}[H0]\label{prob:viii28-03}
Why is \(H^0(X;G)\cong G^c\) for \(c\) path components?
\end{problem}
\begin{solution}
A zero-cocycle must take the same value on endpoints of every path, hence is constant on each component; there are no negative-degree coboundaries.
\end{solution}

\begin{problem}[Sphere]\label{prob:viii28-04}
Compute integral cohomology of \(S^n\).
\end{problem}
\begin{solution}
The homology is free in degrees 0 and \(n\), so the cohomology UCT gives \(\mathbb Z\) in those degrees and zero elsewhere.
\end{solution}

\begin{problem}[Kronecker]\label{prob:viii28-05}
Why is the evaluation pairing well defined?
\end{problem}
\begin{solution}
Changing the cycle by a boundary does not change a cocycle's value, and changing the cocycle by a coboundary does not change its value on a cycle.
\end{solution}

\begin{problem}[Circle generator]\label{prob:viii28-06}
Interpret a generator of \(H^1(S^1;\mathbb Z)\).
\end{problem}
\begin{solution}
It is a cohomology class evaluating to \(1\) on the fundamental one-cycle.
\end{solution}

\begin{problem}[Torus H1]\label{prob:viii28-07}
Compute \(H^1(T^2;\mathbb Z)\).
\end{problem}
\begin{solution}
Since \(H_1(T^2)=\mathbb Z^2\) is free, \(H^1\cong\operatorname{Hom}(\mathbb Z^2,\mathbb Z)\cong\mathbb Z^2\).
\end{solution}

\begin{problem}[RP2 H2]\label{prob:viii28-08}
Use cohomology UCT to compute \(H^2(\mathbb RP^2;\mathbb Z)\).
\end{problem}
\begin{solution}
The Hom term from \(H_2=0\) vanishes; the Ext term from \(H_1=\mathbb Z/2\) is \(\mathbb Z/2\).
\end{solution}

\begin{problem}[RP2 H1]\label{prob:viii28-09}
Compute \(H^1(\mathbb RP^2;\mathbb Z)\).
\end{problem}
\begin{solution}
\(\operatorname{Hom}(\mathbb Z/2,\mathbb Z)=0\) and Ext from \(H_0=\mathbb Z\) vanishes, so \(H^1=0\).
\end{solution}

\begin{problem}[Relative disk]\label{prob:viii28-10}
Compute \(H^n(D^n,S^{n-1};G)\).
\end{problem}
\begin{solution}
The pair has one relative \(n\)-class; cohomology is \(G\) in degree \(n\) and zero otherwise.
\end{solution}

\begin{problem}[Annulus pair]\label{prob:viii28-11}
Why does \(H^*(S^1\times I,S^1\times\{0\};G)=0\)?
\end{problem}
\begin{solution}
The chosen boundary circle is a deformation retract of the annulus, so the pair is homotopy equivalent to \((A,A)\).
\end{solution}

\begin{problem}[Connecting map]\label{prob:viii28-12}
Describe the cohomology connecting homomorphism geometrically.
\end{problem}
\begin{solution}
Extend a cocycle on \(A\) to a cochain on \(X\); its coboundary vanishes on \(A\) and defines a relative cocycle one degree higher.
\end{solution}

\begin{problem}[Cellular cochains]\label{prob:viii28-13}
For a finite CW complex with \(c_n\) \(n\)-cells, what is \(C^n_{\rm cell}(X;G)\)?
\end{problem}
\begin{solution}
It is \(\operatorname{Hom}(\mathbb Z^{c_n},G)\cong G^{c_n}\).
\end{solution}

\begin{problem}[Finite graph]\label{prob:viii28-14}
Compute \(H^1\) of a connected graph of rank \(r\) over \(\mathbb Z\).
\end{problem}
\begin{solution}
Its \(H_1\) is \(\mathbb Z^r\), so \(H^1\cong\mathbb Z^r\).
\end{solution}

\begin{problem}[Moore cohomology]\label{prob:viii28-15}
If \(H_n(M)=\mathbb Z/m\) and other reduced homology vanishes, where does integral cohomology torsion appear?
\end{problem}
\begin{solution}
\(H^{n+1}(M;\mathbb Z)\cong\operatorname{Ext}(\mathbb Z/m,\mathbb Z)\cong\mathbb Z/m\).
\end{solution}

\begin{problem}[Free case]\label{prob:viii28-16}
Why does free integral homology make cohomology a dual group?
\end{problem}
\begin{solution}
Ext vanishes, leaving \(H^n(X;G)\cong\operatorname{Hom}(H_n(X),G)\).
\end{solution}

\begin{problem}[Homotopy]\label{prob:viii28-17}
Why do homotopic maps induce equal cohomology maps?
\end{problem}
\begin{solution}
The induced chain maps are chain homotopic; dualizing yields cochain-homotopic maps, which agree on cohomology.
\end{solution}

\begin{problem}[Reduced]\label{prob:viii28-18}
For \(n>0\), compare \(H^n\) and reduced \(H^n\).
\end{problem}
\begin{solution}
They agree; only degree zero is altered.
\end{solution}

\begin{problem}[Coefficient field]\label{prob:viii28-19}
What simplification occurs over a field for finite-dimensional chain groups?
\end{problem}
\begin{solution}
Cochain groups are ordinary vector-space duals and Ext phenomena over the field disappear.
\end{solution}

\begin{problem}[SVG annulus]\label{prob:viii28-20}
What concept should the annulus SVG emphasize?
\end{problem}
\begin{solution}
The deformation-retract geometry together with the restriction/connecting portion of the cohomology long exact sequence.
\end{solution}
