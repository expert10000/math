\section{Supplementary worked examples}
\label{sec:viii22-pedagogy}

\begin{example}[Chain homotopy kills the difference on homology]\label{ex:viii22-ped-01}
If chain maps $f,g:C_*\to D_*$ satisfy
\[
g-f=d_DK+Kd_C,
\]
then for a cycle $z$ the difference $g(z)-f(z)=d_DK(z)$ is a boundary. Hence chain-homotopic maps induce the same homology map.
\end{example}

\begin{example}[The prism operator]\label{ex:viii22-ped-02}
A topological homotopy $H:X\times I\to Y$ produces a prism operator $P:C_n(X)\to C_{n+1}(Y)$ satisfying
\[
g_\#-f_\#=\partial P+P\partial.
\]
This chain identity is the concrete bridge from continuous deformation to homology invariance.
\end{example}

\begin{example}[Homotopy inverse at chain level]\label{ex:viii22-ped-03}
When chain maps $f:C\to D$ and $g:D\to C$ have composites chain homotopic to the identities, $f$ and $g$ induce mutually inverse homology maps even if the composites are not literally equal to the identity on chains.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Degree of K]\label{exr:viii22-ped-01}
What degree does a chain homotopy $K$ have?
\end{exercise}
\begin{hint}
Compare degrees in $dK+Kd$.
\end{hint}
\begin{solution}
$K:C_n\to D_{n+1}$ has degree $+1$.
\end{solution}

\begin{exercise}[Cycle difference]\label{exr:viii22-ped-02}
If $z$ is a cycle and $g-f=dK+Kd$, simplify $(g-f)(z)$.
\end{exercise}
\begin{hint}
$dz=0$.
\end{hint}
\begin{solution}
$(g-f)(z)=dK(z)$, a boundary.
\end{solution}

\begin{exercise}[Boundary compatibility]\label{exr:viii22-ped-03}
Why are $f$ and $g$ required to be chain maps?
\end{exercise}
\begin{hint}
Homology classes must map cycles to cycles and boundaries to boundaries.
\end{hint}
\begin{solution}
Chain-map identities $df=fd$ and $dg=gd$ ensure induced homology maps are defined.
\end{solution}

\begin{exercise}[Zero-homotopic map]\label{exr:viii22-ped-04}
If $f=dK+Kd$, what map does $f$ induce on homology?
\end{exercise}
\begin{hint}
Compare $f$ to the zero chain map.
\end{hint}
\begin{solution}
$f_*=0$.
\end{solution}

\begin{exercise}[Identity contraction preview]\label{exr:viii22-ped-05}
What equation makes $\operatorname{id}_C$ chain homotopic to zero?
\end{exercise}
\begin{hint}
Set $f=0$ and $g=\operatorname{id}$.
\end{hint}
\begin{solution}
$\operatorname{id}_C=dK+Kd$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Equality on homology]\label{exr:viii22-ped-06}
Prove chain-homotopic chain maps induce equal maps on homology.
\end{exercise}
\begin{hint}
Evaluate the chain-homotopy equation on cycles.
\end{hint}
\begin{solution}
The difference on a cycle is a boundary, so the resulting homology classes agree.
\end{solution}

\begin{exercise}[Composition stability]\label{exr:viii22-ped-07}
Show postcomposing chain-homotopic maps with a chain map preserves chain homotopy.
\end{exercise}
\begin{hint}
Compose the homotopy operator.
\end{hint}
\begin{solution}
If $g-f=dK+Kd$ and $u$ is a chain map, then $ug-uf=d(uK)+(uK)d$.
\end{solution}

\begin{exercise}[Precomposition stability]\label{exr:viii22-ped-08}
Show precomposing chain-homotopic maps with a chain map preserves chain homotopy.
\end{exercise}
\begin{hint}
Use $Kv$ as the new homotopy.
\end{hint}
\begin{solution}
For a chain map $v$, $(g-f)v=d(Kv)+(Kv)d$.
\end{solution}

\begin{exercise}[Transitivity]\label{exr:viii22-ped-09}
Prove chain homotopy is transitive.
\end{exercise}
\begin{hint}
Add homotopy operators.
\end{hint}
\begin{solution}
If $g-f=dK+Kd$ and $h-g=dL+Ld$, then $h-f=d(K+L)+(K+L)d$.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Equal homology maps imply chain homotopy]\label{exr:viii22-ped-10}
Test: if $f_*=g_*$ on homology, then $f$ and $g$ are chain homotopic.
\end{exercise}
\begin{hint}
Chain homotopy is stronger than equality after passing to homology.
\end{hint}
\begin{solution}
False in general; homology forgets extension-level chain information.
\end{solution}

\begin{exercise}[Homotopy operator degree zero]\label{exr:viii22-ped-11}
Test: a chain homotopy has degree $0$.
\end{exercise}
\begin{hint}
Check the degrees of $dK$.
\end{hint}
\begin{solution}
False. It has degree $+1$ so that $dK+Kd$ has degree $0$.
\end{solution}

\begin{exercise}[Any graded maps]\label{exr:viii22-ped-12}
Test: the equation $g-f=dK+Kd$ makes sense without $f,g$ being chain maps.
\end{exercise}
\begin{hint}
The equation can be written, but induced homology maps may not exist.
\end{hint}
\begin{solution}
As an algebraic equality it may be written, but the standard conclusion about induced homology requires $f$ and $g$ to be chain maps.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Homotopy invariance pipeline]\label{exr:viii22-ped-13}
Summarize how a geometric homotopy proves equality of induced homology maps.
\end{exercise}
\begin{hint}
Insert the singular-chain functor between geometry and homology.
\end{hint}
\begin{solution}
A geometric homotopy yields the prism chain homotopy; the chain identity makes the two induced maps differ by boundaries on cycles, hence they agree on homology.
\end{solution}

\begin{exercise}[Algorithmic certificate]\label{exr:viii22-ped-14}
Why can an explicit matrix $K$ be a certificate that two finite chain maps induce the same homology map?
\end{exercise}
\begin{hint}
Check one linear identity.
\end{hint}
\begin{solution}
Verifying $g-f=dK+Kd$ is a finite matrix calculation and directly certifies equality on homology.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Chain-homotopy equivalence]\label{exr:viii22-ped-15}
Show a chain-homotopy equivalence is a quasi-isomorphism.
\end{exercise}
\begin{hint}
Use homotopy inverses.
\end{hint}
\begin{solution}
The induced maps of the two chain maps compose to identities on homology, so each is an isomorphism.
\end{solution}

\begin{exercise}[Derived viewpoint preview]\label{exr:viii22-ped-16}
Why is it useful to identify chain-homotopic maps before taking homology?
\end{exercise}
\begin{hint}
They are indistinguishable by homology and behave well under composition.
\end{hint}
\begin{solution}
Quotienting by chain homotopy forms the homotopy category of complexes, where homology factors naturally and homotopy-equivalent complexes become isomorphic.
\end{solution}
