\chapter{Chain Homotopies}
\label{ch:viii22-chain-homotopies}

Chain homotopy is the algebraic shadow of topological homotopy.  It compares
chain maps without requiring equality and is exactly the equivalence relation
needed for homology: chain-homotopic maps induce the same maps on homology.

\section{Definition}

\begin{definition}[Chain homotopy]
\label{def:viii22-chain-homotopy}
Let \(f,g:C_\bullet\to D_\bullet\) be chain maps.  A chain homotopy
\(s:f\simeq g\) is a family
\[
s_n:C_n\to D_{n+1}
\]
such that
\[
g-f=\partial s+s\partial.
\]
\end{definition}

\section{Degree convention}
The chain differential has degree \(-1\), while a chain homotopy has degree
\(+1\).  Thus both terms
\[
\partial s,\qquad s\partial
\]
have degree \(0\).

\section{Homology consequence}
\begin{theorem}[Homology invariance]
\label{thm:viii22-homology-invariance}
If \(f\simeq g\) as chain maps, then
\[
H_n(f)=H_n(g)
\]
for every \(n\).
\end{theorem}

\begin{proof}
For a cycle \(z\),
\[
g(z)-f(z)=\partial s(z)+s(\partial z)=\partial s(z),
\]
so the two images differ by a boundary.
\end{proof}

\section{Reflexivity}
The zero operator gives
\[
f\simeq f.
\]

\section{Symmetry}
If \(s:f\simeq g\), then
\[
-s:g\simeq f.
\]

\section{Transitivity}
If \(s:f\simeq g\) and \(t:g\simeq h\), then
\[
s+t:f\simeq h.
\]

Hence chain homotopy is an equivalence relation on chain maps.

\section{Compatibility with composition}
If \(f\simeq g:C\to D\) and \(u:D\to E\) is a chain map, then
\[
uf\simeq ug.
\]
Similarly, if \(v:B\to C\), then
\[
fv\simeq gv.
\]

\section{Chain-homotopy equivalence}
\begin{definition}[Chain-homotopy equivalence]
\label{def:viii22-chain-equivalence}
Chain complexes \(C\) and \(D\) are chain-homotopy equivalent if there are
chain maps
\[
f:C\to D,\qquad g:D\to C
\]
with
\[
gf\simeq\operatorname{id}_C,
\qquad
fg\simeq\operatorname{id}_D.
\]
\end{definition}

\begin{theorem}
\label{thm:viii22-chain-equivalence-homology}
Chain-homotopy-equivalent complexes have naturally isomorphic homology groups.
\end{theorem}

\section{Null-homotopic chain maps}
A chain map \(f:C\to D\) is null-homotopic if
\[
f\simeq0.
\]
Equivalently there are maps \(s_n:C_n\to D_{n+1}\) satisfying
\[
f=\partial s+s\partial.
\]

Every null-homotopic chain map induces the zero map on homology.

\section{The prism operator revisited}
VIII/20 constructed, from a homotopy \(F:X\times I\to Y\), a prism operator
\[
P:C_n^{\rm sing}(X)\to C_{n+1}^{\rm sing}(Y)
\]
with
\[
g_\#-f_\#=\partial P+P\partial.
\]
Thus topological homotopy becomes chain homotopy.

\section{Homotopy of simplicial maps}
After subdivision or through appropriate simplicial homotopy data, simplicial
maps can likewise give chain-homotopic simplicial chain maps.

\section{Chain homotopy and exact complexes}
An exact complex need not automatically come equipped with a contraction.
Chain homotopy provides the language for stating the stronger property that the
identity itself is null-homotopic.

\section{Hom-complex viewpoint}
For graded homomorphisms \(T:C\to D\), define
\[
\delta(T)=\partial_D T-(-1)^{|T|}T\partial_C.
\]
For \(|s|=1\),
\[
\delta(s)=\partial s+s\partial.
\]
Thus \(g-f=\delta(s)\): chain-homotopic maps differ by a coboundary in the
Hom-complex.

\section{Matrices}
For finite free complexes, a chain homotopy is represented by matrices
\(S_n\) satisfying
\[
G_n-F_n=D^D_{n+1}S_n+S_{n-1}D^C_n.
\]
This converts chain-homotopy questions into linear Diophantine equations.

\section{Direct sums}
Chain homotopies add componentwise under direct sums.

\section{Tensor-product preview}
For tensor products, chain homotopies interact with Koszul signs.  The full
product machinery is deferred to VIII/27.

\section{Relative chains}
A homotopy of maps of pairs produces a chain homotopy on relative chain
complexes because the prism operator preserves the distinguished subcomplex.

\section{Augmented complexes}
For augmented complexes, chain homotopies may be required to respect the
augmentation.  This is useful when reduced homology is the natural invariant.

\section{Uniqueness up to chain homotopy}
Many chain-level constructions are not canonical on the nose but are canonical
up to chain homotopy.  Since homology kills this ambiguity, homology maps remain
canonical.

\section{Why this notion matters}
Chain homotopy is the correct algebraic equivalence because it is:
\begin{itemize}
\item weaker than equality of chain maps;
\item strong enough to force equality on homology;
\item stable under composition;
\item produced naturally by topological homotopies.
\end{itemize}

\section{Bridge to contractions}
A chain contraction is precisely a chain homotopy
\[
\operatorname{id}_C\simeq0.
\]
VIII/23 studies this special but powerful case and its relation to acyclic
complexes.

\section{Solved problem dossiers}

\begin{problem}[Cycle argument]\label{prob:viii22-01}
Prove directly that chain-homotopic maps induce equal homology maps.
\end{problem}
\begin{solution}
For a cycle \(z\), \(g(z)-f(z)=\partial s(z)\), so their classes agree.
\end{solution}

\begin{problem}[Equivalence relation]\label{prob:viii22-02}
Verify reflexivity, symmetry and transitivity.
\end{problem}
\begin{solution}
Use \(0\), \(-s\), and \(s+t\), respectively.
\end{solution}

\begin{problem}[Postcomposition]\label{prob:viii22-03}
If \(f\simeq g\) and \(u\) is a chain map, find a homotopy from \(uf\) to \(ug\).
\end{problem}
\begin{solution}
Use \(us\); the chain-map identity moves \(u\) past the differential.
\end{solution}

\begin{problem}[Precomposition]\label{prob:viii22-04}
If \(f\simeq g\) and \(v\) is a chain map, find a homotopy from \(fv\) to \(gv\).
\end{problem}
\begin{solution}
Use \(sv\).
\end{solution}

\begin{problem}[Null map]\label{prob:viii22-05}
Show a null-homotopic map induces zero in homology.
\end{problem}
\begin{solution}
It is chain-homotopic to the zero chain map, so the induced maps agree.
\end{solution}

\begin{problem}[Homotopy equivalence]\label{prob:viii22-06}
Show chain-homotopy equivalence implies homology isomorphism.
\end{problem}
\begin{solution}
Apply homology to \(gf\simeq\mathrm{id}\) and \(fg\simeq\mathrm{id}\).
\end{solution}

\begin{problem}[Strict isomorphism]\label{prob:viii22-07}
Why is every chain isomorphism a chain-homotopy equivalence?
\end{problem}
\begin{solution}
Use its strict inverse; the composites equal the identities, hence are homotopic via zero.
\end{solution}

\begin{problem}[Prism]\label{prob:viii22-08}
Interpret the prism identity algebraically.
\end{problem}
\begin{solution}
It says the singular chain maps induced by homotopic topological maps differ by \(\partial P+P\partial\).
\end{solution}

\begin{problem}[Relative prism]\label{prob:viii22-09}
Why does the prism operator descend to relative chains for a homotopy of pairs?
\end{problem}
\begin{solution}
Prisms on chains lying in the subspace remain in the target subspace, so the operator respects the quotient.
\end{solution}

\begin{problem}[Matrix criterion]\label{prob:viii22-10}
State the matrix equation for a chain homotopy.
\end{problem}
\begin{solution}
\(G_n-F_n=D^D_{n+1}S_n+S_{n-1}D^C_n\).
\end{solution}

\begin{problem}[Zero differential]\label{prob:viii22-11}
If both complexes have zero differentials, when are two chain maps chain homotopic?
\end{problem}
\begin{solution}
The homotopy equation becomes \(g-f=0\), so exactly when the maps are equal.
\end{solution}

\begin{problem}[Concentrated complex]\label{prob:viii22-12}
Let \(C\) be concentrated in one degree. What does chain homotopy between endomorphisms reduce to?
\end{problem}
\begin{solution}
Again all differentials vanish, so homotopy is equality.
\end{solution}

\begin{problem}[Two-term example]\label{prob:viii22-13}
Explain why nontrivial chain homotopies can occur only when adjacent target degrees exist.
\end{problem}
\begin{solution}
The operator \(s_n:C_n\to D_{n+1}\) must have somewhere to land; if all \(D_{n+1}=0\), that component vanishes.
\end{solution}

\begin{problem}[Hom-complex]\label{prob:viii22-14}
Explain \(g-f=\delta(s)\).
\end{problem}
\begin{solution}
The differential in the Hom-complex is \(\delta(s)=\partial_Ds+s\partial_C\) for degree one.
\end{solution}

\begin{problem}[Canonical up to homotopy]\label{prob:viii22-15}
Why can a chain-level construction still define a canonical homology map if choices are chain homotopic?
\end{problem}
\begin{solution}
All choices induce the same map on homology.
\end{solution}

\begin{problem}[Direct sums]\label{prob:viii22-16}
Combine homotopies \(s^\alpha:f^\alpha\simeq g^\alpha\).
\end{problem}
\begin{solution}
The direct sum \(\bigoplus s^\alpha\) satisfies the homotopy identity componentwise.
\end{solution}

\begin{problem}[Topological composition]\label{prob:viii22-17}
Relate postcomposition of a homotopy \(F\) to chain homotopy.
\end{problem}
\begin{solution}
Postcomposing \(F\) by a continuous map corresponds to postcomposing its prism chain homotopy by the induced chain map.
\end{solution}

\begin{problem}[Augmentation]\label{prob:viii22-18}
What extra condition is natural for augmented chain homotopies?
\end{problem}
\begin{solution}
They should be compatible with the augmentation so the homotopy descends correctly to reduced complexes.
\end{solution}

\begin{problem}[Homology cannot detect]\label{prob:viii22-19}
Can homology distinguish chain maps that are chain homotopic?
\end{problem}
\begin{solution}
No; by definition of chain-homotopy invariance, their induced maps are identical.
\end{solution}

\begin{problem}[Bridge]\label{prob:viii22-20}
Why is a chain contraction a special case?
\end{problem}
\begin{solution}
It is a chain homotopy from \(\mathrm{id}_C\) to \(0\), i.e. \(\mathrm{id}_C=\partial s+s\partial\).
\end{solution}

\begin{problem}[Two streams]\label{prob:viii22-21}
Why is this chapter broader than VIII/20's prism proof?
\end{problem}
\begin{solution}
It treats chain homotopy abstractly for arbitrary chain complexes, not only singular chains arising from topological homotopies.
\end{solution}

\begin{problem}[Coefficient stability]\label{prob:viii22-22}
Why can chain-homotopy identities be transported after coefficient change?
\end{problem}
\begin{solution}
Applying an additive coefficient functor preserves sums and compositions in the defining identity, subject to the usual tensor exactness caveats for later homology calculations.
\end{solution}

\begin{problem}[Naturality]\label{prob:viii22-23}
Explain why chain homotopy is compatible with functorial constructions.
\end{problem}
\begin{solution}
Because the identity \(g-f=\partial s+s\partial\) is preserved by additive functors that commute with the relevant differentials.
\end{solution}

\begin{problem}[Final principle]\label{prob:viii22-24}
State the main invariant content.
\end{problem}
\begin{solution}
Chain-homotopy classes of chain maps, not raw chain maps, are what homology sees.
\end{solution}
