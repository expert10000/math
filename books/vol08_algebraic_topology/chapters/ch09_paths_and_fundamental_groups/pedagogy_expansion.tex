\section{Supplementary worked examples}
\label{sec:viii09-pedagogy}

\begin{example}[Fundamental group of the circle]\label{ex:viii09-ped-01}
Loops $\gamma_k(t)=e^{2\pi i kt}$ represent all classes in $\pi_1(S^1,1)$, and concatenation adds winding numbers.  Thus $\pi_1(S^1)\cong\mathbb Z$.
\end{example}

\begin{example}[Product loops]\label{ex:viii09-ped-02}
A loop in $X\times Y$ is a pair of loops, and endpoint-fixing homotopies act coordinatewise.  Consequently $\pi_1(X\times Y,(x_0,y_0))\cong\pi_1(X,x_0)\times\pi_1(Y,y_0)$.
\end{example}

\begin{example}[Simply connected disk]\label{ex:viii09-ped-03}
Every loop in $D^n$ contracts by the straight-line contraction of the disk.  Hence $\pi_1(D^n)=0$, illustrating how contractibility forces trivial fundamental group.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Loop endpoints]\label{exr:viii09-ped-01}
What endpoint conditions define a loop based at $x_0$?
\end{exercise}
\begin{hint}
Both ends of $I$ map to the base point.
\end{hint}
\begin{solution}
$\gamma(0)=\gamma(1)=x_0$.
\end{solution}

\begin{exercise}[Constant loop]\label{exr:viii09-ped-02}
Write the identity element of $\pi_1(X,x_0)$.
\end{exercise}
\begin{hint}
Use the constant path.
\end{hint}
\begin{solution}
It is the class of $c_{x_0}(t)=x_0$.
\end{solution}

\begin{exercise}[Inverse loop]\label{exr:viii09-ped-03}
Given $\gamma$, write a representative of its inverse class.
\end{exercise}
\begin{hint}
Reverse time.
\end{hint}
\begin{solution}
$\overline\gamma(t)=\gamma(1-t)$ represents $[\gamma]^{-1}$.
\end{solution}

\begin{exercise}[Circle product]\label{exr:viii09-ped-04}
In $\pi_1(S^1)\cong\mathbb Z$, what class is $[\gamma_2]*[\gamma_{-5}]$?
\end{exercise}
\begin{hint}
Concatenation adds winding numbers.
\end{hint}
\begin{solution}
It corresponds to $2+(-5)=-3$.
\end{solution}

\begin{exercise}[Torus group]\label{exr:viii09-ped-05}
Compute $\pi_1(S^1\times S^1)$.
\end{exercise}
\begin{hint}
Use the product theorem.
\end{hint}
\begin{solution}
It is $\mathbb Z\times\mathbb Z$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Well-defined product]\label{exr:viii09-ped-06}
Prove loop concatenation descends to endpoint-homotopy classes.
\end{exercise}
\begin{hint}
Concatenate homotopies in the path variable.
\end{hint}
\begin{solution}
If $\alpha\simeq\alpha'$ and $\beta\simeq\beta'$ rel endpoints, piece the homotopies on the two halves of the interval; this gives $\alpha*\beta\simeq\alpha'*\beta'$ rel endpoints.
\end{solution}

\begin{exercise}[Group inverse]\label{exr:viii09-ped-07}
Prove $[\gamma][\overline\gamma]=1$ in $\pi_1$.
\end{exercise}
\begin{hint}
Shrink the out-and-back path to the base point.
\end{hint}
\begin{solution}
There is an endpoint-fixed homotopy that progressively shortens the traversal of $\gamma$ followed by its reverse, yielding the constant loop.
\end{solution}

\begin{exercise}[Functoriality]\label{exr:viii09-ped-08}
Prove a based map $f:(X,x_0)\to(Y,y_0)$ induces a homomorphism $f_*:\pi_1(X,x_0)\to\pi_1(Y,y_0)$.
\end{exercise}
\begin{hint}
Postcompose loops and homotopies with $f$.
\end{hint}
\begin{solution}
Define $f_*[\gamma]=[f\circ\gamma]$. It is well defined by homotopy compatibility and respects concatenation because $f\circ(\alpha*\beta)=(f\circ\alpha)*(f\circ\beta)$ up to the same parameterization.
\end{solution}

\begin{exercise}[Contractible group]\label{exr:viii09-ped-09}
Prove a contractible path-connected space has trivial fundamental group.
\end{exercise}
\begin{hint}
Use the contraction to homotope every loop to the constant loop.
\end{hint}
\begin{solution}
Composing a loop with the contraction gives a free contraction; after arranging the based condition (or using a contraction fixing the chosen point when available and change-of-basepoint), every class is trivial.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Fundamental group always abelian]\label{exr:viii09-ped-10}
Test: $\pi_1(X)$ is always abelian.
\end{exercise}
\begin{hint}
Wedges of circles give free groups.
\end{hint}
\begin{solution}
False.  Fundamental groups can be highly nonabelian.
\end{solution}

\begin{exercise}[Free homotopy defines group]\label{exr:viii09-ped-11}
Test: the fundamental group uses free homotopy classes of loops.
\end{exercise}
\begin{hint}
The base point must remain fixed.
\end{hint}
\begin{solution}
False. It uses homotopies relative to the endpoints/base point.
\end{solution}

\begin{exercise}[Homeomorphism required]\label{exr:viii09-ped-12}
Test: homotopy equivalent path-connected spaces can have different fundamental groups.
\end{exercise}
\begin{hint}
$\pi_1$ is a homotopy invariant up to basepoint transport.
\end{hint}
\begin{solution}
False in the based/isomorphism sense: a homotopy equivalence induces an isomorphism of fundamental groups.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Obstacle detection]\label{exr:viii09-ped-13}
Why does $\pi_1$ detect holes relevant to path planning?
\end{exercise}
\begin{hint}
Nonhomotopic loops cannot be continuously deformed through admissible paths.
\end{hint}
\begin{solution}
Distinct loop classes encode qualitatively different ways of winding around obstacles, so the group organizes path-homotopy constraints ignored by metric distance alone.
\end{solution}

\begin{exercise}[Presentation from CW structure]\label{exr:viii09-ped-14}
How does a 2-dimensional CW complex lead toward a presentation of $\pi_1$?
\end{exercise}
\begin{hint}
1-cells generate loops and 2-cell boundaries impose relations.
\end{hint}
\begin{solution}
After choosing a spanning tree/basepoint, remaining 1-cells supply generators; each 2-cell attaching loop becomes a relation via van Kampen.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Figure-eight preview]\label{exr:viii09-ped-15}
Predict the fundamental group of a wedge of two circles.
\end{exercise}
\begin{hint}
No 2-cell imposes a relation between the two basic loops.
\end{hint}
\begin{solution}
It is the free group $F_2$ on the two circle loops.
\end{solution}

\begin{exercise}[Idempotent endomorphism]\label{exr:viii09-ped-16}
If a based self-map $f:S^1\to S^1$ satisfies $f\circ f\simeq f$ based, what winding numbers are possible?
\end{exercise}
\begin{hint}
The induced map on $\pi_1(S^1)=\mathbb Z$ is multiplication by an integer $d$.
\end{hint}
\begin{solution}
The relation gives $d^2=d$, hence $d=0$ or $1$.
\end{solution}
