\chapter{Homotopy Invariance}
\label{ch:viii20-homotopy-invariance}

Homology is useful only if it ignores deformations that preserve homotopy type.
The key theorem is that homotopic maps induce the same homomorphism on homology.
At chain level, the difference between the two induced chain maps is a boundary
commutator.  For singular chains this is implemented by the prism operator.

\section{Chain homotopies}

Let
\[
f_\bullet,g_\bullet:C_\bullet\to D_\bullet
\]
be chain maps.

\begin{definition}[Chain homotopy]
\label{def:viii20-chain-homotopy}
A chain homotopy from \(f\) to \(g\) is a family of homomorphisms
\[
P_n:C_n\to D_{n+1}
\]
such that
\[
g_n-f_n
=
\partial_{n+1}P_n+P_{n-1}\partial_n.
\]
\end{definition}

The operator \(P\) has degree \(+1\).

\section{Chain-homotopic maps induce equal homology maps}

\begin{theorem}[Chain-homotopy invariance]
\label{thm:viii20-chain-homotopy-invariance}
If chain maps \(f,g:C_\bullet\to D_\bullet\) are chain homotopic, then
\[
H_n(f)=H_n(g)
\]
for every \(n\).
\end{theorem}

\begin{proof}
If \(z\) is a cycle, then
\[
g(z)-f(z)
=
\partial P(z)+P(\partial z)
=
\partial P(z),
\]
so \(f(z)\) and \(g(z)\) differ by a boundary and represent the same homology
class.
\end{proof}

\section{Topological homotopies}

Suppose
\[
f,g:X\to Y
\]
are homotopic through
\[
H:X\times I\to Y.
\]

The goal is to construct a chain homotopy between the induced singular chain
maps
\[
f_\#,g_\#:C_*^{\rm sing}(X)\to C_*^{\rm sing}(Y).
\]

\section{Prisms}

For a singular simplex
\[
\sigma:\Delta^n\to X,
\]
the homotopy gives a map
\[
\Delta^n\times I\to Y.
\]

The prism \(\Delta^n\times I\) can be subdivided into \(n+1\)
\((n+1)\)-simplices.

\begin{definition}[Prism operator]
\label{def:viii20-prism}
The prism operator
\[
P_n:C_n(X)\to C_{n+1}(Y)
\]
is the signed sum of the singular \((n+1)\)-simplices obtained by this
standard prism subdivision followed by \(H\circ(\sigma\times\operatorname{id})\).
\end{definition}

\section{Prism identity}

\begin{theorem}[Prism identity]
\label{thm:viii20-prism-identity}
The prism operator satisfies
\[
g_\#-f_\#
=
\partial P+P\partial.
\]
\end{theorem}

Thus \(P\) is a chain homotopy between \(f_\#\) and \(g_\#\).

\section{Homotopy invariance theorem}

\begin{theorem}[Homotopic maps induce the same map]
\label{thm:viii20-homotopy-invariance}
If
\[
f\simeq g:X\to Y,
\]
then
\[
f_*=g_*:H_n(X)\to H_n(Y)
\]
for every \(n\).
\end{theorem}

\section{Homotopy equivalence}

\begin{corollary}[Homotopy equivalent spaces have isomorphic homology]
\label{cor:viii20-homotopy-equivalence}
If
\[
X\simeq Y,
\]
then
\[
H_n(X)\cong H_n(Y)
\]
for all \(n\).
\end{corollary}

Indeed, if \(f:X\to Y\) and \(g:Y\to X\) are homotopy inverses, then
\[
g_*f_*=(g\circ f)_*=(\operatorname{id}_X)_*,
\]
and similarly \(f_*g_*=\operatorname{id}\).

\section{Contractible spaces}

\begin{corollary}[Homology of a contractible space]
\label{cor:viii20-contractible}
If \(X\) is nonempty and contractible, then
\[
H_0(X)\cong\mathbb Z,
\qquad
H_n(X)=0\quad(n>0).
\]
Equivalently,
\[
\widetilde H_n(X)=0
\]
for every \(n\).
\end{corollary}

\section{Deformation retracts}

If \(A\subset X\) is a deformation retract, then
\[
A\simeq X.
\]

\begin{corollary}[Deformation retract]
\label{cor:viii20-deformation-retract}
The inclusion
\[
A\hookrightarrow X
\]
induces homology isomorphisms in every degree.
\end{corollary}

\section{Strong deformation retracts}

A strong deformation retract gives the same conclusion, with the homotopy fixed
on \(A\).  The additional relative control is important in pairs and CW
constructions but does not alter the absolute homology conclusion.

\section{Retracts and split maps}

If
\[
r:X\to A,
\qquad
i:A\hookrightarrow X,
\qquad
r i=\operatorname{id}_A,
\]
then
\[
r_* i_*=\operatorname{id}_{H_*(A)}.
\]

Thus \(i_*\) is injective and \(r_*\) is surjective even without assuming a
deformation retraction.

\section{Null-homotopic maps}

If
\[
f:X\to Y
\]
is null-homotopic, then on reduced homology
\[
\widetilde f_*=0.
\]

For positive degrees,
\[
f_*:H_n(X)\to H_n(Y)
\]
is zero.

\section{Degree and homotopy}

For maps
\[
f,g:S^n\to S^n,
\]
homotopy invariance of top homology gives
\[
f\simeq g
\quad\Longrightarrow\quad
\deg f=\deg g.
\]

This recovers the degree obstruction from VIII/03 inside homology.

\section{No-retraction theorem via homology}

Suppose
\[
r:D^{n+1}\to S^n
\]
were a retraction.  Then inclusion
\[
i:S^n\hookrightarrow D^{n+1}
\]
would satisfy
\[
r_*i_*=\operatorname{id}_{H_n(S^n)}.
\]

But
\[
H_n(D^{n+1})=0,
\]
so \(i_*=0\), contradiction.

\begin{theorem}[Homological no-retraction]
\label{thm:viii20-no-retraction}
There is no retraction
\[
D^{n+1}\to S^n.
\]
\end{theorem}

\section{Homology detects noncontractibility}

If a connected space has
\[
\widetilde H_n(X)\ne0
\]
for some \(n\), then \(X\) is not contractible.

Thus:
\[
S^n,\quad T^n,\quad \mathbb RP^{2k+1}
\]
and many other spaces are immediately seen to be noncontractible.

\section{Homology is not complete}

The converse is false: spaces can have the same homology groups without being
homotopy equivalent.

Homology is a powerful but coarse invariant.  It is abelian and loses much
noncommutative information visible to the fundamental group.

\section{Acyclic does not mean contractible}

A space is acyclic if its reduced homology vanishes.  Acyclic spaces need not
be contractible.

Thus
\[
\widetilde H_*(X)=0
\]
is necessary but not sufficient for contractibility.

\section{Homotopy invariance for pairs}

If maps of pairs
\[
f,g:(X,A)\to(Y,B)
\]
are homotopic through maps of pairs, then
\[
f_*=g_*:H_n(X,A)\to H_n(Y,B).
\]

The prism construction descends to relative quotient chain complexes.

\section{Naturality with exact sequences}

Homotopy invariance is compatible with the long exact sequence of a pair.
Homotopic maps of pairs induce the same morphism of long exact sequences.

\section{Cellular consequence}

If two CW complexes are homotopy equivalent, cellular homology calculations
for either must give isomorphic homology groups.  This justifies simplifying
cell structures through homotopy-preserving operations before computation.

\section{Chain-homotopy viewpoint}

The topological statement
\[
f\simeq g
\]
is converted into the algebraic statement
\[
f_\#\simeq_{\rm ch} g_\#.
\]

This is the first major instance of a recurring principle:
\[
\text{topological deformation}
\quad\leadsto\quad
\text{algebraic chain homotopy}.
\]

VIII/22 develops chain homotopies as algebraic objects in their own right.

\section{Bridge to Euler characteristic}

For finite CW complexes, homology ranks determine Euler characteristic through
the Euler--Poincare formula.  Since homology is homotopy invariant, Euler
characteristic becomes a homotopy invariant as well.  VIII/21 develops this
systematically.

\section{Solved problem dossiers}

\begin{problem}[Cycle proof]\label{prob:viii20-01}
Prove chain-homotopic maps induce equal maps on homology.
\end{problem}
\begin{solution}
If \(z\) is a cycle, \(g(z)-f(z)=\partial P(z)+P\partial z=\partial P(z)\), so the two images differ by a boundary.
\end{solution}

\begin{problem}[Homotopy inverse]\label{prob:viii20-02}
Let \(f:X\to Y\), \(g:Y\to X\) be homotopy inverses. Prove \(f_*\) is an isomorphism.
\end{problem}
\begin{solution}
Homotopy invariance gives \(g_*f_*=(gf)_*=(\operatorname{id}_X)_*\) and \(f_*g_*=\operatorname{id}_{H_*(Y)}\), so \(g_*\) is the inverse.
\end{solution}

\begin{problem}[Contractible disk]\label{prob:viii20-03}
Compute the homology of \(D^n\) from homotopy invariance.
\end{problem}
\begin{solution}
\(D^n\simeq *\), so \(H_0(D^n)=\mathbb Z\) and all positive homology vanishes.
\end{solution}

\begin{problem}[Deformation retract annulus]\label{prob:viii20-04}
Compute the homology of an annulus by deformation retracting to its core circle.
\end{problem}
\begin{solution}
The annulus is homotopy equivalent to \(S^1\), so \(H_0=\mathbb Z,\ H_1=\mathbb Z\), and higher groups vanish.
\end{solution}

\begin{problem}[Punctured plane]\label{prob:viii20-05}
Compute \(H_*(\mathbb R^2\setminus\{0\})\).
\end{problem}
\begin{solution}
Radial deformation retract gives \(S^1\), hence \(H_0=\mathbb Z,\ H_1=\mathbb Z\), higher groups zero.
\end{solution}

\begin{problem}[Punctured Euclidean space]\label{prob:viii20-06}
Compute \(H_*(\mathbb R^{n+1}\setminus\{0\})\).
\end{problem}
\begin{solution}
It deformation retracts onto \(S^n\), so homology is \(\mathbb Z\) in degrees \(0,n\) and zero otherwise.
\end{solution}

\begin{problem}[No retraction]\label{prob:viii20-07}
Give the homological proof that \(D^{n+1}\) does not retract onto \(S^n\).
\end{problem}
\begin{solution}
A retraction would give \(r_*i_*=\operatorname{id}\) on \(H_n(S^n)=\mathbb Z\), but \(i_*\) factors through \(H_n(D^{n+1})=0\), contradiction.
\end{solution}

\begin{problem}[Null map]\label{prob:viii20-08}
If \(f:S^n\to Y\) is null-homotopic, what is \(f_*\) on \(H_n\) for \(n>0\)?
\end{problem}
\begin{solution}
Zero, because \(f\) is homotopic to a constant map and positive homology of a point vanishes.
\end{solution}

\begin{problem}[Degree obstruction]\label{prob:viii20-09}
Why can maps \(S^n\to S^n\) of different degree not be homotopic?
\end{problem}
\begin{solution}
The induced map on \(H_n(S^n)\cong\mathbb Z\) is multiplication by degree. Homotopic maps induce the same homology map.
\end{solution}

\begin{problem}[Antipodal even sphere]\label{prob:viii20-10}
Use homology to distinguish the antipodal map from the identity on even \(S^n\).
\end{problem}
\begin{solution}
The identity induces multiplication by \(1\) on \(H_n\); the antipodal map has degree \(-1\) for even \(n\), so the induced maps differ.
\end{solution}

\begin{problem}[Retract splitting]\label{prob:viii20-11}
Suppose \(A\) is a retract of \(X\). What algebraic information follows on homology?
\end{problem}
\begin{solution}
The inclusion-induced map is split injective and the retraction-induced map is split surjective; \(H_*(A)\) occurs as a direct summand in the appropriate algebraic sense.
\end{solution}

\begin{problem}[Noncontractible torus]\label{prob:viii20-12}
Why is \(T^2\) not contractible?
\end{problem}
\begin{solution}
\(H_1(T^2)\cong\mathbb Z^2\ne0\), while a contractible space has zero positive homology.
\end{solution}

\begin{problem}[Projective noncontractibility]\label{prob:viii20-13}
Why is \(\mathbb RP^2\) not contractible?
\end{problem}
\begin{solution}
\(H_1(\mathbb RP^2)\cong\mathbb Z/2\ne0\).
\end{solution}

\begin{problem}[Acyclic warning]\label{prob:viii20-14}
Why can vanishing reduced homology not prove contractibility?
\end{problem}
\begin{solution}
Homology forgets nonabelian and higher homotopy information; there exist noncontractible acyclic spaces.
\end{solution}

\begin{problem}[Relative deformation]\label{prob:viii20-15}
If \((X,A)\) deformation retracts as a pair onto \((Y,B)\), what happens to relative homology?
\end{problem}
\begin{solution}
The pair maps are homotopy equivalences through maps of pairs, so they induce isomorphisms \(H_n(X,A)\cong H_n(Y,B)\).
\end{solution}

\begin{problem}[Prism in dimension zero]\label{prob:viii20-16}
What does the prism operator do to a singular 0-simplex \(x\)?
\end{problem}
\begin{solution}
The homotopy path \(t\mapsto H(x,t)\) is a singular 1-simplex whose boundary is \(g(x)-f(x)\), exactly the degree-zero prism identity.
\end{solution}

\begin{problem}[Prism boundary meaning]\label{prob:viii20-17}
Geometrically, what are the endpoint terms in the boundary of a prism?
\end{problem}
\begin{solution}
The top and bottom faces are the singular simplices obtained from \(g\circ\sigma\) and \(f\circ\sigma\); the side faces combine into \(P(\partial\sigma)\).
\end{solution}

\begin{problem}[Identity homotopy]\label{prob:viii20-18}
What chain homotopy is needed between \(\operatorname{id}_\#\) and itself?
\end{problem}
\begin{solution}
The zero operator \(P=0\) satisfies \(\operatorname{id}-\operatorname{id}=0=\partial P+P\partial\).
\end{solution}

\begin{problem}[Composition with homotopy]\label{prob:viii20-19}
If \(f_0\simeq f_1:X\to Y\) and \(h:Y\to Z\), compare \(hf_0\) and \(hf_1\) on homology.
\end{problem}
\begin{solution}
They are homotopic, so \((hf_0)_*=(hf_1)_*\); equivalently \(h_*f_{0*}=h_*f_{1*}\).
\end{solution}

\begin{problem}[Precomposition]\label{prob:viii20-20}
If \(f_0\simeq f_1:X\to Y\) and \(k:W\to X\), compare \(f_0k\) and \(f_1k\).
\end{problem}
\begin{solution}
They are homotopic via \(H(k(w),t)\), hence induce the same homology map.
\end{solution}

\begin{problem}[Euler consequence]\label{prob:viii20-21}
Why must finite CW complexes of different Euler characteristic fail to be homotopy equivalent?
\end{problem}
\begin{solution}
Euler characteristic equals the alternating sum of homology ranks and is therefore preserved by homotopy equivalence.
\end{solution}
