
\chapter{Lefschetz Theory}
\label{ch:viii35-lefschetz-theory}

Lefschetz theory is a synthesis theorem.  A self-map acts linearly on homology,
fixed points appear geometrically as intersections of a graph with a diagonal,
and Poincaré duality proves that the two counts agree.

\section{The Lefschetz number}
Let $X$ be a finite CW complex and $f:X\to X$ a continuous map.

\begin{definition}[Lefschetz number]
\label{def:viii35-lefschetz-number}
With coefficients in a field such as $\mathbb Q$,
\[
L(f)
=
\sum_{k\ge0}
(-1)^k
\operatorname{tr}
\bigl(f_*:H_k(X;\mathbb Q)\to H_k(X;\mathbb Q)\bigr).
\]
\end{definition}

\section{Homotopy invariance}
If $f\simeq g$, then $f_*=g_*$ on homology, hence
\[
L(f)=L(g).
\]

\section{The fixed-point theorem}
\begin{theorem}[Lefschetz fixed-point theorem]
\label{thm:viii35-lefschetz-fixed-point}
For a self-map of a suitable compact triangulable space, if
\[
L(f)\ne0,
\]
then $f$ has a fixed point.
\end{theorem}

The converse is false: fixed points can occur with total index zero.

\section{The identity map}
For the identity,
\[
L(\operatorname{id}_X)
=
\sum_k(-1)^k\dim H_k(X;\mathbb Q)
=
\chi(X).
\]
Thus Euler characteristic is the Lefschetz number of the identity.

\section{Contractible spaces and Brouwer}
If $X$ is connected and contractible, only $H_0$ contributes:
\[
L(f)=1.
\]
Applied to $D^n$, this gives Brouwer's fixed-point theorem.

\section{Sphere maps}
If $f:S^n\to S^n$ has degree $d$, then
\[
L(f)=1+(-1)^n d.
\]
Whenever this number is nonzero, $f$ must have a fixed point.

\section{Constant maps}
A constant self-map of a connected finite CW complex induces the identity on
$H_0$ and zero on positive-dimensional reduced homology, so
\[
L(f)=1.
\]

\section{The graph and diagonal}
For a self-map $f:M\to M$ of a closed manifold, define
\[
\Gamma_f=\{(x,f(x)):x\in M\},
\qquad
\Delta=\{(x,x):x\in M\}.
\]
Then
\[
x\in\operatorname{Fix}(f)
\quad\Longleftrightarrow\quad
(x,f(x))\in\Gamma_f\cap\Delta.
\]

\section{Intersection-theoretic meaning}
Under the usual transversality and orientation hypotheses,
\[
L(f)=\Gamma_f\cdot\Delta.
\]
This is the geometric bridge from VIII/34 to the homological trace formula.

\section{Local fixed-point index}
If a smooth fixed point $x$ is nondegenerate, so
\[
\det(I-Df_x)\ne0,
\]
its local index is
\[
\operatorname{ind}_x(f)=\operatorname{sign}\det(I-Df_x).
\]

\begin{theorem}[Lefschetz--Hopf]
\label{thm:viii35-lefschetz-hopf}
When the fixed points are isolated and the standard hypotheses hold,
\[
L(f)=\sum_{x\in\operatorname{Fix}(f)}\operatorname{ind}_x(f).
\]
\end{theorem}

\section{Cancellation}
The theorem only says that a nonzero Lefschetz number forces a fixed point.
If positive and negative local indices cancel, fixed points can exist while
\[
L(f)=0.
\]

\section{Torus endomorphisms}
A linear torus map induced by
\[
A\in M_2(\mathbb Z)
\]
acts by $1$ on $H_0$, by $A$ on $H_1$, and by $\det A$ on $H_2$.  Therefore
\[
L(f)=1-\operatorname{tr}(A)+\det(A)
=\det(I-A).
\]

\section{A fixed-point criterion on the torus}
If
\[
\det(I-A)\ne0,
\]
then the induced torus map has nonzero Lefschetz number and therefore a fixed
point.

\section{Poincaré-duality proof philosophy}
The diagonal class in $M\times M$ is expressed using dual bases in homology
and cohomology.  Pulling it against the graph of $f$ converts geometric
intersection into the alternating trace of $f_*$.

\section{Supertrace viewpoint}
The Lefschetz number is the supertrace of the graded homology endomorphism:
\[
L(f)=\operatorname{str}(f_*).
\]
This compactly packages the alternating signs.

\section{Endpoint of the canonical arc}
The chapter closes the Volume VIII progression
\[

\section{Solved dossiers}
\begin{problem}
Define the Lefschetz number of a self-map of a finite CW complex.
\end{problem}
\begin{solution}
$L(f)=\sum_k(-1)^k\operatorname{tr}(f_*:H_k(X;\mathbb Q)\to H_k(X;\mathbb Q))$.
\end{solution}

\begin{problem}
Why may rational homology be used in the basic trace formula?
\end{problem}
\begin{solution}
It removes torsion from the finite-dimensional trace calculation while preserving the classical Lefschetz number.
\end{solution}

\begin{problem}
State the Lefschetz fixed-point theorem.
\end{problem}
\begin{solution}
If $L(f)\ne0$ for a self-map of a suitable compact triangulable space, then $f$ has a fixed point.
\end{solution}

\begin{problem}
What is $L(\operatorname{id}_X)$?
\end{problem}
\begin{solution}
$\chi(X)$.
\end{solution}

\begin{problem}
How does Lefschetz imply Brouwer for a map $D^n\to D^n$?
\end{problem}
\begin{solution}
$D^n$ is contractible, so only $H_0$ contributes and $L(f)=1\ne0$.
\end{solution}

\begin{problem}
Compute $L(f)$ for $f:S^n\to S^n$ of degree $d$.
\end{problem}
\begin{solution}
$L(f)=1+(-1)^n d$.
\end{solution}

\begin{problem}
What condition on the degree of a sphere map guarantees a fixed point?
\end{problem}
\begin{solution}
$1+(-1)^n d\ne0$.
\end{solution}

\begin{problem}
What is the Lefschetz number of a constant self-map of a connected finite complex?
\end{problem}
\begin{solution}
$1$.
\end{solution}

\begin{problem}
Why is the converse of the Lefschetz theorem false?
\end{problem}
\begin{solution}
A map can have fixed points whose local indices cancel, so $L(f)=0$ does not imply fixed-point-free.
\end{solution}

\begin{problem}
Why is $L(f)$ homotopy invariant?
\end{problem}
\begin{solution}
Homotopic maps induce the same homomorphisms on homology and therefore the same traces.
\end{solution}

\begin{problem}
Describe fixed points using graph and diagonal.
\end{problem}
\begin{solution}
$x$ is fixed exactly when $(x,f(x))$ lies in both $\Gamma_f$ and $\Delta$.
\end{solution}

\begin{problem}
What does the global Lefschetz number measure geometrically?
\end{problem}
\begin{solution}
Under transversality/nondegeneracy hypotheses it is the signed intersection number $\Gamma_f\cdot\Delta$.
\end{solution}

\begin{problem}
What is a local fixed-point index in the nondegenerate smooth case?
\end{problem}
\begin{solution}
It is the local intersection sign; equivalently the sign of $\det(I-Df_x)$ when that determinant is nonzero.
\end{solution}

\begin{problem}
State the Lefschetz--Hopf refinement.
\end{problem}
\begin{solution}
For isolated fixed points satisfying the usual hypotheses, $L(f)$ is the sum of their local fixed-point indices.
\end{solution}

\begin{problem}
Compute the Lefschetz number of a torus map induced by $A\in M_2(\mathbb Z)$.
\end{problem}
\begin{solution}
$L(f)=1-\operatorname{tr}(A)+\det(A)=\det(I-A)$.
\end{solution}

\begin{problem}
What does $\det(I-A)\ne0$ imply for the induced torus map?
\end{problem}
\begin{solution}
Its Lefschetz number is nonzero, so it has a fixed point.
\end{solution}

\begin{problem}
What happens for a fixed-point-free map?
\end{problem}
\begin{solution}
Necessarily $L(f)=0$.
\end{solution}

\begin{problem}
How does the identity map connect Lefschetz theory to Euler characteristic?
\end{problem}
\begin{solution}
Its local/global fixed-point count is encoded by $L(\mathrm{id})=\chi(X)$.
\end{solution}

\begin{problem}
How does Lefschetz theory use Poincaré duality?
\end{problem}
\begin{solution}
On manifolds, duality converts the graph--diagonal intersection class into the alternating trace on homology.
\end{solution}

\begin{problem}
Why is VIII/35 a natural endpoint for Volume VIII?
\end{problem}
\begin{solution}
It synthesizes homology, cohomology, cup/cap products, Poincaré duality and intersection theory into a computable fixed-point invariant.
\end{solution}
