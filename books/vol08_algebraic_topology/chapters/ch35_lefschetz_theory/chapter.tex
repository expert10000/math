
\chapter{Lefschetz Theory}
\label{ch:viii35-lefschetz-theory}

Lefschetz theory is a synthesis theorem.  A self-map acts linearly on homology,
fixed points appear geometrically as intersections of a graph with a diagonal,
and Poincaré duality proves that the two counts agree.

\section{The Lefschetz number}
Let $X$ be a finite CW complex and $f:X\to X$ a continuous map.

\begin{definition}[Lefschetz number]
\label{def:viii35-lefschetz-number}
With coefficients in a field such as $\mathbb Q$,
\[
L(f)
=
\sum_{k\ge0}
(-1)^k
\operatorname{tr}
\bigl(f_*:H_k(X;\mathbb Q)\to H_k(X;\mathbb Q)\bigr).
\]
\end{definition}

\section{Homotopy invariance}
If $f\simeq g$, then $f_*=g_*$ on homology, hence
\[
L(f)=L(g).
\]

\section{The fixed-point theorem}
\begin{theorem}[Lefschetz fixed-point theorem]
\label{thm:viii35-lefschetz-fixed-point}
For a self-map of a suitable compact triangulable space, if
\[
L(f)\ne0,
\]
then $f$ has a fixed point.
\end{theorem}

The converse is false: fixed points can occur with total index zero.

\section{The identity map}
For the identity,
\[
L(\operatorname{id}_X)
=
\sum_k(-1)^k\dim H_k(X;\mathbb Q)
=
\chi(X).
\]
Thus Euler characteristic is the Lefschetz number of the identity.

\section{Contractible spaces and Brouwer}
If $X$ is connected and contractible, only $H_0$ contributes:
\[
L(f)=1.
\]
Applied to $D^n$, this gives Brouwer's fixed-point theorem.

\section{Sphere maps}
If $f:S^n\to S^n$ has degree $d$, then
\[
L(f)=1+(-1)^n d.
\]
Whenever this number is nonzero, $f$ must have a fixed point.

\section{Constant maps}
A constant self-map of a connected finite CW complex induces the identity on
$H_0$ and zero on positive-dimensional reduced homology, so
\[
L(f)=1.
\]

\section{The graph and diagonal}
For a self-map $f:M\to M$ of a closed manifold, define
\[
\Gamma_f=\{(x,f(x)):x\in M\},
\qquad
\Delta=\{(x,x):x\in M\}.
\]
Then
\[
x\in\operatorname{Fix}(f)
\quad\Longleftrightarrow\quad
(x,f(x))\in\Gamma_f\cap\Delta.
\]

\section{Intersection-theoretic meaning}
Under the usual transversality and orientation hypotheses,
\[
L(f)=\Gamma_f\cdot\Delta.
\]
This is the geometric bridge from VIII/34 to the homological trace formula.

\section{Local fixed-point index}
If a smooth fixed point $x$ is nondegenerate, so
\[
\det(I-Df_x)\ne0,
\]
its local index is
\[
\operatorname{ind}_x(f)=\operatorname{sign}\det(I-Df_x).
\]

\begin{theorem}[Lefschetz--Hopf]
\label{thm:viii35-lefschetz-hopf}
When the fixed points are isolated and the standard hypotheses hold,
\[
L(f)=\sum_{x\in\operatorname{Fix}(f)}\operatorname{ind}_x(f).
\]
\end{theorem}

\section{Cancellation}
The theorem only says that a nonzero Lefschetz number forces a fixed point.
If positive and negative local indices cancel, fixed points can exist while
\[
L(f)=0.
\]

\section{Torus endomorphisms}
A linear torus map induced by
\[
A\in M_2(\mathbb Z)
\]
acts by $1$ on $H_0$, by $A$ on $H_1$, and by $\det A$ on $H_2$.  Therefore
\[
L(f)=1-\operatorname{tr}(A)+\det(A)
=\det(I-A).
\]

\section{A fixed-point criterion on the torus}
If
\[
\det(I-A)\ne0,
\]
then the induced torus map has nonzero Lefschetz number and therefore a fixed
point.

\section{Poincaré-duality proof philosophy}
The diagonal class in $M\times M$ is expressed using dual bases in homology
and cohomology.  Pulling it against the graph of $f$ converts geometric
intersection into the alternating trace of $f_*$.

\section{Supertrace viewpoint}
The Lefschetz number is the supertrace of the graded homology endomorphism:
\[
L(f)=\operatorname{str}(f_*).
\]
This compactly packages the alternating signs.

\section{Endpoint of the canonical arc}
The chapter closes the Volume VIII progression
\[
\text{homology}
\to
\text{cohomology}
\to
\text{cup/cap products}
\to
\text{Thom and Euler classes}
\to
\text{Poincaré duality}
\to
\text{intersection}
\to
\text{fixed-point theory}.
\]

\section{Solved dossiers}
\begin{problem}
Define the Lefschetz number of a self-map of a finite CW complex.
\end{problem}
\begin{solution}
$L(f)=\sum_k(-1)^k\operatorname{tr}(f_*:H_k(X;\mathbb Q)\to H_k(X;\mathbb Q))$.
\end{solution}

\begin{problem}
Why may rational homology be used in the basic trace formula?
\end{problem}
\begin{solution}
It removes torsion from the finite-dimensional trace calculation while preserving the classical Lefschetz number.
\end{solution}

\begin{problem}
State the Lefschetz fixed-point theorem.
\end{problem}
\begin{solution}
If $L(f)\ne0$ for a self-map of a suitable compact triangulable space, then $f$ has a fixed point.
\end{solution}

\begin{problem}
What is $L(\operatorname{id}_X)$?
\end{problem}
\begin{solution}
$\chi(X)$.
\end{solution}

\begin{problem}
How does Lefschetz imply Brouwer for a map $D^n\to D^n$?
\end{problem}
\begin{solution}
$D^n$ is contractible, so only $H_0$ contributes and $L(f)=1\ne0$.
\end{solution}

\begin{problem}
Compute $L(f)$ for $f:S^n\to S^n$ of degree $d$.
\end{problem}
\begin{solution}
$L(f)=1+(-1)^n d$.
\end{solution}

\begin{problem}
What condition on the degree of a sphere map guarantees a fixed point?
\end{problem}
\begin{solution}
$1+(-1)^n d\ne0$.
\end{solution}

\begin{problem}
What is the Lefschetz number of a constant self-map of a connected finite complex?
\end{problem}
\begin{solution}
$1$.
\end{solution}

\begin{problem}
Why is the converse of the Lefschetz theorem false?
\end{problem}
\begin{solution}
A map can have fixed points whose local indices cancel, so $L(f)=0$ does not imply fixed-point-free.
\end{solution}

\begin{problem}
Why is $L(f)$ homotopy invariant?
\end{problem}
\begin{solution}
Homotopic maps induce the same homomorphisms on homology and therefore the same traces.
\end{solution}

\begin{problem}
Describe fixed points using graph and diagonal.
\end{problem}
\begin{solution}
$x$ is fixed exactly when $(x,f(x))$ lies in both $\Gamma_f$ and $\Delta$.
\end{solution}

\begin{problem}
What does the global Lefschetz number measure geometrically?
\end{problem}
\begin{solution}
Under transversality/nondegeneracy hypotheses it is the signed intersection number $\Gamma_f\cdot\Delta$.
\end{solution}

\begin{problem}
What is a local fixed-point index in the nondegenerate smooth case?
\end{problem}
\begin{solution}
It is the local intersection sign; equivalently the sign of $\det(I-Df_x)$ when that determinant is nonzero.
\end{solution}

\begin{problem}
State the Lefschetz--Hopf refinement.
\end{problem}
\begin{solution}
For isolated fixed points satisfying the usual hypotheses, $L(f)$ is the sum of their local fixed-point indices.
\end{solution}

\begin{problem}
Compute the Lefschetz number of a torus map induced by $A\in M_2(\mathbb Z)$.
\end{problem}
\begin{solution}
$L(f)=1-\operatorname{tr}(A)+\det(A)=\det(I-A)$.
\end{solution}

\begin{problem}
What does $\det(I-A)\ne0$ imply for the induced torus map?
\end{problem}
\begin{solution}
Its Lefschetz number is nonzero, so it has a fixed point.
\end{solution}

\begin{problem}
What happens for a fixed-point-free map?
\end{problem}
\begin{solution}
Necessarily $L(f)=0$.
\end{solution}

\begin{problem}
How does the identity map connect Lefschetz theory to Euler characteristic?
\end{problem}
\begin{solution}
Its local/global fixed-point count is encoded by $L(\mathrm{id})=\chi(X)$.
\end{solution}

\begin{problem}
How does Lefschetz theory use Poincaré duality?
\end{problem}
\begin{solution}
On manifolds, duality converts the graph--diagonal intersection class into the alternating trace on homology.
\end{solution}

\begin{problem}
Why is VIII/35 a natural endpoint for Volume VIII?
\end{problem}
\begin{solution}
It synthesizes homology, cohomology, cup/cap products, Poincaré duality and intersection theory into a computable fixed-point invariant.
\end{solution}

\section{Exercises, Hints, and Solutions}
\begin{exercise}\label{exr:viii35-01}
Compute $L(\operatorname{id}_{S^n})$.
\end{exercise}
\begin{hint}
Use $L(\mathrm{id})=\chi$.
\end{hint}
\begin{solution}
For the identity, \(L(\mathrm{id}_X)=\chi(X)\). Hence
\[
L(\mathrm{id}_{S^n})=\chi(S^n)=1+(-1)^n,
\]
equal to \(2\) for even \(n\) and \(0\) for odd \(n\).
\end{solution}

\begin{exercise}\label{exr:viii35-02}
Compute $L(f)$ for a degree-$d$ self-map of $S^n$.
\end{exercise}
\begin{hint}
Only $H_0$ and $H_n$ contribute.
\end{hint}
\begin{solution}
Only \(H_0(S^n)\) and \(H_n(S^n)\) are nonzero. The induced maps have traces \(1\) and \(d\), so
\[
L(f)=1+(-1)^n d.
\]
\end{solution}

\begin{exercise}\label{exr:viii35-03}
Determine for which $d$ the Lefschetz theorem forces a fixed point on $S^2$.
\end{exercise}
\begin{hint}
Set $1+d\ne0$.
\end{hint}
\begin{solution}
For \(S^2\),
\[
L(f)=1+d.
\]
Lefschetz forces a fixed point whenever \(d\ne-1\). The theorem is silent when \(d=-1\).
\end{solution}

\begin{exercise}\label{exr:viii35-04}
Do the same on $S^1$.
\end{exercise}
\begin{hint}
Set $1-d\ne0$.
\end{hint}
\begin{solution}
For \(S^1\),
\[
L(f)=1-d.
\]
Thus a fixed point is forced whenever \(d\ne1\).
\end{solution}

\begin{exercise}\label{exr:viii35-05}
Compute the Lefschetz number of a constant map on a connected finite CW complex.
\end{exercise}
\begin{hint}
Only $H_0$ contributes nontrivially.
\end{hint}
\begin{solution}
A constant map on a connected space induces the identity on \(H_0\) and zero on positive reduced homology. Therefore
\[
L(f)=1.
\]
\end{solution}

\begin{exercise}\label{exr:viii35-06}
Deduce Brouwer's fixed-point theorem for $D^n$.
\end{exercise}
\begin{hint}
Use contractibility.
\end{hint}
\begin{solution}
Every self-map \(f:D^n\to D^n\) has \(L(f)=1\), since the disk is contractible and connected. The Lefschetz fixed-point theorem therefore guarantees a fixed point. This is Brouwer's theorem.
\end{solution}

\begin{exercise}\label{exr:viii35-07}
Show that homotopic self-maps have the same Lefschetz number.
\end{exercise}
\begin{hint}
Use homotopy invariance of induced homology maps.
\end{hint}
\begin{solution}
Homotopic maps induce identical maps on every homology group. Their traces agree degree by degree, hence their alternating trace sums agree:
\[
f\simeq g\implies L(f)=L(g).
\]
\end{solution}

\begin{exercise}\label{exr:viii35-08}
Show that a fixed-point-free map must have Lefschetz number zero.
\end{exercise}
\begin{hint}
Use the contrapositive of Lefschetz.
\end{hint}
\begin{solution}
The Lefschetz theorem says \(L(f)\ne0\Rightarrow\operatorname{Fix}(f)\ne\varnothing\). Taking the contrapositive gives
\[
\operatorname{Fix}(f)=\varnothing\Rightarrow L(f)=0.
\]
\end{solution}

\begin{exercise}\label{exr:viii35-09}
Give an example where $L(f)=0$ but fixed points may still occur.
\end{exercise}
\begin{hint}
Cancellation of local indices is possible.
\end{hint}
\begin{solution}
The identity on \(S^1\) has every point fixed but
\[
L(\mathrm{id}_{S^1})=\chi(S^1)=0.
\]
Thus \(L(f)=0\) does not imply that the fixed-point set is empty.
\end{solution}

\begin{exercise}\label{exr:viii35-10}
Write the graph $\Gamma_f$ and diagonal $\Delta$ for a self-map of a manifold.
\end{exercise}
\begin{hint}
Use $x\mapsto(x,f(x))$.
\end{hint}
\begin{solution}
The graph and diagonal are
\[
\Gamma_f=\{(x,f(x)):x\in M\},\qquad
\Delta=\{(x,x):x\in M\}\subset M\times M.
\]
\end{solution}

\begin{exercise}\label{exr:viii35-11}
Explain why their intersections are fixed points.
\end{exercise}
\begin{hint}
Solve $(x,f(x))=(x,x)$.
\end{hint}
\begin{solution}
A point lies in both submanifolds exactly when
\[
(x,f(x))=(x,x),
\]
which is equivalent to \(f(x)=x\). Hence \(\Gamma_f\cap\Delta\) is naturally the fixed-point set.
\end{solution}

\begin{exercise}\label{exr:viii35-12}
Relate a transverse intersection sign to local index.
\end{exercise}
\begin{hint}
Orient the graph and diagonal.
\end{hint}
\begin{solution}
At a transverse intersection of \(\Gamma_f\) and \(\Delta\), the orientations of their tangent spaces determine a local intersection sign. For a nondegenerate smooth fixed point this sign is
\[
\operatorname{sign}\det(I-Df_x),
\]
the local fixed-point index.
\end{solution}

\begin{exercise}\label{exr:viii35-13}
Compute $L(f)$ for the torus map induced by $A=\begin{pmatrix}2&0\\0&2\end{pmatrix}$.
\end{exercise}
\begin{hint}
Use $1-\operatorname{tr}A+\det A$.
\end{hint}
\begin{solution}
For \(A=\begin{pmatrix}2&0\\0&2\end{pmatrix}\),
\[
L(f)=1-\operatorname{tr}A+\det A=1-4+4=1.
\]
Equivalently \(\det(I-A)=\det(-I)=1\), so a fixed point is forced.
\end{solution}

\begin{exercise}\label{exr:viii35-14}
Compute it for $A=\begin{pmatrix}1&1\\0&1\end{pmatrix}$.
\end{exercise}
\begin{hint}
Compute trace and determinant.
\end{hint}
\begin{solution}
For \(A=\begin{pmatrix}1&1\\0&1\end{pmatrix}\),
\[
L(f)=1-2+1=0.
\]
Indeed \(I-A\) is singular, so Lefschetz gives no forced fixed point.
\end{solution}

\begin{exercise}\label{exr:viii35-15}
Verify $L(f)=\det(I-A)$ for a general $2\times2$ integer matrix.
\end{exercise}
\begin{hint}
Expand $\det(I-A)$.
\end{hint}
\begin{solution}
For \(A=\begin{pmatrix}a&b\\c&d\end{pmatrix}\),
\[
\det(I-A)=(1-a)(1-d)-bc
=1-(a+d)+(ad-bc)
=1-\operatorname{tr}A+\det A.
\]
This is exactly the torus Lefschetz number.
\end{solution}

\begin{exercise}\label{exr:viii35-16}
Determine when the torus trace formula guarantees a fixed point.
\end{exercise}
\begin{hint}
Require $\det(I-A)\ne0$.
\end{hint}
\begin{solution}
The trace formula gives \(L(f)=\det(I-A)\). Therefore
\[
\det(I-A)\ne0
\]
is sufficient for the induced torus map to have a fixed point.
\end{solution}

\begin{exercise}\label{exr:viii35-17}
Compute $L$ for an orientation-reversing degree $-1$ map of $S^2$.
\end{exercise}
\begin{hint}
Use $1+d$ in even dimension.
\end{hint}
\begin{solution}
An orientation-reversing map \(S^2\to S^2\) of degree \(-1\) has
\[
L(f)=1+(-1)=0.
\]
Thus Lefschetz alone does not force a fixed point.
\end{solution}

\begin{exercise}\label{exr:viii35-18}
Compute $L$ for the antipodal map on $S^n$.
\end{exercise}
\begin{hint}
The antipodal degree is $(-1)^{n+1}$.
\end{hint}
\begin{solution}
The antipodal map on \(S^n\) has degree \((-1)^{n+1}\). Hence
\[
L(a)=1+(-1)^n(-1)^{n+1}=1-1=0.
\]
\end{solution}

\begin{exercise}\label{exr:viii35-19}
Compare the conclusion with the actual fixed-point set.
\end{exercise}
\begin{hint}
Check that antipodal maps have no fixed points.
\end{hint}
\begin{solution}
The antipodal map has no fixed points because \(x=-x\) on a sphere would imply \(x=0\), which is impossible. Its Lefschetz number is correspondingly zero, consistent with the theorem.
\end{solution}

\begin{exercise}\label{exr:viii35-20}
Use the identity map to recover Euler characteristic.
\end{exercise}
\begin{hint}
Sum alternating Betti numbers.
\end{hint}
\begin{solution}
For the identity,
\[
L(\mathrm{id}_X)=\sum_k(-1)^k\operatorname{tr}(\mathrm{id}|H_k)
=\sum_k(-1)^k b_k
=\chi(X).
\]
\end{solution}

\begin{exercise}\label{exr:viii35-21}
Express the Lefschetz number as an alternating supertrace.
\end{exercise}
\begin{hint}
Regard the total homology as a graded vector space.
\end{hint}
\begin{solution}
If \(H_*(X;\mathbb Q)=H_{\mathrm{even}}\oplus H_{\mathrm{odd}}\) as a graded vector space, then
\[
L(f)=\operatorname{tr}(f_*|H_{\mathrm{even}})
-\operatorname{tr}(f_*|H_{\mathrm{odd}}),
\]
which is the supertrace.
\end{solution}

\begin{exercise}\label{exr:viii35-22}
Explain the role of finite-dimensional homology in defining the trace.
\end{exercise}
\begin{hint}
Use a field such as $\mathbb Q$.
\end{hint}
\begin{solution}
The trace is defined for finite-dimensional vector spaces. A finite CW complex has finite-dimensional \(H_k(X;\mathbb Q)\), and only finitely many degrees are nonzero, so the alternating trace sum is finite and well-defined.
\end{solution}

\begin{exercise}\label{exr:viii35-23}
Describe how Poincaré duality enters the graph--diagonal proof.
\end{exercise}
\begin{hint}
Dualize the diagonal class.
\end{hint}
\begin{solution}
Poincaré duality identifies the homology class of the diagonal with a sum built from dual homology/cohomology bases. Pairing the graph class with this diagonal class converts the graph--diagonal intersection number into the alternating traces of \(f_*\).
\end{solution}

\begin{exercise}\label{exr:viii35-24}
Summarize homology trace $\leftrightarrow$ intersection number $\leftrightarrow$ fixed point.
\end{exercise}
\begin{hint}
Track the three descriptions of the same number.
\end{hint}
\begin{solution}
The same integer has three forms:
\[
L(f)=\sum_k(-1)^k\operatorname{tr}(f_*|H_k)
=\Gamma_f\cdot\Delta
=\sum_{x\in\operatorname{Fix}(f)}\operatorname{ind}_x(f)
\]
when the local-index hypotheses hold. This is the algebraic, geometric, and local fixed-point synthesis of Lefschetz theory.
\end{solution}
