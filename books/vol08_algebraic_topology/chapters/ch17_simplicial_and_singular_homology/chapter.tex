\chapter{Simplicial and Singular Homology}
\label{ch:viii17-simplicial-singular-homology}

Chain complexes become topological invariants once their generators are chosen
from a space.  Simplicial homology uses oriented simplices in a triangulation.
Singular homology replaces a triangulation by all continuous maps from standard
simplices into the space.  The two theories agree on triangulable spaces, but
singular homology is defined for arbitrary topological spaces and is therefore
the canonical general theory.

\section{Simplicial chain groups}

Let \(K\) be a simplicial complex.

\begin{definition}[Simplicial chain group]
\label{def:viii17-simplicial-chain}
The group \(C_n(K)\) is the free abelian group generated by oriented
\(n\)-simplices of \(K\), with the relation that reversing orientation changes
sign.
\end{definition}

\section{Simplicial boundary}

For an oriented simplex,
\[
[v_0,\dots,v_n],
\]
define
\[
\partial_n[v_0,\dots,v_n]
=
\sum_{i=0}^n
(-1)^i
[v_0,\dots,\widehat v_i,\dots,v_n].
\]

\begin{theorem}[Boundary squares to zero]
\label{thm:viii17-boundary-square}
\[
\partial_{n-1}\partial_n=0.
\]
\end{theorem}

Every codimension-two face appears twice with opposite signs.

\section{Simplicial homology}

\begin{definition}[Simplicial homology]
\label{def:viii17-simplicial-homology}
\[
H_n(K)
=
\ker\partial_n/\operatorname{im}\partial_{n+1}.
\]
\end{definition}

If \(X\cong|K|\), one often writes these groups as homology groups of the
triangulated space.

\section{Degree zero}

For a simplicial complex,
\[
H_0(K)
\]
records path components.

\begin{theorem}[Zeroth homology]
\label{thm:viii17-h0}
If \(K\) has \(c\) path components, then
\[
H_0(K)\cong\mathbb Z^c.
\]
\end{theorem}

\section{The circle}

Triangulate \(S^1\) by a polygon.  There is one independent one-cycle going
around the loop and no two-simplex filling it.

\begin{theorem}[Homology of the circle]
\label{thm:viii17-circle}
\[
H_0(S^1)\cong\mathbb Z,
\qquad
H_1(S^1)\cong\mathbb Z,
\qquad
H_n(S^1)=0\quad(n\ge2).
\]
\end{theorem}

\section{The sphere}

For \(n\ge1\),
\[
H_k(S^n)\cong
\begin{cases}
\mathbb Z,&k=0,n,\\
0,&\text{otherwise}.
\end{cases}
\]

The top-dimensional class records the fundamental cycle of the oriented
sphere.

\section{The disk}

Because a disk is contractible,
\[
H_0(D^n)\cong\mathbb Z,
\qquad
H_k(D^n)=0\quad(k>0).
\]

A direct simplicial calculation sees every positive-dimensional cycle as a
boundary after filling the triangulation.

\section{Reduced homology}

\begin{definition}[Reduced homology]
\label{def:viii17-reduced}
The reduced homology groups \(\widetilde H_n(X)\) are defined using the
augmented chain complex.  For nonempty \(X\),
\[
H_0(X)\cong \widetilde H_0(X)\oplus\mathbb Z.
\]
\end{definition}

If \(X\) is path connected,
\[
\widetilde H_0(X)=0.
\]

For a point,
\[
\widetilde H_n(*)=0
\]
for every \(n\).

\section{Singular simplices}

Let
\[
\Delta^n
=
\left\{
(t_0,\dots,t_n)\in\mathbb R^{n+1}:
t_i\ge0,\ \sum t_i=1
\right\}.
\]

\begin{definition}[Singular simplex]
\label{def:viii17-singular-simplex}
A singular \(n\)-simplex in \(X\) is a continuous map
\[
\sigma:\Delta^n\to X.
\]
\end{definition}

\section{Singular chain groups}

\begin{definition}[Singular chains]
\label{def:viii17-singular-chain}
The singular chain group
\[
C_n^{\mathrm{sing}}(X)
\]
is the free abelian group generated by all singular \(n\)-simplices in \(X\).
\end{definition}

Unlike simplicial chains, there are generally infinitely many generators even
for very simple spaces.

\section{Face maps and singular boundary}

Let
\[
\delta_i:\Delta^{n-1}\to\Delta^n
\]
include the \(i\)th face.  Define
\[
\partial\sigma
=
\sum_{i=0}^n(-1)^i\sigma\circ\delta_i.
\]

The same cancellation proves
\[
\partial^2=0.
\]

\section{Singular homology}

\begin{definition}[Singular homology]
\label{def:viii17-singular-homology}
\[
H_n^{\mathrm{sing}}(X)
=
H_n(C_*^{\mathrm{sing}}(X)).
\]
\end{definition}

This construction requires no triangulation.

\section{Functoriality}

A continuous map
\[
f:X\to Y
\]
acts on a singular simplex by composition:
\[
\sigma\longmapsto f\circ\sigma.
\]

\begin{theorem}[Induced chain map]
\label{thm:viii17-induced-chain}
The maps
\[
f_\#:C_n^{\mathrm{sing}}(X)\to C_n^{\mathrm{sing}}(Y)
\]
defined by
\[
f_\#(\sigma)=f\circ\sigma
\]
form a chain map.
\end{theorem}

Hence every continuous map induces
\[
f_*:H_n(X)\to H_n(Y).
\]

\section{Functoriality laws}

\[
(\operatorname{id}_X)_*=\operatorname{id}_{H_n(X)},
\qquad
(g\circ f)_*=g_*\circ f_*.
\]

Thus homology is a functor from spaces and continuous maps to graded abelian
groups.

\section{Simplicial maps and homology}

A simplicial map
\[
f:K\to L
\]
induces a simplicial chain map by sending an oriented simplex to its oriented
image when the dimension is preserved, and to zero when vertices collapse and
the image has smaller dimension.

This agrees with the induced map on singular homology after geometric
realization.

\section{Simplicial versus singular homology}

\begin{theorem}[Comparison theorem]
\label{thm:viii17-comparison}
For a simplicial complex \(K\),
\[
H_n^{\mathrm{simp}}(K)
\cong
H_n^{\mathrm{sing}}(|K|).
\]
\end{theorem}

The proof uses subdivision and approximation machinery.  The theorem allows
combinatorial calculations to compute the general topological invariant.

\section{Homology of disjoint unions}

For a disjoint union
\[
X=\coprod_\alpha X_\alpha,
\]
singular simplices lie in individual components, so
\[
H_n(X)
\cong
\bigoplus_\alpha H_n(X_\alpha).
\]

\section{Wedges in positive degree}

For well-pointed spaces and positive degrees, reduced homology satisfies the
wedge formula
\[
\widetilde H_n(X\vee Y)
\cong
\widetilde H_n(X)\oplus\widetilde H_n(Y).
\]

This predicts, for example,
\[
H_1(S^1\vee S^1)\cong\mathbb Z^2.
\]

\section{Homology and holes}

Homology does not record all homotopy information.  Instead it linearizes
cycles:
\[
\text{cycles that do not bound}
\quad\leadsto\quad
\text{homology classes}.
\]

The groups are abelian even when the fundamental group is not.

\section{First homology and abelianization}

\begin{theorem}[First homology]
\label{thm:viii17-h1-abelianization}
For a path-connected space,
\[
H_1(X;\mathbb Z)
\cong
\pi_1(X)_{\mathrm{ab}},
\]
the abelianization of the fundamental group.
\end{theorem}

This result provides a direct bridge between the covering/fundamental-group arc
and homology.

\section{Homotopy invariance preview}

Homotopic maps induce the same maps in singular homology.  The full chain-level
proof uses prism operators and chain homotopies and is deferred to VIII/20,
with chain-homotopy machinery developed further in VIII/22.

\section{Bridge to cellular homology}

Simplicial and singular chain groups can be very large.  A CW complex offers a
smaller chain complex with one generator per cell.  VIII/18 derives that
cellular chain complex from the skeletal filtration and computes its boundary
maps from attaching maps.

\section{Solved problem dossiers}

\begin{problem}[Triangle boundary squared]\label{prob:viii17-01}
Verify \(\partial^2[v_0,v_1,v_2]=0\).
\end{problem}
\begin{solution}
First \(\partial[v_0,v_1,v_2]=[v_1,v_2]-[v_0,v_2]+[v_0,v_1]\). Applying \(\partial[v_i,v_j]=[v_j]-[v_i]\), every vertex cancels.
\end{solution}

\begin{problem}[Polygonal circle]\label{prob:viii17-02}
Compute simplicial \(H_1\) of an \(m\)-gon with no 2-simplices.
\end{problem}
\begin{solution}
The sum of all coherently oriented edges is a cycle. The kernel of \(\partial_1\) has rank one and there are no 2-boundaries, so \(H_1\cong\mathbb Z\).
\end{solution}

\begin{problem}[Filled triangle]\label{prob:viii17-03}
Compute \(H_1\) of a filled 2-simplex.
\end{problem}
\begin{solution}
The boundary cycle of the triangle is exactly the boundary of the 2-simplex, so it dies. The disk is contractible, hence \(H_1=0\).
\end{solution}

\begin{problem}[Two components]\label{prob:viii17-04}
Compute \(H_0\) of two disjoint circles.
\end{problem}
\begin{solution}
There are two path components, so \(H_0\cong\mathbb Z^2\).
\end{solution}

\begin{problem}[Reduced zero homology]\label{prob:viii17-05}
Compute \(\widetilde H_0\) of a space with \(c\) nonempty path components.
\end{problem}
\begin{solution}
Ordinary \(H_0\cong\mathbb Z^c\); removing the diagonal augmentation summand leaves \(\mathbb Z^{c-1}\).
\end{solution}

\begin{problem}[Sphere top cycle]\label{prob:viii17-06}
Why does an oriented triangulated sphere define a top-dimensional cycle?
\end{problem}
\begin{solution}
Every codimension-one face belongs to two top simplices with opposite induced orientations, so those face contributions cancel in the total boundary.
\end{solution}

\begin{problem}[Disk boundary class]\label{prob:viii17-07}
Why is the boundary sphere class zero in the homology of the disk?
\end{problem}
\begin{solution}
The boundary fundamental cycle is the boundary of the sum of oriented top-dimensional simplices filling the disk.
\end{solution}

\begin{problem}[Singular 0-simplices]\label{prob:viii17-08}
What are singular 0-simplices in \(X\)?
\end{problem}
\begin{solution}
\(\Delta^0\) is a point, so singular 0-simplices are exactly points of \(X\).
\end{solution}

\begin{problem}[Singular 1-simplex]\label{prob:viii17-09}
What familiar object is a singular 1-simplex?
\end{problem}
\begin{solution}
A continuous map \(I\cong\Delta^1\to X\), i.e. a path with an orientation.
\end{solution}

\begin{problem}[Boundary of a singular path]\label{prob:viii17-10}
For a singular 1-simplex \(\sigma\), what is \(\partial\sigma\)?
\end{problem}
\begin{solution}
It is the endpoint minus the starting point: \(\sigma(1)-\sigma(0)\).
\end{solution}

\begin{problem}[Functorial boundary]\label{prob:viii17-11}
Verify \(f_\#\partial=\partial f_\#\) on a singular simplex.
\end{problem}
\begin{solution}
Both expressions are \(\sum_i(-1)^i f\circ\sigma\circ\delta_i\).
\end{solution}

\begin{problem}[Constant map on positive homology]\label{prob:viii17-12}
What does a constant map induce on reduced positive-degree homology?
\end{problem}
\begin{solution}
It factors through a point, whose reduced homology vanishes, so the induced map is zero.
\end{solution}

\begin{problem}[Identity map]\label{prob:viii17-13}
What map does \(\operatorname{id}_X\) induce on homology?
\end{problem}
\begin{solution}
The identity on every \(H_n(X)\).
\end{solution}

\begin{problem}[Circle degree map]\label{prob:viii17-14}
For \(f:S^1\to S^1\) of degree \(m\), what is \(f_*:H_1(S^1)\to H_1(S^1)\)?
\end{problem}
\begin{solution}
Under \(H_1(S^1)\cong\mathbb Z\), it is multiplication by \(m\).
\end{solution}

\begin{problem}[Two-circle wedge]\label{prob:viii17-15}
Interpret \(H_1(S^1\vee S^1)\) using abelianization.
\end{problem}
\begin{solution}
\(\pi_1\cong F_2\), whose abelianization is \(\mathbb Z^2\), so \(H_1\cong\mathbb Z^2\).
\end{solution}

\begin{problem}[Torus first homology]\label{prob:viii17-16}
Use \(\pi_1(T^2)\cong\mathbb Z^2\) to compute \(H_1(T^2)\).
\end{problem}
\begin{solution}
The group is already abelian, so its abelianization is \(\mathbb Z^2\).
\end{solution}

\begin{problem}[Projective-plane first homology]\label{prob:viii17-17}
Use \(\pi_1(\mathbb RP^2)\cong\mathbb Z/2\) to compute \(H_1\).
\end{problem}
\begin{solution}
The fundamental group is abelian already, so \(H_1(\mathbb RP^2)\cong\mathbb Z/2\).
\end{solution}

\begin{problem}[Simply connected space]\label{prob:viii17-18}
What is \(H_1(X)\) for path-connected simply connected \(X\)?
\end{problem}
\begin{solution}
The fundamental group is trivial, so its abelianization and therefore \(H_1\) are zero.
\end{solution}

\begin{problem}[Disjoint-union chains]\label{prob:viii17-19}
Why do singular chain groups split over components?
\end{problem}
\begin{solution}
The connected simplex \(\Delta^n\) has image in a single component, so every singular generator belongs to one component and the chain group is the direct sum of component chain groups.
\end{solution}

\begin{problem}[Comparison theorem use]\label{prob:viii17-20}
Why is simplicial homology computationally useful if singular homology is the general invariant?
\end{problem}
\begin{solution}
For a triangulated space, the comparison theorem identifies the small combinatorial simplicial chain complex with singular homology, avoiding infinitely many singular generators.
\end{solution}

\begin{problem}[Simplicial collapse map]\label{prob:viii17-21}
If a simplicial map sends an edge's two vertices to the same vertex, what happens to that oriented edge on chains?
\end{problem}
\begin{solution}
Its image has dimension zero, so the induced simplicial 1-chain map sends the edge to zero.
\end{solution}

\begin{problem}[Orientation sign]\label{prob:viii17-22}
Why must simplicial chains impose sign under odd vertex permutations?
\end{problem}
\begin{solution}
Without the sign convention, adjacent top simplices would not cancel correctly along shared faces and the alternating boundary formula would fail to encode orientation.
\end{solution}

\begin{problem}[Homotopy-invariance boundary]\label{prob:viii17-23}
Why is the proof that homotopic maps induce the same map postponed?
\end{problem}
\begin{solution}
It requires a chain homotopy, concretely a prism operator on singular chains. VIII/20 develops this proof after the basic homology constructions are established.
\end{solution}

\begin{problem}[Bridge to cellular chains]\label{prob:viii17-24}
Why can cellular homology be much smaller than singular homology?
\end{problem}
\begin{solution}
A CW complex with finitely many cells has one cellular generator per cell, while singular chains contain every continuous simplex map. The cellular boundary encodes attaching-map degrees.
\end{solution}
