\section{Supplementary worked examples}
\label{sec:viii17-pedagogy}

\begin{example}[Singular homology of a point]\label{ex:viii17-ped-01}
Although the singular chain groups of a point contain many degenerate simplices, the resulting homology is $H_0(*)\cong\mathbb Z$ and $H_n(*)=0$ for $n>0$.  Homology discards most chain-level redundancy.
\end{example}

\begin{example}[Simplicial circle]\label{ex:viii17-ped-02}
For a triangulated circle, one independent 1-cycle survives modulo boundaries, giving $H_1(S^1)\cong\mathbb Z$ and $H_0(S^1)\cong\mathbb Z$.  Singular homology gives the same groups.
\end{example}

\begin{example}[Comparison principle]\label{ex:viii17-ped-03}
For a simplicial complex $K$, simplicial chains include into singular chains of $|K|$.  The comparison theorem shows this inclusion induces isomorphisms on homology, so combinatorial and continuous models agree.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Point homology]\label{exr:viii17-ped-01}
State $H_n(*)$ with integer coefficients.
\end{exercise}
\begin{hint}
Only connected-component information survives in degree zero.
\end{hint}
\begin{solution}
$H_0(*)\cong\mathbb Z$ and $H_n(*)=0$ for $n>0$.
\end{solution}

\begin{exercise}[Circle homology]\label{exr:viii17-ped-02}
State the integral homology of $S^1$.
\end{exercise}
\begin{hint}
There is one component and one essential loop.
\end{hint}
\begin{solution}
$H_0(S^1)\cong\mathbb Z$, $H_1(S^1)\cong\mathbb Z$, and $H_n=0$ for $n\ge2$.
\end{solution}

\begin{exercise}[Sphere homology]\label{exr:viii17-ped-03}
State the reduced integral homology of $S^n$.
\end{exercise}
\begin{hint}
Only the top-dimensional class survives.
\end{hint}
\begin{solution}
$\widetilde H_k(S^n)\cong\mathbb Z$ for $k=n$ and is zero otherwise.
\end{solution}

\begin{exercise}[Disconnected $H_0$]\label{exr:viii17-ped-04}
If a space has $c$ path components under the standard niceness assumptions, what is $H_0$?
\end{exercise}
\begin{hint}
One generator survives per component.
\end{hint}
\begin{solution}
$H_0\cong\mathbb Z^c$.
\end{solution}

\begin{exercise}[Triangulated disk]\label{exr:viii17-ped-05}
What is the homology of a triangulated disk?
\end{exercise}
\begin{hint}
A disk is contractible.
\end{hint}
\begin{solution}
$H_0\cong\mathbb Z$ and $H_n=0$ for $n>0$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Simplicial cycles map to singular cycles]\label{exr:viii17-ped-06}
Show the inclusion of simplicial chains into singular chains commutes with boundary.
\end{exercise}
\begin{hint}
Each oriented affine simplex has the same alternating face restrictions.
\end{hint}
\begin{solution}
The singular boundary of the affine characteristic simplex is exactly the signed sum of its simplicial faces, so the inclusion is a chain map.
\end{solution}

\begin{exercise}[Triangulation invariance]\label{exr:viii17-ped-07}
Explain why two triangulations of the same polyhedron yield isomorphic homology.
\end{exercise}
\begin{hint}
Both compare to singular homology of the same space.
\end{hint}
\begin{solution}
Each simplicial homology is naturally isomorphic to singular homology of the common underlying polyhedron, hence the groups are isomorphic independently of triangulation.
\end{solution}

\begin{exercise}[Homeomorphism invariance]\label{exr:viii17-ped-08}
Prove a homeomorphism induces isomorphisms on singular homology.
\end{exercise}
\begin{hint}
Use the induced chain maps of the map and its inverse.
\end{hint}
\begin{solution}
Functoriality gives induced homology maps whose compositions are identities, so they are inverse isomorphisms.
\end{solution}

\begin{exercise}[Contractible spaces]\label{exr:viii17-ped-09}
Prove a contractible space has the homology of a point.
\end{exercise}
\begin{hint}
Homotopy invariance is the key theorem.
\end{hint}
\begin{solution}
A contraction gives a homotopy equivalence with a point, hence isomorphisms on all homology groups.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Singular depends on triangulation]\label{exr:viii17-ped-10}
Test: singular homology requires choosing a triangulation and changes if the triangulation changes.
\end{exercise}
\begin{hint}
Singular simplices are all continuous maps from standard simplices.
\end{hint}
\begin{solution}
False. Singular homology is defined directly for topological spaces and is triangulation-independent.
\end{solution}

\begin{exercise}[Simplicial always available]\label{exr:viii17-ped-11}
Test: every topological space is already a simplicial complex.
\end{exercise}
\begin{hint}
Triangulability is an extra property.
\end{hint}
\begin{solution}
False; singular homology is designed to work without a triangulation.
\end{solution}

\begin{exercise}[Same homology implies homeomorphic]\label{exr:viii17-ped-12}
Test: spaces with isomorphic homology groups are homeomorphic.
\end{exercise}
\begin{hint}
Homology is not a complete invariant.
\end{hint}
\begin{solution}
False. Many nonhomeomorphic and even non-homotopy-equivalent spaces share the same homology groups.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Mesh verification]\label{exr:viii17-ped-13}
Why compare a mesh's simplicial homology with a known singular-homology target?
\end{exercise}
\begin{hint}
It checks that the discretization has the intended topology.
\end{hint}
\begin{solution}
For a mesh approximating a known space, matching Betti numbers/torsion provides a strong topological sanity check independent of metric fidelity.
\end{solution}

\begin{exercise}[Continuous-to-discrete bridge]\label{exr:viii17-ped-14}
What is gained by the simplicial-singular comparison theorem computationally?
\end{exercise}
\begin{hint}
Use finite combinatorics without changing the invariant.
\end{hint}
\begin{solution}
One may compute with sparse finite boundary matrices of a triangulation while knowing the result equals the intrinsic singular homology of the underlying space.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Subdivision chain equivalence]\label{exr:viii17-ped-15}
Explain how repeated subdivision can refine geometry without changing homology.
\end{exercise}
\begin{hint}
Subdivision maps are chain-homotopy equivalences.
\end{hint}
\begin{solution}
The subdivision operator and a suitable simplicial approximation induce inverse maps on homology up to chain homotopy, so refinement preserves the groups.
\end{solution}

\begin{exercise}[Comparison synthesis]\label{exr:viii17-ped-16}
Describe the logical route from a triangulated surface to a topological invariant independent of that mesh.
\end{exercise}
\begin{hint}
Build simplicial chains, compare to singular chains, then use topological invariance.
\end{hint}
\begin{solution}
Compute simplicial homology from the mesh; the comparison theorem identifies it with singular homology of the realization; singular homology is invariant under homeomorphism/homotopy, so the answer reflects the space rather than the chosen triangulation.
\end{solution}
