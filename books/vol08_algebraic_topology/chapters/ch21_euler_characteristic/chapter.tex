\chapter{Euler Characteristic}
\label{ch:viii21-euler-characteristic}

Euler characteristic compresses a finite cell decomposition into a single
integer.  Remarkably, the same integer can be computed from homology ranks, so
it is independent of the chosen finite CW or simplicial structure and is
preserved by homotopy equivalence.  This chapter develops the combinatorial,
homological, and structural forms of the invariant.

\section{Cell-count definition}

Let \(X\) be a finite CW complex with \(c_n\) cells of dimension \(n\).

\begin{definition}[Euler characteristic]
\label{def:viii21-euler}
\[
\chi(X)=\sum_{n\ge0}(-1)^n c_n.
\]
\end{definition}

Only finitely many terms are nonzero.

\section{Simplicial form}

If a finite simplicial complex has \(f_n\) \(n\)-simplices, then
\[
\chi(K)=\sum_n(-1)^n f_n.
\]

For a triangulated space \(X\cong|K|\), this agrees with the CW definition.

\section{Basic examples}

For a finite discrete set with \(m\) points,
\[
\chi=m.
\]

For an interval,
\[
\chi=2-1=1.
\]

For a polygonal circle,
\[
\chi=V-E=0.
\]

For the standard two-cell sphere structure,
\[
\chi(S^n)=1+(-1)^n.
\]

\section{Graphs}

For a finite graph,
\[
\chi(X)=V-E.
\]

If the graph is connected and has fundamental-group rank \(r\), then
\[
r=E-V+1,
\]
so
\[
\chi(X)=1-r.
\]

Thus a tree has
\[
\chi=1.
\]

\section{Closed orientable surfaces}

The standard CW structure on a genus-\(g\) orientable surface has one
0-cell, \(2g\) 1-cells, and one 2-cell.  Hence
\[
\chi(\Sigma_g)=2-2g.
\]

In particular,
\[
\chi(S^2)=2,
\qquad
\chi(T^2)=0.
\]

\section{Real projective spaces}

The standard CW structure on \(\mathbb RP^n\) has one cell in every dimension
\(0,\dots,n\), so
\[
\chi(\mathbb RP^n)
=
\sum_{k=0}^n(-1)^k.
\]

Therefore
\[
\chi(\mathbb RP^n)=
\begin{cases}
1,&n\text{ even},\\
0,&n\text{ odd}.
\end{cases}
\]

\section{Euler--Poincare formula}

Let \(X\) be a finite CW complex.  Write
\[
b_n=\operatorname{rank}H_n(X;\mathbb Z).
\]

\begin{theorem}[Euler--Poincare]
\label{thm:viii21-euler-poincare}
\[
\chi(X)
=
\sum_n(-1)^n b_n.
\]
\end{theorem}

Torsion in homology does not contribute to the rank sum.

\section{Algebraic proof}

For a finite chain complex of finite-rank free abelian groups,
\[
C_n,
\]
write
\[
Z_n=\ker\partial_n,
\qquad
B_n=\operatorname{im}\partial_{n+1}.
\]

From
\[
0\to Z_n\to C_n\to B_{n-1}\to0
\]
we obtain
\[
\operatorname{rank}C_n
=
\operatorname{rank}Z_n+\operatorname{rank}B_{n-1}.
\]

From
\[
0\to B_n\to Z_n\to H_n\to0
\]
we obtain
\[
\operatorname{rank}Z_n
=
\operatorname{rank}B_n+\operatorname{rank}H_n.
\]

Alternating sums telescope, leaving the homology-rank formula.

\section{Independence of cell structure}

\begin{corollary}[Well-definedness]
\label{cor:viii21-well-defined}
The alternating cell count is independent of the chosen finite CW structure.
\end{corollary}

Indeed both cell decompositions compute the same homology ranks.

\section{Homotopy invariance}

\begin{corollary}[Homotopy invariance]
\label{cor:viii21-homotopy}
If finite CW complexes \(X\) and \(Y\) are homotopy equivalent, then
\[
\chi(X)=\chi(Y).
\]
\end{corollary}

This follows immediately from VIII/20.

\section{Contractible finite complexes}

If \(X\) is a nonempty finite contractible CW complex, then
\[
H_0(X)\cong\mathbb Z,
\qquad
H_n(X)=0\ (n>0),
\]
so
\[
\chi(X)=1.
\]

Therefore a finite CW complex with
\[
\chi(X)\ne1
\]
cannot be contractible.

The converse is false.

\section{Disjoint unions}

For finite CW complexes,
\[
\chi(X\sqcup Y)=\chi(X)+\chi(Y).
\]

This is immediate from either cell counts or homology direct sums.

\section{Wedges}

For connected finite CW complexes with chosen basepoints,
\[
\chi(X\vee Y)
=
\chi(X)+\chi(Y)-1.
\]

The subtraction corrects for identifying two zero-cells into one.

More generally,
\[
\chi\left(\bigvee_{j=1}^rX_j\right)
=
\sum_j\chi(X_j)-(r-1).
\]

\section{Product formula}

\begin{theorem}[Product formula]
\label{thm:viii21-product}
For finite CW complexes,
\[
\chi(X\times Y)
=
\chi(X)\chi(Y).
\]
\end{theorem}

One proof uses product cells: an \(i\)-cell times a \(j\)-cell contributes an
\((i+j)\)-cell, and the alternating double sum factors.

\section{Examples of products}

Since
\[
\chi(S^1)=0,
\]
every finite CW product
\[
S^1\times X
\]
has
\[
\chi=0.
\]

For example,
\[
\chi(T^n)=0
\]
for every \(n\ge1\).

Also,
\[
\chi(S^2\times S^2)=4.
\]

\section{Inclusion--exclusion}

Suppose finite subcomplexes \(A,B\) cover a finite CW complex \(X\):
\[
X=A\cup B.
\]

\begin{theorem}[Euler inclusion--exclusion]
\label{thm:viii21-inclusion-exclusion}
\[
\chi(X)
=
\chi(A)+\chi(B)-\chi(A\cap B).
\]
\end{theorem}

This follows from cell counts when the decompositions are compatible, and more
generally from Mayer--Vietoris/homological arguments.

\section{Attaching one cell}

If \(Y\) is obtained from \(X\) by attaching one \(n\)-cell, then
\[
\chi(Y)=\chi(X)+(-1)^n.
\]

Thus:
\begin{itemize}
\item adding a 0-cell increases \(\chi\) by \(1\);
\item adding a 1-cell decreases \(\chi\) by \(1\);
\item adding a 2-cell increases \(\chi\) by \(1\).
\end{itemize}

The attaching map does not affect this raw Euler change.

\section{Relative Euler characteristic}

For a finite CW pair \((X,A)\), define
\[
\chi(X,A)=\chi(X)-\chi(A).
\]

Equivalently, if the relative homology groups have finite rank,
\[
\chi(X,A)
=
\sum_n(-1)^n\operatorname{rank}H_n(X,A).
\]

\section{Exact sequences and Euler sums}

For a finite exact sequence of finite-rank abelian groups
\[
0\to A_0\to A_1\to\cdots\to A_m\to0,
\]
the alternating sum of ranks is zero.

This elementary fact is the algebraic reason Euler characteristic behaves well
with long exact sequences.

\section{Euler characteristic and coverings}

If
\[
p:\widetilde X\to X
\]
is a finite \(d\)-sheeted covering of finite CW complexes, then
\[
\chi(\widetilde X)=d\,\chi(X).
\]

Each cell downstairs lifts to \(d\) cells upstairs.

\section{Covering examples}

For the \(d\)-fold circle cover,
\[
0=\chi(S^1)=d\cdot0.
\]

For a finite-sheeted covering
\[
\widetilde\Sigma\to\Sigma_g
\]
of closed orientable surfaces,
\[
\chi(\widetilde\Sigma)
=
d(2-2g).
\]

This constrains the genus upstairs.

\section{Odd-dimensional closed manifolds preview}

For closed orientable odd-dimensional manifolds, Poincare duality later implies
\[
\chi(M)=0.
\]

That theorem belongs to VIII/33; here it is only a preview of how deeper
homological symmetry controls Euler characteristic.

\section{What Euler characteristic cannot see}

Euler characteristic is extremely coarse.  Spaces with the same \(\chi\) can
have very different topology.

For example,
\[
\chi(S^1)=0
\qquad\text{and}\qquad
\chi(T^2)=0,
\]
yet their homology and fundamental groups are different.

\section{Bridge to homological machinery}

The Euler--Poincare argument is a rank shadow of chain-level algebra.  The next
part develops that algebra more deeply through chain homotopies, contractions,
mapping cones, coefficient changes, universal coefficient formulas, and
products.

\section{Solved problem dossiers}

\begin{problem}[Triangle disk]\label{prob:viii21-01}
Compute Euler characteristic of a filled triangle.
\end{problem}
\begin{solution}
There are \(3\) vertices, \(3\) edges, and \(1\) face, so \(\chi=3-3+1=1\).
\end{solution}

\begin{problem}[Tetrahedral sphere]\label{prob:viii21-02}
Compute \(\chi\) of the boundary of a tetrahedron.
\end{problem}
\begin{solution}
\(V=4,E=6,F=4\), so \(\chi=4-6+4=2\).
\end{solution}

\begin{problem}[Square with diagonal]\label{prob:viii21-03}
Compute \(\chi\) of a square triangulated by one diagonal.
\end{problem}
\begin{solution}
\(V=4,E=5,F=2\), so \(\chi=4-5+2=1\).
\end{solution}

\begin{problem}[Graph rank]\label{prob:viii21-04}
A connected finite graph has \(V=10,E=13\). Compute its fundamental-group rank and Euler characteristic.
\end{problem}
\begin{solution}
The rank is \(E-V+1=4\), and \(\chi=V-E=-3=1-4\).
\end{solution}

\begin{problem}[Rose]\label{prob:viii21-05}
Compute \(\chi\) of the wedge of \(r\) circles.
\end{problem}
\begin{solution}
One vertex and \(r\) one-cells give \(\chi=1-r\).
\end{solution}

\begin{problem}[Genus three]\label{prob:viii21-06}
Compute \(\chi(\Sigma_3)\).
\end{problem}
\begin{solution}
\(2-2\cdot3=-4\).
\end{solution}

\begin{problem}[Projective plane]\label{prob:viii21-07}
Compute \(\chi(\mathbb RP^2)\) from cells and homology.
\end{problem}
\begin{solution}
Cell count gives \(1-1+1=1\). Homology ranks are \(1,0,0\) since \(H_1\) is torsion, so Euler--Poincare also gives \(1\).
\end{solution}

\begin{problem}[Projective 3-space]\label{prob:viii21-08}
Compute \(\chi(\mathbb RP^3)\).
\end{problem}
\begin{solution}
Cell count \(1-1+1-1=0\). Homology ranks \(b_0=b_3=1\) give \(1-1=0\).
\end{solution}

\begin{problem}[Torus]\label{prob:viii21-09}
Compute \(\chi(T^2)\) from homology.
\end{problem}
\begin{solution}
The Betti numbers are \(1,2,1\), so \(\chi=1-2+1=0\).
\end{solution}

\begin{problem}[Sphere product]\label{prob:viii21-10}
Compute \(\chi(S^2\times S^2)\).
\end{problem}
\begin{solution}
\(\chi(S^2)=2\), so the product formula gives \(4\).
\end{solution}

\begin{problem}[Circle product]\label{prob:viii21-11}
Why does \(S^1\times X\) have zero Euler characteristic for finite CW \(X\)?
\end{problem}
\begin{solution}
\(\chi(S^1)=0\), so the product formula gives \(0\cdot\chi(X)=0\).
\end{solution}

\begin{problem}[Wedge of sphere and circle]\label{prob:viii21-12}
Compute \(\chi(S^2\vee S^1)\).
\end{problem}
\begin{solution}
\(\chi(S^2)+\chi(S^1)-1=2+0-1=1\).
\end{solution}

\begin{problem}[Three-sphere wedge]\label{prob:viii21-13}
Compute \(\chi(S^2\vee S^2\vee S^2)\).
\end{problem}
\begin{solution}
The sum is \(6\), and identifying three basepoints costs \(2\), so \(\chi=4\).
\end{solution}

\begin{problem}[Attach a one-cell]\label{prob:viii21-14}
If \(X\) has \(\chi=5\), what is the Euler characteristic after attaching one 1-cell?
\end{problem}
\begin{solution}
It decreases by \(1\), giving \(4\).
\end{solution}

\begin{problem}[Attach a two-cell]\label{prob:viii21-15}
Continue from the previous space by attaching one 2-cell.
\end{problem}
\begin{solution}
It increases by \(1\), returning to \(5\).
\end{solution}

\begin{problem}[Contractibility obstruction]\label{prob:viii21-16}
A finite CW complex has \(\chi=0\). What can you conclude?
\end{problem}
\begin{solution}
It cannot be contractible, since every nonempty finite contractible CW complex has Euler characteristic \(1\).
\end{solution}

\begin{problem}[Same Euler warning]\label{prob:viii21-17}
Why can \(\chi=0\) not distinguish \(S^1\) from \(T^2\)?
\end{problem}
\begin{solution}
Both have Euler characteristic zero, but their homology and fundamental groups differ. Euler characteristic is only one alternating integer.
\end{solution}

\begin{problem}[Covering surface genus]\label{prob:viii21-18}
Let \(\widetilde\Sigma\to\Sigma_2\) be a 3-sheeted cover of closed orientable surfaces. Find the genus upstairs.
\end{problem}
\begin{solution}
\(\chi(\Sigma_2)=-2\), so upstairs \(\chi=-6\). Solve \(2-2\tilde g=-6\), giving \(\tilde g=4\).
\end{solution}

\begin{problem}[Cover of a torus]\label{prob:viii21-19}
What Euler characteristic must every finite-sheeted cover of \(T^2\) have?
\end{problem}
\begin{solution}
Since \(\chi(T^2)=0\), the covering formula gives \(0\).
\end{solution}

\begin{problem}[Relative Euler disk pair]\label{prob:viii21-20}
Compute \(\chi(D^n,S^{n-1})\).
\end{problem}
\begin{solution}
\(\chi(D^n)-\chi(S^{n-1})=1-[1+(-1)^{n-1}]=(-1)^n\), matching the single relative \(n\)-cell.
\end{solution}

\begin{problem}[Exact sequence rank cancellation]\label{prob:viii21-21}
Why does a finite exact sequence have zero alternating rank sum?
\end{problem}
\begin{solution}
Each image rank appears once with each sign when the groups are decomposed into successive kernels/images, so the contributions telescope.
\end{solution}

\begin{problem}[Euler--Poincare torsion]\label{prob:viii21-22}
Why does \(\mathbb Z/5\) in homology contribute nothing to Euler characteristic?
\end{problem}
\begin{solution}
Euler--Poincare uses free ranks (Betti numbers); a finite torsion group has rank zero.
\end{solution}

\begin{problem}[Subdivision]\label{prob:viii21-23}
Why does barycentric subdivision preserve Euler characteristic?
\end{problem}
\begin{solution}
It preserves the homeomorphism type, hence homology and Euler characteristic. Combinatorially the larger simplex counts have the same alternating sum.
\end{solution}

\begin{problem}[Bridge to duality]\label{prob:viii21-24}
What later theorem forces \(\chi=0\) for closed orientable odd-dimensional manifolds?
\end{problem}
\begin{solution}
Poincare duality in VIII/33 pairs Betti numbers symmetrically across complementary dimensions, causing alternating cancellation.
\end{solution}
