\section{Supplementary worked examples}
\label{sec:viii02-pedagogy}

\begin{example}[Contracting a star-shaped set]\label{ex:viii02-ped-01}
If $X\subset\mathbb R^n$ is star-shaped about $x_0$, $H(x,t)=(1-t)x+tx_0$ contracts $X$ to $x_0$.  Thus star-shaped spaces are contractible even when they are not convex.
\end{example}

\begin{example}[A deformation retract]\label{ex:viii02-ped-02}
The punctured plane deformation retracts onto $S^1$ by $H(x,t)=((1-t)+t/\|x\|)x$.  Hence $\mathbb R^2\setminus\{0\}$ and $S^1$ have the same homotopy type without being homeomorphic.
\end{example}

\begin{example}[Projection from a cylinder]\label{ex:viii02-ped-03}
The projection $S^1\times I\to S^1$ and inclusion $S^1\hookrightarrow S^1\times I$ are homotopy inverses.  One composite is the identity and the other is deformed to the identity by sliding the interval coordinate.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Homotopy inverse equations]\label{exr:viii02-ped-01}
State the equations defining a homotopy equivalence $f:X\to Y$ with inverse candidate $g$.
\end{exercise}
\begin{hint}
The composites need only be homotopic to identities.
\end{hint}
\begin{solution}
Require $g\circ f\simeq\operatorname{id}_X$ and $f\circ g\simeq\operatorname{id}_Y$.
\end{solution}

\begin{exercise}[Contract a disk]\label{exr:viii02-ped-02}
Give an explicit contraction of $D^n$ to the origin.
\end{exercise}
\begin{hint}
Use scalar multiplication.
\end{hint}
\begin{solution}
$H(x,t)=(1-t)x$ stays in $D^n$ and ends at $0$.
\end{solution}

\begin{exercise}[Cylinder retraction]\label{exr:viii02-ped-03}
Write a deformation retraction of $X\times I$ onto $X\times\{0\}$.
\end{exercise}
\begin{hint}
Move only the interval coordinate.
\end{hint}
\begin{solution}
$H((x,s),t)=(x,(1-t)s)$ fixes $X\times\{0\}$ pointwise.
\end{solution}

\begin{exercise}[Retract maps]\label{exr:viii02-ped-04}
For a retraction $r:X\to A$ with inclusion $i:A\hookrightarrow X$, compute $r\circ i$.
\end{exercise}
\begin{hint}
A retraction is the identity on $A$.
\end{hint}
\begin{solution}
$r\circ i=\operatorname{id}_A$.
\end{solution}

\begin{exercise}[Homotopy type transfer]\label{exr:viii02-ped-05}
If $X\simeq Y$ and $Y\simeq Z$, what should be the homotopy type relation between $X$ and $Z$?
\end{exercise}
\begin{hint}
Compose homotopy equivalences.
\end{hint}
\begin{solution}
$X\simeq Z$; composition of homotopy equivalences is a homotopy equivalence.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Contractible implies identity null]\label{exr:viii02-ped-06}
Prove $X$ is contractible iff $\operatorname{id}_X$ is null-homotopic.
\end{exercise}
\begin{hint}
A contraction is exactly such a homotopy.
\end{hint}
\begin{solution}
A contraction joins $\operatorname{id}_X$ to a constant; conversely a null-homotopy of the identity is a contraction.
\end{solution}

\begin{exercise}[Deformation retract gives equivalence]\label{exr:viii02-ped-07}
Prove a deformation retract $A\subset X$ is homotopy equivalent to $X$.
\end{exercise}
\begin{hint}
Use inclusion and retraction.
\end{hint}
\begin{solution}
$r\circ i=\operatorname{id}_A$, while $i\circ r\simeq\operatorname{id}_X$ by the deformation retraction.
\end{solution}

\begin{exercise}[Equivalence transitivity]\label{exr:viii02-ped-08}
Prove the composite of two homotopy equivalences is a homotopy equivalence.
\end{exercise}
\begin{hint}
Compose chosen homotopy inverses in reverse order.
\end{hint}
\begin{solution}
If $g$ and $k$ are inverse candidates for $f$ and $h$, then $g\circ k$ is an inverse candidate for $h\circ f$; composition compatibility transports the homotopies.
\end{solution}

\begin{exercise}[Maps into contractible target]\label{exr:viii02-ped-09}
Prove any two maps $X\to Y$ are homotopic when $Y$ is contractible.
\end{exercise}
\begin{hint}
First deform each map to the contraction point.
\end{hint}
\begin{solution}
Postcompose the contraction of $Y$ with each map, obtaining homotopies to the same constant map; concatenate one with the reverse of the other.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Equivalent means homeomorphic]\label{exr:viii02-ped-10}
Test: homotopy equivalent spaces must be homeomorphic.
\end{exercise}
\begin{hint}
Compare $S^1$ and $S^1\times I$.
\end{hint}
\begin{solution}
False.  They are homotopy equivalent but have different dimensions/boundary behavior.
\end{solution}

\begin{exercise}[Retract means deformation retract]\label{exr:viii02-ped-11}
Test: every retract is automatically a deformation retract.
\end{exercise}
\begin{hint}
A retraction alone supplies no homotopy from $i\circ r$ to the identity.
\end{hint}
\begin{solution}
False; deformation retraction is strictly stronger.
\end{solution}

\begin{exercise}[Contractible means one point]\label{exr:viii02-ped-12}
Test: every contractible space is a singleton.
\end{exercise}
\begin{hint}
Use $D^n$ or an interval.
\end{hint}
\begin{solution}
False.  Contractibility is a homotopy-type statement, not equality of underlying sets.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Model reduction]\label{exr:viii02-ped-13}
Why is replacing a space by a deformation retract useful before computing invariants?
\end{exercise}
\begin{hint}
Homotopy invariants are unchanged.
\end{hint}
\begin{solution}
The retract often has fewer cells or simpler geometry while preserving homotopy type, so later algebraic invariants can be computed on the simpler model.
\end{solution}

\begin{exercise}[Robot configuration intuition]\label{exr:viii02-ped-14}
Interpret contractibility as absence of homotopy-level obstacles in a configuration region.
\end{exercise}
\begin{hint}
Every configuration can be continuously moved to one reference configuration.
\end{hint}
\begin{solution}
A contraction supplies a coherent motion of all points toward one reference point; this does not encode metric feasibility, but it certifies trivial homotopy type.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Punctured plane]\label{exr:viii02-ped-15}
Use a deformation retraction to reduce the homotopy type of $\mathbb R^2\setminus\{0\}$.
\end{exercise}
\begin{hint}
Normalize the radius.
\end{hint}
\begin{solution}
Radial deformation retracts the punctured plane onto $S^1$, so all homotopy invariants can be computed on the circle.
\end{solution}

\begin{exercise}[Two-sided synthesis]\label{exr:viii02-ped-16}
Given explicit maps $f:X\to Y$ and $g:Y\to X$, list the exact data needed to certify a homotopy equivalence.
\end{exercise}
\begin{hint}
Maps alone are insufficient.
\end{hint}
\begin{solution}
One must provide continuous homotopies $g f\simeq\operatorname{id}_X$ and $f g\simeq\operatorname{id}_Y$, with any required based or relative conditions checked separately.
\end{solution}
