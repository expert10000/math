\section{Supplementary worked examples}
\label{sec:viii33-pedagogy}

\begin{example}[Poincare duality on a closed oriented surface]\label{ex:viii33-ped-01}
For a closed oriented surface $\Sigma_g$,
\[
H^1(\Sigma_g;\mathbb Z)\xrightarrow{\;\cap[\Sigma_g]\;}H_1(\Sigma_g;\mathbb Z)
\]
is an isomorphism. The cup-product pairing on $H^1$ therefore corresponds to the algebraic intersection pairing on one-cycles.
\end{example}

\begin{example}[Duality on a sphere]\label{ex:viii33-ped-02}
For $S^n$, cap product with the fundamental class identifies $H^0(S^n)$ with $H_n(S^n)$ and $H^n(S^n)$ with $H_0(S^n)$. The two nonzero cohomology groups are exchanged with complementary homological degrees.
\end{example}

\begin{example}[Submanifold dual classes]\label{ex:viii33-ped-03}
An oriented codimension-$k$ submanifold $N\subset M^n$ determines a class $\operatorname{PD}[N]\in H^k(M)$. For a complementary cycle $Z$, evaluating $\operatorname{PD}[N]$ on $[Z]$ gives the algebraic intersection number $N\cdot Z$ when transversality holds.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Degree reversal]\label{exr:viii33-ped-01}
For a closed oriented $n$-manifold, which homology group is dual to $H^k(M)$?
\end{exercise}
\begin{hint}
Cap with $[M]$ lowers degree by $k$.
\end{hint}
\begin{solution}
$H_{n-k}(M)$.
\end{solution}

\begin{exercise}[Fundamental class]\label{exr:viii33-ped-02}
Where does the integral fundamental class of a connected closed oriented $n$-manifold live?
\end{exercise}
\begin{hint}
Top homology.
\end{hint}
\begin{solution}
$[M]\in H_n(M;\mathbb Z)\cong\mathbb Z$.
\end{solution}

\begin{exercise}[Sphere duality]\label{exr:viii33-ped-03}
What is the Poincare dual of the point class in $H_0(S^n)$?
\end{exercise}
\begin{hint}
Complementary degree is $n$.
\end{hint}
\begin{solution}
It is a generator of $H^n(S^n)$, up to orientation sign.
\end{solution}

\begin{exercise}[Dual of whole manifold]\label{exr:viii33-ped-04}
What cohomology class is dual to $[M]$ itself?
\end{exercise}
\begin{hint}
Complementary degree is zero.
\end{hint}
\begin{solution}
The unit $1\in H^0(M)$.
\end{solution}

\begin{exercise}[Betti symmetry]\label{exr:viii33-ped-05}
What equality of Betti numbers follows over a field?
\end{exercise}
\begin{hint}
Dual groups have complementary degrees.
\end{hint}
\begin{solution}
$b_k(M)=b_{n-k}(M)$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Cap-product isomorphism]\label{exr:viii33-ped-06}
State Poincare duality using cap product.
\end{exercise}
\begin{hint}
Use the fundamental class.
\end{hint}
\begin{solution}
$\alpha\mapsto \alpha\cap[M]$ is an isomorphism $H^k(M;R)\to H_{n-k}(M;R)$ under the standard orientation/coefficient hypotheses.
\end{solution}

\begin{exercise}[Intersection from cup product]\label{exr:viii33-ped-07}
Prove the intersection pairing of complementary cycles can be computed by cup product of their dual classes.
\end{exercise}
\begin{hint}
Evaluate the product on $[M]$.
\end{hint}
\begin{solution}
If $a,b$ are complementary, then $a\cdot b=\langle \operatorname{PD}(a)\smile\operatorname{PD}(b),[M]\rangle$, with the standard sign conventions.
\end{solution}

\begin{exercise}[Duality implies top homology]\label{exr:viii33-ped-08}
Show a connected closed oriented manifold has $H_n(M;\mathbb Z)\cong\mathbb Z$.
\end{exercise}
\begin{hint}
Apply duality to $H^0(M)\cong\mathbb Z$.
\end{hint}
\begin{solution}
Poincare duality identifies $H^0(M)$ with $H_n(M)$, giving $\mathbb Z$.
\end{solution}

\begin{exercise}[Cohomology-homology rank symmetry]\label{exr:viii33-ped-09}
Explain why duality plus field coefficients forces the Poincare polynomial to be palindromic.
\end{exercise}
\begin{hint}
Use $b_k=b_{n-k}$.
\end{hint}
\begin{solution}
The coefficient of $t^k$ equals that of $t^{n-k}$, so $P_M(t)=t^nP_M(t^{-1})$.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Boundary harmless]\label{exr:viii33-ped-10}
Test: ordinary Poincare duality in the same form holds for compact manifolds with boundary.
\end{exercise}
\begin{hint}
Relative groups enter.
\end{hint}
\begin{solution}
False. One uses Poincare--Lefschetz duality, relating absolute cohomology to relative homology or vice versa.
\end{solution}

\begin{exercise}[Orientation irrelevant over Z]\label{exr:viii33-ped-11}
Test: every closed manifold has integral Poincare duality with untwisted coefficients.
\end{exercise}
\begin{hint}
Nonorientable manifolds lack an integral fundamental class.
\end{hint}
\begin{solution}
False. One needs orientability or appropriate local coefficients; mod-$2$ duality remains available.
\end{solution}

\begin{exercise}[Duality says groups equal same degree]\label{exr:viii33-ped-12}
Test: Poincare duality identifies $H^k(M)$ with $H_k(M)$.
\end{exercise}
\begin{hint}
Degrees are complementary.
\end{hint}
\begin{solution}
False in general; the homology degree is $n-k$.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Geometric query]\label{exr:viii33-ped-13}
How can a codimension-$k$ feature be stored cohomologically for repeated intersection queries?
\end{exercise}
\begin{hint}
Use its Poincare-dual class.
\end{hint}
\begin{solution}
Store $\operatorname{PD}[N]\in H^k(M)$; pairing it with varying complementary cycles returns their algebraic intersections with $N$.
\end{solution}

\begin{exercise}[Sanity check for manifolds]\label{exr:viii33-ped-14}
How can Betti-number symmetry be used to detect an error in a computed homology table of a closed oriented manifold?
\end{exercise}
\begin{hint}
Compare complementary degrees.
\end{hint}
\begin{solution}
If $b_k\ne b_{n-k}$, either the computation, manifold assumptions, or orientation/coefficient assumptions are wrong.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Surface pairing]\label{exr:viii33-ped-15}
For $T^2$, choose dual degree-one classes $a,b$. Compute $\langle a\smile b,[T^2]\rangle$ after compatible orientation choices.
\end{exercise}
\begin{hint}
Use the standard meridian-longitude basis.
\end{hint}
\begin{solution}
It is $1$; reversing the order gives $-1$ over $\mathbb Z$.
\end{solution}

\begin{exercise}[Duality synthesis]\label{exr:viii33-ped-16}
Explain how Thom classes lead to Poincare duality for embedded submanifolds.
\end{exercise}
\begin{hint}
Use the normal bundle and a tubular neighborhood.
\end{hint}
\begin{solution}
The normal Thom class extends to the ambient manifold and becomes the cohomological dual of the submanifold; cap and cup products then convert intersections of geometric cycles into algebraic pairings.
\end{solution}
