\section{Supplementary worked examples}
\label{sec:viii16-pedagogy}

\begin{example}[A two-term complex]\label{ex:viii16-ped-01}
For $C_1\cong\mathbb Z\xrightarrow{m}C_0\cong\mathbb Z$, with no other nonzero groups, $H_1=\ker(m)$ and $H_0=\mathbb Z/m$.  Thus a single integer differential creates torsion as a cokernel.
\end{example}

\begin{example}[Circle chain complex]\label{ex:viii16-ped-02}
The cellular chain complex of a circle with one 0-cell and one 1-cell is $0\to\mathbb Z\xrightarrow{0}\mathbb Z\to0$, yielding $H_1\cong\mathbb Z$ and $H_0\cong\mathbb Z$.
\end{example}

\begin{example}[Why $d^2=0$ matters]\label{ex:viii16-ped-03}
The condition $d_nd_{n+1}=0$ says every boundary is automatically a cycle, so $\operatorname{im}d_{n+1}\subseteq\ker d_n$ and the quotient defining homology is meaningful.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Cycle group]\label{exr:viii16-ped-01}
Define $Z_n(C)$ for a chain complex.
\end{exercise}
\begin{hint}
Cycles are killed by the outgoing differential.
\end{hint}
\begin{solution}
$Z_n(C)=\ker(d_n:C_n\to C_{n-1})$.
\end{solution}

\begin{exercise}[Boundary group]\label{exr:viii16-ped-02}
Define $B_n(C)$.
\end{exercise}
\begin{hint}
Boundaries come from one degree higher.
\end{hint}
\begin{solution}
$B_n(C)=\operatorname{im}(d_{n+1}:C_{n+1}\to C_n)$.
\end{solution}

\begin{exercise}[Homology]\label{exr:viii16-ped-03}
Define $H_n(C)$.
\end{exercise}
\begin{hint}
Quotient cycles by boundaries.
\end{hint}
\begin{solution}
$H_n(C)=Z_n(C)/B_n(C)$.
\end{solution}

\begin{exercise}[Zero differential]\label{exr:viii16-ped-04}
If all differentials vanish, what is $H_n(C)$?
\end{exercise}
\begin{hint}
Then every chain is a cycle and only zero is a boundary.
\end{hint}
\begin{solution}
$H_n(C)\cong C_n$.
\end{solution}

\begin{exercise}[Multiplication complex]\label{exr:viii16-ped-05}
For $0\to\mathbb Z\xrightarrow{4}\mathbb Z\to0$, compute the homology in the two nonzero degrees.
\end{exercise}
\begin{hint}
Multiplication by 4 is injective with cokernel $\mathbb Z/4$.
\end{hint}
\begin{solution}
The upper homology is $0$ and the lower is $\mathbb Z/4$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Boundaries are cycles]\label{exr:viii16-ped-06}
Prove $B_n\subseteq Z_n$.
\end{exercise}
\begin{hint}
Use $d_nd_{n+1}=0$.
\end{hint}
\begin{solution}
If $x=d_{n+1}y$, then $d_nx=d_nd_{n+1}y=0$, so $x\in\ker d_n$.
\end{solution}

\begin{exercise}[Homology functor on chain maps]\label{exr:viii16-ped-07}
Show a chain map $f:C\to D$ sends cycles to cycles and boundaries to boundaries.
\end{exercise}
\begin{hint}
Use $d_Df=fd_C$.
\end{hint}
\begin{solution}
For a cycle $z$, $d_Df(z)=f(d_Cz)=0$; for a boundary $d_Cc$, $f(d_Cc)=d_Df(c)$ is a boundary. Hence $f$ induces $H_n(f)$.
\end{solution}

\begin{exercise}[Composition]\label{exr:viii16-ped-08}
Prove induced homology maps respect composition.
\end{exercise}
\begin{hint}
Apply the maps to representatives.
\end{hint}
\begin{solution}
$H(g\circ f)[z]=[(g\circ f)(z)]=H(g)(H(f)[z])$, so homology is functorial.
\end{solution}

\begin{exercise}[Isomorphism of complexes]\label{exr:viii16-ped-09}
Prove a chain isomorphism induces isomorphisms on all homology groups.
\end{exercise}
\begin{hint}
Its inverse is also a chain map.
\end{hint}
\begin{solution}
Functoriality makes the induced maps inverse to each other on homology.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Any graded map is chain map]\label{exr:viii16-ped-10}
Test: any family $f_n:C_n\to D_n$ defines a chain map.
\end{exercise}
\begin{hint}
It must commute with differentials.
\end{hint}
\begin{solution}
False; the chain condition $d_Df=fd_C$ is essential.
\end{solution}

\begin{exercise}[Cycles equal homology]\label{exr:viii16-ped-11}
Test: $H_n(C)=Z_n(C)$ always.
\end{exercise}
\begin{hint}
Boundaries may be nonzero.
\end{hint}
\begin{solution}
False. Homology is the quotient $Z_n/B_n$.
\end{solution}

\begin{exercise}[$d^2=0$ optional]\label{exr:viii16-ped-12}
Test: the equation $d^2=0$ is merely a convenient convention.
\end{exercise}
\begin{hint}
Without it, boundaries need not be cycles.
\end{hint}
\begin{solution}
False. It is necessary for the homology quotient to be defined in the usual way.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Matrix computation]\label{exr:viii16-ped-13}
How is homology computed from finite free chain complexes in practice?
\end{exercise}
\begin{hint}
Represent differentials by integer matrices.
\end{hint}
\begin{solution}
Compute kernels and images, often via Smith normal form; free ranks and invariant factors then give the homology groups.
\end{solution}

\begin{exercise}[Error checking]\label{exr:viii16-ped-14}
Why is verifying $D_nD_{n+1}=0$ a useful implementation test for boundary matrices?
\end{exercise}
\begin{hint}
It is the chain-complex identity.
\end{hint}
\begin{solution}
A nonzero product signals inconsistent orientations/incidence or assembly errors before any homology computation is trusted.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Smith form preview]\label{exr:viii16-ped-15}
For a differential represented by diagonal Smith form $\operatorname{diag}(d_1,\ldots,d_r,0,\ldots)$, what information becomes visible?
\end{exercise}
\begin{hint}
Nonzero invariant factors control cokernel torsion; zeros control kernel rank.
\end{hint}
\begin{solution}
The diagonal immediately separates rank, free directions, and torsion factors $\mathbb Z/d_i$, enabling explicit kernel/cokernel and hence homology calculations.
\end{solution}

\begin{exercise}[Exactness synthesis]\label{exr:viii16-ped-16}
At one position $C_{n+1}\to C_n\to C_{n-1}$, relate exactness to homology.
\end{exercise}
\begin{hint}
Exactness means image equals kernel.
\end{hint}
\begin{solution}
The complex is exact at $C_n$ iff $B_n=Z_n$, which is equivalent to $H_n(C)=0$.
\end{solution}
