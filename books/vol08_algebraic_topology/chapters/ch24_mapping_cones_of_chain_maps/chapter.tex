\chapter{Mapping Cones of Chain Maps}
\label{ch:viii24-mapping-cones-chain-maps}

The mapping cone of a chain map packages the map into a new chain complex.
Its homology measures the failure of the map to be a quasi-isomorphism, while
its contractibility detects chain-homotopy equivalences.

\section{Definition}
Let \(f:C_\bullet\to D_\bullet\) be a chain map.

\begin{definition}[Mapping cone]
\label{def:viii24-cone}
Define
\[
\operatorname{Cone}(f)_n=D_n\oplus C_{n-1}
\]
with differential
\[
d(d,c)=\bigl(\partial_Dd+f(c),-\partial_Cc\bigr).
\]
\end{definition}

\section{Chain condition}
\begin{proposition}
\label{prop:viii24-square-zero}
The cone differential satisfies
\[
d^2=0.
\]
\end{proposition}

The verification uses exactly the chain-map identity
\[
\partial_Df=f\partial_C.
\]

\section{Shifted complex}
Define the suspension/shift \(C[1]\) by
\[
C[1]_n=C_{n-1},
\qquad
\partial_{C[1]}=-\partial_C.
\]
Then there is a short exact sequence
\[
0\to D\to\operatorname{Cone}(f)\to C[1]\to0.
\]

\section{Cone exact sequence}
The short exact sequence yields
\[
\cdots\to
H_n(D)\to
H_n(\operatorname{Cone}(f))
\to
H_{n-1}(C)
\xrightarrow{H_{n-1}(f)}
H_{n-1}(D)
\to\cdots.
\]

Equivalently one can arrange it as
\[
\cdots\to H_n(C)\xrightarrow{H_n(f)}H_n(D)
\to H_n(\operatorname{Cone}(f))
\to H_{n-1}(C)\to\cdots.
\]

\section{Cone of the identity}
\begin{theorem}[Identity cone contracts]
\label{thm:viii24-id-cone}
\[
\operatorname{Cone}(\operatorname{id}_C)
\]
is contractible.
\end{theorem}

\section{Cone of the zero map}
If \(f=0\), then
\[
\operatorname{Cone}(0)\cong D\oplus C[1]
\]
as chain complexes.

Thus
\[
H_n(\operatorname{Cone}(0))
\cong
H_n(D)\oplus H_{n-1}(C).
\]

\section{Quasi-isomorphisms}
\begin{definition}[Quasi-isomorphism]
\label{def:viii24-quasi}
A chain map \(f:C\to D\) is a quasi-isomorphism if
\[
H_n(f):H_n(C)\to H_n(D)
\]
is an isomorphism for every \(n\).
\end{definition}

\begin{theorem}[Cone criterion]
\label{thm:viii24-quasi-cone}
A chain map \(f\) is a quasi-isomorphism if and only if
\[
H_n(\operatorname{Cone}(f))=0
\]
for all \(n\).
\end{theorem}

\section{Chain-homotopy equivalences}
\begin{theorem}[Homotopy-equivalence cone]
\label{thm:viii24-homotopy-cone}
If \(f\) is a chain-homotopy equivalence, then
\[
\operatorname{Cone}(f)
\]
is contractible.
\end{theorem}

Thus chain-homotopy equivalence is stronger than quasi-isomorphism, matching
contractible versus merely acyclic cones.

\section{Cone of an inclusion}
If \(i:A\hookrightarrow B\) is an inclusion of chain complexes, the cone
provides a chain model closely related to the quotient \(B/A\).

Under suitable splitting/cofibration-style algebraic hypotheses, the cone and
quotient are quasi-isomorphic.

\section{Relative chains}
For a subspace \(A\subset X\), the inclusion
\[
C_*(A)\to C_*(X)
\]
has a cone whose homology is naturally related to relative homology
\[
H_*(X,A).
\]

This mirrors the topological mapping-cone picture from VIII/07.

\section{Topological versus algebraic cones}
A map of spaces
\[
f:A\to X
\]
has a topological mapping cone \(C_f\).  Applying singular chains gives a chain
complex naturally compared with
\[
\operatorname{Cone}(C_*(f)).
\]
The two constructions encode the same cofiber geometry up to the expected
chain equivalences.

\section{Functoriality}
A commutative square of chain maps induces a chain map between mapping cones.

\section{Homotopy-commutative squares}
If a square commutes only up to chain homotopy, a corrected cone map can still
be built by including the homotopy term.  This is one reason mapping cones are
central in derived constructions.

\section{Connecting morphism}
The cone exact sequence realizes the map
\[
H_{n-1}(C)\to H_{n-1}(D)
\]
as the homology map induced by \(f\).  Thus the cone is a universal algebraic
device for measuring this map.

\section{Two-term example}
For
\[
f:\mathbb Z\to\mathbb Z,\qquad f=\times m
\]
with both complexes concentrated in degree \(0\), the cone is the two-term
complex
\[
0\to\mathbb Z\xrightarrow{\times m}\mathbb Z\to0.
\]
Hence for \(m\ne0\),
\[
H_0(\operatorname{Cone}(f))\cong\mathbb Z/m.
\]

\section{Moore-space connection}
The multiplication-by-\(m\) cone is the algebraic analogue of the degree-\(m\)
cell attachment producing a Moore space.  Both produce \(\mathbb Z/m\) torsion.

\section{Cone of an isomorphism}
If \(f\) is a strict chain isomorphism, its cone is contractible.  This is the
strict special case of the homotopy-equivalence theorem.

\section{Cone and exactness}
If
\[
0\to C\xrightarrow{f}D\to Q\to0
\]
is a short exact sequence, the cone of \(f\) is frequently quasi-isomorphic to
the quotient complex \(Q\).

\section{Distinguished-triangle preview}
The pattern
\[
C\xrightarrow f D\to\operatorname{Cone}(f)\to C[1]
\]
is the prototype of a distinguished triangle.  Derived categories formalize
this pattern, but no derived-category machinery is needed in this volume.

\section{Why signs matter}
The minus sign in the shifted differential is essential.  Without it, the cone
differential would generally fail to square to zero.

\section{Bridge to coefficients}
The next chapter changes coefficient groups.  Mapping-cone calculations remain
valuable there because tensoring a cone produces the expected cone of the
tensored map, subject to standard exactness considerations.

\section{Solved problem dossiers}

\begin{problem}[Square-zero check]\label{prob:viii24-01}
Verify the cone differential squares to zero.
\end{problem}
\begin{solution}
Compute \(d^2(d,c)\): the first component is \(\partial_D^2d+\partial_Df(c)-f\partial_Cc=0\) by the chain-map identity; the second is \(\partial_C^2c=0\).
\end{solution}

\begin{problem}[Identity cone]\label{prob:viii24-02}
Explain why \(\operatorname{Cone}(\mathrm{id})\) contracts.
\end{problem}
\begin{solution}
An explicit degree-one operator swaps the shifted and unshifted components appropriately; equivalently the cone models an algebraic cylinder pair and is chain-homotopy equivalent to zero.
\end{solution}

\begin{problem}[Zero cone]\label{prob:viii24-03}
Compute the homology of \(\operatorname{Cone}(0)\).
\end{problem}
\begin{solution}
The differential is block diagonal, so the cone is \(D\oplus C[1]\) and \(H_n\cong H_n(D)\oplus H_{n-1}(C)\).
\end{solution}

\begin{problem}[Quasi-isomorphism criterion]\label{prob:viii24-04}
Prove the cone criterion using the long exact sequence.
\end{problem}
\begin{solution}
Exactness shows all cone homology vanishes exactly when every \(H_n(C)\to H_n(D)\) is both injective and surjective.
\end{solution}

\begin{problem}[Homotopy equivalence]\label{prob:viii24-05}
Why does chain-homotopy equivalence imply contractible cone?
\end{problem}
\begin{solution}
A homotopy inverse and the two chain homotopies can be assembled into an explicit cone contraction.
\end{solution}

\begin{problem}[Strict isomorphism]\label{prob:viii24-06}
Deduce the cone of a chain isomorphism contracts.
\end{problem}
\begin{solution}
A strict isomorphism is a chain-homotopy equivalence with zero homotopies.
\end{solution}

\begin{problem}[Multiplication by \(m\)]\label{prob:viii24-07}
Compute the cone of \(\times m:\mathbb Z\to\mathbb Z\) concentrated in degree zero.
\end{problem}
\begin{solution}
It is \(0\to\mathbb Z\xrightarrow{\times m}\mathbb Z\to0\), with \(H_0=\mathbb Z/m\) and \(H_1=0\) for \(m\ne0\).
\end{solution}

\begin{problem}[Case \(m=0\)]\label{prob:viii24-08}
Compute the same cone for \(m=0\).
\end{problem}
\begin{solution}
The differential is zero, so \(H_1\cong\mathbb Z\) and \(H_0\cong\mathbb Z\).
\end{solution}

\begin{problem}[Case \(m=1\)]\label{prob:viii24-09}
What happens for \(m=1\)?
\end{problem}
\begin{solution}
The differential is an isomorphism, so the cone is acyclic and in fact contractible.
\end{solution}

\begin{problem}[Relative model]\label{prob:viii24-10}
Relate the cone of \(C_*(A)\to C_*(X)\) to \(H_*(X,A)\).
\end{problem}
\begin{solution}
The cone is a standard chain model quasi-isomorphic to the quotient relative complex, up to the conventional degree arrangement.
\end{solution}

\begin{problem}[Topological cone]\label{prob:viii24-11}
Why should singular chains on a topological mapping cone compare with the algebraic cone?
\end{problem}
\begin{solution}
Both encode the same cofiber construction: one geometrically before applying chains, the other algebraically after applying the chain functor.
\end{solution}

\begin{problem}[Map of cones]\label{prob:viii24-12}
Given a strictly commutative square, define the obvious map on cones.
\end{problem}
\begin{solution}
Apply the target map on the \(D\)-summand and the source map on the shifted \(C\)-summand; commutativity makes it a chain map.
\end{solution}

\begin{problem}[Homotopy square]\label{prob:viii24-13}
Why is an extra term needed if the square commutes only up to chain homotopy?
\end{problem}
\begin{solution}
The failure of strict commutation appears in the cone differential; the chosen homotopy supplies exactly the correction needed to cancel it.
\end{solution}

\begin{problem}[Quotient comparison]\label{prob:viii24-14}
When does the cone of an inclusion resemble the quotient complex?
\end{problem}
\begin{solution}
For a suitably split inclusion, the cone contains a contractible copy of the subcomplex and reduces up to chain homotopy/quasi-isomorphism to the quotient.
\end{solution}

\begin{problem}[Acyclic versus contractible]\label{prob:viii24-15}
Why can an acyclic cone fail to be contractible?
\end{problem}
\begin{solution}
Acyclicity detects only quasi-isomorphism of \(f\); contractibility of the cone requires the stronger chain-homotopy-equivalence property.
\end{solution}

\begin{problem}[Moore analogy]\label{prob:viii24-16}
Explain the algebraic/topological parallel for multiplication by \(m\).
\end{problem}
\begin{solution}
The chain cone has \(\mathbb Z/m\) homology, matching the cellular chain complex of a Moore space built by degree-\(m\) attachment.
\end{solution}

\begin{problem}[Triangle]\label{prob:viii24-17}
Why is the cone sequence naturally written \(C\to D\to\operatorname{Cone}(f)\to C[1]\)?
\end{problem}
\begin{solution}
The inclusion of \(D\) and projection to the shifted \(C\)-summand form the short exact cone sequence and its associated cofiber pattern.
\end{solution}

\begin{problem}[Sign]\label{prob:viii24-18}
Show what fails if the shifted differential had no minus sign.
\end{problem}
\begin{solution}
The cross terms in \(d^2\) would add instead of canceling, leaving generally \(2f\partial_C\) or equivalent nonzero terms depending on convention.
\end{solution}

\begin{problem}[Homology map failure]\label{prob:viii24-19}
What does nonzero \(H_n(\operatorname{Cone}(f))\) tell you?
\end{problem}
\begin{solution}
It witnesses failure of \(H_*(f)\) to be an isomorphism in adjacent degrees, as organized by the cone long exact sequence.
\end{solution}

\begin{problem}[Coefficient bridge]\label{prob:viii24-20}
Why are cones useful with changed coefficients?
\end{problem}
\begin{solution}
They convert the question whether a coefficient-changed map remains a quasi-isomorphism into acyclicity of the corresponding coefficient-changed cone.
\end{solution}

\begin{problem}[Derived preview]\label{prob:viii24-21}
What later formalism abstracts mapping-cone triangles?
\end{problem}
\begin{solution}
Derived and triangulated categories, though this volume only uses the concrete chain-complex construction.
\end{solution}

\begin{problem}[Final principle]\label{prob:viii24-22}
State the main diagnostic role of a cone.
\end{problem}
\begin{solution}
The cone measures the failure of a chain map to be an equivalence: acyclic cone means quasi-isomorphism; contractible cone means chain-homotopy equivalence.
\end{solution}
