\section{Supplementary worked examples}
\label{sec:viii31-pedagogy}

\begin{example}[Thom class of a trivial oriented bundle]\label{ex:viii31-ped-01}
For the oriented trivial rank-$r$ bundle $\pi:B\times\mathbb R^r\to B$, the Thom class is the fiberwise orientation class
\[
u\in H^r(B\times D^r,\;B\times S^{r-1}).
\]
Cup product with $u$ gives the Thom isomorphism $H^k(B)\cong H^{k+r}(D(E),S(E))$.
\end{example}

\begin{example}[The zero section under the Thom isomorphism]\label{ex:viii31-ped-02}
The zero section $s:B\to E$ pulls the Thom class back to the Euler class:
\[
e(E)=s^*u.
\]
Thus the same class that generates relative cohomology in every fiber also measures the obstruction to moving the zero section off itself.
\end{example}

\begin{example}[Why orientation is required]\label{ex:viii31-ped-03}
A Thom class over $\mathbb Z$ must restrict to a chosen generator of $H^r(D^r,S^{r-1};\mathbb Z)$ in every fiber. If orientations reverse around a loop, these local generators cannot be chosen coherently; twisted coefficients or mod-$2$ coefficients repair this failure.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Degree of Thom class]\label{exr:viii31-ped-01}
What is the degree of the Thom class of an oriented rank-$r$ real bundle?
\end{exercise}
\begin{hint}
Look at the top relative cohomology of the fiber disk pair.
\end{hint}
\begin{solution}
It lies in degree $r$.
\end{solution}

\begin{exercise}[Fiber restriction]\label{exr:viii31-ped-02}
To what class must a Thom class restrict on each fiber pair $(D^r,S^{r-1})$?
\end{exercise}
\begin{hint}
Use the orientation generator.
\end{hint}
\begin{solution}
It restricts to a generator of $H^r(D^r,S^{r-1})\cong\mathbb Z$.
\end{solution}

\begin{exercise}[Thom shift]\label{exr:viii31-ped-03}
What degree shift appears in the Thom isomorphism?
\end{exercise}
\begin{hint}
Cup with a degree-$r$ class.
\end{hint}
\begin{solution}
$H^k(B)\cong H^{k+r}(D(E),S(E))$.
\end{solution}

\begin{exercise}[Euler pullback]\label{exr:viii31-ped-04}
Express the Euler class in terms of the Thom class and zero section.
\end{exercise}
\begin{hint}
Pull back along $s:B\to E$.
\end{hint}
\begin{solution}
$e(E)=s^*u$.
\end{solution}

\begin{exercise}[Trivial bundle Euler]\label{exr:viii31-ped-05}
What is the Euler class of a trivial oriented positive-rank bundle?
\end{exercise}
\begin{hint}
It has a nowhere-zero section.
\end{hint}
\begin{solution}
It vanishes.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Thom map is module map]\label{exr:viii31-ped-06}
Show the Thom isomorphism is given by $\alpha\mapsto\pi^*\alpha\smile u$.
\end{exercise}
\begin{hint}
Check degrees and fiber restriction.
\end{hint}
\begin{solution}
The pullback carries $\alpha$ to the total space and cup product with the Thom class supplies the fiber orientation, giving the canonical degree-shifted isomorphism.
\end{solution}

\begin{exercise}[Naturality]\label{exr:viii31-ped-07}
Explain the naturality of Thom classes under orientation-preserving pullback.
\end{exercise}
\begin{hint}
Pull back the bundle and its disk/sphere pair.
\end{hint}
\begin{solution}
For $f:B'\to B$, the Thom class of $f^*E$ is the pullback of $u_E$ under the induced map of Thom pairs, provided orientations are pulled back compatibly.
\end{solution}

\begin{exercise}[Uniqueness]\label{exr:viii31-ped-08}
Why is an integral Thom class unique once the orientation is fixed?
\end{exercise}
\begin{hint}
Compare two candidates fiberwise and use the Thom isomorphism in degree zero.
\end{hint}
\begin{solution}
Their difference restricts trivially to every fiber; the relative cohomology group is a free rank-one $H^0(B)$-module on each connected component, fixing the class by the chosen orientation.
\end{solution}

\begin{exercise}[Orientation obstruction]\label{exr:viii31-ped-09}
Explain why a nonorientable real bundle need not have an integral Thom class.
\end{exercise}
\begin{hint}
Transport a fiber generator around an orientation-reversing loop.
\end{hint}
\begin{solution}
The generator returns with opposite sign, so no globally coherent integral fiber orientation class exists.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Every bundle has integral Thom class]\label{exr:viii31-ped-10}
Test: every real vector bundle has a Thom class over $\mathbb Z$.
\end{exercise}
\begin{hint}
Orientability is the obstruction.
\end{hint}
\begin{solution}
False. Nonorientable bundles require twisted coefficients or a coefficient system such as $\mathbb F_2$.
\end{solution}

\begin{exercise}[Thom class lives in ordinary cohomology of E]\label{exr:viii31-ped-11}
Test: the Thom class is naturally an element of $H^r(E)$.
\end{exercise}
\begin{hint}
Boundary behavior matters.
\end{hint}
\begin{solution}
False in the disk-bundle model; it is naturally a relative class in $H^r(D(E),S(E))$, equivalently compactly supported in the total space.
\end{solution}

\begin{exercise}[No rank shift]\label{exr:viii31-ped-12}
Test: the Thom isomorphism preserves cohomological degree.
\end{exercise}
\begin{hint}
Cup with $u$.
\end{hint}
\begin{solution}
False. It shifts degree upward by the bundle rank.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Intersection localization]\label{exr:viii31-ped-13}
How does the Thom class turn a submanifold into a cohomology class of the ambient manifold?
\end{exercise}
\begin{hint}
Use a tubular neighborhood and extend the Thom class.
\end{hint}
\begin{solution}
The normal bundle Thom class is transferred through a tubular neighborhood to a class supported near the submanifold, yielding its Poincare dual.
\end{solution}

\begin{exercise}[Zero-locus counting]\label{exr:viii31-ped-14}
How can the Thom/Euler formalism count zeros of a transverse section?
\end{exercise}
\begin{hint}
Intersect the section with the zero section.
\end{hint}
\begin{solution}
A transverse section pulls the Thom class back to the Euler class; evaluating it on the base fundamental class gives the signed number of zeros.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Trivial rank-r bundle]\label{exr:viii31-ped-15}
Write the Thom isomorphism for $B\times\mathbb R^r$ explicitly.
\end{exercise}
\begin{hint}
Use the generator $u$ of the fiber pair.
\end{hint}
\begin{solution}
$\alpha\mapsto \operatorname{pr}_B^*\alpha\smile u$ identifies $H^k(B)$ with $H^{k+r}(B\times D^r,B\times S^{r-1})$.
\end{solution}

\begin{exercise}[Thom synthesis]\label{exr:viii31-ped-16}
Explain how one class links orientation, relative cohomology, Euler classes, and Poincare duality.
\end{exercise}
\begin{hint}
Follow the Thom class through restriction, pullback, and tubular neighborhoods.
\end{hint}
\begin{solution}
The Thom class is the coherent fiber orientation generator; cup product with it gives the Thom isomorphism, zero-section pullback gives Euler class, and transfer from normal bundles produces Poincare-dual classes of submanifolds.
\end{solution}
