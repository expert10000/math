
\chapter{Thom Classes}
\label{ch:viii31-thom-classes}

Thom classes turn the local orientation data of a vector bundle into a global
cohomology class.  They are the bridge from vector bundles to Euler classes,
Gysin maps, Poincaré duality, and intersection theory.

\section{Disk and sphere bundles}
Let $\pi:E\to B$ be a rank-$r$ real vector bundle equipped with a bundle
metric. Define
\[
D(E)=\{v\in E:\|v\|\le1\},\qquad S(E)=\{v\in E:\|v\|=1\}.
\]
The pair $(D(E),S(E))$ is the natural relative space in which the Thom class
lives.

\section{Fiber orientation}
An orientation of $E$ is a continuous choice of orientation on every fiber.
For each $b\in B$,
\[
(D(E_b),S(E_b))\cong(D^r,S^{r-1}),\qquad
H^r(D^r,S^{r-1};R)\cong R.
\]

\begin{definition}[Thom class]
\label{def:viii31-thom-class}
For an oriented rank-$r$ vector bundle $E\to B$, a Thom class is
\[
u_E\in H^r(D(E),S(E);R)
\]
whose restriction to every oriented fiber pair is the chosen generator.
\end{definition}

\section{The Thom isomorphism}
\begin{theorem}[Thom isomorphism]
\label{thm:viii31-thom-isomorphism}
For an oriented rank-$r$ vector bundle over a suitable paracompact base,
\[
\Phi_E:H^k(B;R)\longrightarrow H^{k+r}(D(E),S(E);R),
\qquad
\alpha\longmapsto \pi^*\alpha\smile u_E
\]
is an isomorphism.
\end{theorem}

\section{Why relative cohomology appears}
The sphere bundle is the fiberwise boundary of the disk bundle.  Relative
cohomology records the transverse orientation class while forgetting boundary
data.

\section{The zero section}
The zero section $s:B\to E$ is a homotopy inverse to the projection of the disk
bundle, but the relative pair remembers the fiber orientation.

\section{Naturality}
For $f:Y\to B$, the pulled-back bundle satisfies
\[
\widetilde f^{\,*}(u_E)=u_{f^*E}.
\]
\begin{theorem}[Naturality]
\label{thm:viii31-thom-naturality}
The square formed by pullback on base cohomology and pullback on relative
cohomology commutes with the two Thom isomorphisms.
\end{theorem}

\section{Compactly supported formulation}
Under the standard local-compactness hypotheses,
\[
H^{k+r}(D(E),S(E);R)\cong H_c^{k+r}(E;R),
\]
hence
\[
H^k(B;R)\cong H_c^{k+r}(E;R).
\]

\section{Trivial bundles and suspension}
For $E=B\times\mathbb R^r$,
\[
\operatorname{Th}(E)=D(E)/S(E)\simeq B_+\wedge S^r,
\]
so the Thom shift becomes an $r$-fold suspension phenomenon.

\section{Orientation reversal}
Reversing the orientation negates the integral Thom class.  Over
$\mathbb Z/2$, the sign disappears.

\section{Normal bundles and tubular neighborhoods}
If $i:N\hookrightarrow M$ is a closed embedded submanifold with oriented normal
bundle $\nu$, a tubular neighborhood identifies a neighborhood of $N$ with
$D(\nu)$.  The normal Thom class is therefore a cohomological representative
concentrated transversely to $N$.

\section{The Thom space and collapse}
\begin{definition}[Thom space]
\label{def:viii31-thom-space}
\[
\operatorname{Th}(E)=D(E)/S(E).
\]
\end{definition}
Collapsing the complement of a tubular neighborhood gives the Thom collapse
\[
M\longrightarrow \operatorname{Th}(\nu).
\]

\section{Gysin viewpoint}
Cup product with a Thom class, followed by excision and extension, produces
wrong-way maps in cohomology.  This is the local mechanism behind geometric
duality.

\section{From Thom to Euler}
Pulling the Thom class back by the zero section gives
\[
e(E)=s^*(u_E)\in H^r(B;R).
\]
This is the central bridge to VIII/32.

\section{From normal Thom classes to dual cycles}
For an oriented embedded submanifold, the normal Thom class extends to ambient
cohomology and becomes the class Poincaré dual to that submanifold.  This is
the bridge to VIII/33 and VIII/34.

\section{Solved dossiers}
\begin{problem}
Compute the relative cohomology of a fiber pair $(D^r,S^{r-1})$.
\end{problem}
\begin{solution}
It is $\mathbb Z$ in degree $r$ and zero otherwise, with the generator fixed by orientation.
\end{solution}

\begin{problem}
Why is the Thom class naturally relative?
\end{problem}
\begin{solution}
The sphere bundle is the fiberwise boundary; relative cohomology retains the orientation class of the disk fiber while killing boundary data.
\end{solution}

\begin{problem}
Find the Thom class for a trivial oriented line bundle.
\end{problem}
\begin{solution}
It is the pullback of the generator of $H^1(D^1,S^0)$ along the fiber factor.
\end{solution}

\begin{problem}
State the rank-five Thom shift.
\end{problem}
\begin{solution}
$H^k(B)\cong H^{k+5}(D(E),S(E))$.
\end{solution}

\begin{problem}
What happens under orientation reversal?
\end{problem}
\begin{solution}
The integral Thom class changes sign.
\end{solution}

\begin{problem}
Why do $\mathbb Z/2$ coefficients remove the orientability requirement?
\end{problem}
\begin{solution}
The two signs agree modulo two.
\end{solution}

\begin{problem}
Describe the Thom space of a trivial rank-$r$ bundle.
\end{problem}
\begin{solution}
It is naturally $B_+\wedge S^r$.
\end{solution}

\begin{problem}
What role does the zero section play?
\end{problem}
\begin{solution}
It identifies the base in the total space and pulls the Thom class back to the Euler class.
\end{solution}

\begin{problem}
What does a tubular neighborhood provide?
\end{problem}
\begin{solution}
It identifies a neighborhood of an embedded submanifold with the disk bundle of its normal bundle.
\end{solution}

\begin{problem}
Define the Thom collapse.
\end{problem}
\begin{solution}
Collapse the complement of a tubular neighborhood to the basepoint of the Thom space.
\end{solution}

\begin{problem}
State naturality under $f:Y\to B$.
\end{problem}
\begin{solution}
For the pulled-back orientation, $\widetilde f^{\,*}u_E=u_{f^*E}$.
\end{solution}

\begin{problem}
How is compact support related to the Thom pair?
\end{problem}
\begin{solution}
Under standard hypotheses, $H^*(D(E),S(E))\cong H_c^*(E)$.
\end{solution}

\begin{problem}
Where does the bundle rank appear?
\end{problem}
\begin{solution}
As the cohomological degree shift.
\end{solution}

\begin{problem}
How does the Thom class encode normal orientation?
\end{problem}
\begin{solution}
Its restriction to each normal disk fiber is the chosen orientation generator.
\end{solution}

\begin{problem}
What is the prototype of a cohomological wrong-way map?
\end{problem}
\begin{solution}
Cup with the Thom class, followed by excision and extension from a tubular neighborhood.
\end{solution}

\begin{problem}
Relate the Thom space to reduced cohomology.
\end{problem}
\begin{solution}
$\widetilde H^{k+r}(\operatorname{Th}(E))\cong H^k(B)$.
\end{solution}

\begin{problem}
Why is a bundle metric useful?
\end{problem}
\begin{solution}
It defines disk and sphere subbundles without changing the underlying homotopy-theoretic data.
\end{solution}

\begin{problem}
What characteristic class comes next?
\end{problem}
\begin{solution}
$e(E)=s^*u_E$.
\end{solution}

\begin{problem}
How does a submanifold acquire a dual cohomology class?
\end{problem}
\begin{solution}
Transfer the Thom class of its oriented normal bundle into the ambient manifold.
\end{solution}

\begin{problem}
Why is the Thom theorem a bridge theorem?
\end{problem}
\begin{solution}
It connects bundle orientation, relative cohomology, embeddings, Gysin maps, Euler classes, and duality.
\end{solution}

\section{Exercises, Hints, and Solutions}
\begin{exercise}\label{exr:viii31-01}
Compute $H^*(D^r,S^{r-1};\mathbb Z)$.
\end{exercise}
\begin{hint}
Use $D^r/S^{r-1}\cong S^r$.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-01 append
For positive rank, use reduced cohomology of the quotient and identify the single nonzero degree; choose its generator by an orientation of the disk.
\end{hint}
\begin{solution}
Since \(D^r/S^{r-1}\cong S^r\), relative cohomology is reduced cohomology of \(S^r\):
\[
H^q(D^r,S^{r-1};\mathbb Z)\cong
\begin{cases}\mathbb Z,&q=r,\\0,&q\ne r.\end{cases}
\]
The generator is the orientation class of the disk fiber.
\end{solution}

\begin{exercise}\label{exr:viii31-02}
Write the Thom isomorphism for a rank-three bundle over $S^2$.
\end{exercise}
\begin{hint}
Shift degree by the rank.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-02 append
Use integral coefficients and an orientation of the rank-three bundle; list the nonzero base degrees before adding three to each.
\end{hint}
\begin{solution}
For a rank-three oriented bundle \(E\to S^2\),
\[
\Phi_E:H^k(S^2)\xrightarrow{\cong}H^{k+3}(D(E),S(E)),
\qquad \alpha\mapsto\pi^*\alpha\smile u_E.
\]
Thus \(1\in H^0(S^2)\) maps to \(u_E\in H^3\), while the degree-two generator maps to a degree-five relative class.
\end{solution}

\begin{exercise}\label{exr:viii31-03}
Check the fiber normalization for a trivial oriented line bundle.
\end{exercise}
\begin{hint}
Restrict to each fiber pair.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-03 append
On \((D^1,S^0)\), choose the generator agreeing with the fiber's orientation and check that the same choice is made continuously over the base.
\end{hint}
\begin{solution}
For \(E=B\times\mathbb R\), the fiber pair over \(b\) is \((D^1,S^0)\). Choose \(u_E\) so that
\[
u_E|_{(D^1,S^0)}=1\in H^1(D^1,S^0;\mathbb Z).
\]
This fiberwise generator condition is exactly the Thom normalization.
\end{solution}

\begin{exercise}\label{exr:viii31-04}
Show that reversing orientation changes the integral Thom class by a sign.
\end{exercise}
\begin{hint}
Track the chosen generator.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-04 append
Restrict both candidate classes to a fiber and compare their generators; uniqueness of the normalized Thom class then determines the global sign.
\end{hint}
\begin{solution}
Reversing the orientation reverses the chosen generator of every fiber group \(H^r(D^r,S^{r-1};\mathbb Z)\). By uniqueness of the normalized class, the new Thom class is \(-u_E\).
\end{solution}

\begin{exercise}\label{exr:viii31-05}
Explain why mod-$2$ Thom theory does not require orientability.
\end{exercise}
\begin{hint}
Reduce signs modulo two.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-05 append
Compare local generators on overlapping trivializations. The transition can change their signs, but that change is invisible in \(\mathbb F_2\).
\end{hint}
\begin{solution}
The only ambiguity in an integral fiber generator is its sign. In \(\mathbb Z/2\), \(1=-1\), so local generators glue without an orientation choice. Hence every real vector bundle has a mod-two Thom class.
\end{solution}

\begin{exercise}\label{exr:viii31-06}
Identify the Thom space of a trivial rank-$r$ bundle.
\end{exercise}
\begin{hint}
Use $B_+\wedge S^r$.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-06 append
Write the quotient \((B\times D^r)/(B\times S^{r-1})\); the extra basepoint in \(B_+\) ensures the entire sphere subbundle is collapsed to one point.
\end{hint}
\begin{solution}
For \(E=B\times\mathbb R^r\),
\[
\operatorname{Th}(E)=D(E)/S(E)\cong B_+\wedge(D^r/S^{r-1})
\cong B_+\wedge S^r.
\]
\end{solution}

\begin{exercise}\label{exr:viii31-07}
Compute its reduced cohomology when the base is $S^1$.
\end{exercise}
\begin{hint}
Use suspension.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-07 append
Use the rank shift on ordinary \(H^*(S^1)\), retaining its degree-zero class; dropping that class would lose one of the two reduced Thom-space groups.
\end{hint}
\begin{solution}
With \(B=S^1\),
\[
\operatorname{Th}(E)\simeq S^1_+\wedge S^r\simeq S^r\vee S^{r+1}.
\]
Therefore its reduced integral cohomology is \(\mathbb Z\) in degrees \(r\) and \(r+1\), and zero otherwise.
\end{solution}

\begin{exercise}\label{exr:viii31-08}
Describe the pullback Thom class under a map $f:Y\to B$.
\end{exercise}
\begin{hint}
Compute the pulled-back bundle first.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-08 append
Use the pullback metric and orientation so that the bundle map preserves disk/sphere pairs, then check the fiber-generator normalization.
\end{hint}
\begin{solution}
If \(f:Y\to B\), the bundle map \(\widetilde f:D(f^*E)\to D(E)\) covering \(f\) pulls the Thom class back:
\[
u_{f^*E}=\widetilde f^{\,*}u_E.
\]
Its restriction to every pulled-back fiber is the same oriented generator.
\end{solution}

\begin{exercise}\label{exr:viii31-09}
Verify naturality of the Thom isomorphism.
\end{exercise}
\begin{hint}
Apply naturality of cup products.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-09 append
Compare \(\widetilde f^{\,*}\pi^*\alpha\) with \((\pi')^*f^*\alpha\), and use multiplicativity of pullback on the product with the Thom class.
\end{hint}
\begin{solution}
For \(\alpha\in H^k(B)\),
\[
\widetilde f^{\,*}(\pi^*\alpha\smile u_E)
=(\pi')^*(f^*\alpha)\smile u_{f^*E}.
\]
Thus pullback commutes with the two Thom isomorphisms.
\end{solution}

\begin{exercise}\label{exr:viii31-10}
Describe the normal bundle of an equatorial $S^{n-1}\subset S^n$.
\end{exercise}
\begin{hint}
The normal direction is one-dimensional.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-10 append
Embed the equator as \(x_{n+1}=0\) in the unit sphere; the constant last-coordinate direction is tangent to \(S^n\) along the equator and normal to it.
\end{hint}
\begin{solution}
The equator \(S^{n-1}\subset S^n\) has a one-dimensional normal bundle. A continuous unit normal can be chosen pointing toward one hemisphere, so this normal line bundle is trivial:
\[
\nu\cong S^{n-1}\times\mathbb R.
\]
\end{solution}

\begin{exercise}\label{exr:viii31-11}
Construct a tubular neighborhood in that example.
\end{exercise}
\begin{hint}
Use a product collar or exponential map.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-11 append
For small \(t\), test \((x,t)\mapsto(\cos t\,x,\sin t)\); check injectivity and identify the equator at \(t=0\).
\end{hint}
\begin{solution}
Using the normal coordinate, a tubular neighborhood is
\[
S^{n-1}\times(-\varepsilon,\varepsilon)\longrightarrow S^n,
\]
with \(S^{n-1}\times\{0\}\) identified with the equator. Radial/geodesic normal displacement gives the embedding.
\end{solution}

\begin{exercise}\label{exr:viii31-12}
Describe the corresponding Thom collapse map.
\end{exercise}
\begin{hint}
Collapse the complement.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-12 append
Choose a closed tubular disk neighborhood and collapse its boundary together with the exterior; the identified boundary is the sphere subbundle in the Thom quotient.
\end{hint}
\begin{solution}
Collapse the complement of the chosen tubular neighborhood to one point, and identify the neighborhood with \(D(\nu)\). This gives
\[
S^n\longrightarrow D(\nu)/S(\nu)=\operatorname{Th}(\nu).
\]
\end{solution}

\begin{exercise}\label{exr:viii31-13}
Relate relative cohomology of $(D(E),S(E))$ to the Thom quotient.
\end{exercise}
\begin{hint}
Use reduced cohomology of a quotient.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-13 append
Apply the relative-to-reduced quotient theorem to the good disk/sphere pair; the resulting class belongs to reduced rather than ordinary quotient cohomology.
\end{hint}
\begin{solution}
For any good pair,
\[
H^q(D(E),S(E))\cong \widetilde H^q(D(E)/S(E)).
\]
Since \(D(E)/S(E)=\operatorname{Th}(E)\), relative Thom cohomology is exactly reduced cohomology of the Thom quotient.
\end{solution}

\begin{exercise}\label{exr:viii31-14}
Explain the compact-support formulation for $E=\mathbb R^r$.
\end{exercise}
\begin{hint}
Compactify the fiber direction.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-14 append
Here the base is a point, so compactifying the fiber is the same as one-point compactification of the total space; identify the added point with infinity.
\end{hint}
\begin{solution}
For the bundle \(\mathbb R^r\to\{\ast\}\), one-point compactification gives \(S^r\). Hence
\[
H_c^q(\mathbb R^r;\mathbb Z)\cong\widetilde H^q(S^r;\mathbb Z),
\]
which is \(\mathbb Z\) in degree \(r\) and zero otherwise.
\end{solution}

\begin{exercise}\label{exr:viii31-15}
Track the degree shift in a rank-five example.
\end{exercise}
\begin{hint}
Write the rank explicitly.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-15 append
Write a general base degree \(k\), add the rank five, and keep the result in relative bundle cohomology rather than in the base group.
\end{hint}
\begin{solution}
For a rank-five oriented bundle,
\[
H^k(B)\xrightarrow{\cong}H^{k+5}(D(E),S(E)).
\]
For example a class in \(H^2(B)\) corresponds to a relative class in degree \(7\).
\end{solution}

\begin{exercise}\label{exr:viii31-16}
Compare Thom classes obtained from two bundle metrics.
\end{exercise}
\begin{hint}
Metrics are homotopic through positive-definite metrics.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-16 append
Interpolate the two positive-definite metrics and use fiberwise radial rescaling; verify that it preserves fiber orientations and the boundary sphere pairs.
\end{hint}
\begin{solution}
Two bundle metrics are joined by a continuous path of positive-definite metrics. Fiberwise radial rescaling identifies their disk/sphere pairs through an isotopy, and the normalized fiber generator is preserved. Thus the induced Thom classes correspond.
\end{solution}

\begin{exercise}\label{exr:viii31-17}
Show that an orientation-preserving bundle isomorphism preserves Thom classes.
\end{exercise}
\begin{hint}
Use the fiber orientation generator.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-17 append
Pull back the target class and restrict to each source fiber; orientation preservation gives the same generator, which determines the class uniquely.
\end{hint}
\begin{solution}
An orientation-preserving bundle isomorphism sends each oriented fiber generator to the oriented fiber generator. Pullback of the target Thom class therefore satisfies the defining normalization, so by uniqueness it equals the source Thom class.
\end{solution}

\begin{exercise}\label{exr:viii31-18}
Analyze an orientation-reversing bundle isomorphism.
\end{exercise}
\begin{hint}
Orientation reversal changes its sign.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-18 append
Track the sign of the induced map on \(H^r(D^r,S^{r-1};\mathbb Z)\); in mod-two coefficients the two possibilities become equal.
\end{hint}
\begin{solution}
An orientation-reversing bundle isomorphism sends the fiber orientation generator to its negative. Consequently the pulled-back integral Thom class is \(-u_E\); modulo two there is no sign change.
\end{solution}

\begin{exercise}\label{exr:viii31-19}
Describe the first step in the Gysin construction for an embedding.
\end{exercise}
\begin{hint}
Start inside a tubular neighborhood.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-19 append
For a closed oriented embedding of codimension \(r\), first multiply the pulled-back base class by the normal Thom class, obtaining relative degree increased by \(r\).
\end{hint}
\begin{solution}
For an oriented embedding \(i:N\hookrightarrow M\), first choose a tubular neighborhood \(D(\nu)\subset M\) of the normal bundle. Cup with \(u_\nu\) on that neighborhood, then use excision and extension to ambient cohomology. This is the geometric core of the Gysin map.
\end{solution}

\begin{exercise}\label{exr:viii31-20}
Explain how the zero section meets every fiber disk.
\end{exercise}
\begin{hint}
The zero section selects the zero vector.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-20 append
The zero vector has norm zero, so it lies in the disk interior and never on the unit sphere; this makes the zero section a map of pairs from \((B,\varnothing)\).
\end{hint}
\begin{solution}
The zero section is \(s(b)=0_b\). Since \(0_b\) lies in the interior of every fiber disk \(D(E_b)\), the image \(s(B)\) meets every disk fiber at its center and never lies in the sphere boundary.
\end{solution}

\begin{exercise}\label{exr:viii31-21}
Compute $s^*u_E$ for a trivial positive-rank bundle.
\end{exercise}
\begin{hint}
A trivial positive-rank bundle admits a nonzero constant section.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-21 append
Choose a nonzero constant fiber vector and normalize it. Homotope this section to the zero section and use the vanishing of the Thom class on the sphere boundary.
\end{hint}
\begin{solution}
A trivial positive-rank bundle has a nowhere-zero constant section, so its Euler class vanishes. Since \(e(E)=s^*u_E\),
\[
s^*u_E=0.
\]
\end{solution}

\begin{exercise}\label{exr:viii31-22}
Preview why a nowhere-zero section constrains $s^*u_E$.
\end{exercise}
\begin{hint}
Compare zero and nonzero sections.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-22 append
Normalize the nowhere-zero section into \(S(E)\); its pullback of the relative Thom class vanishes, while linear interpolation compares it with the zero section in \(D(E)\).
\end{hint}
\begin{solution}
A nowhere-zero section can be normalized to a section of \(S(E)\). The Euler class is the obstruction represented by the zero section's pullback of the Thom class; a section avoiding zero makes that obstruction vanish. Hence a nowhere-zero section forces \(s^*u_E=e(E)=0\).
\end{solution}

\begin{exercise}\label{exr:viii31-23}
Relate a normal Thom class to the cohomology class of a submanifold.
\end{exercise}
\begin{hint}
Extend the normal Thom class into ambient cohomology.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-23 append
Use the oriented normal bundle of a closed submanifold; pass from its Thom class through the ambient relative group before forgetting supports.
\end{hint}
\begin{solution}
For an oriented embedded submanifold \(N\subset M\), take the Thom class of its oriented normal bundle, identify the disk bundle with a tubular neighborhood, and extend by excision into \(M\). The resulting ambient cohomology class is \(\operatorname{PD}[N]\).
\end{solution}

\begin{exercise}\label{exr:viii31-24}
Summarize the chain Thom class $\to$ Euler class $\to$ Poincaré duality.
\end{exercise}
\begin{hint}
Track the degree change at every arrow.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii31-24 append
Assume the base is a closed oriented \(n\)-manifold: the rank-\(r\) Thom class gives a degree-\(r\) Euler class and then a degree-\((n-r)\) homology class.
\end{hint}
\begin{solution}
The chain is
\[
u_E\in H^r(D(E),S(E))
\;\xrightarrow{s^*}\;
e(E)\in H^r(B)
\;\xrightarrow{\frown[B]}\;
H_{\dim B-r}(B).
\]
Thus the Thom class produces the Euler class, and cap product with the fundamental class converts it into its geometric dual cycle.
\end{solution}
