\section{Supplementary worked examples}
\label{sec:viii07-pedagogy}

\begin{example}[Cone of the identity]\label{ex:viii07-ped-01}
The mapping cone of $\operatorname{id}_X$ is contractible: one glues the cone $CX$ to $X$ along the identity, and the resulting space is naturally homeomorphic to the cone with its base retained as the attached copy.
\end{example}

\begin{example}[Cone of a constant map]\label{ex:viii07-ped-02}
For a based constant map $f:X\to Y$, the mapping cone is homotopy equivalent to $Y\vee\Sigma X$.  The cone base is collapsed at one point of $Y$, leaving the suspension contribution.
\end{example}

\begin{example}[Degree-$m$ cone]\label{ex:viii07-ped-03}
The mapping cone of a degree-$m$ map $S^1\to S^1$ is a two-dimensional CW complex with one cell in dimensions $0,1,2$ and cellular differential $d_2=m$, producing $H_1\cong\mathbb Z/m$ for $m\ne0$.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Cone definition]\label{exr:viii07-ped-01}
Describe the cone $CX$ as a quotient of $X\times I$.
\end{exercise}
\begin{hint}
Collapse one end of the cylinder.
\end{hint}
\begin{solution}
$CX=X\times I/(X\times\{1\}\text{ collapsed to a point})$ (up to the chosen endpoint convention).
\end{solution}

\begin{exercise}[Mapping cone gluing]\label{exr:viii07-ped-02}
What is glued to $Y$ in the mapping cone of $f:X\to Y$?
\end{exercise}
\begin{hint}
Attach the base of $CX$ using $f$.
\end{hint}
\begin{solution}
One forms $C_f=Y\cup_f CX$, identifying $(x,0)$ in the cone base with $f(x)\in Y$.
\end{solution}

\begin{exercise}[Identity cone]\label{exr:viii07-ped-03}
What is the homotopy type of the mapping cone of $\operatorname{id}_X$?
\end{exercise}
\begin{hint}
The identity attachment fills the copy of $X$ by its cone.
\end{hint}
\begin{solution}
It is contractible.
\end{solution}

\begin{exercise}[Point target]\label{exr:viii07-ped-04}
What is the mapping cone of the map $X\to *$?
\end{exercise}
\begin{hint}
The cone base is collapsed to the target point.
\end{hint}
\begin{solution}
It is the reduced suspension $\Sigma X$ in the based setting.
\end{solution}

\begin{exercise}[Degree-$m$ homology]\label{exr:viii07-ped-05}
For the cone of degree $m:S^1\to S^1$, compute $H_1$ when $m\ne0$.
\end{exercise}
\begin{hint}
Use the cellular differential $\mathbb Z\xrightarrow{m}\mathbb Z$.
\end{hint}
\begin{solution}
$H_1\cong\operatorname{coker}(m)=\mathbb Z/m$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Identity cone contraction]\label{exr:viii07-ped-06}
Prove the cone of the identity is contractible.
\end{exercise}
\begin{hint}
Use the cone coordinate to slide all points to the cone vertex.
\end{hint}
\begin{solution}
The attached copy of $X$ is exactly the cone base; the ordinary cone contraction descends to the mapping cone.
\end{solution}

\begin{exercise}[Homotopy functoriality]\label{exr:viii07-ped-07}
Explain why homotopic maps have homotopy-equivalent mapping cones under CW/cofibration hypotheses.
\end{exercise}
\begin{hint}
Use the homotopy as gluing data over $X\times I$.
\end{hint}
\begin{solution}
A mapping-cylinder comparison interpolates the attaching maps and gives maps between the two cones that are homotopy inverses.
\end{solution}

\begin{exercise}[Cofiber sequence intuition]\label{exr:viii07-ped-08}
Why does $X\xrightarrow fY\to C_f$ behave like a quotient that kills the image of $f$ homotopically?
\end{exercise}
\begin{hint}
The cone on $X$ is contractible and fills each $f(x)$ along a cone line.
\end{hint}
\begin{solution}
After adjoining $CX$, the map $X\to C_f$ through $Y$ becomes null-homotopic; $C_f$ is the homotopy-theoretic cofiber of $f$.
\end{solution}

\begin{exercise}[Suspension case]\label{exr:viii07-ped-09}
Prove $C_{X\to *}\cong\Sigma X$ in the reduced based model.
\end{exercise}
\begin{hint}
Both constructions collapse the two cylinder ends appropriately.
\end{hint}
\begin{solution}
The target point collapses the base $X\times\{0\}$ while the cone collapses $X\times\{1\}$, exactly yielding the suspension quotient.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Cone always contractible]\label{exr:viii07-ped-10}
Test: every mapping cone is contractible.
\end{exercise}
\begin{hint}
Only the ordinary cone is automatically contractible.
\end{hint}
\begin{solution}
False; degree-$m$ mapping cones can have nontrivial homology.
\end{solution}

\begin{exercise}[Cone remembers only image]\label{exr:viii07-ped-11}
Test: mapping cones depend only on the set-theoretic image of $f$.
\end{exercise}
\begin{hint}
The attaching homotopy class matters.
\end{hint}
\begin{solution}
False. Different attaching degrees with the same image can give different homology.
\end{solution}

\begin{exercise}[Cone of zero is just $Y$]\label{exr:viii07-ped-12}
Test: the cone of a constant map $X\to Y$ is homotopy equivalent to $Y$.
\end{exercise}
\begin{hint}
There is generally a suspension summand.
\end{hint}
\begin{solution}
False; in the based setting it is typically $Y\vee\Sigma X$.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Relation-to-homology]\label{exr:viii07-ped-13}
Why are mapping cones useful for turning a map into an object whose homology measures failure of isomorphism?
\end{exercise}
\begin{hint}
Cone homology fits into a long exact sequence with the source and target.
\end{hint}
\begin{solution}
The cofiber sequence yields an exact sequence in homology, so kernels and cokernels of $f_*$ appear as pieces of the cone's homology.
\end{solution}

\begin{exercise}[Data structure analogy]\label{exr:viii07-ped-14}
Interpret a mapping cone as recording both a target and a certificate that the source map becomes null.
\end{exercise}
\begin{hint}
Each source point receives a cone path to one vertex.
\end{hint}
\begin{solution}
The attached cone supplies explicit null-homotopy paths for the composite $X\to Y\to C_f$, while retaining the rest of $Y$; it packages a map together with its homotopy cofiber.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Moore space]\label{exr:viii07-ped-15}
Use a mapping cone to construct a space with reduced homology $\mathbb Z/m$ in degree one.
\end{exercise}
\begin{hint}
Take the degree-$m$ map on $S^1$.
\end{hint}
\begin{solution}
The mapping cone of $S^1\xrightarrow{m}S^1$ has cellular chain differential $d_2=m$, so $\widetilde H_1\cong\mathbb Z/m$ and the adjacent reduced groups vanish.
\end{solution}

\begin{exercise}[Cone synthesis]\label{exr:viii07-ped-16}
For $f:S^n\to S^n$ of degree $m$, predict the key cellular differential in $C_f$.
\end{exercise}
\begin{hint}
There is an $n$-cell from the target and an $(n+1)$-cell from the cone.
\end{hint}
\begin{solution}
The cellular differential $C_{n+1}\to C_n$ is multiplication by $m$, so the cone converts degree into torsion homology.
\end{solution}
