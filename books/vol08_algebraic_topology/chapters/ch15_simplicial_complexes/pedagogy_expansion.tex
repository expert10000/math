\section{Supplementary worked examples}
\label{sec:viii15-pedagogy}

\begin{example}[Boundary of a triangle]\label{ex:viii15-ped-01}
An oriented 2-simplex $[v_0v_1v_2]$ has boundary $[v_1v_2]-[v_0v_2]+[v_0v_1]$.  The alternating signs ensure shared edges cancel when adjacent triangles are summed with compatible orientations.
\end{example}

\begin{example}[Triangulated circle]\label{ex:viii15-ped-02}
A polygon with cyclically arranged edges is a simplicial model of $S^1$.  Its 1-cycle is the sum of oriented edges; every vertex appears once with sign $+1$ and once with sign $-1$ in the boundary.
\end{example}

\begin{example}[Triangulated sphere]\label{ex:viii15-ped-03}
The boundary of a tetrahedron is a simplicial complex homeomorphic to $S^2$.  The signed sum of its four oriented triangular faces is a 2-cycle because every edge appears twice with opposite induced orientations.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[0-simplex faces]\label{exr:viii15-ped-01}
How many proper faces does a 0-simplex have in the ordinary nonempty-face convention?
\end{exercise}
\begin{hint}
A vertex has no nonempty lower-dimensional face.
\end{hint}
\begin{solution}
None.
\end{solution}

\begin{exercise}[1-simplex boundary]\label{exr:viii15-ped-02}
Compute $\partial[v_0v_1]$.
\end{exercise}
\begin{hint}
Delete one vertex at a time with alternating signs.
\end{hint}
\begin{solution}
$\partial[v_0v_1]=[v_1]-[v_0]$.
\end{solution}

\begin{exercise}[2-simplex boundary]\label{exr:viii15-ped-03}
Compute $\partial[v_0v_1v_2]$.
\end{exercise}
\begin{hint}
Use the alternating face formula.
\end{hint}
\begin{solution}
$[v_1v_2]-[v_0v_2]+[v_0v_1]$.
\end{solution}

\begin{exercise}[Dimension]\label{exr:viii15-ped-04}
What is the dimension of a simplex with $k+1$ vertices?
\end{exercise}
\begin{hint}
A $k$-simplex has $k+1$ vertices.
\end{hint}
\begin{solution}
Its dimension is $k$.
\end{solution}

\begin{exercise}[Euler count tetrahedron]\label{exr:viii15-ped-05}
Compute $V-E+F$ for the boundary of a tetrahedron.
\end{exercise}
\begin{hint}
Count 4 vertices, 6 edges, 4 faces.
\end{hint}
\begin{solution}
$4-6+4=2$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Boundary squared]\label{exr:viii15-ped-06}
Prove $\partial^2=0$ on an oriented simplex.
\end{exercise}
\begin{hint}
Each codimension-two face is obtained by deleting two vertices in two orders.
\end{hint}
\begin{solution}
The two deletion orders contribute opposite signs, so every codimension-two face cancels pairwise.
\end{solution}

\begin{exercise}[Shared-edge cancellation]\label{exr:viii15-ped-07}
Explain why internal edges cancel in the boundary of an oriented triangulated surface chain.
\end{exercise}
\begin{hint}
Adjacent faces induce opposite orientations on their common edge.
\end{hint}
\begin{solution}
Each internal edge appears in two face boundaries with opposite signs, leaving only boundary edges of the surface.
\end{solution}

\begin{exercise}[Subcomplex closure]\label{exr:viii15-ped-08}
Prove a collection of simplices closed under faces forms a simplicial subcomplex.
\end{exercise}
\begin{hint}
Check exactly the defining face condition.
\end{hint}
\begin{solution}
Every face of every selected simplex remains selected, so the collection inherits all required incidence data and is a subcomplex.
\end{solution}

\begin{exercise}[Simplicial map determined by vertices]\label{exr:viii15-ped-09}
Why is a simplicial map determined by its action on vertices?
\end{exercise}
\begin{hint}
A simplex is the set/span of its vertices.
\end{hint}
\begin{solution}
Once vertex images are fixed, each simplex must map to the simplex spanned by those images; affine extension on geometric realizations is then unique.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Any vertex map simplicial]\label{exr:viii15-ped-10}
Test: every function between vertex sets defines a simplicial map.
\end{exercise}
\begin{hint}
Images of simplices must again span simplices.
\end{hint}
\begin{solution}
False; the simplex condition can fail.
\end{solution}

\begin{exercise}[Orientation changes simplex]\label{exr:viii15-ped-11}
Test: reversing orientation creates a different underlying simplex.
\end{exercise}
\begin{hint}
Orientation is extra algebraic data.
\end{hint}
\begin{solution}
False. The geometric simplex is unchanged; its oriented chain generator changes sign.
\end{solution}

\begin{exercise}[Triangulation unique]\label{exr:viii15-ped-12}
Test: a topological space has at most one triangulation.
\end{exercise}
\begin{hint}
Subdivision already gives alternatives.
\end{hint}
\begin{solution}
False. Spaces can admit many distinct triangulations.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Mesh topology]\label{exr:viii15-ped-13}
Why is a simplicial complex a natural data model for a triangle mesh?
\end{exercise}
\begin{hint}
It separates combinatorial adjacency from geometric coordinates.
\end{hint}
\begin{solution}
Vertices, edges, and faces encode incidence combinatorially; geometric realization assigns positions.  Topological chains can therefore be computed from connectivity independent of embedding.
\end{solution}

\begin{exercise}[Boundary extraction]\label{exr:viii15-ped-14}
How can the simplicial boundary operator identify boundary edges of an oriented triangle mesh?
\end{exercise}
\begin{hint}
Sum all oriented faces.
\end{hint}
\begin{solution}
Internal edges cancel in pairs, while edges incident to only one face remain in the total boundary chain, directly extracting the mesh boundary.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Sphere cycle]\label{exr:viii15-ped-15}
Write the conceptual fundamental 2-cycle of the tetrahedral sphere.
\end{exercise}
\begin{hint}
Orient all four faces coherently.
\end{hint}
\begin{solution}
Take the signed sum of the four oriented triangular faces induced by an orientation of the tetrahedron boundary; shared edges cancel, so the chain has zero boundary.
\end{solution}

\begin{exercise}[Subdivision synthesis]\label{exr:viii15-ped-16}
Why should subdividing a simplicial complex preserve the homology one intends to compute?
\end{exercise}
\begin{hint}
Subdivision changes combinatorics but not the underlying polyhedron.
\end{hint}
\begin{solution}
A subdivision has homeomorphic geometric realization; the induced chain maps are chain-homotopy equivalences, so homology remains unchanged while mesh resolution changes.
\end{solution}
