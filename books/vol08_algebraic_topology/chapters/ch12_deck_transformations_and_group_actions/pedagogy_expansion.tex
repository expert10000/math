\section{Supplementary worked examples}
\label{sec:viii12-pedagogy}

\begin{example}[Deck group of the circle universal cover]\label{ex:viii12-ped-01}
For $p:\mathbb R\to S^1$, every deck transformation is $T_k(t)=t+k$ with $k\in\mathbb Z$.  Hence $\operatorname{Deck}(p)\cong\mathbb Z$.
\end{example}

\begin{example}[Deck group of $z\mapsto z^n$]\label{ex:viii12-ped-02}
For $p_n:S^1\to S^1$, multiplication by an $n$th root of unity preserves fibers and commutes with $p_n$.  These rotations form a cyclic deck group of order $n$.
\end{example}

\begin{example}[Free properly discontinuous action]\label{ex:viii12-ped-03}
If a discrete group $G$ acts freely and properly discontinuously on a suitable space $E$, the quotient map $E\to E/G$ is a covering and $G$ acts by deck transformations.  Regular coverings arise naturally this way.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Deck equation]\label{exr:viii12-ped-01}
State the defining equation for a deck transformation $h:E\to E$ over $B$.
\end{exercise}
\begin{hint}
It must preserve the projection.
\end{hint}
\begin{solution}
$p\circ h=p$, with $h$ a homeomorphism of $E$.
\end{solution}

\begin{exercise}[Integer translation]\label{exr:viii12-ped-02}
Check $T_k(t)=t+k$ is a deck transformation of $\mathbb R\to S^1$.
\end{exercise}
\begin{hint}
Exponentials are periodic by integers.
\end{hint}
\begin{solution}
$p(t+k)=e^{2\pi i(t+k)}=e^{2\pi it}=p(t)$, and $T_k$ is a homeomorphism.
\end{solution}

\begin{exercise}[Root rotation]\label{exr:viii12-ped-03}
For $p_n(z)=z^n$, check $h_\zeta(z)=\zeta z$ is a deck transformation when $\zeta^n=1$.
\end{exercise}
\begin{hint}
Compute $(\zeta z)^n$.
\end{hint}
\begin{solution}
$p_n(\zeta z)=\zeta^nz^n=z^n=p_n(z)$.
\end{solution}

\begin{exercise}[Composition law]\label{exr:viii12-ped-04}
What is the composition law for deck transformations of $\mathbb R\to S^1$?
\end{exercise}
\begin{hint}
Compose translations.
\end{hint}
\begin{solution}
$T_kT_\ell=T_{k+\ell}$.
\end{solution}

\begin{exercise}[Fiber action]\label{exr:viii12-ped-05}
Where can a deck transformation send a point $e\in E$?
\end{exercise}
\begin{hint}
Use $p(h(e))=p(e)$.
\end{hint}
\begin{solution}
It must remain in the same fiber $p^{-1}(p(e))$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Deck determined by one point]\label{exr:viii12-ped-06}
Prove a deck transformation of a connected cover is determined by its value at one point.
\end{exercise}
\begin{hint}
Two deck maps are lifts of the same map $p:E\to B$.
\end{hint}
\begin{solution}
If two deck transformations agree at one point, map-lift uniqueness on connected $E$ forces them to agree everywhere.
\end{solution}

\begin{exercise}[Free action on total space]\label{exr:viii12-ped-07}
Prove the deck group of a connected cover acts freely on $E$.
\end{exercise}
\begin{hint}
A deck map fixing one point agrees there with the identity.
\end{hint}
\begin{solution}
By the previous uniqueness principle, a deck transformation with a fixed point equals the identity.
\end{solution}

\begin{exercise}[Quotient construction]\label{exr:viii12-ped-08}
Explain why a free properly discontinuous group action yields evenly covered quotient neighborhoods.
\end{exercise}
\begin{hint}
Choose $U$ disjoint from all nontrivial translates.
\end{hint}
\begin{solution}
The quotient restricts to a homeomorphism on each translate $gU$, and the inverse image of its quotient image is the disjoint union of these translates.
\end{solution}

\begin{exercise}[Regular cover transitivity]\label{exr:viii12-ped-09}
Show that if deck transformations act transitively on one fiber of a connected cover, they act transitively on every fiber.
\end{exercise}
\begin{hint}
Transport points along lifted paths.
\end{hint}
\begin{solution}
Path lifting conjugates the fiber actions along any base path; a deck transformation matching two points over one basepoint propagates to the corresponding endpoints over another.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Deck maps can fix points]\label{exr:viii12-ped-10}
Test: a nonidentity deck transformation of a connected cover may fix isolated points.
\end{exercise}
\begin{hint}
Use lift uniqueness.
\end{hint}
\begin{solution}
False.  The deck action is free.
\end{solution}

\begin{exercise}[Every cover regular]\label{exr:viii12-ped-11}
Test: the deck group always acts transitively on fibers.
\end{exercise}
\begin{hint}
That is the definition/characterization of regular coverings, not all coverings.
\end{hint}
\begin{solution}
False. Nonregular connected covers can have smaller deck groups.
\end{solution}

\begin{exercise}[Deck group equals fiber]\label{exr:viii12-ped-12}
Test: for every connected cover, deck-group cardinality equals fiber cardinality.
\end{exercise}
\begin{hint}
Only in regular covers does the action become transitive.
\end{hint}
\begin{solution}
False in general; a deck transformation is determined by a point image, so its size is at most the fiber size, with equality in regular cases.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Symmetry extraction]\label{exr:viii12-ped-13}
Why is the deck group a useful algebraic summary of a covering?
\end{exercise}
\begin{hint}
It records global symmetries invisible in one local sheet.
\end{hint}
\begin{solution}
Deck transformations permute sheets compatibly over every base point, encoding how the covering repeats globally and connecting the cover to subgroup normalizers.
\end{solution}

\begin{exercise}[Quotient reconstruction]\label{exr:viii12-ped-14}
When can a covering be recovered as a quotient by its deck group?
\end{exercise}
\begin{hint}
Think regular connected coverings.
\end{hint}
\begin{solution}
For a regular connected cover, the deck group acts freely and transitively on fibers, and the base is naturally the quotient $E/\operatorname{Deck}(p)$ under standard hypotheses.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Normalizer formula preview]\label{exr:viii12-ped-15}
For a connected cover corresponding to $H\le\pi_1(B)$, what group controls deck transformations?
\end{exercise}
\begin{hint}
Use the normalizer of $H$.
\end{hint}
\begin{solution}
Under the classification hypotheses, $\operatorname{Deck}(p)\cong N(H)/H$; if $H$ is normal, this becomes $\pi_1(B)/H$.
\end{solution}

\begin{exercise}[Monodromy versus deck symmetry]\label{exr:viii12-ped-16}
Distinguish the action of $\pi_1(B)$ on a fiber by path lifting from the deck-group action.
\end{exercise}
\begin{hint}
One comes from loops downstairs; the other from homeomorphisms upstairs.
\end{hint}
\begin{solution}
Monodromy permutes a chosen fiber via lifted loop endpoints and exists for every cover; deck transformations are global symmetries of the total space.  For regular covers the two structures align through the quotient group.
\end{solution}
