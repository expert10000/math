\chapter{Deck Transformations and Group Actions}
\label{ch:viii12-deck-transformations}

A covering space has internal symmetries that preserve every point of the base.
These symmetries are deck transformations.  On connected covers they are
extremely rigid: one point determines the whole transformation.  Their group
acts on fibers, regular coverings are characterized by transitivity of this
action, and universal covers recover the base fundamental group as a deck
group.

\section{Deck transformations}

\begin{definition}[Deck transformation]
\label{def:viii12-deck}
Let
\[
p:\widetilde X\to X
\]
be a covering.  A deck transformation is a homeomorphism
\[
h:\widetilde X\to\widetilde X
\]
such that
\[
p\circ h=p.
\]
\end{definition}

The deck transformations form a group under composition:
\[
\operatorname{Deck}(p).
\]

\section{Deck maps preserve fibers}

\begin{proposition}[Fiber preservation]
\label{prop:viii12-fiber}
For every \(x\in X\),
\[
h\bigl(p^{-1}(x)\bigr)=p^{-1}(x)
\]
for each deck transformation \(h\).
\end{proposition}

Thus the deck group acts by permutations on every fiber.

\section{Rigidity on connected covers}

\begin{theorem}[One point determines a deck transformation]
\label{thm:viii12-one-point}
If \(\widetilde X\) is connected, two deck transformations that agree at one
point are equal everywhere.
\end{theorem}

\begin{proof}
Both maps are lifts of the same covering map
\[
p:\widetilde X\to X
\]
through the covering \(p\).  Apply uniqueness of lifts from VIII/11.
\end{proof}

\begin{corollary}[Free action]
\label{cor:viii12-free}
If a deck transformation of a connected covering fixes one point, it is the
identity.  Hence the deck group acts freely on \(\widetilde X\).
\end{corollary}

\section{The universal circle cover}

For
\[
p:\mathbb R\to S^1,\qquad p(t)=e^{2\pi it},
\]
the translations
\[
T_n(t)=t+n,\qquad n\in\mathbb Z,
\]
are deck transformations.

\begin{theorem}[Deck group of the circle cover]
\label{thm:viii12-circle-deck}
\[
\operatorname{Deck}(\mathbb R\to S^1)\cong\mathbb Z.
\]
\end{theorem}

\section{Finite circle covers}

For
\[
p_n:S^1\to S^1,\qquad p_n(z)=z^n,
\]
the rotations
\[
R_k(z)=e^{2\pi ik/n}z
\]
are deck transformations.

\begin{theorem}[Deck group of \(z^n\)]
\label{thm:viii12-power-deck}
\[
\operatorname{Deck}(p_n)\cong\mathbb Z/n.
\]
\end{theorem}

\section{Antipodal covering}

For \(n\ge2\),
\[
q:S^n\to\mathbb RP^n
\]
has the nontrivial deck transformation
\[
a(x)=-x.
\]

\begin{proposition}[Projective deck group]
\label{prop:viii12-projective-deck}
\[
\operatorname{Deck}(S^n\to\mathbb RP^n)\cong\mathbb Z/2.
\]
\end{proposition}

\section{Deck action on a fiber}

Fix \(x_0\in X\).  Restriction gives an action
\[
\operatorname{Deck}(p)\curvearrowright p^{-1}(x_0).
\]

Because the action is free for connected covers, no nonidentity deck
transformation fixes a fiber point.

\section{Regular coverings}

\begin{definition}[Regular covering]
\label{def:viii12-regular}
A connected covering is regular, normal, or Galois if the deck group acts
transitively on one, equivalently every, fiber.
\end{definition}

\begin{theorem}[Normal-subgroup criterion]
\label{thm:viii12-normal}
Under the standard covering-classification hypotheses, a connected pointed
cover corresponding to
\[
H\le \pi_1(X,x_0)
\]
is regular if and only if
\[
H\trianglelefteq\pi_1(X,x_0).
\]
\end{theorem}

\section{Deck group as a quotient}

\begin{theorem}[Deck group of a regular cover]
\label{thm:viii12-quotient}
If a connected covering corresponds to a normal subgroup
\[
H\trianglelefteq G=\pi_1(X,x_0),
\]
then
\[
\operatorname{Deck}(p)\cong G/H.
\]
\end{theorem}

For a universal cover \(H=1\), so
\[
\operatorname{Deck}(\widetilde X\to X)
\cong
\pi_1(X,x_0)
\]
under the standard hypotheses.

\section{Universal covers and group actions}

If \(\widetilde X\) is a universal cover, the fundamental group acts on
\(\widetilde X\) by deck transformations.  The quotient by this action recovers
the base:
\[
X\cong \widetilde X/\pi_1(X).
\]

This is the geometric realization of the fundamental group as a symmetry group
of the universal cover.

\section{Properly discontinuous actions}

A group \(G\) acting on a space \(\widetilde X\) is properly discontinuous in
the covering-space sense when every point has a neighborhood whose distinct
translates do not overlap in a way that destroys the local quotient model.

\begin{theorem}[Quotient construction]
\label{thm:viii12-quotient-cover}
If \(G\) acts freely and properly discontinuously on a suitable space
\(\widetilde X\), then
\[
\widetilde X\to\widetilde X/G
\]
is a covering map, and \(G\) embeds into its deck group.
\end{theorem}

Under standard connectedness hypotheses and an effective action realizing all
deck symmetries, the deck group is exactly \(G\).

\section{Translations on the line}

The action
\[
\mathbb Z\curvearrowright\mathbb R,
\qquad
n\cdot t=t+n,
\]
is free and properly discontinuous.  Its quotient is
\[
\mathbb R/\mathbb Z\cong S^1.
\]

\section{Lattice actions}

For a lattice
\[
\Lambda\cong\mathbb Z^n
\]
acting by translations on \(\mathbb R^n\),
\[
\mathbb R^n/\Lambda
\]
is an \(n\)-torus, and
\[
\mathbb R^n\to\mathbb R^n/\Lambda
\]
is a universal covering map with deck group \(\Lambda\).

In particular,
\[
\operatorname{Deck}(\mathbb R^2\to T^2)\cong\mathbb Z^2.
\]

\section{Monodromy versus deck action}

Monodromy is the action
\[
\pi_1(X,x_0)\curvearrowright p^{-1}(x_0)
\]
obtained by lifting loops.  The deck group also acts on the same fiber.  These
actions are related but not identical in arbitrary covers.

For a regular cover, the deck action is transitive and can be identified with
a quotient of the fundamental group.

\section{The normalizer formula}

Let a connected pointed cover correspond to
\[
H\le G=\pi_1(X,x_0).
\]

\begin{theorem}[Normalizer formula]
\label{thm:viii12-normalizer}
Under the standard classification hypotheses,
\[
\operatorname{Deck}(p)\cong N_G(H)/H,
\]
where
\[
N_G(H)=\{g\in G:gHg^{-1}=H\}.
\]
\end{theorem}

If \(H\) is normal, then \(N_G(H)=G\), recovering the regular-cover quotient
formula.

\section{Deck transformations and basepoint choices}

A deck transformation can move the chosen point
\[
\tilde x_0\in p^{-1}(x_0)
\]
to another point of the same fiber.  Changing the pointed cover in this way
conjugates the associated subgroup.  This explains why unpointed connected
coverings correspond to conjugacy classes of subgroups.

\section{Automorphisms of coverings}

Deck transformations are precisely automorphisms of the covering object over
the fixed base \(X\).  Thus
\[
\operatorname{Deck}(p)
=
\operatorname{Aut}_X(\widetilde X).
\]

\section{Regularity examples}

The universal circle cover is regular because the trivial subgroup of
\(\mathbb Z\) is normal.  The \(n\)-fold circle cover corresponds to
\(n\mathbb Z\trianglelefteq\mathbb Z\), so it is regular with deck group
\[
\mathbb Z/n.
\]

The universal projective cover corresponds to the trivial subgroup of
\(\mathbb Z/2\), hence is regular.

\section{Bridge to \(SU(2)\to SO(3)\)}

The next chapter studies one of the most important geometric two-sheeted
covers:
\[
SU(2)\to SO(3).
\]
Its kernel is
\[
\{\pm I\},
\]
and the nontrivial deck transformation is multiplication by \(-I\).  This
provides a Lie-group realization of the same covering and quotient principles
developed here.

\section{Solved problem dossiers}

\begin{problem}[Circle translations]\label{prob:viii12-01}
Prove \(T_n(t)=t+n\) is a deck transformation of \(\mathbb R\to S^1\).
\end{problem}
\begin{solution}
\(p(T_n(t))=e^{2\pi i(t+n)}=e^{2\pi it}=p(t)\), and \(T_n\) is a homeomorphism.
\end{solution}

\begin{problem}[All circle deck maps]\label{prob:viii12-02}
Show every deck transformation of \(\mathbb R\to S^1\) is an integer translation.
\end{problem}
\begin{solution}
A deck map sends \(0\) to a point in the fiber \(\mathbb Z\), say \(n\).  The translation \(T_n\) is another deck map with the same value at \(0\); connected-cover rigidity gives equality.
\end{solution}

\begin{problem}[Finite circle rotations]\label{prob:viii12-03}
Verify \(R_k(z)=e^{2\pi ik/n}z\) is a deck map of \(p_n(z)=z^n\).
\end{problem}
\begin{solution}
\(p_n(R_k(z))=e^{2\pi ik}z^n=z^n=p_n(z)\).
\end{solution}

\begin{problem}[All finite circle deck maps]\label{prob:viii12-04}
Why are there exactly \(n\) deck transformations of \(z\mapsto z^n\)?
\end{problem}
\begin{solution}
A deck transformation is determined by the image of \(1\), which can be any of the \(n\) points in the fiber over \(1\).  Each choice is realized by one root-of-unity rotation.
\end{solution}

\begin{problem}[Antipodal deck map]\label{prob:viii12-05}
Verify \(a(x)=-x\) is a deck transformation of \(S^n\to\mathbb RP^n\).
\end{problem}
\begin{solution}
The quotient map sends \(x\) and \(-x\) to the same projective line, so \(qa=q\).
\end{solution}

\begin{problem}[Projective deck group]\label{prob:viii12-06}
Why are identity and antipodal map the only deck transformations of the connected cover \(S^n\to\mathbb RP^n\)?
\end{problem}
\begin{solution}
A deck transformation is determined by the image of one point, and the fiber contains only \(x\) and \(-x\).
\end{solution}

\begin{problem}[Free action proof]\label{prob:viii12-07}
Prove the deck action on a connected covering is free.
\end{problem}
\begin{solution}
If \(h(\tilde x)=\tilde x\), then \(h\) and the identity deck map agree at one point.  Lift uniqueness makes them equal.
\end{solution}

\begin{problem}[Universal-cover quotient]\label{prob:viii12-08}
Explain why a universal cover can be viewed as a quotient by its deck group.
\end{problem}
\begin{solution}
The deck group acts freely and properly discontinuously, and its orbits are fibers for a regular universal cover.  The quotient therefore recovers the base.
\end{solution}

\begin{problem}[Torus lattice cover]\label{prob:viii12-09}
Show \(\mathbb R^2\to\mathbb R^2/\mathbb Z^2\) has deck group \(\mathbb Z^2\).
\end{problem}
\begin{solution}
Each lattice translation is a deck map.  Any deck map is determined by the image of \(0\), which must be another point of the fiber \(\mathbb Z^2\), so it is the corresponding translation.
\end{solution}

\begin{problem}[Regularity of \(z^n\)]\label{prob:viii12-10}
Why is the \(n\)-fold circle cover regular?
\end{problem}
\begin{solution}
It corresponds to \(n\mathbb Z\trianglelefteq\mathbb Z\). Equivalently, the root-of-unity deck rotations act transitively on every fiber.
\end{solution}

\begin{problem}[Quotient deck group]\label{prob:viii12-11}
Use subgroup theory to compute the deck group of the cover corresponding to \(n\mathbb Z\le\mathbb Z\).
\end{problem}
\begin{solution}
Since \(\mathbb Z\) is abelian, \(n\mathbb Z\) is normal and the deck group is \(\mathbb Z/n\mathbb Z\).
\end{solution}

\begin{problem}[Universal circle quotient]\label{prob:viii12-12}
Apply the regular-cover formula to the trivial subgroup of \(\mathbb Z\).
\end{problem}
\begin{solution}
The deck group is \(\mathbb Z/0\cong\mathbb Z\), matching integer translations of \(\mathbb R\).
\end{solution}

\begin{problem}[Normalizer example]\label{prob:viii12-13}
If \(H\le G\) is self-normalizing, what is the deck group of the corresponding connected cover?
\end{problem}
\begin{solution}
The normalizer formula gives \(N_G(H)/H=H/H\), so the deck group is trivial.
\end{solution}

\begin{problem}[Normal subgroup example]\label{prob:viii12-14}
If \(H\trianglelefteq G\) has index \(5\), what can be said about the regular cover?
\end{problem}
\begin{solution}
It is five-sheeted and its deck group \(G/H\) has five elements.
\end{solution}

\begin{problem}[Deck versus monodromy]\label{prob:viii12-15}
Explain one difference between deck action and monodromy action.
\end{problem}
\begin{solution}
Monodromy is an action of the base fundamental group on a fiber obtained by lifting loops.  Deck action is an action of automorphisms of the entire covering.  They coincide only through additional regularity identifications.
\end{solution}

\begin{problem}[Connected-cover rigidity]\label{prob:viii12-16}
Why can a deck transformation not permute one fiber arbitrarily?
\end{problem}
\begin{solution}
The image of one point determines the whole map, but not every fiber permutation need extend continuously to a deck transformation.  Only permutations compatible with the global covering structure do.
\end{solution}

\begin{problem}[Trivial cover automorphisms]\label{prob:viii12-17}
Describe deck transformations of \(X\times F\to X\) when \(X\) is connected and \(F\) is discrete.
\end{problem}
\begin{solution}
Any fixed permutation \(\sigma\in\operatorname{Sym}(F)\) defines \(h(x,a)=(x,\sigma(a))\). Conversely continuity over connected \(X\) forces the permutation to be constant, so the deck group is \(\operatorname{Sym}(F)\).
\end{solution}

\begin{problem}[Why connectedness mattered]\label{prob:viii12-18}
Why does one-point rigidity fail for a disconnected covering space?
\end{problem}
\begin{solution}
A deck map can agree with another on one component but differ on another component, so agreement at one point need not control all components.
\end{solution}

\begin{problem}[Conjugate subgroup viewpoint]\label{prob:viii12-19}
How does moving \(\tilde x_0\) in the fiber change the associated subgroup?
\end{problem}
\begin{solution}
A path/monodromy element relating the two fiber points conjugates \(H=p_*\pi_1(\widetilde X,\tilde x_0)\) inside \(G=\pi_1(X,x_0)\).
\end{solution}

\begin{problem}[Regular cover fiber action]\label{prob:viii12-20}
Why does transitivity of the deck group on a fiber imply every fiber point is equivalent as a pointed cover?
\end{problem}
\begin{solution}
For any two fiber points, a deck transformation carries one to the other while commuting with the projection, giving an isomorphism of the corresponding pointed covering structures.
\end{solution}

\begin{problem}[Deck group as quotient symmetry]\label{prob:viii12-21}
Interpret \(G/H\) geometrically for a regular cover.
\end{problem}
\begin{solution}
Cosets label sheets/fiber positions, and quotient elements act by translating those positions; this algebraic action is realized by deck transformations.
\end{solution}

\begin{problem}[Proper discontinuity of translations]\label{prob:viii12-22}
Why is the integer-translation action on \(\mathbb R\) properly discontinuous?
\end{problem}
\begin{solution}
For any \(t\), choose an interval of length less than \(1\). Its nonzero integer translates are disjoint from it, giving a local quotient sheet.
\end{solution}

\begin{problem}[Free action necessity]\label{prob:viii12-23}
What goes wrong for covering quotients if a group action has a fixed point?
\end{problem}
\begin{solution}
At a fixed point the quotient map cannot look locally like disjoint homeomorphic sheets indexed by group elements; distinct orbit representatives collapse at the fixed point.
\end{solution}

\begin{problem}[Bridge to \(SU(2)\)]\label{prob:viii12-24}
What deck symmetry should appear in the cover \(SU(2)\to SO(3)\)?
\end{problem}
\begin{solution}
The kernel is \(\{\pm I\}\).  Multiplication by \(-I\) is the nontrivial deck transformation, so the deck group is \(\mathbb Z/2\).
\end{solution}
