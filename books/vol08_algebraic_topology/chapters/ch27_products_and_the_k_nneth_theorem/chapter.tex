\chapter{Products and the Künneth Theorem}
\label{ch:viii27-products-kunneth}

Products of spaces lead to tensor products of chains.  The Künneth theorem
expresses the homology of \(X\times Y\) from the homology of the factors, with
Tor terms measuring torsion interactions.

\section{Product cells}
If \(e^p\subset X\) and \(e^q\subset Y\) are cells, then
\[
e^p\times e^q
\]
is a cell of dimension \(p+q\) in the product CW structure.

This immediately gives
\[
\chi(X\times Y)=\chi(X)\chi(Y)
\]
for finite CW complexes.

\section{Tensor product of chain complexes}
For chain complexes \(C,D\), define
\[
(C\otimes D)_n
=
\bigoplus_{p+q=n}C_p\otimes D_q.
\]

The differential on homogeneous tensors is
\[
\partial(c\otimes d)
=
\partial c\otimes d
+
(-1)^p c\otimes\partial d,
\qquad c\in C_p.
\]

\section{Koszul sign}
The sign \((-1)^p\) is necessary for
\[
\partial^2=0.
\]

\section{Eilenberg--Zilber principle}
For spaces \(X,Y\), singular chains on the product are chain-homotopy
equivalent to the tensor product:
\[
C_*^{\rm sing}(X\times Y)
\simeq
C_*^{\rm sing}(X)\otimes C_*^{\rm sing}(Y).
\]

This is the chain-level bridge from topology to algebra.

\section{Cross product}
The Eilenberg--Zilber construction yields the homology cross product
\[
\times:
H_p(X)\otimes H_q(Y)
\to
H_{p+q}(X\times Y).
\]

\section{Künneth theorem over \(\mathbb Z\)}
\begin{theorem}[Künneth theorem]
\label{thm:viii27-kunneth}
For suitable spaces such as CW complexes, there is a natural short exact
sequence
\[
0\to
\bigoplus_{p+q=n}
H_p(X)\otimes H_q(Y)
\longrightarrow
H_n(X\times Y)
\longrightarrow
\bigoplus_{p+q=n-1}
\operatorname{Tor}(H_p(X),H_q(Y))
\to0.
\]
\end{theorem}

The sequence splits noncanonically.

\section{Field coefficients}
Over a field \(k\), Tor vanishes and Künneth becomes
\[
H_n(X\times Y;k)
\cong
\bigoplus_{p+q=n}
H_p(X;k)\otimes_k H_q(Y;k).
\]

\section{Product of circles}
Since
\[
H_0(S^1)=H_1(S^1)=\mathbb Z,
\]
Künneth gives
\[
H_0(T^2)=\mathbb Z,
\qquad
H_1(T^2)=\mathbb Z^2,
\qquad
H_2(T^2)=\mathbb Z.
\]

\section{Higher tori}
For
\[
T^n=(S^1)^n,
\]
\[
H_k(T^n;\mathbb Z)\cong\mathbb Z^{\binom nk}.
\]

The total homology ranks are the binomial coefficients.

\section{Products of spheres}
For \(S^m\times S^n\), homology is free and concentrated in degrees
\[
0,\quad m,\quad n,\quad m+n,
\]
with the middle degrees combining if \(m=n\).

\section{\(S^2\times S^2\)}
\[
H_0=\mathbb Z,\quad
H_2=\mathbb Z^2,\quad
H_4=\mathbb Z,
\]
and all other groups vanish.

\section{Cylinder}
\[
S^1\times I\simeq S^1.
\]
Künneth agrees: \(I\) has only \(H_0=\mathbb Z\), so tensoring introduces no
new positive-dimensional classes.

\section{Product with a contractible space}
If \(Y\) is nonempty and contractible,
\[
X\times Y\simeq X,
\]
and Künneth reduces correspondingly.

\section{Torsion interaction}
If both factors have torsion, Tor can create homology one degree higher than the
naive tensor terms predict.

\section{Example with projective spaces}
Since
\[
H_1(\mathbb RP^2)=\mathbb Z/2,
\]
the product \(\mathbb RP^2\times\mathbb RP^2\) has tensor torsion and a Tor
contribution.  In particular,
\[
\operatorname{Tor}(\mathbb Z/2,\mathbb Z/2)\cong\mathbb Z/2
\]
appears in degree \(3\).

\section{Moore-space products}
For Moore spaces, the tensor and Tor terms can be read directly from the cyclic
torsion groups:
\[
\mathbb Z/m\otimes\mathbb Z/n
\cong
\operatorname{Tor}(\mathbb Z/m,\mathbb Z/n)
\cong
\mathbb Z/\gcd(m,n).
\]

\section{Naturality}
The Künneth short exact sequence is natural in maps of both factors.

\section{Nonnatural splitting}
As with UCT, the splitting is generally not natural.  The short exact sequence,
rather than a chosen direct-sum decomposition, is the canonical object.

\section{Reduced Künneth}
For pointed connected spaces, reduced homology often gives a cleaner product
formula by separating the \(H_0\) terms.

\section{Relative Künneth preview}
Pairs and products of pairs admit relative forms of Künneth under suitable
hypotheses.  These are useful in manifold and bundle calculations.

\section{Euler characteristic from Künneth}
Over a field, alternating dimensions factor:
\[
\chi(X\times Y)
=
\chi(X)\chi(Y).
\]
This recovers the product formula from homology.

\section{Betti-number convolution}
Over a field,
\[
b_n(X\times Y)
=
\sum_{p+q=n}b_p(X)b_q(Y).
\]

Thus Poincaré polynomials multiply:
\[
P_{X\times Y}(t)=P_X(t)P_Y(t).
\]

\section{Poincaré polynomial}
Define
\[
P_X(t)=\sum_n b_n(X)t^n.
\]
For the torus,
\[
P_{T^n}(t)=(1+t)^n.
\]

\section{Chain-homotopy input}
The Eilenberg--Zilber and Alexander--Whitney maps are inverse up to chain
homotopy, not literally inverse.  The machinery of VIII/22 is exactly what
allows them to induce inverse homology isomorphisms.

\section{Why the theorem matters}
Künneth replaces a potentially large product-space chain calculation with:
\begin{itemize}
\item tensor products of known homology groups;
\item Tor corrections for torsion.
\end{itemize}

\section{Bridge to cohomology}
Products become even richer in cohomology because the diagonal map and cross
product combine to form the cup product.  VIII/28 begins cohomology, and
VIII/29 develops its multiplicative structure.

\section{Solved problem dossiers}

\begin{problem}[Torus]\label{prob:viii27-01}
Compute \(H_*(S^1\times S^1)\) with Künneth.
\end{problem}
\begin{solution}
All groups are free, so no Tor occurs: tensor terms give \(\mathbb Z\) in degrees 0 and 2 and \(\mathbb Z^2\) in degree 1.
\end{solution}

\begin{problem}[Three-torus]\label{prob:viii27-02}
Compute the ranks of \(H_k(T^3)\).
\end{problem}
\begin{solution}
They are \(\binom30,\binom31,\binom32,\binom33=1,3,3,1\).
\end{solution}

\begin{problem}[Four-torus]\label{prob:viii27-03}
Compute \(b_2(T^4)\).
\end{problem}
\begin{solution}
\(\binom42=6\).
\end{solution}

\begin{problem}[Sphere product]\label{prob:viii27-04}
Compute \(H_*(S^2\times S^3)\).
\end{problem}
\begin{solution}
Free groups \(\mathbb Z\) occur in degrees 0,2,3,5 and vanish elsewhere.
\end{solution}

\begin{problem}[Equal spheres]\label{prob:viii27-05}
Compute \(H_2(S^2\times S^2)\).
\end{problem}
\begin{solution}
There are two tensor contributions \(H_2(S^2)\otimes H_0(S^2)\) and \(H_0\otimes H_2\), giving \(\mathbb Z^2\).
\end{solution}

\begin{problem}[Top class]\label{prob:viii27-06}
Where does the top class of \(S^m\times S^n\) come from?
\end{problem}
\begin{solution}
From \(H_m(S^m)\otimes H_n(S^n)\cong\mathbb Z\) in total degree \(m+n\).
\end{solution}

\begin{problem}[Contractible factor]\label{prob:viii27-07}
Use Künneth for \(X\times I\).
\end{problem}
\begin{solution}
Only \(H_0(I)=\mathbb Z\) is nonzero, so \(H_n(X\times I)\cong H_n(X)\).
\end{solution}

\begin{problem}[RP2 product Tor]\label{prob:viii27-08}
Locate a Tor contribution in \(\mathbb RP^2\times\mathbb RP^2\).
\end{problem}
\begin{solution}
Since both factors have \(H_1=\mathbb Z/2\), \(\operatorname{Tor}(H_1,H_1)=\mathbb Z/2\) contributes in total degree \(1+1+1=3\).
\end{solution}

\begin{problem}[RP2 product H2]\label{prob:viii27-09}
Give tensor contributions to \(H_2(\mathbb RP^2\times\mathbb RP^2)\).
\end{problem}
\begin{solution}
The \(H_1\otimes H_1\) term gives \(\mathbb Z/2\), while integral \(H_2\) of each factor vanishes.
\end{solution}

\begin{problem}[Moore gcd]\label{prob:viii27-10}
For Moore torsion groups \(\mathbb Z/12\) and \(\mathbb Z/18\), compute tensor and Tor.
\end{problem}
\begin{solution}
Both are \(\mathbb Z/6\).
\end{solution}

\begin{problem}[Field Kunneth]\label{prob:viii27-11}
Why is Künneth especially simple over \(\mathbb F_p\)?
\end{problem}
\begin{solution}
Vector spaces are flat and Tor vanishes, leaving a direct sum of tensor products.
\end{solution}

\begin{problem}[Betti convolution]\label{prob:viii27-12}
If \(P_X=1+2t+t^2\) and \(P_Y=1+t\), compute \(P_{X\times Y}\).
\end{problem}
\begin{solution}
Multiply to get \(1+3t+3t^2+t^3\).
\end{solution}

\begin{problem}[Euler product]\label{prob:viii27-13}
Use Poincaré polynomials to recover \(\chi(X\times Y)\).
\end{problem}
\begin{solution}
Euler characteristic is \(P(-1)\); since product polynomials multiply, \(\chi(X\times Y)=\chi(X)\chi(Y)\).
\end{solution}

\begin{problem}[Cross product]\label{prob:viii27-14}
What homology class does \(a\times b\) have degree?
\end{problem}
\begin{solution}
If \(a\in H_p(X)\) and \(b\in H_q(Y)\), then \(a\times b\in H_{p+q}(X\times Y)\).
\end{solution}

\begin{problem}[Naturality]\label{prob:viii27-15}
How does \(f\times g\) act on cross products?
\end{problem}
\begin{solution}
\((f\times g)_*(a\times b)=f_*(a)\times g_*(b)\).
\end{solution}

\begin{problem}[Nonnatural split]\label{prob:viii27-16}
Why is the Künneth exact sequence preferable to a chosen direct sum?
\end{problem}
\begin{solution}
The exact sequence is natural; a splitting usually depends on arbitrary choices.
\end{solution}

\begin{problem}[Reduced product]\label{prob:viii27-17}
Why can reduced homology simplify product formulas?
\end{problem}
\begin{solution}
It separates the ubiquitous degree-zero \(\mathbb Z\) terms, making the genuinely positive-dimensional cross terms more transparent.
\end{solution}

\begin{problem}[Eilenberg--Zilber]\label{prob:viii27-18}
What is the role of chain homotopy in the theorem?
\end{problem}
\begin{solution}
The Alexander--Whitney and shuffle/Eilenberg--Zilber maps compose to maps chain-homotopic to identities, so they induce actual inverse homology isomorphisms.
\end{solution}

\begin{problem}[Koszul sign]\label{prob:viii27-19}
Verify the cross terms in \(\partial^2(c\otimes d)\) cancel.
\end{problem}
\begin{solution}
The second application of the differential produces cross terms with opposite signs because the degree of \(\partial c\) is \(p-1\).
\end{solution}

\begin{problem}[Product cells]\label{prob:viii27-20}
How does a \(p\)-cell times a \(q\)-cell contribute?
\end{problem}
\begin{solution}
It gives a \((p+q)\)-cell, explaining both the tensor grading and Euler multiplicativity.
\end{solution}

\begin{problem}[Two streams]\label{prob:viii27-21}
Why is VIII/27 sourced from both topology-13 and topology-17?
\end{problem}
\begin{solution}
One stream carries the main product/Künneth theorem while the other carries complementary product calculations and homological machinery; both have the same canonical destination.
\end{solution}

\begin{problem}[Bridge]\label{prob:viii27-22}
How do products lead toward cup products?
\end{problem}
\begin{solution}
The cohomology cross product composed with the diagonal \(X\to X\times X\) produces the cup product.
\end{solution}

\begin{problem}[Final principle]\label{prob:viii27-23}
What does Künneth compute?
\end{problem}
\begin{solution}
It reconstructs homology of a product from factor homologies via tensor products plus Tor corrections.
\end{solution}

\begin{problem}[Comparison with UCT]\label{prob:viii27-24}
State the analogy with VIII/26.
\end{problem}
\begin{solution}
UCT changes coefficients for one complex; Künneth combines two complexes. Both are controlled by tensor products corrected by Tor.
\end{solution}
