\section{Supplementary worked examples}
\label{sec:viii19-pedagogy}

\begin{example}[The fundamental relative class of a disk]\label{ex:viii19-ped-01}
For the pair $(D^n,S^{n-1})$, the long exact sequence and the contractibility of $D^n$ give
\[
H_k(D^n,S^{n-1})\cong
\begin{cases}
\mathbb Z,&k=n,\\
0,&k\ne n.
\end{cases}
\]
The generator is represented by the oriented $n$-cell; its boundary becomes zero only after passing to relative chains.
\end{example}

\begin{example}[Relative homology as homology of a quotient]\label{ex:viii19-ped-02}
For a good pair $(X,A)$, collapsing $A$ to a point gives
\[
H_n(X,A)\cong \widetilde H_n(X/A).
\]
Thus $(D^n,S^{n-1})$ becomes $S^n$, turning the relative fundamental class into the ordinary reduced fundamental class of the sphere.
\end{example}

\begin{example}[Exactness detects what survives an inclusion]\label{ex:viii19-ped-03}
In the long exact sequence
\[
H_n(A)\xrightarrow{i_*}H_n(X)\xrightarrow{j_*}H_n(X,A)\xrightarrow{\partial}H_{n-1}(A),
\]
exactness says that an absolute class dies in relative homology exactly when it came from $A$, while the boundary of a relative class measures its residual boundary in $A$.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Relative chain group]\label{exr:viii19-ped-01}
Write $C_n(X,A)$ in terms of singular chain groups.
\end{exercise}
\begin{hint}
Quotient out chains supported in $A$.
\end{hint}
\begin{solution}
$C_n(X,A)=C_n(X)/C_n(A)$.
\end{solution}

\begin{exercise}[Empty subspace]\label{exr:viii19-ped-02}
Compute $H_n(X,\varnothing)$.
\end{exercise}
\begin{hint}
The relative chain quotient changes nothing.
\end{hint}
\begin{solution}
$H_n(X,\varnothing)\cong H_n(X)$.
\end{solution}

\begin{exercise}[Identical pair]\label{exr:viii19-ped-03}
Compute $H_n(X,X)$.
\end{exercise}
\begin{hint}
The relative chain complex is zero.
\end{hint}
\begin{solution}
$H_n(X,X)=0$ in every degree.
\end{solution}

\begin{exercise}[Disk pair]\label{exr:viii19-ped-04}
Compute $H_n(D^n,S^{n-1})$ and $H_k(D^n,S^{n-1})$ for $k\ne n$.
\end{exercise}
\begin{hint}
Use the long exact sequence of the pair.
\end{hint}
\begin{solution}
The only nonzero group is $H_n(D^n,S^{n-1})\cong\mathbb Z$.
\end{solution}

\begin{exercise}[Boundary map of a relative class]\label{exr:viii19-ped-05}
What does the connecting map $\partial:H_n(X,A)\to H_{n-1}(A)$ do to a relative cycle represented by $c$?
\end{exercise}
\begin{hint}
Take the ordinary boundary before passing to homology.
\end{hint}
\begin{solution}
Since $\partial c$ lies in $C_{n-1}(A)$, the connecting map sends $[c]$ to $[\partial c]$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Exactness at absolute homology]\label{exr:viii19-ped-06}
Prove $\ker j_*=\operatorname{im} i_*$ in the long exact sequence of a pair.
\end{exercise}
\begin{hint}
Translate $j_*[z]=0$ into a relative boundary statement.
\end{hint}
\begin{solution}
If $j_*[z]=0$, then $z=\partial c+a$ with $a\in C_n(A)$, so $[z]=i_*[a]$; the reverse inclusion follows because chains from $A$ vanish relatively.
\end{solution}

\begin{exercise}[Deformation retract pair]\label{exr:viii19-ped-07}
Show that if $A$ is a deformation retract of $X$, then $H_n(X,A)=0$ for all $n$.
\end{exercise}
\begin{hint}
The inclusion induces an isomorphism on homology.
\end{hint}
\begin{solution}
In the long exact sequence, $i_*:H_n(A)\to H_n(X)$ is an isomorphism in every degree, forcing all relative groups to vanish.
\end{solution}

\begin{exercise}[Naturality]\label{exr:viii19-ped-08}
Explain why a map of pairs $f:(X,A)\to(Y,B)$ induces a morphism between the two long exact sequences.
\end{exercise}
\begin{hint}
Use the induced chain map on quotients.
\end{hint}
\begin{solution}
$f_\#$ preserves subcomplexes, descends to relative chains, and commutes with boundaries; the connecting maps are therefore natural.
\end{solution}

\begin{exercise}[Reduced quotient relation]\label{exr:viii19-ped-09}
Under the usual good-pair hypothesis, prove the relation $H_n(X,A)\cong\widetilde H_n(X/A)$.
\end{exercise}
\begin{hint}
Compare the relative chain model with chains after collapsing $A$.
\end{hint}
\begin{solution}
The quotient map induces a chain equivalence between the relative complex and the reduced chain complex of $X/A$, hence an isomorphism on homology.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Naive quotient formula]\label{exr:viii19-ped-10}
Test: $H_n(X,A)$ is always $H_n(X)/H_n(A)$.
\end{exercise}
\begin{hint}
Look at the connecting homomorphism in the long exact sequence.
\end{hint}
\begin{solution}
False. The relative group also records classes whose boundary lies in $A$; the connecting map can contribute information not captured by a quotient of homology groups.
\end{solution}

\begin{exercise}[Separate homology determines the pair]\label{exr:viii19-ped-11}
Test: knowing $H_*(X)$ and $H_*(A)$ determines $H_*(X,A)$.
\end{exercise}
\begin{hint}
The inclusion map matters.
\end{hint}
\begin{solution}
False. The long exact sequence depends on the induced map $H_*(A)\to H_*(X)$, not only on the two groups separately.
\end{solution}

\begin{exercise}[Relative zero-cycles]\label{exr:viii19-ped-12}
Test: $H_0(X,A)$ is always zero when $A$ is nonempty.
\end{exercise}
\begin{hint}
Consider components of $X$ disjoint from $A$.
\end{hint}
\begin{solution}
False. $H_0(X,A)$ detects components of $X$ not met by $A$; it vanishes when $A$ meets every path component of $X$.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Cell attachment]\label{exr:viii19-ped-13}
Explain why relative homology is the natural language for adding an $n$-cell.
\end{exercise}
\begin{hint}
Compare $(X^n,X^{n-1})$.
\end{hint}
\begin{solution}
The relative group isolates the newly added cells: $H_k(X^n,X^{n-1})$ is free on the $n$-cells when $k=n$ and zero otherwise.
\end{solution}

\begin{exercise}[Local topology]\label{exr:viii19-ped-14}
How can $H_n(X,X-\{x\})$ be used as a local invariant near $x$?
\end{exercise}
\begin{hint}
Collapse everything away from a small neighborhood of $x$.
\end{hint}
\begin{solution}
It records homology supported near $x$; for an $n$-manifold it is locally $\mathbb Z$ in degree $n$ for orientable local coefficients and zero in other degrees.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Sphere from a pair]\label{exr:viii19-ped-15}
Recover $\widetilde H_*(S^n)$ from the pair $(D^n,S^{n-1})$.
\end{exercise}
\begin{hint}
Use $D^n/S^{n-1}\cong S^n$.
\end{hint}
\begin{solution}
The relative computation gives $\widetilde H_n(S^n)\cong\mathbb Z$ and zero in all other degrees.
\end{solution}

\begin{exercise}[Exact-sequence synthesis]\label{exr:viii19-ped-16}
Given $H_*(A)$, $H_*(X)$, and the inclusion map, describe a systematic workflow for computing $H_*(X,A)$.
\end{exercise}
\begin{hint}
Use exactness degree by degree.
\end{hint}
\begin{solution}
Insert the known groups and maps into the long exact sequence, compute kernels and cokernels of the inclusion maps, and resolve the short exact pieces determined by exactness.
\end{solution}
