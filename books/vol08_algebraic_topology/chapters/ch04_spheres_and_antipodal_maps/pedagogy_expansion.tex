\section{Supplementary worked examples}
\label{sec:viii04-pedagogy}

\begin{example}[Antipodal quotient]\label{ex:viii04-ped-01}
Identifying $x\sim -x$ on $S^n$ produces real projective space $\mathbb{RP}^n$.  The quotient map $S^n\to\mathbb{RP}^n$ is a two-sheeted covering for $n\ge1$.
\end{example}

\begin{example}[Antipodal degree parity]\label{ex:viii04-ped-02}
The antipodal map on $S^n$ has degree $(-1)^{n+1}$: it preserves orientation when $n$ is odd and reverses orientation when $n$ is even.
\end{example}

\begin{example}[Odd maps and antipodal symmetry]\label{ex:viii04-ped-03}
Maps satisfying $f(-x)=-f(x)$ respect the antipodal involution and therefore descend to maps on projective space.  Equivariance converts a symmetry condition upstairs into a well-defined quotient map downstairs.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Antipodal square]\label{exr:viii04-ped-01}
Compute $A\circ A$ for $A(x)=-x$.
\end{exercise}
\begin{hint}
Apply the sign twice.
\end{hint}
\begin{solution}
$A^2=\operatorname{id}_{S^n}$.
\end{solution}

\begin{exercise}[Fixed points]\label{exr:viii04-ped-02}
Does the antipodal map on $S^n$ have fixed points?
\end{exercise}
\begin{hint}
Solve $x=-x$ on the unit sphere.
\end{hint}
\begin{solution}
No; $x=-x$ implies $x=0$, which is not on $S^n$.
\end{solution}

\begin{exercise}[Degree on $S^3$]\label{exr:viii04-ped-03}
Compute the degree of the antipodal map on $S^3$.
\end{exercise}
\begin{hint}
Use $(-1)^{n+1}$.
\end{hint}
\begin{solution}
$(-1)^4=1$.
\end{solution}

\begin{exercise}[Degree on $S^2$]\label{exr:viii04-ped-04}
Compute the degree of the antipodal map on $S^2$.
\end{exercise}
\begin{hint}
Use the parity formula.
\end{hint}
\begin{solution}
$(-1)^3=-1$.
\end{solution}

\begin{exercise}[Quotient fiber]\label{exr:viii04-ped-05}
How many points are in a fiber of $S^n\to\mathbb{RP}^n$?
\end{exercise}
\begin{hint}
Each projective point is a line through the origin.
\end{hint}
\begin{solution}
Exactly two: $x$ and $-x$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Free action]\label{exr:viii04-ped-06}
Prove the antipodal $\mathbb Z/2$-action on $S^n$ is free.
\end{exercise}
\begin{hint}
A nontrivial stabilizer would give a fixed point.
\end{hint}
\begin{solution}
The nonidentity element sends $x$ to $-x$, and $x=-x$ has no unit-vector solution, so all stabilizers are trivial.
\end{solution}

\begin{exercise}[Quotient map is covering]\label{exr:viii04-ped-07}
Explain why the antipodal quotient is locally a two-sheeted covering.
\end{exercise}
\begin{hint}
Choose a small neighborhood disjoint from its antipode.
\end{hint}
\begin{solution}
For sufficiently small $U$ around $x$, $U\cap(-U)=\varnothing$; the quotient evenly covers its image by $U$ and $-U$.
\end{solution}

\begin{exercise}[Equivariant descent]\label{exr:viii04-ped-08}
Prove an odd map $f:S^n\to S^m$ descends to $\mathbb{RP}^n\to\mathbb{RP}^m$.
\end{exercise}
\begin{hint}
Check independence of the representative $x$ versus $-x$.
\end{hint}
\begin{solution}
Since $f(-x)=-f(x)$, the two representatives have the same projective image, so the quotient universal property gives a continuous descended map.
\end{solution}

\begin{exercise}[Orientation parity]\label{exr:viii04-ped-09}
Derive the orientation behavior of the antipodal map from its degree.
\end{exercise}
\begin{hint}
Orientation preserving means degree $+1$.
\end{hint}
\begin{solution}
$(-1)^{n+1}=1$ exactly for odd $n$, and equals $-1$ for even $n$.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Antipodal has fixed points]\label{exr:viii04-ped-10}
Test: the antipodal map always has two fixed points.
\end{exercise}
\begin{hint}
Do not confuse it with a reflection.
\end{hint}
\begin{solution}
False. It has no fixed points on any sphere.
\end{solution}

\begin{exercise}[Quotient is sphere]\label{exr:viii04-ped-11}
Test: $S^n/(x\sim -x)$ is always homeomorphic to $S^n$.
\end{exercise}
\begin{hint}
The quotient is projective space.
\end{hint}
\begin{solution}
False in general; it is $\mathbb{RP}^n$.
\end{solution}

\begin{exercise}[Degree always minus one]\label{exr:viii04-ped-12}
Test: the antipodal map always reverses orientation.
\end{exercise}
\begin{hint}
Check the parity of $n$.
\end{hint}
\begin{solution}
False. It preserves orientation for odd-dimensional spheres.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Projective modeling]\label{exr:viii04-ped-13}
Why is antipodal identification a natural model for unoriented directions?
\end{exercise}
\begin{hint}
A line does not distinguish $x$ from $-x$.
\end{hint}
\begin{solution}
Unit vectors encode directions with an orientation; quotienting by the antipodal relation forgets the sign and yields the space of unoriented lines.
\end{solution}

\begin{exercise}[Symmetry reduction]\label{exr:viii04-ped-14}
How does equivariance simplify maps between quotient spaces?
\end{exercise}
\begin{hint}
Work upstairs where spheres are simpler.
\end{hint}
\begin{solution}
An equivariant map between covering spaces descends automatically, allowing calculations on spheres before passing to projective quotients.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Odd-map obstruction]\label{exr:viii04-ped-15}
Suppose an odd map $S^n\to S^n$ were null-homotopic through odd maps. What extra structure must such a homotopy preserve?
\end{exercise}
\begin{hint}
Track antipodal equivariance at every time.
\end{hint}
\begin{solution}
It must satisfy $H(-x,t)=-H(x,t)$ for all $(x,t)$, so it descends to a homotopy on $\mathbb{RP}^n$; ordinary null-homotopy alone does not guarantee this.
\end{solution}

\begin{exercise}[Parity synthesis]\label{exr:viii04-ped-16}
For which $n$ can the antipodal map be homotopic to the identity based solely on degree obstruction?
\end{exercise}
\begin{hint}
Compare degrees.
\end{hint}
\begin{solution}
Degree does not obstruct a homotopy when $(-1)^{n+1}=1$, i.e. odd $n$; for even $n$ the degrees differ, so no homotopy exists.
\end{solution}
