\chapter{Cup Products}
\label{ch:viii29-cup-products}

Cohomology becomes substantially richer when its groups are multiplied.
The cup product turns \(H^*(X;R)\) into a graded ring and detects topology that
the additive groups alone can miss.

\section{Cochain-level cup product}

Let \(R\) be a commutative coefficient ring.  For singular cochains
\[
\alpha\in C^p(X;R),
\qquad
\beta\in C^q(X;R),
\]
define their cup product using the Alexander--Whitney front/back face formula.

\begin{definition}[Cup product]
\label{def:viii29-cup}
For a singular \((p+q)\)-simplex
\[
\sigma:[v_0,\dots,v_{p+q}]\to X,
\]
set
\[
(\alpha\smile\beta)(\sigma)
=
\alpha(\sigma|[v_0,\dots,v_p])\,
\beta(\sigma|[v_p,\dots,v_{p+q}]).
\]
\end{definition}

\section{Leibniz rule}

\begin{theorem}[Coboundary derivation rule]
\label{thm:viii29-leibniz}
\[
\delta(\alpha\smile\beta)
=
\delta\alpha\smile\beta
+
(-1)^p\alpha\smile\delta\beta.
\]
\end{theorem}

Therefore cocycles multiply to cocycles, and changing a factor by a coboundary
changes the product by a coboundary.

\section{Product on cohomology}

The cochain product descends to
\[
\smile:
H^p(X;R)\times H^q(X;R)
\to H^{p+q}(X;R).
\]

\section{Cohomology ring}

\begin{definition}[Cohomology ring]
\label{def:viii29-ring}
The graded group
\[
H^*(X;R)=\bigoplus_{n\ge0}H^n(X;R)
\]
with cup product is the cohomology ring of \(X\).
\end{definition}

The unit is the class
\[
1\in H^0(X;R)
\]
represented by the constant \(1\)-valued zero-cochain on a connected space.

\section{Associativity}
Cup product is associative on cohomology:
\[
(a\smile b)\smile c=a\smile(b\smile c).
\]

\section{Naturality}

\begin{theorem}[Naturality]
\label{thm:viii29-naturality}
For \(f:X\to Y\),
\[
f^*(a\smile b)=f^*a\smile f^*b.
\]
\end{theorem}

Thus \(f^*:H^*(Y;R)\to H^*(X;R)\) is a graded-ring homomorphism.

\section{Graded commutativity}

\begin{theorem}[Graded commutativity]
\label{thm:viii29-graded-comm}
For
\[
a\in H^p(X;R),
\qquad
b\in H^q(X;R),
\]
one has
\[
a\smile b=(-1)^{pq}b\smile a.
\]
\end{theorem}

This equality generally holds only after passing to cohomology, not strictly on
the standard cochain-level cup product.

\section{Odd-degree squares}

If \(a\) has odd degree and \(2\) is invertible in \(R\), then
\[
a\smile a=0.
\]

Over characteristic \(2\), the sign disappears, so odd-degree squares need not
vanish for sign reasons.

\section{Sphere ring}

For \(n>0\),
\[
H^*(S^n;\mathbb Z)
\cong
\mathbb Z[u]/(u^2),
\qquad |u|=n,
\]
where \(u^2=0\) for dimensional reasons.

\section{Circle ring}

\[
H^*(S^1;\mathbb Z)
\cong
\Lambda_{\mathbb Z}(a),
\qquad |a|=1.
\]

Thus
\[
a^2=0.
\]

\section{Torus ring}

Let \(a,b\in H^1(T^2;\mathbb Z)\) be dual to the two fundamental one-cycles.
Then
\[
H^*(T^2;\mathbb Z)
\cong
\Lambda_{\mathbb Z}(a,b).
\]

The top class is
\[
a\smile b\in H^2(T^2;\mathbb Z)\cong\mathbb Z,
\]
with
\[
b\smile a=-a\smile b.
\]

\section{Higher tori}

For
\[
T^n=(S^1)^n,
\]
\[
H^*(T^n;\mathbb Z)
\cong
\Lambda_{\mathbb Z}(a_1,\dots,a_n),
\qquad |a_i|=1.
\]

This multiplicative structure refines the binomial-rank computation from
Künneth.

\section{Real projective space mod 2}

Let
\[
x\in H^1(\mathbb RP^n;\mathbb F_2)
\]
be the degree-one generator.

A fundamental result is
\[
H^*(\mathbb RP^n;\mathbb F_2)
\cong
\mathbb F_2[x]/(x^{n+1}),
\qquad |x|=1.
\]

Thus
\[
x^k\ne0\quad(0\le k\le n).
\]

\section{Why the ring is stronger}

As graded vector spaces, many spaces can have the same Betti numbers.  The cup
product records which classes multiply nontrivially and can distinguish such
spaces.

\section{Example: wedge versus product}

Both
\[
S^1\vee S^1\vee S^2
\]
and
\[
T^2
\]
have additive homology ranks
\[
b_0=1,\quad b_1=2,\quad b_2=1.
\]

But on the wedge, products of positive-degree classes from different wedge
summands vanish, whereas on the torus
\[
a\smile b
\]
generates \(H^2\).

Hence the spaces are not homotopy equivalent.

\section{Cross product and diagonal}

The cup product may be described conceptually as
\[
a\smile b
=
\Delta^*(a\times b),
\]
where
\[
\Delta:X\to X\times X,
\qquad
\Delta(x)=(x,x),
\]
is the diagonal and \(\times\) is the cohomology cross product.

\section{Kronecker pairing compatibility}

Cup products interact with homology cross products through evaluation:
cohomology products detect product cycles and, on manifolds, later encode
intersection pairings.

\section{Relative cup products}

There are relative forms
\[
H^p(X,A;R)\times H^q(X,B;R)
\to
H^{p+q}(X,A\cup B;R)
\]
under suitable hypotheses.

These become important for Thom classes and duality.

\section{Cup length}

A basic ring invariant is cup length: the largest number of positive-degree
classes whose product is nonzero.

Cup length provides lower bounds for several topological complexity
invariants.

\section{Maps and ring obstructions}

If \(f:X\to Y\) is a homotopy equivalence, then
\[
f^*:H^*(Y;R)\to H^*(X;R)
\]
is a graded-ring isomorphism.

Therefore a mismatch of cohomology rings obstructs homotopy equivalence.

Cohomology rings are the natural receptacle for characteristic classes of
bundles.  VIII/30 first develops vector bundles and clutching; Thom and Euler

\section{Solved problem dossiers}

\begin{problem}[Leibniz]\label{prob:viii29-01}
Why does the cup product descend to cohomology?
\end{problem}
\begin{solution}
The Leibniz rule shows cocycles multiply to cocycles; if one factor changes by a coboundary, the product changes by a coboundary.
\end{solution}

\begin{problem}[Naturality]\label{prob:viii29-02}
Prove \(f^*(a\smile b)=f^*a\smile f^*b\).
\end{problem}
\begin{solution}
The Alexander--Whitney face formula commutes with precomposition by \(f\), so the equality already holds on cochains.
\end{solution}

\begin{problem}[Odd square]\label{prob:viii29-03}
Let \(a\in H^1(X;\mathbb Q)\). Show \(a^2=0\).
\end{problem}
\begin{solution}
Graded commutativity gives \(a^2=-a^2\), so \(2a^2=0\); over \(\mathbb Q\), \(2\) is invertible.
\end{solution}

\begin{problem}[Circle]\label{prob:viii29-04}
Compute the integral cohomology ring of \(S^1\).
\end{problem}
\begin{solution}
It is \(\Lambda(a)\) with \(|a|=1\); dimensional reasons and graded commutativity give \(a^2=0\).
\end{solution}

\begin{problem}[Torus]\label{prob:viii29-05}
Describe the ring \(H^*(T^2;\mathbb Z)\).
\end{problem}
\begin{solution}
It is the exterior algebra on \(a,b\in H^1\), with top generator \(a\smile b\) and \(b\smile a=-a\smile b\).
\end{solution}

\begin{problem}[Three-torus]\label{prob:viii29-06}
How many degree-two ring generators arise as products in \(T^3\)?
\end{problem}
\begin{solution}
The products \(a_1a_2,a_1a_3,a_2a_3\) give three independent degree-two classes.
\end{solution}

\begin{problem}[Top torus class]\label{prob:viii29-07}
What generates \(H^n(T^n;\mathbb Z)\)?
\end{problem}
\begin{solution}
The product \(a_1\smile\cdots\smile a_n\).
\end{solution}

\begin{problem}[RP2 mod2]\label{prob:viii29-08}
Describe \(H^*(\mathbb RP^2;\mathbb F_2)\).
\end{problem}
\begin{solution}
\(\mathbb F_2[x]/(x^3)\) with \(|x|=1\); thus \(x^2\) is the nonzero top class.
\end{solution}

\begin{problem}[RP3 mod2]\label{prob:viii29-09}
Describe the nonzero powers of the degree-one generator.
\end{problem}
\begin{solution}
\(1,x,x^2,x^3\) are nonzero and \(x^4=0\).
\end{solution}

\begin{problem}[Wedge versus torus]\label{prob:viii29-10}
Explain why \(S^1\vee S^1\vee S^2\) is not homotopy equivalent to \(T^2\) despite equal Betti numbers.
\end{problem}
\begin{solution}
The wedge has trivial products between positive-degree classes from different summands, while the torus has \(a\smile b\ne0\).
\end{solution}

\begin{problem}[Diagonal]\label{prob:viii29-11}
Explain \(a\smile b=\Delta^*(a\times b)\).
\end{problem}
\begin{solution}
Take the cohomology cross product on \(X\times X\) and pull it back along the diagonal; the Alexander--Whitney formula realizes this construction.
\end{solution}

\begin{problem}[Ring map obstruction]\label{prob:viii29-12}
If \(f:X\to Y\) is a homotopy equivalence, what must happen to cup products?
\end{problem}
\begin{solution}
\(f^*\) is an isomorphism preserving multiplication, so the graded cohomology rings must be isomorphic.
\end{solution}

\begin{problem}[Cup length]\label{prob:viii29-13}
What does a nonzero product \(a_1\cdots a_r\) prove?
\end{problem}
\begin{solution}
The cup length is at least \(r\), providing a multiplicative lower bound on topological complexity.
\end{solution}

\begin{problem}[Characteristic two]\label{prob:viii29-14}
Why need an odd-degree mod-2 class not square to zero?
\end{problem}
\begin{solution}
Graded commutativity contributes the sign \(-1\), but \(-1=1\) over \(\mathbb F_2\), so no sign-forced cancellation occurs.
\end{solution}

\begin{problem}[Sphere]\label{prob:viii29-15}
Why is the square of the top sphere class zero?
\end{problem}
\begin{solution}
It would lie in degree \(2n\), where \(H^{2n}(S^n)=0\) for \(n>0\).
\end{solution}

\begin{problem}[Relative product]\label{prob:viii29-16}
State a useful relative cup product.
\end{problem}
\begin{solution}
\(H^p(X,A)\times H^q(X,B)\to H^{p+q}(X,A\cup B)\).
\end{solution}

\begin{problem}[Pairing]\label{prob:viii29-17}
How can a cup product be tested against homology?
\end{problem}
\begin{solution}
Evaluate \(a\smile b\) on an appropriate \((p+q)\)-cycle using the Kronecker pairing.
\end{solution}

\begin{problem}[Product vs additive]\label{prob:viii29-18}
What extra information does the ring contain beyond cohomology groups?
\end{problem}
\begin{solution}
It records which additive classes multiply to nonzero higher-degree classes and the relations among these products.
\end{solution}

\begin{problem}[SVG torus]\label{prob:viii29-19}
What should the torus-vs-wedge SVG show?
\end{problem}
\begin{solution}
Identical Betti-number rows but different multiplication: a nonzero \(a\smile b\) cell for the torus and zero mixed products for the wedge.
\end{solution}

\begin{problem}[Bridge]\label{prob:viii29-20}
Why are cup products relevant to bundles?
\end{problem}
\begin{solution}
Characteristic classes lie in cohomology and multiply; their ring relations encode bundle and manifold structure.
\end{solution}

\begin{problem}[Override provenance]\label{prob:viii29-21}
Why is cup-product material assigned here rather than to Poincaré duality?
\end{problem}
\begin{solution}
The migration ledger contains a raw-source-verified targeted override for cup-product/cohomology-ring sections into VIII/29.
\end{solution}

\begin{problem}[Final principle]\label{prob:viii29-22}
State the main improvement from cohomology groups to the cohomology ring.
\end{problem}
\begin{solution}
The ring retains multiplicative interactions between classes, making it strictly more informative than the graded groups alone.
\end{solution}
