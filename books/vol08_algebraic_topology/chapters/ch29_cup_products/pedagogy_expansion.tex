\section{Supplementary worked examples}
\label{sec:viii29-pedagogy}

\begin{example}[The torus cohomology ring]\label{ex:viii29-ped-01}
Let $a,b\in H^1(T^2;\mathbb Z)$ be dual to the two circle factors. Then
\[
a\smile a=b\smile b=0,\qquad a\smile b=-b\smile a,
\]
and $a\smile b$ generates $H^2(T^2;\mathbb Z)$. Thus the ring remembers how the two independent one-dimensional directions combine to form the fundamental class.
\end{example}

\begin{example}[Cup products on a sphere]\label{ex:viii29-ped-02}
If $u\in H^n(S^n)$ with $n>0$, then $u\smile u$ lies in $H^{2n}(S^n)=0$. Hence the positive-degree part of the sphere cohomology ring has trivial multiplication.
\end{example}

\begin{example}[Projective-space power structure]\label{ex:viii29-ped-03}
With $\mathbb F_2$ coefficients, $\mathbb{RP}^m$ has a degree-one class $x$ whose powers survive until dimension $m$:
\[
H^*(\mathbb{RP}^m;\mathbb F_2)\cong \mathbb F_2[x]/(x^{m+1}).
\]
This ring structure is much richer than the list of Betti numbers alone.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Degree of product]\label{exr:viii29-ped-01}
If $\alpha\in H^p(X)$ and $\beta\in H^q(X)$, what is the degree of $\alpha\smile\beta$?
\end{exercise}
\begin{hint}
Degrees add.
\end{hint}
\begin{solution}
It lies in $H^{p+q}(X)$.
\end{solution}

\begin{exercise}[Unit]\label{exr:viii29-ped-02}
What class is the multiplicative unit in the cohomology ring of a connected space?
\end{exercise}
\begin{hint}
Look in $H^0(X)$.
\end{hint}
\begin{solution}
The class of the constant function $1$ in $H^0(X)$ is the unit.
\end{solution}

\begin{exercise}[Sphere square]\label{exr:viii29-ped-03}
For $u\in H^n(S^n)$ with $n>0$, compute $u^2$.
\end{exercise}
\begin{hint}
Check $H^{2n}(S^n)$.
\end{hint}
\begin{solution}
$u^2=0$.
\end{solution}

\begin{exercise}[Torus top product]\label{exr:viii29-ped-04}
For degree-one generators $a,b$ of $T^2$, what group contains $a\smile b$?
\end{exercise}
\begin{hint}
Degrees add to $2$.
\end{hint}
\begin{solution}
$a\smile b\in H^2(T^2)\cong\mathbb Z$, and with compatible orientations it is a generator.
\end{solution}

\begin{exercise}[Graded sign]\label{exr:viii29-ped-05}
State graded commutativity for $\alpha\in H^p$, $\beta\in H^q$.
\end{exercise}
\begin{hint}
Interchanging factors contributes a sign.
\end{hint}
\begin{solution}
$\alpha\smile\beta=(-1)^{pq}\beta\smile\alpha$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Naturality]\label{exr:viii29-ped-06}
Prove pullback preserves cup products.
\end{exercise}
\begin{hint}
Use naturality of the cochain-level diagonal approximation.
\end{hint}
\begin{solution}
For $f:X\to Y$, $f^*(\alpha\smile\beta)=f^*\alpha\smile f^*\beta$, so $f^*$ is a graded-ring homomorphism.
\end{solution}

\begin{exercise}[Odd square over Z]\label{exr:viii29-ped-07}
Show that if $\alpha$ has odd degree and cohomology has no $2$-torsion in degree $2|\alpha|$, then $\alpha^2=0$.
\end{exercise}
\begin{hint}
Use graded commutativity.
\end{hint}
\begin{solution}
$\alpha^2=-\alpha^2$, hence $2\alpha^2=0$; without $2$-torsion this forces $\alpha^2=0$.
\end{solution}

\begin{exercise}[Product detects dimension]\label{exr:viii29-ped-08}
If $\alpha_1\smile\cdots\smile\alpha_r\ne0$, prove the total degree cannot exceed the cohomological dimension.
\end{exercise}
\begin{hint}
The product lives in the sum degree.
\end{hint}
\begin{solution}
A nonzero class must live in a nonzero cohomology group, so the total degree is bounded by the largest degree with nonzero cohomology.
\end{solution}

\begin{exercise}[Ring distinguishes spaces]\label{exr:viii29-ped-09}
Explain how different products can distinguish spaces with the same additive cohomology groups.
\end{exercise}
\begin{hint}
An additive isomorphism need not preserve multiplication.
\end{hint}
\begin{solution}
If one space has a nonzero product of classes where the corresponding product must vanish in the other, their cohomology rings are not isomorphic and the spaces cannot be homotopy equivalent.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Cup product strictly commutative]\label{exr:viii29-ped-10}
Test: $\alpha\smile\beta=\beta\smile\alpha$ over $\mathbb Z$ in all degrees.
\end{exercise}
\begin{hint}
Remember the graded sign.
\end{hint}
\begin{solution}
False. The sign $(-1)^{pq}$ appears.
\end{solution}

\begin{exercise}[All positive products vanish]\label{exr:viii29-ped-11}
Test: positive-degree cup products always vanish because degree increases.
\end{exercise}
\begin{hint}
Consider the torus.
\end{hint}
\begin{solution}
False. On $T^2$, two degree-one classes have a nonzero degree-two product.
\end{solution}

\begin{exercise}[Additive cohomology enough]\label{exr:viii29-ped-12}
Test: Betti numbers determine the cohomology ring.
\end{exercise}
\begin{hint}
Multiplication is extra structure.
\end{hint}
\begin{solution}
False; Betti numbers record only graded dimensions.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Cup-length obstruction]\label{exr:viii29-ped-13}
How can a nonzero long cup product obstruct a space from being too simple?
\end{exercise}
\begin{hint}
Use cup length as a homotopy invariant.
\end{hint}
\begin{solution}
A nonzero product of many positive-degree classes gives a lower bound on invariants such as Lusternik--Schnirelmann category and rules out homotopy types whose rings have shorter cup length.
\end{solution}

\begin{exercise}[Orientation pairing]\label{exr:viii29-ped-14}
On a closed oriented surface, how does the cup product encode intersection of one-dimensional classes?
\end{exercise}
\begin{hint}
Evaluate the degree-two product on the fundamental class.
\end{hint}
\begin{solution}
The integer $\langle \alpha\smile\beta,[\Sigma]\rangle$ is the algebraic intersection pairing of the Poincare-dual one-cycles.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[RP2 mod 2]\label{exr:viii29-ped-15}
Let $x\in H^1(\mathbb{RP}^2;\mathbb F_2)$. What are $x^2$ and $x^3$?
\end{exercise}
\begin{hint}
Use $\mathbb F_2[x]/(x^3)$.
\end{hint}
\begin{solution}
$x^2$ is the nonzero top class and $x^3=0$.
\end{solution}

\begin{exercise}[Ring synthesis]\label{exr:viii29-ped-16}
Why is the cup product a decisive upgrade from cohomology groups to a cohomology ring?
\end{exercise}
\begin{hint}
It records interactions among classes.
\end{hint}
\begin{solution}
The graded groups say which classes exist; the cup product records which classes combine nontrivially, yielding finer homotopy information, intersection pairings, characteristic classes, and duality structures.
\end{solution}
