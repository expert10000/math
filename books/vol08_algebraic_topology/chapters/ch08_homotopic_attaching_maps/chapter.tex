\chapter{Homotopic Attaching Maps}
\label{ch:viii08-homotopic-attaching}

The topology of an adjunction space depends on its attaching map, but it does
not depend on that map pointwise.  Under the standard CW/cofibration
hypotheses, deforming the attaching map through a homotopy preserves the
homotopy type of the resulting space.  This chapter makes that principle
precise and develops the homotopy-extension mechanism behind it.

\section{The basic problem}

Let
\[
\varphi_0,\varphi_1:S^{n-1}\to X
\]
be homotopic maps.  Form
\[
Y_i=X\cup_{\varphi_i}D^n.
\]
The two quotient spaces need not be literally identical, but one expects
\[
Y_0\simeq Y_1.
\]

\section{Homotopy extension property}

\begin{definition}[Homotopy extension property]
\label{def:viii08-hep}
A pair \((X,A)\) has the homotopy extension property if, whenever
\[
f:X\to Y
\]
and
\[
H:A\times I\to Y
\]
satisfy \(H(a,0)=f(a)\), there exists
\[
\widetilde H:X\times I\to Y
\]
with
\[
\widetilde H(x,0)=f(x),
\qquad
\widetilde H|_{A\times I}=H.
\]
\end{definition}

\begin{definition}[Cofibration]
\label{def:viii08-cofibration}
An inclusion \(i:A\hookrightarrow X\) is a cofibration if the pair
\((X,A)\) has the homotopy extension property.
\end{definition}

\section{CW pairs are cofibrations}

\begin{theorem}[CW subcomplex inclusion]
\label{thm:viii08-cw-cofibration}
If \(A\) is a CW subcomplex of a CW complex \(X\), then
\[
A\hookrightarrow X
\]
is a cofibration.
\end{theorem}

This theorem is one of the main reasons CW complexes form a convenient
category for homotopy theory.

\section{Homotopic attachment theorem}

\begin{theorem}[Homotopic attaching maps]
\label{thm:viii08-homotopic-attachment}
Let \(X\) be a CW complex and let
\[
\varphi_0,\varphi_1:S^{n-1}\to X
\]
be homotopic.  Then
\[
X\cup_{\varphi_0}D^n
\simeq
X\cup_{\varphi_1}D^n.
\]
\end{theorem}

The conclusion is homotopy equivalence, not generally homeomorphism.

\section{Idea of the proof}

Let
\[
H:S^{n-1}\times I\to X
\]
be a homotopy from \(\varphi_0\) to \(\varphi_1\).  One can insert a collar
between the old space and the disk boundary and use \(H\) along this collar.
The resulting intermediate adjunction space contains both attachments as
deformation retracts.

Equivalently, one can use a mapping-cylinder construction to replace the
attaching map by a homotopic representative without changing homotopy type.

\section{A cylinder model}

Attach
\[
S^{n-1}\times I
\]
to \(X\) using \(H\), and place \(D^n\) at the far end.  At one end the boundary
is attached by \(\varphi_0\); at the other by \(\varphi_1\).  Sliding along the
cylinder provides the comparison homotopies.

\begin{proposition}[Interpolation space]
\label{prop:viii08-interpolation}
There is a space \(Z_H\) constructed from the homotopy \(H\) that deformation
retracts onto each of
\[
X\cup_{\varphi_0}D^n
\quad\text{and}\quad
X\cup_{\varphi_1}D^n.
\]
\end{proposition}

\section{Constant versus null-homotopic attachment}

A null-homotopic attaching map need not be literally constant, but it produces
the same homotopy type as a constant attachment.

\begin{corollary}[Null attachment splitting]
\label{cor:viii08-null-attachment}
If
\[
\varphi:S^{n-1}\to X
\]
is null-homotopic, then
\[
X\cup_\varphi D^n
\simeq
X\vee S^n
\]
under the usual pointed CW hypotheses.
\end{corollary}

\section{Degree classifies circle attachments up to homotopy}

For maps
\[
S^1\to S^1,
\]
homotopy classes are classified by degree.

\begin{corollary}[Degree determines two-cell attachment type]
\label{cor:viii08-degree}
If
\[
\varphi,\psi:S^1\to S^1
\]
have the same degree, then
\[
S^1\cup_\varphi D^2
\simeq
S^1\cup_\psi D^2.
\]
\end{corollary}

This does not say the two quotient spaces are homeomorphic for arbitrary
representatives; it says the homotopy type is determined by the homotopy class
of the attachment.

\section{Several cells}

Suppose a family of \(n\)-cells is attached by maps
\[
\varphi_\alpha:S^{n-1}\to X.
\]
If each \(\varphi_\alpha\) is replaced by a homotopic map
\(\psi_\alpha\), then under the standard CW hypotheses the resulting spaces are
homotopy equivalent.

\begin{theorem}[Simultaneous homotopic attachments]
\label{thm:viii08-simultaneous}
Homotopic replacement of a family of attaching maps preserves the homotopy type
of the corresponding CW attachment stage.
\end{theorem}

\section{Relative version}

If the attaching-map homotopy is fixed on a specified subspace of the boundary,
the comparison can be performed relative to the corresponding attached
subspace.  Relative homotopies therefore fit naturally into the attachment
theory developed in VIII/01.

\section{Mapping-cone formulation}

For maps
\[
f_0,f_1:A\to X
\]
with \(f_0\simeq f_1\), mapping cones satisfy
\[
C_{f_0}\simeq C_{f_1}
\]
under the standard cofibration/CW assumptions.

\begin{theorem}[Homotopy invariance of mapping cones]
\label{thm:viii08-cones}
Homotopic maps between suitable CW spaces have homotopy-equivalent mapping
cones.
\end{theorem}

This is the general form of the cell-attachment theorem.

\section{Why cofibrations matter}

A naive quotient argument can fail if the inclusion being pushed out behaves
badly.  Cofibrations guarantee that a homotopy on the attaching subspace can be
extended and controlled inside the ambient construction.

\begin{warning}[Do not replace cofibration hypotheses silently]
\label{warn:viii08-cofibration}
The theorem is robust for CW attachments because sphere-boundary inclusions and
CW-subcomplex inclusions are cofibrations.  For arbitrary quotient
constructions, one should not assume homotopy invariance without checking the
relevant hypotheses.
\end{warning}

\section{Application: different formulas, same cell complex homotopy type}

Two polygonal attaching loops can look quite different as parametrized maps but
represent the same element of
\[
[S^1,X].
\]
When they are homotopic, attaching a two-cell along either loop yields the same
homotopy type.

\section{Application to null loops}

If a loop
\[
\varphi:S^1\to X
\]
is null-homotopic, adding a two-cell along \(\varphi\) has homotopy type
\[
X\vee S^2.
\]
This is often easier to recognize than the original quotient.

\section{Dependence only on homotopy class}

The correct conceptual object attached to a cell is not the raw parametrized
formula
\[
\varphi:S^{n-1}\to X,
\]
but its homotopy class
\[
[\varphi]\in[S^{n-1},X],
\]
as far as the resulting homotopy type is concerned.

\section{Limits of the conclusion}

Homotopy-equivalent attachment spaces may differ in geometric or quotient-level
details.  The theorem does not preserve a chosen cell parametrization, metric,
smooth structure, or literal point-set presentation.

\begin{warning}[Homotopy equivalence is the invariant]
\label{warn:viii08-homeomorphism}
Never strengthen the conclusion
\[
X\cup_{\varphi_0}D^n\simeq X\cup_{\varphi_1}D^n
\]
to homeomorphism without additional proof.
\end{warning}

\section{Bridge to the fundamental group}

For two-cell attachments to a graph, the homotopy class of the attaching loop
is encoded by an element of the fundamental group.  Thus the next chapter's
group law on loops gives an algebraic language for the attachment data of
two-dimensional CW complexes.

\section{Solved problem dossiers}

\begin{problem}[Null attachment]\label{prob:viii08-01}
Let \(\varphi:S^{n-1}\to X\) be null-homotopic.  Derive the splitting of the attachment.
\end{problem}
\begin{solution}
Choose a homotopy from \(\varphi\) to a constant map \(c\).  The homotopic-attaching-map theorem gives \(X\cup_\varphi D^n\simeq X\cup_cD^n\), and the constant attachment has homotopy type \(X\vee S^n\).
\end{solution}

\begin{problem}[Degree-three representatives]\label{prob:viii08-02}
Let \(\varphi,\psi:S^1\to S^1\) both have degree \(3\). Compare the two attached spaces.
\end{problem}
\begin{solution}
Circle maps of equal degree are homotopic. Therefore \(S^1\cup_\varphi D^2\simeq S^1\cup_\psi D^2\).
\end{solution}

\begin{problem}[Degree mismatch]\label{prob:viii08-03}
Suppose \(\deg\varphi=2\) and \(\deg\psi=3\). Can VIII/08 identify their attachments?
\end{problem}
\begin{solution}
No.  The maps are not homotopic, so the theorem does not apply.  Later cellular homology in fact distinguishes the corresponding degree attachments.
\end{solution}

\begin{problem}[Constant versus wiggly null loop]\label{prob:viii08-04}
A loop in \(X\) is visibly complicated but null-homotopic. What two-cell attachment homotopy type results?
\end{problem}
\begin{solution}
Replace the loop by the constant loop through a null-homotopy.  The result is homotopy equivalent to \(X\vee S^2\).
\end{solution}

\begin{problem}[Cylinder interpolation]\label{prob:viii08-05}
Explain how \(S^{n-1}\times I\) records a homotopy between two attachments.
\end{problem}
\begin{solution}
At time \(0\) its base is glued to \(X\) by \(\varphi_0\), at time \(1\) by \(\varphi_1\), and intermediate slices follow \(H(-,t)\).  Sliding along this collar produces comparison deformation retracts.
\end{solution}

\begin{problem}[Why not homeomorphic?]\label{prob:viii08-06}
Why does the theorem not automatically produce a homeomorphism between the endpoint quotient spaces?
\end{problem}
\begin{solution}
The construction controls deformations and deformation retracts, not pointwise quotient identifications.  Homotopy equivalence is invariant under the interpolation, while homeomorphism would require stronger point-set control.
\end{solution}

\begin{problem}[Mapping-cone corollary]\label{prob:viii08-07}
Deduce the homotopy invariance of \(C_f\) from the attachment principle.
\end{problem}
\begin{solution}
A mapping cone is obtained by attaching the cone \(CA\) along its base \(A\) using \(f\).  Replacing \(f\) by a homotopic map changes the attachment through a homotopy, hence preserves the cone homotopy type under the cofibration hypotheses.
\end{solution}

\begin{problem}[CW-pair role]\label{prob:viii08-08}
Why does the theorem apply cleanly to a cell attachment in a CW complex?
\end{problem}
\begin{solution}
The boundary inclusion \(S^{n-1}\hookrightarrow D^n\) and CW-subcomplex inclusions are cofibrations, so the required homotopies can be extended in the pushout construction.
\end{solution}

\begin{problem}[Two-cell graph attachment]\label{prob:viii08-09}
Let \(X\) be a graph and let two loops \(\alpha,\beta:S^1\to X\) be homotopic rel basepoint. Compare the attached two-complexes.
\end{problem}
\begin{solution}
Attaching one two-cell along \(\alpha\) or \(\beta\) produces homotopy-equivalent spaces.  The based homotopy is stronger than needed for the unbased conclusion.
\end{solution}

\begin{problem}[Relative attachment]\label{prob:viii08-10}
Suppose a homotopy of attaching maps is fixed on a subset \(B\subset S^{n-1}\). What extra control should the comparison preserve?
\end{problem}
\begin{solution}
The interpolation can be performed relative to the image of \(B\), so the resulting homotopy equivalence can preserve the corresponding attached subspace.
\end{solution}

\begin{problem}[Family of null attachments]\label{prob:viii08-11}
Attach \(r\) \(n\)-cells to \(X\) by null-homotopic maps. Predict the homotopy type.
\end{problem}
\begin{solution}
Replace all attaching maps simultaneously by constants.  Under the standard pointed hypotheses the result is homotopy equivalent to \(X\vee\bigvee_{j=1}^rS^n\).
\end{solution}

\begin{problem}[HEP extension]\label{prob:viii08-12}
Given \(f:X\to Y\) and a homotopy \(H:A\times I\to Y\) starting at \(f|_A\), what does HEP produce?
\end{problem}
\begin{solution}
It produces \(\widetilde H:X\times I\to Y\) extending \(H\) and satisfying \(\widetilde H(-,0)=f\).
\end{solution}

\begin{problem}[Subcomplex example]\label{prob:viii08-13}
Give a concrete cofibration from the previous chapters.
\end{problem}
\begin{solution}
The inclusion \(S^{n-1}\hookrightarrow D^n\) is an inclusion of a CW subcomplex and therefore a cofibration.
\end{solution}

\begin{problem}[Attachment simplification]\label{prob:viii08-14}
Why is it legitimate to replace an attaching loop by a simpler loop in the same homotopy class before computing invariants?
\end{problem}
\begin{solution}
VIII/08 ensures the replacement preserves the homotopy type of the resulting CW complex, and homotopy invariants therefore have the same values.
\end{solution}

\begin{problem}[Same word, different parametrization]\label{prob:viii08-15}
Two polygon boundary maps traverse the same loop word but with different speeds and pauses. Compare their attachments.
\end{problem}
\begin{solution}
Reparameterization gives a homotopy of the loops, so their two-cell attachments are homotopy equivalent.
\end{solution}

\begin{problem}[Reverse parametrization warning]\label{prob:viii08-16}
Is a loop necessarily homotopic to the same geometric loop traversed backward?
\end{problem}
\begin{solution}
No.  Reversal represents the inverse fundamental-group element and need not be homotopic to the original loop.  VIII/09 makes this distinction algebraic.
\end{solution}

\begin{problem}[Degree-zero circle attachment]\label{prob:viii08-17}
Let \(\varphi:S^1\to S^1\) have degree \(0\). Identify the attachment homotopy type.
\end{problem}
\begin{solution}
Every degree-zero circle map is null-homotopic, so \(S^1\cup_\varphi D^2\simeq S^1\vee S^2\).
\end{solution}

\begin{problem}[Degree-one circle attachment]\label{prob:viii08-18}
Let \(\varphi:S^1\to S^1\) have degree \(1\). Compare its attachment with the identity attachment.
\end{problem}
\begin{solution}
\(\varphi\) is homotopic to the identity, so the attachment is homotopy equivalent to \(S^1\cup_{\operatorname{id}}D^2\cong D^2\), hence contractible.
\end{solution}

\begin{problem}[Homotopy-class principle]\label{prob:viii08-19}
State precisely what attachment data VIII/08 says can be simplified.
\end{problem}
\begin{solution}
For homotopy-type questions under the standard hypotheses, the attaching map may be replaced by any representative of its homotopy class in \([S^{n-1},X]\).
\end{solution}

\begin{problem}[Bridge to \(\pi_1\)]\label{prob:viii08-20}
Why is the fundamental group the natural next tool for two-cell attachments?
\end{problem}
\begin{solution}
An attaching map for a two-cell is a loop \(S^1\to X^1\).  Based homotopy classes of loops form \(\pi_1(X^1)\), converting the attaching data into group elements and words.
\end{solution}
