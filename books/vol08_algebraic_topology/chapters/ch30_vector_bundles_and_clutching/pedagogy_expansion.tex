\section{Supplementary worked examples}
\label{sec:viii30-pedagogy}

\begin{example}[The Mobius line bundle]\label{ex:viii30-ped-01}
Glue $[0,1]\times\mathbb R$ by $(0,v)\sim(1,-v)$. The result is a real line bundle over $S^1$ with nontrivial clutching sign. It has no nowhere-zero section compatible with a global choice of orientation, distinguishing it from the trivial cylinder bundle.
\end{example}

\begin{example}[Clutching a complex line bundle on S2]\label{ex:viii30-ped-02}
Write $S^2=D^2_+\cup D^2_-$ and glue trivial complex line bundles along the equator by
\[
g_k:S^1\to U(1),\qquad g_k(z)=z^k.
\]
Different winding numbers $k$ yield different complex line bundles; $k$ becomes the first Chern number.
\end{example}

\begin{example}[Tangent bundle of the circle]\label{ex:viii30-ped-03}
The tangent bundle $TS^1$ is trivial: the vector field $v(z)=iz$ is continuous and nowhere zero, giving a global frame. This contrasts with higher-dimensional sphere tangent bundles where global frames are constrained by Euler classes.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Bundle chart]\label{exr:viii30-ped-01}
What data locally identifies a rank-$r$ vector bundle with a product?
\end{exercise}
\begin{hint}
Choose a trivialization over an open set.
\end{hint}
\begin{solution}
A local trivialization is a homeomorphism or bundle isomorphism $\pi^{-1}(U)\cong U\times\mathbb F^r$ respecting projection and linear fiber structure.
\end{solution}

\begin{exercise}[Transition function]\label{exr:viii30-ped-02}
Where does a transition function take values for a real rank-$r$ vector bundle?
\end{exercise}
\begin{hint}
It changes one local frame to another.
\end{hint}
\begin{solution}
It takes values in $GL_r(\mathbb R)$.
\end{solution}

\begin{exercise}[Trivial line bundle section]\label{exr:viii30-ped-03}
Give a nowhere-zero section of $S^1\times\mathbb R\to S^1$.
\end{exercise}
\begin{hint}
Use a constant fiber vector.
\end{hint}
\begin{solution}
$s(x)=(x,1)$ is a nowhere-zero section.
\end{solution}

\begin{exercise}[Clutching domain for Sn]\label{exr:viii30-ped-04}
For a bundle over $S^n=D^n_+\cup D^n_-$, on what space is the clutching map defined?
\end{exercise}
\begin{hint}
Use the equatorial overlap.
\end{hint}
\begin{solution}
On $S^{n-1}$.
\end{solution}

\begin{exercise}[Mobius clutching]\label{exr:viii30-ped-05}
What is the clutching datum of the Mobius line bundle over $S^1$?
\end{exercise}
\begin{hint}
The transition across the overlap reverses sign.
\end{hint}
\begin{solution}
It uses the nontrivial component of $GL_1(\mathbb R)=\mathbb R^\times$, represented by multiplication by $-1$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Equivalent clutching maps]\label{exr:viii30-ped-06}
Explain why homotopic clutching maps yield isomorphic bundles under the standard sphere classification setup.
\end{exercise}
\begin{hint}
Extend the homotopy over the collar of the equator.
\end{hint}
\begin{solution}
A homotopy of transition functions gives a bundle over $S^n\times I$ restricting to the two clutched bundles at the ends; standard bundle homotopy invariance identifies them.
\end{solution}

\begin{exercise}[Trivial bundle criterion]\label{exr:viii30-ped-07}
Show a null-homotopic clutching map produces a trivial bundle over $S^n$.
\end{exercise}
\begin{hint}
Deform the transition map to the identity.
\end{hint}
\begin{solution}
After the clutching map is deformed to the constant identity transition, the two trivial hemisphere bundles glue to the global product bundle.
\end{solution}

\begin{exercise}[Frame and triviality]\label{exr:viii30-ped-08}
Prove $r$ pointwise linearly independent global sections trivialize a rank-$r$ bundle.
\end{exercise}
\begin{hint}
Use the sections as a basis in every fiber.
\end{hint}
\begin{solution}
The map $(x,a_1,\dots,a_r)\mapsto\sum a_is_i(x)$ is a continuous fiberwise linear isomorphism from the trivial bundle to the given bundle.
\end{solution}

\begin{exercise}[Pullback bundle]\label{exr:viii30-ped-09}
Explain why pulling a bundle back along $f:X\to B$ again gives a vector bundle.
\end{exercise}
\begin{hint}
Pull back each local trivialization.
\end{hint}
\begin{solution}
Over $f^{-1}(U)$, the pullback is identified with $f^{-1}(U)\times\mathbb F^r$, with transition functions obtained by composing those of the original bundle with $f$.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Local trivial means global trivial]\label{exr:viii30-ped-10}
Test: every vector bundle is globally trivial because it is locally trivial.
\end{exercise}
\begin{hint}
The obstruction is in transition functions.
\end{hint}
\begin{solution}
False. Nontrivial gluing, such as the Mobius bundle, prevents a global product trivialization.
\end{solution}

\begin{exercise}[Nowhere-zero section trivializes every rank]\label{exr:viii30-ped-11}
Test: one nowhere-zero section trivializes any vector bundle.
\end{exercise}
\begin{hint}
One section supplies only one frame vector.
\end{hint}
\begin{solution}
False for rank $r>1$; it splits off a trivial line under suitable hypotheses but does not provide a full frame.
\end{solution}

\begin{exercise}[All clutching maps equivalent]\label{exr:viii30-ped-12}
Test: any two clutching maps with the same domain and target define isomorphic bundles.
\end{exercise}
\begin{hint}
Homotopy class matters.
\end{hint}
\begin{solution}
False. Distinct homotopy classes can classify nonisomorphic bundles.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Gauge transition check]\label{exr:viii30-ped-13}
How can transition functions be used to verify a computational atlas for a bundle?
\end{exercise}
\begin{hint}
Check cocycle identities on overlaps.
\end{hint}
\begin{solution}
Numerically or symbolically one checks $g_{ij}g_{jk}=g_{ik}$ on triple overlaps and invertibility on pairwise overlaps; failures reveal inconsistent local frames.
\end{solution}

\begin{exercise}[Hairy-ball preview]\label{exr:viii30-ped-14}
Why would a nowhere-zero section of $TS^2$ be topologically significant?
\end{exercise}
\begin{hint}
It would split off a trivial line subbundle.
\end{hint}
\begin{solution}
Such a section would contradict the nonzero Euler class of $TS^2$; this is the bundle-theoretic form of the hairy-ball theorem.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Degree-k clutching]\label{exr:viii30-ped-15}
For $g_k(z)=z^k$ and $g_\ell(z)=z^\ell$ from $S^1$ to $U(1)$, when are they homotopic?
\end{exercise}
\begin{hint}
Compare winding numbers.
\end{hint}
\begin{solution}
They are homotopic exactly when $k=\ell$.
\end{solution}

\begin{exercise}[Bundle synthesis]\label{exr:viii30-ped-16}
Describe the workflow from local trivializations to global bundle topology.
\end{exercise}
\begin{hint}
Extract transitions, impose cocycle conditions, and study their homotopy class.
\end{hint}
\begin{solution}
Local frames determine transition functions; compatibility makes them a cocycle; changes of frame modify the cocycle by gauge transformations; the residual homotopy/cohomology class records the global obstruction to triviality.
\end{solution}
