\chapter{Vector Bundles and Clutching}
\label{ch:viii30-vector-bundles-clutching}

A vector bundle is a continuously varying family of vector spaces over a base
space.  Local triviality makes bundles look like products in small
neighborhoods, while transition functions record the global twisting.
Clutching constructs bundles over spheres by gluing trivial bundles across the
equator.

\section{Vector bundles}

\begin{definition}[Real vector bundle]
\label{def:viii30-vector-bundle}
A rank-\(k\) real vector bundle over \(X\) is a map
\[
\pi:E\to X
\]
such that each fiber
\[
E_x=\pi^{-1}(x)
\]
is a \(k\)-dimensional real vector space and every \(x\in X\) has a
neighborhood \(U\) with a homeomorphism
\[
\Phi:\pi^{-1}(U)\to U\times\mathbb R^k
\]
that is linear on fibers.
\end{definition}

Complex vector bundles are defined similarly with \(\mathbb C^k\).

\section{Local trivializations}

A local trivialization identifies nearby fibers with one fixed model vector
space.  The bundle can be globally nontrivial even though every small piece is
a product.

\section{Transition functions}

If \(\Phi_i,\Phi_j\) are trivializations over \(U_i,U_j\), then on the overlap
\[
U_i\cap U_j
\]
their change of coordinates has the form
\[
(x,v)\longmapsto(x,g_{ij}(x)v)
\]
with
\[
g_{ij}:U_i\cap U_j\to GL_k(\mathbb R).
\]

They satisfy
\[
g_{ii}=I,\qquad
g_{ij}g_{jk}=g_{ik}.
\]

\section{Trivial bundles}

The product
\[
X\times\mathbb R^k\to X
\]
is the trivial rank-\(k\) real bundle.

A rank-\(k\) bundle is trivial if it is isomorphic to this product bundle.

\section{Sections}

\begin{definition}[Section]
\label{def:viii30-section}
A section is a map
\[
s:X\to E
\]
with
\[
\pi\circ s=\operatorname{id}_X.
\]
\end{definition}

The zero vector in every fiber defines the zero section.

A nowhere-zero section is one satisfying
\[
s(x)\ne0
\]
for every \(x\).

\section{Bundle maps}

A bundle map over \(f:X\to Y\) is a map of total spaces carrying fibers
linearly into fibers and covering \(f\).

An isomorphism of bundles over the same base is fiberwise linear and a
homeomorphism on total spaces.

\section{Pullback bundles}

Given
\[
f:Y\to X
\]
and a vector bundle \(E\to X\), define
\[
f^*E
=
\{(y,e)\in Y\times E:f(y)=\pi(e)\}.
\]

\begin{theorem}[Pullback]
\label{thm:viii30-pullback}
The projection
\[
f^*E\to Y
\]
is a vector bundle of the same rank.
\end{theorem}

\section{Whitney sum}

For bundles \(E,F\to X\), define
\[
(E\oplus F)_x=E_x\oplus F_x.
\]

This is the Whitney sum bundle
\[
E\oplus F\to X.
\]

\section{Subbundles and quotient bundles}

A vector subbundle \(F\subset E\) has fiberwise vector subspaces of constant
rank with local triviality.  Under suitable hypotheses one can form the
quotient bundle
\[
E/F.
\]

\section{Line bundles}

Rank-one bundles are line bundles.

Real line bundles have transition functions in
\[
GL_1(\mathbb R)=\mathbb R^\times.
\]
After choosing a metric, the structure group reduces to
\[
O(1)=\{\pm1\}.
\]

\section{The Möbius line bundle}

The Möbius strip is the total space of a nontrivial real line bundle over
\(S^1\).  It can be constructed from
\[
[0,1]\times\mathbb R
\]
by the identification
\[
(0,v)\sim(1,-v).
\]

The sign change is the twisting transition function.

\section{No nowhere-zero section criterion for line bundles}

A real line bundle is trivial if and only if it admits a nowhere-zero section.

Hence the Möbius line bundle has no nowhere-zero global section.

\section{Tangent bundles}

For a smooth \(n\)-manifold \(M\),
\[
TM=\coprod_{x\in M}T_xM
\]
is a rank-\(n\) vector bundle.

The tangent bundle of \(S^1\) is trivial.

The tangent bundle of \(S^2\) is nontrivial; the hairy-ball theorem is a
manifestation of the absence of a nowhere-zero section.

\section{Bundle metrics}

Every vector bundle over a paracompact base admits a fiberwise inner product.
This allows the structure group of a real rank-\(k\) bundle to be reduced from
\(GL_k(\mathbb R)\) to \(O(k)\).

For complex bundles one similarly reduces to \(U(k)\).

\section{Clutching over a sphere}

Write
\[
S^n=D^n_+\cup D^n_-,
\qquad
D^n_+\cap D^n_-\simeq S^{n-1}.
\]

A bundle is trivial over each hemisphere because each hemisphere is
contractible.  The global bundle is therefore determined by how the two
trivial bundles are glued along the equator.

\begin{definition}[Clutching map]
\label{def:viii30-clutching}
A clutching map for a rank-\(k\) real bundle over \(S^n\) is a map
\[
g:S^{n-1}\to GL_k(\mathbb R)
\]
used to identify
\[
(x,v)_+\sim(x,g(x)v)_-
\]
on the equator.
\end{definition}

\section{Homotopic clutching maps}

\begin{theorem}[Clutching homotopy invariance]
\label{thm:viii30-clutching-homotopy}
Homotopic clutching maps determine isomorphic vector bundles over \(S^n\).
\end{theorem}

After reducing structure group, one can work with
\[
g:S^{n-1}\to O(k)
\]
for real bundles or
\[
g:S^{n-1}\to U(k)
\]
for complex bundles.

\section{Classification by clutching}

Under the standard rank/base hypotheses, isomorphism classes of rank-\(k\)
bundles over \(S^n\) are classified by homotopy classes
\[
[S^{n-1},GL_k(\mathbb R)]
\]
or equivalently, after metric reduction,
\[
[S^{n-1},O(k)].
\]

For complex bundles use \(U(k)\).

\section{Real line bundles over the circle}

A rank-one real bundle over \(S^1\) is clutched along
\[
S^0=\{-1,+1\}.
\]
Up to isomorphism there are two possibilities:
\begin{itemize}
\item the trivial cylinder bundle;
\item the Möbius bundle.
\end{itemize}

\section{Complex line bundles over \(S^2\)}

A complex line bundle over \(S^2\) is clutched by
\[
g:S^1\to U(1)\cong S^1.
\]
Its homotopy class is classified by an integer winding number.

This integer will later be identified with the first Chern class after the
appropriate characteristic-class machinery is introduced.

\section{Tangent bundle clutching}

The tangent bundle of a sphere can be described by a clutching map on the
equator.  The nontriviality of this clutching map encodes the global twisting
of the tangent bundle.

\section{Stable viewpoint}

Adding a trivial bundle can simplify classification:
\[
E\sim_{\rm st}F
\quad\Longleftrightarrow\quad
E\oplus\varepsilon^r\cong F\oplus\varepsilon^s
\]
for suitable trivial summands.

This is the entry point to \(K\)-theory, beyond the scope of this volume.

\section{SVG visual laboratory}

This chapter ships with editable SVG assets for:
\begin{itemize}
\item local trivializations and transition functions, including the Möbius
twist;
\item hemisphere clutching over \(S^n\), with the equatorial map
\(S^{n-1}\to O(k)\).
\end{itemize}

\section{Bridge to Thom classes}

A vector bundle has a total space, a zero section, and a disk/sphere bundle.
Cohomology of these pairs produces the Thom class.  VIII/31 develops that
construction and the Thom isomorphism.

\section{Solved problem dossiers}

\begin{problem}[Transition cocycle]\label{prob:viii30-01}
Derive \(g_{ij}g_{jk}=g_{ik}\).
\end{problem}
\begin{solution}
Changing coordinates from chart \(k\) to \(j\) and then \(j\) to \(i\) must equal the direct change from \(k\) to \(i\).
\end{solution}

\begin{problem}[Pullback]\label{prob:viii30-02}
Describe the fiber of \(f^*E\to Y\) over \(y\).
\end{problem}
\begin{solution}
It is naturally the fiber \(E_{f(y)}\), so the pullback has the same rank.
\end{solution}

\begin{problem}[Whitney rank]\label{prob:viii30-03}
If \(E\) has rank \(r\) and \(F\) rank \(s\), what is \(\operatorname{rank}(E\oplus F)\)?
\end{problem}
\begin{solution}
\(r+s\).
\end{solution}

\begin{problem}[Mobius gluing]\label{prob:viii30-04}
Why is \((0,v)\sim(1,-v)\) a nontrivial line-bundle gluing?
\end{problem}
\begin{solution}
Going once around the base reverses the fiber coordinate, so no consistent nonzero orientation/vector can return to itself globally.
\end{solution}

\begin{problem}[Mobius section]\label{prob:viii30-05}
Why has the Möbius line bundle no nowhere-zero section?
\end{problem}
\begin{solution}
A nowhere-zero section would trivialize a real line bundle, contradicting the nontrivial sign monodromy.
\end{solution}

\begin{problem}[Cylinder]\label{prob:viii30-06}
What clutching produces the trivial real line bundle over \(S^1\)?
\end{problem}
\begin{solution}
The constant \(+1\) gluing gives the cylinder.
\end{solution}

\begin{problem}[Two line bundles]\label{prob:viii30-07}
Why are there exactly two real line bundles over \(S^1\) up to isomorphism?
\end{problem}
\begin{solution}
After metric reduction, the clutching data take values in \(O(1)=\{\pm1\}\), yielding trivial and sign-twisted classes.
\end{solution}

\begin{problem}[Hemisphere triviality]\label{prob:viii30-08}
Why is a bundle over each hemisphere of \(S^n\) trivial?
\end{problem}
\begin{solution}
Each hemisphere is contractible; vector bundles over a contractible paracompact base are trivial.
\end{solution}

\begin{problem}[Clutching homotopy]\label{prob:viii30-09}
Why do homotopic clutching maps give isomorphic bundles?
\end{problem}
\begin{solution}
A homotopy of transition data extends across a collar of the equator and gives equivalent gluing data.
\end{solution}

\begin{problem}[Complex line S2]\label{prob:viii30-10}
Classify complex line bundles over \(S^2\) by clutching.
\end{problem}
\begin{solution}
Clutching maps are \(S^1\to U(1)\cong S^1\), classified by degree/winding number \(\mathbb Z\).
\end{solution}

\begin{problem}[Winding zero]\label{prob:viii30-11}
What complex line bundle corresponds to clutching degree zero?
\end{problem}
\begin{solution}
The clutching map is null-homotopic, so the bundle is trivial.
\end{solution}

\begin{problem}[Bundle metric]\label{prob:viii30-12}
Why does a metric reduce the structure group?
\end{problem}
\begin{solution}
Choose orthonormal frames in each trivialization; changes between orthonormal frames lie in \(O(k)\) or \(U(k)\).
\end{solution}

\begin{problem}[Tangent circle]\label{prob:viii30-13}
Construct a nowhere-zero section of \(TS^1\).
\end{problem}
\begin{solution}
The unit tangent vector field \(s(e^{it})=ie^{it}\) is continuous and nowhere zero, trivializing the line bundle.
\end{solution}

\begin{problem}[Hairy sphere]\label{prob:viii30-14}
What bundle fact is expressed by the hairy-ball theorem on \(S^2\)?
\end{problem}
\begin{solution}
The tangent bundle \(TS^2\) has no nowhere-zero section and is therefore nontrivial.
\end{solution}

\begin{problem}[Bundle isomorphism]\label{prob:viii30-15}
What must a bundle isomorphism preserve?
\end{problem}
\begin{solution}
It covers the identity on the base and is a linear isomorphism on every fiber, with a homeomorphic total-space map.
\end{solution}

\begin{problem}[Stable equivalence]\label{prob:viii30-16}
What does adding a trivial summand do conceptually?
\end{problem}
\begin{solution}
It can erase unstable distinctions while retaining stable bundle information, leading toward \(K\)-theory.
\end{solution}

\begin{problem}[Clutching tangent sphere]\label{prob:viii30-17}
Why can the tangent bundle of \(S^n\) be encoded on \(S^{n-1}\)?
\end{problem}
\begin{solution}
Trivialize over northern and southern hemispheres; their mismatch along the equator is the clutching map.
\end{solution}

\begin{problem}[SVG transition]\label{prob:viii30-18}
What should the transition-function SVG show?
\end{problem}
\begin{solution}
Two overlapping base charts, product-like fibers above each, and a matrix \(g_{ij}(x)\) converting fiber coordinates on the overlap.
\end{solution}

\begin{problem}[SVG clutching]\label{prob:viii30-19}
What should the clutching SVG show?
\end{problem}
\begin{solution}
Two trivial hemisphere bundles glued along the equator by \(g:S^{n-1}\to O(k)\), making the global twist visually explicit.
\end{solution}

\begin{problem}[Bridge]\label{prob:viii30-20}
Why does the zero section matter for Thom theory?
\end{problem}
\begin{solution}
The Thom class lives in relative cohomology of the bundle total space relative to the complement of the zero section, or equivalently the disk/sphere bundle pair.
\end{solution}

\begin{problem}[Corpus ownership]\label{prob:viii30-21}
Why does this chapter own both vector-bundle and clutching legacy streams?
\end{problem}
\begin{solution}
The migration map sends both explicit sections and all inherited theorem/problem descendants in topology-14 to VIII/30.
\end{solution}

\begin{problem}[Final principle]\label{prob:viii30-22}
What does clutching convert a global bundle problem into?
\end{problem}
\begin{solution}
A homotopy-class problem for a map from the equator into the structure group.
\end{solution}
