\section{Supplementary worked examples}
\label{sec:viii34-pedagogy}

\begin{example}[The intersection form of CP2]\label{ex:viii34-ped-01}
For the complex projective plane, $H_2(\mathbb{CP}^2;\mathbb Z)\cong\mathbb Z$ is generated by the class of a projective line. Two generic projective lines meet once positively, so the intersection form is the $1\times1$ matrix
\[
[1].
\]
\end{example}

\begin{example}[The hyperbolic form on S2 x S2]\label{ex:viii34-ped-02}
Let $a=[S^2\times\{*\}]$ and $b=[\{*\}\times S^2]$. Self-intersections are zero while $a\cdot b=1$, giving
\[
Q_{S^2\times S^2}=
\begin{pmatrix}
0&1\\
1&0
\end{pmatrix}.
\]
\end{example}

\begin{example}[Orientation reversal]\label{ex:viii34-ped-03}
Reversing the orientation of a closed oriented $4$-manifold changes every oriented intersection number by a sign. Hence
\[
Q_{-M}=-Q_M,
\]
so the signature changes from $\sigma(M)$ to $-\sigma(M)$.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Definition in dimension four]\label{exr:viii34-ped-01}
Define the intersection form of a closed oriented $4$-manifold on the free part of $H_2(M;\mathbb Z)$.
\end{exercise}
\begin{hint}
Intersect representative surfaces or use cup product.
\end{hint}
\begin{solution}
$Q_M(x,y)=\langle \operatorname{PD}(x)\smile\operatorname{PD}(y),[M]\rangle$.
\end{solution}

\begin{exercise}[CP2 matrix]\label{exr:viii34-ped-02}
Write the intersection matrix of $\mathbb{CP}^2$ in the projective-line basis.
\end{exercise}
\begin{hint}
A line has self-intersection $+1$.
\end{hint}
\begin{solution}
The matrix is $[1]$.
\end{solution}

\begin{exercise}[S2xS2 matrix]\label{exr:viii34-ped-03}
Write the matrix in the product-sphere basis.
\end{exercise}
\begin{hint}
Each factor has zero self-intersection and the factors meet once.
\end{hint}
\begin{solution}
$\begin{pmatrix}0&1\\1&0\end{pmatrix}$.
\end{solution}

\begin{exercise}[Signature of CP2]\label{exr:viii34-ped-04}
Compute $\sigma(\mathbb{CP}^2)$.
\end{exercise}
\begin{hint}
Count positive minus negative eigenvalues of $[1]$.
\end{hint}
\begin{solution}
$\sigma=1$.
\end{solution}

\begin{exercise}[Signature of S2xS2]\label{exr:viii34-ped-05}
Compute the signature of the hyperbolic form.
\end{exercise}
\begin{hint}
Its eigenvalues are $1$ and $-1$.
\end{hint}
\begin{solution}
$\sigma=0$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Symmetry]\label{exr:viii34-ped-06}
Prove the intersection form on a closed oriented $4$-manifold is symmetric.
\end{exercise}
\begin{hint}
Degree-two cup products commute because $(-1)^{2\cdot2}=1$.
\end{hint}
\begin{solution}
Graded commutativity gives $\alpha\smile\beta=\beta\smile\alpha$ in degree two, so $Q(x,y)=Q(y,x)$.
\end{solution}

\begin{exercise}[Unimodularity]\label{exr:viii34-ped-07}
Explain why the intersection form on the free part is unimodular.
\end{exercise}
\begin{hint}
Use Poincare duality modulo torsion.
\end{hint}
\begin{solution}
Poincare duality identifies the free lattice $H_2/\mathrm{tors}$ with its integral dual, so the pairing has determinant $\pm1$ in any integral basis.
\end{solution}

\begin{exercise}[Basis change]\label{exr:viii34-ped-08}
Show changing an integral basis by $A\in GL_n(\mathbb Z)$ transforms the matrix by $Q\mapsto A^TQA$.
\end{exercise}
\begin{hint}
Expand the bilinear pairing on new basis vectors.
\end{hint}
\begin{solution}
The coordinates of both arguments transform by $A$, yielding the congruence transformation $A^TQA$.
\end{solution}

\begin{exercise}[Orientation reversal]\label{exr:viii34-ped-09}
Prove $Q_{-M}=-Q_M$.
\end{exercise}
\begin{hint}
The fundamental class changes sign.
\end{hint}
\begin{solution}
Evaluation on $[-M]=-[M]$ multiplies every pairing value by $-1$.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Form determined by Betti number]\label{exr:viii34-ped-10}
Test: $b_2(M)$ determines the intersection form.
\end{exercise}
\begin{hint}
Many noncongruent forms have the same rank.
\end{hint}
\begin{solution}
False. Rank alone does not determine parity, signature, or lattice structure.
\end{solution}

\begin{exercise}[Intersection matrix basis invariant entrywise]\label{exr:viii34-ped-11}
Test: the matrix entries are independent of basis.
\end{exercise}
\begin{hint}
Only the equivalence class of the bilinear form is invariant.
\end{hint}
\begin{solution}
False. Matrices change by integral congruence under basis changes.
\end{solution}

\begin{exercise}[Self-intersection always zero]\label{exr:viii34-ped-12}
Test: every oriented surface class in a $4$-manifold has zero self-intersection.
\end{exercise}
\begin{hint}
Consider a projective line in $\mathbb{CP}^2$.
\end{hint}
\begin{solution}
False; the generator of $H_2(\mathbb{CP}^2)$ has self-intersection $+1$.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Geometric collision matrix]\label{exr:viii34-ped-13}
How can a basis of embedded surfaces be used to construct an intersection matrix computationally?
\end{exercise}
\begin{hint}
Perturb to transverse position and count signed intersections.
\end{hint}
\begin{solution}
Each matrix entry is obtained by making two representatives transverse and summing local signs; diagonal entries use a push-off determined by the normal bundle.
\end{solution}

\begin{exercise}[Classification diagnostic]\label{exr:viii34-ped-14}
Why is signature more informative than $b_2$ alone?
\end{exercise}
\begin{hint}
It remembers the positive/negative decomposition of the form.
\end{hint}
\begin{solution}
Two manifolds with the same $b_2$ can have different signatures, so the signature distinguishes intersection lattices that rank cannot.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Connected sum form]\label{exr:viii34-ped-15}
Under standard orientations, what happens to intersection forms under connected sum of closed oriented $4$-manifolds?
\end{exercise}
\begin{hint}
Middle homology splits.
\end{hint}
\begin{solution}
$Q_{M\#N}\cong Q_M\oplus Q_N$.
\end{solution}

\begin{exercise}[Intersection synthesis]\label{exr:viii34-ped-16}
Explain how cup product, Poincare duality, and embedded representatives produce the same intersection form.
\end{exercise}
\begin{hint}
Translate homology classes to degree-two cohomology and evaluate.
\end{hint}
\begin{solution}
Poincare duality turns surfaces into degree-two classes; their cup product evaluated on $[M]$ equals the signed geometric intersection count, linking the algebraic matrix directly to geometry.
\end{solution}
