\chapter{The Universal Coefficient Theorem}
\label{ch:viii26-universal-coefficient}

The universal coefficient theorem converts integral homology into homology with
arbitrary coefficients.  The price for tensoring is a Tor correction term.

\section{Statement for homology}
\begin{theorem}[Universal coefficient theorem for homology]
\label{thm:viii26-uct}
For a space \(X\) and abelian group \(G\), there is a natural short exact
sequence
\[
0\to H_n(X;\mathbb Z)\otimes G
\longrightarrow H_n(X;G)
\longrightarrow
\operatorname{Tor}\bigl(H_{n-1}(X;\mathbb Z),G\bigr)
\to0.
\]
\end{theorem}

The sequence splits, but not naturally in general.

\section{Meaning}
The first term is the expected coefficient change from degree \(n\).  The Tor
term imports torsion information from degree \(n-1\).

\section{Free homology}
If \(H_{n-1}(X;\mathbb Z)\) is free abelian, then
\[
\operatorname{Tor}(H_{n-1}(X),G)=0,
\]
and therefore
\[
H_n(X;G)\cong H_n(X;\mathbb Z)\otimes G.
\]

\section{Field coefficients}
For \(G=\mathbb F_p\),
\[
A\otimes\mathbb F_p\cong A/pA,
\]
while
\[
\operatorname{Tor}(A,\mathbb F_p)
\]
detects \(p\)-torsion in \(A\).

\section{Cyclic formulas}
For positive integers \(m,n\),
\[
\mathbb Z/m\otimes\mathbb Z/n
\cong
\mathbb Z/\gcd(m,n),
\]
and
\[
\operatorname{Tor}(\mathbb Z/m,\mathbb Z/n)
\cong
\mathbb Z/\gcd(m,n).
\]

\section{Free summands}
\[
\operatorname{Tor}(\mathbb Z,G)=0,
\qquad
\mathbb Z\otimes G\cong G.
\]

\section{Finite direct sums}
Tensor and Tor distribute over finite direct sums, so finitely generated
integral homology can be handled one cyclic summand at a time.

\section{Projective plane}
For \(\mathbb RP^2\),
\[
H_0=\mathbb Z,\quad H_1=\mathbb Z/2,\quad H_2=0.
\]
With \(G=\mathbb F_2\), UCT gives
\[
H_1(\mathbb RP^2;\mathbb F_2)\cong\mathbb F_2
\]
from the tensor term and
\[
H_2(\mathbb RP^2;\mathbb F_2)\cong\mathbb F_2
\]
from the Tor term.

\section{Why a top mod-2 class appears}
The integral top group \(H_2(\mathbb RP^2;\mathbb Z)\) vanishes, yet mod-2
\(H_2\) is nonzero.  UCT explains this as
\[
\operatorname{Tor}(H_1(\mathbb RP^2),\mathbb F_2)
\cong\mathbb F_2.
\]

\section{Moore spaces}
For a Moore space with
\[
\widetilde H_n(M;\mathbb Z)=\mathbb Z/m,
\]
UCT gives coefficient homology in degrees \(n\) and \(n+1\):
\[
\widetilde H_n(M;G)\cong (\mathbb Z/m)\otimes G,
\]
\[
\widetilde H_{n+1}(M;G)
\cong
\operatorname{Tor}(\mathbb Z/m,G).
\]

\section{Rational coefficients}
Since \(\mathbb Q\) is torsion-free and divisible, for finitely generated
homology the Tor contribution vanishes and finite torsion tensors to zero.
Thus
\[
H_n(X;\mathbb Q)
\cong
H_n(X;\mathbb Z)_{\rm free}\otimes\mathbb Q.
\]

\section{Proof idea}
Singular chain groups are free abelian.  One analyzes
\[
0\to Z_n\to C_n\to B_{n-1}\to0
\]
and the quotient
\[
0\to B_n\to Z_n\to H_n\to0,
\]
then tensors with \(G\).  The possible failure of exactness is precisely
encoded by Tor.

\section{Free resolutions}
For an abelian group \(A\), choose a free resolution
\[
0\to F_1\to F_0\to A\to0.
\]
Then
\[
\operatorname{Tor}(A,G)
=
\ker(F_1\otimes G\to F_0\otimes G).
\]

\section{Cyclic resolution}
For \(A=\mathbb Z/m\), use
\[
0\to\mathbb Z\xrightarrow{\times m}\mathbb Z\to\mathbb Z/m\to0.
\]
Tensoring with \(G\) shows
\[
\operatorname{Tor}(\mathbb Z/m,G)
=
\ker(\times m:G\to G).
\]

\section{Tensor cyclic quotient}
The same resolution gives
\[
(\mathbb Z/m)\otimes G\cong G/mG.
\]

Thus the UCT terms match the direct Moore-space calculation from VIII/25.

\section{Naturality}
The UCT short exact sequence is natural in both \(X\) and \(G\).

A map \(X\to Y\) induces a commutative diagram of UCT sequences, as does a
homomorphism of coefficient groups.

\section{Nonnatural splitting}
Although the UCT short exact sequence splits as a sequence of abelian groups,
there is generally no preferred splitting compatible with all maps.

Therefore one should treat
\[
H_n(X;G)
\]
as an extension unless a canonical identification is available.

\section{Dimension over a field}
For finite-type homology and \(G=\mathbb F_p\),
\[
\dim_{\mathbb F_p}H_n(X;\mathbb F_p)
\]
receives contributions from the free rank of \(H_n(X;\mathbb Z)\), from
\(p\)-torsion in \(H_n\), and from \(p\)-torsion in \(H_{n-1}\).

\section{Betti numbers}
Over \(\mathbb Q\), the dimension is exactly the integral Betti number:
\[
b_n=\dim_{\mathbb Q}H_n(X;\mathbb Q).
\]

\section{Relative UCT}
The same theorem applies to relative chain complexes:
\[
0\to H_n(X,A;\mathbb Z)\otimes G
\to H_n(X,A;G)
\to
\operatorname{Tor}(H_{n-1}(X,A;\mathbb Z),G)
\to0.
\]

\section{Reduced UCT}
Reduced homology satisfies the same formula in the expected degrees.

\section{Computational workflow}
For finitely generated integral homology:
\begin{enumerate}
\item decompose each group into free and cyclic torsion summands;
\item tensor the degree-\(n\) group with \(G\);
\item compute Tor from the degree-\((n-1)\) group;
\item assemble the resulting short exact sequence.
\end{enumerate}

\section{Bridge to Künneth}
UCT concerns one chain complex with changed coefficients.  The Künneth theorem
concerns tensor products of two chain complexes.  Again, a tensor term gives the
main answer and Tor provides the correction.

\section{Solved problem dossiers}

\begin{problem}[RP2 H2 mod2]\label{prob:viii26-01}
Use UCT to compute \(H_2(\mathbb RP^2;\mathbb F_2)\).
\end{problem}
\begin{solution}
The tensor term is zero because integral \(H_2=0\). The Tor term is \(\operatorname{Tor}(\mathbb Z/2,\mathbb F_2)\cong\mathbb F_2\).
\end{solution}

\begin{problem}[RP2 H1 mod2]\label{prob:viii26-02}
Compute \(H_1(\mathbb RP^2;\mathbb F_2)\).
\end{problem}
\begin{solution}
\((\mathbb Z/2)\otimes\mathbb F_2\cong\mathbb F_2\), while Tor from \(H_0=\mathbb Z\) vanishes.
\end{solution}

\begin{problem}[RP2 mod3]\label{prob:viii26-03}
Compute positive homology with \(\mathbb F_3\).
\end{problem}
\begin{solution}
Both tensor and Tor of \(\mathbb Z/2\) with \(\mathbb F_3\) vanish, so positive homology is zero.
\end{solution}

\begin{problem}[Moore mod p]\label{prob:viii26-04}
Apply UCT to \(M(\mathbb Z/m,n)\) with \(\mathbb F_p\).
\end{problem}
\begin{solution}
If \(p\mid m\), tensor gives \(\mathbb F_p\) in degree \(n\) and Tor gives \(\mathbb F_p\) in degree \(n+1\); otherwise both vanish.
\end{solution}

\begin{problem}[Cyclic G]\label{prob:viii26-05}
Compute \((\mathbb Z/12)\otimes\mathbb Z/18\) and Tor.
\end{problem}
\begin{solution}
Both are \(\mathbb Z/6\).
\end{solution}

\begin{problem}[Free plus torsion]\label{prob:viii26-06}
Let \(H_n=\mathbb Z^2\oplus\mathbb Z/6\). Tensor with \(\mathbb F_2\).
\end{problem}
\begin{solution}
The free part gives \(\mathbb F_2^2\) and the torsion summand gives \(\mathbb F_2\), so the tensor term is \(\mathbb F_2^3\).
\end{solution}

\begin{problem}[Adjacent Tor]\label{prob:viii26-07}
If \(H_{n-1}=\mathbb Z/4\), what Tor contribution appears with \(\mathbb F_2\)?
\end{problem}
\begin{solution}
\(\operatorname{Tor}(\mathbb Z/4,\mathbb F_2)\cong\mathbb F_2\).
\end{solution}

\begin{problem}[Rationalization]\label{prob:viii26-08}
Let \(H_n=\mathbb Z^3\oplus\mathbb Z/12\). Compute \(H_n(-;\mathbb Q)\) assuming adjacent Tor vanishes.
\end{problem}
\begin{solution}
\(\mathbb Q^3\).
\end{solution}

\begin{problem}[Cyclic resolution]\label{prob:viii26-09}
Use the standard resolution to compute \(\operatorname{Tor}(\mathbb Z/m,G)\).
\end{problem}
\begin{solution}
Tensor \(0\to\mathbb Z\xrightarrow m\mathbb Z\to\mathbb Z/m\to0\); Tor is the kernel of multiplication by \(m\) on \(G\).
\end{solution}

\begin{problem}[Tensor quotient]\label{prob:viii26-10}
Use the same resolution to compute \(\mathbb Z/m\otimes G\).
\end{problem}
\begin{solution}
It is the cokernel of multiplication by \(m\), namely \(G/mG\).
\end{solution}

\begin{problem}[Natural map]\label{prob:viii26-11}
What happens to UCT under \(f:X\to Y\)?
\end{problem}
\begin{solution}
Tensor, coefficient homology, and Tor maps form a commutative diagram of short exact UCT sequences.
\end{solution}

\begin{problem}[Nonnatural split]\label{prob:viii26-12}
Why should one not write a canonical direct sum in UCT?
\end{problem}
\begin{solution}
A splitting exists as groups, but no choice is generally compatible with all maps; only the short exact sequence is canonical.
\end{solution}

\begin{problem}[Relative example]\label{prob:viii26-13}
Apply UCT to \(H_n(D^n,S^{n-1};G)\).
\end{problem}
\begin{solution}
Integral relative homology is \(\mathbb Z\) in degree \(n\), so coefficient relative homology is \(G\) there and zero elsewhere.
\end{solution}

\begin{problem}[Sphere]\label{prob:viii26-14}
Use UCT to compute \(H_n(S^n;G)\).
\end{problem}
\begin{solution}
Integral top homology is free \(\mathbb Z\) and adjacent lower homology is zero for \(n>1\), so top homology is \(G\).
\end{solution}

\begin{problem}[Circle]\label{prob:viii26-15}
Use UCT to compute \(H_1(S^1;G)\).
\end{problem}
\begin{solution}
\(\mathbb Z\otimes G\cong G\), and Tor from \(H_0=\mathbb Z\) vanishes.
\end{solution}

\begin{problem}[Betti]\label{prob:viii26-16}
Why does rational homology recover Betti numbers?
\end{problem}
\begin{solution}
Finite torsion disappears and Tor vanishes, leaving one \(\mathbb Q\) for each free integral generator.
\end{solution}

\begin{problem}[Mod p dimension]\label{prob:viii26-17}
Why can \(\dim H_n(X;\mathbb F_p)\) exceed \(b_n\)?
\end{problem}
\begin{solution}
Besides free rank, \(p\)-torsion in \(H_n\) contributes through tensor and \(p\)-torsion in \(H_{n-1}\) contributes through Tor.
\end{solution}

\begin{problem}[UCT workflow]\label{prob:viii26-18}
Given finitely generated integral homology, what is the first computational step?
\end{problem}
\begin{solution}
Decompose each group into free and cyclic torsion summands, then apply tensor and Tor summandwise.
\end{solution}

\begin{problem}[Extension issue]\label{prob:viii26-19}
What information can remain after tensor and Tor are known?
\end{problem}
\begin{solution}
One must still know how the two terms fit into the short exact sequence; the theorem says it splits noncanonically for homology over \(\mathbb Z\).
\end{solution}

\begin{problem}[Coefficient consistency]\label{prob:viii26-20}
How does UCT confirm direct cellular mod-\(p\) calculations?
\end{problem}
\begin{solution}
Both methods produce the same coefficient homology; UCT explains which classes come from tensor and which are new Tor classes.
\end{solution}

\begin{problem}[Bridge]\label{prob:viii26-21}
How does Künneth resemble UCT?
\end{problem}
\begin{solution}
Both have a primary tensor expression and a Tor correction measuring failure of exactness.
\end{solution}

\begin{problem}[Final principle]\label{prob:viii26-22}
What does UCT save computationally?
\end{problem}
\begin{solution}
It avoids rebuilding the entire coefficient chain complex once integral homology is known.
\end{solution}
