\section{Supplementary worked examples}
\label{sec:viii26-pedagogy}

\begin{example}[UCT for homology with a cyclic coefficient group]\label{ex:viii26-ped-01}
If $H_n(X;\mathbb Z)$ and $H_{n-1}(X;\mathbb Z)$ are known, then
\[
0\to H_n(X)\otimes G\to H_n(X;G)\to
\operatorname{Tor}(H_{n-1}(X),G)\to0.
\]
For $G=\mathbb Z/m$, both terms reduce to explicit gcd computations on cyclic summands.
\end{example}

\begin{example}[UCT over a field]\label{ex:viii26-ped-02}
For a field $F$, the homological UCT still records torsion through Tor when $F=\mathbb F_p$, whereas with $F=\mathbb Q$ all torsion vanishes and
\[
H_n(X;\mathbb Q)\cong H_n(X;\mathbb Z)\otimes\mathbb Q.
\]
\end{example}

\begin{example}[Noncanonical splitting]\label{ex:viii26-ped-03}
The UCT short exact sequence splits as a sequence of abelian groups, but generally not naturally. Thus one may identify the isomorphism type of $H_n(X;G)$ without obtaining a preferred decomposition into tensor and Tor summands.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Free input]\label{exr:viii26-ped-01}
If $H_{n-1}(X)$ is free, what happens to the Tor term in homological UCT?
\end{exercise}
\begin{hint}
Tor of a free abelian group vanishes.
\end{hint}
\begin{solution}
The Tor term is zero.
\end{solution}

\begin{exercise}[Rational coefficients]\label{exr:viii26-ped-02}
What is $\operatorname{Tor}(A,\mathbb Q)$ for an abelian group $A$?
\end{exercise}
\begin{hint}
$\mathbb Q$ is flat over $\mathbb Z$.
\end{hint}
\begin{solution}
It vanishes.
\end{solution}

\begin{exercise}[Cyclic tensor]\label{exr:viii26-ped-03}
Compute $(\mathbb Z/6)\otimes\mathbb Z/4$.
\end{exercise}
\begin{hint}
Use the gcd formula.
\end{hint}
\begin{solution}
It is $\mathbb Z/2$.
\end{solution}

\begin{exercise}[Cyclic Tor]\label{exr:viii26-ped-04}
Compute $\operatorname{Tor}(\mathbb Z/6,\mathbb Z/4)$.
\end{exercise}
\begin{hint}
Use the same gcd.
\end{hint}
\begin{solution}
It is $\mathbb Z/2$.
\end{solution}

\begin{exercise}[Free rank over Q]\label{exr:viii26-ped-05}
If $H_n(X)\cong\mathbb Z^b\oplus T$, compute $H_n(X;\mathbb Q)$.
\end{exercise}
\begin{hint}
Torsion tensors to zero with $\mathbb Q$.
\end{hint}
\begin{solution}
$H_n(X;\mathbb Q)\cong\mathbb Q^b$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Derive cyclic Tor]\label{exr:viii26-ped-06}
Derive $\operatorname{Tor}(\mathbb Z/n,G)$ from $0\to\mathbb Z\xrightarrow{n}\mathbb Z\to\mathbb Z/n\to0$.
\end{exercise}
\begin{hint}
Tensor the resolution with $G$.
\end{hint}
\begin{solution}
The Tor group is the kernel of multiplication by $n$ on $G$, i.e. $G[n]$.
\end{solution}

\begin{exercise}[Naturality]\label{exr:viii26-ped-07}
Explain what naturality of the UCT means for a map $X\to Y$.
\end{exercise}
\begin{hint}
All terms and arrows should commute with induced maps.
\end{hint}
\begin{solution}
The homology maps, tensor maps, Tor maps, and coefficient homology maps form a commutative diagram of short exact sequences.
\end{solution}

\begin{exercise}[Splitting caveat]\label{exr:viii26-ped-08}
Why does existence of a splitting not give a canonical projection onto the Tor summand?
\end{exercise}
\begin{hint}
The theorem does not choose a natural splitting.
\end{hint}
\begin{solution}
Different algebraic choices produce different complements, so no functorial projection is determined in general.
\end{solution}

\begin{exercise}[Torsion shift]\label{exr:viii26-ped-09}
Prove $p$-torsion in $H_{n-1}(X)$ can contribute to $H_n(X;\mathbb F_p)$.
\end{exercise}
\begin{hint}
Inspect the Tor term.
\end{hint}
\begin{solution}
$\operatorname{Tor}(\mathbb Z/p,\mathbb F_p)\cong\mathbb F_p$, so such torsion appears one degree higher in coefficient homology.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[UCT sequence canonically splits]\label{exr:viii26-ped-10}
Test: the homological UCT has a natural splitting for all spaces and coefficients.
\end{exercise}
\begin{hint}
The splitting is generally noncanonical.
\end{hint}
\begin{solution}
False. A splitting exists for abelian groups, but not naturally in the data.
\end{solution}

\begin{exercise}[No torsion means no coefficient change]\label{exr:viii26-ped-11}
Test: if $H_n(X)$ is torsion-free, then $H_n(X;G)=H_n(X)\otimes G$.
\end{exercise}
\begin{hint}
Look one degree lower.
\end{hint}
\begin{solution}
False unless the relevant Tor term from $H_{n-1}(X)$ also vanishes.
\end{solution}

\begin{exercise}[Q determines integral homology]\label{exr:viii26-ped-12}
Test: rational homology determines integral homology.
\end{exercise}
\begin{hint}
Rationalization loses torsion.
\end{hint}
\begin{solution}
False; spaces with different torsion can have identical rational homology.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Predict mod-p ranks]\label{exr:viii26-ped-13}
How can UCT predict a jump in mod-$p$ Betti number?
\end{exercise}
\begin{hint}
Count both tensor and Tor contributions.
\end{hint}
\begin{solution}
A $p$-torsion summand in $H_n$ contributes by tensoring, while one in $H_{n-1}$ contributes through Tor, so either can increase the $\mathbb F_p$ dimension.
\end{solution}

\begin{exercise}[Cross-check computation]\label{exr:viii26-ped-14}
How can UCT validate a direct boundary-matrix computation with coefficients?
\end{exercise}
\begin{hint}
Compute the expected group independently from integral homology.
\end{hint}
\begin{solution}
Comparing the direct coefficient result with the tensor-plus-Tor prediction catches rank, torsion, or indexing errors.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Moore-space pattern]\label{exr:viii26-ped-15}
Suppose $\widetilde H_k(X)\cong\mathbb Z/m$ and other reduced integral homology vanishes. Describe where mod-$p$ homology can appear when $p\mid m$.
\end{exercise}
\begin{hint}
Use tensor in degree $k$ and Tor in degree $k+1$.
\end{hint}
\begin{solution}
An $\mathbb F_p$ class appears in degree $k$ from tensor and another in degree $k+1$ from Tor.
\end{solution}

\begin{exercise}[UCT synthesis]\label{exr:viii26-ped-16}
Explain the conceptual role of the tensor and Tor terms.
\end{exercise}
\begin{hint}
One changes existing classes; the other measures failure of tensoring to preserve exactness.
\end{hint}
\begin{solution}
The tensor term transports degree-$n$ integral classes to the new coefficients, while Tor records torsion obstructions from degree $n-1$ created by the nonexactness of tensor product.
\end{solution}
