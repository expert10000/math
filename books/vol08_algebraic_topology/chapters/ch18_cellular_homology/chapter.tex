\chapter{Cellular Homology}
\label{ch:viii18-cellular-homology}

For a CW complex, singular chains are usually far larger than necessary.
Cellular homology compresses the calculation to one generator for each cell.
The cellular boundary map is determined by how each \(n\)-cell attaches to the
\((n-1)\)-skeleton.  This makes the topology of CW constructions directly
computable from degrees of attaching maps.

\section{CW filtration and relative groups}

Let
\[
X^0\subset X^1\subset X^2\subset\cdots
\]
be the skeletal filtration of a CW complex.

The cellular chain group in degree \(n\) is built from
\[
H_n(X^n,X^{n-1}).
\]

\begin{theorem}[Relative cell decomposition]
\label{thm:viii18-relative-cells}
If \(X\) has \(c_n\) \(n\)-cells, then
\[
H_k(X^n,X^{n-1})
\cong
\begin{cases}
\mathbb Z^{c_n},&k=n,\\
0,&k\ne n.
\end{cases}
\]
\end{theorem}

Thus every oriented \(n\)-cell contributes one free generator.

\section{Cellular chain groups}

\begin{definition}[Cellular chains]
\label{def:viii18-cellular-chain}
Define
\[
C_n^{\mathrm{cell}}(X)
=
H_n(X^n,X^{n-1}).
\]
After choosing orientations of the \(n\)-cells,
\[
C_n^{\mathrm{cell}}(X)\cong\mathbb Z^{c_n}.
\]
\end{definition}

\section{Cellular boundary}

The cellular boundary is the composite
\[
d_n:
H_n(X^n,X^{n-1})
\longrightarrow
H_{n-1}(X^{n-1},X^{n-2})
\]
obtained from the connecting map in the long exact sequence of the triple
\[
(X^n,X^{n-1},X^{n-2}).
\]

\begin{theorem}[Cellular chain complex]
\label{thm:viii18-chain-complex}
\[
\cdots\to
C_n^{\mathrm{cell}}(X)
\xrightarrow{d_n}
C_{n-1}^{\mathrm{cell}}(X)
\to\cdots
\]
is a chain complex, and its homology is naturally isomorphic to singular
homology:
\[
H_n(C_*^{\mathrm{cell}}(X))
\cong
H_n(X).
\]
\end{theorem}

\section{Boundary coefficients from attaching maps}

Let \(e_\alpha^n\) be an \(n\)-cell with attaching map
\[
\varphi_\alpha:S^{n-1}\to X^{n-1}.
\]
For an \((n-1)\)-cell \(e_\beta^{n-1}\), collapse
\[
X^{n-1}\setminus e_\beta^{n-1}
\]
to a point.  The resulting quotient is a sphere
\[
X^{n-1}/(X^{n-1}\setminus e_\beta^{n-1})
\cong S^{n-1}.
\]
The composite
\[
S^{n-1}
\xrightarrow{\varphi_\alpha}
X^{n-1}
\longrightarrow
S^{n-1}
\]
has an integer degree.

\begin{theorem}[Cellular boundary formula]
\label{thm:viii18-boundary-degree}
The coefficient of \(e_\beta^{n-1}\) in
\[
d_n(e_\alpha^n)
\]
is the degree of this composite map.
\end{theorem}

This is the computational heart of cellular homology.

\section{A CW complex with no adjacent dimensions}

If there are no \((n-1)\)-cells, then
\[
C_{n-1}^{\mathrm{cell}}(X)=0
\]
and therefore
\[
d_n=0.
\]

This makes the minimal CW structure of a sphere immediate.

\section{Spheres}

For the CW structure on \(S^n\) with one \(0\)-cell and one \(n\)-cell,
\[
C_k=
\begin{cases}
\mathbb Z,&k=0,n,\\
0,&\text{otherwise}.
\end{cases}
\]
All differentials vanish, giving
\[
H_k(S^n)\cong
\begin{cases}
\mathbb Z,&k=0,n,\\
0,&\text{otherwise}.
\end{cases}
\]

\section{The torus}

The standard torus CW structure has one \(0\)-cell, two \(1\)-cells, and one
\(2\)-cell:
\[
C_2\cong\mathbb Z,
\qquad
C_1\cong\mathbb Z^2,
\qquad
C_0\cong\mathbb Z.
\]

The two-cell attaching word is
\[
aba^{-1}b^{-1}.
\]
Its total degree on each one-cell is zero, so
\[
d_2=0.
\]
Since the one-cells are loops at one vertex,
\[
d_1=0.
\]

Hence
\[
H_2(T^2)\cong\mathbb Z,\qquad
H_1(T^2)\cong\mathbb Z^2,\qquad
H_0(T^2)\cong\mathbb Z.
\]

\section{Real projective spaces}

The standard CW structure on
\[
\mathbb RP^n
\]
has one cell in each dimension \(0,\dots,n\).

\begin{theorem}[Projective cellular differential]
\label{thm:viii18-rp-differential}
With suitable orientations,
\[
d_k=
\begin{cases}
0,&k\text{ odd},\\
\times2,&k\text{ even}.
\end{cases}
\]
\end{theorem}

Thus the cellular chain complex alternates:
\[
\cdots
\xrightarrow{0}
\mathbb Z
\xrightarrow{2}
\mathbb Z
\xrightarrow{0}
\mathbb Z
\xrightarrow{2}
\mathbb Z
\xrightarrow{0}
\mathbb Z.
\]

\section{Homology of real projective space}

For \(0<k<n\),
\[
H_k(\mathbb RP^n)\cong
\begin{cases}
\mathbb Z/2,&k\text{ odd},\\
0,&k\text{ even}.
\end{cases}
\]

At the top dimension,
\[
H_n(\mathbb RP^n)\cong
\begin{cases}
\mathbb Z,&n\text{ odd},\\
0,&n\text{ even}.
\end{cases}
\]

\section{Degree-\(m\) two-cell attachments}

Consider
\[
X_m=S^1\cup_mD^2.
\]
Its cellular chain complex is
\[
0\to\mathbb Z
\xrightarrow{\times m}
\mathbb Z
\xrightarrow{0}
\mathbb Z
\to0.
\]

For \(m\ne0\),
\[
H_2(X_m)=0,
\qquad
H_1(X_m)\cong\mathbb Z/m,
\qquad
H_0(X_m)\cong\mathbb Z.
\]

For \(m=0\),
\[
X_0\simeq S^1\vee S^2
\]
and both \(H_1\) and \(H_2\) are \(\mathbb Z\).

\section{Moore spaces}

For \(n\ge1\), attach an \((n+1)\)-cell to \(S^n\) by a degree-\(m\) map:
\[
M(\mathbb Z/m,n)
=
S^n\cup_mD^{n+1}.
\]

The relevant cellular boundary is
\[
\mathbb Z
\xrightarrow{\times m}
\mathbb Z.
\]

Hence, for \(m\ne0\),
\[
\widetilde H_k(M(\mathbb Z/m,n))
\cong
\begin{cases}
\mathbb Z/m,&k=n,\\
0,&k\ne n.
\end{cases}
\]

\section{Lens-space pattern}

A lens space admits a CW structure with one cell in each dimension
\[
0,1,2,3.
\]
The cellular boundary maps encode the finite cyclic fundamental-group data,
and a standard three-dimensional lens space \(L(p,q)\) has homology
\[
H_0\cong\mathbb Z,
\qquad
H_1\cong\mathbb Z/p,
\qquad
H_2=0,
\qquad
H_3\cong\mathbb Z.
\]

The parameter \(q\) affects finer topology but not these integral homology
groups.

\section{Closed orientable surfaces}

A genus-\(g\) orientable surface has one \(0\)-cell, \(2g\) one-cells, and one
\(2\)-cell.  The two-cell attaches by a product of commutators.

The total exponent of every one-cell in that attaching word is zero, so
\[
d_2=0.
\]
Also \(d_1=0\).

Therefore
\[
H_2(\Sigma_g)\cong\mathbb Z,
\qquad
H_1(\Sigma_g)\cong\mathbb Z^{2g},
\qquad
H_0(\Sigma_g)\cong\mathbb Z.
\]

\section{Nonorientable surface pattern}

For a closed nonorientable surface built from one vertex, \(g\) one-cells, and
one two-cell, the two-cell boundary contains squared generators.  Cellular
homology therefore produces a nontrivial multiplication-by-\(2\) component and
detects \(2\)-torsion in first homology.

\section{Euler characteristic}

For a finite CW complex,
\[
\chi(X)=\sum_n(-1)^n c_n.
\]

Since
\[
C_n^{\mathrm{cell}}(X)\cong\mathbb Z^{c_n},
\]
the Euler--Poincare principle later gives
\[
\chi(X)
=
\sum_n(-1)^n\operatorname{rank}H_n(X).
\]

\section{Cellular maps and induced maps}

A cellular map
\[
f:X\to Y
\]
induces maps between the skeletal pairs and hence a chain map
\[
C_*^{\mathrm{cell}}(X)\to C_*^{\mathrm{cell}}(Y).
\]

The induced homology map agrees with the usual singular-homology map.

\section{Why cellular homology is efficient}

For a finite CW complex, the cellular chain groups have ranks equal to the
finite cell counts.  By contrast, singular chain groups are typically
infinitely generated in every positive degree.

The price for this compression is that the cellular differential requires
careful analysis of attaching maps.

The definition
\[

\]

exact sequence systematically, explaining the connecting maps that underlie
the cellular boundary operator.

\section{Solved problem dossiers}

\begin{problem}[Minimal sphere]\label{prob:viii18-01}
Compute cellular homology of the two-cell CW structure on \(S^n\).
\end{problem}
\begin{solution}
The chain groups are \(\mathbb Z\) in degrees \(n\) and \(0\) and zero elsewhere. Since there is no adjacent nonzero target, every differential is zero, giving \(H_0=H_n=\mathbb Z\).
\end{solution}

\begin{problem}[Circle]\label{prob:viii18-02}
Compute cellular homology of the one-vertex one-edge circle.
\end{problem}
\begin{solution}
The chain complex is \(0\to\mathbb Z\xrightarrow0\mathbb Z\to0\), so \(H_1\cong H_0\cong\mathbb Z\).
\end{solution}

\begin{problem}[Two-circle wedge]\label{prob:viii18-03}
Compute cellular homology of \(S^1\vee S^1\).
\end{problem}
\begin{solution}
One vertex and two loop one-cells give \(0\to\mathbb Z^2\xrightarrow0\mathbb Z\to0\), so \(H_1\cong\mathbb Z^2\).
\end{solution}

\begin{problem}[Torus]\label{prob:viii18-04}
Compute all cellular homology groups of the standard torus.
\end{problem}
\begin{solution}
The chain complex is \(0\to\mathbb Z\xrightarrow0\mathbb Z^2\xrightarrow0\mathbb Z\to0\), hence \(H_2=\mathbb Z,\ H_1=\mathbb Z^2,\ H_0=\mathbb Z\).
\end{solution}

\begin{problem}[Genus two surface]\label{prob:viii18-05}
Compute \(H_*(\Sigma_2)\).
\end{problem}
\begin{solution}
One 0-cell, four 1-cells, and one 2-cell with commutator-product attaching word give zero differentials, so \(H_2=\mathbb Z,\ H_1=\mathbb Z^4,\ H_0=\mathbb Z\).
\end{solution}

\begin{problem}[Degree-three attachment]\label{prob:viii18-06}
Compute \(H_*(S^1\cup_3D^2)\).
\end{problem}
\begin{solution}
The chain complex has \(d_2=\times3\) and \(d_1=0\). Thus \(H_2=0,\ H_1=\mathbb Z/3,\ H_0=\mathbb Z\).
\end{solution}

\begin{problem}[Degree-one attachment]\label{prob:viii18-07}
Compute the homology of \(S^1\cup_1D^2\).
\end{problem}
\begin{solution}
Multiplication by one is an isomorphism, so positive homology vanishes and \(H_0=\mathbb Z\), agreeing with the disk model.
\end{solution}

\begin{problem}[Degree-zero attachment]\label{prob:viii18-08}
Compute the homology of \(S^1\cup_0D^2\).
\end{problem}
\begin{solution}
All differentials vanish, giving \(H_2=\mathbb Z,\ H_1=\mathbb Z,\ H_0=\mathbb Z\), as for \(S^1\vee S^2\).
\end{solution}

\begin{problem}[Projective plane]\label{prob:viii18-09}
Compute cellular homology of \(\mathbb RP^2\).
\end{problem}
\begin{solution}
The complex is \(0\to\mathbb Z\xrightarrow2\mathbb Z\xrightarrow0\mathbb Z\to0\). Hence \(H_2=0,\ H_1=\mathbb Z/2,\ H_0=\mathbb Z\).
\end{solution}

\begin{problem}[Projective 3-space]\label{prob:viii18-10}
Compute cellular homology of \(\mathbb RP^3\).
\end{problem}
\begin{solution}
The complex is \(0\to\mathbb Z\xrightarrow0\mathbb Z\xrightarrow2\mathbb Z\xrightarrow0\mathbb Z\to0\). Thus \(H_3=\mathbb Z,\ H_2=0,\ H_1=\mathbb Z/2,\ H_0=\mathbb Z\).
\end{solution}

\begin{problem}[Projective parity]\label{prob:viii18-11}
Why is top homology of \(\mathbb RP^n\) nonzero exactly for odd \(n\)?
\end{problem}
\begin{solution}
The top differential is zero for odd \(n\), giving kernel \(\mathbb Z\); for even \(n\) it is multiplication by \(2\), whose kernel is zero.
\end{solution}

\begin{problem}[Moore \(M(\mathbb Z/5,n)\)]\label{prob:viii18-12}
Compute its reduced homology.
\end{problem}
\begin{solution}
The only relevant differential is \(\mathbb Z\xrightarrow5\mathbb Z\), so \(\widetilde H_n\cong\mathbb Z/5\) and all other reduced groups vanish.
\end{solution}

\begin{problem}[Moore \(m=1\)]\label{prob:viii18-13}
What homology does \(S^n\cup_1D^{n+1}\) have?
\end{problem}
\begin{solution}
The multiplication-by-one differential is an isomorphism, so all reduced homology vanishes; the space is homologically contractible.
\end{solution}

\begin{problem}[Lens space]\label{prob:viii18-14}
Explain why \(H_1(L(p,q))\) is \(\mathbb Z/p\) in the standard cell model.
\end{problem}
\begin{solution}
The one-cell and two-cell portion contains a cellular differential whose essential coefficient is \(p\), so the relevant cokernel is \(\mathbb Z/p\).
\end{solution}

\begin{problem}[Lens parameter \(q\)]\label{prob:viii18-15}
Why can cellular integral homology fail to distinguish \(L(p,q)\) for different \(q\)?
\end{problem}
\begin{solution}
The standard cellular boundary ranks and the torsion coefficient \(p\) do not depend on \(q\). The parameter affects finer invariants beyond these homology groups.
\end{solution}

\begin{problem}[Euler torus]\label{prob:viii18-16}
Compute the torus Euler characteristic from its cells.
\end{problem}
\begin{solution}
\(\chi=1-2+1=0\).
\end{solution}

\begin{problem}[Euler genus \(g\)]\label{prob:viii18-17}
Compute \(\chi(\Sigma_g)\) from the standard cells.
\end{problem}
\begin{solution}
\(\chi=1-2g+1=2-2g\).
\end{solution}

\begin{problem}[Euler projective plane]\label{prob:viii18-18}
Compute \(\chi(\mathbb RP^2)\).
\end{problem}
\begin{solution}
There is one cell in dimensions 0,1,2, so \(\chi=1-1+1=1\).
\end{solution}

\begin{problem}[Boundary coefficient meaning]\label{prob:viii18-19}
Explain how an attaching map contributes an integer to a cellular differential.
\end{problem}
\begin{solution}
Project the attaching sphere onto the quotient sphere corresponding to one lower-dimensional cell; the degree of that sphere map is the incidence coefficient.
\end{solution}

\begin{problem}[Orientation change]\label{prob:viii18-20}
What happens to a cellular boundary matrix if the orientation of one cell is reversed?
\end{problem}
\begin{solution}
The corresponding basis generator changes sign, so one row or column changes sign. Homology is unchanged because this is a basis change by a unit.
\end{solution}

\begin{problem}[Cell subdivision]\label{prob:viii18-21}
Why should subdividing a CW or simplicial model not change the final homology?
\end{problem}
\begin{solution}
Subdivision changes the chain complex but not the underlying topological space; cellular/simplicial homology agrees with singular homology, which is an invariant of the space.
\end{solution}

\begin{problem}[Support-figure role]\label{prob:viii18-22}
Where do the mapped quotient/lens/Moore support diagrams belong conceptually?
\end{problem}
\begin{solution}
They illustrate cellular attaching maps and the resulting boundary matrices, so their canonical mathematical home is VIII/18 rather than an independent chapter.
\end{solution}

\begin{problem}[Cellular versus singular size]\label{prob:viii18-23}
Compare chain sizes for the minimal CW sphere.
\end{problem}
\begin{solution}
Cellular chains have only two generators total, while singular chains have infinitely many singular simplices in every relevant degree. Both yield the same homology.
\end{solution}

\begin{problem}[Bridge to relative exactness]\label{prob:viii18-24}
Why does the cellular construction naturally lead to VIII/19?
\end{problem}
\begin{solution}
Each cellular group is a relative group \(H_n(X^n,X^{n-1})\), and the cellular differential is built from connecting homomorphisms in exact sequences of pairs and triples.
\end{solution}
