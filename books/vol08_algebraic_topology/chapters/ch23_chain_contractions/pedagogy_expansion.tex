\section{Supplementary worked examples}
\label{sec:viii23-pedagogy}

\begin{example}[Contracting a cone complex]\label{ex:viii23-ped-01}
For a cone-like chain complex, a degree-$+1$ operator $s$ with
\[
ds+sd=\operatorname{id}
\]
is a chain contraction. Every cycle $z$ then satisfies $z=d\,s(z)$, so all positive-degree homology vanishes.
\end{example}

\begin{example}[Augmented simplex]\label{ex:viii23-ped-02}
The augmented simplicial chain complex of a simplex admits an explicit contraction by coning every simplex to a fixed vertex. This algebraically expresses the geometric contractibility of the simplex.
\end{example}

\begin{example}[A split acyclic complex]\label{ex:viii23-ped-03}
For the two-term complex $0\to A\xrightarrow{\operatorname{id}}A\to0$, define $s$ to be the identity from degree $0$ back to degree $1$. Then $ds+sd=\operatorname{id}$, giving an explicit contraction rather than merely observing that homology vanishes.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Contraction equation]\label{exr:viii23-ped-01}
State the chain-contraction identity.
\end{exercise}
\begin{hint}
The identity map is chain homotopic to zero.
\end{hint}
\begin{solution}
$ds+sd=\operatorname{id}_C$.
\end{solution}

\begin{exercise}[Acyclic consequence]\label{exr:viii23-ped-02}
If $C_*$ has a chain contraction, what is $H_*(C)$?
\end{exercise}
\begin{hint}
Apply the identity to a cycle.
\end{hint}
\begin{solution}
All homology groups vanish (or reduced homology in an augmented topological model).
\end{solution}

\begin{exercise}[Cycle primitive]\label{exr:viii23-ped-03}
For a cycle $z$, express it as a boundary using a contraction $s$.
\end{exercise}
\begin{hint}
$dz=0$.
\end{hint}
\begin{solution}
$z=ds(z)$.
\end{solution}

\begin{exercise}[Two-term identity complex]\label{exr:viii23-ped-04}
Contract $0\to A\xrightarrow{\mathrm{id}}A\to0$.
\end{exercise}
\begin{hint}
Send degree $0$ back to degree $1$ by the identity.
\end{hint}
\begin{solution}
Take $s_0=\operatorname{id}_A$ and all other $s_n=0$; then $ds+sd=\operatorname{id}$.
\end{solution}

\begin{exercise}[Direct sum]\label{exr:viii23-ped-05}
If $C$ and $D$ are contractible chain complexes, contract $C\oplus D$.
\end{exercise}
\begin{hint}
Use the direct sum of the two contractions.
\end{hint}
\begin{solution}
$s_C\oplus s_D$ satisfies $d(s_C\oplus s_D)+(s_C\oplus s_D)d=\operatorname{id}$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Contraction implies null identity]\label{exr:viii23-ped-06}
Prove a chain contraction is exactly a chain homotopy from $\operatorname{id}$ to $0$.
\end{exercise}
\begin{hint}
Compare definitions.
\end{hint}
\begin{solution}
The defining equation for a chain homotopy between $\operatorname{id}$ and $0$ is precisely $\operatorname{id}=ds+sd$.
\end{solution}

\begin{exercise}[Homology vanishing]\label{exr:viii23-ped-07}
Prove contractible complexes are acyclic.
\end{exercise}
\begin{hint}
Apply the contraction to cycles.
\end{hint}
\begin{solution}
If $dz=0$, then $z=ds(z)$, hence every cycle is a boundary.
\end{solution}

\begin{exercise}[Transport contraction]\label{exr:viii23-ped-08}
Show an isomorphic chain complex to a contractible one is contractible.
\end{exercise}
\begin{hint}
Conjugate the contraction by the chain isomorphism.
\end{hint}
\begin{solution}
If $\phi:C\to D$ is an isomorphism and $s$ contracts $C$, then $\phi s\phi^{-1}$ contracts $D$.
\end{solution}

\begin{exercise}[Tensor contraction]\label{exr:viii23-ped-09}
Under standard sign conventions, explain why tensoring a contractible complex with a free complex often remains contractible.
\end{exercise}
\begin{hint}
Use $s\otimes\operatorname{id}$ with the total differential.
\end{hint}
\begin{solution}
The graded Leibniz signs make the cross terms cancel, leaving the identity on the tensor product.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[Acyclic implies contractible]\label{exr:viii23-ped-10}
Test: every acyclic chain complex is chain contractible.
\end{exercise}
\begin{hint}
Splittings need not exist.
\end{hint}
\begin{solution}
False over general rings; acyclicity does not guarantee the exact sequences of cycles and boundaries split.
\end{solution}

\begin{exercise}[Topological contractibility automatic]\label{exr:viii23-ped-11}
Test: a topologically contractible space gives an unaugmented singular chain complex with zero $H_0$.
\end{exercise}
\begin{hint}
Remember ordinary $H_0$ of a point.
\end{hint}
\begin{solution}
False. Ordinary singular homology has $H_0\cong\mathbb Z$; the reduced or augmented complex is the acyclic object.
\end{solution}

\begin{exercise}[Contraction unique]\label{exr:viii23-ped-12}
Test: if a contraction exists, it is unique.
\end{exercise}
\begin{hint}
Different homotopy operators can satisfy the same identity.
\end{hint}
\begin{solution}
False; contractions are generally highly nonunique.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Eliminating cells]\label{exr:viii23-ped-13}
How does an explicit contraction justify removing an algebraically canceling pair of generators?
\end{exercise}
\begin{hint}
The pair forms a contractible summand.
\end{hint}
\begin{solution}
A contractible direct summand contributes no homology and may be eliminated while preserving the homology of the remaining complex.
\end{solution}

\begin{exercise}[Numerical simplification]\label{exr:viii23-ped-14}
Why are contractions useful in computational homology?
\end{exercise}
\begin{hint}
They produce explicit reductions rather than only isomorphism statements.
\end{hint}
\begin{solution}
A contraction supplies chain-level maps and homotopies that compress a large complex to a smaller one while certifying that homology is preserved.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Cone contraction]\label{exr:viii23-ped-15}
Sketch the contraction of the augmented chain complex of a simplex using a chosen vertex $v$.
\end{exercise}
\begin{hint}
Cone each simplex to $v$.
\end{hint}
\begin{solution}
Define $s(\sigma)=v*\sigma$ with the usual orientation convention; the boundary formula for a cone gives $ds+sd=\operatorname{id}$.
\end{solution}

\begin{exercise}[Strong reduction]\label{exr:viii23-ped-16}
Explain the difference between knowing $H_*(C)=0$ and possessing a chain contraction.
\end{exercise}
\begin{hint}
One is a homology statement; the other is explicit chain data.
\end{hint}
\begin{solution}
A contraction is a constructive null-homotopy of the identity and can be composed, transported, and used algorithmically; vanishing homology alone provides none of that extra structure.
\end{solution}
