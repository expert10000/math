\section{Supplementary worked examples}
\label{sec:viii06-pedagogy}

\begin{example}[CW structure on the torus]\label{ex:viii06-ped-01}
A torus admits one 0-cell, two 1-cells $a,b$, and one 2-cell attached along the commutator word $aba^{-1}b^{-1}$.  This compact cell structure exposes both its fundamental group relation and its cellular chains.
\end{example}

\begin{example}[Projective space skeletons]\label{ex:viii06-ped-02}
Real projective space has one cell in each dimension $0,1,\ldots,n$.  Its skeletal filtration $\mathbb{RP}^0\subset\cdots\subset\mathbb{RP}^n$ makes induction and cellular homology particularly efficient.
\end{example}

\begin{example}[Spheres as minimal CW complexes]\label{ex:viii06-ped-03}
$S^n$ has a CW structure with exactly one 0-cell and one $n$-cell.  This illustrates that a CW decomposition need not resemble a triangulation and can be dramatically smaller.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Skeleton definition]\label{exr:viii06-ped-01}
What is the $n$-skeleton $X^n$ of a CW complex?
\end{exercise}
\begin{hint}
Collect cells of dimension at most $n$.
\end{hint}
\begin{solution}
$X^n$ is the subspace obtained from all cells of dimensions $\le n$.
\end{solution}

\begin{exercise}[Sphere cells]\label{exr:viii06-ped-02}
How many cells occur in the standard two-cell CW structure on $S^n$?
\end{exercise}
\begin{hint}
Use dimensions $0$ and $n$.
\end{hint}
\begin{solution}
Two cells: one 0-cell and one $n$-cell.
\end{solution}

\begin{exercise}[Torus chain ranks]\label{exr:viii06-ped-03}
For the standard torus CW structure, state the ranks of $C_0,C_1,C_2$.
\end{exercise}
\begin{hint}
Count cells in each dimension.
\end{hint}
\begin{solution}
The ranks are $1,2,1$ respectively.
\end{solution}

\begin{exercise}[Wedge of circles]\label{exr:viii06-ped-04}
Give a CW structure on a wedge of $r$ circles.
\end{exercise}
\begin{hint}
Use a common 0-cell.
\end{hint}
\begin{solution}
Use one 0-cell and $r$ 1-cells, each with both endpoints attached to the common vertex.
\end{solution}

\begin{exercise}[Finite CW Euler count]\label{exr:viii06-ped-05}
For a finite CW complex with $c_n$ cells in dimension $n$, write the cell-count expression for Euler characteristic.
\end{exercise}
\begin{hint}
Use alternating signs.
\end{hint}
\begin{solution}
$\chi(X)=\sum_n(-1)^n c_n$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Skeleta are subcomplexes]\label{exr:viii06-ped-06}
Prove each skeleton $X^n$ is a CW subcomplex of $X$.
\end{exercise}
\begin{hint}
Its cells are closed under taking attaching boundaries into lower dimensions.
\end{hint}
\begin{solution}
By construction every $k$-cell with $k\le n$ attaches to $X^{k-1}\subset X^n$, so their union carries the induced CW structure.
\end{solution}

\begin{exercise}[Closure finite principle]\label{exr:viii06-ped-07}
Explain why the closure of a cell in a CW complex meets only finitely many cells under the closure-finite axiom.
\end{exercise}
\begin{hint}
This is one of the defining finiteness conditions.
\end{hint}
\begin{solution}
The characteristic map of a closed disk meets the lower skeleton through a compact boundary and the CW closure-finite condition requires only finitely many cell intersections.
\end{solution}

\begin{exercise}[Weak topology]\label{exr:viii06-ped-08}
State the weak topology test for a CW complex and explain its role.
\end{exercise}
\begin{hint}
Test closedness on every closed cell/finite skeleton piece.
\end{hint}
\begin{solution}
A set is closed when its intersection with each closed cell (equivalently each skeleton in the standard formulation) is closed; this ensures the topology is determined by the attaching process.
\end{solution}

\begin{exercise}[Cellular induction]\label{exr:viii06-ped-09}
Why can many maps/properties on CW complexes be constructed inductively over skeleta?
\end{exercise}
\begin{hint}
Pass from $X^{n-1}$ to $X^n$ by attaching $n$-cells.
\end{hint}
\begin{solution}
At each step the extension problem reduces to extending data over disks from their boundary spheres, allowing dimension-by-dimension induction.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[CW equals simplicial]\label{exr:viii06-ped-10}
Test: every CW decomposition is a simplicial complex decomposition.
\end{exercise}
\begin{hint}
Cells need not be simplices and attaching maps need not be injective.
\end{hint}
\begin{solution}
False. CW complexes allow general disks and boundary identifications.
\end{solution}

\begin{exercise}[One CW structure]\label{exr:viii06-ped-11}
Test: a space has at most one CW structure.
\end{exercise}
\begin{hint}
Compare a circle with different subdivisions.
\end{hint}
\begin{solution}
False. The same space can admit many CW structures with very different cell counts.
\end{solution}

\begin{exercise}[Finite dimensional means finite cells]\label{exr:viii06-ped-12}
Test: every finite-dimensional CW complex has finitely many cells.
\end{exercise}
\begin{hint}
Dimension bounds cell dimensions, not their number.
\end{hint}
\begin{solution}
False; there may be infinitely many cells in a fixed dimension.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Computation compression]\label{exr:viii06-ped-13}
Why can a coarse CW structure outperform a triangulation for algebraic computations?
\end{exercise}
\begin{hint}
Chain ranks equal cell counts.
\end{hint}
\begin{solution}
Fewer cells produce much smaller cellular chain groups and boundary matrices while preserving the topology.
\end{solution}

\begin{exercise}[Incremental topology]\label{exr:viii06-ped-14}
Explain why the skeletal filtration is natural for persistent or staged calculations.
\end{exercise}
\begin{hint}
Each stage adds one dimension of cells.
\end{hint}
\begin{solution}
The filtration isolates how topology changes as higher-dimensional cells are attached, making inductive exact sequences and cellular differentials natural.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Torus relation]\label{exr:viii06-ped-15}
Explain how the torus 2-cell attaching word encodes commutativity of the two fundamental loops.
\end{exercise}
\begin{hint}
The attached disk kills its boundary loop.
\end{hint}
\begin{solution}
The word $aba^{-1}b^{-1}$ becomes null-homotopic, imposing $ab=ba$ in the fundamental group.
\end{solution}

\begin{exercise}[CW synthesis]\label{exr:viii06-ped-16}
For a finite CW complex, explain the bridge from geometric cell counts to Euler characteristic and then to homology ranks.
\end{exercise}
\begin{hint}
Use cellular chains and the Euler-Poincare identity.
\end{hint}
\begin{solution}
Cell counts give alternating ranks of cellular chain groups; cancellation of boundary ranks shows this equals the alternating sum of homology ranks, so $\chi$ is both combinatorial and homological.
\end{solution}
