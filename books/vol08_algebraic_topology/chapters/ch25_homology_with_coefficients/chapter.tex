\chapter{Homology with Coefficients}
\label{ch:viii25-homology-coefficients}

Integral homology is only one member of a family of invariants.  Replacing the
coefficient group \(\mathbb Z\) by an abelian group \(G\) can reveal torsion,
simplify calculations, and produce operations such as Bockstein connecting
homomorphisms.

\section{Chains with coefficients}
For a chain complex \(C_\bullet\) of free abelian groups and an abelian group
\(G\), define
\[
C_n(C;G)=C_n\otimes G
\]
with differential
\[
\partial\otimes\operatorname{id}_G.
\]

\begin{definition}[Homology with coefficients]
\label{def:viii25-coefficients}
\[
H_n(C;G)=H_n(C_\bullet\otimes G).
\]
For a space \(X\),
\[
H_n(X;G)=H_n(C_*^{\rm sing}(X)\otimes G).
\]
\end{definition}

\section{Integer coefficients}
Taking \(G=\mathbb Z\) recovers ordinary integral homology.

\section{Field coefficients}
If \(G=\mathbb F_p=\mathbb Z/p\), every chain group becomes a vector space over
\(\mathbb F_p\).  Boundary matrices may be reduced modulo \(p\), and homology
can be computed by ordinary linear algebra.

\section{Cellular chains with coefficients}
For a CW complex,
\[
C_n^{\rm cell}(X;G)
\cong
C_n^{\rm cell}(X;\mathbb Z)\otimes G.
\]
Thus the same cellular boundary matrices are interpreted in \(G\).

\section{Degree-\(m\) attachment}
For
\[
X_m=S^1\cup_mD^2,
\]
the coefficient chain complex is
\[
0\to G\xrightarrow{\times m}G\to G\to0.
\]
Hence the result depends strongly on whether multiplication by \(m\) is
injective or zero in \(G\).

\section{Projective plane mod 2}
The integral cellular differential of \(\mathbb RP^2\) is multiplication by
\(2\):
\[
0\to\mathbb Z\xrightarrow{2}\mathbb Z\xrightarrow0\mathbb Z\to0.
\]
With \(\mathbb F_2\)-coefficients, this differential becomes zero.

Therefore
\[
H_k(\mathbb RP^2;\mathbb F_2)\cong\mathbb F_2
\qquad
(k=0,1,2).
\]

\section{Projective spaces mod 2}
More generally, every cellular differential in \(\mathbb RP^n\) becomes zero
modulo \(2\), so
\[
H_k(\mathbb RP^n;\mathbb F_2)\cong\mathbb F_2
\qquad
0\le k\le n.
\]

\section{Moore spaces}
For
\[
M(\mathbb Z/m,n)=S^n\cup_mD^{n+1},
\]
the coefficient chain complex contains
\[
G\xrightarrow{\times m}G.
\]
Thus
\[
H_{n+1}(M;G)=\ker(\times m:G\to G),
\]
and
\[
\widetilde H_n(M;G)=G/mG.
\]

\section{Example with mod \(p\)}
For \(G=\mathbb F_p\):
\begin{itemize}
\item if \(p\nmid m\), multiplication by \(m\) is an isomorphism;
\item if \(p\mid m\), multiplication by \(m\) is zero.
\end{itemize}

So mod-\(p\) homology detects exactly the \(p\)-primary part of this Moore
attachment.

\section{Tensor products of abelian groups}
Basic identities include
\[
\mathbb Z\otimes G\cong G,
\qquad
\mathbb Z/m\otimes G\cong G/mG,
\]
and
\[
\mathbb Z/m\otimes\mathbb Z/n
\cong
\mathbb Z/\gcd(m,n).
\]

These formulas foreshadow the universal coefficient theorem.

\section{Tor preview}
Tensoring is right exact but not exact in general.  The failure of left
exactness is measured by
\[
\operatorname{Tor}_1^{\mathbb Z}(-,G).
\]

For cyclic groups,
\[
\operatorname{Tor}(\mathbb Z/m,\mathbb Z/n)
\cong
\mathbb Z/\gcd(m,n).
\]

VIII/26 makes this correction precise.

\section{Short exact coefficient sequences}
A short exact sequence of coefficient groups
\[
0\to G'\to G\to G''\to0
\]
produces a short exact sequence of singular chain complexes because singular
chain groups are free abelian.

Therefore it yields a long exact sequence in homology:
\[
\cdots\to H_n(X;G')
\to H_n(X;G)
\to H_n(X;G'')
\xrightarrow{\beta}
H_{n-1}(X;G')\to\cdots.
\]

\section{Bockstein homomorphism}
The connecting homomorphism
\[
\beta:H_n(X;G'')\to H_{n-1}(X;G')
\]
is called a Bockstein homomorphism for the chosen coefficient sequence.

A central example is
\[
0\to\mathbb Z\xrightarrow{\times p}\mathbb Z\to\mathbb Z/p\to0.
\]

\section{Bockstein and torsion}
The Bockstein can detect when a mod-\(p\) class comes from \(p\)-torsion in
integral homology.

For projective spaces this relates the rich mod-\(2\) homology to integral
\(2\)-torsion.

\section{Naturality}
Continuous maps induce homomorphisms on homology with arbitrary coefficients,
and coefficient homomorphisms
\[
G\to H
\]
also induce maps
\[
H_n(X;G)\to H_n(X;H).
\]

Thus homology is functorial in both the space and the coefficient group.

\section{Reduced homology}
Reduced homology with coefficients is defined using the augmented coefficient
chain complex.  For a connected space,
\[
\widetilde H_0(X;G)=0.
\]

\section{Relative coefficients}
For a pair \((X,A)\),
\[
H_n(X,A;G)
=
H_n(C_*(X,A)\otimes G).
\]
Long exact sequences of pairs remain valid with coefficients.

\section{Orientation with coefficients}
A manifold that is not orientable over \(\mathbb Z\) may become orientable over
\(\mathbb F_2\), because sign reversal is invisible modulo \(2\).

This phenomenon is developed systematically in the manifold chapters.

\section{Characteristic dependence}
Different coefficient fields can produce different Betti numbers.  One should
therefore always record the coefficient ring when reporting homology:
\[
H_n(X;\mathbb Z),
\quad
H_n(X;\mathbb Q),
\quad
H_n(X;\mathbb F_p)
\]
need not contain equivalent information.

\section{Rational homology}
Tensoring with \(\mathbb Q\) kills finite torsion.  Thus rational homology
records the free part of integral homology.

\section{Support-diagram provenance}
The legacy support corpus contains explicit Moore-space, Bockstein, and
projective-space coefficient diagrams mapped to this chapter.  Their
mathematical roles are integrated here; exact visual-instance reconciliation is
reserved for the post-VIII/35 audit.

\section{Bridge to UCT}
Computing directly after tensoring chains works, but the universal coefficient
theorem gives a far more efficient answer in terms of integral homology:
a tensor term plus a Tor correction.  VIII/26 develops that theorem.

\section{Solved problem dossiers}

\begin{problem}[RP2 mod 2]\label{prob:viii25-01}
Compute \(H_*(\mathbb RP^2;\mathbb F_2)\) cellularly.
\end{problem}
\begin{solution}
The integral differential \(d_2=2\) becomes zero modulo 2, as does \(d_1\); hence one copy of \(\mathbb F_2\) survives in degrees 0,1,2.
\end{solution}

\begin{problem}[RP2 mod 3]\label{prob:viii25-02}
Compute positive homology of \(\mathbb RP^2;\mathbb F_3\).
\end{problem}
\begin{solution}
Multiplication by 2 is invertible mod 3, so \(H_2=H_1=0\).
\end{solution}

\begin{problem}[Degree m mod p]\label{prob:viii25-03}
For \(S^1\cup_mD^2\), compute \(H_1(-;\mathbb F_p)\).
\end{problem}
\begin{solution}
It is the cokernel of multiplication by \(m\) on \(\mathbb F_p\): zero if \(p\nmid m\), and \(\mathbb F_p\) if \(p\mid m\).
\end{solution}

\begin{problem}[Moore mod p]\label{prob:viii25-04}
Compute reduced homology of \(M(\mathbb Z/m,n)\) over \(\mathbb F_p\).
\end{problem}
\begin{solution}
If \(p\nmid m\), it vanishes; if \(p\mid m\), copies of \(\mathbb F_p\) occur in degrees \(n\) and \(n+1\).
\end{solution}

\begin{problem}[Tensor cyclic]\label{prob:viii25-05}
Compute \(\mathbb Z/12\otimes\mathbb Z/18\).
\end{problem}
\begin{solution}
It is \(\mathbb Z/\gcd(12,18)\cong\mathbb Z/6\).
\end{solution}

\begin{problem}[Tensor with Q]\label{prob:viii25-06}
Compute \(\mathbb Z/m\otimes\mathbb Q\).
\end{problem}
\begin{solution}
Zero, since multiplication by \(m\) is invertible on \(\mathbb Q\).
\end{solution}

\begin{problem}[Rational projective]\label{prob:viii25-07}
Compute rational homology of \(\mathbb RP^2\).
\end{problem}
\begin{solution}
Integral \(H_1=\mathbb Z/2\) is killed by tensoring with \(\mathbb Q\); only \(H_0\cong\mathbb Q\) remains.
\end{solution}

\begin{problem}[Rational torus]\label{prob:viii25-08}
Compute \(H_1(T^2;\mathbb Q)\).
\end{problem}
\begin{solution}
\(\mathbb Z^2\otimes\mathbb Q\cong\mathbb Q^2\).
\end{solution}

\begin{problem}[Coefficient map]\label{prob:viii25-09}
What map does \(\mathbb Z\to\mathbb Z/p\) induce?
\end{problem}
\begin{solution}
It gives reduction mod \(p\): \(H_n(X;\mathbb Z)\to H_n(X;\mathbb Z/p)\), though not every mod-\(p\) class need arise this way because of Tor/Bockstein effects.
\end{solution}

\begin{problem}[Bockstein source]\label{prob:viii25-10}
Explain the Bockstein from \(0\to\mathbb Z\xrightarrow p\mathbb Z\to\mathbb Z/p\to0\).
\end{problem}
\begin{solution}
It is the connecting map \(H_n(X;\mathbb Z/p)\to H_{n-1}(X;\mathbb Z)\).
\end{solution}

\begin{problem}[Bockstein meaning]\label{prob:viii25-11}
What can a nonzero Bockstein indicate?
\end{problem}
\begin{solution}
It indicates an obstruction to lifting a mod-\(p\) class to integral coefficients and often detects \(p\)-torsion one degree lower.
\end{solution}

\begin{problem}[Field computation]\label{prob:viii25-12}
Why are mod-\(p\) cellular calculations often easier?
\end{problem}
\begin{solution}
Boundary matrices become matrices over a field, so Gaussian elimination computes kernels and images directly.
\end{solution}

\begin{problem}[Orientation mod 2]\label{prob:viii25-13}
Why can every manifold be locally oriented mod 2?
\end{problem}
\begin{solution}
Changing an orientation sign multiplies by \(-1\), which equals \(1\) in \(\mathbb F_2\).
\end{solution}

\begin{problem}[Relative coefficients]\label{prob:viii25-14}
State the coefficient LES of a pair.
\end{problem}
\begin{solution}
\(\cdots\to H_n(A;G)\to H_n(X;G)\to H_n(X,A;G)\to H_{n-1}(A;G)\to\cdots\).
\end{solution}

\begin{problem}[Disjoint union]\label{prob:viii25-15}
How does coefficient homology behave on disjoint unions?
\end{problem}
\begin{solution}
Chains split by components, so \(H_n(\coprod X_\alpha;G)\cong\bigoplus H_n(X_\alpha;G)\).
\end{solution}

\begin{problem}[Sphere coefficients]\label{prob:viii25-16}
Compute \(H_*(S^n;G)\).
\end{problem}
\begin{solution}
For nonempty \(G\), \(H_0\cong G\), \(H_n\cong G\), and other groups vanish.
\end{solution}

\begin{problem}[Circle coefficients]\label{prob:viii25-17}
Compute \(H_1(S^1;G)\).
\end{problem}
\begin{solution}
It is \(G\), since the cellular differential is zero.
\end{solution}

\begin{problem}[Moore G/mG]\label{prob:viii25-18}
Explain the quotient \(G/mG\).
\end{problem}
\begin{solution}
It is the cokernel of the degree-\(m\) cellular boundary \(G\xrightarrow{\times m}G\).
\end{solution}

\begin{problem}[Moore kernel]\label{prob:viii25-19}
Explain the extra \(H_{n+1}\) coefficient group.
\end{problem}
\begin{solution}
It is the kernel of multiplication by \(m\) on \(G\), which may be nonzero even though integral top homology vanishes.
\end{solution}

\begin{problem}[Mod 2 projective spaces]\label{prob:viii25-20}
Compute \(H_k(\mathbb RP^n;\mathbb F_2)\).
\end{problem}
\begin{solution}
Every cellular differential becomes zero, so it is \(\mathbb F_2\) in every degree \(0\le k\le n\).
\end{solution}

\begin{problem}[Coefficient dependence]\label{prob:viii25-21}
Give a space whose Betti numbers depend on the field.
\end{problem}
\begin{solution}
\(\mathbb RP^2\): over \(\mathbb Q\), positive homology vanishes; over \(\mathbb F_2\), both \(H_1\) and \(H_2\) are nonzero.
\end{solution}

\begin{problem}[Tor preview]\label{prob:viii25-22}
Compute \(\operatorname{Tor}(\mathbb Z/12,\mathbb Z/18)\).
\end{problem}
\begin{solution}
It is also \(\mathbb Z/6\), illustrating that tensor and Tor have the same cyclic gcd size but occur in different UCT degrees.
\end{solution}
