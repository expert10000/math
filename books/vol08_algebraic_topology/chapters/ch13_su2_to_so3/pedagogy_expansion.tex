\section{Supplementary worked examples}
\label{sec:viii13-pedagogy}

\begin{example}[Unit quaternions as $S^3$]\label{ex:viii13-ped-01}
$SU(2)$ is diffeomorphic to the unit quaternions $S^3$.  A unit quaternion $q$ acts on the imaginary quaternions $\operatorname{Im}\mathbb H\cong\mathbb R^3$ by $v\mapsto qvq^{-1}$, giving a rotation.
\end{example}

\begin{example}[Kernel $\{\pm1\}$]\label{ex:viii13-ped-02}
The conjugation action of $q$ and $-q$ on imaginary quaternions is identical.  These are exactly the elements in the kernel, so the map $SU(2)\to SO(3)$ is two-to-one.
\end{example}

\begin{example}[Double cover and fundamental group]\label{ex:viii13-ped-03}
Because $SU(2)\cong S^3$ is simply connected and covers $SO(3)$ with deck group $\mathbb Z/2$, it is the universal cover.  Consequently $\pi_1(SO(3))\cong\mathbb Z/2$.
\end{example}

\section{Graded supplementary exercises}
The following exercises progress from direct computation and construction to proof, hypothesis testing, application, and synthesis.

\subsection*{Standard computations and constructions}

\begin{exercise}[Dimension]\label{exr:viii13-ped-01}
What is the real dimension of $SU(2)$?
\end{exercise}
\begin{hint}
Identify it with unit quaternions.
\end{hint}
\begin{solution}
It is 3, since the unit quaternions form $S^3\subset\mathbb R^4$.
\end{solution}

\begin{exercise}[Kernel elements]\label{exr:viii13-ped-02}
Which two elements of $SU(2)$ map to the identity rotation in $SO(3)$?
\end{exercise}
\begin{hint}
Central unit quaternions act trivially by conjugation.
\end{hint}
\begin{solution}
$q=1$ and $q=-1$.
\end{solution}

\begin{exercise}[Fiber size]\label{exr:viii13-ped-03}
How many points are in each fiber of $SU(2)\to SO(3)$?
\end{exercise}
\begin{hint}
It is a double cover.
\end{hint}
\begin{solution}
Exactly two.
\end{solution}

\begin{exercise}[Deck involution]\label{exr:viii13-ped-04}
Write the nontrivial deck transformation of the double cover.
\end{exercise}
\begin{hint}
Swap $q$ with the other point in its fiber.
\end{hint}
\begin{solution}
$q\mapsto -q$.
\end{solution}

\begin{exercise}[Fundamental group]\label{exr:viii13-ped-05}
Compute $\pi_1(SO(3))$ using the universal cover.
\end{exercise}
\begin{hint}
The deck group of a universal cover identifies with the fundamental group.
\end{hint}
\begin{solution}
$\pi_1(SO(3))\cong\mathbb Z/2$.
\end{solution}

\subsection*{Proofs}

\begin{exercise}[Conjugation preserves norm]\label{exr:viii13-ped-06}
Prove $v\mapsto qvq^{-1}$ preserves the Euclidean norm on imaginary quaternions.
\end{exercise}
\begin{hint}
Quaternion norm is multiplicative.
\end{hint}
\begin{solution}
$|qvq^{-1}|=|q||v||q|^{-1}=|v|$ for unit $q$; conjugation also preserves the imaginary subspace, hence gives an orthogonal transformation.
\end{solution}

\begin{exercise}[Orientation]\label{exr:viii13-ped-07}
Explain why the induced orthogonal transformation lies in $SO(3)$ rather than merely $O(3)$.
\end{exercise}
\begin{hint}
Use continuity and the value at $q=1$.
\end{hint}
\begin{solution}
The determinant is a continuous map from connected $SU(2)$ to $\{\pm1\}$ and equals $+1$ at the identity, so it is always $+1$.
\end{solution}

\begin{exercise}[Kernel]\label{exr:viii13-ped-08}
Prove the kernel of the conjugation homomorphism is $\{\pm1\}$.
\end{exercise}
\begin{hint}
An element commuting with every imaginary quaternion is central in $\mathbb H$.
\end{hint}
\begin{solution}
Such a unit quaternion must be real, hence equals $\pm1$.
\end{solution}

\begin{exercise}[Universal cover]\label{exr:viii13-ped-09}
Prove the double cover is universal.
\end{exercise}
\begin{hint}
Use $SU(2)\cong S^3$.
\end{hint}
\begin{solution}
$S^3$ is simply connected, so the connected covering $SU(2)\to SO(3)$ has simply connected total space and is universal.
\end{solution}

\subsection*{Counterexamples and hypothesis tests}

\begin{exercise}[$SU(2)=SO(3)$]\label{exr:viii13-ped-10}
Test: $SU(2)$ and $SO(3)$ are isomorphic as Lie groups.
\end{exercise}
\begin{hint}
The map has a two-element kernel.
\end{hint}
\begin{solution}
False. $SU(2)$ is a double cover of $SO(3)$, not isomorphic to it.
\end{solution}

\begin{exercise}[Rotation has unique lift]\label{exr:viii13-ped-11}
Test: every rotation has a unique representing unit quaternion.
\end{exercise}
\begin{hint}
Remember the kernel.
\end{hint}
\begin{solution}
False. Each rotation has exactly two lifts $q$ and $-q$.
\end{solution}

\begin{exercise}[$SO(3)$ simply connected]\label{exr:viii13-ped-12}
Test: $SO(3)$ is simply connected because it is connected.
\end{exercise}
\begin{hint}
Connectedness is much weaker.
\end{hint}
\begin{solution}
False. Its fundamental group is $\mathbb Z/2$.
\end{solution}

\subsection*{Applications and investigations}

\begin{exercise}[Quaternion rotation]\label{exr:viii13-ped-13}
Why are unit quaternions numerically attractive for representing rotations?
\end{exercise}
\begin{hint}
They lift $SO(3)$ to a sphere without Euler-angle singularities.
\end{hint}
\begin{solution}
A unit quaternion provides a smooth normalized four-coordinate representation; the only redundancy is the harmless sign pair $q\sim -q$, avoiding gimbal-lock coordinate singularities.
\end{solution}

\begin{exercise}[Spinorial sign]\label{exr:viii13-ped-14}
Interpret the fact that a $2\pi$ rotation represents the nontrivial loop in $SO(3)$ while a $4\pi$ rotation becomes null-homotopic.
\end{exercise}
\begin{hint}
Lift the loop to $SU(2)$.
\end{hint}
\begin{solution}
A $2\pi$ rotation lifts from $1$ to $-1$ and therefore does not close; traversing twice reaches $1$ again, reflecting the order-two fundamental group.
\end{solution}

\subsection*{Challenge problems}

\begin{exercise}[Loop lifting]\label{exr:viii13-ped-15}
Describe the lift of a generator of $\pi_1(SO(3))$ to $SU(2)$.
\end{exercise}
\begin{hint}
A generator should switch the two points over the identity rotation.
\end{hint}
\begin{solution}
Starting at $1$, its lift ends at $-1$.  Squaring the loop lifts to a closed path, consistent with order two.
\end{solution}

\begin{exercise}[Cover synthesis]\label{exr:viii13-ped-16}
Connect the algebraic kernel, deck group, and fundamental group in the $SU(2)\to SO(3)$ example.
\end{exercise}
\begin{hint}
All three involve the same $\{\pm1\}$ symmetry.
\end{hint}
\begin{solution}
The kernel is $\{\pm1\}$; multiplication by $-1$ is the nontrivial deck transformation; since the cover is universal, this deck group is canonically isomorphic to $\pi_1(SO(3))\cong\mathbb Z/2$.
\end{solution}
