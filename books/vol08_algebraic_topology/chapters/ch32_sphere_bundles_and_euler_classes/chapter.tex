
\chapter{Sphere Bundles and Euler Classes}
\label{ch:viii32-sphere-bundles-euler}

The unit sphere bundle converts a nowhere-zero vector field into a genuine
section problem.  The Euler class measures the primary obstruction to solving
that problem globally.

\section{Unit sphere bundles}
For a rank-$r$ real vector bundle $E\to B$ with metric,
\[
S(E)=\{v\in E:\|v\|=1\}\longrightarrow B
\]
has fiber $S^{r-1}$.

\section{Nowhere-zero sections}
A nowhere-zero section of $E$ normalizes to a section of $S(E)$, and every
section of $S(E)$ is a nowhere-zero section of $E$.

\section{Euler class}
\begin{definition}[Euler class]
\label{def:viii32-euler-class}
For an oriented rank-$r$ bundle with Thom class $u_E$ and zero section
$s:B\to E$,
\[
e(E)=s^*u_E\in H^r(B;R).
\]
\end{definition}

\section{The obstruction theorem}
\begin{theorem}[Nowhere-zero section obstruction]
\label{thm:viii32-euler-obstruction}
If an oriented vector bundle admits a nowhere-zero section, then
\[
e(E)=0.
\]
\end{theorem}
The converse is an obstruction-theoretic statement that requires appropriate
hypotheses; the Euler class is the primary obstruction, not in every setting a
complete classifier of sections.

\section{Naturality}
For $f:Y\to B$,
\[
e(f^*E)=f^*e(E).
\]

\section{Whitney sums}
For oriented bundles,
\[
e(E\oplus F)=e(E)\smile e(F).
\]
In particular, adding a positive-rank trivial summand forces the Euler class to
vanish.

\section{Sphere bundles and the Gysin sequence}
For an oriented rank-$r$ bundle, the sphere bundle $S(E)\to B$ has a long exact
Gysin sequence.  The characteristic map in the base is cup product with
$e(E)$:
\[
\cdots\to H^{k-r}(B)\xrightarrow{\smile e(E)}H^k(B)
\to H^k(S(E))\to H^{k-r+1}(B)\to\cdots.
\]

\section{Tangent bundles}
For a closed oriented $n$-manifold,
\[
e(TM)\in H^n(M).
\]
Its evaluation on the fundamental class is the Euler number.

\begin{theorem}[Euler class and Euler characteristic]
\label{thm:viii32-euler-characteristic}
For a closed oriented manifold $M$,
\[
\langle e(TM),[M]\rangle=\chi(M).
\]
\end{theorem}

\section{The hairy-ball theorem}
Since $\chi(S^{2m})=2$, the tangent Euler class of an even-dimensional sphere
is nonzero.  Thus $S^{2m}$ admits no nowhere-zero tangent vector field.

\section{Odd-dimensional manifolds}
A closed oriented odd-dimensional manifold has Euler characteristic zero.  In
particular, the tangent Euler number vanishes.

\section{Zeros of transverse sections}
For an oriented rank-$r$ bundle over an oriented $r$-manifold, a generic
section has isolated transverse zeros.  Each zero has a local sign, and the
signed sum equals the Euler number.

\section{Poincaré--Hopf viewpoint}
For $E=TM$, the preceding statement becomes the Poincaré--Hopf theorem: the
sum of indices of isolated zeros of a vector field equals $\chi(M)$.

\section{Zero loci as dual cycles}
More generally, a transverse section of an oriented rank-$r$ bundle has a
codimension-$r$ zero set whose fundamental class is Poincaré dual to $e(E)$.
This is the direct bridge to VIII/33.

\section{Solved dossiers}
\begin{problem}
Show that a nowhere-zero section of $E$ is equivalent to a section of $S(E)$ after choosing a metric.
\end{problem}
\begin{solution}
Normalize the vector field fiberwise; conversely include the unit sphere bundle in $E$.
\end{solution}

\begin{problem}
Define the Euler class from the Thom class.
\end{problem}
\begin{solution}
For an oriented rank-$r$ bundle, $e(E)=s^*u_E\in H^r(B)$.
\end{solution}

\begin{problem}
Why does a nowhere-zero section force $e(E)=0$?
\end{problem}
\begin{solution}
Move the zero section through a homotopy to a disjoint section; the pulled-back Thom class vanishes.
\end{solution}

\begin{problem}
Compute the Euler class of a trivial positive-rank oriented bundle.
\end{problem}
\begin{solution}
It is zero because a constant nowhere-zero section exists.
\end{solution}

\begin{problem}
What is the Euler class of an oriented line bundle with a nowhere-zero section?
\end{problem}
\begin{solution}
Zero.
\end{solution}

\begin{problem}
State multiplicativity for Whitney sums.
\end{problem}
\begin{solution}
$e(E\oplus F)=e(E)\smile e(F)$ for oriented bundles.
\end{solution}

\begin{problem}
Relate $e(TM)$ to the Euler characteristic.
\end{problem}
\begin{solution}
For a closed oriented manifold, $\langle e(TM),[M]\rangle=\chi(M)$.
\end{solution}

\begin{problem}
Evaluate the Euler number of $TS^2$.
\end{problem}
\begin{solution}
It equals $\chi(S^2)=2$.
\end{solution}

\begin{problem}
What happens for a closed oriented odd-dimensional manifold?
\end{problem}
\begin{solution}
Its Euler characteristic, hence tangent Euler number, is zero.
\end{solution}

\begin{problem}
Why does $S^{2m}$ have no nowhere-zero tangent vector field?
\end{problem}
\begin{solution}
Its Euler class pairs to $2\ne0$.
\end{solution}

\begin{problem}
Why is an Euler class an obstruction rather than always a complete classifier?
\end{problem}
\begin{solution}
Vanishing is necessary for a nonzero section, but further obstructions can exist outside the standard obstruction-theoretic range.
\end{solution}

\begin{problem}
Where does the Euler class appear in the Gysin sequence?
\end{problem}
\begin{solution}
It is the class whose cup-product map links adjacent base-cohomology terms.
\end{solution}

\begin{problem}
Describe the sphere bundle of a rank-$r$ bundle.
\end{problem}
\begin{solution}
It has fiber $S^{r-1}$ and records nonzero directions in each vector fiber.
\end{solution}

\begin{problem}
What is the geometric meaning of a zero of a transverse section?
\end{problem}
\begin{solution}
It contributes a local index; the signed sum represents the Euler number.
\end{solution}

\begin{problem}
State the Poincaré--Hopf connection.
\end{problem}
\begin{solution}
For an oriented closed manifold, the sum of vector-field indices equals $\chi(M)$.
\end{solution}

\begin{problem}
What coefficient system avoids orientability issues?
\end{problem}
\begin{solution}
Mod-$2$ coefficients.
\end{solution}

\begin{problem}
How does pullback affect Euler classes?
\end{problem}
\begin{solution}
$e(f^*E)=f^*e(E)$.
\end{solution}

\begin{problem}
What is the Euler class of $E\oplus\varepsilon^1$?
\end{problem}
\begin{solution}
Zero, since the trivial summand gives a nowhere-zero section; equivalently multiplicativity includes $e(\varepsilon^1)=0$.
\end{solution}

\begin{problem}
How does VIII/32 lead to duality?
\end{problem}
\begin{solution}
The zero locus of a transverse section represents the Poincaré dual of the Euler class.
\end{solution}

\begin{problem}
Why are sphere bundles visually natural here?
\end{problem}
\begin{solution}
A global section is literally a continuous choice of one point on every sphere fiber.
\end{solution}
