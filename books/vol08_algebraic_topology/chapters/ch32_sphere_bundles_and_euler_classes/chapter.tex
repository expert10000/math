
\chapter{Sphere Bundles and Euler Classes}
\label{ch:viii32-sphere-bundles-euler}

The unit sphere bundle converts a nowhere-zero vector field into a genuine
section problem.  The Euler class measures the primary obstruction to solving
that problem globally.

\section{Unit sphere bundles}
For a rank-$r$ real vector bundle $E\to B$ with metric,
\[
S(E)=\{v\in E:\|v\|=1\}\longrightarrow B
\]
has fiber $S^{r-1}$.

\section{Nowhere-zero sections}
A nowhere-zero section of $E$ normalizes to a section of $S(E)$, and every
section of $S(E)$ is a nowhere-zero section of $E$.

\section{Euler class}
\begin{definition}[Euler class]
\label{def:viii32-euler-class}
For an oriented rank-$r$ bundle with Thom class $u_E$ and zero section
$s:B\to E$,
\[
e(E)=s^*u_E\in H^r(B;R).
\]
\end{definition}

\section{The obstruction theorem}
\begin{theorem}[Nowhere-zero section obstruction]
\label{thm:viii32-euler-obstruction}
If an oriented vector bundle admits a nowhere-zero section, then
\[
e(E)=0.
\]
\end{theorem}
The converse is an obstruction-theoretic statement that requires appropriate
hypotheses; the Euler class is the primary obstruction, not in every setting a
complete classifier of sections.

\section{Naturality}
For $f:Y\to B$,
\[
e(f^*E)=f^*e(E).
\]

\section{Whitney sums}
For oriented bundles,
\[
e(E\oplus F)=e(E)\smile e(F).
\]
In particular, adding a positive-rank trivial summand forces the Euler class to
vanish.

\section{Sphere bundles and the Gysin sequence}
For an oriented rank-$r$ bundle, the sphere bundle $S(E)\to B$ has a long exact
Gysin sequence.  The characteristic map in the base is cup product with
$e(E)$:
\[
\cdots\to H^{k-r}(B)\xrightarrow{\smile e(E)}H^k(B)
\to H^k(S(E))\to H^{k-r+1}(B)\to\cdots.
\]

\section{Tangent bundles}
For a closed oriented $n$-manifold,
\[
e(TM)\in H^n(M).
\]
Its evaluation on the fundamental class is the Euler number.

\begin{theorem}[Euler class and Euler characteristic]
\label{thm:viii32-euler-characteristic}
For a closed oriented manifold $M$,
\[
\langle e(TM),[M]\rangle=\chi(M).
\]
\end{theorem}

\section{The hairy-ball theorem}
Since $\chi(S^{2m})=2$, the tangent Euler class of an even-dimensional sphere
is nonzero.  Thus $S^{2m}$ admits no nowhere-zero tangent vector field.

\section{Odd-dimensional manifolds}
A closed oriented odd-dimensional manifold has Euler characteristic zero.  In
particular, the tangent Euler number vanishes.

\section{Zeros of transverse sections}
For an oriented rank-$r$ bundle over an oriented $r$-manifold, a generic
section has isolated transverse zeros.  Each zero has a local sign, and the
signed sum equals the Euler number.

\section{Poincaré--Hopf viewpoint}
For $E=TM$, the preceding statement becomes the Poincaré--Hopf theorem: the
sum of indices of isolated zeros of a vector field equals $\chi(M)$.

\section{Zero loci as dual cycles}
More generally, a transverse section of an oriented rank-$r$ bundle has a
codimension-$r$ zero set whose fundamental class is Poincaré dual to $e(E)$.
This is the direct bridge to VIII/33.

\section{Solved dossiers}
\begin{problem}
Show that a nowhere-zero section of $E$ is equivalent to a section of $S(E)$ after choosing a metric.
\end{problem}
\begin{solution}
Normalize the vector field fiberwise; conversely include the unit sphere bundle in $E$.
\end{solution}

\begin{problem}
Define the Euler class from the Thom class.
\end{problem}
\begin{solution}
For an oriented rank-$r$ bundle, $e(E)=s^*u_E\in H^r(B)$.
\end{solution}

\begin{problem}
Why does a nowhere-zero section force $e(E)=0$?
\end{problem}
\begin{solution}
Move the zero section through a homotopy to a disjoint section; the pulled-back Thom class vanishes.
\end{solution}

\begin{problem}
Compute the Euler class of a trivial positive-rank oriented bundle.
\end{problem}
\begin{solution}
It is zero because a constant nowhere-zero section exists.
\end{solution}

\begin{problem}
What is the Euler class of an oriented line bundle with a nowhere-zero section?
\end{problem}
\begin{solution}
Zero.
\end{solution}

\begin{problem}
State multiplicativity for Whitney sums.
\end{problem}
\begin{solution}
$e(E\oplus F)=e(E)\smile e(F)$ for oriented bundles.
\end{solution}

\begin{problem}
Relate $e(TM)$ to the Euler characteristic.
\end{problem}
\begin{solution}
For a closed oriented manifold, $\langle e(TM),[M]\rangle=\chi(M)$.
\end{solution}

\begin{problem}
Evaluate the Euler number of $TS^2$.
\end{problem}
\begin{solution}
It equals $\chi(S^2)=2$.
\end{solution}

\begin{problem}
What happens for a closed oriented odd-dimensional manifold?
\end{problem}
\begin{solution}
Its Euler characteristic, hence tangent Euler number, is zero.
\end{solution}

\begin{problem}
Why does $S^{2m}$ have no nowhere-zero tangent vector field?
\end{problem}
\begin{solution}
Its Euler class pairs to $2\ne0$.
\end{solution}

\begin{problem}
Why is an Euler class an obstruction rather than always a complete classifier?
\end{problem}
\begin{solution}
Vanishing is necessary for a nonzero section, but further obstructions can exist outside the standard obstruction-theoretic range.
\end{solution}

\begin{problem}
Where does the Euler class appear in the Gysin sequence?
\end{problem}
\begin{solution}
It is the class whose cup-product map links adjacent base-cohomology terms.
\end{solution}

\begin{problem}
Describe the sphere bundle of a rank-$r$ bundle.
\end{problem}
\begin{solution}
It has fiber $S^{r-1}$ and records nonzero directions in each vector fiber.
\end{solution}

\begin{problem}
What is the geometric meaning of a zero of a transverse section?
\end{problem}
\begin{solution}
It contributes a local index; the signed sum represents the Euler number.
\end{solution}

\begin{problem}
State the Poincaré--Hopf connection.
\end{problem}
\begin{solution}
For an oriented closed manifold, the sum of vector-field indices equals $\chi(M)$.
\end{solution}

\begin{problem}
What coefficient system avoids orientability issues?
\end{problem}
\begin{solution}
Mod-$2$ coefficients.
\end{solution}

\begin{problem}
How does pullback affect Euler classes?
\end{problem}
\begin{solution}
$e(f^*E)=f^*e(E)$.
\end{solution}

\begin{problem}
What is the Euler class of $E\oplus\varepsilon^1$?
\end{problem}
\begin{solution}
Zero, since the trivial summand gives a nowhere-zero section; equivalently multiplicativity includes $e(\varepsilon^1)=0$.
\end{solution}

\begin{problem}
How does VIII/32 lead to duality?
\end{problem}
\begin{solution}
The zero locus of a transverse section represents the Poincaré dual of the Euler class.
\end{solution}

\begin{problem}
Why are sphere bundles visually natural here?
\end{problem}
\begin{solution}
A global section is literally a continuous choice of one point on every sphere fiber.
\end{solution}

\section{Exercises, Hints, and Solutions}
\begin{exercise}\label{exr:viii32-01}
Construct the unit sphere bundle of a trivial rank-two bundle over $S^1$.
\end{exercise}
\begin{hint}
Normalize fiberwise.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-01 append
Use the product Euclidean metric and impose unit norm in each \(\mathbb R^2\) fiber; identify both the total sphere-bundle space and its projection.
\end{hint}
\begin{solution}
For \(E=S^1\times\mathbb R^2\), the unit vectors in each fiber form \(S^1\), so
\[
S(E)\cong S^1\times S^1\to S^1.
\]
It is the trivial circle bundle.
\end{solution}

\begin{exercise}\label{exr:viii32-02}
Show that a nonzero section normalizes to a sphere-bundle section.
\end{exercise}
\begin{hint}
% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-02 replace_misdirected_clue
Define \(\widehat s(b)=s(b)/\|s(b)\|\). Check that the denominator is continuous and never zero, and that the normalized vector remains over \(b\).
\end{hint}
\begin{solution}
If \(s(b)\ne0\) for all \(b\), define \(\widehat s(b)=s(b)/\|s(b)\|\). Continuity of the norm gives a continuous section \(\widehat s:B\to S(E)\). Conversely, including \(S(E)\subset E\) turns every sphere-bundle section into a nowhere-zero section.
\end{solution}

\begin{exercise}\label{exr:viii32-03}
Prove naturality $e(f^*E)=f^*e(E)$.
\end{exercise}
\begin{hint}
Apply Thom naturality.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-03 append
Draw the square formed by the bundle map and the two zero sections, then combine its commuting pullbacks with naturality of the oriented Thom class.
\end{hint}
\begin{solution}
Thom naturality gives \(u_{f^*E}=\widetilde f^{\,*}u_E\). Pulling back along the zero sections yields
\[
e(f^*E)=s_Y^*u_{f^*E}=s_Y^*\widetilde f^{\,*}u_E=f^*s_B^*u_E=f^*e(E).
\]
\end{solution}

\begin{exercise}\label{exr:viii32-04}
Compute the Euler class of a trivial rank-three bundle.
\end{exercise}
\begin{hint}
Use a constant vector.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-04 append
A fixed nonzero vector gives a section in every product fiber; apply the implication from existence of such a section to vanishing Euler class.
\end{hint}
\begin{solution}
A trivial rank-three bundle admits a constant nowhere-zero section. Therefore its Euler class is zero:
\[
e(B\times\mathbb R^3)=0\in H^3(B).
\]
\end{solution}

\begin{exercise}\label{exr:viii32-05}
Use multiplicativity to compute $e(E\oplus\varepsilon^1)$.
\end{exercise}
\begin{hint}
Use multiplicativity.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-05 append
Use compatible orientations on the direct sum; one factor in the Euler cup product is the Euler class of the trivial line.
\end{hint}
\begin{solution}
By multiplicativity,
\[
e(E\oplus\varepsilon^1)=e(E)\smile e(\varepsilon^1).
\]
The trivial line has a nowhere-zero section, so \(e(\varepsilon^1)=0\), and the product vanishes.
\end{solution}

\begin{exercise}\label{exr:viii32-06}
Show that $e(TS^2)$ pairs to $2$.
\end{exercise}
\begin{hint}
Pair with the fundamental class.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-06 append
Normalize the positive generator \(u\) by \(\langle u,[S^2]\rangle=1\); Poincare--Hopf then determines the coefficient of \(u\) in the tangent Euler class.
\end{hint}
\begin{solution}
For the oriented two-sphere,
\[
\langle e(TS^2),[S^2]\rangle=\chi(S^2)=2.
\]
Thus the Euler class is twice the preferred generator of \(H^2(S^2;\mathbb Z)\).
\end{solution}

\begin{exercise}\label{exr:viii32-07}
Show that $TS^2$ has no nowhere-zero section.
\end{exercise}
\begin{hint}
Use nonvanishing of the pairing.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-07 append
Use the contrapositive of the necessary Euler-class condition; a nonzero evaluation proves the cohomology class itself cannot vanish.
\end{hint}
\begin{solution}
If \(TS^2\) had a nowhere-zero section, the Euler class would vanish. But its evaluation on \([S^2]\) is \(2\), so \(e(TS^2)\ne0\). Hence no nowhere-zero tangent vector field exists.
\end{solution}

\begin{exercise}\label{exr:viii32-08}
Compute the Euler number of $TS^{2m}$.
\end{exercise}
\begin{hint}
Use $\chi(S^{2m})=2$.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-08 append
Apply the tangent Euler-number theorem on the closed oriented even sphere and compute its Euler characteristic from its two-cell CW structure.
\end{hint}
\begin{solution}
For every \(m\ge1\),
\[
\langle e(TS^{2m}),[S^{2m}]\rangle=\chi(S^{2m})=2.
\]
So the tangent Euler number of every even-dimensional sphere is \(2\).
\end{solution}

\begin{exercise}\label{exr:viii32-09}
Explain why the Euler number of $TS^{2m+1}$ vanishes.
\end{exercise}
\begin{hint}
Use $\chi(S^{2m+1})=0$.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-09 append
Besides the Euler calculation, view the odd sphere inside \(\mathbb C^{m+1}\); multiplication by \(i\) gives a tangent vector perpendicular to the radial vector and never zero.
\end{hint}
\begin{solution}
For \(S^{2m+1}\),
\[
\langle e(TS^{2m+1}),[S^{2m+1}]\rangle=\chi(S^{2m+1})=0.
\]
Indeed odd spheres admit explicit nowhere-zero tangent vector fields.
\end{solution}

\begin{exercise}\label{exr:viii32-10}
Relate the hairy-ball theorem to the Euler class.
\end{exercise}
\begin{hint}
Translate vector fields into sections.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-10 append
A tangent vector field is a section of \(TS^{2m}\); its normalization would produce a section of the unit tangent sphere bundle, contradicting the nonzero Euler class.
\end{hint}
\begin{solution}
A tangent vector field is a section of \(TM\), and after normalization a nowhere-zero vector field is a section of \(S(TM)\). Since \(e(TS^{2m})\ne0\), the Euler obstruction rules out such a section. This is the hairy-ball theorem.
\end{solution}

\begin{exercise}\label{exr:viii32-11}
Write the relevant fragment of the Gysin sequence for an oriented circle bundle.
\end{exercise}
\begin{hint}
Take fiber $S^1$.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-11 append
The sphere fiber has dimension one, but the vector-bundle rank is two; use rank two for the cup-with-Euler degree shift.
\end{hint}
\begin{solution}
For an oriented rank-two bundle \(E\to B\), the sphere fiber is \(S^1\), and the Gysin sequence contains
\[
\cdots\to H^{k-2}(B)\xrightarrow{\smile e(E)}H^k(B)
\to H^k(S(E))\to H^{k-1}(B)\xrightarrow{\smile e(E)}H^{k+1}(B)\to\cdots.
\]
\end{solution}

\begin{exercise}\label{exr:viii32-12}
Identify the cup-with-Euler-class map in that sequence.
\end{exercise}
\begin{hint}
Look for cup product by $e(E)$.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-12 append
Identify the map as \(H^{k-r}(B)\to H^k(B)\) and check that multiplication by the degree-\(r\) class lands in the stated target.
\end{hint}
\begin{solution}
In the Gysin sequence the characteristic map from base cohomology to base cohomology is precisely
\[
\alpha\longmapsto \alpha\smile e(E).
\]
Its degree shift equals the rank of the oriented vector bundle.
\end{solution}

\begin{exercise}\label{exr:viii32-13}
Describe zeros of a generic section of an oriented rank-$r$ bundle over an $r$-manifold.
\end{exercise}
\begin{hint}
Use transversality.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-13 append
Transversality subtracts the fiber rank from the base dimension. Compactness turns the resulting closed zero-dimensional zero set into a finite set.
\end{hint}
\begin{solution}
If \(\operatorname{rank}E=r=\dim B\), transversality to the zero section makes the zero set of a generic section a submanifold of dimension \(r-r=0\). On a compact base it is therefore a finite set of isolated zeros.
\end{solution}

\begin{exercise}\label{exr:viii32-14}
Assign signs to isolated transverse zeros.
\end{exercise}
\begin{hint}
Compare orientations of base and fiber.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-14 append
At a zero the vertical derivative is intrinsically \(T_xB\to E_x\); compare its determinant sign using the specified orientations.
\end{hint}
\begin{solution}
At an isolated transverse zero \(x\), trivialize \(E\) and orient both base and fiber. The derivative of the local section is an isomorphism \(T_xB\to E_x\). Assign sign \(+1\) if it preserves orientation and \(-1\) if it reverses orientation.
\end{solution}

\begin{exercise}\label{exr:viii32-15}
Relate their signed sum to the Euler number.
\end{exercise}
\begin{hint}
Sum local degrees.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-15 append
For a closed oriented base of the same dimension as the rank, sum the local degrees and compare them with evaluation on its fundamental class.
\end{hint}
\begin{solution}
The local signs are local degrees. Their sum equals the Euler number:
\[
\sum_{x:s(x)=0}\operatorname{ind}_x(s)
=\langle e(E),[B]\rangle.
\]
For \(E=TB\) this is Poincaré--Hopf.
\end{solution}

\begin{exercise}\label{exr:viii32-16}
Explain the mod-$2$ version without orientation.
\end{exercise}
\begin{hint}
Ignore signs modulo two.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-16 append
Count transverse zeros modulo two and use the mod-two fundamental class; no sign comparison of the base and fiber is then needed.
\end{hint}
\begin{solution}
With \(\mathbb Z/2\) coefficients, the signs \(+1\) and \(-1\) coincide. Thus the zero locus is counted modulo two, and the mod-two Thom/Euler obstruction is defined without orienting the bundle.
\end{solution}

\begin{exercise}\label{exr:viii32-17}
Compare Euler classes before and after pullback to a point.
\end{exercise}
\begin{hint}
Pullback to the fiber over the chosen point.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-17 append
Apply naturality to the inclusion of the chosen point and inspect its positive-degree cohomology; retaining the positive-rank assumption is essential.
\end{hint}
\begin{solution}
Pulling a positive-rank bundle back to a point gives a vector space over a point. Since \(H^r(\{\ast\})=0\) for \(r>0\),
\[
e(i^*E)=i^*e(E)=0.
\]
This is consistent with every vector space admitting nonzero vectors.
\end{solution}

\begin{exercise}\label{exr:viii32-18}
Analyze a product bundle over $S^2\times S^2$.
\end{exercise}
\begin{hint}
Use Künneth and naturality.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-18 append
Use the bundle chosen in the solution, \(p_1^*TS^2\oplus p_2^*TS^2\), not a trivial product bundle. Apply naturality and multiply the two pulled-back Euler classes.
\end{hint}
\begin{solution}
For example, on \(S^2\times S^2\) let
\[
E=p_1^*TS^2\oplus p_2^*TS^2.
\]
Then
\[
e(E)=p_1^*e(TS^2)\smile p_2^*e(TS^2),
\]
and its evaluation on \([S^2\times S^2]\) is \(2\cdot2=4\).
\end{solution}

\begin{exercise}\label{exr:viii32-19}
Use Whitney-sum multiplicativity in a tangent-bundle example.
\end{exercise}
\begin{hint}
Split the tangent bundle if possible.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-19 append
Use \(T(M\times N)\cong p_M^*TM\oplus p_N^*TN\) with the product orientation; for Euler numbers assume both manifolds are closed and oriented.
\end{hint}
\begin{solution}
Since \(T(M\times N)\cong p_M^*TM\oplus p_N^*TN\),
\[
e(T(M\times N))=p_M^*e(TM)\smile p_N^*e(TN).
\]
Evaluating gives \(\chi(M\times N)=\chi(M)\chi(N)\), illustrating Whitney-sum multiplicativity.
\end{solution}

\begin{exercise}\label{exr:viii32-20}
Explain why vanishing Euler class need not classify a bundle.
\end{exercise}
\begin{hint}
Remember that characteristic classes are generally not complete invariants.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-20 append
Separate bundle classification from section existence: Euler data alone need not distinguish bundles, and a vanishing primary obstruction may leave higher section obstructions.
\end{hint}
\begin{solution}
The Euler class is only one characteristic class. In higher rank or on complicated bases, two nonisomorphic bundles can share the same Euler class but differ in other characteristic data such as Stiefel--Whitney or Pontryagin classes. Therefore \(e(E)=0\) alone is not a complete classification.
\end{solution}

\begin{exercise}\label{exr:viii32-21}
Describe the first obstruction to a section of $S(E)$.
\end{exercise}
\begin{hint}
Use obstruction theory on the sphere fiber.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-21 append
For oriented rank \(r\ge2\), start with the \((r-2)\)-connected sphere fiber and locate the first extension obstruction in degree \(r\); higher-dimensional bases may have further obstructions.
\end{hint}
\begin{solution}
For an oriented rank-\(r\) bundle, the fiber of \(S(E)\) is \(S^{r-1}\). The primary obstruction to extending a section lies in
\[
H^r(B;\pi_{r-1}(S^{r-1}))\cong H^r(B;\mathbb Z)
\]
(with the appropriate orientation local system), and this obstruction is the Euler class.
\end{solution}

\begin{exercise}\label{exr:viii32-22}
Relate a transverse zero set to a Thom class.
\end{exercise}
\begin{hint}
Pull back the Thom class by the section.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-22 append
Pull back the Thom class as a relative class supported near the section's zero set, then pass to ordinary base cohomology to obtain its Euler class.
\end{hint}
\begin{solution}
A transverse section \(s\) meets the zero section cleanly. Pulling the normal Thom class through this intersection identifies the cohomology class detecting the zero set with \(s^*u_E=e(E)\). Thus the Thom class localizes the zero locus.
\end{solution}

\begin{exercise}\label{exr:viii32-23}
Explain how the zero set becomes a dual homology class.
\end{exercise}
\begin{hint}
Apply Poincaré duality in the base.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-23 append
On a closed oriented \(n\)-manifold, transversality identifies the normal bundle of the zero locus with \(E|_Z\); cap the Euler class with \([B]\) to obtain its \((n-r)\)-class.
\end{hint}
\begin{solution}
If \(Z=s^{-1}(0)\) is transverse, then \(Z\) has codimension \(r\) and its normal bundle is naturally \(E|_Z\). The Thom construction therefore yields
\[
\operatorname{PD}[Z]=e(E).
\]
\end{solution}

\begin{exercise}\label{exr:viii32-24}
Summarize sphere bundle $\to$ section obstruction $\to$ Euler class $\to$ duality.
\end{exercise}
\begin{hint}
Track geometry and cohomology simultaneously.

% VIII-PEDAGOGY-HINT-v1.1 exr:viii32-24 append
Keep the logical arrows separate: a sphere-bundle section is equivalent to a nowhere-zero section and forces \(e(E)=0\); the reverse implication needs additional obstruction-theoretic hypotheses.
\end{hint}
\begin{solution}
The geometry and algebra line up as
\[
\text{section of }S(E)
\Longleftrightarrow
\text{nowhere-zero section of }E,
\]
while failure of such a section is detected first by \(e(E)\). For a transverse section, its zero set represents \(\operatorname{PD}(e(E))\), leading directly to Poincaré duality.
\end{solution}
