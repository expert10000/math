% Robust Poincaré disk geodesic between boundary angles a,b (degrees).
% Handles the diameter case (antipodal endpoints) where det=0.
% Robust Poincaré disk geodesic between boundary angles a,b (degrees).
\newcommand{\DiskGeodesic}[3][]{%
	\pgfmathsetmacro{\ax}{cos(#2)}%
	\pgfmathsetmacro{\ay}{sin(#2)}%
	\pgfmathsetmacro{\bx}{cos(#3)}%
	\pgfmathsetmacro{\by}{sin(#3)}%
	
	% det = ax*by - ay*bx
	\pgfmathsetmacro{\det}{\ax*\by - \ay*\bx}%
	\pgfmathsetmacro{\detabs}{abs(\det)}%
	
	% diameter / antipodal case: det ~ 0
	\ifdim \detabs pt < 0.000001pt
	\draw[#1] (\ax,\ay) -- (\bx,\by);
	\else
	\pgfmathsetmacro{\h}{(\by-\ay)/\det}%
	\pgfmathsetmacro{\k}{(\ax-\bx)/\det}%
	\pgfmathsetmacro{\Rgeo}{sqrt(\h*\h+\k*\k-1)}%
	\pgfmathsetmacro{\tA}{atan2(\ay-\k,\ax-\h)}%
	\pgfmathsetmacro{\tB}{atan2(\by-\k,\bx-\h)}%
	\pgfmathsetmacro{\dAB}{\tB-\tA}%
	\pgfmathsetmacro{\dAB}{ifthenelse(\dAB>180,\dAB-360,\dAB)}%
	\pgfmathsetmacro{\dAB}{ifthenelse(\dAB<-180,\dAB+360,\dAB)}%
	\pgfmathsetmacro{\tEnd}{\tA+\dAB}%
	\draw[#1] (\h,\k) ++(\tA:\Rgeo)
	arc[start angle=\tA, end angle=\tEnd, radius=\Rgeo];
	\fi
}

\subsubsection*{1) Warm-up: base $D^2$ (contractible) $\Rightarrow$ everything is trivial, parallel transport can be flat}

Let
\[
M = D^2 \subset \mathbb{R}^2.
\]

\paragraph{(a) Trivial vector bundle and sections.}
Take
\[
E = D^2 \times \mathbb{R}^2,
\qquad
\pi(x,v)=x.
\]
A section is just a map
\[
s(x)=(x,\,V(x)),
\]
i.e.\ a vector field
\[
V:D^2\to \mathbb{R}^2.
\]

\paragraph{(b) Frame bundle (principal $\mathrm{GL}(2)$).}
The frame bundle
\[
F(E)\to D^2
\]
is also trivial:
\[
F(E)\cong D^2\times \mathrm{GL}(2,\mathbb{R}),
\]
and a (global) section is a global frame---e.g.\ the constant frame
\[
(e_1,e_2).
\]

\paragraph{(c) A flat connection and parallel transport.}
Take the flat connection (think: the usual derivative in $\mathbb{R}^2$).
Its curvature is
\[
0,
\]
so parallel transport along any loop is the identity (path-independent on simply-connected sets).

\paragraph{Moral.}
Over
\[
D^2
\]
you can always pick global sections/frames and choose flat connections.


\subsubsection*{2) Base $S^2$ (curved) $\Rightarrow$ tangent bundle is nontrivial, Levi--Civita transport has nontrivial holonomy}

Let
\[
M=S^2\subset \mathbb{R}^3
\]
with the round metric (unit sphere).

\paragraph{(a) Tangent bundle and sections.}
Take the tangent bundle
\[
\pi:TS^2\to S^2,
\qquad
\pi(p,v)=p,
\]
whose fiber at $p$ is the tangent plane
\[
(TS^2)_p = T_pS^2.
\]
A section
\[
s:S^2\to TS^2
\]
is a tangent vector field:
\[
s(p)=(p,\,V(p)),\qquad V(p)\in T_pS^2.
\]
(There is no nowhere-vanishing global tangent field on $S^2$, hence no global frame of $TS^2$.)

\paragraph{(b) Orthonormal frame bundle (principal $\mathrm{SO}(2)$).}
Define the oriented orthonormal frame bundle
\[
P := F_{\mathrm{SO}}^+(S^2)
= \bigl\{(p,e_1,e_2): p\in S^2,\ (e_1,e_2)\ \text{oriented orthonormal basis of }T_pS^2\bigr\}.
\]
The projection
\[
\pi:P\to S^2,\qquad \pi(p,e_1,e_2)=p
\]
makes $P$ a principal $\mathrm{SO}(2)$-bundle, with right action given by rotating the frame:
\[
(p,e_1,e_2)\cdot R
=
(p,\,e_1',e_2'),
\qquad
R\in \mathrm{SO}(2).
\]

\paragraph{(c) Levi--Civita connection as a principal connection.}
The Levi--Civita connection on $TS^2$ induces a principal connection on $P$:
equivalently, a choice of horizontal subspaces
\[
T_uP = H_u \oplus V_u
\qquad (u\in P),
\]
where $V_u$ are the vertical (fiber) directions.

In spherical coordinates $(\theta,\phi)$ on $S^2$, consider the orthonormal tangent frame
\[
e_\theta = \partial_\theta,
\qquad
e_\phi = \frac{1}{\sin\theta}\,\partial_\phi.
\]
Then the Levi--Civita connection can be encoded by a single $\mathfrak{so}(2)$-valued $1$-form
\[
\omega = \cos\theta\, d\phi
\]
(up to a sign convention), describing how the frame rotates along motion on the sphere.

\paragraph{(d) Parallel transport and holonomy (``area gives rotation'').}
Given a curve
\[
\gamma:[0,1]\to S^2,
\]
parallel transport for Levi--Civita means: a vector field $V(t)\in T_{\gamma(t)}S^2$ satisfying
\[
\nabla_{\dot\gamma(t)}V(t)=0.
\]
For a closed loop $\gamma=\partial\Omega$, parallel transport typically returns a vector rotated by a holonomy angle.
On the unit sphere (Gaussian curvature $K\equiv 1$), for sufficiently nice loops,
\[
\text{holonomy angle} = \int_{\Omega} K\, dA = \mathrm{Area}(\Omega).
\]

\emph{Concrete triangle example.}
Let $\Omega$ be the spherical triangle with vertices:
north pole $N$ and two points $A,B$ on the equator whose longitudes differ by $\alpha$,
with edges: meridian $N\to A$, equator $A\to B$, meridian $B\to N$.
Then the enclosed area is
\[
\mathrm{Area}(\Omega)=\alpha
\]
(unit sphere), hence parallel transport around the triangle rotates tangent vectors by angle
\[
\alpha.
\]

\paragraph{Moral.}
Over
\[
S^2
\]
the tangent bundle and frame bundle are globally nontrivial, and Levi--Civita parallel transport around loops produces nontrivial holonomy determined by curvature (on the unit sphere: by area).


\subsubsection*{3) Hopf fibration: a nontrivial principal $S^1$-bundle over $S^2$ with an explicit connection}

Let
\[
S^3=\{(z_1,z_2)\in \mathbb{C}^2:\ |z_1|^2+|z_2|^2=1\}.
\]

\paragraph{(a) Principal $S^1$-bundle structure.}
Let $S^1$ act on $S^3$ by
\[
e^{i\psi}\cdot (z_1,z_2) = (e^{i\psi}z_1,\ e^{i\psi}z_2).
\]
The quotient space is
\[
S^3/S^1 \cong \mathbb{C}P^1 \cong S^2,
\]
so we get a principal bundle
\[
S^1 \hookrightarrow S^3 \xrightarrow{\,p\,} S^2,
\]
called the Hopf fibration.

A concrete Hopf map into $\mathbb{R}^3$ (identifying $S^2\subset \mathbb{R}^3$) is
\[
p(z_1,z_2)=
\Bigl(
2\Re(z_1\overline{z_2}),\ 
2\Im(z_1\overline{z_2}),\ 
|z_1|^2-|z_2|^2
\Bigr),
\]
which indeed lies on $S^2$.

\paragraph{(b) Local sections (local triviality).}
Cover $S^2$ by two charts $U_+$ (remove the south pole) and $U_-$ (remove the north pole).
Over each chart, the bundle is trivial:
\[
p^{-1}(U_\pm)\cong U_\pm\times S^1,
\]
so one can choose local sections
\[
s_\pm:U_\pm\to S^3.
\]
On the overlap $U_+\cap U_-$ the sections differ by a transition function
\[
g_{+-}:U_+\cap U_-\to S^1,
\]
which encodes the global nontriviality of the bundle.

\paragraph{(c) An explicit connection $1$-form.}
There is a standard $S^1$-connection on the Hopf bundle.
In Euler-angle coordinates $(\theta,\phi,\psi)$ on $S^3$ adapted to the fibration, one can take
\[
\omega = d\psi + \cos\theta\, d\phi.
\]
Here:
\begin{itemize}
	\item $\psi$ is the fiber coordinate (along $S^1$),
	\item $(\theta,\phi)$ are coordinates on the base $S^2$.
\end{itemize}

Equivalently (in complex coordinates), one often writes the same connection as
\[
\omega = \Im\!\bigl(\overline{z_1}\,dz_1+\overline{z_2}\,dz_2\bigr),
\]
restricted to $S^3$.

\paragraph{(d) Curvature and holonomy.}
The curvature $2$-form is
\[
\Omega = d\omega,
\]
and (up to a conventional normalization factor) it pulls back the area form on $S^2$.
Hence the holonomy around a loop $\gamma=\partial\Omega$ in the base is governed by the enclosed area:
transport around $\gamma$ produces a phase in $S^1$ determined by integrating curvature over the spanned region.

\paragraph{Moral.}
The Hopf fibration is a clean, explicit example of:
a nontrivial principal bundle over $S^2$,
local sections but no global section compatible with trivialization,
and a canonical connection whose curvature detects area (and thus yields nontrivial holonomy).
