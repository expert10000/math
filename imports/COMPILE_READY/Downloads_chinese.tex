	\section*{1) Krótkie wiadomości na start (do czatu)}
	\begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=4pt]
		\item 你好！ — \textit{Nǐ hǎo!} — Cześć! / Hi!
		\item 嗨！ — \textit{Hāi!} — Hej! / Hey!
		\item 你怎么样？ — \textit{Nǐ zěnmeyàng?} — Jak się masz? / How are you?
		\item 最近怎么样？ — \textit{Zuìjìn zěnmeyàng?} — Co słychać ostatnio? / How have you been?
		\item 你在忙吗？ — \textit{Nǐ zài máng ma?} — Jesteś zajęty/a? / Are you busy?
		\item 现在方便聊吗？ — \textit{Xiànzài fāngbiàn liáo ma?} — Możesz teraz pogadać? / Is it convenient to chat now?
		\item 好久不见！ — \textit{Hǎojiǔ bú jiàn!} — Dawno się nie widzieliśmy! / Long time no see!
		\item 认识你很高兴！ — \textit{Rènshi nǐ hěn gāoxìng!} — Miło Cię poznać! / Nice to meet you!
		\item 我叫… 你呢？ — \textit{Wǒ jiào … nǐ ne?} — Mam na imię… a Ty? / My name is… and you?
		\item 你叫什么名字？ — \textit{Nǐ jiào shénme míngzi?} — Jak masz na imię? / What’s your name?
		\item 你是哪里人？ — \textit{Nǐ shì nǎlǐ rén?} — Skąd jesteś? / Where are you from?
		\item 谢谢你！ — \textit{Xièxie nǐ!} — Dzięki! / Thank you!
	\end{enumerate}
	
	\section*{2) Przydatne „słowa czatowe” (krótkie)}
	\begin{itemize}[leftmargin=*, itemsep=3pt]
		\item 好的 — \textit{Hǎo de} — OK / sure
		\item 可以 — \textit{Kěyǐ} — Można / can
		\item 不可以 — \textit{Bù kěyǐ} — Nie można / can’t
		\item 没问题 — \textit{Méi wèntí} — Nie ma problemu / no problem
		\item 等一下 — \textit{Děng yíxià} — Chwila / wait a sec
		\item 马上 — \textit{Mǎshàng} — Zaraz / right away
		\item 抱歉 — \textit{Bàoqiàn} — Przepraszam / sorry
		\item 对不起 — \textit{Duìbuqǐ} — Przepraszam / sorry
		\item 真的吗？ — \textit{Zhēn de ma?} — Serio? / really?
		\item 太好了！ — \textit{Tài hǎo le!} — Super! / great!
		\item 我懂了 — \textit{Wǒ dǒng le} — Rozumiem / got it
		\item 我不懂 — \textit{Wǒ bù dǒng} — Nie rozumiem / I don’t understand
		\item 再说一遍 — \textit{Zài shuō yí biàn} — Powtórz proszę / say again
		\item 慢一点 — \textit{Màn yìdiǎn} — Wolniej / slower
		\item 没关系 — \textit{Méi guānxi} — Nic nie szkodzi / it’s ok
	\end{itemize}
	
	\section*{3) Proste zdania „na codziennie”}
	\begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=4pt]
		\item 你今天怎么样？ — \textit{Nǐ jīntiān zěnmeyàng?} — Jak dziś? / How are you today?
		\item 你在做什么？ — \textit{Nǐ zài zuò shénme?} — Co robisz? / What are you doing?
		\item 我在学习中文。 — \textit{Wǒ zài xuéxí Zhōngwén.} — Uczę się chińskiego. / I’m learning Chinese.
		\item 我来自波兰。 — \textit{Wǒ láizì Bōlán.} — Jestem z Polski. / I’m from Poland.
		\item 你会说英语吗？ — \textit{Nǐ huì shuō Yīngyǔ ma?} — Mówisz po angielsku? / Do you speak English?
		\item 我会一点儿中文。 — \textit{Wǒ huì yìdiǎnr Zhōngwén.} — Trochę umiem chiński. / I speak a little Chinese.
		\item 我喜欢… — \textit{Wǒ xǐhuan …} — Lubię… / I like…
		\item 你喜欢什么？ — \textit{Nǐ xǐhuan shénme?} — Co lubisz? / What do you like?
		\item 我们可以做朋友吗？ — \textit{Wǒmen kěyǐ zuò péngyou ma?} — Możemy być znajomymi? / Can we be friends?
		\item 你有微信吗？ — \textit{Nǐ yǒu Wēixìn ma?} — Masz WeChat? / Do you have WeChat?
		\item 我可以加你微信吗？ — \textit{Wǒ kěyǐ jiā nǐ Wēixìn ma?} — Mogę dodać Cię na WeChat? / Can I add you on WeChat?
		\item 你现在有空吗？ — \textit{Nǐ xiànzài yǒu kòng ma?} — Masz teraz czas? / Are you free now?
		\item 我有点忙。 — \textit{Wǒ yǒudiǎn máng.} — Trochę jestem zajęty/a. / I’m a bit busy.
		\item 我们晚点聊。 — \textit{Wǒmen wǎndiǎn liáo.} — Pogadajmy później. / Let’s chat later.
		\item 好的，没问题。 — \textit{Hǎo de, méi wèntí.} — OK, nie ma problemu. / Okay, no problem.
		\item 明天见！ — \textit{Míngtiān jiàn!} — Do jutra! / See you tomorrow!
		\item 晚安！ — \textit{Wǎn’ān!} — Dobranoc! / Good night!
		\item 祝你今天开心！ — \textit{Zhù nǐ jīntiān kāixīn!} — Miłego dnia! / Have a nice day!
		\item 谢谢你的帮助。 — \textit{Xièxie nǐ de bāngzhù.} — Dzięki za pomoc. / Thanks for your help.
		\item 不客气。 — \textit{Bú kèqi.} — Nie ma za co. / You’re welcome.
	\end{enumerate}
