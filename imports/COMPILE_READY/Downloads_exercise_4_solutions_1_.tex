\section*{Definitions and Examples}

\paragraph{Supremum and Infimum.}
For a set $S \subset \mathbb{R}$:
\begin{itemize}
  \item The \emph{supremum} (least upper bound) is the smallest $M \in \mathbb{R}$ such that $x \leq M$ for all $x \in S$. Denoted $\sup S$.
  \item The \emph{infimum} (greatest lower bound) is the largest $m \in \mathbb{R}$ such that $m \leq x$ for all $x \in S$. Denoted $\inf S$.
\end{itemize}
\textbf{Example.} For $S=(0,1)$, $\sup S = 1$, $\inf S = 0$.

\paragraph{Partitions, Upper and Lower Sums.}
A \emph{partition} of $[a,b]$ is a finite set
\[
P = \{x_0, x_1, \dots, x_n\}, \quad a = x_0 < x_1 < \cdots < x_n = b.
\]
For a bounded function $f:[a,b]\to\mathbb{R}$:
\[
U(f,P) = \sum_{i=1}^n M_i (x_i - x_{i-1}), \quad M_i = \sup\{ f(x) : x \in [x_{i-1}, x_i]\},
\]
\[
L(f,P) = \sum_{i=1}^n m_i (x_i - x_{i-1}), \quad m_i = \inf\{ f(x) : x \in [x_{i-1}, x_i]\}.
\]
\textbf{Example.} For $f(x) = x$ on $[0,1]$ with $P=\{0,0.5,1\}$:
\[
U(f,P) = 0.75, \quad L(f,P) = 0.25.
\]

\paragraph{Integrability.}
A bounded $f:[a,b]\to\mathbb{R}$ is \emph{Riemann integrable} if
\[
\sup_P L(f,P) = \inf_P U(f,P).
\]
\textbf{Example.} $f(x)=x$ on $[0,1]$ is integrable with $\int_0^1 x dx = \tfrac{1}{2}$.

\paragraph{Characteristic Function.}
For $A \subset \mathbb{R}$, the \emph{characteristic function} is
\[
\chi_A(x) = \begin{cases}
1 & x \in A, \\
0 & x \notin A.
\end{cases}
\]
\textbf{Example.} If $A=[0,\tfrac{1}{2}]$, then $\int_0^1 \chi_A(x) dx = \tfrac{1}{2}$.

\section*{Solutions}

\subsection*{Exercise 4.1}
\begin{solution}
\begin{enumerate}[label=\alph*)]
\item $\sup_{x\in A}(-f(x)) = -\inf_{x\in A} f(x)$.
\item $\inf_{x\in A}(-f(x)) = -\sup_{x\in A} f(x)$.
\item $\sup_{x\in A}\lambda f(x) = \lambda \sup_{x\in A} f(x)$ for $\lambda \geq 0$.
\item $\sup_{x\in A}(f(x)+g(x)) \leq \sup f + \sup g$.
\item $\inf_{x\in A}(f(x)+g(x)) \geq \inf f + \inf g$.
\item $\inf_{x\in A}\lambda f(x) = \lambda \inf f(x)$.
\item $|\sup f| \leq \sup |f|$.
\item $|\inf f| \leq \sup |f|$.
\end{enumerate}
\end{solution}

\subsection*{Exercise 4.2}
\begin{solution}
\begin{enumerate}[label=\alph*)]
\item $U(-f,P) = -L(f,P)$.
\item $L(-f,P) = -U(f,P)$.
\item $U(\lambda f,P) = \lambda U(f,P)$ for $\lambda \geq 0$.
\item $L(\lambda f,P) = \lambda L(f,P)$.
\item $U(f+g,P) \leq U(f,P) + U(g,P)$.
\item $L(f+g,P) \geq L(f,P) + L(g,P)$.
\end{enumerate}
\end{solution}

\subsection*{Exercise 4.3}
\begin{solution}
If $P \preceq Q$, then $U(f,Q) \leq U(f,P)$ and $L(f,Q) \geq L(f,P)$. Hence
\[
0 \leq U(f,Q)-L(f,Q) \leq U(f,P)-L(f,P).
\]
\end{solution}

\subsection*{Exercise 4.4}
\begin{solution}
The function $\chi_A$ is Riemann integrable since it has only finitely many discontinuities. Moreover,
\[
\int_{-1}^1 \chi_A(x)\, dx = 0,
\]
because a finite set has measure zero.
\end{solution}
