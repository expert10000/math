\section{Gauss' Divergence Theorem --- Sources and Closed Flux}
\label{sec:gauss-theorem}

Let $V$ be a volume with outward-oriented closed boundary $\partial V$. Then
\[
\boxed{\iint_{\partial V}\mathbf F\cdot d\mathbf A
=\iiint_V\nabla\cdot\mathbf F\,dV.}
\]
The net flux through the boundary equals the total source strength inside.

\begin{workedexample}
\textbf{Flux of a radial expansion field.}\par
For $\mathbf F=(x,y,z)$ inside a sphere of radius $R$,
\[
\nabla\cdot\mathbf F=3.
\]
Therefore
\[
\iint_{\partial V}\mathbf F\cdot d\mathbf A
=\iiint_V3\,dV=3\left(\frac{4\pi R^3}{3}\right)=4\pi R^3.
\]
On the sphere, $\mathbf F=R\hat{\mathbf n}$, so direct integration gives the same result.
\end{workedexample}

\subsection{Singular sources}
The inverse-square field $\mathbf F=\mathbf r/r^3$ has zero divergence wherever $r\neq0$, yet its flux through any sphere centered at the origin is $4\pi$. The origin is a singularity and cannot be ignored. In distributional language,
\[
\nabla\cdot\frac{\mathbf r}{r^3}=4\pi\delta^{(3)}(\mathbf r).
\]

\begin{keyidea}[Connection to electrostatics]
For $\nabla\cdot\mathbf E=\rho/\varepsilon_0$, the divergence theorem yields
\[
\iint_{\partial V}\mathbf E\cdot d\mathbf A
=\frac{1}{\varepsilon_0}\iiint_V\rho\,dV
=\frac{Q_{\mathrm{enc}}}{\varepsilon_0}.
\]
The differential and integral forms of Gauss' law are therefore the same statement at different scales.
\end{keyidea}
