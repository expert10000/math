\section{Thirty Integrated Worked Examples}
\label{sec:ch5-worked-examples}

The examples below develop a working language for complex arithmetic, plane mappings, analyticity, and elementary contour integration. Each solution is paired with an interpretation so that symbolic manipulation remains connected to geometry and physics.

\subsection{Arithmetic, polar form, and roots}

\begin{workedexample}
\textbf{Cartesian arithmetic.}\par
\textbf{Problem.} Compute $(3-2i)+(1+5i)$ and $(3-2i)(1+5i)$.
\begin{solution}
Addition gives $4+3i$. Multiplication gives $3+15i-2i-10i^2=13+13i$.

\textbf{Interpretation.} Addition combines displacement vectors; multiplication mixes scaling and rotation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Division by conjugation.}\par
\textbf{Problem.} Simplify $(2+3i)/(1-i)$.
\begin{solution}
Multiply numerator and denominator by $1+i$:
\[
\frac{2+3i}{1-i}=\frac{(2+3i)(1+i)}{2}=\frac{-1+5i}{2}.
\]

\textbf{Interpretation.} Conjugation converts a complex denominator into the real number $|1-i|^2$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Modulus and argument.}\par
\textbf{Problem.} Write $z=-1+\sqrt3 i$ in polar form.
\begin{solution}
$r=2$ and the point lies in quadrant II, so $\theta=2\pi/3$. Hence $z=2e^{2\pi i/3}$.

\textbf{Interpretation.} Cartesian coordinates locate the point; polar data separate distance from direction.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{A product in polar form.}\par
\textbf{Problem.} Multiply $z_1=2e^{i\pi/6}$ and $z_2=3e^{-i\pi/4}$.
\begin{solution}
\[
z_1z_2=6e^{i(\pi/6-\pi/4)}=6e^{-i\pi/12}.
\]

\textbf{Interpretation.} Moduli multiply while arguments add.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{De Moivre's theorem.}\par
\textbf{Problem.} Evaluate $(\cos\theta+i\sin\theta)^5$.
\begin{solution}
By de Moivre's theorem,
\[
(\cos\theta+i\sin\theta)^5=\cos5\theta+i\sin5\theta.
\]

\textbf{Interpretation.} Repeated multiplication performs repeated rotation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Cube roots of unity.}\par
\textbf{Problem.} Solve $z^3=1$.
\begin{solution}
\[
z_k=e^{2\pi i k/3},\qquad k=0,1,2.
\]
Thus $1$, $-\tfrac12+\tfrac{\sqrt3}{2}i$, and $-\tfrac12-\tfrac{\sqrt3}{2}i$ are the roots.

\textbf{Interpretation.} The solutions are equally spaced vertices of a regular triangle on the unit circle.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Square roots of a complex number.}\par
\textbf{Problem.} Find the square roots of $3+4i$.
\begin{solution}
Let $(a+ib)^2=3+4i$. Then $a^2-b^2=3$ and $2ab=4$. Since $a^2+b^2=5$, one obtains $a=2$, $b=1$. The roots are $\pm(2+i)$.

\textbf{Interpretation.} Squaring doubles argument and squares modulus; taking square roots reverses both operations with two possible angles.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Euler's identity.}\par
\textbf{Problem.} Evaluate $e^{i\pi}$.
\begin{solution}
Euler's formula gives $e^{i\pi}=\cos\pi+i\sin\pi=-1$.

\textbf{Interpretation.} A rotation by $\pi$ on the unit circle reaches the point $-1$.
\end{solution}
\end{workedexample}

\subsection{Mappings and complex functions}

\begin{workedexample}
\textbf{Affine mapping.}\par
\textbf{Problem.} Describe $w=(1+i)z+2$.
\begin{solution}
The factor $1+i=\sqrt2e^{i\pi/4}$ rotates by $\pi/4$ and scales by $\sqrt2$; adding $2$ translates two units along the real axis.

\textbf{Interpretation.} Every nonconstant affine complex map is a similarity transformation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{The square map.}\par
\textbf{Problem.} Map the ray $z=re^{i\theta_0}$ under $w=z^2$.
\begin{solution}
\[
w=r^2e^{2i\theta_0}.
\]
The ray angle doubles and radial distances square.

\textbf{Interpretation.} The map stretches nonuniformly and doubles angular separation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Inversion.}\par
\textbf{Problem.} Find the image of $z=re^{i\theta}$ under $w=1/z$.
\begin{solution}
\[
w=r^{-1}e^{-i\theta}.
\]

\textbf{Interpretation.} Inversion reciprocates distance from the origin and reverses orientation angle.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{A line under inversion.}\par
\textbf{Problem.} Determine the image of the vertical line $x=a\neq0$ under $w=1/z$.
\begin{solution}
Writing $w=u+iv$ and $z=1/w=(u-iv)/(u^2+v^2)$ gives
\[
\frac{u}{u^2+v^2}=a,
\]
or $(u-1/(2a))^2+v^2=1/(4a^2)$.

\textbf{Interpretation.} A line not through the origin maps to a circle through the origin.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Exponential periodicity.}\par
\textbf{Problem.} Solve $e^z=1$.
\begin{solution}
With $z=x+iy$, $e^z=e^x(\cos y+i\sin y)$. Therefore $x=0$ and $y=2\pi n$, so $z=2\pi in$.

\textbf{Interpretation.} The complex exponential wraps horizontal strips around the punctured plane.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Complex sine on the imaginary axis.}\par
\textbf{Problem.} Evaluate $\sin(iy)$.
\begin{solution}
Using $\sin z=(e^{iz}-e^{-iz})/(2i)$,
\[
\sin(iy)=i\sinh y.
\]

\textbf{Interpretation.} Trigonometric and hyperbolic functions are different slices of the same complex exponential structure.
\end{solution}
\end{workedexample}

\subsection{Differentiability and analyticity}

\begin{workedexample}
\textbf{Derivative of $z^2$.}\par
\textbf{Problem.} Compute the complex derivative of $f(z)=z^2$.
\begin{solution}
\[
\frac{f(z+h)-f(z)}{h}=2z+h\longrightarrow2z.
\]
Thus $f'(z)=2z$ for all $z$.

\textbf{Interpretation.} The local scale factor is $|2z|$ and the local rotation is $\arg(2z)$ away from the origin.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Failure of differentiability for conjugation.}\par
\textbf{Problem.} Test $f(z)=\bar z$ at the origin.
\begin{solution}
Along real increments, $\bar h/h=1$; along imaginary increments, $\bar h/h=-1$. The limit depends on direction, so no complex derivative exists.

\textbf{Interpretation.} Complex differentiability is much stricter than differentiability in two real variables.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Cauchy--Riemann test.}\par
\textbf{Problem.} Determine whether $f(z)=x^2-y^2+i2xy$ is analytic.
\begin{solution}
Here $u=x^2-y^2$, $v=2xy$. Since $u_x=2x=v_y$ and $u_y=-2y=-v_x$, the Cauchy--Riemann equations hold everywhere. In fact $f(z)=z^2$.

\textbf{Interpretation.} The real and imaginary parts are locked together as harmonic conjugates.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{A nonanalytic real-valued function.}\par
\textbf{Problem.} Test $f(z)=|z|^2$.
\begin{solution}
$u=x^2+y^2$, $v=0$. The equations require $2x=0$ and $2y=0$, so they hold only at the origin, not on a neighbourhood. The function is nowhere analytic.

\textbf{Interpretation.} Satisfying the equations at one isolated point is not enough for analyticity.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Finding a harmonic conjugate.}\par
\textbf{Problem.} Given $u=x^2-y^2$, find $v$ so that $u+iv$ is analytic.
\begin{solution}
From $v_y=u_x=2x$, $v=2xy+g(x)$. Then $v_x=2y+g'(x)=-u_y=2y$, so $g$ is constant. Thus $v=2xy+C$.

\textbf{Interpretation.} The analytic function is $z^2+iC$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Harmonicity.}\par
\textbf{Problem.} Verify that the real part of $e^z$ is harmonic.
\begin{solution}
$u=e^x\cos y$. Then $u_{xx}=e^x\cos y$ and $u_{yy}=-e^x\cos y$, so $\nabla^2u=0$.

\textbf{Interpretation.} Real and imaginary parts of analytic functions solve Laplace's equation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Constructing an analytic function from a potential.}\par
\textbf{Problem.} Let $u=e^x\cos y$. Find the analytic function.
\begin{solution}
The harmonic conjugate is $v=e^x\sin y+C$, so $f(z)=e^z+iC$.

\textbf{Interpretation.} Equipotential and stream-function curves meet orthogonally where $f'(z)\neq0$.
\end{solution}
\end{workedexample}

\subsection{Elementary contour integrals}

\begin{workedexample}
\textbf{Integral along a line segment.}\par
\textbf{Problem.} Evaluate $\int_C z\,dz$ from $0$ to $1+i$ along the straight segment.
\begin{solution}
Set $z(t)=(1+i)t$, $0\le t\le1$. Then
\[
\int_0^1 (1+i)t(1+i)\,dt=\frac{(1+i)^2}{2}=i.
\]

\textbf{Interpretation.} Because $z$ has the antiderivative $z^2/2$, the value depends only on endpoints.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Integral around a circle.}\par
\textbf{Problem.} Evaluate $\oint_{|z|=R}z^n\,dz$ for integer $n$.
\begin{solution}
With $z=Re^{it}$,
\[
\oint z^n\,dz=iR^{n+1}\int_0^{2\pi}e^{i(n+1)t}\,dt.
\]
The result is $0$ unless $n=-1$, in which case it is $2\pi i$.

\textbf{Interpretation.} The exceptional integrand $1/z$ detects one winding around the origin.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Orientation reversal.}\par
\textbf{Problem.} What happens to $\int_C f(z)\,dz$ if the path orientation is reversed?
\begin{solution}
The reversed parametrization changes $dz$ by a minus sign, hence
\[
\int_{-C}f(z)\,dz=-\int_Cf(z)\,dz.
\]

\textbf{Interpretation.} Complex line integrals are oriented geometric quantities.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Path independence for an entire integrand.}\par
\textbf{Problem.} Evaluate $\int_C e^z\,dz$ from $z_0$ to $z_1$.
\begin{solution}
Since $e^z$ is its own antiderivative,
\[
\int_Ce^z\,dz=e^{z_1}-e^{z_0}
\]
for every piecewise smooth path $C$.

\textbf{Interpretation.} Analytic antiderivatives convert a path problem into endpoint data.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{A path-dependent integral.}\par
\textbf{Problem.} Compare $\int_C\bar z\,dz$ from $0$ to $1+i$ along two paths.
\begin{solution}
Along the straight line $z=(1+i)t$, the integral is $\int_0^1(1-i)t(1+i)dt=1$. Along the broken path $0\to1\to1+i$, the two pieces give $1/2$ and $1/2+i$, totaling $1+i$.

\textbf{Interpretation.} The nonanalytic integrand has no global antiderivative, so the route matters.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Winding number preview.}\par
\textbf{Problem.} Evaluate $\oint_C dz/z$ when $C$ winds twice counterclockwise around the origin.
\begin{solution}
Each positive winding contributes $2\pi i$, so the result is $4\pi i$.

\textbf{Interpretation.} The integral records topological winding rather than metric details of the curve.
\end{solution}
\end{workedexample}

\subsection{Integrated applications}

\begin{workedexample}
\textbf{Phasor differentiation.}\par
\textbf{Problem.} Differentiate $Ae^{i\omega t}$ repeatedly.
\begin{solution}
Each derivative multiplies the phasor by $i\omega$:
\[
\frac{d^n}{dt^n}Ae^{i\omega t}=(i\omega)^nAe^{i\omega t}.
\]

\textbf{Interpretation.} Differentiation becomes algebraic multiplication, one reason complex notation simplifies oscillations.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Series impedance.}\par
\textbf{Problem.} Find the impedance of a resistor and inductor in series.
\begin{solution}
With time dependence $e^{i\omega t}$, $V_R=RI$ and $V_L=i\omega LI$. Therefore
\[
Z=\frac{V}{I}=R+i\omega L.
\]

\textbf{Interpretation.} Resistance and reactance are the real and imaginary components of one response coefficient.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Interference of two amplitudes.}\par
\textbf{Problem.} Let $\psi_1=A$ and $\psi_2=Ae^{i\phi}$. Find $|\psi_1+\psi_2|^2$.
\begin{solution}
\[
|A(1+e^{i\phi})|^2=2A^2(1+\cos\phi)=4A^2\cos^2(\phi/2).
\]

\textbf{Interpretation.} Relative phase changes measurable intensity even though a common global phase does not.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Lowest-Landau-level polynomial preview.}\par
\textbf{Problem.} Describe the zeros of $\psi(z)=z^m e^{-|z|^2/(4\ell_B^2)}$.
\begin{solution}
The Gaussian never vanishes, while $z^m$ has a zero of multiplicity $m$ at the origin. Its phase winds by $2\pi m$ around that point.

\textbf{Interpretation.} Analytic zeros encode vortices and angular momentum in two-dimensional quantum states.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Complex potential for uniform flow.}\par
\textbf{Problem.} Interpret $F(z)=Uz$ in two-dimensional ideal flow.
\begin{solution}
$F=\phi+i\psi=U x+iUy$. Thus $\phi=Ux$ and $\psi=Uy$; the velocity is uniform and directed along $x$.

\textbf{Interpretation.} Analytic functions package velocity potential and stream function into one object.
\end{solution}
\end{workedexample}
