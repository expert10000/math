\section{Twenty-Five Integrated Worked Examples}
\label{sec:ch2-worked-examples}

These examples are deliberately cumulative.  Each states the governing idea,
performs the calculation, and closes with a physical or geometric reading of
the result.

\begin{workedexample}
\textbf{Domain and level set.}\par
\textbf{Problem.} For $f(x,y)=\sqrt{4-x^2-y^2}$, determine the domain and the level set $f=1$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
Reality of the square root requires $x^2+y^2\le4$.  Setting $f=1$ gives $4-x^2-y^2=1$, hence $x^2+y^2=3$, a circle of radius $\sqrt3$.

\textbf{Interpretation.} The graph is the upper hemisphere of radius 2; horizontal slicing produces circular level sets.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{A removable limit.}\par
\textbf{Problem.} Evaluate $\lim_{x\to2}(x^2-4)/(x-2)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
Factor $x^2-4=(x-2)(x+2)$.  For $x\ne2$ the quotient equals $x+2$, so the limit is $4$.

\textbf{Interpretation.} The value at the missing point is irrelevant to the limit; only nearby values matter.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Two-path test.}\par
\textbf{Problem.} Show that $f(x,y)=xy/(x^2+y^2)$ has no limit at $(0,0)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
Along $y=x$, $f=1/2$.  Along $y=-x$, $f=-1/2$.  Since two paths give different values, the limit does not exist.

\textbf{Interpretation.} Multivariable approach has infinitely many directions, so one-dimensional intuition must be strengthened.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Derivative from first principles.}\par
\textbf{Problem.} Use the definition to differentiate $f(x)=x^2$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$[f(x+h)-f(x)]/h=[2xh+h^2]/h=2x+h$.  Taking $h\to0$ gives $f'(x)=2x$.

\textbf{Interpretation.} The derivative is the coefficient of the best local linear approximation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Linearization error.}\par
\textbf{Problem.} Approximate $\sqrt{4.1}$ by linearizing $f(x)=\sqrt{x}$ at $x=4$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$f(4)=2$ and $f'(4)=1/4$. Thus $f(4.1)\approx2+(1/4)(0.1)=2.025$.

\textbf{Interpretation.} A nonlinear function is replaced locally by its tangent line; the neglected error is second order.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Velocity and acceleration.}\par
\textbf{Problem.} For $x(t)=t^3-3t^2+2t$, find $v$ and $a$ at $t=2$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$v=3t^2-6t+2$ and $a=6t-6$. Therefore $v(2)=2$ and $a(2)=6$.

\textbf{Interpretation.} Position, velocity, and acceleration are successive time derivatives.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Product and chain rules.}\par
\textbf{Problem.} Differentiate $y=x^2e^{3x}$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$y'=2xe^{3x}+3x^2e^{3x}=e^{3x}(2x+3x^2)$.

\textbf{Interpretation.} The product rule combines independent changes; the chain rule propagates change through composition.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Implicit differentiation.}\par
\textbf{Problem.} For $x^2+xy+y^2=7$, find $dy/dx$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
Differentiating gives $2x+y+x y'+2y y'=0$, hence $y'=-(2x+y)/(x+2y)$.

\textbf{Interpretation.} The curve can have a well-defined tangent even where solving globally for $y(x)$ is inconvenient.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Partial derivatives.}\par
\textbf{Problem.} For $T=x^2y+e^{yz}$, compute $T_x,T_y,T_z$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$T_x=2xy$, $T_y=x^2+ze^{yz}$, and $T_z=ye^{yz}$.

\textbf{Interpretation.} A partial derivative varies one coordinate while holding the others fixed.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Total differential.}\par
\textbf{Problem.} Estimate the change in $f=x^2y$ from $(2,3)$ under $(dx,dy)=(0.01,-0.02)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$df=2xy\,dx+x^2\,dy$. At $(2,3)$, $df=12(0.01)+4(-0.02)=0.04$.

\textbf{Interpretation.} The differential combines all first-order coordinate effects into one predicted change.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Tangent plane.}\par
\textbf{Problem.} Find the tangent plane to $z=x^2+xy$ at $(1,2,3)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$f_x=2x+y=4$ and $f_y=x=1$. Thus $z-3=4(x-1)+(y-2)$.

\textbf{Interpretation.} The plane is the graph of the derivative, the best local linear model of the surface.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Gradient.}\par
\textbf{Problem.} For $f=x^2+2y^2+3z^2$, find $\nabla f$ at $(1,-1,2)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\nabla f=(2x,4y,6z)$, hence $\nabla f(1,-1,2)=(2,-4,12)$.

\textbf{Interpretation.} The gradient points in the direction of greatest instantaneous increase.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Directional derivative.}\par
\textbf{Problem.} For $f=x^2+y^2$, compute the derivative at $(1,2)$ toward $(4,6)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
The direction is $(3,4)$, so $\hat u=(3/5,4/5)$. Since $\nabla f=(2,4)$, $D_{\hat u}f=2(3/5)+4(4/5)=22/5$.

\textbf{Interpretation.} Direction must be normalized; otherwise the result includes an arbitrary speed factor.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Steepest descent.}\par
\textbf{Problem.} For $V=x^2+xy+3y^2$, find the steepest-descent direction at $(1,1)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\nabla V=(2x+y,x+6y)=(3,7)$. The unit steepest-descent direction is $-(3,7)/\sqrt{58}$.

\textbf{Interpretation.} For a potential energy, negative gradient gives the force direction.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Jacobian matrix.}\par
\textbf{Problem.} For $F(u,v)=(u^2-v,uv)$, compute $J_F(1,2)$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$J_F=\begin{pmatrix}2u&-1\\v&u\end{pmatrix}$, hence $J_F(1,2)=\begin{pmatrix}2&-1\\2&1\end{pmatrix}$.

\textbf{Interpretation.} The Jacobian is the local linear transformation carrying input perturbations to output perturbations.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Jacobian determinant.}\par
\textbf{Problem.} For polar coordinates $x=r\cos\theta$, $y=r\sin\theta$, find the area scale factor.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\det\partial(x,y)/\partial(r,\theta)=\det\begin{pmatrix}\cos\theta&-r\sin\theta\\\sin\theta&r\cos\theta\end{pmatrix}=r$.

\textbf{Interpretation.} A small polar rectangle $dr\,d	heta$ maps to area $r\,dr\,d	heta$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Multivariable chain rule.}\par
\textbf{Problem.} Let $z=x^2y$, $x=t^2$, $y=e^t$. Find $dz/dt$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$z=t^4e^t$, so $dz/dt=e^t(4t^3+t^4)$. Equivalently, $z_xx'+z_yy'$ gives the same result.

\textbf{Interpretation.} The chain rule sums change transmitted through every dependency path.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Critical point classification.}\par
\textbf{Problem.} Classify the origin for $f=x^2+4xy+y^2$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
The Hessian is $\begin{pmatrix}2&4\\4&2\end{pmatrix}$ with determinant $-12<0$, so the origin is a saddle point.

\textbf{Interpretation.} A zero gradient is only a candidate; the Hessian determines the local quadratic geometry.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Constrained optimization.}\par
\textbf{Problem.} Maximize $xy$ subject to $x^2+y^2=2$ in the first quadrant.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\nabla(xy)=\lambda\nabla(x^2+y^2)$ gives $y=2\lambda x$ and $x=2\lambda y$, hence $x=y$. The constraint gives $x=y=1$, so the maximum is $1$.

\textbf{Interpretation.} At a constrained extremum, the objective gradient is normal to the constraint surface.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Radial derivative.}\par
\textbf{Problem.} For $f(\mathbf r)=r^{-1}$ in three dimensions, compute $\nabla f$ for $r\ne0$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
Since $df/dr=-r^{-2}$ and $\nabla r=\mathbf r/r$, $\nabla(r^{-1})=-\mathbf r/r^3$.

\textbf{Interpretation.} Inverse-square fields arise as gradients of inverse-distance potentials.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Divergence.}\par
\textbf{Problem.} For $\mathbf F=(x^2,xy,z^2)$, compute $\nabla\cdot\mathbf F$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\nabla\cdot\mathbf F=2x+x+2z=3x+2z$.

\textbf{Interpretation.} Positive divergence represents local net outflow; negative divergence represents local convergence.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Curl.}\par
\textbf{Problem.} For $\mathbf F=(-y,x,0)$, compute $\nabla\times\mathbf F$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\nabla\times\mathbf F=(0,0,\partial_x x-\partial_y(-y))=(0,0,2)$.

\textbf{Interpretation.} The field has uniform local rotation about the $z$ axis.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Laplacian.}\par
\textbf{Problem.} Compute $\nabla^2 e^{-\alpha(x^2+y^2)}$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
Differentiating twice gives $\nabla^2 f=[4\alpha^2(x^2+y^2)-4\alpha]e^{-\alpha(x^2+y^2)}$.

\textbf{Interpretation.} The Laplacian compares a field value with its local neighborhood and drives diffusion and wave equations.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Complex phase gradient.}\par
\textbf{Problem.} For $\psi=Ae^{i\mathbf k\cdot\mathbf r}$, compute $\nabla\psi$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$\nabla\psi=i\mathbf k\psi$.

\textbf{Interpretation.} A plane wave is an eigenfunction of the gradient; its phase gradient is the wavevector.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Error propagation.}\par
\textbf{Problem.} If $R=V/I$, estimate $dR$ for small independent changes $dV,dI$.

\textbf{Theory.} Identify the relevant derivative or local-linear principle before calculating.

\begin{solution}
$dR=(1/I)dV-(V/I^2)dI$. Thus $dR/R=dV/V-dI/I$ to first order.

\textbf{Interpretation.} Differentials propagate measurement errors through a derived quantity.
\end{solution}
\end{workedexample}
