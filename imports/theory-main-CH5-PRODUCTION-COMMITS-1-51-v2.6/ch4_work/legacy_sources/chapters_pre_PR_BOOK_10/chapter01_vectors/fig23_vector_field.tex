\begin{tikzpicture}[scale=.92]
  \draw[axis] (-3.2,0)--(3.5,0) node[right] {$x$};
  \draw[axis] (0,-3.0)--(0,3.3) node[above] {$y$};

  \foreach \x in {-2,-1,0,1,2}{
    \foreach \y in {-2,-1,0,1,2}{
      \pgfmathsetmacro{\dx}{-.32*\y}
      \pgfmathsetmacro{\dy}{.32*\x}
      \draw[vector secondary,line width=.8pt] (\x,\y)--++(\dx,\dy);
    }
  }

  \draw[force] (1.6,0)--(1.6,1.15) node[above,label] {$\mathbf F(1.6,0)$};
  \node[concept node,text width=5.0cm,anchor=west] at (4.1,.1) {
    A vector field assigns one vector to every point:
    \[
      \mathbf F(x,y)=(-y,x).
    \]
    Here the field circulates counterclockwise around the origin.
  };
\end{tikzpicture}
