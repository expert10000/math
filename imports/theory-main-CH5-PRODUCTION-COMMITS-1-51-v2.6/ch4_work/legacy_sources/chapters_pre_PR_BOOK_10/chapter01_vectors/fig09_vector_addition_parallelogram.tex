% Parallelogram rule for vector addition
\begin{tikzpicture}[scale=1.0]
  \coordinate (O) at (0,0);
  \coordinate (A) at (4.0,.8);
  \coordinate (B) at (1.45,2.85);
  \coordinate (R) at (5.45,3.65);
  \fill[SoftBlue] (O)--(A)--(R)--(B)--cycle;
  \draw[construction] (A)--(R);
  \draw[construction] (B)--(R);
  \draw[vector] (O)--(A) node[midway,below,label] {$\mathbf A$};
  \draw[vector secondary] (O)--(B) node[midway,left,label] {$\mathbf B$};
  \draw[force] (O)--(R) node[midway,above left,label] {$\mathbf A+\mathbf B$};
  \fill[HighlightGold] (R) circle (2.1pt);
\end{tikzpicture}
