% Orthonormal unit basis in three dimensions
\begin{tikzpicture}[scale=1.08]
  \coordinate (O) at (0,0);
  \draw[vector] (O)--(3.2,0) node[right,label] {$\hat{\mathbf i}$};
  \draw[vector secondary] (O)--(-1.7,-1.15) node[below left,label] {$\hat{\mathbf j}$};
  \draw[field] (O)--(0,3.0) node[above,label] {$\hat{\mathbf k}$};
  \node[below right] at (O) {$O$};
  \draw[PhysicsBlue!55,line width=.8pt] (.45,0)--(.45,.45)--(0,.45);
  \node[concept node,text width=4.0cm] at (4.7,2.1) {$|\hat{\mathbf i}|=|\hat{\mathbf j}|=|\hat{\mathbf k}|=1$\\[1mm]
  $\hat{\mathbf i}\cdot\hat{\mathbf j}=\hat{\mathbf j}\cdot\hat{\mathbf k}=\hat{\mathbf k}\cdot\hat{\mathbf i}=0$};
\end{tikzpicture}
