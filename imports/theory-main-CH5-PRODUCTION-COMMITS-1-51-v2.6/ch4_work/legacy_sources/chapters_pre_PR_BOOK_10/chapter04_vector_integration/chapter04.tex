\chapter{Vector Integration and Integral Theorems}
\chapteropeningart
{From local fields to global physics}
{Integration accumulates infinitesimal contributions along curves, across
surfaces, and throughout volumes, connecting local differential laws with
global physical quantities.}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\chapterquote
{``Differentiation reveals what happens locally; integration reveals what
happens as a whole.''}
{}

\begin{roadmapbox}
Local fields $\longrightarrow$ line integrals $\longrightarrow$ surface flux
$\longrightarrow$ volume integrals $\longrightarrow$ integral theorems.
\end{roadmapbox}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\learningobjectives

\begin{itemize}
  \item explain the distinction between local and global descriptions of physical systems;
  \item identify line, surface, and volume integrals and their physical meanings;
  \item understand the geometric ideas of work, circulation, and flux;
  \item recognize how Green's theorem, Gauss's theorem, and Stokes' theorem connect boundaries with interiors;
  \item relate integral formulations to conservation laws and quantum probability.
\end{itemize}
