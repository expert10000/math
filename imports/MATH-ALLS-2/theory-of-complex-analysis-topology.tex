	\section*{Metric spaces: openness via neighborhoods and closedness via distance}
	
	\paragraph{Problem.}
	Let $(X,d)$ be a metric space.
	\begin{enumerate}[label=(\alph*)] % or simply [(a)] thanks to shortlabels
		\item Show that $U\subset X$ is open in $X$ if and only if $U$ is a neighborhood of all of its points; i.e., for every $x\in U$ there exists $r>0$ with $B(x,r)\subset U$.
		\item Show that $A\subset X$ is closed in $X$ if and only if
		\[
		A=\{x\in X:\ \inf_{y\in A} d(x,y)=0\}.
		\]
	\end{enumerate}
	
	\paragraph{Background.}
	For $E\subset X$ and $x\in X$, define the point-to-set distance
	\[
	d(x,E):=\inf\{d(x,y):y\in E\}\in[0,\infty].
	\]
	The open ball is $B(x,r)=\{y\in X: d(x,y)<r\}$. A set $U$ is \emph{open} if for each $x\in U$ there exists $r>0$ with $B(x,r)\subset U$. A set $F$ is \emph{closed} if $X\setminus F$ is open, equivalently if $F$ contains all of its limit points. The closure $\overline{E}$ is the set of all $x$ such that every open ball around $x$ meets $E$.
	
	\bigskip
	
	\paragraph{Solution to (a).}
	\emph{($\Rightarrow$)} Suppose $U$ is open and let $x\in U$. By the metric-space definition of open, there exists $r>0$ with $B(x,r)\subset U$. Thus $U$ is a neighborhood of $x$. Since $x$ was arbitrary, $U$ is a neighborhood of each of its points.
	
	\smallskip
	\noindent
	\emph{($\Leftarrow$)} Conversely, assume that for every $x\in U$ there exists $r_x>0$ with $B(x,r_x)\subset U$. This is precisely the definition that $U$ is open in a metric space. Hence $U$ is open.
	
	\bigskip
	
	\paragraph{Solution to (b).}
	Set $S:=\{x\in X: d(x,A)=0\}$. We first show $S=\overline{A}$.
	
	\smallskip
	\noindent
	($\subset$) Let $x\in S$, so $d(x,A)=0$. For any $\varepsilon>0$ there exists $a\in A$ with $d(x,a)<\varepsilon$. Hence $B(x,\varepsilon)\cap A\neq\emptyset$ for all $\varepsilon>0$, so $x\in\overline{A}$.
	
	\smallskip
	\noindent
	($\supset$) Let $x\in\overline{A}$. Then for every $\varepsilon>0$, $B(x,\varepsilon)\cap A\neq\emptyset$, so there exists $a_\varepsilon\in A$ with $d(x,a_\varepsilon)<\varepsilon$. Taking the infimum over $A$ yields $d(x,A)=0$, hence $x\in S$.
	
	\smallskip
	Therefore $S=\overline{A}$. Now, $A$ is closed if and only if $A=\overline{A}$; by the previous identity, this is equivalent to $A=S=\{x\in X: d(x,A)=0\}$, proving the claim.
	
	\bigskip
	
	\paragraph{Examples.}
	\begin{enumerate}
		\item In $(\mathbb{R},|\cdot|)$, $U=(0,1)$ is open since for $x\in(0,1)$ the ball $B\!\left(x,\min\{x,1-x\}\right)\subset(0,1)$. The set $A=[0,1]$ is closed and equals $\{x: d(x,[0,1])=0\}$.
		\item In a discrete metric space, every set is open and closed. Here $d(x,A)=0$ iff $x\in A$, so indeed $A=\{x: d(x,A)=0\}$ holds for all $A$.
		\item In $X=\mathbb{R}^2\setminus\{0\}$ (Euclidean metric), $U=\{x:\|x\|>1\}$ is open; $A=\{x:\|x\|\ge1\}$ is closed in $X$ and equals $\{x\in X: d(x,A)=0\}$.
	\end{enumerate}
	
\section*{Infinite intersections of open sets need not be open}

\paragraph{Problem.}
Let $(X,d)$ be a metric space. In the definition of the topology of $(X,d)$, why don’t we require arbitrary (possibly infinite) intersections of open sets to be open? Give a counterexample with $X=\mathbb{R}$.

\paragraph{Background.}
In a metric space $(X,d)$, a set $U\subset X$ is open if for every $x\in U$ there exists $r>0$ such that the open ball $B(x,r)=\{y\in X:\ d(x,y)<r\}\subset U$. It follows immediately that:
\begin{itemize}
\item arbitrary unions of open sets are open;
\item finite intersections of open sets are open (take the minimum of finitely many witness radii).
\end{itemize}
However, these arguments do not extend to infinite families of open sets.

\paragraph{Solution (counterexample in $\mathbb{R}$).}
For each $n\in\mathbb{N}$ define the open interval
\[
U_n := (-1/n,\,1/n).
\]
Each $U_n$ is open in $\mathbb{R}$ (with the usual metric). Consider their infinite intersection:
\[
\bigcap_{n=1}^\infty U_n = \{0\}.
\]
The set $\{0\}$ is not open in $\mathbb{R}$, since for any $r>0$ the open ball $B(0,r)=(-r,r)$ contains points different from $0$. Therefore, an infinite intersection of open sets need not be open.

\paragraph{Discussion.}
If we demanded that arbitrary intersections of open sets be open in every topology, then in $\mathbb{R}$ the intersection of all open neighborhoods of a point $x$ would be $\{x\}$, forcing every singleton to be open; since any subset is a union of singletons, every set would be open. This would collapse the usual topology to the discrete topology, which is not intended.

\paragraph{Additional examples.}
In any metric space $(X,d)$ and any $x\in X$, the decreasing family of open balls $B(x,1/n)$ consists of open sets with
\[
\bigcap_{n\ge1} B(x,1/n) = \{x\}.
\]
This set is open if and only if the metric is discrete. Hence, outside of the discrete case, infinite intersections of open sets commonly fail to be open.

\section*{Convergence in metric spaces: $\varepsilon$--$N$ vs neighborhoods and uniqueness}


\paragraph{Problem.}
Let $(X,d)$ be a metric space and $\{x_n\}_{n=1}^\infty\subset X$. We say $x_n\to x\in X$ if
\[
(\ast)\qquad \forall \varepsilon>0\ \exists N(\varepsilon)\in\mathbb{N}\ \text{ such that }\ d(x_n,x)<\varepsilon\ \text{ for all } n\ge N(\varepsilon).
\]
\begin{enumerate}
\item[(a)] Show that $x_n\to x$ is equivalent to:
\[
(\ast\ast)\qquad \forall V\subset X\ \text{with }V\text{ a neighborhood of }x,\ \exists N(V)\ \text{ such that } x_n\in V\ \text{ for all } n\ge N(V).
\]
\item[(b)] Using $(\ast)$, prove that if $\{x_n\}$ converges in $X$, then its limit is unique.
\end{enumerate}

\paragraph{Background.}
A set $V\subset X$ is a neighborhood of $x$ if there exists $r>0$ with the open ball $B(x,r)=\{y\in X: d(x,y)<r\}$ contained in $V$. The triangle inequality states $d(x,y)\le d(x,z)+d(z,y)$, and $d(x,y)=0$ implies $x=y$.

\paragraph{Solution to (a).}
\emph{($\ast\Rightarrow\ast\ast$)} Let $V$ be a neighborhood of $x$. Then $\exists r>0$ with $B(x,r)\subset V$. By $(\ast)$ with $\varepsilon=r$, there is $N(r)$ such that $d(x_n,x)<r$ for all $n\ge N(r)$, hence $x_n\in B(x,r)\subset V$. Set $N(V):=N(r)$.

\smallskip
\noindent
\emph{($\ast\ast\Rightarrow\ast$)} Fix $\varepsilon>0$. The ball $B(x,\varepsilon)$ is a neighborhood of $x$, so by $(\ast\ast)$ there exists $N$ with $x_n\in B(x,\varepsilon)$ for all $n\ge N$, i.e.\ $d(x_n,x)<\varepsilon$. Thus $(\ast)$ holds.

\paragraph{Solution to (b): uniqueness of the limit.}
Suppose $x_n\to x$ and $x_n\to y$. Let $\varepsilon>0$. Choose $N_1$ with $d(x_n,x)<\varepsilon/2$ for all $n\ge N_1$ and $N_2$ with $d(x_n,y)<\varepsilon/2$ for all $n\ge N_2$. For $n\ge N:=\max\{N_1,N_2\}$,
\[
d(x,y)\le d(x,x_n)+d(x_n,y)<\frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\varepsilon.
\]
Since $\varepsilon$ is arbitrary, $d(x,y)=0$, hence $x=y$. Therefore limits are unique.

\paragraph{Examples.}
\begin{enumerate}
\item In $(\mathbb{R},|\cdot|)$, $x_n=1/n\to 0$. Given a neighborhood $V$ of $0$, pick $r>0$ with $(-r,r)\subset V$ and then choose $N>\tfrac{1}{r}$ so that $1/n<r$ for $n\ge N$.
\item The sequence $x_n=(-1)^n$ does not converge: for $\varepsilon=1$, no single $x\in\mathbb{R}$ satisfies $|x_n-x|<1$ for all large $n$.
\item In the discrete metric, $x_n\to x$ iff $x_n=x$ eventually. Indeed, take $\varepsilon=\tfrac12$, then $d(x_n,x)<\tfrac12$ means $x_n=x$.
\end{enumerate}
	
	
\section*{Uniform limits: sums and products}

\paragraph{Problem.}
Let $(f_n)$ and $(g_n)$ be sequences of real-valued functions on a set $X$ with
$f_n \xRightarrow{\text{\rm unif}} f$ and $g_n \xRightarrow{\text{\rm unif}} g$.
\begin{enumerate}[label=(\alph*)]
\item Show that $f_n+g_n \xRightarrow{\text{\rm unif}} f+g$.
\item Show that $f_n g_n \to fg$ pointwise, but that $f_n g_n$ need not converge uniformly to $fg$ in general. If $f$ and $g$ are bounded, does the latter conclusion change? What if $f$ is bounded but $g$ is not?
\end{enumerate}

\paragraph{Background.}
Uniform convergence means $\|f_n-f\|_\infty=\sup_{x\in X}|f_n(x)-f(x)|\to 0$; it implies pointwise convergence. For products we use
\[
|f_n g_n - fg| \le |f_n|\,|g_n-g| + |g|\,|f_n-f|.
\]
If $f_n\Rightarrow f$, then $f_n$ are eventually uniformly bounded.

\paragraph{Solution to (a).}
Let $\varepsilon>0$. Choose $N_1$ with $\sup_X|f_n-f|<\varepsilon/2$ for $n\ge N_1$ and $N_2$ with $\sup_X|g_n-g|<\varepsilon/2$ for $n\ge N_2$. For $n\ge N=\max\{N_1,N_2\}$,
\[
\sup_{x\in X}|(f_n+g_n)-(f+g)| \le \sup_X|f_n-f|+\sup_X|g_n-g| < \varepsilon,
\]
hence $f_n+g_n \Rightarrow f+g$.

\paragraph{Solution to (b): pointwise and failure of uniformity in general.}
For fixed $x\in X$, $f_n(x)\to f(x)$ and $g_n(x)\to g(x)$, hence $f_n(x)g_n(x)\to f(x)g(x)$; thus the convergence is pointwise.

To see that uniform convergence may fail, take $X=\mathbb{R}$, $f\equiv 0$, $g(x)=x$, $f_n\equiv 1/n$, and $g_n\equiv g$. Then $f_n\Rightarrow f$ and $g_n\Rightarrow g$, but
\[
\sup_{x\in\mathbb{R}} |f_n(x)g_n(x)-f(x)g(x)|=\sup_{x\in\mathbb{R}}\frac{|x|}{n}=\infty
\]
for all $n$, so $(f_n g_n)$ does not converge uniformly to $fg$.

\paragraph{When $f$ and $g$ are bounded.}
Assume $\|f\|_\infty\le M_f$ and $\|g\|_\infty\le M_g$. From uniform convergence, for $n$ large
\(
\sup_X|f_n|\le M_f+1
\)
and
\(
\sup_X|g_n-g|<1.
\)
Then for all sufficiently large $n$,
\[
\sup_X |f_n g_n - fg|
\le (M_f+1)\,\sup_X|g_n-g| + M_g\,\sup_X|f_n-f| \xrightarrow[n\to\infty]{} 0.
\]
Hence $f_n g_n \xRightarrow{\text{\rm unif}} fg$.

\paragraph{If $f$ is bounded but $g$ is not.}
Uniform convergence of the product need not hold; the counterexample above has $f\equiv 0$ bounded while $g(x)=x$ is unbounded.

\section*{Uniform convergence and sequential continuity}

\paragraph{Problem.}
Let $(f_n)$ be a sequence of continuous real-valued functions on $[0,1]^k\subset\mathbb{R}^k$, and suppose that $f_n \Rightarrow f$ uniformly. Let $(x_n)$ be a sequence in $[0,1]^k$ with $x_n\to x$.
\begin{enumerate}[label=(\alph*)]
\item Show that $f_n(x_n)\to f(x)$. (Uniform convergence preserves sequential continuity.)
\item Is (a) still true if $f_n$ converges pointwise, but not necessarily uniformly, to $f$?
\end{enumerate}

\paragraph{Background.}
Uniform convergence means $\sup_{y\in[0,1]^k}|f_n(y)-f(y)|\to 0$. If each $f_n$ is continuous and $f_n\Rightarrow f$ uniformly, then $f$ is continuous.

\paragraph{Solution to (a).}
Since $f_n\Rightarrow f$ and each $f_n$ is continuous, $f$ is continuous. Then for every $n$,
\[
|f_n(x_n)-f(x)|
\le |f_n(x_n)-f(x_n)| + |f(x_n)-f(x)|.
\]
The first term is bounded by $\sup_{y}|f_n(y)-f(y)|\xrightarrow[n\to\infty]{}0$, and the second term tends to $0$ because $f$ is continuous and $x_n\to x$. Hence $f_n(x_n)\to f(x)$.

\paragraph{Solution to (b).}
No. Consider $[0,1]$ and $f_n(x)=x^n$. Then $f_n\to f$ pointwise, where $f(x)=0$ for $x\in[0,1)$ and $f(1)=1$. Let $x_n=1-\tfrac1n\to 1$. Then
\[
f_n(x_n)=(1-1/n)^n \longrightarrow e^{-1}\neq 1=f(1).
\]
Thus the conclusion of (a) can fail for merely pointwise convergence.

\section*{Which of the following are metrics?}

\subsection*{Metric axioms (refresher)}
A function $d:X\times X\to[0,\infty)$ is a metric if, for all $x,y,z\in X$,
\begin{align*}
\text{(M1) Nonnegativity/identity:}\quad& d(x,y)\ge0,\ \ d(x,y)=0 \iff x=y.\\[0.35em]
\text{(M2) Symmetry:}\quad& d(x,y)=d(y,x).\\[0.35em]
\text{(M3) Triangle inequality:}\quad& d(x,z)\le d(x,y)+d(y,z).
\end{align*}

\subsection*{(a) On $\mathbb{R}^n$: $d(x,y)=\min\{|x_1-y_1|,\ldots,|x_n-y_n|\}$}
\emph{Not a metric.} The identity of indiscernibles fails. For $x=(0,0)$ and $y=(1,0)$ in $\mathbb{R}^2$,
$d(x,y)=\min\{1,0\}=0$ while $x\ne y$.

\subsection*{(b) On $\mathbb{C}$}
\[
d(z,w)=
\begin{cases}
|z-w|,& \text{if $z$ and $w$ lie on the same line through $0$},\\
|z|+|w|,& \text{otherwise.}
\end{cases}
\]
\emph{This is a metric.} Nonnegativity and symmetry are clear. If $z,w$ are collinear, $d(z,w)=|z-w|=0$ implies $z=w$; if they are not collinear, $d(z,w)=|z|+|w|=0$ forces $z=w=0$, hence again $z=w$.
For the triangle inequality, fix $u\in\mathbb{C}$. If $z,w$ are collinear,
\[ d(z,w)=|z-w|\le |z-u|+|u-w|\le d(z,u)+d(u,w). \]
If $z,w$ are not collinear, then $d(z,w)=|z|+|w|$. Consider:
\begin{itemize}
\item $u$ collinear with $z$ but not $w$:
$d(z,u)+d(u,w)=|z-u|+(|u|+|w|)\ge ||z|-|u||+|u|+|w|\ge |z|+|w|$.
\item $u$ collinear with $w$ but not $z$: symmetric.
\item $u$ collinear with neither:
$d(z,u)+d(u,w)=(|z|+|u|)+(|u|+|w|)=|z|+|w|+2|u|\ge |z|+|w|$.
\end{itemize}
Thus $d$ satisfies the triangle inequality.

\subsection*{(c) On $X=\mathbb{N}^{\mathbb{N}}$ (sequences of naturals)}
Let $N$ be the minimal index with $x_N\ne y_N$ (if none, set $d(x,y)=0$), and define
\[
d(x,y)=
\begin{cases}
0,& x=y,\\
2^{-N},& \text{otherwise.}
\end{cases}
\]
\emph{This is an ultrametric.} Symmetry and $d=0\Rightarrow x=y$ are immediate.
If $x$ and $y$ agree up to $k$ and $y$ and $z$ agree up to $k$, then $x$ and $z$ agree up to $k$, hence
$d(x,z)\le \max\{d(x,y),d(y,z)\}\le d(x,y)+d(y,z)$.

\subsection*{Ultrametrics (definition \& properties)}
\textbf{Definition.} A metric $d$ is an \emph{ultrametric} if
\[
d(x,z)\le \max\{d(x,y),\,d(y,z)\}\qquad(\text{all }x,y,z).
\]
\textbf{Consequences.} (i) Triangles are isosceles with small base; (ii) open balls are nested or disjoint;
(iii) if $d(x,y)<r$ and $d(y,z)<r$ then $d(x,z)<r$.

\subsection*{(d) On closed, nonempty, bounded subsets of $(M,d')$: Hausdorff distance}
\[
d(A,B)=\max\!\left\{\sup_{a\in A}\inf_{b\in B} d'(a,b),\ \sup_{b\in B}\inf_{a\in A} d'(a,b)\right\}.
\]
\emph{This is a metric.} Nonnegativity and symmetry are clear; finiteness uses boundedness.
If $d(A,B)=0$, then each $a\in A$ has $d'(a,b_k)\to0$ with $b_k\in B$, so $a\in B$ by closedness; thus $A\subseteq B$ and symmetrically $B\subseteq A$, hence $A=B$.
For the triangle inequality, for any $a\in A$,
\[
\inf_{c\in C} d'(a,c)\le \inf_{b\in B}\!\big(d'(a,b)+\inf_{c\in C} d'(b,c)\big)
\le \inf_{b\in B} d'(a,b)+\sup_{b\in B}\inf_{c\in C} d'(b,c).
\]
Taking $\sup_{a\in A}$ and the symmetric estimate gives $d(A,C)\le d(A,B)+d(B,C)$.

\subsection*{Examples}
\begin{itemize}
\item (a) In $\mathbb{R}^2$, $d((0,0),(1,0))=0$ though the points differ.
\item (b) $d(1,i)=|1|+|i|=2$; $d(1,-1)=2$; $d(-1,i)=2$.
\item (c) $x=(1,1,1,\dots)$, $y=(1,2,0,\dots)$ $\Rightarrow d(x,y)=2^{-2}=1/4$.
\item (d) In $(\mathbb{R},|\cdot|)$, $d([0,1],[1,3])=1$.
\end{itemize}

\section*{Metrics, Ultrametrics, and Counterexamples: a Toolbox}

\subsection*{Classic metrics (worked examples)}
\begin{itemize}
\item \textbf{Absolute value / Euclidean / Manhattan / Chebyshev.}
On $\mathbb{R}^n$,
\[
d_p(x,y)=\Big(\sum_{i=1}^n |x_i-y_i|^p\Big)^{1/p}\quad (1\le p<\infty),\qquad
d_\infty(x,y)=\max_{1\le i\le n}|x_i-y_i|.
\]
Triangle follows from Minkowski; $d_\infty$ is the $p\to\infty$ limit.
\item \textbf{Discrete metric.} $d(x,y)=\mathbf 1_{x\ne y}$ on any set.
\item \textbf{Graph shortest--path.} On the vertices of a connected graph with nonnegative edge weights: $d(u,v)=$ length of a shortest $u$--$v$ path. Triangle via path concatenation.
\item \textbf{Hamming distance.} On $\{0,1,\dots,m\}^n$: $d(x,y)=|\{i:\ x_i\ne y_i\}|$.
\item \textbf{Levenshtein (edit) distance.} Minimum number (or total weight) of edits (insert/delete/substitute) to transform one string into another. Triangle by composing optimal edit scripts.
\item \textbf{Mahalanobis.} With $\Sigma\succ0$,
\[
d(x,y)=\sqrt{(x-y)^\top \Sigma^{-1}(x-y)}.
\]
\item \textbf{Angular (geodesic) distance.} On the unit sphere (or projective space),
\[
d(u,v)=\arccos\big(\langle u,v\rangle\big).
\]
\item \textbf{Hausdorff distance.} On closed, bounded subsets of a metric space $(M,d')$,
\[
d_H(A,B)=\max\!\left\{\sup_{a\in A}\inf_{b\in B} d'(a,b),\ \sup_{b\in B}\inf_{a\in A} d'(a,b)\right\}.
\]
\item \textbf{Jaccard distance (finite universe).} For finite $A,B\subseteq U$,
\[
d_J(A,B)=1-\frac{|A\cap B|}{|A\cup B|}.
\]
\item \textbf{Bounded transform.} If $d$ is a metric, then $\tilde d(x,y)=\min\{1,d(x,y)\}$ is a metric.
\end{itemize}

\subsection*{Ultrametrics (strong triangle inequality)}
\paragraph{Definition.} A metric $d$ is an \emph{ultrametric} if
\[
d(x,z)\ \le\ \max\{d(x,y),\,d(y,z)\}\qquad\text{for all }x,y,z.
\]
\paragraph{Consequences.}
\begin{itemize}
\item Every triangle is isosceles with small base: at least two sides are equal and not smaller than the third.
\item Open balls are either nested or disjoint (the ``nested balls'' property).
\item If $d(x,y)<r$ and $d(y,z)<r$, then automatically $d(x,z)<r$.
\end{itemize}
\paragraph{Examples.}
\begin{itemize}
\item \textbf{$p$--adic metric} on $\mathbb{Q}$ (or $\mathbb{Z}$): for $x\ne y$,
\[
d_p(x,y)=p^{-v_p(x-y)},\quad d_p(x,x)=0,
\]
where $v_p$ is the $p$--adic valuation. This satisfies the strong triangle.
\item \textbf{Baire/prefix metric} on sequences $x=(x_i),y=(y_i)$ over any alphabet: $d(x,y)=0$ if $x=y$, else $d(x,y)=2^{-N}$ where $N$ is the first index with $x_N\ne y_N$.
\item \textbf{Tree ultrametric.} In a rooted tree with a level/height function $h$, define $d(u,v)=h(\mathrm{lca}(u,v))$ (or a decreasing function of that height); strong triangle follows from subtree nesting.
\item \textbf{Closure under max and scaling.} If $d_1,d_2$ are ultrametrics, then $\max\{d_1,d_2\}$ is ultrametric; so is $c\,d_1$ for $c>0$.
\end{itemize}

\subsection*{Counterexamples and non--metrics}
\begin{itemize}
\item \textbf{Min--coordinate ``distance''} on $\mathbb{R}^n$:
\[
d(x,y)=\min_{1\le i\le n}|x_i-y_i|.
\]
Not a metric: identity fails (e.g.\ $(0,0)$ and $(1,0)$ give $d=0$ but points differ).
\item \textbf{Squared Euclidean} $d(x,y)=\|x-y\|_2^2$. Not a metric: triangle fails (e.g.\ $0,1,2\in\mathbb{R}$: $d(0,2)=4>d(0,1)+d(1,2)=2$).
\item \textbf{Cosine dissimilarity} $1-\cos\theta$ (with $\theta$ angle). Not a metric (triangle fails). Prefer angular distance $\arccos$.
\item \textbf{Dynamic Time Warping (DTW)} on time series. Not a metric (violates triangle).
\item \textbf{Jaccard with measure on infinite sets}
\[
d(A,B)=1-\frac{\mu(A\cap B)}{\mu(A\cup B)}.
\]
In general only a pseudometric (distinct sets of equal a.e.\ class can have distance $0$).
\item \textbf{Mahalanobis with $\Sigma\succeq0$ but not $\succ0$.} Only a pseudometric: $d(x,y)=0$ can occur for $x\ne y$ along the nullspace.
\item \textbf{Ultrametric non--examples.}
\begin{itemize}
\item Standard $|\cdot|$ on $\mathbb{R}$ is \emph{not} ultrametric: with $0,1,2$, $d(0,2)=2$ but $\max\{d(0,1),d(1,2)\}=1$.
\item Hamming is not ultrametric in general: on $\{0,1\}^2$, $x=(0,0)$, $y=(0,1)$, $z=(1,1)$ give $d(x,z)=2>\max\{1,1\}$.
\end{itemize}
\end{itemize}

\subsection*{Closure tricks (to build new metrics)}
\begin{itemize}
\item If $d$ is a metric and $f:[0,\infty)\to[0,\infty)$ is increasing, concave, and $f(0)=0$, then $f\circ d$ is a metric.
Examples: $f(t)=\min\{1,t\}$; $f(t)=t^\alpha$ with $0<\alpha\le1$.
\item If $d_1,d_2$ are metrics, then $d_1+d_2$ and $\max\{d_1,d_2\}$ are metrics.
(But $\min\{d_1,d_2\}$ need not be a metric.)
\end{itemize}

\section*{Metric spaces vs. cosmological (FLRW) metric; what ``flat'' means}

\subsection*{1. Two notions of metric}
\paragraph{Metric space (analysis).}
A function $d:X\times X\to[0,\infty)$ is a metric if, for all $x,y,z\in X$,
\begin{align*}
\text{(M1)}\;& d(x,y)\ge 0,\quad d(x,y)=0 \iff x=y,\\
\text{(M2)}\;& d(x,y)=d(y,x),\\
\text{(M3)}\;& d(x,z)\le d(x,y)+d(y,z).
\end{align*}
It induces a topology and genuine (global) distances between points of $X$.

\paragraph{Spacetime metric (general relativity).}
A smooth symmetric $(0,2)$--tensor $g$ of Lorentzian signature $(-,+,+,+)$ on a 4D manifold.
Infinitesimal intervals are measured by
\[
ds^2 \;=\; g_{\mu\nu}(x)\,dx^\mu dx^\nu.
\]
Because of the indefinite signature, $ds^2$ can be negative/zero/positive (timelike/null/spacelike),
so $g$ is \emph{not} a metric in the metric-space sense. However, on a \emph{spatial slice} (fixing $t$),
the induced spatial tensor $g_{ij}$ is positive-definite and yields a bona fide metric space via path-length:
\[
d_{\text{spatial}}(x,y) \;=\; \inf_{\gamma:x\to y}\ \int_\gamma \sqrt{g_{ij}\,dx^i dx^j}\,.
\]

\subsection*{2. The FLRW line element and spatial curvature}
On large scales, the universe is modeled by the FLRW metric:
\[
ds^2 \;=\; -c^2\,dt^2 \;+\; a(t)^2\!\left(\frac{dr^2}{1-k r^2} + r^2\,d\Omega^2\right),
\qquad d\Omega^2 = d\theta^2 + \sin^2\!\theta\,d\phi^2,
\]
where:
\begin{itemize}
\item $a(t)$ is the scale factor (controls expansion),
\item $k\in\{-1,0,+1\}$ sets the \emph{spatial} curvature sign (open/flat/closed),
\item $(r,\theta,\phi)$ are comoving spherical coordinates.
\end{itemize}

\paragraph{Spatially \emph{flat} case ($k=0$).}
The spatial part becomes Euclidean up to scale:
\[
ds^2 \;=\; -c^2\,dt^2 \;+\; a(t)^2\big(dr^2 + r^2 d\Omega^2\big).
\]
Fixing $t=t_0$, the induced spatial metric is
\[
dl^2 \;=\; a(t_0)^2\big(dr^2 + r^2 d\Omega^2\big),
\]
i.e.\ a constant multiple of the standard Euclidean metric on $\mathbb{R}^3$.
Multiplying a metric by a positive constant preserves all metric-space axioms/topology:
distances are simply rescaled by $a(t_0)$.

\subsection*{3. Cosmological distances in the flat case ($k=0$)}
\begin{itemize}
\item \textbf{Comoving distance} $\chi$: the Euclidean distance in comoving coordinates
(it is time-independent for comoving objects).
\item \textbf{Proper distance at time $t$:}
\[
D_{\text{proper}}(t) \;=\; a(t)\,\chi.
\]
\item \textbf{Null curves (light):} photons follow $ds^2=0$, so light-cone relations connect $a(t)$
with observed redshift and with how we infer distances.
\end{itemize}

\subsection*{4. Spatially flat vs.\ fully flat (Minkowski)}
\begin{itemize}
\item \textbf{Spatially flat} ($k=0$): each constant-time slice has Euclidean geometry (up to scale).
\item \textbf{Minkowski spacetime}: globally flat in both space and time; effectively $a(t)\equiv 1$ (in suitable coords).
\end{itemize}
Thus, $k=0$ does \emph{not} imply that the entire spacetime is flat—if $a(t)$ changes, spacetime curvature
(through expansion) is nonzero even though each spatial slice is Euclidean.

\subsection*{5. Where ultrametrics fit (and where they don’t)}
An \emph{ultrametric} is a metric obeying the strong triangle inequality
\[
d(x,z)\ \le\ \max\{d(x,y),\,d(y,z)\}.
\]
It yields hierarchical, tree-like geometry (nested or disjoint balls). Standard cosmological spatial geometry
(FLRW slices with $k=-1,0,+1$) is Riemannian (Euclidean/hyperbolic/spherical), \emph{not} ultrametric.
Ultrametrics arise in $p$--adic models or hierarchical clustering, not in the usual large-scale spatial
metric of the universe.

\subsection*{6. Analysis--to--cosmology bridge (summary)}
Given a spatial slice $(\Sigma_t,g_{ij})$, define a metric-space distance by path-length:
\[
d_{\Sigma_t}(x,y) \;=\; \inf_{\gamma:x\to y} \int_\gamma \sqrt{g_{ij}\,dx^i dx^j}.
\]
For $k=0$, $g_{ij}=a(t)^2\,\delta_{ij}$ in comoving Cartesian coordinates, so
$d_{\Sigma_t}(x,y)=a(t)\,\|\vec x-\vec y\|_2$, a scaled Euclidean metric.

\section*{Hamming and Edit Distances on Fixed-Length Strings}

Let $X_L$ be the set of strings of length $L\in\mathbb{N}$ over a fixed alphabet. For $s,s'\in X_L$ define
\[
d_1(s,s')=\#\{\,i\in\{1,\dots,L\}: s[i]\ne s'[i]\}
\]
and let $d_2(s,s')$ be the minimum number of single-character insertions, deletions, or substitutions needed to transform $s$ into $s'$ (Levenshtein distance with unit costs).

\paragraph{Claim 1. $d_1$ is a metric on $X_L$.}
Nonnegativity and symmetry are immediate. Identity holds since $d_1(s,s')=0$ iff all coordinates match, i.e.\ $s=s'$.
For the triangle inequality, for each $i$,
\[
\mathbf 1_{\{s[i]\ne u[i]\}}
\ \le\
\mathbf 1_{\{s[i]\ne t[i]\}}+\mathbf 1_{\{t[i]\ne u[i]\}},
\]
and summing over $i=1,\dots,L$ yields $d_1(s,u)\le d_1(s,t)+d_1(t,u)$.

\paragraph{Claim 2. $d_2$ is a metric on $X_L$.}
Nonnegativity and identity are clear. Symmetry follows by reversing an edit script (interchanging insertions and deletions, keeping substitutions).
For the triangle inequality, let $\pi_{s\to t}$ and $\pi_{t\to u}$ be minimal scripts of lengths $d_2(s,t)$ and $d_2(t,u)$. Their concatenation is a valid script from $s$ to $u$ of length $d_2(s,t)+d_2(t,u)$, so minimality implies
\[
d_2(s,u)\le d_2(s,t)+d_2(t,u).
\]

\paragraph{Claim 3. On $X_L$, $d_2=d_1$.}
Let $k=d_1(s,s')$ be the number of mismatched positions.
\begin{itemize}
	\item (\emph{Upper bound}) Substitute at each mismatched position: $d_2(s,s')\le k$.
	\item (\emph{Lower bound}) Any edit transforming a length-$L$ string to another length-$L$ string uses equally many insertions and deletions. Using an insertion+deletion to fix one mismatch costs at least $2$, while a substitution costs $1$. Since each of the $k$ mismatches must be changed at least once, $d_2(s,s')\ge k$.
\end{itemize}
Thus $d_2(s,s')=k=d_1(s,s')$ for all $s,s'\in X_L$.

\paragraph{Conclusion.}
Both $d_1$ and $d_2$ are metrics on $X_L$. Moreover, for every $L\in\mathbb{N}\cup\{0\}$,
\[
d_2=d_1 \quad\text{on } X_L,
\]
so the two metrics are (trivially) equivalent and induce the same topology and the same Lipschitz geometry on $X_L$.

\section*{Levenshtein (Edit) Distance: Examples}

\paragraph{Definition.}
For strings $s,t$, the Levenshtein distance $d_2(s,t)$ is the minimum number of single-character \emph{insertions}, \emph{deletions}, or \emph{substitutions} to transform $s$ into $t$ (each edit has unit cost).

\paragraph{Examples.}

\subparagraph{(1) \texttt{kitten} $\to$ \texttt{sitting} \ (distance 3).}
Edits:
\begin{enumerate}
	\item Substitute $k\to s$: \texttt{kitten} $\to$ \texttt{sitten}.
	\item Substitute $e\to i$: \texttt{sitten} $\to$ \texttt{sittin}.
	\item Insert $g$ at end: \texttt{sittin} $\to$ \texttt{sitting}.
\end{enumerate}
Hence $d_2(\texttt{kitten},\texttt{sitting})=3$.

\subparagraph{(2) \texttt{flaw} $\to$ \texttt{lawn} \ (distance 2).}
Edits: delete \texttt{f} and insert \texttt{n} at end, so $d_2=2$.
(Equivalently, substitute $f\to l$ and insert $n$.)

\subparagraph{(3) \texttt{book} $\to$ \texttt{back} \ (distance 2).}
Edits: substitute the two \texttt{o}'s by \texttt{a} and \texttt{c}; thus $d_2=2$.

\subparagraph{(4) \texttt{intention} $\to$ \texttt{execution} \ (distance 5).}
A minimal edit script has $5$ edits (e.g., a sequence of substitutions plus one insertion), and the DP computation confirms $d_2=5$.

\paragraph{Dynamic-programming table for \texttt{flaw} vs \texttt{lawn}.}
Let $D_{i,j}$ be the distance between the prefixes of lengths $i$ and $j$ respectively. Initialize $D_{i,0}=i$, $D_{0,j}=j$, and use
\[
D_{i,j}=\min\left\{D_{i-1,j}+1,\ \ D_{i,j-1}+1,\ \ D_{i-1,j-1}+\mathbf 1_{s_i\ne t_j}\right\}.
\]
Filling the table (rows $s=\texttt{flaw}$, columns $t=\texttt{lawn}$):
\[
\begin{array}{c|ccccc}
	& \varnothing & \texttt{l} & \texttt{a} & \texttt{w} & \texttt{n}\\\hline
	\varnothing & 0 & 1 & 2 & 3 & 4\\
	\texttt{f}  & 1 & 1 & 2 & 3 & 4\\
	\texttt{l}  & 2 & 1 & 2 & 3 & 4\\
	\texttt{a}  & 3 & 2 & 1 & 2 & 3\\
	\texttt{w}  & 4 & 3 & 2 & 1 & 2
\end{array}
\]
Thus $d_2(\texttt{flaw},\texttt{lawn})=2$.

\paragraph{Remarks.}
If $|s|=|t|=L$, then $d_2(s,t)$ equals the Hamming distance: edit optimally via $L$ independent substitutions at mismatched positions.
Weighted variants (unequal edit costs) are computed by the same recurrence with the corresponding costs.

\section*{Problem}
Consider the following subsets of $\mathbb{R}$ with the Euclidean metric. Determine whether they are open or closed.
\begin{enumerate}[label=(\alph*)]
	\item $U=(0,1)\cup(2,3)$.
	\item $U=[0,1]$. What if we replace the metric space $\mathbb{R}$ by the metric space $[-1,1]$, the metric space $(0,1)$, or the metric space $[0,2]$, in each case with the subspace metric inherited from the Euclidean metric?
	\item $U=\displaystyle\bigcup_{n=1}^{\infty}(-n,\sin n)$.
\end{enumerate}

\section*{Solution}

\paragraph{Background (subspace).}
If $Y\subset\mathbb{R}$ has the subspace metric, then $A\subset Y$ is open in $Y$ iff $A=Y\cap O$ for some open $O\subset\mathbb{R}$, and $A$ is closed in $Y$ iff $A=Y\cap F$ for some closed $F\subset\mathbb{R}$.

\paragraph{(a) $U=(0,1)\cup(2,3)$ in $\mathbb{R}$.}
$U$ is a union of open intervals $\Rightarrow$ \emph{open}.  
It is \emph{not closed} since $1$ and $2$ are limit points not in $U$.

\paragraph{(b) $U=[0,1]$ under different ambients.}
\begin{itemize}
	\item In $\mathbb{R}$: $U$ is \emph{closed} (complement open), \emph{not open}.
	\item In $[-1,1]$: $U=[-1,1]\cap(-\infty,1]$ $\Rightarrow$ \emph{closed}; \emph{not open} (any subspace ball at $0$ contains negatives).
	\item In $(0,1)$: $U=[0,1]\cap(0,1)=(0,1)$ equals the whole space $\Rightarrow$ \emph{open and closed}.
	\item In $[0,2]$: $U=[0,2]\cap(-\infty,1]$ $\Rightarrow$ \emph{closed}; \emph{not open} (neighborhoods of $1$ include points $>1$).
\end{itemize}

\paragraph{(c) $U=\displaystyle\bigcup_{n=1}^{\infty}(-n,\sin n)$ in $\mathbb{R}$.}
Each $(-n,\sin n)$ is open, so $U$ is \emph{open}.  
Since $\{\sin n:n\in\mathbb{N}\}$ is dense in $[-1,1]$, every $y<1$ lies in some $(-n,\sin n)$, while $1\notin U$. Hence
\[
U=(-\infty,1),
\]
which is \emph{open} and \emph{not closed} (its closure is $(-\infty,1]$).

\section*{Problem}
Consider the following subsets of $\mathbb{R}^2$ with the Euclidean metric. Determine whether they are open or closed.
\begin{enumerate}[label=(\alph*)]
	\item $\{(x,y): y>0\}$.
	\item $\{(x,y): y\ne 0\}$.
	\item $\{(x,0): 0\le x\le 1\}$.
	\item $\{(x,0): 0< x< 1\}$.
	\item $\{(x,y): y/x\in\mathbb{N}\}\ \cup\ \{(x,y): x=0\}$.
	\item $\{(x,y): y/x\in\mathbb{Q}\}\ \cup\ \{(x,y): x=0\}$.
\end{enumerate}

\section*{Solution}
\paragraph{(a) $\{y>0\}$.}
Open, not closed. It is the preimage of $(0,\infty)$ under the continuous map $(x,y)\mapsto y$, and its closure is $\{y\ge0\}$.

\paragraph{(b) $\{y\ne0\}$.}
Open, not closed. It is $\mathbb{R}^2\setminus\{y=0\}$; the $x$--axis $\{y=0\}$ is closed, hence the set is open. The axis is not open, so the set is not closed.

\paragraph{(c) $\{(x,0): 0\le x\le1\}$.}
Closed, not open. It equals $\{y=0\}\cap\{0\le x\le1\}$ where both factors are closed (the latter is the preimage of $[0,1]$ under the $x$--projection). Any open ball in $\mathbb{R}^2$ leaves the $x$--axis, so the set is not open.

\paragraph{(d) $\{(x,0): 0<x<1\}$.}
Neither open nor closed. It has empty interior in $\mathbb{R}^2$ (hence not open), and its closure is the closed segment in (c), so it is not closed.

\paragraph{(e) $\{y/x\in\mathbb{N}\}\cup\{x=0\}$.}
Closed, not open. Each line $\{y=nx\}$ ($n\in\mathbb{N}$) and the axis $\{x=0\}$ is closed; to see the union is closed, let $(x_k,y_k)$ be in the union with $(x_k,y_k)\to (a,b)$. If $a=0$ the limit lies on $\{x=0\}$. If $a\ne0$, then $x_k\to a$, and the slopes $n_k=y_k/x_k\in\mathbb{N}$ cannot diverge (else $|y_k|=|n_k x_k|\to\infty$), so eventually $n_k$ is constant, say $n$, hence $b=na$. Thus the limit lies in the union. No line has interior, so the set is not open.

\paragraph{(f) $\{y/x\in\mathbb{Q}\}\cup\{x=0\}$.}
Neither open nor closed. Not open because it is a union of lines with empty interior. Not closed because rationals are dense: pick $q_k\in\mathbb{Q}$ with $q_k\to \alpha\notin\mathbb{Q}$ and choose $x_k\to a\ne0$; then $(x_k,q_k x_k)$ belong to the set and $(x_k,q_k x_k)\to (a,\alpha a)\notin$ the set.

\section*{Dense, Nowhere Dense, Open Dense: Definitions and Examples}

\paragraph{Dense.}
Let $(X,\tau)$ be a topological space. A set $A\subset X$ is \emph{dense} in $X$ if
\[
\overline{A}=X,
\]
equivalently, every nonempty open set $U\subset X$ satisfies $U\cap A\neq\varnothing$.

\paragraph{Nowhere dense.}
A set $A\subset X$ is \emph{nowhere dense} if the interior of its closure is empty:
\[
\operatorname{int}(\overline{A})=\varnothing.
\]
Equivalently: for every nonempty open $U\subset X$ there exists a nonempty open $V\subset U$ with
$V\cap A=\varnothing$.

\paragraph{Open dense.}
A set $A\subset X$ is \emph{open dense} if it is both open and dense. (These can be proper subsets of $X$.)

\paragraph{Meagre/comeagre (Baire category).}
A set is \emph{meagre} (first category) if it is a countable union of nowhere dense sets.
Its complement is \emph{comeagre} (residual); equivalently, a comeagre set is a countable intersection of open dense sets.

\bigskip

\subsection*{Examples in $\mathbb{R}$ (Euclidean topology)}
\begin{itemize}
	\item \textbf{Dense but not open:} $\mathbb{Q}$, and also $\mathbb{R}\setminus\mathbb{Q}$ (both have closure $\mathbb{R}$, neither contains an open interval).
	\item \textbf{Open but not dense:} $(0,1)$ (its closure is $[0,1]\ne\mathbb{R}$).
	\item \textbf{Open and dense:} $\mathbb{R}\setminus C$, where $C$ is the Cantor set (complement of a closed nowhere dense set).
	\item \textbf{Closed nowhere dense:} the Cantor set $C$ (closed; $\operatorname{int}(C)=\varnothing$).
	\item \textbf{Nowhere dense (not necessarily closed):} any finite set; $\mathbb{Z}$; $\{1/n:n\in\mathbb{N}\}$.
\end{itemize}

\subsection*{Quick implications (and non-implications)}
\begin{itemize}
	\item Dense $\centernot\Rightarrow$ open (e.g.\ $\mathbb{Q}$).
	\item Open $\centernot\Rightarrow$ dense (e.g.\ $(0,1)$).
	\item Dense \& open \emph{can} occur (e.g.\ $\mathbb{R}\setminus C$).
	\item Nowhere dense $\Rightarrow$ empty interior of $\overline{A}$ (hence $A$ itself has empty interior).
\end{itemize}

	
\section*{Problem}
Let $C([0,1])$ be the set of real-valued continuous functions on $[0,1]$.
With the uniform metric $d_\infty(f,g)=\sup_{x\in[0,1]}|f(x)-g(x)|$, determine whether the following sets are open or closed:
\begin{enumerate}[label=(\alph*)]
	\item $A=\{f\in C([0,1]) : f(1/2)=0\}$;
	\item $B=\{f\in C([0,1]) : \int_0^1 f=0\}$.
\end{enumerate}
Does the answer change if we replace $d_\infty$ with the $L^1$ metric $d_1(f,g)=\int_0^1 |f-g|$?

\section*{Solution}

\paragraph{Uniform (sup) metric.}
\begin{itemize}
	\item[(a)] $A$ is \emph{closed} and not open. The evaluation map $\mathrm{ev}_{1/2}:f\mapsto f(1/2)$ is continuous on $(C([0,1]),\|\cdot\|_\infty)$, hence $A=\mathrm{ev}_{1/2}^{-1}(\{0\})$ is closed. It is not open: for any $f\in A$ and $\varepsilon>0$, add a small continuous bump supported near $1/2$ to obtain $g$ with $\|g-f\|_\infty<\varepsilon$ but $g(1/2)\ne 0$.
	\item[(b)] $B$ is \emph{closed} and not open. The functional $I(f)=\int_0^1 f$ is continuous since $|I(f)-I(g)|\le \int_0^1 |f-g|\le \|f-g\|_\infty$. Thus $B=I^{-1}(\{0\})$ is closed. It is not open: perturb $f$ by a small positive (or negative) bump of arbitrarily small height/support to change the integral while keeping $\|\,\cdot\,\|_\infty$ small.
\end{itemize}

\paragraph{$L^1$ metric.}
\begin{itemize}
	\item[(a)] $A$ is \emph{neither open nor closed}. Evaluation at a point is not continuous in $L^1$.
	\begin{itemize}
		\item Not closed: let $g\equiv 1$, and let $f_n$ be continuous functions with $f_n(1/2)=0$ and $f_n\equiv 1$ outside $(\tfrac12-\tfrac1n,\tfrac12+\tfrac1n)$ (linear ramps in between). Then $f_n\in A$ and $\int_0^1|f_n-g|\le 2/n\to0$, so $f_n\to g$ in $L^1$ with $g\notin A$.
		\item Not open: given $f\in A$ and $\varepsilon>0$, add a very narrow spike near $1/2$ (choose height/width so its area $<\varepsilon$) producing $h$ with $d_1(h,f)<\varepsilon$ but $h(1/2)\ne 0$.
	\end{itemize}
	\item[(b)] $B$ is \emph{closed} and not open. The map $I(f)=\int_0^1 f$ is continuous in $L^1$ because $|I(f)-I(g)|\le \int_0^1|f-g|$. Hence $B=I^{-1}(\{0\})$ is closed. As before, small-area bumps show $B$ is not open.
\end{itemize}

\paragraph{Summary.}
Under $d_\infty$: $A$ closed/not open; $B$ closed/not open.  
Under $d_1$: $A$ neither open nor closed; $B$ closed/not open.

\section*{Evaluation map, continuity in the sup norm, and closedness of a zero set}

\paragraph{Evaluation map.}
For a fixed $x_0\in[0,1]$, the \emph{evaluation map}
\[
\operatorname{ev}_{x_0}: C([0,1]) \longrightarrow \mathbb{R},\qquad
\operatorname{ev}_{x_0}(f)=f(x_0),
\]
assigns to each continuous function its value at $x_0$.

\paragraph{Continuity in the uniform (sup) norm.}
Equip $C([0,1])$ with the norm $\|f\|_\infty=\sup_{x\in[0,1]}|f(x)|$ and the induced metric
$d_\infty(f,g)=\|f-g\|_\infty$. Then for any $f,g\in C([0,1])$,
\[
\big|\operatorname{ev}_{x_0}(f)-\operatorname{ev}_{x_0}(g)\big|
=\big|f(x_0)-g(x_0)\big|
\le \sup_{x\in[0,1]} |f(x)-g(x)|
=\|f-g\|_\infty.
\]
Hence $\operatorname{ev}_{x_0}$ is $1$-Lipschitz, in particular continuous, on $\big(C([0,1]),\|\cdot\|_\infty\big)$.

\paragraph{Closedness of the zero set.}
Let
\[
A=\{\,f\in C([0,1]) : f(x_0)=0\,\}=\operatorname{ev}_{x_0}^{-1}(\{0\}).
\]
Since $\operatorname{ev}_{x_0}$ is continuous and $\{0\}\subset\mathbb{R}$ is closed, $A$ is closed in $C([0,1])$.

\paragraph{Sequence proof (alternative).}
If $f_n\in A$ and $\|f_n-f\|_\infty\to 0$, then
\[
|f(x_0)|=|f(x_0)-f_n(x_0)|\le \|f-f_n\|_\infty \longrightarrow 0,
\]
so $f(x_0)=0$ and $f\in A$. Thus $A$ contains all its limit points and is closed.

\paragraph{Contrast: $L^1$ metric.}
With $d_1(f,g)=\int_0^1 |f-g|$, the map $\operatorname{ev}_{x_0}$ is not continuous in general
(small $L^1$ distance does not control the value at a single point),
so the set $\{f:f(x_0)=0\}$ need not be closed in $\big(C([0,1]),d_1\big)$.

\section*{Continuity and closed sets: preimages vs.\ images}

\paragraph{Fact (preimage of closed is closed).}
If $T:X\to Y$ is continuous and $F\subset Y$ is closed, then
\[
T^{-1}(F)\ \text{ is closed in } X.
\]
\emph{Reason.} Continuity is equivalent to: for every closed $F\subset Y$, the preimage $T^{-1}(F)$ is closed.

\paragraph{Application to the evaluation map.}
Let $\operatorname{ev}_{x_0}:C([0,1])\to\mathbb{R}$, $\operatorname{ev}_{x_0}(f)=f(x_0)$.  
In the sup norm, $\operatorname{ev}_{x_0}$ is $1$-Lipschitz, hence continuous:
\[
|f(x_0)-g(x_0)|\le \|f-g\|_\infty.
\]
Therefore, for $A=\{f\in C([0,1]):f(x_0)=0\}$ we have
\[
A=\operatorname{ev}_{x_0}^{-1}(\{0\}),
\]
and since $\{0\}$ is closed in $\mathbb{R}$, $A$ is closed in $C([0,1])$.

\paragraph{Caution: images of closed sets need not be closed.}
Continuity \emph{does not} imply that $T(F)$ is closed when $F$ is closed.
\[
\text{Example: } f:\mathbb{R}\to\mathbb{R},\ f(x)=e^x.
\quad \mathbb{R}\ \text{is closed, but } f(\mathbb{R})=(0,\infty)\ \text{is not closed.}
\]

\paragraph{When are images of closed sets closed?}
If $X$ is compact and $Y$ is Hausdorff, then for continuous $T:X\to Y$ and any closed $F\subset X$,
$F$ is compact $\Rightarrow$ $T(F)$ is compact $\Rightarrow$ $T(F)$ is closed in $Y$.
(Alternatively, this holds whenever $T$ is a \emph{closed map}.)

\section*{Problem}
Show that the Dirichlet function
\[
\mathbf{1}_{\mathbb{Q}}(x)=
\begin{cases}
	1,& x\in\mathbb{Q},\\
	0,& x\notin\mathbb{Q},
\end{cases}
\]
is not the pointwise limit of any sequence of continuous functions on $\mathbb{R}$.

\section*{Solution}
We use a standard Baire category fact (Baire--Osgood theorem):

\medskip
\emph{If $(f_n)$ are continuous functions on $\mathbb{R}$ and $f_n\to f$ pointwise, then $f$ has at least one point of continuity; in fact, the set of continuity points of $f$ is a dense $G_\delta$.}

\smallskip
Sketch of proof idea: In a Baire space, for each $k\in\mathbb{N}$ one shows the set
\[
G_k=\Bigl\{x\in\mathbb{R}:\ \exists\,\delta>0\text{ with }\operatorname{osc}(f;B(x,\delta))<\tfrac1k\Bigr\}
\]
is open and dense; then $\bigcap_{k=1}^\infty G_k$ is a dense $G_\delta$ consisting of continuity points of $f$.

Now, the Dirichlet function $\mathbf{1}_{\mathbb{Q}}$ is discontinuous at every $x\in\mathbb{R}$, since both $\mathbb{Q}$ and its complement are dense: every neighborhood of $x$ contains rationals and irrationals, so the oscillation in any neighborhood is $1$. Hence $\mathbf{1}_{\mathbb{Q}}$ has no continuity points.

By the theorem, it therefore cannot be the pointwise limit of continuous functions.
	
	
\section*{Problem}
For each sequence $(f_n)$ below, decide whether it converges uniformly on the given set $X$. If yes, find its uniform limit.
\begin{enumerate}[label=(\alph*)]
	\item $f_n(x)=x^{2n}$ on $X=(0,1)$.
	\item $f_n(x)=x e^{-nx}$ on $X=[0,\infty)$.
	\item $f_n(x)=e^{-x^2}\sin(x/n)$ on $X=\mathbb{R}$.
\end{enumerate}

\section*{Solution}

\paragraph{(a) $f_n(x)=x^{2n}$ on $(0,1)$.}
For each fixed $x\in(0,1)$, $x^{2n}\to 0$. However
\[
\sup_{x\in(0,1)}|x^{2n}-0|=\sup_{x\in(0,1)} x^{2n}=1,
\]
since values approach $1$ as $x\to1^{-}$. Hence the convergence is \emph{not uniform} on $(0,1)$.

\paragraph{(b) $f_n(x)=x e^{-nx}$ on $[0,\infty)$.}
Pointwise $f_n(x)\to 0$ for all $x\ge0$. For uniform convergence, maximize $g_n(x)=x e^{-nx}$:
with $y=nx$,
\[
g_n(x)=\frac{y}{n}e^{-y},\qquad \max_{y\ge0} y e^{-y}=e^{-1}\ \text{ at }y=1.
\]
Thus $\sup_{x\ge0} g_n(x)=\frac{1}{en}\to 0$, so $f_n\to 0$ \emph{uniformly} on $[0,\infty)$.

\paragraph{(c) $f_n(x)=e^{-x^2}\sin(x/n)$ on $\mathbb{R}$.}
Pointwise $f_n(x)\to 0$. Moreover,
\[
|f_n(x)|\le e^{-x^2}\cdot\frac{|x|}{n}\quad\text{since }|\sin t|\le |t|,
\]
hence
\[
\sup_{x\in\mathbb{R}}|f_n(x)|
\le \frac{1}{n}\,\sup_{x\in\mathbb{R}} |x|e^{-x^2}
= \frac{1}{n}\cdot \frac{e^{-1/2}}{\sqrt{2}}
\longrightarrow 0.
\]
Therefore $f_n\to 0$ \emph{uniformly} on $\mathbb{R}$.

\paragraph{Summary.}
(a) Not uniform; (b) uniform limit $0$; (c) uniform limit $0$.

\section*{Problem}
Let $(f_n)$ be a sequence of real-valued functions on a set $X$, pointwise bounded in the sense that
\[
\forall x\in X:\quad \sup_{n\in\mathbb{N}} |f_n(x)| \le g(x) < \infty
\]
for some $g:X\to[0,\infty)$.
\begin{enumerate}[label=(\alph*)]
	\item If $X=\mathbb{N}$, show that $(f_n)$ has a pointwise convergent subsequence.
	\item If $X=\mathbb{R}$, does $(f_n)$ still necessarily have a pointwise convergent subsequence?
\end{enumerate}

\section*{Solution}

\paragraph{(a) $X=\mathbb{N}$.}
For each fixed $m\in\mathbb{N}$ the real sequence $(f_n(m))_{n\ge1}$ is bounded, hence has a convergent subsequence.
Construct subsequences by diagonalization:
\begin{itemize}
	\item pick $(n^{(1)}_k)$ so that $f_{n^{(1)}_k}(1)$ converges;
	\item from that, pick $(n^{(2)}_k)$ so that $f_{n^{(2)}_k}(2)$ also converges;
	\item continue inductively.
\end{itemize}
Define $m_k:=n^{(k)}_k$. Then for each fixed $j$, the tail $(f_{m_k}(j))_{k\ge j}$ coincides with a tail of $(f_{n^{(j)}_k}(j))$, hence converges. Therefore $(f_{m_k})$ converges \emph{pointwise on $\mathbb{N}$}.

\paragraph{(b) $X=\mathbb{R}$.}
Not necessarily. Consider the Rademacher functions on $[0,1]$,
\[
r_n(x)=\operatorname{sign}(\sin(2^n\pi x))\in\{-1,1\},
\]
extended to $\mathbb{R}$ periodically. They satisfy $|r_n(x)|\le 1$ for all $x,n$, so the family is pointwise bounded.

We claim no subsequence of $(r_n)$ converges pointwise on $[0,1]$. Let $(r_{n_k})$ be any subsequence. Choose $x\in[0,1]$ whose binary digits at positions $n_k$ alternate $0,1,0,1,\dots$ (fill other digits arbitrarily and avoid a terminating expansion). For this $x$,
\[
r_{n_k}(x)=(-1)^{\text{binary digit of $x$ at position $n_k$}}
\]
alternates between $+1$ and $-1$, hence does not converge. Thus no subsequence converges pointwise on all of $[0,1]$ (and hence not on $\mathbb{R}$).

\paragraph{Conclusion.}
(a) Yes: a pointwise convergent subsequence exists by diagonalization on the countable domain.
(b) No: on $\mathbb{R}$, pointwise boundedness alone does not ensure a pointwise convergent subsequence.

\section*{Problem}
Prove that a bounded sequence in $\mathbb{R}^n$ must have a convergent subsequence.
\emph{Hint:} You know this for $n=1$; try induction on $n$.

\section*{Solution (Bolzano--Weierstrass in $\mathbb{R}^n$)}
\paragraph{Background.}
(1) In $\mathbb{R}$, every bounded sequence has a convergent subsequence (Bolzano--Weierstrass).
(2) In $\mathbb{R}^n$, $x_k=(x_k^{(1)},\dots,x_k^{(n)})\to x$ iff $x_k^{(i)}\to x^{(i)}$ in $\mathbb{R}$ for each $i$.

\paragraph{Induction on $n$.}
For $n=1$ the statement is Bolzano--Weierstrass.

Assume it holds for $n-1$. Let $(x_k)$ be a bounded sequence in $\mathbb{R}^n$. Write $x_k=(a_k,b_k)$ with $a_k\in\mathbb{R}$ and $b_k\in\mathbb{R}^{n-1}$. Since $(a_k)$ is bounded, by Bolzano--Weierstrass there exists a subsequence $(x_{k_1(j)})$ such that $a_{k_1(j)}\to a\in\mathbb{R}$.

The sequence $(b_{k_1(j)})$ is bounded in $\mathbb{R}^{n-1}$, so by the induction hypothesis there is a further subsequence $(x_{k_2(j)})$ of $(x_{k_1(j)})$ with $b_{k_2(j)}\to b\in\mathbb{R}^{n-1}$.

Therefore $x_{k_2(j)}=(a_{k_2(j)},b_{k_2(j)})\to (a,b)$ in $\mathbb{R}^n$ (coordinatewise convergence). Hence $(x_k)$ has a convergent subsequence. \qed
	
	
\section*{Continuity via neighborhoods and preimages of open/closed sets}

\paragraph{Problem.}
Let $(X,d)$ and $(Y,d')$ be metric spaces and $f:X\to Y$.
\begin{enumerate}[label=(\alph*)]
	\item Show that $f$ is continuous at $x\in X$ iff for every neighborhood $V\subset Y$ of $f(x)$, the preimage $U=f^{-1}(V)$ is a neighborhood of $x$.
	\item Deduce: $f$ is continuous on $X$ iff for every open $V\subset Y$, the set $f^{-1}(V)$ is open in $X$.
	\item Show: $f$ is continuous on $X$ iff for every closed $F\subset Y$, the set $f^{-1}(F)$ is closed in $X$.
\end{enumerate}

\paragraph{Background.}
In a metric space, $V$ is a neighborhood of $p$ iff there exists $r>0$ with $B(p,r)\subset V$.
The $\varepsilon$--$\delta$ definition at $x$ says:
\[
\forall\varepsilon>0\ \exists\delta>0\ \text{s.t. } d(x,y)<\delta \implies d'(f(x),f(y))<\varepsilon.
\]

\paragraph{Solution to (a).}
($\Rightarrow$) Suppose $f$ is continuous at $x$ and let $V$ be a neighborhood of $f(x)$. Choose $r>0$ with $B_Y(f(x),r)\subset V$. By continuity at $x$, there is $\delta>0$ such that $d(x,y)<\delta\Rightarrow d'(f(x),f(y))<r$. Hence $B_X(x,\delta)\subset f^{-1}(B_Y(f(x),r))\subset f^{-1}(V)$, so $f^{-1}(V)$ is a neighborhood of $x$.

($\Leftarrow$) Conversely, suppose $f^{-1}(V)$ is a neighborhood of $x$ for every neighborhood $V$ of $f(x)$. Let $\varepsilon>0$ and set $V=B_Y(f(x),\varepsilon)$. Then $f^{-1}(V)$ is a neighborhood of $x$, so $B_X(x,\delta)\subset f^{-1}(V)$ for some $\delta>0$. Thus $d(x,y)<\delta\Rightarrow d'(f(x),f(y))<\varepsilon$. Hence $f$ is continuous at $x$.

\paragraph{Solution to (b).}
($\Rightarrow$) If $f$ is continuous on $X$ and $V\subset Y$ is open, then for each $x\in f^{-1}(V)$ the set $V$ is a neighborhood of $f(x)$, so by (a) $f^{-1}(V)$ is a neighborhood of $x$. Hence $f^{-1}(V)$ is open.

($\Leftarrow$) If $f^{-1}(V)$ is open for every open $V\subset Y$, then fixing $x\in X$ and $\varepsilon>0$ we consider $V=B_Y(f(x),\varepsilon)$. The openness of $f^{-1}(V)$ implies there exists $\delta>0$ with $B_X(x,\delta)\subset f^{-1}(V)$, which is exactly the $\varepsilon$--$\delta$ condition at $x$. Thus $f$ is continuous on $X$.

\paragraph{Solution to (c).}
($\Rightarrow$) If $f$ is continuous and $F\subset Y$ is closed, then $X\setminus f^{-1}(F)=f^{-1}(Y\setminus F)$ is open by (b); hence $f^{-1}(F)$ is closed.

($\Leftarrow$) If $f^{-1}(F)$ is closed for every closed $F\subset Y$, then preimages of open sets (complements of closed sets) are open, so by (b) $f$ is continuous.

\paragraph{Examples.}
\begin{itemize}
	\item $f:\mathbb{R}\to\mathbb{R}$, $f(x)=x^3$ is continuous: inverse images of open intervals are open.
	\item The step function $f=\mathbf{1}_{(0,\infty)}$ is discontinuous at $0$: the neighborhood $(-\tfrac12,\tfrac12)$ of $f(0)=0$ pulls back to $(-\infty,0]$, not a neighborhood of $0$.
	\item Constant maps $f(x)=c$ are continuous: preimages of opens are either $\emptyset$ or $X$.
\end{itemize}
\section*{More continuity examples (via preimages and neighborhoods)}

\begin{enumerate}
	\item \textbf{Polynomials and compositions.}
	$f:\mathbb{R}\to\mathbb{R}$, $f(x)=x^5-3x+1$. Continuous; for any open $V$, $f^{-1}(V)$ is open.
	Compositions and finite sums/products of continuous functions are continuous.
	
	\item \textbf{Rational functions on their domain.}
	$f(x)=\dfrac{x^2-1}{x-1}$ is continuous on $\mathbb{R}\setminus\{1\}$. For open $V\subset\mathbb{R}$,
	$f^{-1}(V)=(\mathbb{R}\setminus\{1\})\cap O$ for some open $O\subset\mathbb{R}$.
	
	\item \textbf{Absolute value.}
	$f(x)=|x|$ is continuous. For $V=(a,b)$ with $0\le a<b$,
	$f^{-1}(V)=(-b,-a)\cup(a,b)$ (open). Also $||x|-|x_0||\le|x-x_0|$.
	
	\item \textbf{Floor function (discontinuous).}
	$\lfloor x\rfloor$ has jump discontinuities at integers. For instance, $V=(-\tfrac12,\tfrac12)$ is a neighborhood of $f(0)=0$, but $f^{-1}(V)=\{0\}$ is not a neighborhood of $0$.
	
	\item \textbf{Indicators of open sets (typically discontinuous).}
	If $A\subset\mathbb{R}$ is nontrivial open and $f=\mathbf{1}_A$, then every boundary point of $A$ is a discontinuity: $f^{-1}\big((\tfrac12,2)\big)=A$ fails to be a neighborhood of any boundary point.
	
	\item \textbf{Distance to a set: $1$-Lipschitz, hence continuous.}
	In a metric space $(X,d)$ with $A\subset X$, define $f(x)=d(x,A)$. Then
	$|f(x)-f(y)|\le d(x,y)$.
	Moreover $f^{-1}\big((0,\varepsilon)\big)=\{x:\ d(x,A)<\varepsilon\}$, the open $\varepsilon$-neighborhood of $A$.
	
	\item \textbf{Projections are continuous (and open).}
	$\pi_1:X\times Y\to X$, $\pi_1(x,y)=x$.
	For open $U\subset X$, $\pi_1^{-1}(U)=U\times Y$ is open.
	If $W\subset X\times Y$ is open, $\pi_1(W)$ is open (projections are open maps).
	
	\item \textbf{Max/min of continuous functions.}
	If $f,g$ are continuous, then
	\[
	\max(f,g)=\tfrac12\big(f+g+|f-g|\big),\qquad
	\min(f,g)=\tfrac12\big(f+g-|f-g|\big)
	\]
	are continuous. Equivalently, $\{\max(f,g)<\alpha\}=\{f<\alpha\}\cap\{g<\alpha\}$ is open.
	
	\item \textbf{Uniform limits preserve continuity.}
	If $f_n\to f$ uniformly and each $f_n$ is continuous, then $f$ is continuous.
	Example: $f_n(x)=\sin(x)/n\to 0$ uniformly on $\mathbb{R}$.
	
	\item \textbf{Inclusion and restriction.}
	If $Y\subset X$ (subspace topology), the inclusion $i:Y\hookrightarrow X$ is continuous.
	If $f:X\to Z$ is continuous, then $f|_Y$ is continuous.
	
	\item \textbf{Continuous need not be closed.}
	$f:\mathbb{R}\to\mathbb{R}$, $f(x)=e^x$ is continuous but sends the closed set $\mathbb{R}$ to $(0,\infty)$, which is not closed.
\end{enumerate}

\section*{Problem}
Let $(f_n)$ be real-valued, uniformly continuous functions on $\mathbb{R}$.
\begin{enumerate}[label=(\alph*)]
	\item If $f_n \Rightarrow f$ uniformly on $\mathbb{R}$, prove that $f$ is uniformly continuous.
	\item Does the conclusion still hold if we only assume $f_n \to f$ pointwise? Give a proof or counterexample.
\end{enumerate}

\section*{Solution}

\paragraph{(a) Uniform limits preserve uniform continuity.}
Fix $\varepsilon>0$. Choose $N$ so that $\|f_N-f\|_\infty<\varepsilon/3$ (uniform convergence).
Since $f_N$ is uniformly continuous, there exists $\delta>0$ such that
$|x-y|<\delta \implies |f_N(x)-f_N(y)|<\varepsilon/3$.
Then for all $x,y$ with $|x-y|<\delta$,
\[
|f(x)-f(y)|
\le |f(x)-f_N(x)| + |f_N(x)-f_N(y)| + |f_N(y)-f(y)|
< \varepsilon/3 + \varepsilon/3 + \varepsilon/3
= \varepsilon.
\]
Hence $f$ is uniformly continuous.

\paragraph{(b) Pointwise convergence need not preserve uniform continuity.}
Consider $f_n(x)=\arctan(n x)$. Each $f_n$ is uniformly continuous (indeed $|f'_n(x)|=\frac{n}{1+n^2x^2}\le 1$).
Pointwise,
\[
f(x)=\lim_{n\to\infty} f_n(x)=
\begin{cases}
	\frac{\pi}{2}, & x>0,\\
	0, & x=0,\\
	-\frac{\pi}{2}, & x<0,
\end{cases}
\]
which is discontinuous at $0$, hence not uniformly continuous. Therefore pointwise convergence does not suffice.

\paragraph{Examples.}
(1) $f_n(x)=\frac{x}{n+1}\to 0$ uniformly; $0$ is uniformly continuous. \quad
(2) $f_n(x)=\sqrt{x^2+\frac1n}\to |x|$ uniformly on $\mathbb{R}$; $|x|$ is uniformly continuous.

\section*{Uniform continuity vs.\ uniform convergence (cheat sheet)}

\paragraph{Setting.} Let $(X,d)$ be a metric space and $f,f_n:X\to\mathbb{R}$.

\subsection*{Definitions}
\begin{itemize}
	\item \textbf{Uniform continuity of $f$.} 
	$f$ is uniformly continuous on $X$ if
	\[
	\forall \varepsilon>0\ \exists \delta>0\ \text{such that}\quad
	d(x,y)<\delta \ \Rightarrow\ |f(x)-f(y)|<\varepsilon
	\quad\text{for all }x,y\in X.
	\]
	(The same $\delta$ works everywhere in $X$.)
	\item \textbf{Uniform convergence of $(f_n)$ to $f$.}
	We write $f_n \Rightarrow f$ on $X$ if
	\[
	\forall \varepsilon>0\ \exists N\in\mathbb{N}\ \text{such that}\quad
	n\ge N \ \Rightarrow\ \sup_{x\in X}|f_n(x)-f(x)|<\varepsilon.
	\]
	(The same $N$ works for all $x\in X$.)
\end{itemize}

\subsection*{What they control}
\begin{itemize}
	\item Uniform continuity controls how \emph{one function} varies with \emph{inputs getting close} ($x\to y$).
	\item Uniform convergence controls how \emph{a sequence of functions} approaches its limit as \emph{the index grows} ($n\to\infty$).
\end{itemize}

\subsection*{Key theorems}
\begin{itemize}
	\item \textbf{Heine--Cantor.} If $X$ is compact and $f$ is continuous on $X$, then $f$ is uniformly continuous.
	\item \textbf{Uniform limit theorem.} If each $f_n$ is continuous and $f_n\Rightarrow f$ uniformly, then $f$ is continuous.
	Moreover, if each $f_n$ is uniformly continuous and $f_n\Rightarrow f$ uniformly, then $f$ is uniformly continuous.
\end{itemize}

\subsection*{Non-implications / pitfalls}
\begin{itemize}
	\item Continuous $\nRightarrow$ uniformly continuous on noncompact domains.  
	Example: $f(x)=x^2$ on $\mathbb{R}$ is not uniformly continuous.
	\item Pointwise convergence $\nRightarrow$ uniform convergence.  
	Example: $f_n(x)=x^n$ on $[0,1]$ converges pointwise to $f=\mathbf{1}_{\{1\}}$ but not uniformly on $(0,1)$.
	\item Pointwise limit of uniformly continuous functions need not be uniformly continuous.  
	Example: $f_n(x)=\arctan(nx)$ (each UC) $\to$ a step function, which is not UC.
\end{itemize}

\subsection*{Quick examples}
\begin{itemize}
	\item \textbf{UC example:} $f(x)=\sin x$ is uniformly continuous on $\mathbb{R}$ (Lipschitz with constant $1$).
	\item \textbf{Uniform convergence example:} $f_n(x)=\frac{x}{n}\Rightarrow 0$ uniformly on $\mathbb{R}$; $0$ is uniformly continuous.
	\item \textbf{Failure of uniform convergence:} $f_n(x)=x^n$ on $[0,1]$ is not uniformly convergent on $(0,1)$ since
	$\sup_{x\in(0,1)}|x^n-0|=1$ for all $n$.
\end{itemize}

\section*{Problem}
Let $X=\mathbb{Q}\cap[0,1]$ with the Euclidean metric.
\begin{enumerate}
	\item[(i)] Show that $X$ is totally bounded.
	\item[(ii)] Using the definition of compactness, show that $X$ is not compact.
	\item[(iii)] Explain why this does not contradict the Heine--Borel theorem.
\end{enumerate}

\section*{Solution}

\paragraph{(i) Total boundedness.}
Let $\varepsilon>0$ and choose $N\in\mathbb{N}$ with $1/N<\varepsilon$. Partition $[0,1]$ into $N$ subintervals
$[k/N,(k+1)/N]$ for $k=0,\dots,N-1$. Since $\mathbb{Q}$ is dense, pick $q_k\in\mathbb{Q}\cap(k/N,(k+1)/N)$.
Then every $x\in X$ lies in some $[k/N,(k+1)/N]$ and satisfies $|x-q_k|<1/N<\varepsilon$. Hence
\[
X\ \subset\ \bigcup_{k=0}^{N-1} B_X(q_k,\varepsilon),
\]
a finite $\varepsilon$-cover. Therefore $X$ is totally bounded.

\paragraph{(ii) Non-compactness via an open cover.}
Fix an irrational $a\in(0,1)$ and define, for $n\in\mathbb{N}$,
\[
U_n = X \cap\Big( (-\infty,a-1/n)\ \cup\ (a+1/n,\infty)\Big).
\]
Each $U_n$ is open in the subspace $X$, and $\{U_n\}_{n\in\mathbb{N}}$ covers $X$:
for any $x\in X$ with $x\ne a$, choose $n>\frac1{|x-a|}$ so that $x\in U_n$.
No finite subfamily covers $X$: if $n_0$ is maximal among a finite selection, then rational points in
$X\cap(a-1/n_0,a+1/n_0)$ remain uncovered. Hence $X$ is not compact.

\emph{Sequential variant.} Choose a sequence $(x_k)\subset X$ with $x_k\to a$ (irrational).
Then $(x_k)$ has no convergent subsequence in $X$ (its only possible limit is $a\notin X$), so $X$ is not compact.

\paragraph{(iii) Consistency with Heine--Borel.}
Heine--Borel states that a subset of $\mathbb{R}$ is compact iff it is closed and bounded.
The set $X$ is bounded but not closed (its closure is $[0,1]$), hence not compact. Moreover, in metric spaces,
compact $\iff$ complete and totally bounded; $X$ is totally bounded but not complete, so it is not compact.
Thus there is no contradiction.

\section*{Total boundedness vs.\ compactness (in metric spaces)}

\paragraph{Key implications (metric spaces).}
Let $(X,d)$ be a metric space.
\begin{enumerate}
	\item \textbf{Compact $\Rightarrow$ complete and totally bounded.}
	\begin{itemize}
		\item Total boundedness: for any $\varepsilon>0$, the open cover $\{B(x,\varepsilon):x\in X\}$ has a finite subcover by compactness.
		\item Completeness: every Cauchy sequence has a convergent subsequence (by compactness); uniqueness of limits in metric spaces forces the whole sequence to converge.
	\end{itemize}
	
	\item \textbf{Totally bounded $+$ complete $\Rightarrow$ compact.}
	\begin{itemize}
		\item Proof sketch (sequential form): given a sequence $(x_n)$ in $X$, total boundedness yields a Cauchy subsequence (by iteratively choosing finer $\varepsilon$-nets). Completeness gives a convergent subsequence; hence $X$ is sequentially compact, which is equivalent to compactness in metric spaces.
	\end{itemize}
	
	\item \textbf{Totally bounded $\centernot\Rightarrow$ compact (in general).}
	\begin{itemize}
		\item Counterexample: $X=\mathbb{Q}\cap[0,1]$ is totally bounded (sit inside a bounded interval), but not compact since it is not complete (e.g.\ a rational sequence converging to an irrational in $[0,1]$ has no limit in $X$).
	\end{itemize}
\end{enumerate}

\paragraph{Euclidean specialization ($\mathbb{R}^n$).}
In $\mathbb{R}^n$ with the Euclidean metric,
\[
\text{compact} \ \Longleftrightarrow\ \text{closed and bounded} \quad\text{(Heine--Borel)}
\]
and also
\[
\text{compact} \ \Longleftrightarrow\ \text{complete and totally bounded}.
\]
Here ``bounded'' already implies ``totally bounded'', and ``closed'' implies completeness (closed subsets of complete spaces are complete). Thus, in $\mathbb{R}^n$,
\[
\boxed{\ \text{closed and bounded} \ \Longleftrightarrow\ \text{complete and totally bounded} \ \Longleftrightarrow\ \text{compact}\ }.
\]


\section*{Totally bounded sets: definition, examples, and counterexamples}

\paragraph{Definition (metric space $(X,d)$).}
A set $A\subset X$ is \emph{totally bounded} if for every $\varepsilon>0$ there exist
$x_1,\dots,x_m\in X$ such that
\[
A \subset \bigcup_{j=1}^m B(x_j,\varepsilon).
\]

\subsection*{Examples}
\begin{enumerate}
	\item \textbf{Compact $\Rightarrow$ totally bounded.}
	Every compact metric space is totally bounded (apply compactness to the cover by all $\varepsilon$-balls).
	
	\item \textbf{Bounded subsets of $\mathbb{R}^n$ (Euclidean).}
	In $\mathbb{R}^n$ with the Euclidean metric, bounded $\Leftrightarrow$ totally bounded:
	cover by finitely many grid cubes of side $<\varepsilon/\sqrt n$.
	
	\item \textbf{$\mathbb{Q}\cap[0,1]$.}
	Pick rationals in a partition of $[0,1]$ into intervals of length $<\varepsilon$ to form a finite $\varepsilon$-net.
	
	\item \textbf{Cantor set (and any compact subset of $\mathbb{R}$).}
	Compact sets are totally bounded.
	
	\item \textbf{Continuous images.}
	If $A$ is totally bounded and $f:X\to Y$ is continuous, then $f(A)$ is totally bounded:
	push forward a finite $\varepsilon$-net; for Lipschitz $f$ with constant $L$, an
	$\varepsilon/L$-net of $A$ maps to an $\varepsilon$-net of $f(A)$.
	
	\item \textbf{Finite products.}
	If $X_1,\dots,X_k$ are totally bounded, then $X_1\times\cdots\times X_k$ is totally bounded
	in any product metric equivalent to $d_\infty((x_i),(y_i))=\max_i d_i(x_i,y_i)$.
\end{enumerate}

\subsection*{Counterexamples (not totally bounded)}
\begin{enumerate}
	\item \textbf{Unbounded subsets of $\mathbb{R}^n$.}
	$\mathbb{R}$, $(0,\infty)$, or $\mathbb{Z}$ (Euclidean metric): for $\varepsilon=1$ no finite union of unit balls covers them.
	
	\item \textbf{Infinite discrete metric space.}
	Let $X$ be infinite with $d(x,y)=\mathbf{1}_{x\ne y}$. Then $\operatorname{diam}(X)=1$ (bounded) but for any $\varepsilon<1$,
	$B(x,\varepsilon)=\{x\}$, so no finite cover exists.
	
	\item \textbf{Unit ball of an infinite-dimensional Banach space (e.g.\ $\ell^2$).}
	$\mathbb{B}=\{x\in\ell^2:\|x\|\le1\}$ is not totally bounded: the unit vectors $e_n\in\mathbb{B}$ satisfy
	$\|e_n-e_m\|=\sqrt2$ for $n\ne m$, so $\varepsilon<1$ balls contain at most one $e_n$.
	
	\item \textbf{$\mathbb{N}$ with $d(m,n)=|m-n|$.}
	Unbounded, hence not totally bounded.
\end{enumerate}

\subsection*{Closure properties (quick reference)}
\begin{itemize}
	\item Subsets of totally bounded sets are totally bounded.
	\item Finite unions of totally bounded sets are totally bounded.
	\item Continuous images of totally bounded sets are totally bounded.
	\item In metric spaces: \emph{compact $\Rightarrow$ totally bounded}; and \emph{compact $\Leftrightarrow$ complete $+$ totally bounded}.
\end{itemize}

\section*{Heine--Borel and Heine--Cantor in Euclidean spaces (sequential proofs)}

\paragraph{Sequential facts in $\mathbb{R}^n$.}
In metric spaces (hence in $\mathbb{R}^n$), compact $\Leftrightarrow$ sequentially compact and
closed $\Leftrightarrow$ sequentially closed. Bolzano--Weierstrass: every bounded sequence in $\mathbb{R}^n$
has a convergent subsequence.

\subsection*{(a) Heine--Borel in $\mathbb{R}^n$: $K$ compact $\iff$ $K$ closed and bounded}

\paragraph{($\Rightarrow$) Compact $\Rightarrow$ closed and bounded.}
Bounded: cover $K$ by the open balls $B(0,m)$ ($m\in\mathbb{N}$). Compactness gives a finite subcover, hence
$K\subset B(0,M)$ for some $M$.

Closed: let $x_k\in K$ with $x_k\to x$. Since $K$ is compact, every sequence has a convergent subsequence
with limit in $K$; but limits in $\mathbb{R}^n$ are unique, so $x\in K$. Thus $K$ is sequentially closed, hence closed.

\paragraph{($\Leftarrow$) Closed and bounded $\Rightarrow$ compact.}
Let $(x_k)\subset K$. Boundedness $\Rightarrow$ by Bolzano--Weierstrass there is a subsequence $x_{k_j}\to x$.
Closedness of $K$ $\Rightarrow$ $x\in K$. Hence every sequence in $K$ admits a convergent subsequence with
limit in $K$: $K$ is sequentially compact, hence compact.

\subsection*{(b) Heine--Cantor on $[a,b]$: continuity $\Rightarrow$ uniform continuity}
Assume $f:[a,b]\to\mathbb{R}$ is continuous but not uniformly continuous. Then there exists $\varepsilon_0>0$
and $x_n,y_n\in[a,b]$ with
\[
|x_n-y_n|<\frac{1}{n}\quad\text{and}\quad |f(x_n)-f(y_n)|\ge \varepsilon_0\ \ \text{for all }n.
\]
Since $[a,b]$ is closed and bounded, it is compact by (a). Thus $(x_n)$ has a subsequence $(x_{n_k})$ with $x_{n_k}\to x^*\in[a,b]$.
Because $|x_{n_k}-y_{n_k}|\to 0$, we also have $y_{n_k}\to x^*$.

By sequential continuity of $f$ at $x^*$, we have
\[
f(x_{n_k}) \to f(x^*), \qquad f(y_{n_k}) \to f(x^*).
\]
Hence $|f(x_{n_k}) - f(y_{n_k})| \to 0$, contradicting $|f(x_{n_k}) - f(y_{n_k})| \ge \varepsilon_0$.
Therefore $f$ is uniformly continuous on $[a,b]$. \qed

\section*{Completeness of $C^k([a,b])$ and continuity of integration/derivative maps}

\section*{Problem}
For each $k\in\mathbb{N}_0$, define the metric space $C^k([a,b])$ as the space of $k$ times continuously differentiable real-valued functions on $[a,b]$ endowed with the metric
\[
d_{C^k}(f,g)=\sum_{j=0}^{k}\ \sup_{x\in[a,b]}\,\Bigl|\frac{d^j f}{dx^j}(x)-\frac{d^j g}{dx^j}(x)\Bigr|.
\]
\begin{enumerate}[label=(\alph*)]
	\item Show that $C^k([a,b])$ is a complete metric space.
	\item Show that the map $f \mapsto \displaystyle\Big(x\mapsto \int_a^x f(y)\,dy\Big)$ is a continuous map from $C^k([a,b])$ to $C^{k+1}([a,b])$.
	\item Show that the map $f \mapsto f'$ is a continuous map from $C^{k+1}([a,b])$ to $C^k([a,b])$.
\end{enumerate}


Let
\[
d_{C^k}(f,g)=\sum_{j=0}^k \sup_{x\in[a,b]} \bigl| f^{(j)}(x)-g^{(j)}(x)\bigr|
\quad\text{on } C^k([a,b]).
\]

\paragraph{(a) $C^k([a,b])$ is complete.}
Let $(f_n)$ be Cauchy in $d_{C^k}$. Then for each $j=0,\dots,k$ the sequence
$(f_n^{(j)})$ is Cauchy in $(C([a,b]),\|\cdot\|_\infty)$, hence converges uniformly to some
$g_j\in C([a,b])$.

Define inductively $g_{j-1}(x)=g_{j-1}(a)+\int_a^x g_j(t)\,dt$ for $j=k,k-1,\dots,1$.
By the uniform convergence of derivatives and the fundamental theorem of calculus,
$g_{j-1}$ is differentiable with $(g_{j-1})'=g_j$. Setting $f:=g_0$, we have
$f\in C^k([a,b])$ and $f^{(j)}=g_j$ for all $0\le j\le k$. Since
$f_n^{(j)}\to f^{(j)}$ uniformly for each $j$, it follows that
$d_{C^k}(f_n,f)\to 0$. Thus $C^k([a,b])$ is complete.

\paragraph{(b) The map $T:C^k([a,b])\to C^{k+1}([a,b])$, $T(f)(x)=\int_a^x f(y)\,dy$, is continuous.}
For $f,g\in C^k$ let $F=T(f)$ and $G=T(g)$. Then $F\in C^{k+1}$ with
$F^{(1)}=f$ and $F^{(j+1)}=f^{(j)}$ for $1\le j\le k$. Moreover,
\[
\|F-G\|_\infty \le (b-a)\,\|f-g\|_\infty,\qquad
\|F^{(j)}-G^{(j)}\|_\infty = \|f^{(j-1)}-g^{(j-1)}\|_\infty\ (1\le j\le k+1).
\]
Hence
\[
d_{C^{k+1}}(T(f),T(g))
\le \max\{b-a,1\}\,\Big(\|f-g\|_\infty+\sum_{j=1}^k \|f^{(j)}-g^{(j)}\|_\infty\Big)
= \max\{b-a,1\}\, d_{C^k}(f,g),
\]
so $T$ is Lipschitz and therefore continuous.

\paragraph{(c) The derivative map $D:C^{k+1}([a,b])\to C^k([a,b])$, $D(f)=f'$, is continuous.}
For $f,g\in C^{k+1}$,
\[
d_{C^k}(f',g')=\sum_{j=0}^k \|f^{(j+1)}-g^{(j+1)}\|_\infty
\le \sum_{j=0}^{k+1} \|f^{(j)}-g^{(j)}\|_\infty
= d_{C^{k+1}}(f,g).
\]
Thus $D$ is $1$-Lipschitz (hence continuous). \qed

	
\section*{Examples and counterexamples for $C^k([a,b])$ with $d_{C^k}$}

\subsection*{Examples}

\begin{enumerate}
	
	\item \textbf{Cauchy $\Rightarrow$ convergent in $C^k$.}
	
	On $[0,1]$, let
	\[
	f_n(x)=\sum_{m=0}^{n}\frac{x^{m}}{m!}.
	\]
	Each $f_n\in C^\infty$.  
	For every fixed $j\in\{0,\dots,k\}$ we have
	\[
	\sup_{x\in[0,1]}\bigl|\,f_n^{(j)}(x)-e^{(\cdot)\,(j)}(x)\,\bigr|\;\longrightarrow\;0.
	\]
	Hence $(f_n)$ is Cauchy in $d_{C^k}$ and therefore
	\[
	f_n \;\xrightarrow[n\to\infty]{}\; e^x \quad\text{in } C^k([0,1]).
	\]
	
	\medskip
	
	\item \textbf{Integration map is Lipschitz $C^k\!\to C^{k+1}$.}
	
	For $f,g\in C^k$ put $F=T(f)$ and $G=T(g)$ with
	\[
	F(x)=\int_a^x f(y)\,dy, \qquad G(x)=\int_a^x g(y)\,dy .
	\]
	Then
	\[
	\|F-G\|_\infty \;\le\; (b-a)\,\|f-g\|_\infty,
	\]
	and for $1\le j\le k+1$,
	\[
	\|F^{(j)}-G^{(j)}\|_\infty \;=\; \|f^{(j-1)}-g^{(j-1)}\|_\infty .
	\]
	Consequently,
	\[
	d_{C^{k+1}}(T(f),T(g))
	\;\le\; \max\{1,b-a\}\; d_{C^k}(f,g).
	\]
	
	\medskip
	
	\item \textbf{Derivative map is $1$-Lipschitz $C^{k+1}\!\to C^{k}$.}
	
	For $f,g\in C^{k+1}$ we have
	\[
	d_{C^{k}}(f',g')
	=\sum_{j=0}^{k} \|f^{(j+1)}-g^{(j+1)}\|_\infty
	\;\le\;
	\sum_{j=0}^{k+1} \|f^{(j)}-g^{(j)}\|_\infty
	= d_{C^{k+1}}(f,g).
	\]
	
	\medskip
	
	\item \textbf{Concrete identity.}
	
	With $f(x)=\sin x$ on $[0,1]$,
	\[
	T(f)(x)=\int_0^x\sin y\,dy = 1-\cos x,
	\]
	so $D(T(f))=f$, while $D(f)=\cos x$ and
	\[
	T(D(f))(x)=\int_0^x \cos y\,dy=\sin x=f(x).
	\]
	
\end{enumerate}

\subsection*{Counterexamples (what can fail if you change norms/spaces)}

\begin{enumerate}
	
	\item \textbf{$(C^1,\|\cdot\|_\infty)$ is not complete.}
	
	Let $f_n(x)=\sqrt{x^2+\tfrac{1}{n}}$.  
	Then $f_n\in C^\infty$ and $f_n\to |x|$ uniformly, but $|x|\notin C^1$.  
	Thus $(C^1([0,1]),\|\cdot\|_\infty)$ is incomplete.
	
	\medskip
	
	\item \textbf{Derivative not continuous from $(C^1,\|\cdot\|_\infty)$ to $(C,\|\cdot\|_\infty)$.}
	
	Take $f_n(x)=\tfrac{1}{n}\sin(n^2 x)$ on $[0,1]$.  
	We have
	\[
	\|f_n\|_\infty=\frac{1}{n}\longrightarrow 0,
	\qquad
	\|f_n'\|_\infty = n \longrightarrow \infty .
	\]
	Hence $D$ is discontinuous with these norms.
	
	\medskip
	
	\item \textbf{Dropping the top derivative breaks completeness.}
	
	On $C^k$ with the truncated metric
	\[
	\tilde d(f,g)=\sum_{j=0}^{k-1}\|f^{(j)}-g^{(j)}\|_\infty,
	\]
	choose $(f_n)\subset C^\infty$ so that $f_n^{(j)}$ converge uniformly for $j<k$,  
	while $f_n^{(k)}$ oscillate without a uniform limit.  
	Then $(f_n)$ is Cauchy for $\tilde d$ but has no limit in $C^k$.
	
	\medskip
	
	\item \textbf{Total boundedness still fails (infinite dimension).}
	
	Closed balls in $(C^k,d_{C^k})$ are complete but not totally bounded.  
	For instance, translate a fixed bump function to disjoint locations;  
	the resulting family lies in the unit ball and remains pairwise well separated.
	
\end{enumerate}

\section*{Examples of failure of total boundedness}

\paragraph{Infinite discrete metric space (bounded but not totally bounded).}
Let $X$ be any infinite set and define
\[
d(x,y)=
\begin{cases}
	0,& x=y,\\
	1,& x\neq y.
\end{cases}
\]
Fix $\varepsilon\in(0,1)$. Then each open ball is a singleton:
\[
B(x,\varepsilon)=\{x\}.
\]
To cover $X$ by $\varepsilon$–balls one needs one ball per point of $X$, hence
infinitely many balls. Therefore no finite $\varepsilon$–cover exists, and $X$ is
\emph{not totally bounded}.

\bigskip

\paragraph{Unit ball in an infinite-dimensional Hilbert space (unboundedly many separated points).}
Consider $\ell^2$ with the norm $\|\cdot\|_2$ and its unit ball
\[
\mathbb{B}=\{x\in\ell^2:\ \|x\|_2\le 1\}.
\]
Let $(e_n)_{n\in\mathbb{N}}$ be the standard unit vectors. Then $e_n\in\mathbb{B}$ for all $n$, and for $m\neq n$,
\[
\|e_n-e_m\|_2=\sqrt{(1-0)^2+(0-1)^2}=\sqrt{2}.
\]
Fix $\varepsilon\in(0,1)$. Any ball of radius $\varepsilon$ contains at most one $e_n$, because
$\|e_n-e_m\|_2=\sqrt{2}>2\varepsilon$ for $m\neq n$. Hence finitely many $\varepsilon$–balls can cover at most
finitely many of the $e_n$, leaving others uncovered. Thus $\mathbb{B}$ is \emph{not totally bounded}.

\section*{Uniform continuity on $[0,\infty)$ — “(a) on one line, rest below” formatting}

\begin{enumerate}[label=(\alph*),leftmargin=*]
	
	\item $f(x)=\inf\{|x-n^2|:n\in\mathbb{N}\}$.\\  \par\medskip
	Write $f(x)=d(x,A)$ with $A=\{n^2:n\in\mathbb{N}\}$. For any $A\subset\mathbb{R}$,
	\[
	|d(x,A)-d(y,A)|\le |x-y|.
	\]
	Hence $f$ is $1$-Lipschitz, therefore uniformly continuous.
	
	\item $f(x)=\sin(x^2)$.\\  \par\medskip
	Let $x_n=\sqrt{\frac{\pi}{2}+2\pi n}$ and $y_n=x_n+\frac{\pi}{2x_n}$. Then
	\[
	\sin(x_n^2)=1,\qquad
	y_n^2=\frac{3\pi}{2}+2\pi n+\frac{\pi^2}{4x_n^2}\to \frac{3\pi}{2}+2\pi n,
	\]
	so $\sin(y_n^2)\to -1$ while $|x_n-y_n|=\frac{\pi}{2x_n}\to0$. Thus $f$ is \emph{not} uniformly
	continuous on $[0,\infty)$.
	
	\item $f(x)=\dfrac{\sin(x^3)}{x^2+1}$.\\  \par\medskip
	Differentiate:
	\[
	f'(x)=\frac{3x^2\cos x^3}{x^2+1}-\frac{2x\sin x^3}{(x^2+1)^2}.
	\]
	Bound the two terms on separate lines:
	\[
	\left|\frac{3x^2\cos x^3}{x^2+1}\right|\le 3,
	\qquad
	\left|\frac{2x\sin x^3}{(x^2+1)^2}\right|\le \frac{2|x|}{(x^2+1)^2}\le 1.
	\]
	Therefore $|f'(x)|\le 4$ for all $x$, so $f$ is globally Lipschitz and uniformly continuous.
	
\end{enumerate}

	
\section*{Compactness vs.\ total boundedness without metrics}

\paragraph{Topological vs.\ uniform spaces.}
In a plain topological space there is no intrinsic notion of total boundedness.
It is a \emph{uniform} notion. Thus we work in a uniform space $(X,\mathcal U)$.

\paragraph{Definitions (uniform space).}
For an entourage $U\in\mathcal U$, the $U$–ball at $x$ is $U[x]=\{y:(x,y)\in U\}$.
\begin{itemize}
	\item $X$ is \emph{totally bounded (precompact)} if for every entourage $U$ there exist
	$x_1,\dots,x_m\in X$ with $X=\bigcup_{i=1}^m U[x_i]$.
	\item $X$ is \emph{complete} if every Cauchy net/filter (with respect to $\mathcal U$) converges.
\end{itemize}

\paragraph{Theorem (uniform-space version).}
For any uniform space $(X,\mathcal U)$:
\begin{enumerate}
	\item If $X$ is compact, then $X$ is complete and totally bounded.
	\item If $X$ is complete and totally bounded, then $X$ is compact.
\end{enumerate}
Moreover, the completion of a totally bounded uniform space is compact (indeed compact
Hausdorff). If the uniformity is separated, the induced topology is Hausdorff.

\paragraph{Remarks.}
Metrics induce uniformities, so the metric-space result
$\text{compact} \Leftrightarrow \text{complete}+\text{totally bounded}$
is a special case. Without a chosen uniformity, “total boundedness” is not defined.

\section*{Examples: $k$-times differentiable but not $(k{+}1)$-times differentiable}

\paragraph{1) Fractional cusp (works for any $k\in\mathbb{N}_0$).}
Let $0<\alpha<1$ and define
\[
f(x)=|x|^{\,k+\alpha}.
\]
Then $f\in C^{k}(\mathbb{R})$ but $f\notin C^{k+1}(\mathbb{R})$.
Indeed, for $j\le k$ one has $f^{(j)}(x)=C_{j}\,\mathrm{sgn}(x)^{\sigma_j}\,|x|^{\,k+\alpha-j}$,
so $f^{(j)}$ is continuous at $0$ (it tends to $0$). However, $f^{(k+1)}(x)$ behaves
like $C\,|x|^{\alpha-1}$ near $0$ and therefore has no finite limit at $0$.

\paragraph{2) Odd power cusp (integer case).}
For $m\in\mathbb{N}_0$,
\[
f(x)=|x|^{\,2m+1}
\]
is $C^{2m}$ but not $C^{2m+1}$ at $x=0$.
Example: $|x|^3$ is $C^2$ with
$f'(x)=3x^2$ for $x>0$ and $f'(x)=-3x^2$ for $x<0$,
$f''(x)=6x$ for $x>0$ and $f''(x)=-6x$ for $x<0$, both extending continuously at $0$ by $0$,
while $f^{(3)}(x)=6$ for $x>0$ and $f^{(3)}(x)=-6$ for $x<0$, so $f^{(3)}$ has a jump at $0$.

\paragraph{3) Oscillatory, flat at $0$.}
Define
\[
f(x)=
\begin{cases}
	x^{k+1}\sin(1/x), & x\neq 0,\\
	0, & x=0.
\end{cases}
\]
Then $f\in C^{k}(\mathbb{R})$ with $f^{(j)}(0)=0$ for $0\le j\le k$,
but $f\notin C^{k+1}$: the $(k{+}1)$st derivative fails to exist/be continuous at $0$
(its expression for $x\neq 0$ involves terms like $x^{-1}\cos(1/x)$).

\paragraph{4) Multivariate version.}
For $x\in\mathbb{R}^n$ and $0<\alpha<1$,
\[
f(x)=\|x\|^{\,k+\alpha}
\]
is $C^{k}$ on $\mathbb{R}^n$ but not $C^{k+1}$ at $x=0$.

\section*{Cusp examples: $|x|^\alpha$ and $|x|^{1+\alpha}$}

\paragraph{Case 1: $f(x)=|x|^{\alpha}$ with $0<\alpha<1$.}
For $x\neq 0$,
\[
f'(x)=\alpha\,|x|^{\alpha-1}\operatorname{sgn}(x).
\]
At $x=0$,
\[
\frac{f(h)-f(0)}{h}=\frac{|h|^\alpha}{h}=\operatorname{sgn}(h)\,|h|^{\alpha-1}\xrightarrow[h\to 0]{}\pm\infty,
\]
since $\alpha-1<0$. Hence $f$ is continuous but not differentiable at $0$:
\[
f\in C^0(\mathbb{R})\setminus C^1(\mathbb{R}).
\]

\paragraph{Case 2: $f(x)=|x|^{1+\alpha}$ with $0<\alpha<1$.}
Let $p=1+\alpha\in(1,2)$. For $x\neq 0$,
\[
f'(x)=p\,|x|^{p-1}\operatorname{sgn}(x)=(1+\alpha)\,|x|^{\alpha}\operatorname{sgn}(x),
\]
so $\lim_{x\to 0} f'(x)=0$; defining $f'(0):=0$ yields $f'\in C^0$.
A second derivative for $x\neq 0$ is
\[
f''(x)=p(p-1)\,|x|^{p-2}=(1+\alpha)\alpha\,|x|^{\alpha-1},
\]
which diverges as $x\to 0$ because $\alpha-1<0$. Therefore $f\notin C^2$ at $0$:
\[
f\in C^1(\mathbb{R})\setminus C^2(\mathbb{R}).
\]

\paragraph{General pattern.}
For integer $k\ge 0$ and $0<\alpha<1$, set $f(x)=|x|^{k+\alpha}$. For $x\neq 0$ and $0\le j\le k$,
\[
f^{(j)}(x)=(k+\alpha)^{\underline{j}}\;|x|^{k+\alpha-j}\,\operatorname{sgn}(x)^{\,j},\qquad
(k+\alpha)^{\underline{j}}=\prod_{m=0}^{j-1}(k+\alpha-m).
\]
Then $f^{(j)}$ extends continuously at $0$ for all $j\le k$, while
\[
f^{(k+1)}(x)\sim C\,|x|^{\alpha-1}\operatorname{sgn}(x)^{\,k+1}\quad (x\to 0),
\]
which is unbounded. Hence $f\in C^{k}$ but $f\notin C^{k+1}$.

\section*{Uniform continuity + integrability $\Rightarrow$ decay at infinity}

\paragraph{Problem.}
Let $f:[0,\infty)\to[0,\infty)$ be uniformly continuous and suppose $\int_0^\infty f(x)\,dx<\infty$.
\begin{enumerate}
	\item[(a)] Show that $\lim_{x\to\infty} f(x)=0$.
	\item[(b)] Is the conclusion still true if $f$ is only continuous? Prove or give a counterexample.
\end{enumerate}

\paragraph{(a) Proof.}
Assume $\limsup_{x\to\infty} f(x)=\varepsilon_0>0$. Let $\varepsilon=\varepsilon_0/2$.
By uniform continuity, there exists $\eta>0$ such that
$|x-y|<\eta \Rightarrow |f(x)-f(y)|<\varepsilon$ for all $x,y\ge0$.
Choose $x_n\to\infty$ with $f(x_n)>\varepsilon_0$. Then for every
$y\in (x_n-\eta/2,\;x_n+\eta/2)$ we have
\[
f(y)\ge f(x_n)-\varepsilon>\varepsilon_0/2.
\]
Taking a subsequence so that these intervals are pairwise disjoint, we obtain
\[
\int_0^\infty f(x)\,dx
\ \ge\ \sum_n \int_{x_n-\eta/2}^{x_n+\eta/2} f(x)\,dx
\ \ge\ \sum_n \frac{\eta}{2}\cdot \frac{\varepsilon_0}{2}
\ =\ \infty,
\]
a contradiction. Hence $\lim_{x\to\infty} f(x)=0$.

\paragraph{(b) Counterexample with $f$ continuous but not uniformly continuous.}
For $n\in\mathbb{N}$ define a triangular bump
\[
\phi_n(x)=\max\!\Big(1-\frac{|x-n|}{w_n},\,0\Big),\qquad w_n=2^{-n},
\]
and set $f(x)=\sum_{n=1}^\infty \phi_n(x)$. The supports are disjoint, so $f$ is continuous and
\[
\int_0^\infty f(x)\,dx=\sum_{n=1}^\infty \int \phi_n(x)\,dx
=\sum_{n=1}^\infty w_n=\sum_{n=1}^\infty 2^{-n}<\infty.
\]
But $f(n)=1$ for all $n$, hence $\limsup_{x\to\infty} f(x)\ge 1$ and the limit at infinity does not exist (in particular, it is not $0$).
Therefore the conclusion fails if we assume only continuity.

\begin{center}\Large\bfseries $\ell_\infty$ (sup metric) is complete \quad+\quad $\ell_2$ (Euclidean metric) is a metric \& complete\end{center}

\textbf{Problem.}

(a) Let $\ell_\infty$ be the space of all bounded real sequences.
Equip it with the metric
\[
d_\infty((x_n),(y_n))=\sup_{n\in\mathbb N}|x_n-y_n|.
\]
Show that $(\ell_\infty,d_\infty)$ is complete.

\medskip
(b) Let
\[
\ell_2=\Big\{(x_n)\in\mathbb R^{\mathbb N}:\sum_{n=1}^\infty x_n^2<\infty\Big\}
\]
and define
\[
d((x_n),(y_n))=\Big(\sum_{n=1}^\infty (x_n-y_n)^2\Big)^{1/2}.
\]
Show that $d$ is a well-defined metric on $\ell_2$ and that $(\ell_2,d)$ is complete.

\medskip\medskip
\textbf{Theory (used).}

\begin{itemize}
	\item A metric space is complete iff every Cauchy sequence converges in the space.
	\item In $\mathbb R$, every Cauchy sequence converges (completeness of $\mathbb R$).
	\item If $x,y\in\ell_2$, then $(x_n-y_n)^2\le 2x_n^2+2y_n^2$, hence $x-y\in\ell_2$.
	\item Minkowski inequality: $\|u+v\|_2\le \|u\|_2+\|v\|_2$.
	\item Fast–Cauchy subsequence trick: from a Cauchy sequence choose a subsequence with
	$\|x_{j+1}-x_j\|\le 2^{-j}$ so the telescoping series converges.
\end{itemize}

\medskip\medskip
\textbf{Solution.}

\textbf{(a) Completeness of $\ell_\infty$.}
Let $(x^{(k)})_{k\ge1}$ be Cauchy in $d_\infty$.

For fixed $n$ we have
\[
|x^{(k)}_n-x^{(m)}_n|\le d_\infty(x^{(k)},x^{(m)}),
\]
so $(x^{(k)}_n)$ is Cauchy in $\mathbb R$ and thus converges.
Set
\[
x_n:=\lim_{k\to\infty}x^{(k)}_n,
\qquad
x:=(x_n)_{n\ge1}.
\]

Choose $k_0$ with
\[
d_\infty(x^{(k)},x^{(k_0)})\le 1
\quad\text{for all }k\ge k_0.
\]
Then for all $n$,
\[
|x_n|\le |x^{(k_0)}_n|+|x_n-x^{(k_0)}_n|
\le \|x^{(k_0)}\|_\infty+1,
\]
hence $x\in\ell_\infty$.

Let $\varepsilon>0$ and pick $K$ so that
\[
d_\infty(x^{(k)},x^{(m)})<\varepsilon/2
\quad\text{for all }k,m\ge K.
\]
Fix $k\ge K$ and let $m\to\infty$.
Then for every $n$,
\[
|x^{(k)}_n-x_n|\le \varepsilon/2.
\]
Taking suprema,
\[
\|x^{(k)}-x\|_\infty\le \varepsilon/2,
\]
so $x^{(k)}\to x$ in $\ell_\infty$.
Therefore $(\ell_\infty,d_\infty)$ is complete.

\medskip
\textbf{(b) $d$ is a metric on $\ell_2$ and $\ell_2$ is complete.}

\emph{Well-definedness.}
For $x,y\in\ell_2$,
\[
(x_n-y_n)^2\le 2x_n^2+2y_n^2,
\]
hence $\sum (x_n-y_n)^2<\infty$ and $d(x,y)<\infty$.

\emph{Metric axioms.}
Nonnegativity and symmetry are clear.
If $d(x,y)=0$ then $\sum (x_n-y_n)^2=0$ and thus $x_n=y_n$ for all $n$.
Minkowski gives the triangle inequality:
\[
\|x-z\|_2=\|(x-y)+(y-z)\|_2
\le \|x-y\|_2+\|y-z\|_2.
\]

\emph{Completeness.}
Let $(x^{(k)})$ be Cauchy in $\ell_2$.
Choose a subsequence $(x^{(k_j)})$ with
\[
\|x^{(k_{j+1})}-x^{(k_j)}\|_2\le 2^{-j}.
\]
Put $u^{(j)}:=x^{(k_{j+1})}-x^{(k_j)}$.
Then
\[
\sum_{j=1}^\infty \|u^{(j)}\|_2<\infty,
\]
so $\sum_{j=1}^\infty u^{(j)}$ converges in $\ell_2$ to some $v$.
Define
\[
x^{(\infty)}:=x^{(k_1)}+v \in \ell_2.
\]
Then $x^{(k_j)}\to x^{(\infty)}$ in $\ell_2$.

Since $(x^{(k)})$ is Cauchy, given $\varepsilon>0$ we can choose
$J$ and $N$ so that
\[
\|x^{(k_J)}-x^{(\infty)}\|_2<\varepsilon/2
\quad\text{and}\quad
\|x^{(k)}-x^{(m)}\|_2<\varepsilon/2
\ \text{for }k,m\ge N.
\]
For $k\ge \max\{N,k_J\}$,
\[
\|x^{(k)}-x^{(\infty)}\|_2
\le \|x^{(k)}-x^{(k_J)}\|_2+\|x^{(k_J)}-x^{(\infty)}\|_2
<\varepsilon.
\]
Hence $x^{(k)}\to x^{(\infty)}\in\ell_2$.
Therefore $(\ell_2,d)$ is complete.

\medskip\medskip
\textbf{Examples.}

\emph{In $\ell_\infty$:}
The unit vectors $e^{(k)}$ satisfy
\[
\|e^{(k)}-e^{(m)}\|_\infty=1
\quad (k\ne m),
\]
so they are not Cauchy.
But for
\[
x^{(k)}=(\underbrace{1/k,\dots,1/k}_{k\ \text{terms}},0,0,\dots)
\]
we have
\[
\|x^{(k)}\|_\infty=1/k\to0.
\]

\medskip
\emph{In $\ell_2$:}
For the same $x^{(k)}$,
\[
\|x^{(k)}\|_2^2=k\cdot(1/k^2)=1/k\to0,
\]
so $x^{(k)}\to0$ in $\ell_2$,
while
\[
\|e^{(k)}-e^{(m)}\|_2=\sqrt2
\quad (k\ne m)
\]
shows $(e^{(k)})$ is not Cauchy.

	
\textbf{Problem.}

On a nonempty complete metric space $(X,d)$, let $f:X\to X$ be continuous and suppose that
for each $x\neq y$ there is $L(x,y)\in(0,1)$ with
\[
d\big(f(x),f(y)\big)\ \le\ L(x,y)\, d(x,y).
\]

\begin{enumerate}[label=(\alph*), leftmargin=2.4em, itemsep=0.6em, topsep=0.6em]
	\item Show that, if $f$ has a fixed point, then it is unique.
	\item Must $f$ be a contraction? Must $f$ have a fixed point?
	Give a proof or counterexample.
	\item Show that, if $(X,d)$ is assumed to be compact,
	then $f$ necessarily has a unique fixed point.
\end{enumerate}

\medskip\medskip
\textbf{Theory.}

A map with $d(fx,fy)<d(x,y)$ for all $x\neq y$ is a \emph{strict contraction} (Edelstein).
Such a map is injective and has at most one fixed point.
On a compact metric space every strict contraction has a fixed point.

\medskip\medskip
\textbf{Solution.}

\emph{(a) Uniqueness.}
If $p,q$ are fixed, then for $p\neq q$,
\[
d(p,q)=d(f(p),f(q))\le L(p,q)\,d(p,q),
\]
which forces $d(p,q)=0$. Hence $p=q$.

\medskip
\emph{(b) Not necessarily Banach.}
On $\mathbb R$ with the usual metric let $f(x)=\log\cosh x$.
Then by the mean value theorem
\[
|f(x)-f(y)|=|\tanh c|\,|x-y|<|x-y|
\]
for some $c$ between $x$ and $y$.
Thus the hypothesis holds, but $\sup|\tanh|=1$ so no single $L<1$ works.

\emph{(b) No fixed point in general.}
Define a strictly increasing $C^1$ bijection $\phi:\mathbb R\to\mathbb R$
with strictly decreasing derivative $\phi'>0$ (e.g. $\phi'(t)=\tfrac12+\frac{1}{1+e^{t}}$).
Let
\[
d(x,y)=|\phi(x)-\phi(y)| \qquad (x,y\in\mathbb R).
\]
Then $(\mathbb R,d)$ is complete.
For $T(x)=x+1$ and $x\neq y$,
\[
\frac{d(Tx,Ty)}{d(x,y)}
=\frac{\phi'(c_2)}{\phi'(c_1)}<1
\]
for suitable $c_1<c_2$, since $\phi'$ is decreasing.
Hence $T$ satisfies the hypothesis but $T$ has no fixed point.

\medskip
\emph{(c) Compact case.}
Let $x_{n+1}=f(x_n)$. By compactness, a subsequence $x_{n_k}\to x^\star$ exists.
Strict contraction implies the whole $(x_n)$ is Cauchy, hence $x_n\to x^\star$.
By continuity,
\[
f(x^\star)=\lim_{n\to\infty} f(x_n)=\lim_{n\to\infty} x_{n+1}=x^\star.
\]
Uniqueness follows from (a).
\qed

\medskip\medskip
\textbf{Examples.}

(1) $f(x)=\log\cosh x$ on $(\mathbb R,|\cdot|)$:
\ $|f(x)-f(y)|=|\tanh c|\,|x-y|$ with $|\tanh c|<1$ but no uniform $L<1$.

(2) On $(\mathbb R,d)$ with $d(x,y)=|\phi(x)-\phi(y)|$ as above,
the shift $x\mapsto x+1$ is a strict contraction and has no fixed point.

\noindent\Large\bfseries Banach contraction, extraction, and Edelstein\par\normalfont\normalsize

\medskip
\textbf{Banach contraction principle.}
Let $(X,d)$ be complete and let $f:X\to X$ satisfy
\[
d(f(x),f(y)) \le L\, d(x,y)\qquad (0<L<1,\ \forall x,y\in X).
\]
Then $f$ has a unique fixed point $x^{*}$ and, for any start $x_{0}$, the iterates $x_{n+1}=f(x_{n})$ converge to $x^{*}$ with
\[
d(x_{n},x^{*}) \le \frac{L^{n}}{1-L}\, d(x_{1},x_{0}).
\]

\medskip
\textit{Extraction (fast Cauchy subsequence).}
From any Cauchy $(x^{k})$ choose $k_{1}<k_{2}<\cdots$ so that
\[
d\big(x^{k_{j+1}},x^{k_{j}}\big)\le 2^{-j}.
\]
Then $\sum_{j} d(x^{k_{j+1}},x^{k_{j}})<\infty$, hence $(x^{k_{j}})$ is Cauchy and converges in a complete space; the original sequence has the same limit.

\medskip
\textbf{Edelstein (strict contraction on compact spaces).}
If $(X,d)$ is compact and
\[
d(f(x),f(y))<d(x,y)\qquad (\forall x\neq y),
\]
then $f$ has a unique fixed point $x^{*}$ and $x_{n+1}=f(x_{n})\to x^{*}$ for any $x_{0}$.

\smallskip
\textit{Sketch.}
For $m>n$,
\[
d(x_{m+1},x_{n+1})=d\big(f(x_{m}),f(x_{n})\big)<d(x_{m},x_{n}),
\]
so orbit distances strictly decrease.
Compactness gives a convergent subsequence $x_{n_{k}}\to x^{*}$; strict decrease forces every subsequence to share the same limit, hence $(x_{n})$ is Cauchy and convergent to $x^{*}$. Continuity yields $f(x^{*})=x^{*}$; uniqueness is immediate.

\textbf{Examples.}
(1) \emph{Edelstein but not Banach on $(\mathbb R,|\cdot|)$.} $f(x)=\log(\cosh x)$.
Mean Value Theorem: $|f(x)-f(y)|=|f'(c)|\,|x-y|=|\tanh c|\,|x-y|<|x-y|$, but $\sup_{t}|\tanh t|=1$ (no single $L<1$).

(2) \emph{Strict but no fixed point without compactness.}
Let $\phi:\mathbb R\to\mathbb R$ be $C^{1}$, strictly increasing, with strictly decreasing $\phi'>0$, and set $d(x,y)=|\phi(x)-\phi(y)|$.
Then $(\mathbb R,d)$ is complete; for $T(x)=x+1$,

\frac{d(Tx,Ty)}{d(x,y)}=\frac{\phi'(c_{2})}{\phi'(c_{1})}<1
\]
for some $c_{1}\in(x,y)$ and $c_{2}\in(x+1,y+1)$, yet $T$ has no fixed point.
	
	
\textbf{Problem.}

Let $0<a,b<\infty$ and let $f:[-a,a]\times[-b,b]\to\mathbb R$ be continuous.

\medskip

Consider the initial value problem
\[
\begin{cases}
	y'(x)=f\big(x,y(x)\big),\\[2pt]
	y(0)=y_0,
\end{cases}
\qquad (x \text{ near } 0).
\]

\medskip

(a)\; Is the solution necessarily unique? Give a proof or a counterexample.

\medskip

(b)\; For every $k\in\mathbb N$ show there is an interval $I_k\ni0$ and $y_k\in C(I_k)$ solving
\[
\begin{cases}
	y_k'(x)=f_k\big(x,y_k(x)\big),\\[2pt]
	y_k(0)=y_0,
\end{cases}
\]
where $f_k$ are Lipschitz on $[-a,a]\times[-b,b]$ and $f_k\to f$ uniformly.

\medskip

What happens as $k\to\infty$?

\medskip

(c)\; Show there is $\varepsilon\in(0,a]$ and a solution $y$ on $(-\varepsilon,\varepsilon)$ of the original IVP.

\medskip

Compare with (b).

\medskip

\textbf{Theory.}

Picard--Lindel\"of: Lipschitz in $y$ $\Rightarrow$ unique local solution.

\medskip

Peano: continuity $\Rightarrow$ local solution (not necessarily unique).

\medskip

Arzel\`a--Ascoli: uniformly bounded and equicontinuous on a compact interval $\Rightarrow$ a uniformly convergent subsequence.

\medskip

Integral form: if $y_k(x)=y_0+\int_0^x f_k(t,y_k(t))\,dt$ and $y_k\to y$, $f_k\to f$ uniformly, then $y(x)=y_0+\int_0^x f(t,y(t))\,dt$.

\medskip

\textbf{Solution.}

\emph{(a) Nonuniqueness in general.}

Take $y'(x)=\sqrt{|y(x)|}$, $y(0)=0$.

Then $y\equiv0$ and $y(x)=\tfrac{x^2}{4}$ for $x\ge0$ (patched with $0$ for $x\le0$)
are distinct solutions.

Hence continuity alone does not guarantee uniqueness.

\medskip

\emph{(b) Lipschitz approximations; behaviour as $k\to\infty$.}

Since $f$ is continuous on the compact rectangle $R=[-a,a]\times[-b,b]$,
there exists $M$ with $|f|\le M$ on $R$.

Choose Lipschitz $f_k:R\to\mathbb R$ with $f_k\to f$ uniformly.

By Picard--Lindel\"of, each IVP with $f_k$ has a unique solution $y_k$ on a maximal interval $I_k\ni0$.

\medskip

Let $\varepsilon:=\min\{a,\ b/M\}$ (interpret $b/M=\infty$ if $M=0$).

Then, for $|x|\le\varepsilon$,
\[
|y_k(x)-y_0|\le \int_0^x |f_k(t,y_k(t))|\,dt \le M|x|\le b,
\]
so $y_k$ is defined on the common interval $I=(-\varepsilon,\varepsilon)$ and takes values in $[-b,b]$.

\medskip

Moreover, for $x,t\in I$,
\[
|y_k(x)-y_k(t)|\le \int_t^x |f_k(s,y_k(s))|\,ds \le M|x-t|,
\]
hence $\{y_k\}$ is uniformly bounded and equicontinuous on $\overline I$.

\medskip

By Arzel\`a--Ascoli there is a subsequence $y_{k_j}\to y$ uniformly on $\overline I$.

Passing to the limit in
\[
y_{k_j}(x)=y_0+\int_0^x f_{k_j}\big(t,y_{k_j}(t)\big)\,dt
\]
gives
\[
y(x)=y_0+\int_0^x f\big(t,y(t)\big)\,dt,
\]
so $y$ solves the original IVP on $I$.

\medskip

As $k\to\infty$, at least a subsequence $y_{k_j}$ converges to a solution of the original problem.

If the original problem is unique, then $y_k\to y$.

If it is nonunique, different subsequences (or different choices of $f_k$) may converge to different solutions.

\medskip

\emph{(c) Existence on $(-\varepsilon,\varepsilon)$.}

The construction in (b) proves Peano existence: there exists a solution
$y$ on $I=(-\varepsilon,\varepsilon)$ with
\[
y(x)=y_0+\int_0^x f\big(t,y(t)\big)\,dt,\qquad |x|<\varepsilon.
\]

Compared with (b): approximating by Lipschitz $f_k$ (unique solvability) and compactness
yield a convergent subsequence and hence existence for the original continuous $f$,
but uniqueness need not hold.

\textbf{Problem.}

On a normed vector space there is a canonical metric
$d(x,y)=\|x-y\|$.

Show that on a finite-dimensional real vector space, every choice of norm (and thus canonical metric) is equivalent.

Is this still true for infinite-dimensional vector spaces?

\medskip
\textbf{Theory.}

Two norms $\|\cdot\|_a,\|\cdot\|_b$ on $V$ are equivalent iff
there exist constants $m,M>0$ with
\[
m\|x\|_b \le \|x\|_a \le M\|x\|_b \qquad \text{for all } x\in V.
\]
Finite-dimensional normed spaces have compact unit spheres for any norm.
Continuous functions on compact sets attain positive minima and finite maxima.

\medskip
\textbf{Solution.}

\emph{Finite dimension $\Rightarrow$ all norms are equivalent.}

Let $\|\cdot\|_a$ and $\|\cdot\|_b$ be norms on a finite-dimensional real space $V$.

Consider the $b$-unit sphere $S_b=\{x\in V:\|x\|_b=1\}$, which is compact.

The map $x\mapsto\|x\|_a$ is continuous, hence attains
\[
m:=\min_{x\in S_b}\|x\|_a>0,
\qquad
M:=\max_{x\in S_b}\|x\|_a<\infty.
\]
For any $x\neq 0$, write $x=\|x\|_b\,u$ with $u\in S_b$. Then
\[
m\|x\|_b \le \|x\|_a = \|x\|_b\,\|u\|_a \le M\|x\|_b.
\]
The inequalities also hold for $x=0$. Therefore the two norms are equivalent.

\medskip
\emph{Infinite dimension $\Rightarrow$ not all norms are equivalent.}

Take $V=C[0,1]$ and define
\[
\|f\|_\infty=\max_{t\in[0,1]}|f(t)|,
\qquad
\|f\|_1=\int_0^1 |f(t)|\,dt.
\]
Let $(f_n)$ be continuous functions with $\|f_n\|_\infty=1$ and
$\|f_n\|_1\to 0$ (e.g., triangular bumps of height $1$ and width $1/n$).
If $\|\cdot\|_\infty$ and $\|\cdot\|_1$ were equivalent, there would exist $m>0$
with $m\|f\|_\infty\le \|f\|_1$ for all $f\in V$, which is contradicted by $(f_n)$.
Hence these norms are \emph{not} equivalent.

\medskip
\textbf{Examples.}

On $\mathbb R^n$ one has the standard estimates
\[
\|x\|_\infty \le \|x\|_2 \le \|x\|_1 \le \sqrt{n}\,\|x\|_2 \le n\,\|x\|_\infty,
\]
so $\|\cdot\|_1,\|\cdot\|_2,\|\cdot\|_\infty$ are equivalent.

On $C[0,1]$, the sequence $(f_n)$ above converges to $0$ in $\|\cdot\|_1$
but not in $\|\cdot\|_\infty$, showing non-equivalence of the norms.

\textbf{Theory (what we’ll use).}

\emph{Picard--Lindel\"of (Banach).}
If $f$ is continuous on $R=[-a,a]\times[-b,b]$ and Lipschitz in $y$ with constant $L$,
then for any $y_0\in[-b,b]$ there exists
\[
\varepsilon\le \min\Big\{a,\ \frac{b}{M},\ \frac{1}{L}\Big\},\qquad M=\sup_R |f|,
\]
such that the IVP $y'(x)=f(x,y(x))$, $y(0)=y_0$ has a unique solution on $[-\varepsilon,\varepsilon]$.
It is the unique fixed point of $T:C([-\varepsilon,\varepsilon])\to C([-\varepsilon,\varepsilon])$,
\[
(Ty)(x)=y_0+\int_0^x f(t,y(t))\,dt,
\]
which is a contraction when $L\varepsilon<1$.

\medskip
\emph{Peano existence.}
If $f$ is merely continuous on $R$, then there exists
$\varepsilon\le \min\{a,b/M\}$ and a (not necessarily unique) solution on $[-\varepsilon,\varepsilon]$.
Proof can be done via polygonal approximations or by Lipschitz approximants $f_k\to f$.

\medskip
\emph{Uniform bounds / domain control.}
If $|f|\le M$ on $R$, any solution satisfies
\[
|y(x)-y_0|\le M|x| \quad \text{for } |x|\le \varepsilon,
\]
hence $y([-\varepsilon,\varepsilon])\subset[-b,b]$ whenever $\varepsilon\le b/M$.

\medskip
\emph{Arzel\`a--Ascoli.}
A uniformly bounded and equicontinuous family in $C([c,d])$ has a uniformly convergent subsequence.
In particular, if $|y_k'|\le M$ a.e., then $|y_k(x)-y_k(t)|\le M|x-t|$ (equicontinuity).

\medskip
\emph{Passing to the limit in integral form.}
If $y_k\to y$ uniformly on $[-\varepsilon,\varepsilon]$, $f_k\to f$ uniformly on $R$,
and $y_k([- \varepsilon,\varepsilon])\subset[-b,b]$, then
\[
\sup_{x}\big|f_k(x,y_k(x))-f(x,y(x))\big|\to0,
\]
so from
\[
y_k(x)=y_0+\int_0^x f_k(t,y_k(t))\,dt
\]
we obtain
\[
y(x)=y_0+\int_0^x f(t,y(t))\,dt,
\]
and hence $y'(x)=f(x,y(x))$ for all $x$.

\textbf{Problem.}

The goal is to guide you through a proof of the Stone–Weierstrass (Bernstein) theorem in one variable:
every continuous $f:K\to\mathbb R$ on a rectangle
$K=[a_1,b_1]\times\cdots\times[a_n,b_n]\subset\mathbb R^n$
can be uniformly approximated by polynomials.
In what follows we take $n=1$.

\medskip
(a)\; Show that, without loss of generality, it is enough to treat $[a_1,b_1]=[0,1]$.

\medskip
(b)\; For $k=0,\dots,n$ let
\[
p_{n,k}(x)=\binom{n}{k}\,x^{k}(1-x)^{\,n-k}.
\]
Show that
\[
\sum_{k=0}^{n} p_{n,k}(x)=1,
\qquad
\sum_{k=0}^{n}\!\left(x-\frac{k}{n}\right)^{\!2} p_{n,k}(x)=\frac{x(1-x)}{n}.
\]

\medskip
(c)\; Define the Bernstein polynomial
\[
p_n(x)=\sum_{k=0}^{n} f\!\left(\frac{k}{n}\right)\,p_{n,k}(x).
\]
Using the identity from (b), show that, for any $\delta>0$,
\[
\begin{aligned}
	|p_n(x)-f(x)|
	&\le
	\sum_{\;|x-\frac{k}{n}|<\delta} \bigl|f(\tfrac{k}{n})-f(x)\bigr|\,p_{n,k}(x)
	\;+\;
	\sum_{\;|x-\frac{k}{n}|\ge \delta} \bigl|f(\tfrac{k}{n})-f(x)\bigr|\,p_{n,k}(x).
\end{aligned}
\]

\medskip
(d)\; Choose $\delta$ appropriately and use the second identity in (b) to conclude that
$p_n \xrightarrow{\ \mathrm{unif}\ } f$ on $[0,1]$.

\medskip
\textbf{Theory (tools).}

\textbf{Affine reduction to $[0,1]$.}
For $a<b$, the map $\phi:[0,1]\to[a,b]$, $\phi(t)=a+(b-a)t$, is a homeomorphism.
If $g=f\circ\phi$ can be uniformly approximated on $[0,1]$ by polynomials $q_m$, then
$q_m\circ\phi^{-1}$ uniformly approximates $f$ on $[a,b]$.
Hence it suffices to work on $[0,1]$.

\medskip
\textbf{Bernstein basis identities.}
For $p_{n,k}(x)=\binom{n}{k}x^k(1-x)^{n-k}$,
\[
\begin{aligned}
	\sum_{k=0}^{n} p_{n,k}(x) &= 1,\\[4pt]
	\sum_{k=0}^{n} \frac{k}{n}\,p_{n,k}(x) &= x,\\[4pt]
	\sum_{k=0}^{n}\!\left(\frac{k}{n}-x\right)^{\!2} p_{n,k}(x) &= \frac{x(1-x)}{n}.
\end{aligned}
\]
(These are normalization, mean, and variance for a $\mathrm{Bin}(n,x)$ law.)

\medskip
\textbf{Bernstein operator.}
\[
B_nf(x):=\sum_{k=0}^{n} f(k/n)\,p_{n,k}(x),
\]
a polynomial of degree $\le n$.

\medskip
\textbf{Modulus of continuity.}
For $f\in C([0,1])$ define
\[
\omega_f(\delta)
:=\sup\{\,|f(u)-f(v)|:\ u,v\in[0,1],\ |u-v|\le\delta\,\},\qquad \delta\ge0.
\]
Then $\omega_f(\delta)\to0$ as $\delta\downarrow0$ (uniform continuity on a compact interval).

\medskip
\textbf{Error splitting and tail estimate.}
For any $\delta>0$,
\[
|B_nf(x)-f(x)|
\le \omega_f(\delta)
\;+\; 2\|f\|_\infty \sum_{|x-\frac{k}{n}|\ge\delta} p_{n,k}(x).
\]
By Chebyshev’s inequality together with the variance identity,
\[
\sum_{|x-\frac{k}{n}|\ge\delta} p_{n,k}(x)
\le \frac{1}{\delta^2}\sum_{k=0}^{n}\!\left(x-\frac{k}{n}\right)^{\!2} p_{n,k}(x)
= \frac{x(1-x)}{n\delta^2}
\le \frac{1}{4n\delta^2}.
\]
Hence
\[
\boxed{\;
	\|B_nf-f\|_\infty
	\le \omega_f(\delta) + \frac{\|f\|_\infty}{2n\delta^2}.
	\;}
\]
Given $\varepsilon>0$, first choose $\delta>0$ so that $\omega_f(\delta)<\varepsilon/2$,
then pick $n$ large with $\frac{\|f\|_\infty}{2n\delta^2}<\varepsilon/2$.
Therefore $B_nf\to f$ uniformly on $[0,1]$.
	
	
	
	\textbf{Solution.}
	
	\emph{(a) Reduction to $[0,1]$.}
	Let $a<b$ and $\phi:[0,1]\to[a,b]$, $\phi(t)=a+(b-a)t$.
	If $g=f\circ\phi$ is uniformly approximated on $[0,1]$ by polynomials $q_n$,
	then $q_n\circ\phi^{-1}$ are polynomials on $[a,b]$ and
	$\|f-(q_n\circ\phi^{-1})\|_{[a,b]}=\|g-q_n\|_{[0,1]}\to0$.
	Thus it suffices to work on $[0,1]$.
	
	\medskip
	\emph{(b) Identities for $p_{n,k}$.}
	For $p_{n,k}(x)=\binom{n}{k}x^k(1-x)^{n-k}$,
	\[
	\sum_{k=0}^{n} p_{n,k}(x)=(x+(1-x))^n=1.
	\]
	Let $X\sim\mathrm{Bin}(n,x)$. Then $p_{n,k}(x)=\mathbb P[X=k]$ and hence
	\[
	\sum_{k=0}^{n}\Big(x-\frac{k}{n}\Big)^{2}p_{n,k}(x)
	=\mathbb E\Big[\Big(x-\frac{X}{n}\Big)^2\Big]
	=\mathrm{Var}\!\Big(\frac{X}{n}\Big)
	=\frac{1}{n^2}\mathrm{Var}(X)
	=\frac{x(1-x)}{n}.
	\]
	(Equivalently, use $\sum k\,p_{n,k}=nx$ and $\sum k^2p_{n,k}=n x(1-x)+n^2x^2$.)
	
	\medskip
	\emph{(c) Error splitting.}
	With $p_n(x)=\sum_{k=0}^{n} f(k/n)\,p_{n,k}(x)$ we have
	\[
	p_n(x)-f(x)=\sum_{k=0}^{n}\big(f(\tfrac{k}{n})-f(x)\big)p_{n,k}(x).
	\]
	Thus, for any $\delta>0$,
	\[
	\begin{aligned}
		|p_n(x)-f(x)|
		&\le
		\sum_{\;|x-\frac{k}{n}|<\delta}\! \bigl|f(\tfrac{k}{n})-f(x)\bigr|\,p_{n,k}(x)
		\;+\;
		\sum_{\;|x-\frac{k}{n}|\ge \delta}\! \bigl|f(\tfrac{k}{n})-f(x)\bigr|\,p_{n,k}(x).
	\end{aligned}
	\]
	
	\medskip
	\emph{(d) Uniform convergence.}
	Let $\omega_f(\delta)=\sup\{|f(u)-f(v)|:\ |u-v|\le\delta\}$.
	Then $\omega_f(\delta)\to0$ as $\delta\downarrow0$.
	The first sum in (c) is $\le \omega_f(\delta)$ (convex combination).
	For the second sum, use $|f(\tfrac{k}{n})-f(x)|\le 2\|f\|_\infty$ and Chebyshev together with (b):
	\[
	\sum_{|x-\frac{k}{n}|\ge\delta} p_{n,k}(x)
	\le \frac{1}{\delta^2}\sum_{k=0}^{n}\Big(x-\frac{k}{n}\Big)^2 p_{n,k}(x)
	= \frac{x(1-x)}{n\delta^2}
	\le \frac{1}{4n\delta^2}.
	\]
	Hence
	\[
	\|p_n-f\|_\infty \le \omega_f(\delta) + \frac{\|f\|_\infty}{2n\delta^2}.
	\]
	Given $\varepsilon>0$, choose $\delta>0$ so that $\omega_f(\delta)<\varepsilon/2$,
	then choose $n$ large with $\frac{\|f\|_\infty}{2n\delta^2}<\varepsilon/2$.
	Therefore $p_n\to f$ uniformly on $[0,1]$.
	
	
	\textbf{Problem.}
	
	Let $(X,d)$ be a compact metric space, and let $C(X)$ denote the space of all real-valued continuous functions on $X$.
	We say that $F\subset C(X)$ is \emph{equicontinuous} if for every $x\in X$ and every $\varepsilon>0$ there exists a neighbourhood $U_x$ of $x$ such that
	\[
	|f(x)-f(y)|<\varepsilon \quad \text{for all } y\in U_x \text{ and all } f\in F.
	\]
	(The neighbourhood $U_x$ is the same for the entire family $F$.)
	
	Show that $F\subset C(X)$ is compact (for the uniform metric) if and only if it is \emph{closed}, \emph{bounded}, and \emph{equicontinuous}.
	
	\medskip
	(a)\; Using results from the lectures, justify that it is enough to show that $F\subset C(X)$ is \emph{totally bounded} if and only if it is \emph{bounded} and \emph{equicontinuous}.
	
	\medskip
	(b)\; Prove the $(\Rightarrow)$ direction: if $F\subset C(X)$ is totally bounded, then it is bounded and equicontinuous.
	
	\medskip
	(c)\; $(\star)$ Prove the $(\Leftarrow)$ direction. \emph{Hint:} use compactness of $X$ to construct a finite subfamily that is a candidate $\varepsilon$-net for $F$.
	
	\medskip
	(d)\; Deduce that if $(f_n)\subset C([a,b])$ is bounded and equicontinuous (for the uniform metric), then there exists a subsequence $(f_{n_k})$ that converges uniformly on $[a,b]$.
	
	\medskip
	\textbf{Theory (tools).}
	
	\textbf{Uniform metric and completeness.}
	On $C(X)$ use $d_\infty(f,g)=\|f-g\|_\infty=\sup_{x\in X}|f(x)-g(x)|$.
	If $X$ is compact then $(C(X),\|\cdot\|_\infty)$ is complete.
	
	\medskip
	\textbf{Compact = complete + totally bounded (metric spaces).}
	A subset of a metric space is compact iff it is complete and totally bounded.
	A closed subset of a complete space is complete.
	Hence it suffices to prove:
	\[
	\boxed{\text{$F$ totally bounded}\ \Longleftrightarrow\ \text{$F$ bounded and equicontinuous.}}
	\]
	
	\medskip
	\textbf{Totally bounded $\Rightarrow$ bounded \& equicontinuous.}
	Let $\varepsilon>0$ and choose a finite $\varepsilon/3$-net $\{g_1,\dots,g_m\}\subset F$ in $\|\cdot\|_\infty$.
	Each $g_i$ is uniformly continuous on compact $X$; choose $\delta>0$ that works for all $i$.
	If $d(x,y)<\delta$ and $f\in F$, pick $i$ with $\|f-g_i\|_\infty<\varepsilon/3$; then
	\[
	|f(x)-f(y)|\le 2\|f-g_i\|_\infty+|g_i(x)-g_i(y)|<\varepsilon.
	\]
	Boundedness follows since a finite $\varepsilon$-net of $F$ is bounded and hence so is $F$.
	
	\medskip
	\textbf{Bounded \& equicontinuous $\Rightarrow$ totally bounded (Arzel\`a–Ascoli).}
	Fix $\varepsilon>0$.
	Equicontinuity gives $\delta>0$ with $d(x,y)<\delta\Rightarrow |f(x)-f(y)|<\varepsilon/3$ for all $f\in F$.
	
	Cover $X$ by finitely many balls $B(x_j,\delta)$, $j=1,\dots,N$.
	If $|f|\le M$ on $X$ for all $f\in F$, partition $[-M,M]$ into intervals of length $\varepsilon/3$.

	For each $f\in F$ choose grid points $a_j$ with $|f(x_j)-a_j|\le \varepsilon/6$ and define a template function that on $B(x_j,\delta)$ takes value $a_j$.

	This finite template family is an $\varepsilon$-net for $F$, hence $F$ is totally bounded.
	
	\medskip
	\textbf{Sequential consequence.}
	On $[a,b]$, any bounded, equicontinuous family in $C([a,b])$ has a uniformly convergent subsequence (Arzel\`a–Ascoli).
	
	
	\textbf{Equicontinuity is essential: counterexamples and remarks.}
	
	\medskip
	\textbf{Setup.}
	Let $(X,d)$ be a metric space and $C(X)$ the space of real--valued continuous
	functions on $X$ with the uniform norm $\|f\|_\infty=\sup_{x\in X}|f(x)|$.
	A family $F\subset C(X)$ is \emph{equicontinuous} if
	\[
	\forall\,x\in X\ \forall\,\varepsilon>0\ \exists\,\delta>0\ \text{s.t.}\ 
	d(x,y)<\delta\ \Longrightarrow\ |f(x)-f(y)|<\varepsilon\quad \text{for all } f\in F.
	\]
	Equivalently, when $X$ is compact, for every $\varepsilon>0$ there exists a \emph{global}
	$\delta>0$ that works for all $x,y\in X$ and all $f\in F$.
	
	\medskip
	\textbf{(1) Bounded but not equicontinuous $\Rightarrow$ not relatively compact.}
	Let $X=[0,1]$ and
	\[
	F=\{\,f_n(x)=x^n:\ n\in\mathbb N\,\}\subset C([0,1]).
	\]
	Then $\|f_n\|_\infty=1$ for all $n$ (bounded), but $F$ is \emph{not} equicontinuous at $x=1$:
	for any $\delta>0$, choose $n$ so large that $(1-\delta)^n<\tfrac12$, hence
	\[
	|f_n(1)-f_n(1-\delta)|=1-(1-\delta)^n>\tfrac12.
	\]
	Moreover $f_n\to \mathbf 1_{\{1\}}$ pointwise, so no subsequence can converge uniformly in $C([0,1])$.
	Thus boundedness alone does not give relative compactness.
	
	\medskip
	\textbf{(2) Equicontinuous but unbounded $\Rightarrow$ not relatively compact.}
	Let $X=[0,1]$ and $F=\{\,c_n:\ c_n(x)\equiv n,\ n\in\mathbb N\,\}$.
	Each $c_n$ is constant, hence $F$ is equicontinuous, but $\|c_n\|_\infty=n\to\infty$,
	so $F$ is not bounded and certainly not totally bounded (hence not relatively compact).
	
	\medskip
	\textbf{(3) Bounded \& equicontinuous, but $X$ noncompact $\Rightarrow$ failure.}
	Let $X=\mathbb R$ and fix $\phi\in C_c^\infty(\mathbb R)$ with $\|\phi\|_\infty=1$ and
	$\operatorname{supp}\phi\subset[-1,1]$. Consider
	\[
	F=\{\,\phi(\,\cdot - t\,):\ t\in\mathbb R\,\}.
	\]
	Then $F$ is uniformly bounded and equicontinuous (derivatives are uniformly bounded),
	but $F$ is \emph{not} totally bounded in $\|\cdot\|_\infty$:
	if $|t-s|>3$ then $\operatorname{supp}(\phi(\cdot-t))\cap\operatorname{supp}(\phi(\cdot-s))=\varnothing$,
	hence $\|\phi(\cdot-t)-\phi(\cdot-s)\|_\infty=1$. No finite $\varepsilon$--net exists for $\varepsilon<1$.
	Thus compactness of the domain $X$ is crucial.
	
	\medskip
	\textbf{(4) Closed and bounded need not be equicontinuous (hence not compact).}
	The unit ball of $C([0,1])$,
	\[
	B=\{\,f\in C([0,1]): \|f\|_\infty\le 1\,\},
	\]
	is closed and bounded but \emph{not} equicontinuous: for any $\delta>0$ one can build a continuous
	``spike'' $s_\delta$ of height $1$ supported in an interval of width $<\delta$.
	Hence $B$ is not compact (reflecting infinite dimensionality).
	
	\medskip
	\textbf{Why equicontinuity matters (Arzel\`a--Ascoli).}
	If $X$ is compact and $F\subset C(X)$ is bounded and equicontinuous, then $F$ is totally bounded in
	$(C(X),\|\cdot\|_\infty)$; the closure $\overline F$ is compact. Conversely, if $F$ is totally bounded,
	then $F$ is bounded and equicontinuous. Thus, on compact $X$,
	\[
	\boxed{\text{$F$ compact } \Longleftrightarrow\ \text{$F$ closed, bounded, and equicontinuous}.}
	\]
	
	\medskip
	\textbf{Proof sketch: totally bounded $\Rightarrow$ equicontinuous.}
	Fix $\varepsilon>0$. Choose a finite $\varepsilon/3$--net $\{g_1,\dots,g_m\}\subset F$ in $\|\cdot\|_\infty$.
	Each $g_i$ is uniformly continuous on compact $X$; pick $\delta>0$ working for all $i$.
	For $d(x,y)<\delta$ and $f\in F$, pick $i$ with $\|f-g_i\|_\infty<\varepsilon/3$:
	\[
	|f(x)-f(y)|\le 2\|f-g_i\|_\infty+|g_i(x)-g_i(y)|<\varepsilon.
	\]
	This $\delta$ is independent of $f$ and $x$, so $F$ is equicontinuous.
	
	\medskip
	\textbf{Proof sketch: bounded \& equicontinuous $\Rightarrow$ totally bounded.}
	Let $\varepsilon>0$. Equicontinuity gives $\delta>0$ with
	$d(x,y)<\delta \Rightarrow |f(x)-f(y)|<\varepsilon/3$ for all $f\in F$.
	Cover compact $X$ by finitely many balls $B(x_j,\delta)$, $j=1,\dots,N$.
	Boundedness gives $|f|\le M$ on $X$ for all $f\in F$; partition $[-M,M]$ into intervals of length $\varepsilon/3$.
	For each $f\in F$, choose grid points $a_j$ with $|f(x_j)-a_j|\le \varepsilon/6$ and define a template function that on
	$B(x_j,\delta)$ takes value $a_j$. This finite template family is an $\varepsilon$--net of $F$, hence $F$ is totally bounded.
	
	\medskip
	\textbf{Takeaway.}
	\emph{Equicontinuity} provides a uniform modulus of continuity for the family and prevents arbitrarily
	narrow spikes; combined with boundedness and compactness of the domain, it yields total boundedness
	(and thus compactness) in the sup norm via Arzel\`a--Ascoli.
	
	\textbf{Relative compactness (precompactness).}
	
	Let $A\subset X$ where $X$ is a topological (or metric) space.
	We say $A$ is \emph{relatively compact} (or \emph{precompact}) if $\overline{A}$ is compact in $X$.
	
	\medskip
	\textbf{Metric characterizations.}
	For $A$ in a metric space:
	\[
	A \text{ is relatively compact }
	\iff \text{ every sequence in } A \text{ has a convergent subsequence in } X
	\iff A \text{ is totally bounded.}
	\]
	If $X$ is complete, then $\overline{A}$ compact $\iff$ $A$ totally bounded.
	
	\medskip
	\textbf{Examples.}
	In $\mathbb{R}^n$, $A$ relatively compact $\iff$ $A$ bounded (Heine--Borel).
	In $C([0,1])$ with $\|\cdot\|_\infty$, bounded \& equicontinuous families are relatively compact (Arzel\`a--Ascoli),
	but the unit ball is not relatively compact.
	
{\large\bfseries Totally bounded sets: definition, examples, and counterexamples.\par}


	
	\medskip
	\textbf{Definition.}
	A subset $A$ of a metric space $(X,d)$ is \emph{totally bounded} if
	for every $\varepsilon>0$ there exist points $x_1,\dots,x_N\in X$
	such that
	\[
	A \subset \bigcup_{i=1}^N B(x_i,\varepsilon).
	\]
	Equivalently, $A$ admits a finite $\varepsilon$--net for each $\varepsilon>0$.
	
	\medskip
	\textbf{Characterizations.}
	\begin{itemize}
		\item $A$ is totally bounded $\iff$ every sequence in $A$ has a Cauchy subsequence.
		\item If $X$ is complete, then $A$ is totally bounded $\iff$ $\overline{A}$ is compact (i.e., $A$ is relatively compact).
		\item Always: totally bounded $\Rightarrow$ bounded (the converse may fail).
	\end{itemize}
	
	\medskip
	\textbf{Finite vs.\ infinite dimension.}
	\begin{itemize}
		\item In $\mathbb{R}^n$ (any finite--dimensional normed space), bounded $\iff$ totally bounded (Heine--Borel).
		\item In infinite--dimensional normed spaces, there exist bounded sets that are not totally bounded.
	\end{itemize}
	
	\medskip
	\textbf{Examples (totally bounded).}
	\begin{itemize}
		\item $[a,b]\subset\mathbb{R}$ and, more generally, any bounded subset of $\mathbb{R}^n$.
		\item $\mathbb{Q}\cap[0,1]$ in the Euclidean metric (subset of a totally bounded set).
		\item Any compact set in a metric space.
	\end{itemize}
	
	\medskip
	\textbf{Counterexamples (not totally bounded).}
	\begin{itemize}
		\item $\mathbb{R}$ or $\mathbb{N}$ in the usual metric (unbounded).
		\item $(\mathbb{R},d)$ with $d(x,y)=\min\{1,|x-y|\}$:
		the space is bounded (diameter $1$) but not totally bounded, since for $\varepsilon<1$
		one needs infinitely many $\varepsilon$-balls to cover $\mathbb{R}$.
		\item The unit ball of an infinite--dimensional Banach space (e.g., $\ell^2$):
		bounded but not totally bounded; the orthonormal vectors $\{e_k\}$ remain $\sqrt2$ apart.
		\item Any infinite discrete metric space $(X,d)$ with $d(x,y)=1$ for $x\neq y$:
		bounded but not totally bounded, because for $\varepsilon<1$
		each $\varepsilon$-ball is a singleton.
	\end{itemize}
	
	\medskip
	\textbf{Basic properties.}
	\begin{itemize}
		\item Subsets, finite unions, and continuous images of totally bounded sets are totally bounded.
		\item Products of totally bounded sets are totally bounded (for the product metric).
	\end{itemize}
	
	{\Large\bfseries $\varepsilon$-nets: finite vs.\ infinite (definition, theory, examples).\par}
	
	\medskip
	\textbf{Definition.}
	Let $(X,d)$ be a metric space and $A\subseteq X$. For $\varepsilon>0$ a set $N\subseteq X$ is an \emph{$\varepsilon$-net for $A$} if
	\[
	A \subseteq \bigcup_{x\in N} B(x,\varepsilon).
	\]
	Equivalently, for every $a\in A$ there exists $x\in N$ with $d(a,x)<\varepsilon$.
	If such $N$ can be chosen finite, we say $A$ admits a \emph{finite $\varepsilon$-net}.
	
	\medskip
	\textbf{Covering and packing numbers.}
	The \emph{covering number} $N(A,\varepsilon)$ is the minimal size of a finite $\varepsilon$-net for $A$ (or $+\infty$ if none exists).
	An \emph{$\varepsilon$-packing} is a set $P\subseteq A$ with the balls $\{B(p,\varepsilon)\}_{p\in P}$ pairwise disjoint;
	its maximal size is the \emph{packing number} $M(A,\varepsilon)$.
	Standard bounds:
	\[
	M(A,2\varepsilon)\ \le\ N(A,\varepsilon)\ \le\ M(A,\varepsilon).
	\]
	
	\medskip
	\textbf{Total boundedness.}
	A set $A$ is \emph{totally bounded} iff $N(A,\varepsilon)<\infty$ for every $\varepsilon>0$.
	In a complete metric space, $A$ is totally bounded $\iff$ $\overline{A}$ is compact (i.e., $A$ is relatively compact).
	
	\medskip
	\textbf{Separability and countable $\varepsilon$-nets.}
	If $X$ is separable with countable dense set $D$, then $D$ is an $\varepsilon$-net for every $\varepsilon>0$ (since $B(x,\varepsilon)\cap D\neq\varnothing$ for all $x$).
	Thus countable $\varepsilon$-nets always exist in separable spaces, though they need not be finite.
	
	\medskip
	\textbf{Examples (finite $\varepsilon$-nets exist).}
	\begin{itemize}
		\item $[0,1]\subset\mathbb R$ with the Euclidean metric. A grid $\{0,\varepsilon,2\varepsilon,\dots,1\}$ is a finite $\varepsilon$-net; $N([0,1],\varepsilon)\sim \lceil 1/\varepsilon\rceil$.
		\item Any bounded subset of $\mathbb R^n$. Cubic lattices produce finite $\varepsilon$-nets; hence bounded $\iff$ totally bounded in finite dimensions.
		\item The unit circle $S^1\subset\mathbb R^2$. Regular $m$-gons yield finite $\varepsilon$-nets with $m\asymp 1/\varepsilon$.
	\end{itemize}
	
	\medskip
	\textbf{Counterexamples (finite $\varepsilon$-nets do not exist for some $\varepsilon>0$).}
	\begin{itemize}
		\item $\mathbb R$ or $\mathbb N$ with the usual metric. For $\varepsilon=1$ no finite $\varepsilon$-net covers the set; hence not totally bounded.
		\item Infinite discrete metric space $(X,d)$ with $d(x,y)=1$ for $x\neq y$. For any $\varepsilon<1$, each $\varepsilon$-ball is a singleton, so every $\varepsilon$-net must be infinite.
		\item Unit ball of an infinite--dimensional Banach space (e.g., $\ell^2$). The orthonormal vectors $\{e_k\}$ satisfy $\|e_i-e_j\|_2=\sqrt 2$, so for $\varepsilon<1$ no finite $\varepsilon$-net exists.
		\item $(\mathbb R,d)$ with $d(x,y)=\min\{1,|x-y|\}$. The space is bounded (diameter $1$), but for any $\varepsilon<1$ one needs infinitely many $\varepsilon$-balls to cover $\mathbb R$.
	\end{itemize}
	
	\medskip
	\textbf{Useful facts.}
	\begin{itemize}
		\item Subsets and finite unions of totally bounded sets are totally bounded.
		\item Continuous images of totally bounded sets are totally bounded.
		\item Products of totally bounded sets are totally bounded (for standard product metrics).
		\item $A$ is totally bounded $\iff$ every sequence in $A$ has a Cauchy subsequence.
	\end{itemize}
