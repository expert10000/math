	\section*{Exercise 1.1}
	Consider the meromorphic function
	\[
	F(z) = \frac{1}{1+z^2} \frac{\cos[(\pi-\theta)z]}{2\sin(\pi z)}.
	\]
	
	\subsection*{a) Poles and residues}
	The poles occur at $z=\pm i$ and $z=n$, $n\in\mathbb{Z}$.  
	At $z=\pm i$:
	\[
	\operatorname{Res}[F,\pm i] = -\frac{\cosh(\pi-\theta)}{4\sinh \pi}.
	\]
	At $z=n\in\mathbb{Z}$:
	\[
	\operatorname{Res}[F,n] = \frac{\cos(n\theta)}{2\pi(1+n^2)}.
	\]
	
	\subsection*{b) The contour $\Gamma^{(N)}$}
	Define
	\[
	\Gamma^{(N)}_1(t) = \Big(N+\tfrac{1}{2}\Big)(1+it), \quad -1\leq t\leq 1,
	\]
	\[
	\Gamma^{(N)}_2(t) = \Big(N+\tfrac{1}{2}\Big)(-t+i), \quad -1\leq t\leq 1,
	\]
	\[
	\Gamma^{(N)}_3(t) = \Big(N+\tfrac{1}{2}\Big)(-1-it), \quad -1\leq t\leq 1,
	\]
	\[
	\Gamma^{(N)}_4(t) = \Big(N+\tfrac{1}{2}\Big)(t-i), \quad -1\leq t\leq 1.
	\]
	Then $\Gamma^{(N)}$ is the closed square contour with vertices
	$\pm(N+\tfrac{1}{2}) \pm i(N+\tfrac{1}{2})$.
	It encloses the poles at $\pm i$ and all integers $n$ with $|n|\leq N$.
	
	\subsection*{c) Bound on the contour integral}
	On $\Gamma^{(N)}$, $|z|\sim N$, so $|F(z)|=O(1/N^2)$.  
	Since the contour has length $O(N)$, we obtain
	\[
	\left| \int_{\Gamma^{(N)}} F(z)\,dz \right| \leq \frac{C}{N},
	\]
	for some constant $C$ independent of $N$.
	
	\subsection*{d) Residue theorem and Fourier series}
	By Cauchy's residue theorem,
	\[
	\int_{\Gamma^{(N)}} F(z)\,dz
	= 2\pi i \left( \operatorname{Res}[F,i]+\operatorname{Res}[F,-i] + \sum_{n=-N}^N \operatorname{Res}[F,n]\right).
	\]
	Hence
	\[
	\left| \frac{\cosh(\pi-\theta)}{2\sinh \pi}
	- \sum_{n=-N}^N \frac{\cos(n\theta)}{1+n^2} \right| \leq \frac{C}{N}.
	\]
	Letting $N\to\infty$, we conclude
	\[
	\frac{\cosh(\pi-\theta)}{2\sinh \pi}
	= \sum_{n=-\infty}^\infty \frac{e^{in\theta}}{1+n^2}.
	\]
	
	This gives the Fourier series expansion, uniformly in $\theta$.

	\section*{Introductory Examples: Simple Meromorphic Maps}
	
	\subsection*{Example 1: $f(z) = \tfrac{1}{z}$}
	
	\textbf{Description.}  
	For $z = re^{i\theta}$ we have
	\[
	\frac{1}{z} = \frac{1}{r} e^{-i\theta}.
	\]
	Thus $f(z)$ inverts the magnitude ($r \mapsto 1/r$) and reflects the angle 
	($\theta \mapsto -\theta$). The unit circle maps to itself with reversed orientation. 
	Lines through the origin are mapped to lines through the origin, reflected in the real axis.

\subsection*{Example: $f(z)=\tfrac{1}{z}$}

\textbf{Description.}  
For $z = re^{i\theta}$ we have
\[
\frac{1}{z} = \frac{1}{r} e^{-i\theta}.
\]
Thus the mapping $f(z)=1/z$ has the following geometric effects:
\begin{itemize}[noitemsep]
	\item \textbf{Inversion of radii:} $r \mapsto 1/r$.
	\item \textbf{Reflection of angles:} $\theta \mapsto -\theta$.
	\item The unit circle $|z|=1$ is invariant, but orientation is reversed.
	\item The interior of the unit disk $|z|<1$ maps to the exterior $|w|>1$, and vice versa.
	\item Rays through the origin map to reflected rays across the real axis.
	\item More generally, lines not through the origin map to circles, and circles not through the origin map to circles (inversion geometry).
\end{itemize}

\subsection*{Example: $f(z)=\tfrac{1}{z+1}$}

\textbf{Description.}  
This mapping is a shifted inversion. Let $w=z+1$. Then
\[
f(z) = \frac{1}{z+1} = \frac{1}{w}.
\]
Thus the inversion is centered at $z=-1$ instead of $z=0$. 
The function has a simple pole at $z=-1$. 
As $|z|\to\infty$, $f(z)\to 0$. 
Key geometric effects:
\begin{itemize}[noitemsep]
	\item Pole at $z=-1$ corresponds to $w=0$ under translation.
	\item The unit circle $|z|=1$ no longer maps to itself, but to another circle in the $w$-plane.
	\item Lines and circles not passing through $z=-1$ map to circles.
	\item Rays and bands relative to $z=-1$ are inverted accordingly.
\end{itemize}

\subsection*{Example: $f(z)=\tfrac{1}{z^2}$}

\textbf{Description.}  
For $z = re^{i\theta}$ we have
\[
f(z) = \frac{1}{z^2} = \frac{1}{r^2} e^{-2i\theta}.
\]
Thus the mapping $f(z)=1/z^2$ has the following geometric effects:
\begin{itemize}[noitemsep]
	\item \textbf{Inversion of radii (squared):} $r \mapsto 1/r^2$.
	\item \textbf{Angle doubling and reflection:} $\theta \mapsto -2\theta$.
	\item The unit circle $|z|=1$ is invariant (maps to itself).
	\item The interior of the unit disk $|z|<1$ maps to the exterior $|w|>1$, and vice versa.
	\item Rays at angle $\theta$ map to rays at angle $-2\theta$.
	\item There is a pole of order 2 at $z=0$.
\end{itemize}

\subsection*{Exercise 1.2 (Periodic Green's function for $-\psi''+\psi=f$)}
Suppose $f\in C^0(\mathbb{R})$ is $2\pi$-periodic, i.e.\ $f(\theta)=f(\theta+2\pi)$. For $\theta\in[0,2\pi)$ define
\begin{equation}\label{eq:psi-split}
	\psi(\theta):=\int_{0}^{\theta} f(\alpha)\,\frac{\cosh(\pi-\theta+\alpha)}{2\sinh\pi}\,d\alpha
	+\int_{\theta}^{2\pi} f(\alpha)\,\frac{\cosh(-\pi-\theta+\alpha)}{2\sinh\pi}\,d\alpha,
\end{equation}
and extend $\psi$ to $\mathbb{R}$ by periodicity $\psi(\theta+2\pi)=\psi(\theta)$.

\paragraph{Background / definition.}
It is convenient to write \eqref{eq:psi-split} using the periodic Green kernel
\begin{equation}\label{eq:Gdef}
	G(\theta,\alpha):=\frac{\cosh\!\big(\pi-|\theta-\alpha|\big)}{2\sinh\pi},\qquad \theta,\alpha\in[0,2\pi).
\end{equation}
Then \eqref{eq:psi-split} is equivalently
\begin{equation}\label{eq:psi-conv}
	\psi(\theta)=\int_{0}^{2\pi} f(\alpha)\,G(\theta,\alpha)\,d\alpha,
\end{equation}
i.e.\ $\psi=L^{-1}f$ for the periodic operator $L:=-\frac{d^2}{d\theta^2}+1$.

\begin{solution}
	\textbf{(a) $\boldsymbol{\psi\in C^0(\mathbb{R})}$.}
	On $[0,2\pi)$ the map $(\theta,\alpha)\mapsto f(\alpha)\,G(\theta,\alpha)$ is continuous and bounded.
	By dominated convergence, $\theta\mapsto\psi(\theta)$ is continuous. The periodic extension is continuous
	since both $f$ and $G$ are $2\pi$-periodic in each variable.
	
	\medskip
	\textbf{(b) First derivative and $\boldsymbol{\psi\in C^1(\mathbb{R})}$.}
	Differentiate \eqref{eq:psi-split} by Leibniz' rule. Using
	\(
	\partial_\theta\cosh(\pi-\theta+\alpha)=-\sinh(\pi-\theta+\alpha)
	\)
	and
	\(
	\partial_\theta\cosh(-\pi-\theta+\alpha)=-\sinh(-\pi-\theta+\alpha),
	\)
	and noting that the boundary terms cancel because
	$
	\frac{\cosh\pi}{2\sinh\pi}-\frac{\cosh\pi}{2\sinh\pi}=0,
	$
	we obtain
	\begin{equation}\label{eq:psi-prime}
		\psi'(\theta)= -\int_0^\theta f(\alpha)\,\frac{\sinh(\pi-\theta+\alpha)}{2\sinh\pi}\,d\alpha
		\;-\; \int_\theta^{2\pi} f(\alpha)\,\frac{\sinh(-\pi-\theta+\alpha)}{2\sinh\pi}\,d\alpha.
	\end{equation}
	Again by dominated convergence, the right-hand side is continuous, hence $\psi\in C^1(\mathbb{R})$.
	
	\medskip
	\textbf{(c) Second derivative and the equation $\boldsymbol{-\psi''+\psi=f}$.}
	Differentiate \eqref{eq:psi-prime} once more. Using
	\(
	\partial_\theta\sinh(\pi-\theta+\alpha)=-\cosh(\pi-\theta+\alpha)
	\)
	and
	\(
	\partial_\theta\sinh(-\pi-\theta+\alpha)=-\cosh(-\pi-\theta+\alpha),
	\)
	Leibniz' rule yields
	\begin{align*}
		\psi''(\theta)
		&=-\frac{f(\theta)\sinh\pi}{2\sinh\pi}+\int_{0}^{\theta} f(\alpha)\,\frac{\cosh(\pi-\theta+\alpha)}{2\sinh\pi}\,d\alpha \\
		&\quad-\frac{f(\theta)\sinh(-\pi)}{2\sinh\pi}+\int_{\theta}^{2\pi} f(\alpha)\,\frac{\cosh(-\pi-\theta+\alpha)}{2\sinh\pi}\,d\alpha \\
		&=-f(\theta)+\psi(\theta).
	\end{align*}
	Thus for all $\theta\in\mathbb{R}$, we have
	\begin{equation*}
		-\psi''(\theta)+\psi(\theta)=f(\theta).
	\end{equation*}
	Since $f$ is continuous, the display shows $\psi\in C^2(\mathbb{R})$.
\end{solution}
