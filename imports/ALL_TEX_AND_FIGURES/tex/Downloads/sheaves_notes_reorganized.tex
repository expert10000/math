\section{Foundations: presheaves, sheaves, and stalks}

\subsection{The central principle}
The three main ideas are:

\begin{enumerate}
	\item every presheaf has an associated sheaf;
	\item the image of a morphism of sheaves is naturally a subsheaf of the target;
	\item exactness of sheaves can be checked on stalks.
\end{enumerate}

All three are governed by the same fundamental principle:

\[
\boxed{
	\text{Sheaves are local objects, and stalks detect local behavior.}
}
\]

A sheaf is not merely a rule assigning an algebraic object to every open set.
It is a rule whose sections are determined by their local behavior and whose
compatible local sections glue uniquely.

The purpose of sheafification is to take a presheaf and force it to satisfy
this local gluing principle.

The purpose of the sheaf image is to replace the global notion of image by the
local notion of image.

The reason exactness can be checked on stalks is that exactness is a local
condition for sheaves.

\subsection{Presheaves}
Let $X$ be a topological space.

\begin{definition}[Presheaf of abelian groups]
	A presheaf of abelian groups on $X$ consists of the following data:
	
	\begin{enumerate}
		\item for every open set $U \subseteq X$, an abelian group $\mathcal F(U)$;
		\item for every inclusion of open sets $V \subseteq U$, a restriction map
		\[
		\rho^U_V : \mathcal F(U) \longrightarrow \mathcal F(V),
		\]
		usually written as
		\[
		s \longmapsto s|_V;
		\]
		\item the restriction maps satisfy
		\[
		\rho^U_U = \id_{\mathcal F(U)}
		\]
		and, whenever $W \subseteq V \subseteq U$,
		\[
		\rho^V_W \circ \rho^U_V = \rho^U_W.
		\]
	\end{enumerate}
\end{definition}

Thus a presheaf is a contravariant functor from the category of open subsets
of $X$ to the category of abelian groups:

\[
\mathcal F : \operatorname{Open}(X)^{op}
\longrightarrow
\mathbf{Ab}.
\]

The word ``contravariant'' means that inclusions go in one direction,

\[
V \subseteq U,
\]

but restriction maps go in the opposite direction,

\[
\mathcal F(U) \longrightarrow \mathcal F(V).
\]

\subsection{Sheaves}
A sheaf is a presheaf satisfying two additional axioms: locality and gluing.

\begin{definition}[Sheaf]
	A presheaf $\mathcal F$ on $X$ is called a sheaf if for every open set
	$U \subseteq X$ and every open cover
	\[
	U = \bigcup_{i \in I} U_i,
	\]
	the following two conditions hold.
	
	\textbf{Locality.}
	If $s,t \in \mathcal F(U)$ and
	\[
	s|_{U_i} = t|_{U_i}
	\]
	for every $i$, then
	\[
	s=t.
	\]
	
	\textbf{Gluing.}
	If we are given sections
	\[
	s_i \in \mathcal F(U_i)
	\]
	such that
	\[
	s_i|_{U_i \cap U_j}
	=
	s_j|_{U_i \cap U_j}
	\]
	for all $i,j$, then there exists a unique section
	\[
	s \in \mathcal F(U)
	\]
	such that
	\[
	s|_{U_i}=s_i
	\]
	for every $i$.
\end{definition}

The slogan is:

\[
\boxed{
	\text{A sheaf is a presheaf whose compatible local data glue uniquely.}
}
\]

\subsection{Stalks and germs}
The stalk is the algebraic object that captures the behavior of a presheaf
arbitrarily close to one point.

\begin{definition}[Stalk]
	Let $\mathcal F$ be a presheaf on $X$ and let $p \in X$.
	The stalk of $\mathcal F$ at $p$ is
	\[
	\mathcal F_p
	=
	\varinjlim_{p \in U} \mathcal F(U),
	\]
	where the direct limit is taken over all open neighborhoods $U$ of $p$.
\end{definition}

If $s \in \mathcal F(U)$ and $p \in U$, the image of $s$ in the stalk
$\mathcal F_p$ is called the germ of $s$ at $p$ and is denoted by

\[
s_p \in \mathcal F_p.
\]

Two sections $s \in \mathcal F(U)$ and $t \in \mathcal F(V)$ have the same germ
at $p$ if and only if there exists an open neighborhood

\[
W \subseteq U \cap V
\]

of $p$ such that

\[
s|_W = t|_W.
\]

Thus

\[
s_p=t_p
\]

means that $s$ and $t$ become equal after restricting to a sufficiently small
neighborhood of $p$.

\begin{remark}
	The stalk $\mathcal F_p$ forgets global behavior and remembers only local
	behavior near $p$.
\end{remark}

\subsection{Why stalks are important}
For sheaves, stalks detect equality.

\begin{proposition}[Equality can be checked on stalks]
	Let $\mathcal F$ be a sheaf on $X$, let $U \subseteq X$ be open, and let
	$s,t \in \mathcal F(U)$.
	
	Then
	\[
	s=t
	\]
	if and only if
	\[
	s_p=t_p
	\]
	for every $p \in U$.
\end{proposition}

\begin{proof}
	If $s=t$, then clearly $s_p=t_p$ for every $p \in U$.
	
	Conversely, suppose that $s_p=t_p$ for every $p \in U$.
	For each $p \in U$, equality of germs means that there exists an open
	neighborhood $V_p \subseteq U$ of $p$ such that
	\[
	s|_{V_p}=t|_{V_p}.
	\]
	The sets $V_p$ cover $U$.
	By the locality axiom for the sheaf $\mathcal F$, we get
	\[
	s=t.
	\]
\end{proof}

\begin{corollary}
	Let
	\[
	f : \mathcal F \longrightarrow \mathcal G
	\]
	be a morphism of sheaves. Then $f=0$ if and only if
	\[
	f_p : \mathcal F_p \longrightarrow \mathcal G_p
	\]
	is the zero map for every $p \in X$.
\end{corollary}

\begin{corollary}
	A morphism of sheaves
	\[
	f : \mathcal F \longrightarrow \mathcal G
	\]
	is an isomorphism if and only if every stalk map
	\[
	f_p : \mathcal F_p \longrightarrow \mathcal G_p
	\]
	is an isomorphism.
\end{corollary}

\section{Sheafification}

\subsection{Sheafification}
A presheaf does not necessarily satisfy locality and gluing.
Sheafification is the process of replacing a presheaf by the closest sheaf
associated to it.

\begin{definition}[Sheafification]
	Let $\mathcal F$ be a presheaf on $X$.
	A sheafification of $\mathcal F$ is a sheaf $\mathcal F^+$ together with a
	morphism of presheaves
	\[
	\theta : \mathcal F \longrightarrow \mathcal F^+
	\]
	such that for every sheaf $\mathcal G$ and every morphism of presheaves
	\[
	\varphi : \mathcal F \longrightarrow \mathcal G,
	\]
	there exists a unique morphism of sheaves
	\[
	\varphi^+ : \mathcal F^+ \longrightarrow \mathcal G
	\]
	such that
	\[
	\varphi = \varphi^+ \circ \theta.
	\]
\end{definition}

Equivalently, every map from $\mathcal F$ into a sheaf factors uniquely through
$\mathcal F^+$.

This is expressed by the diagram

\[
\begin{tikzcd}
	\mathcal F \arrow[r, "\varphi"] \arrow[d, "\theta"'] &
	\mathcal G \\
	\mathcal F^+ \arrow[ur, dashed, "\exists !\,\varphi^+"'] &
\end{tikzcd}
\]

where $\mathcal G$ is a sheaf.

\subsection{Construction of sheafification by stalks}
Let $\mathcal F$ be a presheaf on $X$.

For every $p \in X$, consider the stalk $\mathcal F_p$.
Form the disjoint union

\[
E_{\mathcal F}
=
\coprod_{p \in X} \mathcal F_p.
\]

There is a natural projection

\[
\pi : E_{\mathcal F} \longrightarrow X
\]

defined by

\[
\pi(a)=p
\qquad
\text{if } a \in \mathcal F_p.
\]

If $U \subseteq X$ is open and $s \in \mathcal F(U)$, define

\[
\widetilde{s} : U \longrightarrow E_{\mathcal F}
\]

by

\[
\widetilde{s}(p)=s_p.
\]

The set $\widetilde{s}(U)$ is the family of germs of $s$ over $U$:

\[
\widetilde{s}(U)
=
\{s_p : p \in U\}.
\]

We topologize $E_{\mathcal F}$ by declaring all sets of the form
$\widetilde{s}(U)$ to be a basis of open sets.

Then define

\[
\mathcal F^+(U)
=
\{\sigma : U \to E_{\mathcal F}
\mid
\pi \circ \sigma = \id_U,
\ \sigma \text{ is continuous}\}.
\]

Thus $\mathcal F^+(U)$ is the group of continuous sections of $\pi$ over $U$.

This construction is called the espace \'{e}tal\'{e} construction.

\subsection{Intuition behind the construction}
A section

\[
\sigma \in \mathcal F^+(U)
\]

chooses for every point $p \in U$ a germ

\[
\sigma(p) \in \mathcal F_p.
\]

However, not every arbitrary choice of germs is allowed.
The section $\sigma$ must be continuous with respect to the topology on
$E_{\mathcal F}$.

This continuity means precisely that locally, $\sigma$ comes from an actual
section of the original presheaf $\mathcal F$.

Therefore:

\[
\boxed{
	\mathcal F^+(U)
	=
	\{
	\text{families of germs over } U
	\text{ that are locally represented by sections of } \mathcal F
	\}.
}
\]

So sheafification does not create arbitrary pointwise data.
It creates exactly the locally representable data.

\subsection{The canonical morphism}
There is a natural morphism of presheaves

\[
\theta : \mathcal F \longrightarrow \mathcal F^+.
\]

For every open set $U \subseteq X$, define

\[
\theta_U : \mathcal F(U) \longrightarrow \mathcal F^+(U)
\]

by

\[
\theta_U(s)=\widetilde{s}.
\]

That is,

\[
\theta_U(s)(p)=s_p.
\]

So $\theta$ sends a section to its family of germs.

\begin{proposition}
	The maps $\theta_U$ define a morphism of presheaves
	\[
	\theta : \mathcal F \longrightarrow \mathcal F^+.
	\]
\end{proposition}

\begin{proof}
	Let $V \subseteq U$ be open.
	For $s \in \mathcal F(U)$ and $p \in V$, we have
	\[
	\theta_V(s|_V)(p)
	=
	(s|_V)_p
	=
	s_p
	=
	\theta_U(s)(p).
	\]
	Hence
	\[
	\theta_V(s|_V)=\theta_U(s)|_V.
	\]
	Thus the maps $\theta_U$ are compatible with restrictions.
\end{proof}

\subsection{Sheafification preserves stalks}
One of the most important facts about sheafification is that it does not change
stalks.

\begin{theorem}
	For every $p \in X$, there is a natural isomorphism
	\[
	(\mathcal F^+)_p \cong \mathcal F_p.
	\]
	Under this identification, the induced map
	\[
	\theta_p : \mathcal F_p \longrightarrow (\mathcal F^+)_p
	\]
	is the identity map.
\end{theorem}

\begin{proof}
	Define
	\[
	\Phi_p : (\mathcal F^+)_p \longrightarrow \mathcal F_p
	\]
	by
	\[
	\Phi_p(\sigma_p)=\sigma(p).
	\]
	
	This is well-defined because if two sections of $\mathcal F^+$ have the same
	germ at $p$, then they agree on some neighborhood of $p$, hence have the same
	value at $p$.
	
	We show that $\Phi_p$ is an isomorphism.
	
	First, let $a \in \mathcal F_p$.
	Choose a representative $s \in \mathcal F(U)$ with $p \in U$ such that
	\[
	a=s_p.
	\]
	Then $\widetilde{s} \in \mathcal F^+(U)$ and
	\[
	\Phi_p((\widetilde{s})_p)
	=
	\widetilde{s}(p)
	=
	s_p
	=
	a.
	\]
	Thus $\Phi_p$ is surjective.
	
	Now suppose
	\[
	\Phi_p(\sigma_p)=\Phi_p(\tau_p).
	\]
	Then
	\[
	\sigma(p)=\tau(p)
	\]
	inside $\mathcal F_p$.
	
	Because $\sigma$ and $\tau$ are continuous sections of the espace \'{e}tal\'{e},
	there exist neighborhoods $V,W$ of $p$ and sections
	\[
	s \in \mathcal F(V),
	\qquad
	t \in \mathcal F(W),
	\]
	such that
	\[
	\sigma|_V=\widetilde{s},
	\qquad
	\tau|_W=\widetilde{t}.
	\]
	The equality $\sigma(p)=\tau(p)$ means
	\[
	s_p=t_p.
	\]
	Therefore there exists a smaller neighborhood
	\[
	N \subseteq V \cap W
	\]
	of $p$ such that
	\[
	s|_N=t|_N.
	\]
	Thus
	\[
	\sigma|_N=\tau|_N.
	\]
	Hence
	\[
	\sigma_p=\tau_p.
	\]
	So $\Phi_p$ is injective.
	
	Therefore
	\[
	(\mathcal F^+)_p \cong \mathcal F_p.
	\]
\end{proof}

\begin{remark}
	Sheafification does not change local germs.
	It only repairs the global gluing behavior of the presheaf.
\end{remark}

\subsection{Universal property of sheafification}
\begin{theorem}[Universal property]
	Let $\mathcal F$ be a presheaf and let $\mathcal G$ be a sheaf.
	For every morphism of presheaves
	\[
	\varphi : \mathcal F \longrightarrow \mathcal G,
	\]
	there exists a unique morphism of sheaves
	\[
	\varphi^+ : \mathcal F^+ \longrightarrow \mathcal G
	\]
	such that
	\[
	\varphi=\varphi^+ \circ \theta.
	\]
\end{theorem}

\begin{proof}
	Let $U \subseteq X$ be open and let
	\[
	\sigma \in \mathcal F^+(U).
	\]
	For each $p \in U$, the value $\sigma(p)$ is an element of $\mathcal F_p$.
	The morphism $\varphi$ induces a map on stalks
	\[
	\varphi_p : \mathcal F_p \longrightarrow \mathcal G_p.
	\]
	
	We want to define a section $g \in \mathcal G(U)$ by requiring
	\[
	g_p=\varphi_p(\sigma(p))
	\]
	for all $p \in U$.
	
	This can be done locally.
	Since $\sigma$ is a continuous section of the espace \'{e}tal\'{e}, for every
	$p \in U$ there exists a neighborhood $U_p \subseteq U$ and a section
	\[
	s_p \in \mathcal F(U_p)
	\]
	such that
	\[
	\sigma|_{U_p}=\widetilde{s_p}.
	\]
	Define
	\[
	g_p := \varphi_{U_p}(s_p) \in \mathcal G(U_p).
	\]
	
	On overlaps, these locally defined sections agree.
	Indeed, their germs agree at every point of the overlap, because both are
	obtained by applying $\varphi$ to the same local family of germs.
	Since $\mathcal G$ is a sheaf, equality of germs implies local equality, and
	the sheaf gluing axiom gives a unique section
	\[
	g \in \mathcal G(U)
	\]
	such that
	\[
	g|_{U_p}=g_p
	\]
	locally.
	
	Define
	\[
	\varphi_U^+(\sigma)=g.
	\]
	
	These maps are compatible with restrictions, hence define a morphism of sheaves
	\[
	\varphi^+ : \mathcal F^+ \longrightarrow \mathcal G.
	\]
	
	Now let $s \in \mathcal F(U)$.
	Then
	\[
	\theta_U(s)=\widetilde{s}.
	\]
	For every $p \in U$,
	\[
	(\varphi_U^+(\widetilde{s}))_p
	=
	\varphi_p(s_p)
	=
	(\varphi_U(s))_p.
	\]
	Since $\mathcal G$ is a sheaf, sections with the same germs everywhere are
	equal. Hence
	\[
	\varphi_U^+(\theta_U(s))=\varphi_U(s).
	\]
	Therefore
	\[
	\varphi^+ \circ \theta=\varphi.
	\]
	
	Uniqueness follows because every section of $\mathcal F^+$ is locally of the
	form $\widetilde{s}$.
	Thus any morphism $\psi:\mathcal F^+ \to \mathcal G$ satisfying
	$\psi \circ \theta=\varphi$ must agree with $\varphi^+$ locally, hence globally.
\end{proof}

\subsection{Sheafification as an adjoint functor}
Let

\[
\PreSh_X
\]

denote the category of presheaves of abelian groups on $X$, and let

\[
\Sh_X
\]

denote the category of sheaves of abelian groups on $X$.

There is an inclusion functor

\[
i : \Sh_X \longrightarrow \PreSh_X.
\]

This functor simply regards a sheaf as a presheaf.

Sheafification defines a functor

\[
a : \PreSh_X \longrightarrow \Sh_X,
\qquad
a(\mathcal F)=\mathcal F^+.
\]

\begin{theorem}
	The sheafification functor is left adjoint to the inclusion functor:
	\[
	a \dashv i.
	\]
	Equivalently, for every presheaf $\mathcal F$ and every sheaf $\mathcal G$,
	there is a natural bijection
	\[
	\Hom_{\Sh_X}(\mathcal F^+,\mathcal G)
	\cong
	\Hom_{\PreSh_X}(\mathcal F,i\mathcal G).
	\]
\end{theorem}

\begin{proof}
	This is exactly the universal property of sheafification.
	A morphism
	\[
	\mathcal F \longrightarrow i\mathcal G
	\]
	from the presheaf $\mathcal F$ to the underlying presheaf of the sheaf
	$\mathcal G$ is equivalent to a unique morphism of sheaves
	\[
	\mathcal F^+ \longrightarrow \mathcal G.
	\]
	This correspondence is natural in both $\mathcal F$ and $\mathcal G$.
\end{proof}

Thus:

\[
\boxed{
	\mathcal F^+ \text{ is the best sheaf approximation of } \mathcal F.
}
\]

\subsection{Example: constant presheaf and constant sheaf}
Let $A$ be an abelian group.

Define the constant presheaf by

\[
\mathcal F(U)=A
\]

for every nonempty open set $U$, with all restriction maps equal to the identity.

This presheaf is usually not a sheaf.

Suppose $U$ has two disjoint nonempty open components $U_1$ and $U_2$.
Choose elements $a_1,a_2 \in A$ with $a_1 \neq a_2$.
Then we can define local sections
\[
s_1=a_1 \in \mathcal F(U_1),
\qquad
s_2=a_2 \in \mathcal F(U_2).
\]
Since $U_1 \cap U_2=\varnothing$, these sections are automatically compatible
on overlaps.

If $\mathcal F$ were a sheaf, they would glue to a section of $\mathcal F(U)$.
But a section of $\mathcal F(U)$ is just a single element of $A$, so it cannot
restrict to both $a_1$ and $a_2$.

Therefore the constant presheaf is not generally a sheaf.

Its sheafification is the constant sheaf

\[
\underline{A}.
\]

For each open set $U$,

\[
\underline{A}(U)
=
\{
\text{locally constant functions } U \to A
\}.
\]

Thus:

\[
\boxed{
	\text{The sheafification of the constant presheaf } A
	\text{ is the sheaf of locally constant } A\text{-valued functions.}
}
\]

\section{Images, exactness, and stalkwise criteria}

\subsection{Kernels of morphisms of sheaves}
Let
\[
f:\mathcal F \longrightarrow \mathcal G
\]
be a morphism of sheaves.

The kernel is defined objectwise by

\[
(\ker f)(U)
=
\ker
\left(
f_U:\mathcal F(U)\to\mathcal G(U)
\right).
\]

\begin{proposition}
	The presheaf $\ker f$ is already a sheaf.
\end{proposition}

\begin{proof}
	Let $U=\bigcup_i U_i$ be an open cover.
	
	Suppose $s,t \in (\ker f)(U)$ and
	\[
	s|_{U_i}=t|_{U_i}
	\]
	for every $i$.
	Since $\mathcal F$ is a sheaf, $s=t$.
	Thus locality holds.
	
	Now suppose $s_i \in (\ker f)(U_i)$ are compatible on overlaps.
	Since $\mathcal F$ is a sheaf, there exists a unique section
	\[
	s \in \mathcal F(U)
	\]
	such that
	\[
	s|_{U_i}=s_i.
	\]
	We need to show that $s \in (\ker f)(U)$, i.e.
	\[
	f_U(s)=0.
	\]
	But
	\[
	f_U(s)|_{U_i}
	=
	f_{U_i}(s_i)
	=
	0
	\]
	for every $i$.
	Since $\mathcal G$ is a sheaf, a section that is locally zero is zero.
	Therefore
	\[
	f_U(s)=0.
	\]
	Hence $s \in (\ker f)(U)$.
\end{proof}

So kernels of sheaves are computed objectwise.

\[
\boxed{
	\text{Kernels of sheaf morphisms are computed section by section.}
}
\]

\subsection{Images of morphisms of sheaves}
Images behave differently from kernels.

Let
\[
f:\mathcal F \longrightarrow \mathcal G
\]
be a morphism of sheaves.

For each open set $U$, there is a group homomorphism

\[
f_U:\mathcal F(U)\to\mathcal G(U).
\]

One might try to define the image presheaf by

\[
U \longmapsto \im(f_U).
\]

This is indeed a presheaf, denoted

\[
\operatorname{Im}^{\operatorname{pre}}(f),
\]

where

\[
\operatorname{Im}^{\operatorname{pre}}(f)(U)
=
\im
\left(
f_U:\mathcal F(U)\to\mathcal G(U)
\right).
\]

But this presheaf need not be a sheaf.

The reason is that being globally in the image is stronger than being locally
in the image.

A section
\[
s \in \mathcal G(U)
\]
may locally be of the form
\[
s|_{U_i}=f(t_i),
\]
but there may not exist a single global
\[
t \in \mathcal F(U)
\]
such that
\[
s=f(t).
\]

Therefore the sheaf image is defined as

\[
\im f
=
\left(
\operatorname{Im}^{\operatorname{pre}}(f)
\right)^+.
\]

\subsection{The sheaf image as a subsheaf}
Although the sheaf image is defined by sheafifying the presheaf image, it has
a very concrete description.

\begin{theorem}
	Let
	\[
	f:\mathcal F \longrightarrow \mathcal G
	\]
	be a morphism of sheaves.
	Then $\im f$ is naturally identified with the subsheaf $\mathcal H$ of
	$\mathcal G$ defined by
	
	\[
	\mathcal H(U)
	=
	\left\{
	s \in \mathcal G(U)
	\mid
	s \text{ is locally in the image of } f
	\right\}.
	\]
	
	Equivalently,
	
	\[
	\mathcal H(U)
	=
	\left\{
	s \in \mathcal G(U)
	\ \middle|\
	\forall p \in U,\ \exists V \subseteq U \text{ open with } p \in V,
	\ \exists t \in \mathcal F(V)
	\text{ such that } s|_V=f_V(t)
	\right\}.
	\]
\end{theorem}

\begin{proof}
	First observe that $\mathcal H(U)$ is a subgroup of $\mathcal G(U)$.
	It is closed under addition and inverses because $f_V$ is a group homomorphism.
	
	If $W \subseteq U$ and $s \in \mathcal H(U)$, then $s|_W$ is locally in the
	image of $f$.
	Indeed, if $p \in W$, choose $V \subseteq U$ and $t \in \mathcal F(V)$ such
	that
	\[
	s|_V=f_V(t).
	\]
	Then on $V \cap W$,
	\[
	s|_{V \cap W}
	=
	f_{V \cap W}(t|_{V \cap W}).
	\]
	Hence $s|_W \in \mathcal H(W)$.
	So $\mathcal H$ is a subpresheaf of $\mathcal G$.
	
	Now we prove that $\mathcal H$ is a sheaf.
	Let $U=\bigcup_i U_i$ and suppose
	\[
	s_i \in \mathcal H(U_i)
	\]
	are compatible on overlaps.
	Since $\mathcal G$ is a sheaf, there exists a unique
	\[
	s \in \mathcal G(U)
	\]
	such that
	\[
	s|_{U_i}=s_i.
	\]
	We must show that $s \in \mathcal H(U)$.
	
	Let $p \in U$.
	Choose $i$ such that $p \in U_i$.
	Since $s_i \in \mathcal H(U_i)$, there exists an open neighborhood
	\[
	V \subseteq U_i
	\]
	of $p$ and a section
	\[
	t \in \mathcal F(V)
	\]
	such that
	\[
	s_i|_V=f_V(t).
	\]
	But $s|_V=s_i|_V$, so
	\[
	s|_V=f_V(t).
	\]
	Thus $s$ is locally in the image of $f$.
	Hence $s \in \mathcal H(U)$.
	Therefore $\mathcal H$ is a subsheaf of $\mathcal G$.
	
	The presheaf image
	\[
	\operatorname{Im}^{\operatorname{pre}}(f)
	\]
	maps naturally into $\mathcal H$, since a section globally in the image is
	certainly locally in the image.
	
	By the universal property of sheafification, this map induces a morphism
	\[
	\left(
	\operatorname{Im}^{\operatorname{pre}}(f)
	\right)^+
	\longrightarrow
	\mathcal H.
	\]
	
	Conversely, every section of $\mathcal H$ is locally a section of the presheaf
	image. This is precisely what sheafification adds: compatible locally defined
	sections.
	Therefore
	\[
	\left(
	\operatorname{Im}^{\operatorname{pre}}(f)
	\right)^+
	\cong
	\mathcal H.
	\]
	Thus
	\[
	\im f \cong \mathcal H \subseteq \mathcal G.
	\]
\end{proof}

Hence:

\[
\boxed{
	(\im f)(U)
	=
	\{s \in \mathcal G(U)
	\mid
	s \text{ is locally of the form } f(t)\}.
}
\]

\subsection{Stalks of image sheaves}
\begin{proposition}
	Let
	\[
	f:\mathcal F \longrightarrow \mathcal G
	\]
	be a morphism of sheaves.
	Then for every $p \in X$,
	\[
	(\im f)_p
	=
	\im(f_p),
	\]
	where
	\[
	f_p:\mathcal F_p\to\mathcal G_p
	\]
	is the induced morphism on stalks.
\end{proposition}

\begin{proof}
	A germ in $(\im f)_p$ is represented by a section
	\[
	s \in (\im f)(U)
	\]
	for some neighborhood $U$ of $p$.
	By the local description of $\im f$, after shrinking $U$ if necessary, there is
	a section
	\[
	t \in \mathcal F(U)
	\]
	such that
	\[
	s=f_U(t).
	\]
	Therefore
	\[
	s_p=f_p(t_p),
	\]
	so
	\[
	s_p \in \im(f_p).
	\]
	
	Conversely, if $a \in \im(f_p)$, then there exists
	\[
	b \in \mathcal F_p
	\]
	such that
	\[
	a=f_p(b).
	\]
	Choose a representative
	\[
	t \in \mathcal F(U)
	\]
	of $b$.
	Then
	\[
	f_U(t) \in (\im f)(U),
	\]
	and the germ of $f_U(t)$ at $p$ is $a$.
	Thus
	\[
	a \in (\im f)_p.
	\]
	
	Hence
	\[
	(\im f)_p=\im(f_p).
	\]
\end{proof}

\subsection{Example: the exponential morphism}
Let $X$ be a complex manifold.
Let $\mathcal O_X$ be the sheaf of holomorphic functions and let
$\mathcal O_X^\times$ be the sheaf of nowhere-zero holomorphic functions.

There is a morphism of sheaves

\[
\exp : \mathcal O_X \longrightarrow \mathcal O_X^\times
\]

given by

\[
f \longmapsto e^{2\pi i f}.
\]

Locally, every nowhere-zero holomorphic function has a holomorphic logarithm.
Therefore locally every section of $\mathcal O_X^\times$ is in the image of
$\exp$.

Thus

\[
\im(\exp)=\mathcal O_X^\times
\]

as sheaves.

However, globally, a nowhere-zero holomorphic function need not have a global
logarithm.

For example, on

\[
X=\mathbb C^\times,
\]

the function

\[
z \longmapsto z
\]

has no global holomorphic logarithm.

Therefore the map on global sections

\[
\mathcal O_X(X)
\longrightarrow
\mathcal O_X^\times(X)
\]

need not be surjective, even though the morphism of sheaves is surjective.

This illustrates the important distinction:

\[
\boxed{
	\text{Surjective as a sheaf}
	\neq
	\text{surjective on global sections.}
}
\]

\subsection{Exactness of sheaves}
Let

\[
\cdots
\longrightarrow
\mathcal F_{i-1}
\xrightarrow{d_{i-1}}
\mathcal F_i
\xrightarrow{d_i}
\mathcal F_{i+1}
\longrightarrow
\cdots
\]

be a sequence of sheaves.

\begin{definition}[Exactness of sheaves]
	The sequence is exact at $\mathcal F_i$ if
	\[
	\im(d_{i-1})=\ker(d_i)
	\]
	as subsheaves of $\mathcal F_i$.
\end{definition}

Using the local description of the sheaf image, exactness at $\mathcal F_i$
means the following.

If
\[
s \in \mathcal F_i(U)
\]
satisfies
\[
d_i(s)=0,
\]
then for every point $p \in U$, there exists an open neighborhood
\[
V \subseteq U
\]
of $p$ and a section
\[
t \in \mathcal F_{i-1}(V)
\]
such that
\[
s|_V=d_{i-1}(t).
\]

Therefore:

\[
\boxed{
	\text{Exactness of sheaves means local exactness of sections.}
}
\]

It does not necessarily mean that every global kernel section is globally an
image.

\subsection{Exactness can be checked on stalks}
This is one of the most fundamental theorems in elementary sheaf theory.

\begin{theorem}[Stalkwise criterion for exactness]
	A sequence of sheaves
	\[
	\cdots
	\longrightarrow
	\mathcal F_{i-1}
	\xrightarrow{d_{i-1}}
	\mathcal F_i
	\xrightarrow{d_i}
	\mathcal F_{i+1}
	\longrightarrow
	\cdots
	\]
	is exact if and only if for every point $p \in X$, the induced sequence of
	stalks
	\[
	\cdots
	\longrightarrow
	(\mathcal F_{i-1})_p
	\xrightarrow{(d_{i-1})_p}
	(\mathcal F_i)_p
	\xrightarrow{(d_i)_p}
	(\mathcal F_{i+1})_p
	\longrightarrow
	\cdots
	\]
	is exact as a sequence of abelian groups.
\end{theorem}

\begin{proof}
	Assume first that the sequence of sheaves is exact.
	Then for every $i$,
	\[
	\im(d_{i-1})=\ker(d_i).
	\]
	Taking stalks at $p$, we obtain
	\[
	(\im d_{i-1})_p=(\ker d_i)_p.
	\]
	But stalks commute with kernels:
	\[
	(\ker d_i)_p=\ker((d_i)_p),
	\]
	and stalks of sheaf images are images of stalk maps:
	\[
	(\im d_{i-1})_p=\im((d_{i-1})_p).
	\]
	Therefore
	\[
	\im((d_{i-1})_p)=\ker((d_i)_p).
	\]
	Hence the stalk sequence is exact.
	
	Conversely, assume that for every $p \in X$, the stalk sequence is exact.
	We must show
	\[
	\im(d_{i-1})=\ker(d_i)
	\]
	as sheaves.
	
	First, the composition
	\[
	d_i\circ d_{i-1}
	\]
	is zero.
	Indeed, on stalks,
	\[
	(d_i)_p\circ (d_{i-1})_p=0
	\]
	for every $p$.
	Since morphisms of sheaves are detected on stalks,
	\[
	d_i\circ d_{i-1}=0.
	\]
	Thus
	\[
	\im(d_{i-1})\subseteq \ker(d_i).
	\]
	
	Now let
	\[
	s \in \ker(d_i)(U).
	\]
	Then
	\[
	d_i(s)=0.
	\]
	For every $p \in U$, the germ $s_p$ satisfies
	\[
	(d_i)_p(s_p)=0.
	\]
	Therefore
	\[
	s_p \in \ker((d_i)_p).
	\]
	By exactness of the stalk sequence,
	\[
	s_p \in \im((d_{i-1})_p).
	\]
	Hence there exists a germ
	\[
	t_p \in (\mathcal F_{i-1})_p
	\]
	such that
	\[
	(d_{i-1})_p(t_p)=s_p.
	\]
	Choose a representative
	\[
	t \in \mathcal F_{i-1}(V)
	\]
	on some neighborhood $V \subseteq U$ of $p$.
	Then
	\[
	(d_{i-1}(t))_p=s_p.
	\]
	Equality of germs means that, after possibly shrinking $V$,
	\[
	d_{i-1}(t)=s|_V.
	\]
	Thus $s$ is locally in the image of $d_{i-1}$.
	
	By the local description of the sheaf image,
	\[
	s \in \im(d_{i-1})(U).
	\]
	Therefore
	\[
	\ker(d_i)\subseteq \im(d_{i-1}).
	\]
	Together with the opposite inclusion, this gives
	\[
	\ker(d_i)=\im(d_{i-1}).
	\]
	Thus the sequence is exact.
\end{proof}

\subsection{Exactness versus exactness on global sections}
\begin{warning}
	Exactness of sheaves does not necessarily imply exactness on global sections.
\end{warning}

Suppose

\[
0
\longrightarrow
\mathcal F'
\longrightarrow
\mathcal F
\longrightarrow
\mathcal F''
\longrightarrow
0
\]

is a short exact sequence of sheaves.

This means that the sequence is exact locally, or equivalently on stalks.

However, the sequence of global sections

\[
0
\longrightarrow
\mathcal F'(X)
\longrightarrow
\mathcal F(X)
\longrightarrow
\mathcal F''(X)
\longrightarrow
0
\]

need not be exact on the right.

The map

\[
\mathcal F(X)\longrightarrow \mathcal F''(X)
\]

may fail to be surjective.

The reason is that a global section of $\mathcal F''$ may lift locally to
sections of $\mathcal F$, but the local lifts may fail to glue into one global
section.

This failure is measured by sheaf cohomology.

Thus:

\[
\boxed{
	\text{Sheaf exactness is local, whereas exactness of global sections is global.}
}
\]

\subsection{The exponential sequence}
A classical example is the exponential sequence on a complex manifold $X$:

\[
0
\longrightarrow
\mathbb Z
\longrightarrow
\mathcal O_X
\xrightarrow{\exp(2\pi i\,\cdot)}
\mathcal O_X^\times
\longrightarrow
0.
\]

This is exact as a sequence of sheaves.

The kernel of the exponential map consists of locally constant integer-valued
functions, so it is $\mathbb Z$.

The map is locally surjective because every nowhere-zero holomorphic function
has a local holomorphic logarithm.

However, it need not be surjective on global sections.
On $X=\mathbb C^\times$, the function
\[
z \mapsto z
\]
is a global section of $\mathcal O_X^\times$, but it has no global holomorphic
logarithm.

Thus this exact sequence of sheaves does not necessarily give an exact sequence
of global sections.

\subsection{Sheaves form an abelian category}
The category $\Sh_X$ of sheaves of abelian groups on $X$ is an abelian category.

This means that kernels, cokernels, images, coimages, and exact sequences exist
and behave analogously to the category of abelian groups.

However, some constructions require sheafification.

For a morphism
\[
f:\mathcal F\to\mathcal G,
\]
the kernel is computed objectwise:

\[
(\ker f)(U)
=
\ker
\left(
\mathcal F(U)\to\mathcal G(U)
\right).
\]

But the cokernel is generally computed by first taking the presheaf cokernel
and then sheafifying:

\[
\coker_{\Sh}(f)
=
\left(
U \longmapsto
\coker
\left(
\mathcal F(U)\to\mathcal G(U)
\right)
\right)^+.
\]

Similarly, the image is

\[
\im_{\Sh}(f)
=
\left(
U \longmapsto
\im
\left(
\mathcal F(U)\to\mathcal G(U)
\right)
\right)^+.
\]

Therefore:

\[
\boxed{
	\text{Kernels are objectwise, but images and cokernels must be sheafified.}
}
\]

\subsection{Monomorphisms, epimorphisms, and stalks}
For sheaves of abelian groups, many categorical properties can be checked on
stalks.

\begin{proposition}
	Let
	\[
	f:\mathcal F\to\mathcal G
	\]
	be a morphism of sheaves.
	
	Then:
	
	\begin{enumerate}
		\item $f$ is injective if and only if every
		\[
		f_p:\mathcal F_p\to\mathcal G_p
		\]
		is injective.
		
		\item $f$ is surjective as a morphism of sheaves if and only if every
		\[
		f_p:\mathcal F_p\to\mathcal G_p
		\]
		is surjective.
		
		\item $f$ is an isomorphism if and only if every
		\[
		f_p:\mathcal F_p\to\mathcal G_p
		\]
		is an isomorphism.
	\end{enumerate}
\end{proposition}

\begin{proof}
	The first statement follows because injectivity is equivalent to the vanishing
	of the kernel, and kernels are detected on stalks.
	
	The second statement follows because surjectivity of a sheaf morphism means
	that its image sheaf is the whole target sheaf.
	This can be checked on stalks:
	\[
	(\im f)_p=\im(f_p).
	\]
	Thus $\im f=\mathcal G$ if and only if
	\[
	\im(f_p)=\mathcal G_p
	\]
	for every $p$.
	
	The third statement follows from the first two.
\end{proof}


\subsection*{Overview}
The theory of sheafification, sheaf images, and stalkwise exactness is built
around the following chain of ideas.

First, a presheaf assigns data to open sets, but it may fail to glue compatible
local data.

Second, sheafification fixes this problem:

\[
\mathcal F \longmapsto \mathcal F^+.
\]

It preserves all stalks:

\[
(\mathcal F^+)_p\cong \mathcal F_p.
\]

It satisfies the universal property:

\[
\Hom_{\Sh_X}(\mathcal F^+,\mathcal G)
\cong
\Hom_{\PreSh_X}(\mathcal F,i\mathcal G).
\]

Thus sheafification is left adjoint to the inclusion

\[
i:\Sh_X\to\PreSh_X.
\]

Third, the image of a sheaf morphism is not merely the objectwise image.
It is the sheaf of sections locally in the image:

\[
(\im f)(U)
=
\{s\in\mathcal G(U)
\mid
s \text{ is locally of the form } f(t)\}.
\]

Fourth, exactness of sheaves is local exactness:

\[
\im(d_{i-1})=\ker(d_i)
\]

means that every local solution of

\[
d_i(s)=0
\]

is locally of the form

\[
s=d_{i-1}(t).
\]

Finally, because stalks capture local behavior, exactness can be checked on
stalks:

\[
\boxed{
	\text{A sequence of sheaves is exact}
	\iff
	\text{the corresponding sequence of stalks is exact at every point.}
}
\]

This is one of the main reasons stalks are indispensable in sheaf theory,
algebraic geometry, complex geometry, differential geometry, and cohomology.


	
	\section{Examples of sheaves}
	
	The examples below illustrate the following central principle of sheaf theory:
	
	\[
	\boxed{
		\text{Sheaf theory replaces global algebra by local algebra plus gluing.}
	}
	\]
	
	A sheaf is not merely a collection of groups $\F(U)$ attached to open sets
	$U \subseteq X$. A sheaf remembers how local data glue to global data.
	
	The main operational rule is:
	
	\[
	\boxed{
		\text{To understand sheaves, look at stalks.}
	}
	\]
	
	The examples are organized around three ideas:
	
	\begin{enumerate}
		\item sheafification;
		\item image sheaves;
		\item exactness checked on stalks.
	\end{enumerate}
	
	\subsection{Example: constant presheaf versus constant sheaf}
	
	Let
	
	\[
	X=\{p,q\}
	\]
	
	with the discrete topology, and let $A$ be an abelian group.
	
	Define the constant presheaf $\F$ by
	
	\[
	\F(U)=A
	\]
	
	for every nonempty open set $U \subseteq X$, with all restriction maps equal to
	the identity map.
	
	Thus
	
	\[
	\F(\{p\})=A,
	\qquad
	\F(\{q\})=A,
	\qquad
	\F(\{p,q\})=A.
	\]
	
	Now consider the open cover
	
	\[
	\{p,q\}=\{p\}\cup\{q\}.
	\]
	
	Choose elements
	
	\[
	a_p\in \F(\{p\})=A,
	\qquad
	a_q\in \F(\{q\})=A.
	\]
	
	Since
	
	\[
	\{p\}\cap\{q\}=\varnothing,
	\]
	
	there is no compatibility condition to check. Hence the two local sections
	$a_p$ and $a_q$ are automatically compatible.
	
	If $\F$ were a sheaf, these compatible local sections would glue to a section
	
	\[
	a\in \F(\{p,q\})=A
	\]
	
	such that
	
	\[
	a|_{\{p\}}=a_p,
	\qquad
	a|_{\{q\}}=a_q.
	\]
	
	But because the restriction maps are identity maps, this would mean
	
	\[
	a=a_p
	\qquad
	\text{and}
	\qquad
	a=a_q.
	\]
	
	Therefore we must have
	
	\[
	a_p=a_q.
	\]
	
	If $a_p\neq a_q$, no such global section exists.
	
	Thus the constant presheaf is not a sheaf.
	
	Its sheafification is the constant sheaf $\underline A$, where
	
	\[
	\underline A(U)
	=
	\{\text{locally constant functions } U\to A\}.
	\]
	
	For the discrete two-point space,
	
	\[
	\underline A(\{p,q\})
	=
	A\times A.
	\]
	
	Thus the sheafification adds the missing glued sections:
	
	\[
	\F^+(\{p,q\})=A\times A.
	\]
	
	Therefore
	
	\[
	\boxed{
		\text{The sheafification of the constant presheaf } A
		\text{ is the locally constant sheaf } \underline A.
	}
	\]
	
	\subsection{Example: bounded continuous functions on $\mathbb R$}
	
	Let
	
	\[
	X=\mathbb R.
	\]
	
	Define a presheaf $\B$ by
	
	\[
	\B(U)
	=
	\{\text{bounded continuous functions } U\to \mathbb R\}.
	\]
	
	This is a presheaf because the restriction of a bounded continuous function is
	again bounded and continuous.
	
	However, $\B$ is not a sheaf.
	
	For each positive integer $n$, let
	
	\[
	U_n=(-n,n).
	\]
	
	The open sets $U_n$ cover $\mathbb R$:
	
	\[
	\mathbb R=\bigcup_{n=1}^{\infty} U_n.
	\]
	
	On each $U_n$, define
	
	\[
	f_n:U_n\to \mathbb R,
	\qquad
	f_n(x)=x.
	\]
	
	Each $f_n$ is bounded on $U_n$, because
	
	\[
	|x|<n
	\qquad
	\text{for } x\in (-n,n).
	\]
	
	Therefore
	
	\[
	f_n\in \B(U_n).
	\]
	
	On overlaps,
	
	\[
	U_m\cap U_n=(-\min(m,n),\min(m,n)),
	\]
	
	the functions agree:
	
	\[
	f_m|_{U_m\cap U_n}=f_n|_{U_m\cap U_n}.
	\]
	
	Hence the family $(f_n)$ is compatible.
	
	If $\B$ were a sheaf, the $f_n$ would glue to a section
	
	\[
	f\in \B(\mathbb R).
	\]
	
	The glued function is necessarily
	
	\[
	f(x)=x.
	\]
	
	But the function
	
	\[
	x\mapsto x
	\]
	
	is not bounded on $\mathbb R$. Therefore
	
	\[
	f\notin \B(\mathbb R).
	\]
	
	Thus $\B$ is not a sheaf.
	
	The sheafification of $\B$ is the sheaf of locally bounded continuous functions.
	On $\mathbb R$, every continuous function is locally bounded, so
	
	\[
	\B^+
	=
	\C^0_{\mathbb R},
	\]
	
	where $\C^0_{\mathbb R}$ denotes the sheaf of continuous real-valued functions.
	
	This example shows that boundedness is a global condition, not a local
	condition.
	
	\[
	\boxed{
		\text{Sheaves only remember local conditions.}
	}
	\]
	
	\subsection{Example: locally constant functions}
	
	Let $A$ be an abelian group and let $X$ be any topological space.
	
	The constant sheaf $\underline A$ is defined by
	
	\[
	\underline A(U)
	=
	\{\text{locally constant functions } U\to A\}.
	\]
	
	This is a sheaf.
	
	If $U$ is connected, then every locally constant function $U\to A$ is constant.
	Therefore
	
	\[
	\underline A(U)\cong A
	\]
	
	when $U$ is connected.
	
	However, if $U$ has several connected components, then a locally constant
	function may take different values on different components.
	
	For example, suppose
	
	\[
	U=U_1\sqcup U_2
	\]
	
	is the disjoint union of two open connected components. Then
	
	\[
	\underline A(U)\cong A\times A.
	\]
	
	More generally, if $U$ has connected components $(U_\alpha)_{\alpha\in I}$,
	then
	
	\[
	\underline A(U)
	\cong
	\prod_{\alpha\in I} A.
	\]
	
	Thus the constant presheaf and the constant sheaf are different.
	
	The constant presheaf says:
	
	\[
	\F(U)=A.
	\]
	
	The constant sheaf says:
	
	\[
	\underline A(U)
	=
	\text{one copy of } A \text{ for each connected component of } U.
	\]
	
	So:
	
	\[
	\boxed{
		\text{constant presheaf } A
		\quad
		\xrightarrow{\text{sheafification}}
		\quad
		\text{locally constant sheaf } \underline A.
	}
	\]
	
	\subsection{Example: functions vanishing at one point}
	
	Let
	
	\[
	X=\mathbb R.
	\]
	
	Let $\C^0$ be the sheaf of continuous real-valued functions on $\mathbb R$.
	Fix the point $0\in\mathbb R$.
	
	Define a subsheaf
	
	\[
	\I_0\subseteq \C^0
	\]
	
	by
	
	\[
	\I_0(U)
	=
	\{f\in \C^0(U): f(0)=0\}
	\]
	
	if $0\in U$, and
	
	\[
	\I_0(U)=\C^0(U)
	\]
	
	if $0\notin U$.
	
	Thus $\I_0$ is the sheaf of continuous functions vanishing at $0$.
	
	There is an inclusion morphism
	
	\[
	\I_0\hookrightarrow \C^0.
	\]
	
	Its image is exactly $\I_0$.
	
	Let us examine the stalks.
	
	If $p\neq 0$, then locally near $p$ we can choose a neighborhood not containing
	$0$. On such a neighborhood, there is no condition of vanishing at $0$.
	Therefore
	
	\[
	(\I_0)_p=(\C^0)_p
	\qquad
	\text{for } p\neq 0.
	\]
	
	At $p=0$, the stalk
	
	\[
	(\I_0)_0
	\]
	
	consists of germs of continuous functions vanishing at $0$.
	
	Hence
	
	\[
	(\I_0)_p
	=
	\begin{cases}
		(\C^0)_p, & p\neq 0,\\[4pt]
		\{f_0\in(\C^0)_0 : f(0)=0\}, & p=0.
	\end{cases}
	\]
	
	This example shows that a subsheaf can impose a condition only at a single
	point.
	
	\[
	\boxed{
		\text{A local condition may appear only in one stalk.}
	}
	\]
	
	\subsection{Example: an image sheaf larger than the objectwise image}
	
	Let
	
	\[
	X=\mathbb C^\times
	=
	\mathbb C\setminus\{0\}.
	\]
	
	Let $\Ocal_X$ be the sheaf of holomorphic functions on $X$, and let
	$\Ocal_X^\times$ be the sheaf of nowhere-zero holomorphic functions.
	
	There is a morphism of sheaves
	
	\[
	\exp:\Ocal_X\to \Ocal_X^\times
	\]
	
	given by
	
	\[
	f\longmapsto e^{2\pi i f}.
	\]
	
	Locally, every nowhere-zero holomorphic function has a holomorphic logarithm.
	Therefore, locally, every section of $\Ocal_X^\times$ lies in the image of
	$\exp$.
	
	Hence the sheaf image is
	
	\[
	\im(\exp)=\Ocal_X^\times.
	\]
	
	However, globally, the function
	
	\[
	g(z)=z
	\]
	
	on $\mathbb C^\times$ has no global holomorphic logarithm.
	
	Indeed, if there were a holomorphic function
	
	\[
	h:\mathbb C^\times\to\mathbb C
	\]
	
	such that
	
	\[
	e^{2\pi i h(z)}=z,
	\]
	
	then $h$ would define a global branch of the logarithm on $\mathbb C^\times$.
	But such a branch does not exist.
	
	Therefore
	
	\[
	z\notin
	\im\left(
	\Ocal_X(X)\to \Ocal_X^\times(X)
	\right).
	\]
	
	Thus
	
	\[
	\boxed{
		\im_{\text{sheaf}}(\exp)=\Ocal_X^\times,
	}
	\]
	
	but
	
	\[
	\boxed{
		\im\left(
		\Ocal_X(X)\to \Ocal_X^\times(X)
		\right)
		\neq
		\Ocal_X^\times(X).
	}
	\]
	
	This is one of the most important examples of the difference between global
	image and sheaf image.
	
	\[
	\boxed{
		\text{Sheaf image means locally image, not globally image.}
	}
	\]
	
	\subsection{Example: the exponential sequence}
	
	On a complex manifold $X$, there is an exact sequence of sheaves
	
	\[
	0
	\longrightarrow
	\mathbb Z
	\longrightarrow
	\Ocal_X
	\xrightarrow{\exp(2\pi i\,\cdot)}
	\Ocal_X^\times
	\longrightarrow
	0.
	\]
	
	Here $\mathbb Z$ denotes the constant sheaf of locally constant integer-valued
	functions.
	
	The map
	
	\[
	\mathbb Z\to \Ocal_X
	\]
	
	is the inclusion of locally constant integer-valued functions into holomorphic
	functions.
	
	The map
	
	\[
	\Ocal_X\to \Ocal_X^\times
	\]
	
	is
	
	\[
	f\mapsto e^{2\pi i f}.
	\]
	
	We check exactness.
	
	First, the kernel of the exponential map consists of holomorphic functions
	$f$ satisfying
	
	\[
	e^{2\pi i f}=1.
	\]
	
	Locally, this means
	
	\[
	f\in \mathbb Z.
	\]
	
	Thus
	
	\[
	\ker(\exp)=\mathbb Z.
	\]
	
	Second, the exponential map is locally surjective. Indeed, every nowhere-zero
	holomorphic function has a local holomorphic logarithm.
	
	Therefore
	
	\[
	\im(\exp)=\Ocal_X^\times
	\]
	
	as sheaves.
	
	Hence the sequence
	
	\[
	0
	\longrightarrow
	\mathbb Z
	\longrightarrow
	\Ocal_X
	\xrightarrow{\exp(2\pi i\,\cdot)}
	\Ocal_X^\times
	\longrightarrow
	0
	\]
	
	is exact as a sequence of sheaves.
	
	However, it need not be exact on global sections.
	
	For example, if
	
	\[
	X=\mathbb C^\times,
	\]
	
	then
	
	\[
	z\mapsto z
	\]
	
	is a global section of $\Ocal_X^\times$, but it has no global holomorphic
	logarithm.
	
	Therefore the map
	
	\[
	\Ocal_X(X)\to \Ocal_X^\times(X)
	\]
	
	is not surjective.
	
	This example shows:
	
	\[
	\boxed{
		\text{Exactness of sheaves is local exactness.}
	}
	\]
	
	It also shows:
	
	\[
	\boxed{
		\text{A surjective morphism of sheaves need not be surjective on global sections.}
	}
	\]
	
	\subsection{Example: the smooth de Rham complex}
	
	Let $X$ be a smooth manifold.
	
	Let $\C^\infty_X$ be the sheaf of smooth real-valued functions, and let
	$\Omega_X^k$ be the sheaf of smooth differential $k$-forms.
	
	We have the de Rham complex of sheaves:
	
	\[
	0
	\longrightarrow
	\mathbb R
	\longrightarrow
	\C^\infty_X
	\xrightarrow{d}
	\Omega_X^1
	\xrightarrow{d}
	\Omega_X^2
	\xrightarrow{d}
	\cdots.
	\]
	
	This sequence is exact as a sequence of sheaves.
	
	The reason is the Poincaré lemma.
	
	The Poincaré lemma says that every closed differential form is locally exact.
	
	That is, if
	
	\[
	d\omega=0,
	\]
	
	then locally there exists a differential form $\eta$ such that
	
	\[
	\omega=d\eta.
	\]
	
	This is precisely sheaf exactness.
	
	However, globally, a closed form need not be exact.
	
	For example, on the circle $S^1$, the angular form
	
	\[
	d\theta
	\]
	
	is closed but not globally exact.
	
	Therefore the sequence of global sections
	
	\[
	0
	\longrightarrow
	\mathbb R
	\longrightarrow
	\C^\infty_X(X)
	\xrightarrow{d}
	\Omega_X^1(X)
	\xrightarrow{d}
	\Omega_X^2(X)
	\xrightarrow{d}
	\cdots
	\]
	
	need not be exact.
	
	Thus:
	
	\[
	\boxed{
		\text{The de Rham complex is locally exact, but not necessarily globally exact.}
	}
	\]
	
	Global obstructions are measured by de Rham cohomology.
	
	\subsection{Example: exactness with a skyscraper sheaf}
	
	Let
	
	\[
	X=\mathbb R.
	\]
	
	Let $\C^0$ be the sheaf of continuous real-valued functions, and let $\I_0$ be
	the subsheaf of continuous functions vanishing at $0$.
	
	Define the skyscraper sheaf $\mathbb R_0$ at $0$ by
	
	\[
	\mathbb R_0(U)
	=
	\begin{cases}
		\mathbb R, & 0\in U,\\
		0, & 0\notin U.
	\end{cases}
	\]
	
	There is a morphism
	
	\[
	\C^0\to \mathbb R_0
	\]
	
	defined as follows.
	
	If $0\in U$, send
	
	\[
	f\in \C^0(U)
	\]
	
	to
	
	\[
	f(0)\in \mathbb R.
	\]
	
	If $0\notin U$, the target is $0$.
	
	The kernel of this morphism consists exactly of those continuous functions
	which vanish at $0$. Hence
	
	\[
	\ker(\C^0\to \mathbb R_0)=\I_0.
	\]
	
	Therefore we obtain an exact sequence of sheaves
	
	\[
	0
	\longrightarrow
	\I_0
	\longrightarrow
	\C^0
	\longrightarrow
	\mathbb R_0
	\longrightarrow
	0.
	\]
	
	Let us check exactness on stalks.
	
	If $p\neq 0$, then
	
	\[
	(\mathbb R_0)_p=0.
	\]
	
	Also,
	
	\[
	(\I_0)_p=(\C^0)_p,
	\]
	
	because near $p\neq 0$ the condition of vanishing at $0$ is invisible.
	
	Thus the stalk sequence at $p\neq 0$ is
	
	\[
	0
	\longrightarrow
	(\C^0)_p
	\longrightarrow
	(\C^0)_p
	\longrightarrow
	0
	\longrightarrow
	0,
	\]
	
	which is exact.
	
	At $p=0$, we get
	
	\[
	0
	\longrightarrow
	(\I_0)_0
	\longrightarrow
	(\C^0)_0
	\longrightarrow
	\mathbb R
	\longrightarrow
	0.
	\]
	
	The map
	
	\[
	(\C^0)_0\to\mathbb R
	\]
	
	sends the germ $f_0$ to the value
	
	\[
	f(0).
	\]
	
	Its kernel is exactly
	
	\[
	(\I_0)_0.
	\]
	
	Therefore the stalk sequence is exact at $0$.
	
	Hence the sheaf sequence is exact because it is exact on every stalk.
	
	\[
	\boxed{
		\text{This example shows exactness checked point by point.}
	}
	\]
	
	\subsection{Example: a sheaf morphism surjective on stalks but not on global sections}
	
	Let
	
	\[
	X=S^1.
	\]
	
	Consider the morphism of sheaves
	
	\[
	d:\C^\infty_{S^1}\to \Omega^1_{S^1}.
	\]
	
	Here $\C^\infty_{S^1}$ is the sheaf of smooth functions on $S^1$, and
	$\Omega^1_{S^1}$ is the sheaf of smooth $1$-forms.
	
	Locally, every smooth $1$-form has a primitive. Therefore $d$ is surjective as
	a morphism of sheaves.
	
	Equivalently, for every point $p\in S^1$, the induced map on stalks
	
	\[
	d_p:(\C^\infty_{S^1})_p\to(\Omega^1_{S^1})_p
	\]
	
	is surjective.
	
	However, the map on global sections
	
	\[
	d:\C^\infty(S^1)\to \Omega^1(S^1)
	\]
	
	is not surjective.
	
	There exist global $1$-forms on $S^1$ that are not derivatives of global
	smooth functions.
	
	For example, the standard angular form is locally of the form $d\theta$, but
	there is no globally defined smooth single-valued function $\theta$ on $S^1$.
	
	Thus:
	
	\[
	\boxed{
		\text{Surjective on stalks}
		\iff
		\text{surjective as a morphism of sheaves,}
	}
	\]
	
	but
	
	\[
	\boxed{
		\text{surjective as a sheaf morphism}
		\not\Rightarrow
		\text{surjective on global sections.}
	}
	\]
	
	\subsection{Example: quotient sheaf and skyscraper sheaf}
	
	Again let
	
	\[
	X=\mathbb R.
	\]
	
	Consider the inclusion
	
	\[
	\I_0\hookrightarrow \C^0.
	\]
	
	The quotient sheaf is
	
	\[
	\C^0/\I_0.
	\]
	
	We claim that
	
	\[
	\C^0/\I_0\cong \mathbb R_0,
	\]
	
	where $\mathbb R_0$ is the skyscraper sheaf at $0$.
	
	We can see this by looking at stalks.
	
	If $p\neq 0$, then
	
	\[
	(\I_0)_p=(\C^0)_p.
	\]
	
	Therefore
	
	\[
	(\C^0/\I_0)_p=0
	\qquad
	\text{for } p\neq 0.
	\]
	
	At $p=0$, the quotient stalk is
	
	\[
	(\C^0/\I_0)_0
	=
	(\C^0)_0/(\I_0)_0.
	\]
	
	The quotient is isomorphic to $\mathbb R$ via evaluation at $0$:
	
	\[
	[f_0]\longmapsto f(0).
	\]
	
	Thus
	
	\[
	(\C^0/\I_0)_0\cong \mathbb R.
	\]
	
	So the quotient sheaf has stalk $\mathbb R$ at $0$ and zero stalk everywhere
	else. This is exactly the skyscraper sheaf $\mathbb R_0$.
	
	Therefore
	
	\[
	\boxed{
		\C^0/\I_0\cong \mathbb R_0.
	}
	\]
	
	This gives the exact sequence
	
	\[
	0
	\longrightarrow
	\I_0
	\longrightarrow
	\C^0
	\longrightarrow
	\mathbb R_0
	\longrightarrow
	0.
	\]
	
	\subsection{Example: presheaf image that is not a sheaf}
	
	Let
	
	\[
	X=S^1.
	\]
	
	Consider the morphism of sheaves
	
	\[
	d:\C^\infty_{S^1}\to \Omega^1_{S^1}.
	\]
	
	The presheaf image is
	
	\[
	U
	\longmapsto
	\im
	\left(
	d:\C^\infty(U)\to \Omega^1(U)
	\right).
	\]
	
	So it assigns to each open set $U$ the group of exact $1$-forms on $U$.
	
	On a small open arc
	
	\[
	U\subset S^1,
	\]
	
	every $1$-form is exact. Hence on such a $U$,
	
	\[
	\im
	\left(
	d:\C^\infty(U)\to \Omega^1(U)
	\right)
	=
	\Omega^1(U).
	\]
	
	However, on the whole circle $S^1$, not every $1$-form is exact.
	
	For example, the angular form is locally exact but not globally exact.
	
	Therefore the presheaf image fails to satisfy the sheaf gluing axiom.
	Compatible local exact forms may glue to a global form that is not globally
	exact.
	
	The sheafification of this presheaf image is all of $\Omega^1_{S^1}$:
	
	\[
	\im_{\text{sheaf}}(d)=\Omega^1_{S^1}.
	\]
	
	But the objectwise image on global sections satisfies
	
	\[
	\im
	\left(
	d:\C^\infty(S^1)\to \Omega^1(S^1)
	\right)
	\neq
	\Omega^1(S^1).
	\]
	
	Thus this example clearly shows why image sheaves must be sheafified.
	
	\[
	\boxed{
		\text{The objectwise image is not generally a sheaf.}
	}
	\]

	\subsection{Final conclusion}
	
	The examples above all express the same idea:
	
	\[
	\boxed{
		\text{Sheaves are local objects.}
	}
	\]
	
	Sheafification repairs a presheaf by adding missing local gluings.
	
	The sheaf image of a morphism is not the global image on each open set, but
	the sheaf of sections that are locally in the image.
	
	Exactness of sheaves is local exactness, which is why it can be checked on
	stalks.
	
	Thus the fundamental rule is:
	
	\[
	\boxed{
		\text{Local behavior is encoded by stalks, and global sections are obtained by gluing.}
	}
	\]
