\section*{Branches of $\log f$ on simply connected domains — with spaced layout}

\bigskip\hrule\bigskip

\subsection*{Problem}

Let $U$ be a simply connected domain in the complex plane and let $f$ be holomorphic on $U$ with $f(z)\neq 0$ for all $z\in U$.

\medskip

\noindent\textbf{(1)} Show that there exists a holomorphic $F$ on $U$ with $e^{F}=f$ (a branch of $\log f$ on $U$).

\noindent\textbf{(2)} Give an example showing that a holomorphic branch of the \emph{scalar} logarithm need not exist on the image $f(U)$.

\bigskip\hrule\bigskip

\subsection*{Background and tools}

\medskip

\noindent\textbf{(i) Logarithms via primitives.}

If $f$ is holomorphic and never zero on a domain $D$, then $\dfrac{f'}{f}$ is holomorphic on $D$. If there exists a holomorphic $F$ with $F' = \dfrac{f'}{f}$, then
\[
\frac{(e^{F})'}{e^{F}} = F' = \frac{f'}{f}
\qquad\Longrightarrow\qquad
\Big(\log \frac{e^{F}}{f}\Big)' = 0,
\]
so $e^{F}/f$ is constant; adding a constant to $F$ we obtain $e^{F}=f$.

\medskip

\noindent\textbf{(ii) Primitives on simply connected domains.}

If $g$ is holomorphic on a simply connected domain $U$, then $g$ has a global primitive $G$ on $U$; equivalently,
\[
\oint_{\gamma} g(z)\,dz=0 \quad \text{for every closed loop }\gamma\subset U.
\]

\medskip

\noindent\textbf{(iii) Integral and winding criteria.}

For any closed loop $\gamma$ in $\{f\neq 0\}$,
\[
\oint_{\gamma}\frac{f'(z)}{f(z)}\,dz \;=\; 2\pi i \cdot \mathrm{Wind}\!\big(f\circ\gamma,\,0\big).
\]
Thus a holomorphic branch of $\log f$ on a domain $D\subset\{f\neq 0\}$ exists iff the integral above is $0$ for all closed loops in $D$ (equivalently, $f\circ\gamma$ never winds around $0$).

\medskip

\noindent\textbf{(iv) Scalar $\log$ on a planar domain.}

A holomorphic branch of the scalar logarithm on a domain $V\subset\mathbf{C}\setminus\{0\}$ exists iff $V$ is simply connected (i.e.\ $V$ does not wrap around the origin). Annuli about $0$ and $\mathbf{C}\setminus\{0\}$ itself fail this.

\bigskip\hrule\bigskip

\subsection*{Solution to (1): Existence on $U$}

Because $f$ is holomorphic and nonvanishing, the function
\[
g(z) := \frac{f'(z)}{f(z)}
\]
is holomorphic on $U$. Since $U$ is simply connected, there exists a holomorphic primitive $G$ with $G'=g$ on $U$.

\medskip

Define
\[
H(z) := e^{G(z)} .
\]
Then
\[
\frac{H'(z)}{H(z)} = G'(z) = \frac{f'(z)}{f(z)}
\qquad\Longrightarrow\qquad
\Big(\log\frac{H(z)}{f(z)}\Big)'=0 .
\]
Hence $H/f$ is constant on $U$. Choose $c\in\mathbf{C}$ with $e^{c} = f(z_0)/H(z_0)$ for some (hence every) $z_0\in U$, and set
\[
F(z) := G(z) + c .
\]
Then $e^{F(z)} = f(z)$ on $U$, so $F$ is a holomorphic branch of $\log f$.

\medskip

\noindent\textit{Uniqueness up to $2\pi i\mathbf{Z}$.}
If $F_1$ and $F_2$ both satisfy $e^{F}=f$, then $e^{F_1-F_2}\equiv 1$, so $F_1-F_2$ is a constant of the form $2\pi i\,k$ with $k\in\mathbf{Z}$.

\bigskip\hrule\bigskip

\subsection*{Equivalent criteria on $U$}

\medskip

\noindent\textbf{Integral criterion.}
A holomorphic $F$ with $e^{F}=f$ exists on $U$ if and only if
\[
\oint_{\gamma}\frac{f'(z)}{f(z)}\,dz = 0
\quad\text{for every closed loop }\gamma\subset U.
\]

\medskip

\noindent\textbf{Winding criterion.}
Equivalently, a holomorphic branch of $\log f$ exists on $U$ iff
\[
\mathrm{Wind}\!\big(f\circ\gamma,0\big)=0
\quad\text{for every closed loop }\gamma\subset U.
\]

\bigskip\hrule\bigskip

\subsection*{Solution to (2): Examples where a scalar $\log$ on $f(U)$ fails}

\medskip

\noindent\textbf{Global example.}
Let $U=\mathbf{C}$ and $f(z)=e^{z}$. Then $f$ is entire, never zero, and a branch $\log f$ exists on $U$ (take $F(z)=z$). However
\[
f(U)=\mathbf{C}\setminus\{0\},
\]
which is not simply connected. For the unit circle $\Gamma=\{|w|=1\}$,
\[
\oint_{\Gamma}\frac{dw}{w} = 2\pi i \neq 0 .
\]
If a holomorphic scalar logarithm existed on $\mathbf{C}\setminus\{0\}$, its derivative would be $1/w$, forcing this integral to be $0$ for all closed loops—a contradiction.

\medskip

\noindent\textbf{Bounded variant (annulus image).}
Let $U=\{z:\ |z|<2\}$ and $f(z)=e^{z}$. Then $\log f$ exists on $U$ (again $F(z)=z$), but
\[
f(U)=\{e^{x+iy}:\ |x|<2,\ |y|<2\}
= \{w:\ e^{-2}<|w|<e^{2}\},
\]
which is an annulus encircling $0$. Any circle $\{|w|=r\}$ with $e^{-2}<r<e^{2}$ satisfies
\[
\oint \frac{dw}{w} = 2\pi i \neq 0 ,
\]
so no holomorphic scalar logarithm exists on $f(U)$.

\bigskip\hrule\bigskip

\subsection*{Geometric/topological intuition}

\medskip

\noindent The exponential map $\exp:\mathbf{C}\to\mathbf{C}\setminus\{0\}$ is a covering map with deck transformations $z\mapsto z+2\pi i k$ ($k\in\mathbf{Z}$). A global holomorphic scalar logarithm on a target domain $V\subset\mathbf{C}\setminus\{0\}$ would be a single-valued holomorphic inverse on $V$; this exists exactly when every loop in $V$ lifts to a closed loop, i.e.\ when $V$ is simply connected. Annuli and $\mathbf{C}\setminus\{0\}$ are not, so monodromy around $0$ prevents a single-valued holomorphic logarithm there.

\medskip

\noindent For $\log f$ on $U$, the only obstruction would be that some image loop $f(\gamma)$ winds around $0$. On simply connected $U$ with $f\neq 0$, the form $f'/f$ is exact, so all such windings vanish and a branch exists.

\bigskip\hrule\bigskip

\subsection*{Summary}

\medskip

\noindent On any simply connected domain $U$, every holomorphic nonvanishing $f$ admits a holomorphic $F$ with $e^{F}=f$ (unique up to $2\pi i\,\mathbf{Z}$). Nevertheless, a holomorphic scalar logarithm on $f(U)$ can fail to exist when $f(U)$ winds around the origin (e.g.\ $f=\exp$ gives $f(U)=\mathbf{C}\setminus\{0\}$ or an annulus).


\paragraph{Example.} Let $U=\mathbb{C}$ and $f(z)=e^{z}$.

\emph{Existence of $\log f$ on $U$.}
Since $f$ is entire and never zero, $f'/f\equiv 1$ is entire.
Because $U$ is simply connected, $G(z):=z$ is a primitive of $f'/f$.
Define $F(z):=G(z)=z$. Then $e^{F(z)}=e^{z}=f(z)$, so $F$ is a holomorphic branch of $\log f$ on $U$.
Uniqueness holds up to $2\pi i\mathbb{Z}$: if $F_1,F_2$ satisfy $e^{F_j}=f$, then $e^{F_1-F_2}\equiv 1$, hence $F_1-F_2\in 2\pi i\mathbb{Z}$ is a constant.

\medskip

\emph{Failure of a branch of $\log$ on $f(U)$.}
We have $f(U)=\mathbb{C}^\times$.
If there were a holomorphic branch $L$ of the scalar logarithm on $\mathbb{C}^\times$, then $L'(w)=1/w$ and, for every closed loop $\Gamma\subset\mathbb{C}^\times$,
\[
\oint_{\Gamma}\frac{dw}{w}=\oint_{\Gamma}L'(w)\,dw=0.
\]
Taking $\Gamma=\{\,|w|=1\,\}$ yields $\displaystyle\oint_{|w|=1}\frac{dw}{w}=2\pi i\neq 0$, a contradiction.
Therefore, no holomorphic branch of the scalar $\log$ exists on $\mathbb{C}^\times$.

\medskip

\emph{Monodromy/winding viewpoint.}
For any closed loop $\Gamma$ in a domain that avoids $0$,
\[
\oint_{\Gamma}\frac{dw}{w}=2\pi i\cdot \mathrm{Wind}(\Gamma,0).
\]
Thus a global holomorphic branch of $\log$ on a domain $V\subset\mathbb{C}^\times$ exists iff every loop in $V$ has winding number $0$ about $0$ (equivalently, $V$ is simply connected).
Here $V=\mathbb{C}^\times$ has nontrivial loops around $0$, so monodromy obstructs single-valuedness.

\medskip

\emph{Analytic continuation around a circle (explicit jump).}
Start with a local branch of $\log$ near $w=1$ and analytically continue once along the unit circle $w=e^{it}$, $t\in[0,2\pi]$.
The continuation returns to the starting point but accumulates the increment
\[
\int_{0}^{2\pi}\frac{d}{dt}\big(\log(e^{it})\big)\,dt
\;=\;\int_{0}^{2\pi} i\,dt
\;=\;2\pi i,
\]
so the value jumps by $2\pi i$. This nontrivial monodromy forbids a single-valued holomorphic logarithm on $\mathbb{C}^\times$.

\medskip

\emph{Bounded variant (annulus image).}
Let $U=\{z:|z|<2\}$ and $f(z)=e^{z}$. Then $F(z)=z$ gives a branch of $\log f$ on $U$, but
\[
f(U)=\{e^{x+iy}\,:\,|x|<2,\ |y|<2\}
=\{w\in\mathbb{C}^\times: e^{-2}<|w|<e^{2}\},
\]
an annulus encircling $0$. Any circle $\{|w|=r\}$ with $e^{-2}<r<e^{2}$ satisfies $\displaystyle\oint\frac{dw}{w}=2\pi i\neq 0$, so no holomorphic branch of the scalar $\log$ exists on $f(U)$.

\medskip

\emph{Local vs.\ global behavior.}
On any simply connected subdomain $V\subset\mathbb{C}^\times$ that does not wrap around $0$ (e.g.\ a slit plane or a sector of angle $<2\pi$), one \emph{can} choose a holomorphic branch of the scalar $\log$.
The obstruction is purely topological: encircling $0$ forces a $2\pi i$ jump.

\medskip

\emph{Connection with $\arg$ and $f'/f$.}
For $w\neq 0$, $\log w=\log|w|+i\,\arg w$; the impossibility of a global continuous $\arg$ on $\mathbb{C}^\times$ is the same obstruction.
Equivalently, if $L$ existed on $V$, then $(\log\circ f)'=(f'/f)$ would be the derivative of a \emph{global} holomorphic function on $V$.
On simply connected $U$ this always holds for $f'/f$ (hence $\log f$ exists on $U$); on $f(U)$ it fails exactly when some loop winds around $0$.

\medskip

\emph{Takeaway.}
A holomorphic nonvanishing $f$ on a simply connected $U$ always has a holomorphic logarithm $F$ with $e^{F}=f$ (unique up to $2\pi i\mathbb{Z}$).
But a holomorphic scalar logarithm on $f(U)$ may not exist if $f(U)$ winds around $0$ (as with $f=\exp$ giving $\mathbb{C}^\times$ or an annulus).

\begin{itemize}
\item $f$ is entire and never zero, so $f'/f \equiv 1$ is entire.
\item On the simply connected set $U=\mathbb{C}$, a primitive of $f'/f$ is $G(z)=z$.
\item Set $F(z):=G(z)=z$. Then $e^{F(z)}=e^{z}=f(z)$.
Thus $F$ is a (holomorphic) branch of $\log f$ on $U$.
\end{itemize}

\bigskip

\subsection*{2) Why a holomorphic $\log$ cannot exist on $f(U)=\mathbb{C}^{\times}$}

\begin{itemize}
\item $f(U)=\mathbb{C}\setminus\{0\}$ is not simply connected (its fundamental group is $\mathbb{Z}$);
any loop going once around $0$ has nonzero winding number.
\item If a holomorphic branch of the scalar logarithm $\Log$ existed on $\mathbb{C}^{\times}$, then
$(\Log)'(w)=1/w$ and hence, for every closed loop $\Gamma\subset\mathbb{C}^{\times}$,
\[
\oint_{\Gamma}\frac{dw}{w}
\;=\;
\oint_{\Gamma}(\Log)'(w)\,dw
\;=\;0.
\]
But for the unit circle $\{|w|=1\}$,
\[
\oint_{|w|=1}\frac{dw}{w} \;=\; 2\pi i \;\neq\; 0.
\]
Contradiction. Therefore no holomorphic branch of $\log$ exists on $\mathbb{C}^{\times}$.
\end{itemize}

\bigskip

\subsection*{3) Geometry/topology intuition (why the obstruction appears)}

\begin{itemize}
\item $\exp:\mathbb{C}\to\mathbb{C}^{\times}$ is a \emph{universal covering map}.
Adding $2\pi i$ in $z$–space leaves $e^{z}$ unchanged: $e^{z+2\pi i}=e^{z}$.
So loops around $0$ in $\mathbb{C}^{\times}$ lift to paths in $\mathbb{C}$ whose endpoints differ by $2\pi i$.
\item A single-valued holomorphic $\log$ on $\mathbb{C}^{\times}$ would be a global holomorphic inverse to $\exp$;
that would force all lifted loops to close, which they do not—this is the \emph{monodromy} obstruction.
\item Equivalent “metric picture”: horizontal lines $y=c$ map to rays of angle $c$; vertical lines $x=c$ map to circles
$|w|=e^{c}$. The image wraps endlessly around $0$; choosing a continuous argument globally is impossible.
\end{itemize}

\medskip

\noindent\textit{Variant (same phenomenon).}
Take $U=\{\,|z|<2\,\}$ and $f(z)=e^{z}$.
Then
\[
f(U)=\{\,e^{x+iy}:\ |x|<2,\ |y|<2\,\}
=\{\,w\in\mathbb{C}^{\times}:\ e^{-2}<|w|<e^{2}\,\},
\]
an annulus around $0$. The same integral test around any circle in that annulus,
$\displaystyle \oint \frac{dw}{w}=2\pi i$, blocks a global holomorphic $\log$ there.

\section*{Cauchy integral with principal value and boundary limits}

\subsection*{Problem (restated)}
Let $\Gamma$ be a simple smooth positively oriented closed curve in the plane, and let $f$ be continuously differentiable on $\Gamma$.
Define, for $z\in\mathbb{C}\setminus\Gamma$,
\[
F(z):=\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-z}\,d\zeta .
\]
This integral diverges when $z$ approaches a point $\zeta_0\in\Gamma$ with $f(\zeta_0)\neq 0$.
Define the principal value at $\zeta_0$ by
\[
\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_0}\,d\zeta
\;:=\;
\lim_{\varepsilon\to 0^+}\frac{1}{2\pi i}
\int_{\Gamma_\varepsilon}\frac{f(\zeta)}{\zeta-\zeta_0}\,d\zeta,
\]
where $\Gamma_\varepsilon$ is $\Gamma$ with a small open arc of radius $\varepsilon$ around $\zeta_0$ removed (the remaining part is traversed with the original orientation).

\bigskip\hrule\bigskip

\subsection*{Background and basic facts}

\paragraph{Analyticity off the curve.}
For $z\notin\Gamma$ the integrand is continuous on $\Gamma$ and the function $F$ is well defined and holomorphic in each of the two components of $\mathbb{C}\setminus\Gamma$ (the interior and the exterior).
Differentiation under the integral sign gives
\[
F'(z)=\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{(\zeta-z)^2}\,d\zeta,
\qquad z\in\mathbb{C}\setminus\Gamma.
\]

\paragraph{Recovery of interior analytic data.}
If $f$ extends to a function holomorphic in the interior of $\Gamma$ and continuous on $\overline{\mathrm{int}(\Gamma)}$, then the Cauchy integral formula yields
\[
F(z)=f(z)\quad\text{for }z\text{ in the interior of }\Gamma,
\qquad
F(z)=0\quad\text{for }z\text{ in the exterior and }|z|\ \text{large}.
\]

\paragraph{Principal value meaning.}
At $z=\zeta_0\in\Gamma$ the kernel $(\zeta-\zeta_0)^{-1}$ has a simple pole on the contour.
The principal value extracts the finite, symmetric part of the singular integral by removing a small arc around the pole and letting its size shrink to $0$.

\bigskip\hrule\bigskip

\subsection*{Nontangential limits and the Sokhotski--Plemelj formulas}

Let $F^\pm(\zeta_0)$ denote the limits of $F(z)$ as $z\to\zeta_0\in\Gamma$
from the interior $(+)$ and from the exterior $(-)$ along nontangential approaches.
If $f\in C^1(\Gamma)$, then these limits exist and
\[
\boxed{%
\begin{aligned}
F^{+}(\zeta_0) &= \frac{1}{2}\,f(\zeta_0)
\;+\;\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_0}\,d\zeta,\\[6pt]
F^{-}(\zeta_0) &= -\frac{1}{2}\,f(\zeta_0)
\;+\;\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_0}\,d\zeta.
\end{aligned}}
\]
Consequently,
\[
\boxed{\,F^{+}(\zeta_0)-F^{-}(\zeta_0)=f(\zeta_0)\,}
\qquad\text{and}\qquad
\boxed{\,\frac{F^{+}(\zeta_0)+F^{-}(\zeta_0)}{2}
=\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_0}\,d\zeta.}
\]

\bigskip

\subsection*{Sketch of derivation (indentation method)}

Fix $\zeta_0\in\Gamma$ and replace a small arc of $\Gamma$ near $\zeta_0$ by a circle of radius $\varepsilon$ centered at $\zeta_0$, lying inside (for $F^+$) or outside (for $F^-$) the domain. Then
\[
F(z)=\frac{1}{2\pi i}\!\!\int_{\Gamma_\varepsilon}\!\!\frac{f(\zeta)}{\zeta-z}\,d\zeta
\;+\;\frac{1}{2\pi i}\!\!\int_{C_\varepsilon}\!\!\frac{f(\zeta)}{\zeta-z}\,d\zeta,
\]
where $C_\varepsilon$ is the small circular arc. Let $z\to\zeta_0$ from the chosen side.
The first integral tends to the principal value term.
On $C_\varepsilon$, write $f(\zeta)=f(\zeta_0)+o(1)$ and compute
\[
\frac{1}{2\pi i}\int_{C_\varepsilon}\frac{f(\zeta_0)}{\zeta-\zeta_0}\,d\zeta
=\pm\frac{1}{2}f(\zeta_0),
\]
with the sign $+$ for the inner indentation and $-$ for the outer one (by orientation).
Adding the contributions yields the two Plemelj formulas above.

\bigskip\hrule\bigskip

\subsection*{Remarks}

\begin{itemize}
\item The regularity assumption $f\in C^1(\Gamma)$ can be weakened (e.g.\ H\"older continuity suffices for the nontangential limits).
\item If $\Gamma$ is a line, these formulas reduce to the classical Hilbert transform jump relations.
\item Orientation matters: reversing the orientation of $\Gamma$ switches the signs in the $\pm\tfrac12 f(\zeta_0)$ terms.
\end{itemize}

\paragraph{Concrete example on the unit circle.}
Let $\Gamma=\{\zeta:|\zeta|=1\}$ (counterclockwise) and $f(\zeta)=\zeta^{m}$ with $m\in\mathbb{Z}_{\ge0}$.
Define
\[
F(z)=\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-z}\,d\zeta,\qquad z\in\mathbb{C}\setminus\Gamma.
\]

\emph{Inside/outside values.}
By Cauchy’s integral formula, if $|z|<1$ then $F(z)=z^{m}$, while if $|z|>1$ then $F(z)=0$.
Thus $F$ is analytic on each side of $\Gamma$, equals $z^{m}$ in the interior, and vanishes in the exterior.

\medskip

\emph{Boundary limits and principal value.}
For $\zeta_{0}\in\Gamma$, the Sokhotski--Plemelj formulas yield
\[
F^{+}(\zeta_{0})
=\frac{1}{2}\zeta_{0}^{m}
+\operatorname{P.V.}\,\frac{1}{2\pi i}\!\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-\zeta_{0}}\,d\zeta,
\qquad
F^{-}(\zeta_{0})
=-\frac{1}{2}\zeta_{0}^{m}
+\operatorname{P.V.}\,\frac{1}{2\pi i}\!\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-\zeta_{0}}\,d\zeta.
\]
Since $F^{+}(\zeta_{0})=\zeta_{0}^{m}$ and $F^{-}(\zeta_{0})=0$, it follows that
\[
\boxed{\ \operatorname{P.V.}\,\frac{1}{2\pi i}\!\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-\zeta_{0}}\,d\zeta
=\frac{1}{2}\,\zeta_{0}^{m}\ },
\qquad
\boxed{\ F^{+}(\zeta_{0})-F^{-}(\zeta_{0})=\zeta_{0}^{m}=f(\zeta_{0})\ }.
\]

\medskip

\emph{Special case $m=0$.}
For $f(\zeta)\equiv 1$ we have $F(z)=1$ for $|z|<1$ and $F(z)=0$ for $|z|>1$, and on $\Gamma$,
\[
F^{+}(\zeta_{0})=1,\qquad F^{-}(\zeta_{0})=0,\qquad
\operatorname{P.V.}\,\frac{1}{2\pi i}\!\int_{|\zeta|=1}\frac{1}{\zeta-\zeta_{0}}\,d\zeta=\frac{1}{2}.
\]

\paragraph{Inside/outside values (unit circle \(\Gamma=\{|\zeta|=1\}\)).}
Consider
\[
F(z)=\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-z}\,d\zeta,
\qquad m\in\mathbb{Z}_{\ge 0}.
\]

\emph{Case \(|z|<1\) (pole inside).}
As a function of \(\zeta\), the integrand has a simple pole at \(\zeta=z\) lying inside \(\Gamma\).
By Cauchy’s integral formula,
\[
\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-z}\,d\zeta
=\zeta^{m}\big|_{\zeta=z}
=z^{m}.
\]
Equivalently, \(\operatorname{Res}_{\zeta=z}\frac{\zeta^{m}}{\zeta-z}=z^{m}\), so the integral is \(2\pi i\,z^{m}\), and dividing by \(2\pi i\) yields \(F(z)=z^{m}\).

\medskip

\emph{Case \(|z|>1\) (no pole inside).}
Now the pole \(\zeta=z\) lies outside \(\Gamma\), hence the integrand is holomorphic on and inside the contour.
By Cauchy’s theorem,
\[
\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-z}\,d\zeta=0,
\quad\text{i.e. }\; F(z)=0.
\]

\medskip

\emph{Series check for \(|z|>1\) (optional).}
For \(|z|>1\),
\[
\frac{\zeta^{m}}{\zeta-z}
=-\frac{1}{z}\,\frac{\zeta^{m}}{1-\zeta/z}
=-\frac{1}{z}\sum_{k=0}^{\infty}\frac{\zeta^{m+k}}{z^{k}}
\quad\big(|\zeta/z|<1\big).
\]
Integrating termwise over \(|\zeta|=1\) gives \(\displaystyle \oint \zeta^{n}\,d\zeta=0\) for all \(n\neq -1\),
so the whole integral is \(0\), confirming \(F(z)=0\) outside.

	\section*{Cauchy principal value on a smooth curve: existence and formula}
	
	\subsection*{Problem (exact restatement of the two tasks)}
	
	\noindent\textbf{(a)} Show that the principal value is well defined for all $\zeta_{0}\in\Gamma^{\ast}$, namely that the limit
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	\;:=\;\lim_{\varepsilon\downarrow 0}\,\frac{1}{2\pi i}\int_{\Gamma_{\varepsilon}}
	\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	\]
	exists, where $\Gamma$ is a simple smooth positively oriented curve, $f$ is continuously differentiable on $\Gamma^{\ast}$, and $\Gamma_{\varepsilon}$ is the part of $\Gamma$ that is at least $\varepsilon$–away from $\zeta_{0}$.
	
	\medskip
	
	\noindent\textbf{(b)} Show that
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	\;=\;
	\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
	\;+\;\frac{1}{2}\,f(\zeta_{0}).
	\]
	
	\bigskip\hrule\bigskip
	
	\subsection*{Background and notation}
	
	Let $\Gamma$ be a simple $C^{1}$ positively oriented closed curve in the plane and let $f\in C^{1}(\Gamma)$.
	For $z\in\mathbb{C}\setminus\Gamma$ define the Cauchy transform
	\[
	F(z):=\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-z}\,d\zeta.
	\]
	Then $F$ is holomorphic on each component of $\mathbb{C}\setminus\Gamma$.
	As $z\to\zeta_{0}\in\Gamma$, the kernel $(\zeta-z)^{-1}$ has a simple pole on the contour; the principal value removes a small arc around $\zeta_{0}$ and takes the limit as its length shrinks to $0$.
	
	\bigskip\hrule\bigskip
	
	\subsection*{Solution to (a): existence of the principal value}
	
	Fix $\zeta_{0}\in\Gamma$.
	Split the integrand into a regular part and a pure singular kernel:
	\[
	\frac{f(\zeta)}{\zeta-\zeta_{0}}
	\;=\;
	\underbrace{\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}}_{\text{regular near }\zeta_{0}}
	\;+\;
	\underbrace{\frac{f(\zeta_{0})}{\zeta-\zeta_{0}}}_{\text{pure kernel}}.
	\tag{1}
	\]
	
	\emph{Regular part.}
	Since $f\in C^{1}(\Gamma)$, there is a constant $C$ with
	\[
	|f(\zeta)-f(\zeta_{0})|\le C\,|\zeta-\zeta_{0}|
	\quad\text{for }\zeta\text{ near }\zeta_{0}.
	\]
	Hence the quotient $\dfrac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}$ is bounded and continuous on $\Gamma$ (interpreted by continuity at $\zeta_{0}$).
	Therefore
	\[
	\int_{\Gamma_{\varepsilon}}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
	\quad\text{converges as }\varepsilon\downarrow 0,
	\]
	so its contribution to the principal value has a proper limit (indeed, it is just the ordinary integral over $\Gamma$).
	
	\medskip
	
	\emph{Pure kernel.}
	On the small missing arc of radius $\varepsilon$ around $\zeta_{0}$ the integral
	\[
	\int\frac{d\zeta}{\zeta-\zeta_{0}}
	\]
	is $\pm\,\pi i$ in the limit (half of the $2\pi i$ residue), with the sign determined by whether the indentation is taken inside or outside, together with the positive orientation of $\Gamma$.
	Thus the contribution of the pure kernel to the truncated integral is uniformly bounded and tends, in the principal value sense, to the finite number $\dfrac{1}{2}\,2\pi i\cdot \dfrac{f(\zeta_{0})}{2\pi i}=\dfrac{1}{2}f(\zeta_{0})$ (precisely accounted for in part (b) below).
	
	\medskip
	
	Combining the two parts from (1) shows that the limit defining
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	\]
	exists for every $\zeta_{0}\in\Gamma$.
	
	\bigskip\hrule\bigskip
	
	\subsection*{Solution to (b): decomposition formula with the $\tfrac12 f(\zeta_{0})$ term}
	
	Start from the splitting \textnormal{(1)} and integrate over $\Gamma_{\varepsilon}$:
	\[
	\frac{1}{2\pi i}\int_{\Gamma_{\varepsilon}}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	=
	\frac{1}{2\pi i}\int_{\Gamma_{\varepsilon}}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
	+\frac{f(\zeta_{0})}{2\pi i}\int_{\Gamma_{\varepsilon}}\frac{d\zeta}{\zeta-\zeta_{0}}.
	\]
	Let $\varepsilon\downarrow 0$.
	The first term converges to the \emph{ordinary} integral over $\Gamma$ because its integrand is bounded and continuous on $\Gamma$:
	\[
	\lim_{\varepsilon\downarrow 0}\frac{1}{2\pi i}\int_{\Gamma_{\varepsilon}}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
	=
	\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta.
	\]
	For the second term,
	\[
	\lim_{\varepsilon\downarrow 0}\frac{1}{2\pi i}\int_{\Gamma_{\varepsilon}}\frac{d\zeta}{\zeta-\zeta_{0}}
	=\frac{1}{2},
	\]
	because the missing small arc carries half of the $2\pi i$ residue at $\zeta=\zeta_{0}$ (with the positive orientation prescribed in the statement).
	Therefore,
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	=
	\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
	+\frac{1}{2}\,f(\zeta_{0}),
	\]
	as required.
	
	\bigskip\hrule\bigskip
	
	\subsection*{Concrete example on the unit circle}
	
	Let $\Gamma=\{\zeta:|\zeta|=1\}$ (counterclockwise), $\zeta_{0}=e^{it_{0}}$, and $f(\zeta)=\zeta^{m}$ with $m\ge 0$.
	Then
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-\zeta_{0}}\,d\zeta
	=
	\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}-\zeta_{0}^{m}}{\zeta-\zeta_{0}}\,d\zeta
	+\frac{1}{2}\,\zeta_{0}^{m}.
	\]
	But the integrand on the right is regular on the unit circle, and Cauchy’s formula gives the value
	\[
	\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}-\zeta_{0}^{m}}{\zeta-\zeta_{0}}\,d\zeta
	=\frac{1}{2}\,\zeta_{0}^{m},
	\]
	so the principal value equals $\dfrac{1}{2}\zeta_{0}^{m}$.
	This matches the Plemelj jump relations: the interior limit is $\zeta_{0}^{m}$ and the exterior limit is $0$, hence their average is exactly the principal value.
	
	
	% ==============================================================
	% Relation between the principal value and the index (winding number)
	% ==============================================================
	
	\subsection*{Connection between the principal value and the index of the curve}
	
	For a smooth closed curve $\Gamma$ and a point $\zeta_{0}$ not on $\Gamma$, the \emph{index} (or winding number) of $\Gamma$ about $\zeta_{0}$ is defined by
	\[
	\operatorname{Ind}(\Gamma;\zeta_{0})
	\;:=\;
	\frac{1}{2\pi i}\oint_{\Gamma}\frac{d\zeta}{\zeta-\zeta_{0}}.
	\]
	It counts how many times $\Gamma$ winds around $\zeta_{0}$, with orientation.
	For a positively oriented simple Jordan curve,
	\[
	\operatorname{Ind}(\Gamma;\zeta_{0})=
	\begin{cases}
		1,& \zeta_{0}\ \text{inside }\Gamma,\\[4pt]
		0,& \zeta_{0}\ \text{outside }\Gamma.
	\end{cases}
	\]
	
	\medskip
	
	When $\zeta_{0}\in\Gamma$, the integral above diverges because of the singularity at $\zeta=\zeta_{0}$.
	However, its \emph{Cauchy principal value} exists and equals \emph{half} the index:
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\oint_{\Gamma}\frac{d\zeta}{\zeta-\zeta_{0}}
	\;=\;
	\frac{1}{2}\,\operatorname{Ind}(\Gamma;\zeta_{0}).
	\]
	For a positively oriented simple curve, $\operatorname{Ind}(\Gamma;\zeta_{0})=1$,
	so
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\oint_{\Gamma}\frac{d\zeta}{\zeta-\zeta_{0}}
	=\frac{1}{2}.
	\]
	
	\bigskip\hrule\bigskip
	
	\subsection*{General decomposition in terms of the index}
	
	In the decomposition formula
	\[
	\operatorname{P.V.}\,\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	=
	\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
	+\frac{1}{2}\,f(\zeta_{0}),
	\]
	the last term $\tfrac12 f(\zeta_{0})$ comes precisely from the principal value of the pure kernel.
	Replacing the constant $\tfrac12$ by $\tfrac12\operatorname{Ind}(\Gamma;\zeta_{0})$ gives the more general form valid for any winding number:
	\[
	\boxed{%
		\operatorname{P.V.}\,\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
		=
		\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}\,d\zeta
		+\frac{\operatorname{Ind}(\Gamma;\zeta_{0})}{2}\,f(\zeta_{0}).
	}
	\]
	
	\bigskip\hrule\bigskip
	
	\subsection*{Consequences for boundary limits (Plemelj jump)}
	
	Define
	\[
	F(z)=\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)}{\zeta-z}\,d\zeta,
	\qquad z\in\mathbb{C}\setminus\Gamma.
	\]
	Then the nontangential limits from the interior $(+)$ and exterior $(-)$ satisfy
	\[
	F^{\pm}(\zeta_{0})
	=\operatorname{P.V.}\,\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	\ \pm\ \frac{\operatorname{Ind}(\Gamma;\zeta_{0})}{2}\,f(\zeta_{0}),
	\qquad \zeta_{0}\in\Gamma.
	\]
	Hence the \emph{jump relation}
	\[
	\boxed{\,F^{+}(\zeta_{0})-F^{-}(\zeta_{0})
		=\operatorname{Ind}(\Gamma;\zeta_{0})\,f(\zeta_{0}).\,}
	\]
	
	\bigskip
	
	\subsection*{Remarks}
	
	\begin{itemize}
		\item For a positively oriented simple contour, $\operatorname{Ind}(\Gamma;\zeta_{0})=1$, so the jump equals $f(\zeta_{0})$ and the principal value term corresponds to $\tfrac{1}{2}f(\zeta_{0})$.
		\item If $\Gamma$ winds $k$ times around $\zeta_{0}$, then
		\(\operatorname{Ind}(\Gamma;\zeta_{0})=k\),
		and all the $\tfrac{1}{2}$ coefficients are multiplied by $k$.
		\item When $\zeta_{0}$ lies outside $\Gamma$, $\operatorname{Ind}=0$, so the principal value is simply
		\(\frac{1}{2\pi i}\oint_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta=0.\)
	\end{itemize}
	
	\subsection*{(c) Interior/Exterior limits and the jump}
	
	Let $F(z)=\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-z}\,d\zeta$ with $\Gamma$ a simple $C^{1}$ Jordan curve and $f\in C^{1}(\Gamma)$. For $\zeta_{0}\in\Gamma$,
	\[
	\lim_{\substack{z\to\zeta_{0}\\ z\ \mathrm{inside}\ \Gamma}} F(z)
	=\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	+\frac{1}{2}f(\zeta_{0}),
	\]
	\[
	\lim_{\substack{z\to\zeta_{0}\\ z\ \mathrm{outside}\ \Gamma}} F(z)
	=\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	-\frac{1}{2}f(\zeta_{0}).
	\]
	Thus $F$ has a jump $F^{+}(\zeta_{0})-F^{-}(\zeta_{0})=f(\zeta_{0})$.
	
	\medskip
	\emph{Proof sketch.}
	Split $\dfrac{f(\zeta)}{\zeta-\zeta_{0}}
	=\dfrac{f(\zeta)-f(\zeta_{0})}{\zeta-\zeta_{0}}
	+\dfrac{f(\zeta_{0})}{\zeta-\zeta_{0}}$.
	The first term yields the principal value. The second contributes $\pm \tfrac12 f(\zeta_{0})$
	via an indentation around $\zeta_{0}$ (the short circular arc integral equals $\pm \pi i$).
	
	\bigskip
	
	\subsection*{(d) What if $\Gamma$ is not simple / only piecewise smooth? What if $f$ is not $C^{1}$?}
	
	\paragraph{Non-simple curves (multiple windings).}
	Replace $\pm\tfrac12$ by $\pm \tfrac12\,\mathrm{Ind}(\Gamma;\zeta_{0})$:
	\[
	F^{\pm}(\zeta_{0})
	=\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{\Gamma}\frac{f(\zeta)}{\zeta-\zeta_{0}}\,d\zeta
	\ \pm\ \frac{\mathrm{Ind}(\Gamma;\zeta_{0})}{2}\,f(\zeta_{0}),
	\qquad
	F^{+}-F^{-}=\mathrm{Ind}(\Gamma;\zeta_{0})\,f(\zeta_{0}).
	\]
	
	\paragraph{Piecewise smooth / Lipschitz curves.}
	For $\Gamma$ Lipschitz (e.g.\ piecewise $C^{1}$), the same formulas hold for \emph{nontangential} limits. The Cauchy singular integral operator is bounded on $L^{p}(\Gamma)$, $1<p<\infty$, so Plemelj limits exist a.e.
	
	\paragraph{Weaker regularity on $f$.}
	If $f\in C^{0,\alpha}(\Gamma)$ (H\"older), the limits hold pointwise for all $\zeta_{0}\in\Gamma$.
	If $f\in L^{p}(\Gamma)$, $1<p<\infty$, the jump relations hold in the $L^{p}$ (and a.e.) sense.
	
	\medskip
	
	\paragraph{Example (unit circle).}
	For $\Gamma=\{|\zeta|=1\}$ and $f(\zeta)=\zeta^{m}$ ($m\ge0$):
	\[
	F(z)=\begin{cases} z^{m}, & |z|<1,\\ 0, & |z|>1,\end{cases}
	\quad\Rightarrow\quad
	F^{+}(e^{it})-F^{-}(e^{it})=e^{im t}=f(e^{it}),
	\]
	and
	\(
	\operatorname{P.V.}\,\frac{1}{2\pi i}\int_{|\zeta|=1}\frac{\zeta^{m}}{\zeta-e^{it}}\,d\zeta
	=\tfrac12 e^{imt}.
	\)
	
	\paragraph{(ii) Curve only piecewise smooth / Lipschitz (corners allowed).}
	The Plemelj formulas remain valid for \emph{Lipschitz} curves (e.g.\ piecewise $C^{1}$ with finite length) when boundary limits are taken \emph{nontangentially}.
	This is classical in singular–integral theory (Calder\'on–Zygmund): the Cauchy singular integral operator is bounded on $L^{p}(\Gamma)$ for $1<p<\infty$, and the nontangential boundary limits exist almost everywhere.
	
	\medskip
	
	\paragraph{(iii) Regularity of $f$.}
	\begin{itemize}
		\item For classical \emph{pointwise} limits at every boundary point, it suffices to assume $f\in C^{0,\alpha}(\Gamma)$ (H\"older) for some $\alpha\in(0,1]$.
		\item For \emph{almost-everywhere} (a.e.) limits and $L^{p}$ jump relations, it is enough to assume $f\in L^{p}(\Gamma)$ with $1<p<\infty$.
		\item The original hypothesis $f\in C^{1}(\Gamma)$ is more than enough; it guarantees the principal value is well defined and that the interior/exterior limits exist \emph{at every} $\zeta_{0}\in\Gamma$ (no exceptional set).
	\end{itemize}
	
	\section*{Cauchy integrals on the unit circle: analysis of the integrands and evaluation}
	
	\subsection*{Problem}
	Let $\gamma$ be the positively oriented unit circle $\{|z|=1\}$. Use Cauchy Integral Formulas to evaluate
	\[
	\text{(a)}\quad \int_{\gamma}\frac{e^{z}\cos z}{z}\,dz,
	\qquad
	\text{(b)}\quad \int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz.
	\]
	
	\subsection*{Background}
	\begin{itemize}
		\item \textbf{Cauchy Integral Formula (CIF).} If $f$ is holomorphic on and inside $\gamma$ and $a$ lies inside, then
		\[
		\frac{1}{2\pi i}\int_{\gamma}\frac{f(\zeta)}{\zeta-a}\,d\zeta=f(a),
		\qquad
		\frac{1}{2\pi i}\int_{\gamma}\frac{f(\zeta)}{(\zeta-a)^{n+1}}\,d\zeta=\frac{f^{(n)}(a)}{n!}.
		\]
		\item \textbf{Residues.} $\displaystyle \int_{\gamma}g(z)\,dz=2\pi i\sum \operatorname{Res}(g;\text{poles inside }\gamma)$.
		\item $e^{z}$ and $\cos z$ are entire; hence $N(z):=e^{z}\cos z$ is entire.
	\end{itemize}
	
	\subsection*{Anatomy of the integrands}
	Let $N(z)=e^{z}\cos z$.
	\begin{itemize}
		\item \textbf{Zeros of $N$.} $e^{z}$ has no zeros; $\cos z$ has zeros at $z=\frac{\pi}{2}+k\pi$ for $k\in\mathbb{Z}$ (simple zeros), none on $|z|=1$.
		\item \textbf{Poles.} $\dfrac{N(z)}{z}$ has a simple pole at $z=0$. $\dfrac{N(z)}{z^{3}}$ has a triple pole at $z=0$. Because $N(0)=1\neq0$, the pole orders are not reduced.
		\item \textbf{Domains.} Both $\dfrac{N}{z}$ and $\dfrac{N}{z^{3}}$ are holomorphic on $\mathbb{C}\setminus\{0\}$; the only singularity inside $\gamma$ is at $0$.
		\item \textbf{Magnitude along lines.} With $z=x+iy$,
		\[
		|N(z)|=e^{x}\,|\cos(x+iy)|,\qquad
		|\cos(x+iy)|^{2}=\cos^{2}x\,\cosh^{2}y+\sin^{2}x\,\sinh^{2}y.
		\]
		Thus along vertical lines ($x$ fixed) $|N|\sim e^{x}\tfrac{e^{|y|}}{2}$ for large $|y|$; along horizontal lines ($y$ fixed) $|N|$ oscillates with an $e^{x}$ envelope.
	\end{itemize}
	
	\subsection*{Evaluations}
	
	\paragraph{(a) The integral of $\dfrac{N(z)}{z}$.}
	By CIF with $f=N$ and $a=0$,
	\[
	\int_{\gamma}\frac{e^{z}\cos z}{z}\,dz
	=2\pi i\,N(0)
	=2\pi i\,e^{0}\cos 0
	=2\pi i.
	\]
	
	\paragraph{(b) The integral of $\dfrac{N(z)}{z^{3}}$.}
	By the higher-order CIF with $n=2$,
	\[
	\int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz
	=2\pi i\,\frac{N^{(2)}(0)}{2!}.
	\]
	Compute derivatives:
	\[
	N'(z)=e^{z}(\cos z-\sin z),\qquad
	N''(z)=e^{z}(\cos z-\sin z)+e^{z}(-\sin z-\cos z)=-2e^{z}\sin z,
	\]
	so $N''(0)=0$. Therefore
	\[
	\int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz=0.
	\]
	
	\subsection*{Remarks}
	\begin{itemize}
		\item Laurent expansions at $0$ show directly that the coefficient of $1/z$ in $\dfrac{N}{z^{3}}$ is $0$; hence the integral vanishes.
		\item The jump/limit issues do not arise here since the integrands are holomorphic on and inside $\gamma$ except at $0$, which is isolated and handled by CIF.
	\end{itemize}
	
	
	\subsection*{Detailed anatomy of \(N(z)=e^{z}\cos z\)}
	
	\paragraph{Algebraic forms.}
	\[
	N(z)=e^{z}\cos z=\tfrac12\!\left(e^{(1+i)z}+e^{(1-i)z}\right).
	\]
	Thus \(N\) is entire of order \(1\) and exponential type \(\sqrt2\).
	
	\paragraph{Zeros and critical points.}
	\begin{itemize}
		\item Zeros: \(e^{z}\neq0\), so \(N(z)=0\iff \cos z=0\iff z=\frac{\pi}{2}+k\pi\) for \(k\in\mathbb{Z}\) (all simple).
		\item Critical points: \(N'(z)=e^{z}(\cos z-\sin z)=0\iff \tan z=1\iff z=\frac{\pi}{4}+k\pi\) (nondegenerate since \(N''(z)=-2e^{z}\sin z\neq0\) there).
		\item Singularities: none (entire).
	\end{itemize}
	
	\paragraph{Real/imag parts and modulus.}
	For \(z=x+iy\),
	\[
	e^{z}=e^{x}(\cos y+i\sin y),\qquad
	\cos(x+iy)=\cos x\cosh y-i\sin x\sinh y,
	\]
	so
	\[
	\begin{aligned}
		N(z)&=e^{x}\Big(\cos y\cos x\cosh y+\sin y\sin x\sinh y\Big)\\
		&\quad+i\,e^{x}\Big(\sin y\cos x\cosh y-\cos y\sin x\sinh y\Big),
	\end{aligned}
	\]
	and
	\[
	|N(x+iy)|^{2}=e^{2x}\Big(\cos^{2}x\,\cosh^{2}y+\sin^{2}x\,\sinh^{2}y\Big).
	\]
	
	\paragraph{Vertical lines (\(x\) fixed).}
	As \(|y|\to\infty\), \(\cosh y\sim\sinh y\sim \tfrac12 e^{|y|}\), hence
	\[
	|N(x+iy)|\sim \tfrac12\,e^{x+|y|}\qquad(|y|\to\infty).
	\]
	
	\paragraph{Horizontal lines (\(y\) fixed).}
	Using \(|\cos(x+iy)|\in\big[|\sinh y|,|\cosh y|\big]\) as \(x\) varies,
	\[
	|N(x+iy)|\in\big[e^{x}|\sinh y|,\ e^{x}|\cosh y|\big],
	\]
	so \( |N(x+iy)| \) oscillates in \(x\) with an \(e^{x}\) envelope (growth for \(x\to+\infty\), decay for \(x\to-\infty\)).
	
	\paragraph{Directional growth (indicator).}
	Along \(z=re^{i\theta}\),
	\[
	\log |N(re^{i\theta})|\approx r\,\max\{\cos\theta+\sin\theta,\ \cos\theta-\sin\theta\},
	\]
	so the indicator is \(h(\theta)=\max\{\cos\theta+\sin\theta,\ \cos\theta-\sin\theta\}\), with maximum \(\sqrt2\) at \(\theta=\pm\frac{\pi}{4}\).
	
	\paragraph{Series at \(0\).}
	\[
	e^{z}\cos z=1+z+\underbrace{0}_{z^{2}}+\frac{z^{3}}{3}+O(z^{4}),
	\]
	implying, for example,
	\[
	\int_{|z|=1}\frac{e^{z}\cos z}{z^{3}}\,dz=2\pi i\,[z^{-1}]\Big(\frac{e^{z}\cos z}{z^{3}}\Big)=0.
	\]
	
	
	\section*{Exponential integrands on the unit circle}
	
	\subsection*{Anatomy of \(E(z)=e^{z}\)}
	
	\begin{itemize}
		\item \textbf{Entire/type.} \(E\) is entire of order \(1\) and exponential type \(1\).
		\item \textbf{Zeros/critical points.} No zeros; \(E'(z)=e^{z}\neq 0\) everywhere, so no critical points.
		\item \textbf{Modulus and lines.} For \(z=x+iy\),
		\(|E(z)|=e^{x}\).
		Along vertical lines (\(x\) fixed) the modulus is constant and the argument increases linearly with \(y\).
		Along horizontal lines (\(y\) fixed) the modulus is \(e^{x}\) (exponential growth/decay).
		\item \textbf{Directional growth.} On the ray \(z=re^{i\theta}\), \(|E(re^{i\theta})|=e^{r\cos\theta}\); indicator \(h(\theta)=\cos\theta\).
		\item \textbf{Series at \(0\).} \(e^{z}=1+z+\frac{z^{2}}{2}+\frac{z^{3}}{6}+\cdots\).
	\end{itemize}
	
	\subsection*{Cauchy evaluations on \(\gamma=\{|z|=1\}\)}
	
	Let \(f(z)=e^{z}\).
	\paragraph{(a)} Using the Cauchy Integral Formula with \(a=0\),
	\[
	\int_{\gamma}\frac{e^{z}}{z}\,dz
	=2\pi i\,f(0)
	=2\pi i.
	\]
	
	\paragraph{(b)} Using the higher-order Cauchy formula with \(n=2\),
	\[
	\int_{\gamma}\frac{e^{z}}{z^{3}}\,dz
	=2\pi i\,\frac{f^{(2)}(0)}{2!}
	=2\pi i\cdot\frac{1}{2}
	=\pi i.
	\]
	(Here \(f^{(2)}(0)=e^{0}=1\). Equivalently, the \(z^{-1}\)-coefficient in the Laurent expansion of \(e^{z}/z^{3}\) is \(1/2\).)
	
	\section*{Problem}
	Let $\gamma$ be the positively oriented unit circle $\{|z|=1\}$. 
	Use the Cauchy Integral Formula to evaluate
	\[
	\text{(a)}\quad \int_{\gamma}\frac{e^{z}\cos z}{z}\,dz,
	\qquad
	\text{(b)}\quad \int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz.
	\]
	
	\section*{Background and method}
	We recall two standard versions of the Cauchy Integral Formula.
	
	\paragraph{Cauchy Integral Formula (CIF).}
	If $f$ is holomorphic in a neighborhood of the closed contour $\gamma$ and $a$ is a point inside $\gamma$, then
	\[
	f(a)=\frac{1}{2\pi i}\oint_{\gamma}\frac{f(\zeta)}{\zeta-a}\,d\zeta.
	\]
	Equivalently,
	\[
	\oint_{\gamma}\frac{f(\zeta)}{\zeta-a}\,d\zeta = 2\pi i\,f(a).
	\]
	
	\paragraph{Higher-order Cauchy formula.}
	If $f$ is holomorphic near $a$, then for $n\ge0$,
	\[
	f^{(n)}(a)=\frac{n!}{2\pi i}\oint_{\gamma}\frac{f(\zeta)}{(\zeta-a)^{n+1}}\,d\zeta,
	\]
	or equivalently,
	\[
	\oint_{\gamma}\frac{f(\zeta)}{(\zeta-a)^{n+1}}\,d\zeta = 2\pi i\,\frac{f^{(n)}(a)}{n!}.
	\]
	
	---
	
	\section*{Structure of the integrand}
	
	Let
	\[
	f(z)=e^{z}\cos z.
	\]
	Then $f$ is \emph{entire}, since both $e^{z}$ and $\cos z$ are entire.
	
	\paragraph{Zeros.}
	$e^{z}\neq0$ everywhere, and $\cos z=0$ at $z=\frac{\pi}{2}+k\pi$, $k\in\mathbb{Z}$. 
	None of these zeros lie on $|z|=1$, so $f$ has no singularities on $\gamma$.
	
	\paragraph{Singularity structure of the integrands.}
	\[
	\frac{f(z)}{z} \text{ has a simple pole at } z=0, \qquad
	\frac{f(z)}{z^{3}} \text{ has a triple pole at } z=0.
	\]
	Since $f(0)=e^{0}\cos 0=1\neq0$, the pole orders are not reduced.
	
	---
	
	\section*{(a) Evaluation of $\displaystyle \int_{\gamma}\frac{e^{z}\cos z}{z}\,dz$}
	
	By the Cauchy Integral Formula with $a=0$, we have
	\[
	\oint_{\gamma}\frac{f(z)}{z}\,dz=2\pi i\,f(0).
	\]
	Compute $f(0)$:
	\[
	f(0)=e^{0}\cos 0=1.
	\]
	Therefore,
	\[
	\boxed{\int_{\gamma}\frac{e^{z}\cos z}{z}\,dz=2\pi i.}
	\]
	
	---
	
	\section*{(b) Evaluation of $\displaystyle \int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz$}
	
	By the higher-order Cauchy formula with $a=0$ and $n=2$,
	\[
	\oint_{\gamma}\frac{f(z)}{z^{3}}\,dz
	=2\pi i\,\frac{f^{(2)}(0)}{2!}.
	\]
	
	\paragraph{Compute the derivatives.}
	\[
	f'(z)=e^{z}(\cos z-\sin z), \qquad
	f''(z)=e^{z}(\cos z-\sin z)+e^{z}(-\sin z-\cos z)=-2e^{z}\sin z.
	\]
	Thus,
	\[
	f''(0)=-2e^{0}\sin 0=0.
	\]
	Hence
	\[
	\boxed{\int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz=0.}
	\]
	
	---
	
	\section*{Alternative check using the series expansion}
	We can verify the result directly from the Maclaurin series.
	
	\[
	e^{z}=1+z+\frac{z^{2}}{2}+\frac{z^{3}}{6}+\cdots,\qquad
	\cos z=1-\frac{z^{2}}{2}+\frac{z^{4}}{24}-\cdots.
	\]
	Multiplying,
	\[
	e^{z}\cos z
	= (1+z+\tfrac{z^{2}}{2}+\tfrac{z^{3}}{6}+\cdots)(1-\tfrac{z^{2}}{2}+\tfrac{z^{4}}{24}-\cdots)
	= 1+z+0\cdot z^{2}+\tfrac{1}{3}z^{3}+O(z^{4}).
	\]
	
	Then
	\[
	\frac{e^{z}\cos z}{z^{3}}
	=\frac{1}{z^{3}}+\frac{1}{z^{2}}+0\cdot\frac{1}{z}+\tfrac{1}{3}+O(z),
	\]
	and the coefficient of $\frac{1}{z}$ is zero.
	By Cauchy’s theorem (or the residue theorem),
	\[
	\oint_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz=2\pi i\,[z^{-1}]\!\left(\frac{e^{z}\cos z}{z^{3}}\right)=0.
	\]
	
	---
	
	\section*{Summary and interpretation}
	
	\[
	\boxed{
		\begin{aligned}
			(a)\quad &\int_{\gamma}\frac{e^{z}\cos z}{z}\,dz = 2\pi i,\\[4pt]
			(b)\quad &\int_{\gamma}\frac{e^{z}\cos z}{z^{3}}\,dz = 0.
		\end{aligned}
	}
	\]
	
	\paragraph{Geometric intuition.}
	- The integrands have their only singularity at the origin, entirely enclosed by the contour $\gamma$.
	- The first integral measures the value of $f(0)$ via the Cauchy kernel $1/z$.
	- The second integral involves the second derivative of $f$ at $0$; since $f''(0)=0$, its contribution vanishes.
	- In the Laurent expansion viewpoint, the integral picks out the coefficient of $\frac{1}{z}$ in $\frac{f(z)}{z^{3}}$, which here is zero.
	
	
\section*{Mean value property and maximum modulus principle}

\subsection*{Problem (exact)}
Let $f:U\to\mathbb{C}$ be holomorphic on a domain $U$.
\begin{enumerate}
	\item[(a)] Show that, if $\overline{B}(a,r)\subset U$, then
	\[
	f(a)=\frac{1}{2\pi}\int_{0}^{2\pi} f(a+re^{it})\,dt.
	\]
	\item[(b)] Suppose $a\in U$ is such that $|f(a)|=\max_{U}|f|$. Show that $|f|$ must be constant near $a$.
	\item[(c)] Deduce that $f$ is constant on all of $U$.
	\quad\emph{Hint:} Consider $S=\{z\in U: |f(z)|=|f(a)|\}$.
\end{enumerate}

\subsection*{Background}
\paragraph{Cauchy integral formula on circles.}
If $\overline{B}(a,r)\subset U$, then
\[
f(a)=\frac{1}{2\pi i}\int_{|\zeta-a|=r}\frac{f(\zeta)}{\zeta-a}\,d\zeta.
\]
With $\zeta=a+re^{it}$ and $d\zeta=ire^{it}dt$, this becomes the circular average.

\paragraph{Equality in the triangle inequality.}
For an integrable complex function $g$, $\left|\int g\right|\le\int|g|$ with equality iff $g$ has a constant argument almost everywhere on the path of integration.

\paragraph{Cauchy integral for derivatives.}
If $f$ is holomorphic on $\overline{B}(a,r)$, then for $n\ge1$,
\[
f^{(n)}(a)=\frac{n!}{2\pi i}\int_{|\zeta-a|=r}\frac{f(\zeta)}{(\zeta-a)^{n+1}}\,d\zeta.
\]

\subsection*{Solution}

\paragraph{(a) Mean value property.}
By Cauchy’s formula,
\[
f(a)=\frac{1}{2\pi i}\int_{|\zeta-a|=r}\frac{f(\zeta)}{\zeta-a}\,d\zeta
=\frac{1}{2\pi i}\int_{0}^{2\pi}\frac{f(a+re^{it})}{re^{it}}\,(ire^{it})\,dt
=\frac{1}{2\pi}\int_{0}^{2\pi} f(a+re^{it})\,dt.
\]

\paragraph{(b) Local constancy at an interior maximum.}
Assume $|f(a)|=\max_{U}|f|$. For $r>0$ with $\overline{B}(a,r)\subset U$,
\[
|f(a)|
=\left|\frac{1}{2\pi}\int_{0}^{2\pi} f(a+re^{it})\,dt\right|
\le \frac{1}{2\pi}\int_{0}^{2\pi}|f(a+re^{it})|\,dt
\le \frac{1}{2\pi}\int_{0}^{2\pi}|f(a)|\,dt
=|f(a)|.
\]
Thus equality holds throughout. Hence $|f(a+re^{it})|=|f(a)|$ and $f(a+re^{it})$ has a constant argument (the same as $f(a)$) for almost every $t$. Therefore
\[
f(a+re^{it})=f(a)\quad\text{for all }t,
\]
i.e. $f$ is constant on the circle $|z-a|=r$. For such an $r$, Cauchy’s formula for $n\ge1$ gives
\[
f^{(n)}(a)=\frac{n!}{2\pi i}\int_{|\zeta-a|=r}\frac{f(\zeta)}{(\zeta-a)^{n+1}}\,d\zeta
=\frac{n!}{2\pi i}\,f(a)\int_{|\zeta-a|=r}\frac{1}{(\zeta-a)^{n+1}}\,d\zeta=0.
\]
Hence the Taylor series of $f$ at $a$ is the constant $f(a)$; there exists $\rho>0$ such that $f\equiv f(a)$ on $B(a,\rho)$. In particular, $|f|$ is constant near $a$.

\paragraph{(c) Constancy on the whole domain.}
Let $M=|f(a)|$ and $S=\{z\in U: |f(z)|=M\}$. The set $S$ is closed (preimage of $\{M\}$ under the continuous map $z\mapsto|f(z)|$). By part (b), for each $z\in S$ there exists a neighborhood on which $f$ is constant, hence that neighborhood lies in $S$; thus $S$ is open. Since $U$ is connected and $S\neq\varnothing$, we conclude $S=U$, so $|f|$ is constant on $U$, and by (b) (applied at any point) $f$ is constant on $U$.

\subsection*{Examples}
\begin{itemize}
	\item For $f(z)=z^{n}$, $\displaystyle f(0)=0=\frac{1}{2\pi}\int_{0}^{2\pi} (re^{it})^{n}dt$ (mean value).
	\item For $f(z)=e^{z}$ on any disk, $\max_{\overline{B}(a,r)}|f|$ occurs on $|z-a|=r$ (no interior maximum unless $f$ is constant).
\end{itemize}

\section*{Mean Value Property for Holomorphic and Harmonic Functions}

\bigskip\hrule\bigskip

\subsection*{Statement (Holomorphic Case)}

Let $f$ be holomorphic on a domain $U\subset\mathbb{C}$. If the closed disk $\overline{B}(a,r)\subset U$, then
\[
\boxed{\; f(a)\;=\;\frac{1}{2\pi}\,\int_{0}^{2\pi} f\!\big(a+re^{it}\big)\,dt \;}
\]
i.e.\ the value of $f$ at the center equals the average of its values on the surrounding circle.

\bigskip

\subsection*{One-Line Proof}

By Cauchy’s integral formula on the circle $|\zeta-a|=r$,
\[
f(a)=\frac{1}{2\pi i}\int_{|\zeta-a|=r}\frac{f(\zeta)}{\zeta-a}\,d\zeta.
\]
Parametrize $\zeta=a+re^{it}$ so that $d\zeta=ire^{it}\,dt$. Then
\[
f(a)=\frac{1}{2\pi i}\int_{0}^{2\pi}\frac{f(a+re^{it})}{re^{it}}\,(ire^{it})\,dt
=\frac{1}{2\pi}\int_{0}^{2\pi} f(a+re^{it})\,dt.
\]

\bigskip\hrule\bigskip

\subsection*{Harmonic Version}

If $u$ is harmonic on $U$ and $\overline{B}(a,r)\subset U$, then
\[
\boxed{\; u(a)\;=\;\frac{1}{2\pi}\,\int_{0}^{2\pi} u\!\big(a+re^{it}\big)\,dt \;}
\]
(For $f=u+iv$ holomorphic, both $u$ and $v$ are harmonic and satisfy this property.)

\bigskip\hrule\bigskip

\subsection*{Intuition and Consequences}

\medskip

\noindent\textbf{Rigidity / No interior spikes.}
The center value is a circular average; thus a modulus maximum cannot occur inside unless $f$ is constant (maximum modulus principle).

\medskip

\noindent\textbf{Cauchy estimates.}
From circular averages and Cauchy formulas for derivatives, one gets the standard bounds
\[
|f^{(n)}(a)| \le \frac{n!}{r^{n}}\,\max_{|\zeta-a|=r}|f(\zeta)|.
\]

\medskip

\noindent\textbf{Uniqueness.}
For harmonic functions, boundary data determines the interior via Poisson’s formula; the mean value property is a key equivalent.

\bigskip\hrule\bigskip

\subsection*{Examples}

\paragraph{1) Polynomials.}
For $f(z)=z^{n}$ and $a=0$,
\[
\frac{1}{2\pi}\int_{0}^{2\pi} (re^{it})^{n}\,dt
=\begin{cases}
	1,& n=0,\\
	0,& n\ge 1,
\end{cases}
\quad=\ f(0).
\]

\medskip

\paragraph{2) Exponential.}
Let $f(z)=e^{z}$. Then on $|z-a|=r$,
\[
f(a+re^{it})=e^{a}\,e^{re^{it}}
=e^{a}\sum_{k=0}^{\infty}\frac{r^{k}e^{ikt}}{k!}.
\]
Averaging over $t\in[0,2\pi]$ kills all $k\ge 1$, leaving
\[
\frac{1}{2\pi}\int_{0}^{2\pi} f(a+re^{it})\,dt
=e^{a}=f(a).
\]

\medskip

\paragraph{3) Harmonic example.}
If $u(x,y)=x$ (harmonic), then for any circle centered at $a=x_{0}+iy_{0}$,
\[
\frac{1}{2\pi}\int_{0}^{2\pi} u\!\big(a+re^{it}\big)\,dt
=\frac{1}{2\pi}\int_{0}^{2\pi} (x_{0}+r\cos t)\,dt
=x_{0}=u(a).
\]

\bigskip\hrule\bigskip

\subsection*{Equality Case and Maximum Modulus}

If $|f(a)|=\max_{\overline{B}(a,r)}|f|$, then
\[
|f(a)|
=\left|\frac{1}{2\pi}\int_{0}^{2\pi} f(a+re^{it})\,dt\right|
\le \frac{1}{2\pi}\int_{0}^{2\pi}|f(a+re^{it})|\,dt
\le |f(a)|.
\]
Equality forces $f(a+re^{it})$ to have \emph{constant argument} and constant modulus on the circle, hence $f$ is constant on that circle. Cauchy’s formula for derivatives then yields $f^{(n)}(a)=0$ for $n\ge 1$, so $f$ is constant near $a$; on a connected domain, $f$ is constant everywhere.

\section*{Harmonic vs.\ Holomorphic: What Implies What?}

\bigskip\hrule\bigskip

\subsection*{Holomorphic $\Rightarrow$ Harmonic}

Let $f=u+iv$ be holomorphic on $U\subset\mathbb{C}$. Then $u,v\in C^\infty(U)$ and satisfy the Cauchy--Riemann equations
\[
u_x=v_y,\qquad u_y=-\,v_x.
\]
Differentiating and adding,
\[
u_{xx}+u_{yy}=0,\qquad v_{xx}+v_{yy}=0,
\]
so $u$ and $v$ are \emph{harmonic}. Hence every holomorphic function has harmonic real and imaginary parts.

\bigskip\hrule\bigskip

\subsection*{Harmonic $\nRightarrow$ Holomorphic (in general)}

A real-valued harmonic function $u$ is not itself holomorphic (nonconstant real-valued holomorphic maps do not exist by the Open Mapping Theorem).

For a complex-valued $f=u+iv$ to be holomorphic one needs both:
\[
\text{(i) } \Delta u=\Delta v=0 \quad \text{and} \quad
\text{(ii) } u_x=v_y,\; u_y=-v_x.
\]
Harmonicity alone does not enforce the Cauchy--Riemann relations.

\bigskip\hrule\bigskip

\subsection*{Harmonic Conjugates and Topology}

Given harmonic $u$ on $U$, define the $1$-form
\[
\alpha:=-\,u_y\,dx+u_x\,dy.
\]
Then $d\alpha=(u_{xx}+u_{yy})\,dx\wedge dy=\Delta u\,dx\wedge dy=0$, so $\alpha$ is \emph{closed}.

\medskip

\noindent\textbf{Local existence.}
Closed $\Rightarrow$ exact locally: there exists $v$ (locally) with $dv=\alpha$, i.e.
\[
v_y=u_x,\qquad v_x=-u_y.
\]
Thus $f=u+iv$ is \emph{locally} holomorphic.

\medskip

\noindent\textbf{Global existence.}
If $U$ is \emph{simply connected}, closed $\Rightarrow$ exact globally, so there is a global harmonic conjugate $v$ and $f=u+iv$ is globally holomorphic (unique up to an additive constant in $v$).
If $U$ is not simply connected, a global $v$ exists iff all periods of $\alpha$ vanish:
\[
\oint_\gamma \alpha \;=\; 0 \quad \text{for every closed loop }\gamma\subset U.
\]

\bigskip\hrule\bigskip

\subsection*{Examples}

\paragraph{Holomorphic $\Rightarrow$ harmonic.}
$f(z)=e^z$; then $u=e^x\cos y$ and $v=e^x\sin y$ satisfy $\Delta u=\Delta v=0$.

\medskip

\paragraph{Harmonic but no global holomorphic primitive on $\mathbb{C}\setminus\{0\}$.}
$u(z)=\log|z|$ is harmonic on $\mathbb{C}\setminus\{0\}$.
Locally $u=\Re(\log z)$, but a global conjugate would be $\arg z$, which is multivalued; the periods are nonzero, so no global single-valued holomorphic $F$ with $\Re F=u$.

\medskip

\paragraph{Harmonic with a global conjugate.}
$u(x,y)=x^2-y^2$ is harmonic; $v(x,y)=2xy$ satisfies the CR relations, and $f=u+iv=(x+iy)^2=z^2$ is holomorphic.

\medskip

\paragraph{Harmonic components but not holomorphic.}
$f(z)=\overline{z}=x-iy$. Here $u=x$, $v=-y$ are harmonic, but $u_x=1\neq v_y=-1$; the CR equations fail, so $f$ is not holomorphic (it is anti-holomorphic).

\bigskip\hrule\bigskip

\subsection*{Bottom Line}

\begin{itemize}
	\item \textbf{Always:} holomorphic $\Rightarrow$ (real and imaginary parts) harmonic.
	\item \textbf{Locally:} harmonic $u$ is the real part of a holomorphic function.
	\item \textbf{Globally:} harmonic $u$ is the real part of a holomorphic function \emph{iff} the domain is simply connected (or, equivalently, the periods of $-u_y\,dx+u_x\,dy$ vanish).
	\item \textbf{Not sufficient:} harmonicity alone does not imply holomorphicity; the Cauchy--Riemann equations are the extra condition.
\end{itemize}

\section*{Product-of-distances lower bound on the unit circle}

\subsection*{Problem}
Let $w_1,\dots,w_n$ lie on the unit circle $\{|z|=1\}$. Show that there exists $z$ on the unit circle such that
\[
\prod_{k=1}^n |z-w_k| \;\ge\; 1.
\]

\bigskip\hrule\bigskip

\subsection*{Key observation: average of $\log|z-w|$ is zero on the circle}

Fix $w$ with $|w|=1$ and parametrize the circle by $z=e^{it}$, $t\in[0,2\pi]$.
Writing $w=e^{i\phi}$,
\[
|e^{it}-w|
= \bigl|e^{it}-e^{i\phi}\bigr|
= 2\,\bigl|\sin\frac{t-\phi}{2}\bigr|.
\]
Hence
\[
\frac1{2\pi}\int_0^{2\pi}\log|e^{it}-w|\,dt
= \frac1{2\pi}\int_0^{2\pi}\log\!\Bigl(2\bigl|\sin\frac{t-\phi}{2}\bigr|\Bigr)\,dt
= \frac1{2\pi}\int_0^{2\pi}\log\!\Bigl(2\bigl|\sin\frac{u}{2}\bigr|\Bigr)\,du
= 0,
\]
where the shift $u=t-\phi$ and the classical integral
$\displaystyle \int_0^{2\pi}\log\!\bigl(2\sin\frac{u}{2}\bigr)\,du=0$ were used.

\bigskip\hrule\bigskip

\subsection*{Proof of the claim}

Define the polynomial
\[
P(z)=\prod_{k=1}^n (z-w_k).
\]
For $z=e^{it}$ on the unit circle,
\[
\log |P(e^{it})|=\sum_{k=1}^n \log|e^{it}-w_k|.
\]
Averaging over $t\in[0,2\pi]$ and using the previous calculation term-by-term,
\[
\frac1{2\pi}\int_0^{2\pi} \log |P(e^{it})|\,dt
=\sum_{k=1}^n \frac1{2\pi}\int_0^{2\pi} \log|e^{it}-w_k|\,dt
= 0.
\]
If $\log |P(e^{it})|<0$ for all $t$, the average would be negative. Therefore there exists $t_0$ with
\[
\log|P(e^{it_0})|\;\ge\;0
\quad\Longleftrightarrow\quad
|P(e^{it_0})|=\prod_{k=1}^n |e^{it_0}-w_k|\;\ge\;1.
\]
Setting $z=e^{it_0}$ completes the proof.

\bigskip\hrule\bigskip

\subsection*{Remarks and variants}

\begin{itemize}
	\item Equivalently, by Jensen’s formula applied to $P$ (whose zeros lie on $|z|=1$), the mean of $\log|P(e^{it})|$ over the unit circle equals $\log|P(0)|=0$, since $|P(0)|=\prod |w_k|=1$.
	\item For $n=1$, choosing $z=-w_1$ gives $|z-w_1|=2\ge1$. For general $n$, the statement guarantees at least one $z$ with product $\ge1$, though the maximizing $z$ depends on the configuration of the $w_k$.
\end{itemize}

\section*{Entire function with nonnegative real part is constant}

\subsection*{Problem}
Let $f$ be holomorphic on $\mathbb{C}$ and suppose $\operatorname{Re} f(z)\ge 0$ for all $z\in\mathbb{C}$.
Show that $f$ is constant. \hfill\emph{(Hint: consider $\exp(-f(z))$.)}

\bigskip\hrule\bigskip

\subsection*{Background}
\begin{itemize}
	\item \textbf{Liouville's Theorem.} Every bounded entire function is constant.
	\item If $\operatorname{Re} f\ge 0$ and $g(z):=e^{-f(z)}$, then
	\[
	|g(z)|=|e^{-f(z)}|=e^{-\operatorname{Re} f(z)}\le 1,
	\]
	so $g$ is entire and bounded.
\end{itemize}

\bigskip\hrule\bigskip

\subsection*{Solution}
Define $g(z):=e^{-f(z)}$. Then $g$ is entire and satisfies $|g(z)|\le 1$ for all $z\in\mathbb{C}$.
By Liouville's theorem, $g$ is constant: $g(z)\equiv c$ with $|c|\le 1$.

\medskip

Hence $e^{-f(z)}\equiv c$. Choosing any logarithm of the constant $c$,
\[
-f(z)\equiv \log c \qquad\Longrightarrow\qquad f(z)\equiv -\log c,
\]
so $f$ is constant on $\mathbb{C}$.

\bigskip\hrule\bigskip

\subsection*{Remark (optional)}
If $f$ were nonconstant, then $g=e^{-f}$ would be nonconstant and thus an open map.
But $g(\mathbb{C})\subset\{w:|w|\le 1\}$ is not open, a contradiction.
Therefore $f$ must be constant.

\section*{Reverse assumption: \(\operatorname{Re} f\le 0\) implies \(f\) is constant}

\subsection*{Claim}
If \(f\) is entire and \(\operatorname{Re} f(z)\le 0\) for all \(z\in\mathbb C\), then \(f\) is constant.

\bigskip\hrule\bigskip

\subsection*{Proof}
Define \(h(z):=e^{\,f(z)}\). Then \(h\) is entire and
\[
|h(z)|=|e^{\,f(z)}|=e^{\operatorname{Re} f(z)}\le 1\qquad(\forall z\in\mathbb C).
\]
Hence \(h\) is a bounded entire function. By Liouville’s theorem, \(h\) is constant:
\(h(z)\equiv c\) with \(|c|\le 1\).
Taking any logarithm of the constant \(c\), we get \(f(z)\equiv \log c\), a constant function.

\bigskip\hrule\bigskip

\subsection*{Remark (generalization)}
If \(\operatorname{Re} f\) is bounded \emph{above} or \emph{below} on \(\mathbb C\), then \(f\) is constant.
Indeed, use \(e^{\,f}\) when \(\operatorname{Re} f\le M\) or \(e^{-f}\) when \(\operatorname{Re} f\ge m\) to obtain a bounded entire function and apply Liouville.

\section*{Half–plane criteria for constancy of entire functions}

\bigskip\hrule\bigskip

\subsection*{General Lemma (rotation–shift half–plane test)}

Let $f$ be entire. Suppose there exist $\theta\in\mathbb{R}$ and $c\in\mathbb{R}$ such that
\[
\Re\!\big(e^{-i\theta} f(z)\big)\ \ge\ c \qquad (\forall z\in\mathbb{C}).
\]
Then $f$ is constant.

\medskip

\emph{Proof.}
Define
\[
g(z):=\exp\!\big(-\,e^{-i\theta}\,(f(z)-c)\big).
\]
Then
\[
|g(z)|= \exp\!\Big(-\,\Re\!\big(e^{-i\theta}(f(z)-c)\big)\Big)\ \le\ 1,
\]
so $g$ is bounded and entire. By Liouville’s theorem, $g$ is constant; hence $f$ is constant. \qed

\bigskip\hrule\bigskip

\subsection*{Immediate corollaries (choose $\theta$ appropriately)}

\begin{itemize}
	\item \textbf{Real part bounded below.} If $\Re f(z)\ge a$ for all $z$, take $\theta=0$:
	\[
	|e^{-(f-a)}|=e^{-(\Re f-a)}\le 1 \ \Rightarrow \ e^{-(f-a)}\ \text{constant} \ \Rightarrow\ f\ \text{constant}.
	\]
	
	\item \textbf{Real part bounded above.} If $\Re f(z)\le a$ for all $z$, use $e^{\,f-a}$:
	\[
	|e^{\,f-a}|=e^{\Re f-a}\le 1 \ \Rightarrow \ e^{\,f-a}\ \text{constant} \ \Rightarrow\ f\ \text{constant}.
	\]
	
	\item \textbf{Imaginary part bounded below.} If $\Im f(z)\ge b$ for all $z$, take $\theta=\tfrac{\pi}{2}$:
	\[
	|e^{\,i f}|=e^{-\Im f}\le e^{-b} \ \Rightarrow \ e^{\,i f}\ \text{constant} \ \Rightarrow\ f\ \text{constant}.
	\]
	
	\item \textbf{Imaginary part bounded above.} If $\Im f(z)\le b$ for all $z$, take $\theta=-\tfrac{\pi}{2}$:
	\[
	|e^{-\,i f}|=e^{\Im f}\le e^{b} \ \Rightarrow \ e^{-\,i f}\ \text{constant} \ \Rightarrow\ f\ \text{constant}.
	\]
	
	\item \textbf{Tilted half–planes.} If $\Re(e^{-i\theta} f(z))\ge a$ (or $\le a$) for some fixed $\theta$, apply the lemma directly. Thus any entire map with image contained in a rotated/translated half–plane is constant.
\end{itemize}

\bigskip\hrule\bigskip

\subsection*{Strip versions and harmonic viewpoint}

If $a\le \Re(e^{-i\theta} f)\le b$ on $\mathbb{C}$, then $\Re(e^{-i\theta} f)$ is a bounded harmonic function on $\mathbb{C}$, hence constant. The Cauchy--Riemann equations then force the imaginary part to be constant as well, so $f$ is constant.

\medskip

Similarly, if $a\le \Im(e^{-i\theta} f)\le b$, apply the same argument to the harmonic function $\Im(e^{-i\theta} f)$.

\bigskip\hrule\bigskip

\subsection*{Remarks}

\begin{itemize}
	\item The choices $e^{\pm f}$ and $e^{\pm i f}$ are tailored so that their moduli are $e^{\pm \Re f}$ and $e^{\mp \Im f}$, respectively.
	\item This gives a unified proof that an entire function with image in \emph{any} (open or closed) half–plane must be constant.
	\item Mapping a half–plane to the unit disk with a Möbius map (e.g.\ Cayley transform) yields the same conclusion: the composition is bounded entire, hence constant.
\end{itemize}
	
	
\section*{Growth and domination for entire functions}

\subsection*{Problem}
Let $f$ be an entire function.

\begin{enumerate}
	\item[(i)] Prove that $f$ is a polynomial of degree at most $k$ if and only if there exist real constants $M,R>0$ and an integer $k$ such that
	\[
	|f(z)|\le M\,|z|^{k}\qquad\text{for }|z|>R.
	\]
	\item[(ii)] Let $g$ be an entire function such that $|f(z)|\le |g(z)|$ for all $z\in\mathbb C$.
	Show that there exists $c\in\mathbb C$ such that $f(z)=c\,g(z)$ for all $z\in\mathbb C$.
\end{enumerate}

\bigskip\hrule\bigskip

\subsection*{Background}
\paragraph{Cauchy estimates.}
If $f$ is holomorphic on $|z|\le \rho$, then
\[
f^{(n)}(0)=\frac{n!}{2\pi i}\int_{|z|=\rho}\frac{f(z)}{z^{n+1}}\,dz,
\qquad
|f^{(n)}(0)|\le \frac{n!}{\rho^{n}}\max_{|z|=\rho}|f(z)|.
\]

\paragraph{Liouville and removable singularities.}
A bounded entire function is constant. If $h$ is holomorphic on a punctured neighborhood of $a$ and bounded near $a$, then $h$ extends holomorphically to $a$.

\bigskip\hrule\bigskip

\subsection*{Solution to (i)}

\paragraph{``Only if.''}
If $f(z)=a_k z^k+\cdots +a_0$ has degree $\le k$, then for $|z|>1$,
\[
|f(z)|\le \big(|a_k|+\cdots+|a_0|\big)\,|z|^{k}.
\]
Taking $R\ge 1$ and $M=\sum_{j=0}^{k}|a_j|$ gives the claim.

\medskip

\paragraph{``If.''}
Assume there exist $M,R>0$ and integer $k$ with $|f(z)|\le M|z|^{k}$ for $|z|>R$.
Fix $\rho>R$. By Cauchy’s estimate,
\[
|f^{(n)}(0)|\le \frac{n!}{\rho^{n}}\max_{|z|=\rho}|f(z)|
\le \frac{n!}{\rho^{n}}\,M\,\rho^{k}
= M\,n!\,\rho^{\,k-n}.
\]
If $n>k$, then letting $\rho\to\infty$ yields $f^{(n)}(0)=0$. Hence all derivatives of order $>k$ vanish, so $f$ is a polynomial of degree at most $k$.

\bigskip\hrule\bigskip

\subsection*{Solution to (ii)}

If $g\equiv 0$, then $|f|\le |g|$ forces $f\equiv 0$, and $f=cg$ is trivial. Assume $g\not\equiv 0$.

Define $h(z)=\dfrac{f(z)}{g(z)}$ where $g(z)\ne 0$. We show $h$ extends to an entire, bounded function.

Let $a$ be a zero of $g$ of order $m\ge 1$, so $g(z)=(z-a)^{m}g_m(z)$ with $g_m$ holomorphic and $g_m(a)\ne 0$. From $|f|\le |g|$ we have, for $z$ near $a$,
\[
|f(z)|\le C\,|z-a|^{m}
\]
for some $C>0$, hence $f(z)=(z-a)^{m}f_m(z)$ with $f_m$ holomorphic (and bounded) near $a$. Then on the punctured neighborhood,
\[
h(z)=\frac{f(z)}{g(z)}=\frac{f_m(z)}{g_m(z)}.
\]
Since $g_m(a)\neq 0$ and $f_m$ is bounded near $a$, $h$ is bounded near $a$; by the removable singularities theorem, $h$ extends holomorphically to $a$. Doing this at all zeros of $g$, we obtain an entire extension of $h$. Moreover, on the set where $g\neq 0$ we have $|h(z)|=\dfrac{|f(z)|}{|g(z)|}\le 1$, therefore by continuity $|h|\le 1$ everywhere.

Thus $h$ is bounded and entire, so by Liouville, $h\equiv c$ for some $c\in\mathbb C$ (with $|c|\le 1$). Consequently $f(z)=c\,g(z)$ for all $z\in\mathbb C$.

\bigskip\hrule\bigskip

\subsection*{Examples}

\begin{itemize}
	\item If $f(z)=3z^{5}-2z+7$ then $|f(z)|\le (3+2+7)\,|z|^{5}$ for $|z|>1$; the growth condition holds with $k=5$.
	\item If $|f(z)|\le |\sin z|$ on $\mathbb{C}$, then by (ii) there is $c\in\mathbb{C}$ with $f(z)=c\sin z$ for all $z$.
\end{itemize}
	


	
	\section*{Poisson and Schwarz Integral Formulas on the Unit Disk}
	Let $f=u+iv$ be holomorphic in a domain containing the closed unit disk $\overline{\mathbb D}$.
	
	\begin{enumerate}
		\item[(a)] Show that
		\[
		u(0)=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,dt.
		\]
		\item[(b)] Assume that for every $z_0\in \mathbb D$ the map
		\[
		h_{z_0}(z)=\frac{z+z_0}{1+\overline{z_0}\,z}
		\]
		is a bijection of $\mathbb D$ onto itself. Using this map, show that for $z\in\mathbb D$,
		\[
		u(z)=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,
		\frac{1-|z|^{2}}{|e^{it}-z|^{2}}\,dt .
		\]
		\item[(c)] Using the fact that
		\[
		\frac{|w|^{2}-|z|^{2}}{|z-w|^{2}}
		=\Re\!\left(\frac{w+z}{w-z}\right),
		\]
		show that
		\[
		f(z)=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,
		\frac{e^{it}+z}{e^{it}-z}\,dt\;+\; i\,v(0).
		\]
	\end{enumerate}
	
	\section*{Background (first theory)}
	\paragraph{Mean value property.}
	If $u$ is harmonic on a neighborhood of $\overline{\mathbb D}$, then for $0<r\le 1$,
	\[
	u(0)=\frac{1}{2\pi}\int_{0}^{2\pi}u(re^{it})\,dt.
	\]
	
	\paragraph{Poisson kernel on $\mathbb D$.}
	For $z=re^{i\phi}\in\mathbb D$ and boundary point $e^{it}$,
	\[
	P_r(\phi-t)=\frac{1-r^{2}}{|e^{it}-re^{i\phi}|^{2}}
	=\frac{1-|z|^{2}}{|e^{it}-z|^{2}}.
	\]
	If $g\in C(\partial\mathbb D)$ and $U$ is harmonic in $\mathbb D$ with nontangential boundary values $g$, then
	\[
	U(z)=\frac{1}{2\pi}\int_{0}^{2\pi} g(e^{it})\,P_r(\phi-t)\,dt .
	\]
	
	\paragraph{Disk automorphisms.}
	For $z_0\in\mathbb D$,
	\[
	h_{z_0}(z)=\frac{z+z_0}{1+\overline{z_0}z},\qquad
	h_{z_0}(0)=z_0,\qquad h_{z_0}^{-1}=h_{-z_0},
	\]
	is a Möbius automorphism of $\mathbb D$ carrying $\partial\mathbb D$ onto itself and preserving harmonic measure.
	
	\paragraph{Schwarz/Herglotz kernel.}
	For $z\in\mathbb D$,
	\[
	K(z,e^{it})=\frac{e^{it}+z}{e^{it}-z},\qquad
	\Re\,K(z,e^{it})=\frac{1-|z|^{2}}{|e^{it}-z|^{2}}=P_r(\phi-t).
	\]
	
	\section*{Solutions}
	
	\subsection*{(a) Mean value at the center}
	Since $f$ is holomorphic near $\overline{\mathbb D}$, its real part $u$ is harmonic there. The mean value property at the origin gives
	\[
	u(0)=\frac{1}{2\pi}\int_{0}^{2\pi}u(e^{it})\,dt.
	\]
	
	\subsection*{(b) Poisson integral via $h_{z_0}$}
	Fix $z\in\mathbb D$ and put $z_0=z$. Consider the harmonic function
	\[
	U(\zeta):=u\big(h_{z}(\zeta)\big),\qquad \zeta\in\mathbb D.
	\]
	By (a), applied at the origin for $U$,
	\[
	U(0)=\frac{1}{2\pi}\int_{0}^{2\pi} U(e^{it})\,dt
	=\frac{1}{2\pi}\int_{0}^{2\pi} u\!\left(h_{z}(e^{it})\right)\,dt .
	\]
	But $U(0)=u(h_z(0))=u(z)$. Using the boundary change of variables $e^{i\theta}=h_{z}(e^{it})$ and the known identity
	\[
	\frac{dt}{d\theta}
	=\frac{1-|z|^{2}}{|e^{i\theta}-z|^{2}},
	\]
	we obtain
	\[
	u(z)=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{i\theta})\,\frac{1-|z|^{2}}{|e^{i\theta}-z|^{2}}\,d\theta,
	\]
	which is the Poisson integral formula.
	
	\subsection*{(c) Schwarz integral for $f$}
	Define
	\[
	F(z):=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,\frac{e^{it}+z}{e^{it}-z}\,dt.
	\]
	For fixed $z\in\mathbb D$ the integrand is holomorphic in $z$ and integrable in $t$, so $F$ is holomorphic on $\mathbb D$. Taking real parts and using
	\[
	\Re\!\left(\frac{e^{it}+z}{e^{it}-z}\right)=\frac{1-|z|^{2}}{|e^{it}-z|^{2}},
	\]
	we get
	\[
	\Re F(z)=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,\frac{1-|z|^{2}}{|e^{it}-z|^{2}}\,dt
	=u(z)
	\]
	by part (b). Hence $F-f$ is holomorphic with \(\Re(F-f)\equiv0\) on $\mathbb D$, so $F-f\equiv iC$ for some real constant $C$. Evaluating at $z=0$,
	\[
	F(0)=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,dt = u(0),
	\]
	so
	\[
	u(0)+iC=f(0)=u(0)+iv(0)\quad\Rightarrow\quad C=v(0).
	\]
	Therefore
	\[
	f(z)=F(z)+i\,v(0)
	=\frac{1}{2\pi}\int_{0}^{2\pi} u(e^{it})\,\frac{e^{it}+z}{e^{it}-z}\,dt + i\,v(0).
	\]

	\section*{Normalization and quick intuition}
	
	Normalize once:
	\[
	K_t(z)=C\!\big(e^{-it}z\big), 
	\qquad
	C(\zeta)=\frac{1+\zeta}{1-\zeta}\quad\text{(Cayley transform).}
	\]
	So every geometric fact for $K_t$ is a rotation of the corresponding fact for $C$ (which “focuses’’ at the boundary point $1$).
	
	\section*{Natural domain and image}
	
	\[
	\boxed{\ \mathbb{D}=\{\,z:\,|z|<1\,\}\ },
	\qquad
	\boxed{\ K_t(\mathbb{D})=\{\,w:\,\operatorname{Re} w>0\,\}\ \text{(right half-plane)}\ }.
	\]
	Reason: for $|z|<1$,
	\[
	\operatorname{Re} K_t(z)=\frac{1-|z|^2}{\,|e^{it}-z|^2\,}>0.
	\]
	
	\medskip
	
	\textbf{Boundary behavior.}
	\[
	K_t(0)=1,\qquad
	\lim_{z\to e^{it}}K_t(z)=\infty,\qquad
	K_t\!\big(\partial\mathbb{D}\setminus\{e^{it}\}\big)=i\mathbb{R}.
	\]
	Thus $K_t:\mathbb{D}\to\{\operatorname{Re} w>0\}$ is a conformal bijection, sending the boundary point $e^{it}$ to $\infty$ and preserving orientation.
	
	\section*{Inverse and derivative (for checks)}
	
	\[
	K_t^{-1}(w)=e^{it}\,\frac{w-1}{w+1},
	\qquad
	K_t'(z)=\frac{2e^{it}}{(e^{it}-z)^2},
	\qquad
	\bigl|K_t'(z)\bigr|=\frac{2}{|e^{it}-z|^2}.
	\]
	
	\section*{Vertical lines in the image and their preimages}
	
	Fix $c>0$. The vertical line 
	\[
	L_c=\{\,w:\ \operatorname{Re} w=c\,\}
	\]
	pulls back to the level set
	\[
	\operatorname{Re} K_t(z)=\frac{1-|z|^2}{\,|e^{it}-z|^2\,}=c.
	\]
	Rotate by $u=e^{-it}z$ (so $e^{it}\mapsto 1$). For the Cayley map $C$ this is
	\[
	\operatorname{Re} C(u)=\frac{1-|u|^2}{\,|1-u|^2\,}=c
	\iff
	|1-u|^2=c^{-1}(1-|u|^2).
	\]
	Writing $u=x+iy$ gives the circle equation
	\[
	\boxed{\ (x-\tfrac{1}{1+c})^2+y^2=\Bigl(\tfrac{1}{1+c}\Bigr)^2\ }.
	\]
	Hence in the $u$–plane we get a circle of radius $1/(1+c)$ centered at $1/(1+c)$ on the real axis, tangent to $\partial\mathbb{D}$ at $u=1$. Rotating back:
	
	\[
	\boxed{\ \{\operatorname{Re} K_t(z)=c\}\ \text{is a circle in }\mathbb{D}\text{ tangent to }\partial\mathbb{D}\text{ at }\tau=e^{it}. \ }
	\]
	
	As $c\downarrow 0$ the circle expands to $\partial\mathbb{D}$ (except at $\tau$); as $c\uparrow\infty$ it shrinks to $\tau$.
	
	\medskip
	
	\textbf{Vertical bands.}
	The strip $a<\operatorname{Re} w<b$ corresponds to the lens-like region in $\mathbb{D}$
	between the two tangent circles $\operatorname{Re} K_t(z)=a$ and $\operatorname{Re} K_t(z)=b$ (both tangent at $\tau$).
	
	\section*{Horizontal lines in the image and their preimages}
	
	Fix $b\in\mathbb{R}$. The horizontal line
	\[
	H_b=\{\,w:\ \operatorname{Im} w=b\,\}
	\]
	pulls back to $\operatorname{Im} K_t(z)=b$. For the Cayley transform,
	\[
	\operatorname{Im} C(u)=\frac{2\,\operatorname{Im} u}{\,|1-u|^2\,}=b
	\iff
	(1-\operatorname{Re} u)^2+\Bigl(\operatorname{Im} u-\frac{1}{b}\Bigr)^2=\Bigl(\frac{1}{b}\Bigr)^2.
	\]
	Thus for $b\neq 0$,
	\[
	\boxed{\ \{\operatorname{Im} C(u)=b\}\ \text{is a circle tangent to }\partial\mathbb{D}\text{ at }u=1,\ 
		\text{center }(1,1/b),\ \text{radius }1/|b|. }
	\]
	Rotating back by $e^{it}$ shows that $\{\operatorname{Im} K_t(z)=b\}$ is also a circle tangent to $\partial\mathbb{D}$ at $\tau=e^{it}$. The two families
	\[
	\{\operatorname{Re} K_t=c\}_{c>0},\qquad \{\operatorname{Im} K_t=b\}_{b\in\mathbb{R}}
	\]
	are orthogonal (conformality).
	
	\medskip
	
	\textbf{Horizontal bands.}
	The strip $\beta<\operatorname{Im} w<\delta$ pulls back to the region between
	the circles $\operatorname{Im} K_t(z)=\beta$ and $\operatorname{Im} K_t(z)=\delta$ (both tangent at $\tau$).
	
	\section*{Canonical correspondences}
	
	\begin{itemize}
		\item $z=0 \mapsto w=1$.
		\item $z=\tau=e^{it}\mapsto w=\infty$ (simple pole).
		\item $\partial\mathbb{D}\setminus\{\tau\}\mapsto i\mathbb{R}$.
		\item Vertical line $\operatorname{Re} w=c$ $\Longleftrightarrow$ circle $\operatorname{Re} K_t(z)=c$ tangent at $\tau$.
		\item Horizontal line $\operatorname{Im} w=b$ $\Longleftrightarrow$ circle $\operatorname{Im} K_t(z)=b$ tangent at $\tau$.
		\item Vertical/horizontal \emph{bands} $\Longleftrightarrow$ regions between pairs of tangent circles.
	\end{itemize}
	
	\section*{Harmonic meaning}
	
	\[
	\boxed{\ \operatorname{Re} K_t(z)=\frac{1-|z|^2}{\,|\,e^{it}-z\,|^2}=P(z,e^{it})\ }
	\]
	is exactly the Poisson kernel of the unit disk with pole at the boundary point $e^{it}$.
	Hence the circles $\operatorname{Re} K_t(z)=c$ are level sets of harmonic measure;
	$\operatorname{Im} K_t$ is (up to an additive constant in $t$) the harmonic conjugate, whose level sets form the orthogonal tangent pencil.	
	
	
\section*{Setup}
Let $\tau=e^{it}$ and note
\[
K_t(z)=C\!\big(e^{-it}z\big), \qquad C(\zeta)=\frac{1+\zeta}{1-\zeta},\qquad
z=\frac{w-1}{w+1}.
\]
“Vertical’’ and “horizontal’’ in the domain are taken \emph{relative to $\tau$}:
\[
\text{vertical: }\ \Re\!\big(e^{-it}z\big)=a,
\qquad
\text{horizontal: }\ \Im\!\big(e^{-it}z\big)=b .
\]

\section*{Images of vertical domain lines and bands}
Impose $\Re z=a$ with $z=(w-1)/(w+1)$. Writing $w=u+iv$,
\[
\Re\!\Big(\frac{w-1}{w+1}\Big)=a
\iff
\frac{|w|^{2}-1}{\,|w+1|^{2}\,}=a
\iff
\big(u-\tfrac{a}{1-a}\big)^{2}+v^{2}=\Big(\tfrac{1}{1-a}\Big)^{2}.
\]
Thus
\[
\boxed{\ \Re\!\big(e^{-it}z\big)=a\ \Longrightarrow\ 
	K_t(z)\ \text{lies on the circle}\ 
	\big(u-\tfrac{a}{1-a}\big)^{2}+v^{2}=\big(\tfrac{1}{1-a}\big)^{2}. }
\]
In particular:
\begin{itemize}
	\item $a=0$ maps to the unit circle $\ |w|=1$.
	\item $a\to 1^{-}$ produces circles whose limits are vertical lines in the right half-plane.
	\item The band $a<\Re(e^{-it}z)<b$ maps to the region between the two circles
	with centers $\tfrac{a}{1-a}$ and $\tfrac{b}{1-b}$ on the real axis and radii $1/|1-a|$, $1/|1-b|$.
\end{itemize}

\section*{Images of horizontal domain lines and bands}
Impose $\Im z=b$ with $z=(w-1)/(w+1)$. Then
\[
\Im\!\Big(\frac{w-1}{w+1}\Big)=b
\iff
\frac{2\,\Im w}{\,|w+1|^{2}\,}=b
\iff
(u+1)^{2}+\Big(v-\tfrac{1}{b}\Big)^{2}=\Big(\tfrac{1}{b}\Big)^{2}\quad (b\neq0).
\]
Therefore
\[
\boxed{\ \Im\!\big(e^{-it}z\big)=b\ \Longrightarrow\ 
	K_t(z)\ \text{lies on the circle}\ 
	(u+1)^{2}+\big(v-\tfrac{1}{b}\big)^{2}=\big(\tfrac{1}{b}\big)^{2}. }
\]
Special cases:
\begin{itemize}
	\item $b=0$ maps to the positive real axis $\{w>0\}$.
	\item For $b\neq0$ the circle is tangent to the real axis at $w=-1$, with center $(-1,1/b)$ and radius $1/|b|$.
	\item The band $\beta<\Im(e^{-it}z)<\delta$ maps to the region between the two tangent circles corresponding to $b=\beta$ and $b=\delta$.
\end{itemize}

\section*{Orthogonality and Möbius geometry}
The families
\[
\{\Re(e^{-it}z)=a\}_{a\in\mathbb{R}}
\quad\text{and}\quad
\{\Im(e^{-it}z)=b\}_{b\in\mathbb{R}}
\]
are orthogonal in the $z$-plane, and $K_t$ (being holomorphic/Möbius where defined) preserves angles. Hence their images are two orthogonal pencils of circles/lines in the $w$-plane.

\section*{Band dictionary (domain $\leftrightarrow$ image)}
\[
\begin{array}{ccl}
	\text{Domain band} & \Longleftrightarrow & \text{Image region}\\[0.25em]
	a<\Re(e^{-it}z)<b & \Longleftrightarrow &
	\text{between two circles centered on the real axis}\\[0.25em]
	\beta<\Im(e^{-it}z)<\delta & \Longleftrightarrow &
	\text{between two circles tangent at } w=-1
\end{array}
\]

\paragraph{Poisson integral on the unit disk.}
If $U$ is harmonic in $\mathbb D$ and has (nontangential) boundary values $g\in C(\partial\mathbb D)$, then for $z=re^{i\phi}\in\mathbb D$,
\[
U(re^{i\phi})
=\frac{1}{2\pi}\int_{0}^{2\pi} g(e^{it})\,P_r(\phi-t)\,dt,
\qquad
P_r(\theta)=\frac{1-r^2}{1-2r\cos\theta+r^2}
=\frac{1-r^2}{|e^{i\theta}-r|^2}.
\]
Here $P_r(\theta)\ge 0$ and $\frac{1}{2\pi}\int_0^{2\pi}P_r(\theta)\,d\theta=1$, so the value at $re^{i\phi}$ is a weighted average of the boundary data on $\partial\mathbb D$. Moreover,
$P_r(\phi-t)=\Re\!\left(\frac{e^{it}+re^{i\phi}}{e^{it}-re^{i\phi}}\right)$ and
$U(0)=\frac{1}{2\pi}\int_0^{2\pi}g(e^{it})\,dt$.
