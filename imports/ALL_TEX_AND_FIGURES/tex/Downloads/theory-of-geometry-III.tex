\subsection*{Key definitions}

\section{Curvature, Gauss map and fundamental forms}

In this section we collect the basic definitions and results about
curvature of plane curves, the Gauss map and the first and second
fundamental forms of a surface in $\mathbb{R}^3$.  We also introduce
ruled, developable and (strictly) convex surfaces and state the
fundamental theorem of surface theory.

\subsection{Curvature of plane curves}

Let $\gamma:(a,b)\to\mathbb{R}^2$ be a smooth regular plane curve.

\begin{definition}[Arc--length parametrisation]
	We say that $\gamma$ is \emph{parametrised by arc--length} if
	\[
	\|\gamma'(s)\|=1\qquad\text{for all }s\in(a,b).
	\]
	In that case $\gamma'(s)$ is a unit tangent vector along the curve.
\end{definition}

For a unit speed curve, differentiating
$\langle\gamma'(s),\gamma'(s)\rangle=1$ gives
$\langle\gamma'(s),\gamma''(s)\rangle=0$, so $\gamma''(s)$ is always
orthogonal to $\gamma'(s)$ and therefore normal to the curve.

\begin{definition}[Curvature of a plane curve]
	Let $\gamma:(a,b)\to\mathbb{R}^2$ be a plane curve parametrised by
	arc--length.  A \emph{unit normal field} $n_\gamma(s)$ along $\gamma$
	is a smooth map $(a,b)\to\mathbb{R}^2$ such that
	\[
	\|n_\gamma(s)\|=1,\qquad
	\langle\gamma'(s),n_\gamma(s)\rangle=0
	\]
	for all $s$.  We choose $n_\gamma(s)$ so that
	$(\gamma'(s),n_\gamma(s))$ is a positively oriented basis of
	$\mathbb{R}^2$ (the ``turn left'' normal).
	
	The \emph{curvature} of $\gamma$ is the function
	$\kappa:(a,b)\to\mathbb{R}$ such that
	\[
	\gamma''(s) = \kappa(s)\,n_\gamma(s).
	\]
	The quantity $\rho(s):=1/|\kappa(s)|$ is called the \emph{radius of
		curvature}.
\end{definition}

Equivalently, if we write the unit tangent as
\[
\gamma'(s) = (\cos\theta(s),\sin\theta(s)),
\]
where $\theta(s)$ is the angle between $\gamma'(s)$ and the $x$--axis,
then
\[
\gamma''(s) = \theta'(s)\,(-\sin\theta(s),\cos\theta(s))
= \theta'(s)\,n_\gamma(s),
\]
so
\[
\kappa(s) = \theta'(s),
\]
i.e.\ curvature is the rate of change of the tangent angle with respect
to arc--length.

\begin{example}
	\leavevmode
	\begin{enumerate}
		\item Straight line: $\gamma(s)=(s,0)$ is unit speed, with
		$\gamma''(s)=0$, so $\kappa(s)\equiv0$.
		
		\item Circle of radius $R$, traversed counterclockwise:
		\[
		\gamma(s)=(R\cos(s/R),R\sin(s/R)).
		\]
		Then $\gamma'(s)$ has unit length and
		$\gamma''(s)=\frac{1}{R}n_\gamma(s)$, so $\kappa(s)\equiv1/R$.
		
		\item Parabola $y=x^2$, with parametrisation $\tilde\gamma(t)=(t,t^2)$.
		Using the general formula
		\[
		\kappa(t)=\frac{x'(t)y''(t)-y'(t)x''(t)}
		{\bigl((x'(t))^2+(y'(t))^2\bigr)^{3/2}},
		\]
		we obtain
		\[
		\kappa(t)=\frac{2}{(1+4t^2)^{3/2}}>0,
		\]
		largest at $t=0$ and decaying as $|t|\to\infty$.
	\end{enumerate}
\end{example}

\subsection{Surfaces, the Gauss map and the second fundamental form}

Let $\Sigma\subset\mathbb{R}^3$ be an oriented smooth surface, with
unit normal field $n:\Sigma\to S^2$.

\begin{definition}[Gauss map]
	The \emph{Gauss map} of $\Sigma$ is the map
	\[
	N:\Sigma\to S^2,\qquad N(p)=n(p),
	\]
	assigning to each point its unit normal vector (chosen consistently with
	the orientation).
\end{definition}

\begin{definition}[Shape operator and second fundamental form]
	The \emph{shape operator} at $p\in\Sigma$ is the linear map
	$S_p:T_p\Sigma\to T_p\Sigma$ defined by
	\[
	S_p(w) = - (dN)_p(w).
	\]
	The \emph{second fundamental form} at $p$ is the symmetric bilinear form
	\[
	\II_p(u,w) = \langle S_p(u),w\rangle
	= -\langle(dN)_p(u),w\rangle,
	\qquad u,w\in T_p\Sigma.
	\]
\end{definition}

In local coordinates $\sigma(u,v)$, with first fundamental form
coefficients
\[
E=\langle\sigma_u,\sigma_u\rangle,\quad
F=\langle\sigma_u,\sigma_v\rangle,\quad
G=\langle\sigma_v,\sigma_v\rangle,
\]
and second fundamental form coefficients
\[
L=\langle\sigma_{uu},n\rangle,\quad
M=\langle\sigma_{uv},n\rangle,\quad
N=\langle\sigma_{vv},n\rangle,
\]
one has
\[
K = \det S_p
= \frac{LN-M^2}{EG-F^2},
\]
the Gauss curvature at $p$.

\begin{definition}[Normal curvature]
	Let $p\in\Sigma$ and let $v\in T_p\Sigma$ be a unit tangent vector.
	Consider the plane $P = p + \Span(v,n(p))$ and the plane curve
	$\gamma_v$ given by the intersection of $\Sigma$ with $P$, parametrised
	by arc--length and passing through $p$ with $\gamma_v'(0)=v$.
	The \emph{normal curvature} of $\Sigma$ at $p$ in the direction $v$ is
	the curvature $\kappa_v(0)$ of the plane curve $\gamma_v$ at $p$.
\end{definition}

\begin{proposition}[Normal curvature and second fundamental form]
	With notation as above, the normal curvature in the direction $v$ is
	\[
	\kappa_v(0)=\II_p(v,v).
	\]
\end{proposition}

\begin{proof}[Idea of proof]
	Along $\gamma_v$ we have $\langle\gamma_v'(s),n(\gamma_v(s))\rangle=0$.
	Differentiating at $s=0$ gives
	\[
	\langle\gamma_v''(0),n(p)\rangle
	+ \langle\gamma_v'(0),(dn)_p(\gamma_v'(0))\rangle = 0.
	\]
	Using $\gamma_v'(0)=v$ and the definition of $\II_p$ we obtain
	\[
	\langle\gamma_v''(0),n(p)\rangle = \II_p(v,v).
	\]
	As a plane curve in the oriented plane $P$, $\gamma_v$ has curvature
	$\kappa_v(0)$ determined by $\gamma_v''(0)=\kappa_v(0)n(p)$.  Taking the
	inner product with $n(p)$ yields $\kappa_v(0)=\langle\gamma_v''(0),n(p)\rangle
	= \II_p(v,v)$.
\end{proof}

\subsection{Ruled, developable and convex surfaces}

\begin{definition}[Ruled surface]
	A smooth surface $\Sigma\subset\mathbb{R}^3$ is called \emph{ruled} if
	through every point $p\in\Sigma$ there passes at least one straight
	line (a \emph{ruling}) contained entirely in $\Sigma$.
	
	A typical local parametrisation is
	\[
	\sigma(u,v)=c(u)+v\,d(u),
	\]
	where $c(u)$ is a space curve (the \emph{directrix}) and $d(u)$ is a
	direction vector.  For fixed $u$, the map $v\mapsto\sigma(u,v)$ is a
	straight line.
\end{definition}

\begin{example}
	\leavevmode
	\begin{enumerate}
		\item Plane: $\sigma(u,v)=(u,v,0)$.
		\item Cylinder: $\sigma(u,v)=(\cos u,\sin u,v)$.
		\item Cone: $\sigma(u,v)=(v\cos u,v\sin u,v)$.
		\item Helicoid: $\sigma(u,v)=(v\cos u,v\sin u,au)$.
		\item Hyperbolic paraboloid: $\sigma(u,v)=(u,v,uv)$.
	\end{enumerate}
\end{example}

\begin{definition}[Developable surface]
	A regular surface $\Sigma\subset\mathbb{R}^3$ is called
	\emph{developable} if it is locally isometric to the plane (it can be
	unfolded onto $\mathbb{R}^2$ without stretching).  For regular surfaces
	this is equivalent to vanishing Gauss curvature:
	\[
	\Sigma \text{ developable } \quad\Longleftrightarrow\quad K\equiv0.
	\]
\end{definition}

Among ruled surfaces, a smooth ruled surface with $K=0$ is locally a
piece of a plane, cylinder, cone or a tangent developable of a space
curve.  Ruled surfaces with $K\neq0$ (such as the helicoid or the
hyperbolic paraboloid, which have $K<0$) are \emph{non--developable}.

\begin{definition}[Convex and strictly convex surfaces]
	A set $R\subset\mathbb{R}^3$ is \emph{convex} if for any $x,y\in R$ the
	segment $[x,y]$ lies in $R$.  A compact smooth surface
	$\Sigma\subset\mathbb{R}^3$ is called \emph{convex} if it is the
	boundary of a convex region $R$.
	
	We say $\Sigma=\partial R$ is \emph{strictly convex} if $R$ is convex
	and for any $x,y\in R$ one has
	\[
	[x,y]\subset R,\qquad [x,y]\cap\Sigma\subset\{x,y\},
	\]
	i.e.\ no nontrivial line segment lies on the boundary.
\end{definition}

Strict convexity implies convexity, but not conversely (cylinders,
cubes and prisms are convex but not strictly convex).  For smooth
strictly convex surfaces, the Gauss map turns out to be injective and,
under the additional assumption $K>0$, a global diffeomorphism onto
$S^2$.  In that case one obtains the ``convex Gauss--Bonnet theorem''
\[
\int_\Sigma K\,dA = 4\pi.
\]

\subsection{Matrix formulation and the fundamental theorem}

Let $\sigma:V\to\Sigma\subset\mathbb{R}^3$ be an allowable
parametrisation, with $V\subset\mathbb{R}^2$ a connected open set with
coordinates $(u,v)$.  Write $E,F,G$ and $L,M,N$ for the first and
second fundamental form coefficients as above, and let
$n$ be the unit normal along $\sigma$.

Define
\[
\alpha_{ijk}=\langle\sigma_i,\sigma_{jk}\rangle,\qquad i,j,k\in\{u,v\},
\]
where subscripts denote partial derivatives.  Differentiating $E,F,G$
and using symmetry of the inner product, one finds that all
$\alpha_{ijk}$ are determined by $E,F,G$ and their first derivatives.

Let
\[
P=(\sigma_u\;\;\sigma_v\;\;n)
\]
be the $3\times3$ matrix whose columns are $\sigma_u$, $\sigma_v$ and
$n$.  Define
\[
C=\begin{pmatrix}
	E & F & 0\\[2pt]
	F & G & 0\\[2pt]
	0 & 0 & 1
\end{pmatrix},
\]
\[
D_1=\begin{pmatrix}
	\alpha_{uuu} & \alpha_{uvu} & -L\\[2pt]
	\alpha_{vuu} & \alpha_{vvu} & -M\\[2pt]
	L & M & 0
\end{pmatrix},\qquad
D_2=\begin{pmatrix}
	\alpha_{uuv} & \alpha_{uvv} & -M\\[2pt]
	\alpha_{vuv} & \alpha_{vvv} & -N\\[2pt]
	M & N & 0
\end{pmatrix}.
\]

A direct computation using $\langle\sigma_u,n\rangle=
\langle\sigma_v,n\rangle=0$ and $\langle n,n\rangle=1$ shows
\[
P^tP=C.
\]
Similarly, $P_u=(\sigma_{uu}\;\;\sigma_{uv}\;\;n_u)$ gives
\[
P^tP_u = D_1,
\]
and $P_v=(\sigma_{uv}\;\;\sigma_{vv}\;\;n_v)$ gives
\[
P^tP_v = D_2.
\]

Let
\[
A=C^{-1}D_1,\qquad B=C^{-1}D_2.
\]
Using $C=P^tP$ one finds
\[
A = P^{-1}P_u,\qquad B=P^{-1}P_v,
\]
hence
\[
P_u = P A,\qquad P_v = P B.
\]
Taking transposes,
\[
P_u^t = A^tP^t,\qquad P_v^t = B^tP^t.
\]
Thus each column $\xi$ of $P^t$ satisfies the linear first-order PDE
system
\[
\xi_u = A^t\xi,\qquad \xi_v = B^t\xi.
\]

\begin{theorem}[Fundamental theorem of surface theory, uniqueness part]
	Suppose two connected smooth surfaces $\Sigma_1,\Sigma_2\subset
	\mathbb{R}^3$ have the same first and second fundamental forms on a
	parameter domain $V\subset\mathbb{R}^2$.  Then $\Sigma_1$ and $\Sigma_2$
	are congruent, i.e.\ there exists a rigid motion of $\mathbb{R}^3$
	mapping $\Sigma_1$ onto $\Sigma_2$.
\end{theorem}

\begin{proof}[Idea of proof]
	The fundamental forms determine the matrices $C,D_1,D_2$ and hence the
	matrices $A,B$.  For each surface, the associated matrix $P$ (whose
	columns are $\sigma_u,\sigma_v,n$) satisfies $P_u=PA$, $P_v=PB$, so the
	columns of $P^t$ satisfy the same linear system
	$\xi_u=A^t\xi$, $\xi_v=B^t\xi$.
	
	By composing with a rigid motion if necessary, we can arrange that at
	some point $(u_0,v_0)\in V$ the positions and frames agree:
	$\sigma^{(1)}(u_0,v_0)=\sigma^{(2)}(u_0,v_0)$ and
	$P^{(1)}(u_0,v_0)=P^{(2)}(u_0,v_0)$.  Assuming uniqueness of solutions
	for the linear PDE system with such initial data, it follows that
	$P^{(1)}(u,v)=P^{(2)}(u,v)$ for all $(u,v)\in V$.  In particular the
	first two columns (the partial derivatives of the parametrisations)
	coincide, so the two surfaces differ only by the initial rigid motion.
\end{proof}

\section*{Problem 3: Partial converse to Clairaut's relation}

\textbf{Problem.}
Let $\Sigma\subset\mathbb R^3$ be a smooth surface of revolution.
Suppose a smooth curve $\gamma:(a,b)\to\Sigma$ satisfies
\[
\rho(t)\cos\theta(t)=\text{constant},
\]
where $\rho(t)$ is the distance from $\gamma(t)$ to the axis of revolution
and $\theta(t)$ is the angle between $\gamma$ and the parallel at $\gamma(t)$.
Assume that there is no positive--length interval on which $\gamma$ coincides
with a parallel. Show that $\gamma$ is a geodesic.

\medskip
\textbf{Theory.}
A surface of revolution is parametrized by
\[
X(u,v) = (\rho(u)\cos v,\ \rho(u)\sin v,\ z(u)),
\]
with induced metric
\[
g = E(u)\,du^2 + G(u)\,dv^2,\qquad
E(u)=\rho'(u)^2+z'(u)^2,\quad G(u)=\rho(u)^2.
\]
For a unit--speed curve $(u(s),v(s))$, the tangent has components
$u',v'$ and satisfies $E u'^2 + G v'^2 = 1$. The angle $\theta$ between
$\gamma$ and the parallel (direction $\partial_v$) is determined by
\[
\cos\theta = g\!\left(\dot\gamma,\frac{1}{\sqrt{G}}\partial_v\right)
= \sqrt{G}\,v' = \rho\,v'.
\]
Hence $\rho\cos\theta = G v'$.

\medskip
\textbf{Solution.}
Reparametrize $\gamma$ by arc length $s$, so that
\[
E(u)u'^2 + G(u)v'^2 = 1.
\]
The Clairaut condition reads
\[
G(u(s))\,v'(s) = c,\qquad c\in\mathbb R.
\tag{1}
\]
Differentiating,
\[
0 = \frac{d}{ds}(G v')
= G_u u' v' + G v'',
\]
so
\[
v'' + \frac{G_u}{G}u'v' = 0.
\tag{2}
\]
For $g=E(u)\,du^2+G(u)\,dv^2$ the Christoffel symbols with index $2$ are
$\Gamma^2_{12}=\Gamma^2_{21} = \frac{1}{2G}G_u$, so the $v$-geodesic equation
\[
v'' + 2\Gamma^2_{12}u'v' = 0
\]
is exactly (2).

Now differentiate the unit--speed condition:
\[
0 = \frac{d}{ds}(E u'^2 + G v'^2)
= E_u u' u'^2 + 2E u'u'' + G_u u' v'^2 + 2G v'v''.
\]
Using (2) to eliminate $v''$, we obtain (for $u'\neq0$)
\[
2E u'' + E_u u'^2 - G_u v'^2 = 0,
\]
or
\[
u'' + \frac{E_u}{2E}u'^2 - \frac{G_u}{2E}v'^2 = 0.
\tag{3}
\]
For this metric, $\Gamma^1_{11}=\frac{E_u}{2E}$ and
$\Gamma^1_{22}=-\frac{G_u}{2E}$, so (3) is exactly the $u$-geodesic
equation. On any interval where $u'\neq0$, $\gamma$ satisfies both
geodesic equations and hence is a geodesic. The only way $u'\equiv0$
on an interval is that $\gamma$ coincides with a parallel there, which
is excluded by hypothesis. Therefore $\gamma$ is a geodesic.

\medskip
\textbf{Remark.}
For a geodesic on a surface of revolution, $G v'=\rho\cos\theta$ is constant
(Clairaut's relation). This problem shows a partial converse: if a curve
satisfies that relation and is not a parallel, it must be a geodesic.

\section*{Problem 4: Geodesics on a cone}

Let $a>0$ and consider the half-cone
\[
\Sigma = \{(x,y,z)\in\mathbb R^3 \mid z^2 = a(x^2+y^2),\ z>0\}.
\]

\subsubsection*{(a) Local isometry and self--intersection}

\textbf{Problem.}
Show that $\Sigma$ is locally isometric to the Euclidean plane.
By opening the cone into a planar sector, or otherwise, show that
when $a=3$ no geodesic on $\Sigma$ intersects itself, but for $a>3$
there exist geodesics which self--intersect.

\medskip
\textbf{Solution.}
Set $r=\sqrt{x^2+y^2}$, $z=\sqrt{a}\,r$ and parametrize
\[
X(r,\phi) = (r\cos\phi,r\sin\phi,\sqrt{a}\,r),\qquad r>0,\ \phi\in\mathbb R.
\]
Then
\[
g = (1+a)\,dr^2 + r^2\,d\phi^2.
\]
Define $s=\sqrt{1+a}\,r$. Then $ds^2=(1+a)\,dr^2$ and
\[
g = ds^2 + \frac{s^2}{1+a}\,d\phi^2
= ds^2 + s^2\,d\psi^2,\qquad \psi=\frac{\phi}{\sqrt{1+a}}.
\]
This is the Euclidean metric in polar coordinates $(s,\psi)$.
Since $\phi$ has period $2\pi$, the angle $\psi$ has period
\[
\alpha = \frac{2\pi}{\sqrt{1+a}}.
\]
Thus $\Sigma$ is locally isometric to the Euclidean plane sector
\[
S_\alpha = \{(s,\psi) \mid s>0,\ 0<\psi<\alpha\},
\]
with the two radial edges identified by rotation through the angle $\alpha$.

Geodesics on $\Sigma$ correspond to straight lines in $S_\alpha$,
and two points of $S_\alpha$ represent the same point of $\Sigma$
if they are related by rotation about the origin through an integer
multiple of $\alpha$.

If $a=3$ then $\sqrt{1+a}=2$ and $\alpha=\pi$. Rotation by $\pi$ sends a
line $L$ to:
\begin{itemize}
	\item itself, if $L$ passes through the origin (this would correspond to
	a geodesic through the cone apex, which is not in $\Sigma$),
	\item a parallel disjoint line, if $L$ does not pass through the origin.
\end{itemize}
Thus for lines corresponding to geodesics on $\Sigma$, $L$ and its rotated
copy $R_{\pm\pi}(L)$ are disjoint. Hence no geodesic self--intersects when
$a=3$.

If $a>3$ then $\sqrt{1+a}>2$ and $\alpha<\pi$. For any such angle, one can
choose a line $L$ not through the origin so that $L$ intersects its rotated
copy $R_{\alpha}(L)$ in a point $p\neq0$ lying inside the sector $S_\alpha$.
The two points $p$ and $R_{\alpha^{-1}}(p)$ on $L$ correspond to the same
point of $\Sigma$, so the corresponding geodesic on $\Sigma$ passes through
this point twice and thus self--intersects. Therefore for $a>3$ there exist
self--intersecting geodesics.

\subsubsection*{(b) Smallest height and Clairaut}

\textbf{Problem.}
Let $\gamma$ be a geodesic on $\Sigma$ which meets the parallel $z=1$
at angle $\theta_0$ (angle with the parallel). If $\theta_0\neq\pi/2$,
show that the smallest value $z_{\min}$ of $z$ along $\gamma$ is achieved,
and using Clairaut’s relation show that $z_{\min}$ is independent of $a$.
What happens when $\theta_0=\pi/2$?

\medskip
\textbf{Solution.}
The cone is a surface of revolution about the $z$-axis with radius
\[
\rho = r = \frac{z}{\sqrt{a}}.
\]
For a geodesic $\gamma$ on a surface of revolution, Clairaut’s relation
gives
\[
\rho\cos\theta = C,
\]
where $\theta$ is the angle between $\gamma$ and the parallels and $C$ is
constant along $\gamma$.

At the intersection with $z=1$ we have $\rho = 1/\sqrt{a}$, so
\[
C = \frac{1}{\sqrt{a}}\cos\theta_0.
\tag{$*$}
\]

If $\theta_0\neq\pi/2$ then $\cos\theta_0>0$, hence $C>0$. Since
$0\le\theta\le\pi/2$ along $\gamma$, we have $\cos\theta\le1$ and therefore
\[
\rho = \frac{C}{\cos\theta} \ge C.
\]
Thus $\rho$ has a positive lower bound. In the planar sector picture,
$\gamma$ lifts to a straight line $L$ which has a unique closest point to
the origin; at that point the line is tangent to the circle of radius
$\rho_{\min}$, so $\theta=0$ and $\cos\theta=1$. Hence
\[
\rho_{\min} = C.
\]
Consequently
\[
z_{\min} = \sqrt{a}\,\rho_{\min}
= \sqrt{a}\,C
= \sqrt{a}\cdot\frac{1}{\sqrt{a}}\cos\theta_0
= \cos\theta_0,
\]
which is independent of $a$.

If $\theta_0=\pi/2$ then $\cos\theta_0=0$ and $(*)$ gives $C=0$, so
$\rho\cos\theta\equiv0$. Since $\cos\theta\ge0$, this forces
$\cos\theta\equiv0$ and thus $\theta\equiv\pi/2$: the geodesic is a meridian
of the cone, everywhere orthogonal to the parallels. Along such a geodesic
$z\to0$ as one approaches the apex; the infimum of $z$ is $0$, but this
value is not attained because the apex is not contained in $\Sigma$.
	
	
	
\begin{definition}[Surface of revolution]
	Let the $z$-axis be fixed in $\mathbb R^3$ and let
	$(\rho(u),0,z(u))$, $u\in I$, be a smooth plane curve in the $rz$-plane
	with $\rho(u)\ge 0$. The \emph{surface of revolution} obtained by rotating
	this curve about the $z$-axis is
	\[
	\Sigma = \{(\rho(u)\cos v,\ \rho(u)\sin v,\ z(u)) \mid u\in I,\ v\in[0,2\pi)\}.
	\]
	In the coordinates $(u,v)$ the induced metric has the form
	\[
	g = E(u)\,du^2 + G(u)\,dv^2,\qquad
	E(u)=\rho'(u)^2+z'(u)^2,\quad G(u)=\rho(u)^2.
	\]
\end{definition}

\begin{definition}[Angle with parallels]
	Let $\gamma(s)=(u(s),v(s))$ be a unit-speed curve on $\Sigma$.
	The unit tangent vector is
	\[
	T = u'(s)\,\partial_u + v'(s)\,\partial_v,
	\]
	and the unit vector tangent to the parallel at the same point is
	\[
	P = \frac{1}{\sqrt{G(u)}}\,\partial_v.
	\]
	The angle $\theta(s)$ between $\gamma$ and the parallel is defined by
	$\cos\theta = g(T,P)$. A short computation gives
	\[
	\cos\theta = \sqrt{G(u)}\,v'(s) = \rho(u)\,v'(s).
	\]
\end{definition}

\begin{theorem}[Clairaut's relation]
	Let $\Sigma$ be a surface of revolution as above and let
	$\gamma$ be a unit-speed geodesic on $\Sigma$.
	Then the quantity
	\[
	G(u(s))\,v'(s)=\rho(u(s))\cos\theta(s)
	\]
	is constant along $\gamma$. Equivalently,
	\[
	\boxed{\rho\bigl(\gamma(s)\bigr)\cos\theta(s)=C.}
	\]
\end{theorem}

\begin{example}[Sphere]
	On the unit sphere in spherical coordinates $(\varphi,\theta)$
	(colatitude and longitude) we have $\rho=\sin\varphi$, so for a geodesic
	(great circle)
	\[
	\sin\varphi\cos\theta = C.
	\]
	When the geodesic approaches the poles ($\sin\varphi$ small),
	it becomes more tangent to the parallels ($\theta\to 0$), and when it
	crosses the equator ($\varphi=\pi/2$) the angle with the equator is fixed.
\end{example}

\begin{example}[Cone]
	For the cone $z^2=a(x^2+y^2)$, $z>0$, we have $\rho = z/\sqrt{a}$,
	hence a geodesic satisfies
	\[
	\frac{z}{\sqrt{a}}\cos\theta = C.
	\]
	If a geodesic meets the parallel $z=1$ at angle $\theta_0$, then
	$C=\frac{1}{\sqrt{a}}\cos\theta_0$ and the minimal height along the
	geodesic is $z_{\min}=\cos\theta_0$, independent of $a$.
\end{example}

	\subsubsection*{Example: a cone is locally isometric to a planar sector}
	
	We explain why a right circular cone (with its tip removed) is a developable
	surface, i.e.\ locally isometric to a region of the plane.  Geometrically, one
	can cut the cone along a generator and unroll it into a sector of a circle; the
	analytic computation below shows that this unrolling preserves the first
	fundamental form.
	
	\paragraph{Geometric picture.}
	Consider a party-hat cone of slant height $\rho$ and base radius $R_b$.  If we
	cut the cone along one generator and flatten it onto the plane, we obtain a
	circular sector of radius $\rho$ and some angle $\Theta$.  The arc length of
	the base circle equals the arc length of the boundary of the sector:
	\[
	2\pi R_b = \Theta\cdot\rho,
	\]
	hence $\Theta = 2\pi R_b/\rho < 2\pi$.  Gluing the straight sides of this
	sector reproduces the cone.
	
	The key point is that in this process we only \emph{bend} the surface, without
	stretching or compressing it.  Therefore distances on the cone coincide with
	distances on the planar sector: the two surfaces are isometric.
	
	\paragraph{Analytic proof.}
	Consider a right circular cone whose tip is at the origin and whose symmetry
	axis is the $z$--axis.  Fix an angle $\alpha\in(0,\pi/2)$ between any generator
	and the axis.  A convenient parametrisation is
	\[
	X(\rho,\phi)
	= \bigl(\rho\sin\alpha\cos\phi,\;
	\rho\sin\alpha\sin\phi,\;
	\rho\cos\alpha\bigr),
	\qquad \rho>0,\ \phi\in(0,2\pi),
	\]
	where $\rho$ is the \emph{slant distance from the apex} and $\phi$ is the angle
	around the axis.
	
	We compute the first fundamental form.  First derivatives:
	\[
	X_\rho = (\sin\alpha\cos\phi,\;\sin\alpha\sin\phi,\;\cos\alpha),
	\]
	\[
	X_\phi = (-\rho\sin\alpha\sin\phi,\;\rho\sin\alpha\cos\phi,\;0).
	\]
	Thus
	\[
	E = \langle X_\rho,X_\rho\rangle
	= \sin^2\alpha(\cos^2\phi+\sin^2\phi)+\cos^2\alpha = 1,
	\]
	\[
	F = \langle X_\rho,X_\phi\rangle = 0,
	\]
	\[
	G = \langle X_\phi,X_\phi\rangle
	= \rho^2\sin^2\alpha(\sin^2\phi+\cos^2\phi)
	= \rho^2\sin^2\alpha.
	\]
	Therefore the induced metric on the cone is
	\[
	ds^2 = E\,d\rho^2 + 2F\,d\rho\,d\phi + G\,d\phi^2
	= d\rho^2 + \rho^2\sin^2\alpha\,d\phi^2.
	\]
	
	Now consider a planar sector with polar coordinates $(\rho,\theta)$, endowed
	with the usual Euclidean metric
	\[
	ds^2_{\text{plane}} = d\rho^2 + \rho^2\,d\theta^2.
	\]
	Define a map from this sector to the cone by the change of angular variable
	\[
	\theta = \sin\alpha\;\phi
	\qquad\Longleftrightarrow\qquad
	\phi = \frac{\theta}{\sin\alpha}.
	\]
	Then $d\theta = \sin\alpha\,d\phi$, and if we pull back the cone metric via this
	map we obtain
	\[
	d\rho^2 + \rho^2\sin^2\alpha\,d\phi^2
	= d\rho^2 + \rho^2(d\theta)^2
	= ds^2_{\text{plane}}.
	\]
	Thus the map
	\[
	(\rho,\theta) \longmapsto
	\bigl(\rho\sin\alpha\cos(\theta/\sin\alpha),\;
	\rho\sin\alpha\sin(\theta/\sin\alpha),\;
	\rho\cos\alpha\bigr)
	\]
	is an \emph{isometry} between a planar sector
	\[
	\{\,(\rho,\theta)\mid 0<\rho<\infty,\ 0<\theta<\Theta\,\}
	\]
	(with $\Theta = 2\pi\frac{R_b}{\rho}$ as above) and the cone with its apex
	removed.
	
	Distances and angles are preserved because the first fundamental form in these
	coordinates coincides exactly with the Euclidean metric on the sector.
	Consequently the cone (without the tip) is locally isometric to a region of the
	plane and hence is a \emph{developable} surface (its Gaussian curvature
	satisfies $K\equiv0$ away from the vertex).
	
	\begin{remark}
		Globally, the cone is not isometric to the whole plane, but only to a sector of
		angle $\Theta<2\pi$.  However, in small neighbourhoods that do not wrap around
		the apex, the cone and the plane are indistinguishable from the metric point of
		view.  This illustrates the local nature of developability.
	\end{remark}
	
	\subsection*{Theory: Surfaces of revolution}
	
	A \emph{surface of revolution} is obtained by rotating a plane curve around a
	fixed line (the axis of revolution).  We describe the standard case where the
	axis is the $z$--axis.
	
	\paragraph{Generating curve.}
	Let
	\[
	\gamma(u) = (\rho(u),0,z(u)),\qquad u\in I\subset\mathbb{R},
	\]
	be a smooth curve in the $xz$--plane (equivalently, in the $(\rho,z)$
	half–plane), where
	\[
	\rho(u)\ge0\quad\text{is the distance from the $z$--axis},\qquad
	z(u)\in\mathbb{R}\quad\text{is the height}.
	\]
	This curve is called the \emph{profile} or \emph{meridian} of the surface.
	The line of revolution is the \emph{$z$--axis}.
	
	\paragraph{Parametrisation of the surface of revolution.}
	Rotating the point $\gamma(u)$ by an angle $v\in[0,2\pi)$ around the $z$--axis
	gives the point
	\[
	(\rho(u)\cos v,\ \rho(u)\sin v,\ z(u)).
	\]
	Thus a parametrisation of the surface of revolution generated by $\gamma$ is
	\[
	X:I\times[0,2\pi)\to\mathbb{R}^3,\qquad
	X(u,v) = \bigl(\rho(u)\cos v,\ \rho(u)\sin v,\ z(u)\bigr).
	\]
	Here
	\begin{itemize}
		\item $u$ moves along the generating curve (meridians),
		\item $v$ is the rotation angle around the $z$--axis (parallels).
	\end{itemize}
	
	\paragraph{First fundamental form.}
	We compute the partial derivatives:
	\[
	X_u = \bigl(\rho'(u)\cos v,\ \rho'(u)\sin v,\ z'(u)\bigr),
	\]
	\[
	X_v = \bigl(-\rho(u)\sin v,\ \rho(u)\cos v,\ 0\bigr).
	\]
	The coefficients of the first fundamental form are
	\[
	E(u) = \langle X_u,X_u\rangle
	= \rho'(u)^2(\cos^2 v+\sin^2 v)+z'(u)^2
	= \rho'(u)^2+z'(u)^2,
	\]
	\[
	F(u) = \langle X_u,X_v\rangle = 0
	\quad\text{(orthogonal coordinates)},
	\]
	\[
	G(u) = \langle X_v,X_v\rangle
	= \rho(u)^2(\sin^2 v+\cos^2 v)
	= \rho(u)^2.
	\]
	Thus the induced metric (first fundamental form) is
	\[
	g = E(u)\,du^2 + 2F(u)\,du\,dv + G(u)\,dv^2
	= E(u)\,du^2 + G(u)\,dv^2,
	\]
	with
	\[
	E(u)=\rho'(u)^2+z'(u)^2,\qquad G(u)=\rho(u)^2.
	\]
	
	\subsubsection*{Examples of surfaces of revolution}
	
	\begin{enumerate}
		\item \textbf{Cylinder of radius $R$.}
		Take $\rho(u)=R$ and $z(u)=u$:
		\[
		X(u,v) = (R\cos v,\ R\sin v,\ u),\qquad
		E(u)=1,\ G(u)=R^2.
		\]
		
		\item \textbf{Right circular cone.}
		Take $\rho(u)=au$, $z(u)=bu$ with $a,b>0$:
		\[
		X(u,v) = (au\cos v,\ au\sin v,\ bu),\qquad
		E(u)=a^2+b^2,\ G(u)=a^2u^2.
		\]
		
		\item \textbf{Sphere of radius $R$.}
		Using the polar angle $u\in(0,\pi)$ as parameter for the meridian,
		\[
		\rho(u)=R\sin u,\qquad z(u)=R\cos u,
		\]
		so
		\[
		X(u,v) = (R\sin u\cos v,\ R\sin u\sin v,\ R\cos u),
		\]
		and
		\[
		E(u) = R^2,\qquad G(u) = R^2\sin^2 u.
		\]
	\end{enumerate}
	
	\subsection*{Problem 5 – A surface with no minimizing geodesic between two points}
	
	\paragraph{Statement.}
	Give an example of a connected smooth surface $\Sigma\subset\mathbb{R}^3$
	and points $p,q\in\Sigma$ for which the infimum
	\[
	\inf_\gamma L(\gamma),
	\]
	taken over all piecewise smooth curves $\gamma:[a,b]\to\Sigma$ with
	$\gamma(a)=p$ and $\gamma(b)=q$, is strictly smaller than the length of any
	piecewise smooth curve $\gamma$ between $p$ and $q$.
	
	\paragraph{Example and solution.}
	Let
	\[
	\Sigma = \{(x,y,0)\in\mathbb{R}^3 : (x,y)\neq(0,0)\},
	\]
	the plane $z=0$ with the origin removed.  This is a connected smooth surface
	in $\mathbb{R}^3$.
	
	Choose
	\[
	p=(-1,0,0),\qquad q=(1,0,0).
	\]
	In the full plane, the straight line segment from $p$ to $q$ has length $2$
	and passes through the origin.  Since the origin is removed, this segment is
	not contained in $\Sigma$ and is not an admissible curve.
	
	Any piecewise smooth curve $\gamma\subset\Sigma$ joining $p$ to $q$ must go
	around the origin, and so has length strictly greater than $2$.  On the other
	hand, for each $\varepsilon>0$ we can construct a curve that avoids the origin
	but has length arbitrarily close to $2$; for instance, in the $xy$-plane:
	\begin{itemize}
		\item the line segment from $(-1,0)$ to $(-\varepsilon,0)$,
		\item a semicircle of radius $\varepsilon$ around the origin,
		\item the line segment from $(\varepsilon,0)$ to $(1,0)$.
	\end{itemize}
	Lifting to $\Sigma\subset\mathbb{R}^3$ we obtain curves whose lengths tend to
	$2$ as $\varepsilon\to0$, but none of which have length exactly $2$.
	
	Therefore
	\[
	\inf_\gamma L(\gamma)=2
	\]
	but no curve in $\Sigma$ connecting $p$ and $q$ attains this infimum.  Hence
	the infimum of lengths is strictly smaller than the length of any admissible
	curve: there is no minimizing geodesic from $p$ to $q$ on $\Sigma$.

	\subsection*{Problem 6 – A ruled surface along the binormal; $\eta$ is geodesic}
	
	\paragraph{Statement.}
	Let $\eta:[s_0,s_1]\to\mathbb{R}^3$ be an embedded smooth curve parametrised by
	arc--length, with $\eta''(s)\neq0$ for all $s$ (nonzero curvature).
	The \emph{binormal vector} to $\eta$ is the unit vector
	\[
	b(s) = \frac{\eta'(s)\times\eta''(s)}{\|\eta'(s)\times\eta''(s)\|}.
	\]
	Consider the ruled surface with parametrisation
	\[
	\sigma(u,v)=\eta(u)+v\,b(u),\qquad
	u\in(s_0,s_1),\ -\varepsilon<v<\varepsilon,\ \varepsilon>0.
	\]
	\begin{enumerate}
		\item[(i)] Show that $D\sigma|_p$ is injective for $p\in\eta$, and hence is
		injective for all points in the image of $\sigma$ with $|v|$
		sufficiently small.
		\item[(ii)] Deduce that, if $\varepsilon$ is sufficiently small, the
		parametrisation $\sigma$ is an injective map and that the image of
		$\sigma$ defines a smooth surface in $\mathbb{R}^3$.
		\item[(iii)] Show that $\eta$ is a geodesic on the resulting surface.
	\end{enumerate}
	
	\paragraph{(i) Differential at points on $\eta$.}
	We compute
	\[
	\sigma_u(u,v) = \eta'(u)+v\,b'(u),\qquad
	\sigma_v(u,v) = b(u).
	\]
	For points on $\eta$ we have $v=0$, so
	\[
	\sigma_u(u,0)=\eta'(u),\qquad \sigma_v(u,0)=b(u).
	\]
	Since $\eta$ is parametrised by arc--length, $\eta'(u)$ is the unit tangent
	$t(u)$.  Because $\eta''(s)\neq0$, the Frenet frame $(t(u),n(u),b(u))$ is
	well-defined; in particular $b(u)$ is a unit vector orthogonal to $t(u)$, so
	$t(u)$ and $b(u)$ are nonzero and linearly independent.
	
	Hence the columns $\sigma_u(u,0)$ and $\sigma_v(u,0)$ are linearly independent
	for each $u$, and the differential
	\[
	D\sigma_{(u,0)}:\mathbb{R}^2\to\mathbb{R}^3
	\]
	has rank $2$.  By continuity of $\sigma_u$ and $\sigma_v$, this remains true
	for all points with $|v|$ sufficiently small, so $D\sigma$ is injective on a
	thin strip around the curve.
	
	\paragraph{(ii) Injectivity and smooth surface.}
	Since $\eta$ is an embedded curve and $[s_0,s_1]$ is compact, the tubular
	neighbourhood theorem implies that there exists $r>0$ such that the map
	\[
	(u,v,w)\mapsto \eta(u)+v\,n(u)+w\,b(u)
	\]
	is an embedding from $(s_0,s_1)\times(-r,r)^2$ into $\mathbb{R}^3$.  In
	particular, restricting to the binormal direction (say $v=0$) and to
	$|w|<r$ shows that
	\[
	(u,w)\mapsto \eta(u)+w\,b(u)
	\]
	is injective.  Choosing $0<\varepsilon<r$ and writing $v$ in place of $w$ we
	obtain that
	\[
	\sigma(u,v)=\eta(u)+v\,b(u),\qquad u\in(s_0,s_1),\ |v|<\varepsilon,
	\]
	is an injective immersion.  Its image is therefore a smooth surface (an
	embedded $2$--dimensional submanifold) in $\mathbb{R}^3$.
	
	\paragraph{(iii) $\eta$ is a geodesic on the surface.}
	Along the curve $\eta$ in the surface we have $v=0$ and
	\[
	\sigma(u,0)=\eta(u),\qquad
	\sigma_u(u,0)=\eta'(u)=t(u),\qquad
	\sigma_v(u,0)=b(u).
	\]
	The surface normal along $\eta$ is
	\[
	N(u)=\frac{\sigma_u(u,0)\times\sigma_v(u,0)}
	{\|\sigma_u(u,0)\times\sigma_v(u,0)\|}
	= \frac{t(u)\times b(u)}{\|t(u)\times b(u)\|}.
	\]
	In the Frenet frame $(t(u),n(u),b(u))$ we have
	$t(u)\times n(u)=b(u)$; thus $t(u)\times b(u)=-n(u)$, and
	\[
	N(u) = \pm n(u).
	\]
	
	The curvature vector of $\eta$ is
	\[
	\eta''(u) = \kappa(u)\,n(u),
	\]
	where $\kappa(u)>0$ is the curvature.  Therefore $\eta''(u)$ is parallel to
	$N(u)$, i.e.\ it is normal to the surface.  The tangential component of
	$\eta''(u)$ in the surface is zero, so the geodesic curvature of $\eta$ on
	this surface vanishes.  Hence $\eta$ is a geodesic on the ruled surface
	$\sigma(u,v)$.

		\sigma_v(u,0)=b(u).

		since in the Frenet frame $(t,n,b)$ we have $t\times n=b$ and hence

		By the intrinsic characterisation of geodesics, $\eta$ is a geodesic on

	\end{itemize}
	
	\subsection*{Problem 7: Isometries and Möbius maps on the 2--sphere}

		prove that the M\"obius group $\operatorname{PSL}(2,\mathbb{C})$ acts on the abstract smooth surface $S^{2}$ by diffeomorphisms.
		
		\emph{Hint:} Check that the generators $z\mapsto az+b$ and $z\mapsto 1/z$ of the M\"obius group
		act smoothly from the original definition of smooth maps in terms of a smooth atlas.
		\item If a M\"obius map $A$ defines an isometry of $S^{2}$, show that it commutes with the antipodal map
		$a:S^{2}\to S^{2}$ given by $a(x,y,z)=(-x,-y,-z)$.
	\end{enumerate}
	
	\paragraph{Theory / Background.}
	
	\begin{itemize}
		\item The isometry group of the round sphere $(S^{2},g_{\mathrm{round}})$ is the group of restrictions of
		orthogonal transformations $A\in O(3)$.
		Great circles (intersections of $S^{2}$ with $2$--planes through the origin) are exactly the geodesics.
		\item For $p\in S^{2}$ the exponential map
		\[
		\exp_p:T_pS^{2}\to S^{2},\qquad
		\exp_p(v)=\cos |v|\;p+\sin |v|\;\frac{v}{|v|},
		\]
		is a diffeomorphism from $\{v\in T_pS^{2}:|v|<\pi\}$ onto $S^{2}\setminus\{-p\}$.
		\item Stereographic projection $\sigma:S^{2}\setminus\{N\}\to\mathbb{C}$ from the north pole $N$ is a smooth diffeomorphism
		and extends to a homeomorphism $\sigma:S^{2}\to\hat{\mathbb{C}}:=\mathbb{C}\cup\{\infty\}$, $\sigma(N)=\infty$.
		\item M\"obius transformations are maps
		\[
		z\longmapsto \frac{az+b}{cz+d},\qquad
		\begin{pmatrix}a&b\\c&d\end{pmatrix}\in\operatorname{SL}(2,\mathbb{C}),
		\]
		considered up to nonzero scalar multiples; they form the group $\operatorname{PSL}(2,\mathbb{C})$.
		The group is generated by translations $z\mapsto z+b$, dilations/rotations $z\mapsto az$, and inversion $z\mapsto 1/z$.
		\item Any linear map $B:\mathbb{R}^{3}\to\mathbb{R}^{3}$ commutes with the antipodal map $-\mathrm{id}$:
		$B(-x)=-B(x)$.
	\end{itemize}
	
	\paragraph{Solution.}
	
	\subparagraph{(a) Isometries of $S^{2}$ come from $O(3)$.}
	
	Let $f:S^{2}\to S^{2}$ be an isometry.
	
	\emph{Step 1: Extend the derivative.}
	Choose $p\in S^{2}$ and set $q=f(p)$.  
	Because $f$ is an isometry, its differential
	\[
	df_p:T_pS^{2}\to T_qS^{2}
	\]
	is an isometry of inner--product spaces.
	
	Choose orthonormal bases $\{e_1,e_2\}$ of $T_pS^{2}$ and $\{e'_1,e'_2\}$ of $T_qS^{2}$ such that
	$e'_i=df_p(e_i)$. Then $\{e_1,e_2,p\}$ and $\{e'_1,e'_2,q\}$ are orthonormal bases of $\mathbb{R}^{3}$.
	Define $A:\mathbb{R}^{3}\to\mathbb{R}^{3}$ by
	\[
	A(e_i)=e'_i,\qquad A(p)=q.
	\]
	In these bases the matrix of $A$ is orthogonal, so $A\in O(3)$.
	In particular $A(S^{2})=S^{2}$ and $A$ restricts to an isometry of $S^{2}$.
	
	\emph{Step 2: Reduce to an isometry fixing a point with identity derivative.}
	Define
	\[
	g := A^{-1}\circ f:S^{2}\to S^{2}.
	\]
	Then $g$ is an isometry, $g(p)=p$, and
	\[
	dg_p=(A^{-1})_{*}\circ df_p=\mathrm{id}_{T_pS^{2}}.
	\]
	
	\emph{Step 3: Show that $g=\mathrm{id}_{S^{2}}$.}
	Let $v\in T_pS^{2}$ with $|v|<\pi$. Let
	\[
	\gamma_v(t):=\exp_p(tv)
	\]
	be the unique geodesic with $\gamma_v(0)=p$, $\gamma'_v(0)=v$.
	Since $g$ is an isometry and $g(p)=p$, the curve $g\circ\gamma_v$ is a geodesic starting at $p$.
	Its initial velocity is
	\[
	(g\circ\gamma_v)'(0)=dg_p(\gamma_v'(0))=dg_p(v)=v,
	\]
	so $g\circ\gamma_v$ and $\gamma_v$ are geodesics with the same initial data.
	By uniqueness of geodesics, $g(\gamma_v(t))=\gamma_v(t)$ for all $t$ with $|t|<\pi$.
	
	The exponential map $\exp_p$ is a diffeomorphism from
	$\{v\in T_pS^{2}:|v|<\pi\}$ onto $S^{2}\setminus\{-p\}$, so
	\[
	g(x)=x\qquad\text{for all }x\in S^{2}\setminus\{-p\}.
	\]
	By continuity, $g(-p)=\lim_{t\to\pi^-}g(\gamma_v(t))
	=\lim_{t\to\pi^-}\gamma_v(t)=-p$, hence $g$ is the identity on all of $S^{2}$.
	
	\emph{Step 4: Conclusion.}
	We have $f=A\circ g=A|_{S^{2}}$, so $f$ is the restriction to $S^{2}$ of some $A\in O(3)$.
	
	\subparagraph{(b) The M\"obius group acts smoothly on $S^{2}$.}
	
	Let $\sigma:S^{2}\to\hat{\mathbb{C}}$ be stereographic projection (extended by $\sigma(N)=\infty$).
	For $M\in\operatorname{PSL}(2,\mathbb{C})$ define
	\[
	\Phi_M := \sigma^{-1}\circ M\circ\sigma:S^{2}\to S^{2}.
	\]
	
	\emph{Group action.}
	Since $\sigma$ and $M$ are bijections, so is $\Phi_M$. For $M_1,M_2\in\operatorname{PSL}(2,\mathbb{C})$ we have
	\[
	\Phi_{M_1M_2}
	=\sigma^{-1}\circ M_1M_2\circ\sigma
	=(\sigma^{-1}\circ M_1\circ\sigma)\circ(\sigma^{-1}\circ M_2\circ\sigma)
	=\Phi_{M_1}\circ\Phi_{M_2},
	\]
	and clearly $\Phi_{\mathrm{id}}=\mathrm{id}_{S^{2}}$. Thus $\operatorname{PSL}(2,\mathbb{C})$ acts on $S^{2}$.
	
	\emph{Smoothness.}
	Consider the atlas $\{(U_0,\varphi_0),(U_1,\varphi_1)\}$ on $S^{2}$, where
	\[
	U_0=S^{2}\setminus\{N\},\quad \varphi_0=\sigma,\qquad
	U_1=S^{2}\setminus\{S\},\quad \varphi_1= z\mapsto \frac{1}{\sigma(z)}.
	\]
	A map $\Psi:S^{2}\to S^{2}$ is smooth iff each coordinate representation
	$\varphi_i\circ\Psi\circ\varphi_j^{-1}$ is smooth between open subsets of $\mathbb{C}$.
	
	The group $\operatorname{PSL}(2,\mathbb{C})$ is generated by
	\[
	T_{a,b}:z\mapsto az+b,\qquad I:z\mapsto 1/z.
	\]
	These maps are rational functions of $z$, hence smooth on the chart $U_0$.
	On the chart near $\infty$ (where we use coordinates $w=1/z$) their coordinate expressions
	remain rational (e.g.\ $w\mapsto w/(a+bw)$ for $z\mapsto az+b$), hence smooth.
	Therefore each generator induces a smooth diffeomorphism of $S^{2}$.
	
	Compositions and inverses of smooth diffeomorphisms are smooth diffeomorphisms.
	Hence for every $M\in\operatorname{PSL}(2,\mathbb{C})$ the map $\Phi_M$ is a diffeomorphism.
	Thus $\operatorname{PSL}(2,\mathbb{C})$ acts on $S^{2}$ by diffeomorphisms.
	
	\subparagraph{(c) Isometric M\"obius maps commute with the antipodal map.}
	
	Suppose a M\"obius map $A$ induces an isometry of $(S^{2},g_{\mathrm{round}})$.
	By part~(b) this gives a diffeomorphism $f_A:S^{2}\to S^{2}$.
	Since $f_A$ is an isometry, part~(a) implies that there exists $B\in O(3)$ with
	\[
	f_A = B|_{S^{2}}.
	\]
	
	Let $a:S^{2}\to S^{2}$ be the antipodal map $a(x)=-x$.
	Because $B$ is linear, $B(-x)=-B(x)$ for all $x\in\mathbb{R}^{3}$.
	Hence, for every $x\in S^{2}$,
	\[
	f_A(a(x)) = B(-x) = -B(x) = a(B(x)) = a(f_A(x)).
	\]
	Thus $f_A\circ a = a\circ f_A$, i.e.\ the isometric M\"obius map commutes with the antipodal map.
	
	\paragraph{Examples.}
	
	\begin{enumerate}
		\item For $\theta\in\mathbb{R}$, the rotation
		\[
		R_\theta = \begin{pmatrix}
			\cos\theta & -\sin\theta & 0\\
			\sin\theta & \cos\theta & 0\\
			0          & 0          & 1
		\end{pmatrix}\in SO(3)
		\]
		restricts to an isometry of $S^{2}$. In stereographic coordinates $z=x+iy$ this corresponds
		to the M\"obius map $z\mapsto e^{i\theta}z$.
		\item The dilation $z\mapsto 2z$ is a M\"obius transformation but does not come from any
		$B\in O(3)$, hence its induced map on $S^{2}$ is not an isometry of the round metric.
		\item Every element of $SO(3)$ corresponds, via stereographic projection, to a M\"obius map of the form
		\[
		z\longmapsto \frac{\alpha z+\beta}{-\overline{\beta}\,z+\overline{\alpha}},\qquad
		|\alpha|^{2}+|\beta|^{2}=1,
		\]
		and these are precisely the orientation--preserving isometries of $S^{2}$.
	\end{enumerate}
	
	\subsection*{Möbius transformations, \texorpdfstring{$\mathrm{PSL}(2,\mathbb{C})$}{PSL(2,C)}, and the action on $S^{2}$}
	
	In this subsection we explain in detail why the Möbius group can be identified
	with $\mathrm{PSL}(2,\mathbb{C})$ and why, after identifying
	$\widehat{\mathbb{C}} = \mathbb{C}\cup\{\infty\}$ with $S^{2}\subset\mathbb{R}^3$
	via stereographic projection, this group acts on the abstract smooth surface
	$S^2$ by diffeomorphisms.
	
	\subsubsection*{1. The Riemann sphere and stereographic projection}
	
	We consider the unit sphere
	\[
	S^{2} = \{(x,y,t)\in\mathbb{R}^{3} : x^{2}+y^{2}+t^{2}=1\}.
	\]
	Let $N=(0,0,1)$ be the north pole, and let
	\[
	\Pi : S^{2}\setminus\{N\} \longrightarrow \mathbb{R}^{2}\cong\mathbb{C}
	\]
	be the stereographic projection from $N$ onto the plane $\{t=0\}$.  In
	coordinates one checks that
	\[
	\Pi(x,y,t) = \left( \frac{x}{1-t},\,\frac{y}{1-t} \right),
	\]
	so if we identify $\mathbb{R}^{2}$ with $\mathbb{C}$ via
	$(X,Y)\mapsto X+iY$, then
	\[
	\Pi(x,y,t) = z = \frac{x}{1-t} + i\,\frac{y}{1-t}.
	\]
	
	We extend $\Pi$ to a bijection
	\[
	\Phi : S^{2} \longrightarrow \widehat{\mathbb{C}} := \mathbb{C}\cup\{\infty\}
	\]
	by setting $\Phi(N) = \infty$ and $\Phi|_{S^{2}\setminus\{N\}} = \Pi$.
	This is a homeomorphism, and in fact a smooth diffeomorphism in the sense
	of smooth manifolds if we equip $\widehat{\mathbb{C}}$ with the standard
	structure of a 1--dimensional complex manifold (the Riemann sphere).  Its
	inverse $\Phi^{-1}:\widehat{\mathbb{C}}\to S^{2}$ is given by the classical
	formulas (for $z=x+iy\in\mathbb{C}$)
	\[
	\Phi^{-1}(z) =
	\left(
	\frac{2x}{|z|^{2}+1},\,
	\frac{2y}{|z|^{2}+1},\,
	\frac{|z|^{2}-1}{|z|^{2}+1}
	\right),
	\qquad
	\Phi^{-1}(\infty) = N.
	\]
	
	Thus $S^{2}$ and $\widehat{\mathbb{C}}$ are the same smooth surface, viewed
	in two different ways; we will freely pass between them via $\Phi$.
	
	\subsubsection*{2. Möbius transformations on the extended complex plane}
	
	A \emph{Möbius transformation} (or linear fractional transformation) is a
	map $f:\widehat{\mathbb{C}}\to\widehat{\mathbb{C}}$ of the form
	\[
	f(z) = \frac{az+b}{cz+d},
	\qquad a,b,c,d\in\mathbb{C},\quad ad-bc\neq 0,
	\]
	with the usual conventions
	\[
	f\!\left(-\frac{d}{c}\right) = \infty\quad (c\neq 0),\qquad
	f(\infty) =
	\begin{cases}
		\dfrac{a}{c}, & c\neq 0,\\[0.4em]
		\infty, & c=0.
	\end{cases}
	\]
	
	It is well known (and easy to check) that:
	
	\begin{itemize}
		\item Möbius transformations form a group under composition;
		\item the map $z\mapsto z$ is the identity element;
		\item every Möbius transformation is a homeomorphism of
		$\widehat{\mathbb{C}}$ with continuous inverse of the same form.
	\end{itemize}
	
	\subsubsection*{3. Möbius transformations and $2\times 2$ matrices}
	
	To each Möbius transformation $f(z)=\dfrac{az+b}{cz+d}$ we associate the
	matrix
	\[
	A =
	\begin{pmatrix}
		a & b\\
		c & d
	\end{pmatrix}
	\in \mathrm{GL}(2,\mathbb{C}).
	\]
	Composition of Möbius transformations corresponds to matrix multiplication:
	if
	\[
	f_A(z) = \frac{az+b}{cz+d},\qquad
	f_B(z) = \frac{\alpha z+\beta}{\gamma z+\delta},
	\]
	then for all $z\in\widehat{\mathbb{C}}$ for which both sides are defined we
	have
	\[
	(f_B\circ f_A)(z) = f_{BA}(z),
	\]
	where $BA$ is the ordinary product of matrices.
	
	However, the matrix representation is not unique: multiplying the matrix by
	a nonzero scalar does not change the Möbius transformation.  Indeed, if
	$\lambda\in\mathbb{C}^{\times}$, then
	\[
	f_{\lambda A}(z)
	= \frac{(\lambda a)z + \lambda b}{(\lambda c)z + \lambda d}
	= \frac{az+b}{cz+d}
	= f_{A}(z).
	\]
	Thus the same Möbius map corresponds to the whole 1--dimensional family of
	matrices $\lambda A$, $\lambda\neq 0$.
	
	We remove this redundancy by working projectively.  First, note that we may
	restrict attention to matrices with determinant $1$:
	if $A\in\mathrm{GL}(2,\mathbb{C})$ has $\det A\neq 0$, then we may rescale
	$A$ to $\lambda A$ with $\lambda^2\det A = 1$, so that $\lambda A\in
	\mathrm{SL}(2,\mathbb{C})$.  This rescaling does not change $f_A$, so every
	Möbius transformation is represented by \emph{some} matrix in
	$\mathrm{SL}(2,\mathbb{C})$.
	
	Moreover, within $\mathrm{SL}(2,\mathbb{C})$ the only nontrivial scalar
	multiple that preserves determinant $1$ is $-I$, since if
	$\lambda A\in\mathrm{SL}(2,\mathbb{C})$ with $A\in\mathrm{SL}(2,\mathbb{C})$,
	then
	\[
	1 = \det(\lambda A) = \lambda^{2}\det A = \lambda^2,
	\]
	so $\lambda = \pm 1$.  Thus the scalar redundancy inside
	$\mathrm{SL}(2,\mathbb{C})$ is precisely the central subgroup
	$\{\pm I\}$.
	
	We therefore define the \emph{projective special linear group}
	\[
	\mathrm{PSL}(2,\mathbb{C})
	:= \mathrm{SL}(2,\mathbb{C}) / \{\pm I\}.
	\]
	An element of $\mathrm{PSL}(2,\mathbb{C})$ is an equivalence class $[A]$ of
	matrices $A\in\mathrm{SL}(2,\mathbb{C})$, where $A\sim -A$.
	
	The assignment
	\[
	[A] \longmapsto f_A
	\]
	is a well-defined group homomorphism
	\[
	\mathrm{PSL}(2,\mathbb{C})
	\longrightarrow
	\{\text{Möbius transformations on }\widehat{\mathbb{C}}\},
	\]
	which is bijective.  Hence we have a natural group isomorphism
	\[
	\mathrm{PSL}(2,\mathbb{C})
	\;\cong\;
	\{\text{all Möbius transformations on }\widehat{\mathbb{C}}\}.
	\]
	This is the precise sense in which ``the Möbius group is
	$\mathrm{PSL}(2,\mathbb{C})$''.
	
	From the complex--geometric viewpoint, $\widehat{\mathbb{C}}$ is the complex
	projective line $\mathbb{CP}^{1}$ and $\mathrm{PSL}(2,\mathbb{C})$ is the
	group of holomorphic automorphisms of $\mathbb{CP}^{1}$.
	
	\subsubsection*{4. Generators of the Möbius group}
	
	It is a standard fact that the Möbius group is generated by the following
	simple maps:
	\begin{align*}
		T_b(z) &= z + b, && b\in\mathbb{C} \quad\text{(translations)},\\
		D_a(z) &= a z,   && a\in\mathbb{C}^{\times}\quad\text{(dilations/rotations)},\\
		I(z)   &= \frac{1}{z}                     &&\text{(inversion)}.
	\end{align*}
	More precisely, any Möbius transformation $f(z)=\dfrac{az+b}{cz+d}$ with
	$ad-bc\neq 0$ can be written as a finite composition of maps of the form
	$T_b$, $D_a$, and $I$.  For example, when $c\neq 0$ one checks that
	\[
	\frac{az+b}{cz+d} =
	D_{a/c}\circ T_{d/c}\circ I\circ T_{b/a}(z),
	\]
	and similar decompositions hold in all cases.  Thus, to understand the
	smoothness properties of general Möbius maps, it suffices to understand
	these generators.
	
	\subsubsection*{5. Induced maps on the sphere via stereographic projection}
	
	Given any Möbius transformation
	$f:\widehat{\mathbb{C}}\to\widehat{\mathbb{C}}$, we obtain a map
	\[
	F := \Phi^{-1}\circ f\circ\Phi : S^{2} \longrightarrow S^{2}.
	\]
	The exercise asks us to show that $F$ is a \emph{diffeomorphism} of the
	smooth surface $S^{2}$.  Since $\Phi$ and $\Phi^{-1}$ are already smooth
	diffeomorphisms, it is enough to check that $f$ is smooth with smooth
	inverse in the atlas on $\widehat{\mathbb{C}}$ coming from stereographic
	coordinates.
	
	The important observation is that smoothness of $F$ can be verified in
	local coordinates.  On the chart
	\[
	U = S^{2}\setminus\{N\},\qquad \Phi:U\to\mathbb{C},
	\]
	the coordinate expression of $F$ is
	\[
	\Phi\circ F\circ\Phi^{-1} = f:\mathbb{C}\to\mathbb{C},
	\]
	which is clearly smooth wherever it is defined (it is a rational function
	with denominator nonzero).  Near the point corresponding to $\infty$ we
	switch to another stereographic chart; in those coordinates the expression
	for $f$ is again rational, hence smooth, and the chart transition maps
	between the two stereographic charts are themselves Möbius transformations
	(and therefore smooth).  This shows that $F$ is smooth in the sense of
	smooth manifolds.
	
	Because Möbius transformations form a group under composition and inversion,
	$f$ has a Möbius inverse $f^{-1}$, and therefore
	\[
	F^{-1} = \Phi^{-1}\circ f^{-1}\circ \Phi : S^{2}\to S^{2}
	\]
	is also smooth.  Consequently $F$ is a diffeomorphism.
	
	To make this completely concrete it is often convenient to verify the claim
	first for the generators $T_b$, $D_a$ and $I$, and then conclude for all
	Möbius maps by composition.  We briefly spell this out.
	
	\paragraph{Translations and dilations/rotations.}
	Let $f(z)=az+b$ with $a\neq 0$.  Then on the chart $U$,
	\[
	\Phi\circ F\circ\Phi^{-1}(z) = f(z) = az+b,
	\]
	which is entire and hence smooth.  On the overlap with the second
	stereographic chart (from the south pole) the coordinate expression of $F$
	is the composition of $f$ with the smooth chart transition maps; it is
	therefore smooth as well.  The inverse map is represented by the affine map
	$f^{-1}(z) = a^{-1}(z-b)$, so $F$ is a diffeomorphism.
	
	\paragraph{Inversion.}
	Let $f(z) = 1/z$, extended by $f(0)=\infty$, $f(\infty)=0$.  On
	$\mathbb{C}^{\times}$ this is smooth.  In the stereographic chart $U$ away
	from the point $\Phi^{-1}(0)$ (the point corresponding to $z=0$) the map
	\[
	\Phi\circ F\circ\Phi^{-1}(z)=\frac{1}{z}
	\]
	is smooth.  Near $z=\infty$ we switch to the other stereographic chart,
	where the coordinate $w=1/z$ is finite; in that chart $f$ is simply
	$w\mapsto w$, so again we have a smooth expression.  The inverse of $f$ is
	itself ($f^{-1}=f$), so $F$ is a diffeomorphism of $S^{2}$.
	
	\medskip
	
	Since translation, dilation/rotation, and inversion each induce smooth
	diffeomorphisms of $S^{2}$ via
	\[
	F = \Phi^{-1}\circ f\circ\Phi,
	\]
	and every Möbius transformation is a finite composition of these generators,
	we conclude:
	
	\begin{quote}
		For every $[A]\in\mathrm{PSL}(2,\mathbb{C})$ the corresponding Möbius
		transformation $f_A$ determines a diffeomorphism
		\[
		F_A = \Phi^{-1}\circ f_A\circ\Phi : S^{2}\to S^{2}.
		\]
		The assignment $[A]\mapsto F_A$ defines a smooth action of
		$\mathrm{PSL}(2,\mathbb{C})$ on the abstract smooth surface $S^{2}$ by
		diffeomorphisms.
	\end{quote}
	
	Thus, identifying $\widehat{\mathbb{C}}$ with $S^{2}$ via stereographic
	projection, the Möbius group $\mathrm{PSL}(2,\mathbb{C})$ acts on $S^{2}$
	smoothly and by diffeomorphisms, as required in part (b) of the problem.
	
	
	\subsection*{Problem 8: Exotic metrics on the disc and on the square}
	
	\paragraph{(a) Statement.}
	On the open unit disc
	\[
	B(0,1) = \{(u,v)\in\mathbb{R}^2 : u^2+v^2<1\}
	\]
	define an abstract Riemannian metric by
	\[
	g = \frac{du^2+dv^2}{1-u^2-v^2}.
	\]
	Prove directly that diameter segments (straight line segments through the origin)
	are length–minimizing curves between their endpoints. Show that distances with
	respect to this metric are bounded, but areas can be unbounded.
	
	\paragraph{(a) Solution.}
	
	In polar coordinates $u=r\cos\theta$, $v=r\sin\theta$ we have
	\[
	du^2+dv^2 = dr^2 + r^2d\theta^2,\qquad 1-u^2-v^2 = 1-r^2,
	\]
	so
	\[
	g = \frac{dr^2+r^2d\theta^2}{1-r^2}.
	\]
	
	\medskip\noindent
	\emph{Diameters are length–minimizing.}
	Let $\gamma(t)=(r(t),\theta(t))$ be a $C^1$ curve in $B(0,1)$.
	Its speed is
	\[
	|\gamma'(t)|_g
	= \sqrt{\frac{r'(t)^2 + r(t)^2\theta'(t)^2}{1-r(t)^2}}
	\ge \frac{|r'(t)|}{\sqrt{1-r(t)^2}}.
	\]
	Hence
	\[
	L_g(\gamma)
	= \int |\gamma'(t)|_g\,dt
	\ge \int \frac{|r'(t)|}{\sqrt{1-r(t)^2}}\,dt.
	\]
	
	Suppose $\gamma$ joins two points on a fixed diameter, i.e.\ with polar
	coordinates $(r_1,\theta_0)$ and $(r_2,\theta_0)$. Reparametrising, we may
	assume that $r(t)$ is monotone. Then
	\[
	\int \frac{|r'(t)|}{\sqrt{1-r(t)^2}}\,dt
	= \int_{r_1}^{r_2} \frac{dr}{\sqrt{1-r^2}}
	= \bigl|\arcsin r_2 - \arcsin r_1\bigr|.
	\]
	
	Let $\delta$ be the diameter segment
	$\delta(t)=(r(t),\theta_0)$ joining these points. For $\delta$ we have
	$\theta'(t)\equiv 0$, so
	\[
	L_g(\delta)
	= \int \sqrt{\frac{r'(t)^2}{1-r(t)^2}}\,dt
	= \int_{r_1}^{r_2} \frac{dr}{\sqrt{1-r^2}}
	= \bigl|\arcsin r_2 - \arcsin r_1\bigr|.
	\]
	Therefore $L_g(\gamma)\ge L_g(\delta)$, with equality if and only if
	$\theta'(t)\equiv 0$, i.e.\ if $\gamma$ is contained in the same diameter.
	Hence diameter segments are length–minimizing curves between their endpoints.
	
	\medskip\noindent
	\emph{Distances are bounded.}
	For a point of radius $r$ the distance from the origin along a radius is
	\[
	d_g\bigl(0,(r,\theta)\bigr)
	= \int_0^r \frac{ds}{\sqrt{1-s^2}}
	= \arcsin r.
	\]
	Thus for any $p,q\in B(0,1)$ with radii $r_p,r_q$ we have
	\[
	d_g(p,q)
	\le d_g(p,0)+d_g(0,q)
	= \arcsin r_p + \arcsin r_q < \pi.
	\]
	So the metric space $(B(0,1),d_g)$ has finite diameter.
	
	\medskip\noindent
	\emph{Areas can be unbounded.}
	The metric is conformal to the Euclidean metric:
	$g = \lambda(du^2+dv^2)$ with
	\[
	\lambda(u,v) = \frac{1}{1-u^2-v^2}.
	\]
	Hence the area element is
	\[
	dA_g = \lambda\,du\,dv = \frac{du\,dv}{1-u^2-v^2}.
	\]
	
	In polar coordinates the area of the Euclidean disc of radius $r$ is
	\[
	\operatorname{Area}_g(B(0,r))
	= \int_0^{2\pi} \int_0^r \frac{\rho}{1-\rho^2}\,d\rho\,d\theta
	= 2\pi \int_0^r \frac{\rho}{1-\rho^2}\,d\rho.
	\]
	With $t=\rho^2$,
	\[
	\operatorname{Area}_g(B(0,r))
	= \pi\int_0^{r^2} \frac{dt}{1-t}
	= -\pi\ln(1-r^2).
	\]
	As $r\to 1^-$, the quantity $-\ln(1-r^2)$ tends to $+\infty$.
	Thus the areas of such discs become arbitrarily large, so areas
	in this metric are unbounded.
	
	\bigskip
	\paragraph{(b) Statement.}
	Let $V=\{(u,v)\in\mathbb{R}^2: |u|<1,\ |v|<1\}$ and define
	\[
	g_1 = \frac{du^2}{(1-u^2)^2} + \frac{dv^2}{(1-v^2)^2},
	\qquad
	g_2 = \frac{du^2}{(1-v^2)^2} + \frac{dv^2}{(1-u^2)^2}.
	\]
	
	\begin{enumerate}
		\item A map $\gamma:[0,\infty)\to V$ is a \emph{properly embedded path} if, for every
		compact $K\subset V$, the preimage $\gamma^{-1}(K)$ is compact (equivalently,
		for every compact $K$ there is $T$ such that $\gamma(t)\notin K$ for all
		$t\ge T$). Show that any homeomorphism $h:V\to V$
		sends properly embedded paths to properly embedded paths.
		\item Prove that the Riemannian surfaces $(V,g_1)$ and $(V,g_2)$ are not isometric,
		but that there is an area–preserving diffeomorphism between them.
		
		\emph{Hint:} Show that exactly one of the two metrics admits a properly embedded
		path of finite length.
	\end{enumerate}
	
	\paragraph{(b)(i) Solution.}
	Let $\gamma:[0,\infty)\to V$ be a properly embedded path and $h:V\to V$ a homeomorphism.
	For any compact $K\subset V$, $h^{-1}(K)$ is compact because $h$ is a homeomorphism.
	Properness of $\gamma$ implies that $\gamma^{-1}(h^{-1}(K))$ is compact in $[0,\infty)$.
	But
	\[
	(h\circ\gamma)^{-1}(K) = \gamma^{-1}\bigl(h^{-1}(K)\bigr),
	\]
	so $(h\circ\gamma)^{-1}(K)$ is compact. Hence $h\circ\gamma$ is properly embedded.
	
	\paragraph{(b)(ii) Solution.}
	
	\emph{Area–preserving diffeomorphism.}
	For the metric $g_1$ we have
	\[
	E_1(u,v) = \frac{1}{(1-u^2)^2}, \qquad
	G_1(u,v) = \frac{1}{(1-v^2)^2},
	\]
	so its area element is
	\[
	dA_{g_1} = \sqrt{E_1 G_1}\,du\,dv
	= \frac{du\,dv}{(1-u^2)(1-v^2)}.
	\]
	
	For $g_2$,
	\[
	E_2(u,v) = \frac{1}{(1-v^2)^2}, \qquad
	G_2(u,v) = \frac{1}{(1-u^2)^2},
	\]
	whence
	\[
	dA_{g_2} = \sqrt{E_2 G_2}\,du\,dv
	= \frac{du\,dv}{(1-u^2)(1-v^2)}.
	\]
	Thus $dA_{g_1}=dA_{g_2}$ pointwise on $V$, and the identity map
	$\mathrm{id}_V:(V,g_1)\to(V,g_2)$ is a diffeomorphism preserving areas.
	
	\medskip\noindent
	\emph{Properly embedded paths and finite length.}
	
	For $g_2$ consider the path
	\[
	\gamma_2(t) = (\tanh t,0),\qquad t\in[0,\infty).
	\]
	We have $|\tanh t|<1$ and $\tanh t\to 1$ as $t\to\infty$, so $\gamma_2$ tends to
	the boundary point $(1,0)$ and is properly embedded in $V$.
	Along $\gamma_2$, $v\equiv 0$ and $du=\tanh'(t)\,dt=\operatorname{sech}^2 t\,dt$,
	so for
	\[
	g_2 = \frac{du^2}{(1-v^2)^2} + \frac{dv^2}{(1-u^2)^2}
	\]
	we obtain $dv=0$ and $1-v^2=1$, hence $ds_{g_2}=|du|$ and
	\[
	L_{g_2}(\gamma_2)
	= \int_0^\infty |\tanh'(t)|\,dt
	= \int_0^\infty \operatorname{sech}^2 t\,dt
	= \bigl[\tanh t\bigr]_0^\infty = 1 <\infty.
	\]
	Thus $(V,g_2)$ admits a properly embedded path of finite length.
	
	Now take any properly embedded path
	$\gamma_1(t)=(u(t),v(t)):[0,\infty)\to V$ for $g_1$.
	Because $V$ is bounded and $\gamma_1$ is proper, its image must
	escape every compact subset, hence approach the boundary $\partial V$.
	Therefore at least one of the coordinates tends to $\pm 1$
	as $t\to\infty$; without loss of generality assume $u(t)\to 1$.
	
	The speed in the metric $g_1$ is
	\[
	|\gamma_1'(t)|_{g_1}
	= \sqrt{\frac{u'(t)^2}{(1-u(t)^2)^2}
		+\frac{v'(t)^2}{(1-v(t)^2)^2}}
	\ge \frac{|u'(t)|}{1-u(t)^2}.
	\]
	On any interval where $u$ is monotone increasing from $u_0$ to $u_1$ we have
	\[
	\int \frac{|u'(t)|}{1-u(t)^2}\,dt
	= \int_{u_0}^{u_1} \frac{du}{1-u^2}.
	\]
	As $u_1\to 1$, this integral diverges to $+\infty$, so by choosing $t$ large
	we can force the partial length of $\gamma_1$ to be arbitrarily large.
	Hence
	\[
	L_{g_1}(\gamma_1)=\int_0^\infty |\gamma_1'(t)|_{g_1}\,dt = \infty.
	\]
	Thus every properly embedded path in $(V,g_1)$ has infinite length.
	
	\medskip\noindent
	\emph{Non–isometry.}
	Suppose there were an isometry $\Phi:(V,g_1)\to(V,g_2)$.
	By part~(i), $\Phi$ would carry properly embedded paths to properly embedded
	paths, and by definition of isometry it would preserve their lengths.
	But $(V,g_2)$ admits a properly embedded path of finite length,
	while no such path exists in $(V,g_1)$.
	This is a contradiction. Therefore $(V,g_1)$ and $(V,g_2)$ are not isometric.
	
	\qedhere
	
	
	\subsection*{Visualizing the metrics of Problem 8}
	
	In this subsection we illustrate the abstract Riemannian metrics from
	Problem~8 by plotting their most important scale factors.  This helps to see
	geometrically why distances can remain bounded while areas become unbounded,
	and how the two metrics on the square $V$ are related by an isometry.
	
	\subsubsection*{(a) The metric on the disc}
	
	On the open unit disc $B(0,1)\subset\mathbb{R}^2$ we are given the metric
	\[
	g_{\mathbb{D}}
	= \frac{du^2 + dv^2}{1-u^2-v^2}.
	\]
	It is conformal to the Euclidean metric: the length element is
	\[
	ds = \lambda(u,v)\,\sqrt{du^2+dv^2},
	\qquad
	\lambda(u,v) = \frac{1}{\sqrt{1-u^2-v^2}},
	\]
	and the area element is
	\[
	dA = A(u,v)\,du\,dv,
	\qquad
	A(u,v) = \frac{1}{1-u^2-v^2}.
	\]
	The factor $\lambda$ tells us how much \emph{lengths} are stretched, while
	$A$ tells us how much \emph{areas} are stretched, compared to the standard
	Euclidean structure on the disc.
	
	Notice that as $u^2+v^2\to 1$ (approaching the boundary circle), both
	$\lambda$ and $A$ blow up; however, their behaviour in integrals is
	different: the radial integral of $\lambda$ from $0$ to $1$ converges
	(finite distances to the boundary), whereas the radial integral of $A$
	diverges (infinite area near the boundary).
	
	\begin{figure}[h]
		\centering
		\includegraphics[width=0.45\textwidth]{metric_disc_length.png}
		\caption{Unit disc with the conformal factor for lengths
			$\lambda(u,v) = 1/\sqrt{1-u^2-v^2}$.
			Colours indicate how distances are stretched; the factor blows up
			near the boundary $u^2+v^2=1$, but geodesic diameters remain finite.}
	\end{figure}
	
	In the picture above, the colour intensity increases as one approaches the
	circle $u^2+v^2=1$, reflecting that any small Euclidean segment near the
	boundary is scaled by a large factor in the Riemannian metric.  Radial
	segments towards the boundary are still of finite total length because the
	integral of $\lambda$ along a radius converges.
	
	\begin{figure}[h]
		\centering
		\includegraphics[width=0.45\textwidth]{metric_disc_area.png}
		\caption{Unit disc with the area factor
			$A(u,v) = 1/(1-u^2-v^2)$ (clipped in the picture).
			This diverges near the boundary, illustrating why areas can be
			unbounded even though distances are bounded.}
	\end{figure}
	
	The second picture shows the area factor $A(u,v)$.  The extremely bright
	ring near the boundary represents the region where the metric inflates area
	so strongly that the total area of the disc (in this metric) becomes
	infinite, even though the metric diameter of the disc is finite.
	
	\subsubsection*{(b) The two metrics on the square $V$}
	
	Let $V=\{(u,v)\in\mathbb{R}^2 : |u|<1,\ |v|<1\}$ and define
	\[
	g^{(1)} = \frac{du^2}{(1-u^2)^2} + \frac{dv^2}{(1-v^2)^2},\qquad
	g^{(2)} = \frac{du^2}{(1-v^2)^2} + \frac{dv^2}{(1-u^2)^2}.
	\]
	Both metrics are \emph{product-type}: the $u$– and $v$–directions stay
	orthogonal, but the scale factors in these directions depend separately on
	$u$ and $v$.  Each factor $(1-u^2)^{-2}$ or $(1-v^2)^{-2}$ blows up as one
	approaches the corresponding boundary line $u=\pm1$ or $v=\pm1$.  Thus the
	boundary of the square is ``pushed to infinity'' in the associated metric.
	
	For $g^{(1)}$ we have
	\[
	g^{(1)}_{11} = \frac{1}{(1-u^2)^2}, \qquad
	g^{(1)}_{22} = \frac{1}{(1-v^2)^2},
	\]
	so the difference $g^{(1)}_{11}-g^{(1)}_{22}$ measures the anisotropy between
	the horizontal and vertical directions.  Near $u=\pm1$ the horizontal
	direction becomes extremely costly, while near $v=\pm1$ it is the vertical
	direction that is strongly stretched.
	
	\begin{figure}[h]
		\centering
		\includegraphics[width=0.45\textwidth]{metric_square_g1_aniso.png}
		\caption{Open square $V=\{|u|<1,|v|<1\}$ with metric
			$g^{(1)} = \dfrac{du^2}{(1-u^2)^2} + \dfrac{dv^2}{(1-v^2)^2}$.
			The colours show the anisotropy $g^{(1)}_{11}-g^{(1)}_{22}$:
			near $u=\pm1$ the $u$--direction is strongly stretched, near
			$v=\pm1$ the $v$--direction dominates.}
	\end{figure}
	
	For $g^{(2)}$ the roles of $u$ and $v$ are swapped:
	\[
	g^{(2)}_{11} = \frac{1}{(1-v^2)^2}, \qquad
	g^{(2)}_{22} = \frac{1}{(1-u^2)^2}.
	\]
	Consequently, the anisotropy pattern $g^{(2)}_{11}-g^{(2)}_{22}$ is the
	mirror image of the previous one under the swap $(u,v)\mapsto(v,u)$.
	
	\begin{figure}[h]
		\centering
		\includegraphics[width=0.45\textwidth]{metric_square_g2_aniso.png}
		\caption{The same domain with metric
			$g^{(2)} = \dfrac{du^2}{(1-v^2)^2} + \dfrac{dv^2}{(1-u^2)^2}$.
			The anisotropy $g^{(2)}_{11}-g^{(2)}_{22}$ is essentially the
			$u\leftrightarrow v$ swap of the previous figure, reflecting that
			the map $(u,v)\mapsto(v,u)$ is an isometry between $g^{(1)}$ and $g^{(2)}$.}
	\end{figure}
	
	Indeed, the coordinate switch
	\[
	\Phi:V\to V,\qquad \Phi(u,v)=(v,u),
	\]
	sends $g^{(1)}$ to $g^{(2)}$, so it defines an isometry between the two
	abstract Riemannian surfaces.  The last two charts make this visually clear:
	the colour patterns are identical up to the interchange of the horizontal and
	vertical axes.

	\subsection*{Examples for (ii): non--isometry and an area--preserving map}
	
	We now illustrate statement (ii) for the two Riemannian metrics on
	\[
	V = \{(u,v)\in\mathbb{R}^2 : |u|<1,\ |v|<1\},
	\]
	namely
	\[
	g^{(1)} = \frac{du^2}{(1-u^2)^2} + \frac{dv^2}{(1-v^2)^2},
	\qquad
	g^{(2)} = \frac{du^2}{(1-v^2)^2} + \frac{dv^2}{(1-u^2)^2}.
	\]
	The pictures above for $g^{(1)}$ and $g^{(2)}$ show how each metric stretches
	the horizontal and vertical directions differently near the boundary
	$|u|=1$ and $|v|=1$.
	
	\subsubsection*{Properly embedded paths and non--isometry}
	
	Recall that a \emph{properly embedded path} in $V$ is a map
	\[
	\gamma : [0,\infty) \to V
	\]
	such that the preimage of every compact set $K\subset V$ is compact
	(i.e.\ $\gamma^{-1}(K)$ is closed and bounded in $[0,\infty)$).  Since $V$
	is a bounded open subset of $\mathbb{R}^2$, this means that a properly
	embedded path must ``run off'' towards the boundary of $V$ as
	$t\to\infty$; it cannot accumulate in the interior.
	
	\paragraph{The first metric $g^{(1)}$: no finite--length proper rays.}
	For $g^{(1)}$ we can separate the $u$ and $v$ directions by the change of
	variables
	\[
	x = \operatorname{arctanh}(u), \qquad
	y = \operatorname{arctanh}(v).
	\]
	Then
	\[
	dx = \frac{du}{1-u^2},\qquad
	dy = \frac{dv}{1-v^2},
	\]
	so
	\[
	g^{(1)} = \frac{du^2}{(1-u^2)^2} + \frac{dv^2}{(1-v^2)^2}
	= dx^2 + dy^2.
	\]
	Thus $(V,g^{(1)})$ is isometric to the Euclidean plane
	$(\mathbb{R}^2,dx^2+dy^2)$ via the diffeomorphism
	$\Phi(u,v) = (\operatorname{arctanh}u,\operatorname{arctanh}v)$.
	
	In a complete Riemannian manifold such as $\mathbb{R}^2$ with the Euclidean
	metric, any curve of \emph{finite} length has a limit point: if
	$\gamma:[0,\infty)\to\mathbb{R}^2$ has finite length $L$, then it is a
	Cauchy curve and converges to some point in $\mathbb{R}^2$.  Its image is
	contained in a closed ball and hence in a compact set.  The preimage of this
	compact set is all of $[0,\infty)$, which is not compact, so $\gamma$ cannot
	be proper.
	
	Therefore $(\mathbb{R}^2,dx^2+dy^2)$ (and hence $(V,g^{(1)})$) does not
	contain any properly embedded path of finite length.  This is consistent
	with the chart for $g^{(1)}$: both horizontal and vertical directions are
	stretched so strongly near $|u|=1$ and $|v|=1$ that the boundary lies at
	infinite distance.
	
	\paragraph{The second metric $g^{(2)}$: a finite--length proper ray.}
	In contrast, $(V,g^{(2)})$ does admit a properly embedded path of finite
	length.  Consider the vertical path
	\[
	\gamma(t) = (u(t),0),
	\qquad
	u(t) = 1 - e^{-t},
	\qquad t\in[0,\infty).
	\]
	Then $\gamma(t)\in V$ for all $t$ and $u(t)\nearrow 1$ as $t\to\infty$, so
	the path runs out towards the boundary line $u=1$.
	
	Along this path we have $v(t)\equiv 0$ and $v'(t) = 0$, so the metric
	$g^{(2)}$ restricts to
	\[
	ds^2
	= \frac{du^2}{(1-v^2)^2}
	= \frac{du^2}{(1-0)^2}
	= du^2.
	\]
	Since $u'(t) = e^{-t}$, the speed of $\gamma$ is
	\[
	\|\gamma'(t)\|_{g^{(2)}} = |u'(t)| = e^{-t},
	\]
	and its total length is
	\[
	L_{g^{(2)}}(\gamma)
	= \int_0^\infty e^{-t}\,dt
	= 1 < \infty.
	\]
	
	To see that $\gamma$ is properly embedded, let $K\subset V$ be any compact
	set.  There exists $\delta>0$ such that
	\[
	K \subset \{(u,v)\in V : |u|\le 1-\delta,\ |v|\le 1-\delta\}.
	\]
	Choose $T>0$ with $u(T) > 1-\delta$; then $u(t)>1-\delta$ for all $t\ge T$,
	so $\gamma(t)\notin K$ for $t\ge T$.  Thus
	$\gamma^{-1}(K)\subset[0,T]$, which is compact.  Hence $\gamma$ is a
	properly embedded path of finite length.
	
	By part (i), any homeomorphism $V\to V$ (in particular, any isometry)
	must map properly embedded paths to properly embedded paths.  Since
	$(V,g^{(1)})$ has no finite--length proper path but $(V,g^{(2)})$ does,
	the two Riemannian surfaces cannot be isometric.
	
	The charts for $g^{(2)}$ above illustrate this behaviour: along the vertical
	line $v=0$ the coefficient in front of $du^2$ is $1/(1-v^2)^2=1$, so the
	boundary $u=1$ is at finite distance, whereas for $g^{(1)}$ the same
	vertical direction is heavily stretched near $u=\pm1$, placing the boundary
	at infinite distance.
	
	\subsubsection*{Area forms and an area--preserving diffeomorphism}
	
	Although the metrics are not isometric, they induce the \emph{same} area
	form on $V$.  For a diagonal metric
	\[
	g = A(u,v)\,du^2 + B(u,v)\,dv^2,
	\]
	the area element is
	\[
	dA_g = \sqrt{\det g}\,du\,dv
	= \sqrt{A(u,v)B(u,v)}\,du\,dv.
	\]
	
	For $g^{(1)}$ we have
	\[
	A_1(u,v) = \frac{1}{(1-u^2)^2},\qquad
	B_1(u,v) = \frac{1}{(1-v^2)^2},
	\]
	so
	\[
	dA_{g^{(1)}} = \frac{1}{(1-u^2)(1-v^2)}\,du\,dv.
	\]
	
	For $g^{(2)}$,
	\[
	A_2(u,v) = \frac{1}{(1-v^2)^2},\qquad
	B_2(u,v) = \frac{1}{(1-u^2)^2},
	\]
	whence
	\[
	dA_{g^{(2)}}
	= \sqrt{A_2 B_2}\,du\,dv
	= \frac{1}{(1-u^2)(1-v^2)}\,du\,dv
	= dA_{g^{(1)}}.
	\]
	
	Thus the identity map
	\[
	\mathrm{id} : (V,g^{(1)}) \longrightarrow (V,g^{(2)})
	\]
	is an \emph{area--preserving diffeomorphism}: it is a smooth bijection with
	smooth inverse, and it pulls back the area form of $g^{(2)}$ to that of
	$g^{(1)}$.
	
	In summary, the two metrics on $V$ have the same area density (as reflected
	in the similar colour patterns in the charts), but differ in their
	large–scale geometry: one is complete, the other admits a properly embedded
	path of finite length.  This shows that area preservation is a much weaker
	condition than Riemannian isometry.
	
	\subsection*{Diameters and bounded distances in the disc metric}
	
	On the open unit disc $B(0,1)\subset\mathbb{R}^2$ we consider the abstract
	Riemannian metric
	\[
	g = \frac{du^2 + dv^2}{1-u^2-v^2}.
	\]
	Introducing polar coordinates $u = r\cos\theta$, $v = r\sin\theta$ with
	$0\le r<1$, $0\le\theta<2\pi$, we obtain
	\[
	du^2 + dv^2 = dr^2 + r^2 d\theta^2,
	\]
	so
	\[
	ds^2
	= \frac{dr^2 + r^2 d\theta^2}{1-r^2}.
	\]
	
	\paragraph{Diameters are length--minimizing.}
	Let $p$ and $q$ be two points lying on the same diameter; in polar
	coordinates we may assume
	\[
	p = (r_1,\theta_0),\qquad q = (r_2,\theta_0),\qquad 0\le r_1<r_2<1.
	\]
	Consider any smooth curve
	\[
	\gamma(t) = \bigl(r(t),\theta(t)\bigr),\qquad t\in[a,b],
	\]
	with $\gamma(a)=p$, $\gamma(b)=q$.
	Its speed with respect to $g$ is
	\[
	\|\gamma'(t)\|_g
	= \sqrt{\frac{r'(t)^2 + r(t)^2 \theta'(t)^2}{1-r(t)^2}}
	\;\ge\; \frac{|r'(t)|}{\sqrt{1-r(t)^2}},
	\]
	because $r(t)^2\theta'(t)^2\ge 0$.
	
	Therefore the length of $\gamma$ satisfies
	\[
	L_g(\gamma)
	= \int_a^b \|\gamma'(t)\|_g\,dt
	\;\ge\; \int_a^b \frac{|r'(t)|}{\sqrt{1-r(t)^2}}\,dt.
	\]
	If $\gamma$ is monotone in $r$ (which we may assume after reparametrising),
	then $r$ runs from $r_1$ to $r_2$, and we obtain
	\[
	L_g(\gamma)
	\;\ge\; \int_{r_1}^{r_2} \frac{dr}{\sqrt{1-r^2}}.
	\]
	
	Now consider the \emph{radial segment} along the diameter,
	\[
	\gamma_0(t) = (r(t),\theta(t)) = (t,\theta_0),\qquad t\in[r_1,r_2].
	\]
	For this curve we have $\theta'(t)\equiv 0$, hence
	\[
	\|\gamma_0'(t)\|_g
	= \sqrt{\frac{1}{1-t^2}}
	= \frac{1}{\sqrt{1-t^2}},
	\]
	and its length is
	\[
	L_g(\gamma_0)
	= \int_{r_1}^{r_2} \frac{dt}{\sqrt{1-t^2}}
	= \arcsin(r_2) - \arcsin(r_1).
	\]
	We see that $\gamma_0$ attains equality in the inequality above, so
	\emph{among all curves joining $p$ and $q$ the radial segment is of minimal
		length}.  Thus diameters (radial segments at fixed angle) are
	length--minimizing geodesics.
	
	\paragraph{Bounded distances.}
	In particular, the distance from the centre to a boundary point is
	\[
	d_g\bigl((0,0),(1,\theta_0)\bigr)
	= \int_0^1 \frac{dr}{\sqrt{1-r^2}}
	= \Bigl[\arcsin(r)\Bigr]_{0}^{1}
	= \frac{\pi}{2}.
	\]
	So every point of the disc lies at distance at most $\pi/2$ from the origin.
	
	For any two points $p,q\in B(0,1)$, the triangle inequality gives
	\[
	d_g(p,q)
	\le d_g(p,0) + d_g(0,q)
	\le \frac{\pi}{2} + \frac{\pi}{2}
	= \pi.
	\]
	Hence the metric diameter of $(B(0,1),g)$ is at most $\pi$.  In fact, two
	opposite boundary points on a diameter (for instance $(r=1,\theta)$ and
	$(r=1,\theta+\pi)$) are joined by a length--minimizing path consisting of
	two radial segments, and the total length of this diameter is
	\[
	L_{\text{diameter}}
	= 2\cdot d_g(0,\partial B(0,1))
	= 2\cdot\frac{\pi}{2}
	= \pi.
	\]
	
	\paragraph{Visual interpretation.}
	The conformal factor
	\[
	\lambda(r) = \frac{1}{\sqrt{1-r^2}}
	\]
	blows up as $r\to 1$, so small Euclidean segments near the boundary are
	greatly stretched in the Riemannian metric.  Nevertheless, the integral
	\(\int_0^1 \lambda(r)\,dr\) converges, so the boundary circle is at finite
	distance.  This is reflected in the chart of the disc metric (see
	Figure~\ref{fig:disc-metric-length}), where colours become more intense
	near the boundary but radial geodesics still have finite total length.

			$\lambda(u,v) = 1/\sqrt{1-u^2-v^2}$.
			Colours indicate how distances are stretched; the factor blows up
			near the boundary $u^2+v^2=1$, but radial diameters still have
			finite total length.}

	\subsection*{E1. Exponential cone and helicoid}
	
	Show that the surfaces $\Sigma$ and $\Sigma'$ in $\mathbb{R}^{3}$ defined as the
	images of
	\[
	\sigma(u,v) = (u\cos v,\;u\sin v,\;\ln u),
	\qquad
	\tau(u,v)   = (u\cos v,\;u\sin v,\;v),
	\]
	(where $u>0$ and $v>0$) have the same Gauss curvature, but are not locally
	isometric.
	
	\begin{figure}[h]
		\centering
		\includegraphics[width=0.45\textwidth]{exp_cone.png}
		\hfill
		\includegraphics[width=0.45\textwidth]{helicoid.png}
		\caption{The surfaces $\Sigma$ (left) and $\Sigma'$ (right): an `exponential
			cone' and a helicoid.}
	\end{figure}
	
	\paragraph{First fundamental forms.}
	
	For the exponential cone $\Sigma$,
	\[
	\sigma(u,v) = (u\cos v,\;u\sin v,\;\ln u),
	\]
	we have
	\[
	\sigma_u=(\cos v,\ \sin v,\ 1/u),\qquad
	\sigma_v=(-u\sin v,\ u\cos v,\ 0).
	\]
	Thus
	\[
	E_\Sigma = \langle\sigma_u,\sigma_u\rangle
	= 1+\frac{1}{u^2},\qquad
	F_\Sigma = \langle\sigma_u,\sigma_v\rangle = 0,\qquad
	G_\Sigma = \langle\sigma_v,\sigma_v\rangle = u^2,
	\]
	and the first fundamental form is
	\[
	I_\Sigma = \Bigl(1+\frac{1}{u^2}\Bigr)\,du^2 + u^2\,dv^2.
	\]
	
	For the helicoid $\Sigma'$,
	\[
	\tau(u,v) = (u\cos v,\;u\sin v,\;v),
	\]
	we obtain
	\[
	\tau_u=(\cos v,\ \sin v,\ 0),\qquad
	\tau_v=(-u\sin v,\ u\cos v,\ 1),
	\]
	so
	\[
	E_{\Sigma'} = 1,\qquad
	F_{\Sigma'} = 0,\qquad
	G_{\Sigma'} = u^2+1,
	\]
	and
	\[
	I_{\Sigma'} = du^2 + (u^2+1)\,dv^2.
	\]
	
	\paragraph{Gauss curvature.}
	
	For $\Sigma$ we compute
	\[
	\sigma_u\times\sigma_v = (-\cos v,\ -\sin v,\ u),
	\qquad
	\| \sigma_u\times\sigma_v \|^2 = u^2+1,
	\]
	so a unit normal is
	\[
	n = \frac{1}{\sqrt{u^2+1}}(-\cos v,\ -\sin v,\ u).
	\]
	Second derivatives are
	\[
	\sigma_{uu}=(0,0,-1/u^2),\quad
	\sigma_{uv}=(-\sin v,\cos v,0),\quad
	\sigma_{vv}=(-u\cos v,-u\sin v,0).
	\]
	Hence
	\[
	e_\Sigma = \langle\sigma_{uu},n\rangle
	= -\frac{1}{u\sqrt{u^2+1}},\quad
	f_\Sigma = \langle\sigma_{uv},n\rangle = 0,\quad
	g_\Sigma = \langle\sigma_{vv},n\rangle
	= \frac{u}{\sqrt{u^2+1}}.
	\]
	With $E_\Sigma,G_\Sigma$ as above and $F_\Sigma=0$,
	\[
	K_\Sigma
	= \frac{e_\Sigma g_\Sigma - f_\Sigma^{2}}{E_\Sigma G_\Sigma-F_\Sigma^{2}}
	= \frac{-\dfrac{1}{u\sqrt{u^2+1}}\cdot
		\dfrac{u}{\sqrt{u^2+1}}}{(1+u^{-2})u^2}
	= -\frac{1}{(1+u^2)^2}.
	\]
	
	For $\Sigma'$ we have
	\[
	\tau_u\times\tau_v = (\sin v,\ -\cos v,\ u),\qquad
	\|\tau_u\times\tau_v\|^2 = u^2+1,
	\]
	so a unit normal is
	\[
	n' = \frac{1}{\sqrt{u^2+1}}(\sin v,\ -\cos v,\ u).
	\]
	Second derivatives:
	\[
	\tau_{uu}=(0,0,0),\quad
	\tau_{uv}=(-\sin v,\cos v,0),\quad
	\tau_{vv}=(-u\cos v,-u\sin v,0).
	\]
	Thus
	\[
	e_{\Sigma'} = 0,\quad
	f_{\Sigma'} = \langle\tau_{uv},n'\rangle
	= -\frac{1}{\sqrt{u^2+1}},\quad
	g_{\Sigma'} = 0.
	\]
	With $E_{\Sigma'}=1$, $F_{\Sigma'}=0$, $G_{\Sigma'}=u^2+1$,
	\[
	K_{\Sigma'}
	= \frac{e_{\Sigma'}g_{\Sigma'}-f_{\Sigma'}^{2}}
	{E_{\Sigma'}G_{\Sigma'}-F_{\Sigma'}^{2}}
	= \frac{-\dfrac{1}{u^2+1}}{u^2+1}
	= -\frac{1}{(1+u^2)^2}.
	\]
	
	Hence
	\[
	K_\Sigma(u,v) = K_{\Sigma'}(u,v)
	= -\frac{1}{(1+u^2)^2},
	\]
	so the two surfaces have the same Gauss curvature.
	
	\paragraph{They are not locally isometric.}
	
	Suppose there exists a local isometry
	\(\Phi:\Sigma\supset U\to U'\subset\Sigma'\).
	Because an isometry preserves Gauss curvature and
	\[
	K(u) = -\frac{1}{(1+u^2)^2},
	\qquad u>0,
	\]
	is strictly increasing
	($K'(u)=4u/(1+u^2)^3>0$), points with different $u$ have different curvature.
	Thus $\Phi$ must preserve the coordinate $u$: if a point
	with coordinates $(u,v)$ maps to $(\tilde u,\tilde v)$, then
	$K(u)=K(\tilde u)$ implies $\tilde u=u$.
	
	Hence in $(u,v)$–coordinates we can write
	\[
	\Phi(u,v) = (u,\varphi(u,v)).
	\]
	Put $a=\varphi_u$ and $b=\varphi_v$. The Jacobian matrix is
	\[
	D\Phi =
	\begin{pmatrix}
		1 & 0\\[2pt]
		a & b
	\end{pmatrix}.
	\]
	
	The pullback of $I_{\Sigma'}$ under $\Phi$ has coefficients
	\[
	(\Phi^* I_{\Sigma'})_{uu} = 1 + (u^2+1)a^2,
	\]
	\[
	(\Phi^* I_{\Sigma'})_{uv} = (u^2+1)ab,
	\]
	\[
	(\Phi^* I_{\Sigma'})_{vv} = (u^2+1)b^2.
	\]
	These must equal $E_\Sigma, F_\Sigma, G_\Sigma$, i.e.
	\[
	1+\frac{1}{u^2} = 1 + (u^2+1)a^2,\qquad
	0 = (u^2+1)ab,\qquad
	u^2 = (u^2+1)b^2.
	\]
	
	From the middle equation, either $a=0$ or $b=0$.
	If $b=0$ then $u^2=0$, impossible for $u>0$.
	Thus $a=0$, and the first equation gives $1+1/u^2=1$, also impossible.
	Therefore such a local isometry $\Phi$ cannot exist.
	
	Consequently, the exponential cone $\Sigma$ and the helicoid $\Sigma'$ have the
	same Gauss curvature but are not locally isometric.
	
	\subsection*{E2. Moving frame and Gauss' theorema egregium}
	
	Consider a smooth surface $\Sigma\subset\mathbb{R}^3$ with Gauss curvature
	$\kappa$, and an allowable parametrization
	\[
	\sigma:V\to U\subset\Sigma
	\]
	with first fundamental form
	\[
	du^2 + G(u,v)\,dv^2.
	\]
	Let
	\[
	e=\sigma_u,\qquad
	f = \frac{\sigma_v}{\sqrt{G}},\qquad
	n = \frac{\sigma_u\times\sigma_v}{\|\sigma_u\times\sigma_v\|},
	\]
	so that $\langle e,f,n\rangle$ is an orthonormal basis of $\mathbb{R}^3$
	depending on $(u,v)\in V$ (a moving frame).
	
	\begin{enumerate}
		\item \emph{Show that $n=e\times f$.}
		
		Since
		\[
		\sigma_u\cdot\sigma_u=1,\quad
		\sigma_u\cdot\sigma_v=0,\quad
		\sigma_v\cdot\sigma_v=G,
		\]
		we have $|e|=1$, $|f|=1$, and $e\cdot f=0$.
		Moreover,
		\[
		\sigma_u\times\sigma_v
		= e\times(\sqrt{G}\,f)
		= \sqrt{G}\,(e\times f),
		\]
		and $\|\sigma_u\times\sigma_v\|=\sqrt{G}$, so
		\[
		n = \frac{\sigma_u\times\sigma_v}{\|\sigma_u\times\sigma_v\|}
		= e\times f.
		\]
		
		\item \emph{Frame derivative formulas.}
		
		The orthonormality relations
		\[
		e\cdot e = f\cdot f = n\cdot n = 1,\quad
		e\cdot f = e\cdot n = f\cdot n = 0
		\]
		imply, upon differentiating with respect to $u$ and $v$, that
		$e_u,f_u,e_v,f_v$ have no components along themselves. Thus there exist
		functions $\alpha,\beta,\lambda_i,\mu_j$ such that
		\[
		e_u = \alpha f + \lambda_1 n,\quad
		e_v = \beta f + \mu_1 n,\quad
		f_u = -\alpha e + \lambda_2 n,\quad
		f_v = -\beta e + \mu_2 n.
		\]
		
		\item \emph{Show $\alpha=0$ and $\beta=(\sqrt{G})_u$.}
		
		Since $e=\sigma_u$ and $f=\sigma_v/\sqrt{G}$,
		\[
		\alpha = e_u\cdot f
		= \sigma_{uu}\cdot\frac{\sigma_v}{\sqrt{G}}
		= \frac{1}{\sqrt{G}}(\sigma_{uu}\cdot\sigma_v).
		\]
		Differentiating $\sigma_u\cdot\sigma_v=0$ with respect to $u$ gives
		$\sigma_{uu}\cdot\sigma_v + \sigma_u\cdot\sigma_{uv}=0$.
		From $(\sigma_u\cdot\sigma_u)_v=0$ we get $\sigma_{uv}\cdot\sigma_u=0$,
		hence $\sigma_{uu}\cdot\sigma_v=0$ and therefore $\alpha=0$.
		
		Similarly,
		\[
		\beta = e_v\cdot f
		= \sigma_{uv}\cdot\frac{\sigma_v}{\sqrt{G}}
		= \frac{1}{\sqrt{G}}(\sigma_{uv}\cdot\sigma_v).
		\]
		Differentiating $\sigma_v\cdot\sigma_v = G$ with respect to $u$ yields
		\[
		G_u = 2\,\sigma_{uv}\cdot\sigma_v,
		\]
		so
		\[
		\beta = \frac{G_u}{2\sqrt{G}} = (\sqrt{G})_u.
		\]
		
		\item \emph{Show $\lambda_1\mu_2-\lambda_2\mu_1
			= e_u\cdot f_v - f_u\cdot e_v
			= -\beta_u
			= -(\sqrt{G})_{uu}$.}
		
		Using the frame expansions and orthonormality,
		\[
		e_u\cdot f_v
		= (\alpha f + \lambda_1 n)\cdot(-\beta e + \mu_2 n)
		= \lambda_1\mu_2,
		\]
		\[
		f_u\cdot e_v
		= (-\alpha e + \lambda_2 n)\cdot(\beta f + \mu_1 n)
		= \lambda_2\mu_1.
		\]
		Hence
		\[
		e_u\cdot f_v - f_u\cdot e_v
		= \lambda_1\mu_2-\lambda_2\mu_1.
		\]
		
		Since $\beta = e_v\cdot f$, we have
		\[
		\beta_u = (e_v\cdot f)_u
		= e_{vu}\cdot f + e_v\cdot f_u.
		\]
		Using the equality $e_{uv}=e_{vu}$ together with the frame formulas and
		collecting coefficients in the orthonormal basis yields
		\[
		\beta_u = \lambda_2\mu_1-\lambda_1\mu_2
		= -(\lambda_1\mu_2-\lambda_2\mu_1).
		\]
		Thus
		\[
		\lambda_1\mu_2-\lambda_2\mu_1
		= e_u\cdot f_v - f_u\cdot e_v
		= -\beta_u.
		\]
		Since $\beta=(\sqrt{G})_u$, it follows that
		\[
		-\beta_u = -(\sqrt{G})_{uu}.
		\]
		
		\item \emph{Use the Gauss map.}
		
		The Gauss map $n:\Sigma\to S^2$ satisfies
		\[
		Dn(\sigma_u) = n_u,\qquad Dn(\sigma_v) = n_v.
		\]
		The restriction of $Dn$ to the tangent plane is the shape operator whose
		determinant equals the Gauss curvature $\kappa$. In the oriented basis
		$\{e,f\}$ this gives
		\[
		n_u\times n_v = \kappa\,(\sigma_u\times\sigma_v).
		\]
		Taking the scalar product with $n$ and using $n=e\times f$ we obtain
		\[
		(n_u\times n_v)\cdot n
		= \kappa\,(\sigma_u\times\sigma_v)\cdot n
		= \kappa\,\|\sigma_u\times\sigma_v\|
		= \kappa\,\sqrt{G}.
		\]
		On the other hand, expressing $n_u$ and $n_v$ in the moving frame and
		computing $(n_u\times n_v)\cdot n$ gives
		\[
		(n_u\times n_v)\cdot n = \lambda_1\mu_2-\lambda_2\mu_1
		= -(\sqrt{G})_{uu}.
		\]
		Therefore
		\[
		\boxed{\kappa\,\sqrt{G} = -(\sqrt{G})_{uu}}
		\qquad\Longrightarrow\qquad
		\boxed{\kappa = -\dfrac{(\sqrt{G})_{uu}}{\sqrt{G}}}.
		\]
		
	\end{enumerate}
	
	\paragraph{Gauss' theorema egregium.}
	
	The formula
	\[
	\kappa = -\frac{(\sqrt{G})_{uu}}{\sqrt{G}}
	\]
	expresses the Gauss curvature solely in terms of the first fundamental form
	$du^2+G(u,v)\,dv^2$. Hence $\kappa$ depends only on the intrinsic metric of the
	surface. In particular, if two smooth embedded surfaces in $\mathbb{R}^3$
	are isometric, their first fundamental forms agree, and therefore their Gauss
	curvatures coincide. This is Gauss' \emph{theorema egregium}.
	
\subsection*{Examples of isometric and non--isometric surfaces
	with first fundamental form $du^{2}+G(u,v)\,dv^{2}$}

In this section we collect concrete examples illustrating the abstract setting
of Problem~E2, where the first fundamental form of a parametrized surface
has the special form
\[
I = du^{2} + G(u,v)\,dv^{2},
\]
i.e.\ $E=1$, $F=0$ and $G=G(u,v)$.  In all examples below we work with
coordinates $(u,v)$ on a parameter domain $V\subset\mathbb{R}^{2}$ and a
parametrization $\sigma:V\to\Sigma\subset\mathbb{R}^{3}$.

Recall from E2 that for such an \emph{allowable} parametrization the Gauss
curvature satisfies
\[
\kappa = -\frac{(\sqrt{G})_{uu}}{\sqrt{G}}.
\]

\subsubsection*{1. Isometric pair: plane strip and cylinder strip}

We first consider a pair of surfaces which are locally isometric and whose
first fundamental forms are both of the special form $du^{2}+G\,dv^{2}$ with
the same function $G$.

\paragraph{Plane strip.}
Let
\[
\sigma_{\mathrm{pl}}(u,v) = (u,v,0),
\qquad (u,v)\in(0,2\pi)\times(-L,L)
\]
for some fixed $L>0$.  Then
\[
\sigma_{\mathrm{pl},u} = (1,0,0),\qquad
\sigma_{\mathrm{pl},v} = (0,1,0),
\]
so
\[
E = \sigma_{\mathrm{pl},u}\cdot\sigma_{\mathrm{pl},u} = 1,\qquad
F = \sigma_{\mathrm{pl},u}\cdot\sigma_{\mathrm{pl},v} = 0,\qquad
G = \sigma_{\mathrm{pl},v}\cdot\sigma_{\mathrm{pl},v} = 1.
\]
Hence the first fundamental form on this strip of the plane is
\[
I_{\mathrm{pl}} = du^{2} + dv^{2},
\]
which is of the desired type with $G(u,v)\equiv 1$.

From Gauss' formula we obtain
\[
\sqrt{G}=1,\quad (\sqrt{G})_{uu}=0
\quad\Longrightarrow\quad
\kappa_{\mathrm{pl}} = 0.
\]

\paragraph{Cylinder strip.}
Consider the unit cylinder and the parametrization
\[
\sigma_{\mathrm{cyl}}(u,v) = (\cos u,\ \sin u,\ v),\qquad
(u,v)\in(0,2\pi)\times(-L,L).
\]
Then
\[
\sigma_{\mathrm{cyl},u} = (-\sin u,\ \cos u,\ 0),\qquad
\sigma_{\mathrm{cyl},v} = (0,0,1),
\]
so
\[
E = \sigma_{\mathrm{cyl},u}\cdot\sigma_{\mathrm{cyl},u} = 1,\qquad
F = \sigma_{\mathrm{cyl},u}\cdot\sigma_{\mathrm{cyl},v} = 0,\qquad
G = \sigma_{\mathrm{cyl},v}\cdot\sigma_{\mathrm{cyl},v} = 1.
\]
Hence
\[
I_{\mathrm{cyl}} = du^{2} + dv^{2},
\]
again with $G(u,v)\equiv 1$.  The Gauss curvature is therefore
$\kappa_{\mathrm{cyl}}=0$ by the same computation as above.

\paragraph{Isometry between the strip and the cylinder.}
On the common parameter domain $V=(0,2\pi)\times(-L,L)$,
the identity map
\[
\Phi:V\to V,\qquad \Phi(u,v)=(u,v),
\]
satisfies
\[
\Phi^{*}I_{\mathrm{cyl}}=I_{\mathrm{pl}},
\]
because their metric coefficients $E,F,G$ coincide identically.  Thus
$\sigma_{\mathrm{pl}}$ and $\sigma_{\mathrm{cyl}}$ describe locally isometric
surfaces: a rectangular strip of the plane and a corresponding strip of the
cylinder.  Geometrically, one may ``roll'' the strip into the cylinder
without stretching.

\subsubsection*{2. Non--isometric pair: plane (polar form) vs unit sphere}

We now consider two surfaces whose first fundamental forms are both of the
form $du^{2}+G(u,v)\,dv^{2}$, but with different $G$ and therefore different
Gauss curvature.  By the theorema egregium they cannot be locally isometric.

\paragraph{Plane in polar coordinates.}
On the plane (minus the origin) we use polar coordinates:
\[
\sigma_{\mathbb{R}^{2}}(u,v) = (u\cos v,\ u\sin v,\ 0),
\qquad u>0,\ v\in(0,2\pi).
\]
Then
\[
\sigma_{u} = (\cos v,\ \sin v,\ 0),\qquad
\sigma_{v} = (-u\sin v,\ u\cos v,\ 0),
\]
so
\[
E = \sigma_{u}\cdot\sigma_{u} = 1,\qquad
F = \sigma_{u}\cdot\sigma_{v} = 0,\qquad
G = \sigma_{v}\cdot\sigma_{v} = u^{2}.
\]
Hence
\[
I_{\mathbb{R}^{2}} = du^{2} + u^{2}\,dv^{2}.
\]

We compute the Gauss curvature using the E2 formula:
\[
\sqrt{G}=u,\qquad (\sqrt{G})_{u}=1,\qquad (\sqrt{G})_{uu}=0,
\]
and therefore
\[
\kappa_{\mathbb{R}^{2}}
= -\frac{(\sqrt{G})_{uu}}{\sqrt{G}}
= 0,
\]
as expected for the plane.

\paragraph{Unit sphere.}
On the unit sphere $S^{2}\subset\mathbb{R}^{3}$ take spherical coordinates:
\[
\sigma_{S^{2}}(u,v)
= (\sin u\cos v,\ \sin u\sin v,\ \cos u),
\qquad u\in(0,\pi),\ v\in(0,2\pi).
\]
We compute
\[
\sigma_{u} = (\cos u\cos v,\ \cos u\sin v,\ -\sin u),\quad
\sigma_{v} = (-\sin u\sin v,\ \sin u\cos v,\ 0),
\]
and thus
\[
E = \sigma_{u}\cdot\sigma_{u} = 1,\qquad
F = \sigma_{u}\cdot\sigma_{v} = 0,\qquad
G = \sigma_{v}\cdot\sigma_{v} = \sin^{2}u.
\]
The first fundamental form is
\[
I_{S^{2}} = du^{2} + \sin^{2}u\,dv^{2}.
\]

Again applying the curvature formula,
\[
\sqrt{G}=\sin u,\quad
(\sqrt{G})_{u} = \cos u,\quad
(\sqrt{G})_{uu} = -\sin u,
\]
we obtain
\[
\kappa_{S^{2}}
= -\frac{(\sqrt{G})_{uu}}{\sqrt{G}}
= -\frac{-\sin u}{\sin u} = 1.
\]

\paragraph{Non--isometry.}
We have shown that
\[
\kappa_{\mathbb{R}^{2}} = 0,\qquad
\kappa_{S^{2}} = 1.
\]
Since Gauss curvature is an intrinsic invariant of the metric, there can be
no local isometry between any patch of the plane (with the Euclidean metric)
and any patch of the unit sphere (with the round metric).

\subsubsection*{3. Same curvature but not locally isometric:
	exponential cone vs helicoid (from E1)}

For completeness we recall the example from Problem~E1: two surfaces with the
same Gauss curvature, but different first fundamental forms, hence not
locally isometric.

\paragraph{Exponential cone.}
Let
\[
\sigma_{\Sigma}(u,v)
= (u\cos v,\ u\sin v,\ \ln u),\qquad u>0,\ v>0.
\]
One computes
\[
I_{\Sigma}
= \Bigl(1+\frac{1}{u^{2}}\Bigr)du^{2} + u^{2}dv^{2},
\]
and
\[
K_{\Sigma}(u,v)
= -\frac{1}{(1+u^{2})^{2}}.
\]

\paragraph{Helicoid.}
Let
\[
\sigma_{\Sigma'}(u,v)
= (u\cos v,\ u\sin v,\ v),\qquad u>0,\ v>0.
\]
Then
\[
I_{\Sigma'}
= du^{2} + (u^{2}+1)\,dv^{2},
\]
and again
\[
K_{\Sigma'}(u,v)
= -\frac{1}{(1+u^{2})^{2}}.
\]

\paragraph{Conclusion.}
Although $\Sigma$ and $\Sigma'$ have the same Gauss curvature function, their
first fundamental forms differ:
\[
I_{\Sigma}
= \Bigl(1+\frac{1}{u^{2}}\Bigr)du^{2} + u^{2}dv^{2}
\neq
du^{2} + (u^{2}+1)\,dv^{2}
= I_{\Sigma'}.
\]
A direct calculation (cf.\ E1) shows that there is no local isometry between
them.  Thus equal Gauss curvature, by itself, does not guarantee local
isometry; one needs the full metric.

\medskip
% For later inclusion of charts:
% one may draw the $(u,v)$-grid on both the exponential cone and the helicoid,
% and compare how the coordinate squares are distorted on each surface.

We work with an allowable parametrization
\[
\sigma:V\to\Sigma\subset\mathbb{R}^3,\qquad
I = du^{2} + G(u,v)\,dv^{2},
\]
so that
\[
\sigma_{u}\cdot\sigma_{u} = 1,\quad
\sigma_{u}\cdot\sigma_{v} = 0,\quad
\sigma_{v}\cdot\sigma_{v} = G(u,v)>0.
\]

Define
\[
e = \sigma_{u},\qquad
f = \frac{\sigma_{v}}{\sqrt{G}},\qquad
n = \frac{\sigma_{u}\times\sigma_{v}}
{\|\sigma_{u}\times\sigma_{v}\|}.
\]

Then
\[
\|e\|^{2} = e\cdot e = \sigma_{u}\cdot\sigma_{u} = 1,
\]
\[
\|f\|^{2}
= f\cdot f
= \frac{\sigma_{v}\cdot\sigma_{v}}{G}
= \frac{G}{G} = 1,
\]
and
\[
e\cdot f
= \sigma_{u}\cdot\frac{\sigma_{v}}{\sqrt{G}}
= \frac{\sigma_{u}\cdot\sigma_{v}}{\sqrt{G}}
= \frac{F}{\sqrt{G}} = 0.
\]
Thus $e$ and $f$ form an orthonormal basis of the tangent plane
$T_{\sigma(u,v)}\Sigma$.

Moreover,
\[
\sigma_{u}\times\sigma_{v}
= e\times(\sqrt{G}\,f)
= \sqrt{G}\,(e\times f),
\]
so
\[
\|\sigma_{u}\times\sigma_{v}\|
= \sqrt{G}\,\|e\times f\|
= \sqrt{G},
\]
and hence
\[
n
= \frac{\sigma_{u}\times\sigma_{v}}{\|\sigma_{u}\times\sigma_{v}\|}
= \frac{\sqrt{G}\,(e\times f)}{\sqrt{G}}
= e\times f.
\]
Therefore $(e,f,n)$ is an orthonormal frame in $\mathbb{R}^{3}$ along the
surface, consisting of two orthonormal tangent vectors $e,f$ and the unit
normal $n$.

In these coordinates the first fundamental form is
\[
I = du^{2} + G(u,v)\,dv^{2},
\]
which means that the coordinate curves $u\mapsto\sigma(u,v_{0})$ are
parametrized by arc length ($E=1$), the $u$- and $v$-curves are orthogonal
($F=0$), and the scale factor in the $v$-direction is $\sqrt{G(u,v)}$.

\begin{example}[Sphere]
	On the unit sphere with parametrization
	\[
	\sigma(u,v) = (\sin u\cos v,\ \sin u\sin v,\ \cos u),
	\]
	one finds
	\[
	I = du^{2} + \sin^{2}u\,dv^{2},
	\]
	so $G(u,v)=\sin^{2}u$.  The corresponding moving frame
	$e=\sigma_{u}$, $f=\sigma_{v}/\sqrt{G}$, $n=e\times f$ is adapted to the
	meridians and parallels.
\end{example}

\begin{example}[Plane in polar coordinates]
	On the plane (minus the origin) with
	\[
	\sigma(u,v) = (u\cos v,\ u\sin v,\ 0),
	\]
	we obtain
	\[
	I = du^{2} + u^{2}dv^{2},
	\]
	so $G(u,v)=u^{2}$, and again $e=\sigma_{u}$, $f=\sigma_{v}/\sqrt{G}$,
	$n=e\times f$ provide an orthonormal frame adapted to radial lines and circles.
\end{example}

	\caption{An allowable chart on the unit sphere.
		The vectors $e=\sigma_u$, $f=\sigma_v/\sqrt{G}$ and $n=e\times f$

\subsubsection*{Stereographic chart and moving frame}

Stereographic projection from the north pole gives a parametrization
\[
\sigma : \mathbb{R}^2 \longrightarrow S^2 \setminus \{(0,0,1)\}
\]
defined by
\[
\sigma(x,y)
=
\left(
\frac{2x}{x^2 + y^2 + 1},
\frac{2y}{x^2 + y^2 + 1},
\frac{x^2 + y^2 - 1}{x^2 + y^2 + 1}
\right).
\]

Compute the partial derivatives:
\[
\sigma_x = \frac{\partial \sigma}{\partial x},
\qquad
\sigma_y = \frac{\partial \sigma}{\partial y}.
\]

One can check that the first fundamental form is
\[
E = \sigma_x \cdot \sigma_x
= \frac{4}{(1 + x^2 + y^2)^2},
\qquad
F = \sigma_x \cdot \sigma_y = 0,
\]
\[
G = \sigma_y \cdot \sigma_y
= \frac{4}{(1 + x^2 + y^2)^2},
\]
so
\[
I
= \frac{4}{(1 + x^2 + y^2)^2}\,(dx^2 + dy^2).
\]

This is \emph{not} of the ``E2 special form'' $du^2 + G(u,v)\,dv^2$
(here $E = G \neq 1$), but we can still build an orthonormal frame by
normalizing the coordinate vectors:
\[
e = \frac{\sigma_x}{\|\sigma_x\|},
\qquad
f = \frac{\sigma_y}{\|\sigma_y\|},
\qquad
n = e \times f.
\]

Then $(e,f,n)$ is again an orthonormal moving frame along the surface,
just with a different underlying chart.

\begin{figure}[h]
	\centering
	\includegraphics[width=0.55\textwidth]{moving_frame_sphere_stereo.png}
	\caption{Unit sphere with a moving orthonormal frame $(e,f,n)$,
		constructed from stereographic coordinates $(x,y)$ rather than
		spherical (polar) coordinates.}

\paragraph{Chain rule interpretation of $Dn(\sigma_u)=n_u$.}
Let $\sigma:V\to\Sigma\subset\mathbb{R}^3$ be a parametrization and
$N:\Sigma\to S^2$ the Gauss map.  Define
\[
n(u,v) := N(\sigma(u,v)).
\]
Then by the chain rule,
\[
n_u(u,v)
= \frac{\partial}{\partial u} \bigl(N(\sigma(u,v))\bigr)
= DN_{\sigma(u,v)}\bigl(\sigma_u(u,v)\bigr),
\]
and similarly
\[
n_v(u,v)
= DN_{\sigma(u,v)}\bigl(\sigma_v(u,v)\bigr).
\]
This is what is meant by
\[
Dn(\sigma_u) = n_u, \qquad Dn(\sigma_v) = n_v,
\]
where $Dn$ denotes the differential of the Gauss map restricted to the
tangent plane.

\paragraph{Unit sphere in polar coordinates.}
For the unit sphere $S^2$, the Gauss map is the identity
$N(p)=p$.  In spherical coordinates
\[
\sigma(u,v) = (\sin u\cos v,\ \sin u\sin v,\ \cos u),
\]
we have
\[
n(u,v) = N(\sigma(u,v)) = \sigma(u,v),
\]
hence
\[
n_u = \sigma_u,\qquad n_v = \sigma_v.
\]
Since $DN_p = \mathrm{Id}$, we obtain
\[
DN_{\sigma(u,v)}(\sigma_u) = \sigma_u = n_u,\qquad
DN_{\sigma(u,v)}(\sigma_v) = \sigma_v = n_v.
\]

\paragraph{Unit sphere in stereographic coordinates.}
Using stereographic coordinates
\[
\sigma(x,y)
= \left(
\frac{2x}{x^2+y^2+1},
\frac{2y}{x^2+y^2+1},
\frac{x^2+y^2-1}{x^2+y^2+1}
\right),
\]
we again have $N(p)=p$ and therefore $n(x,y)=\sigma(x,y)$, so
\[
n_x = \sigma_x,\qquad n_y = \sigma_y,
\]
and
\[
DN_{\sigma(x,y)}(\sigma_x) = \sigma_x = n_x,\qquad
DN_{\sigma(x,y)}(\sigma_y) = \sigma_y = n_y.
\]
In both charts the equalities $Dn(\sigma_u)=n_u$, $Dn(\sigma_v)=n_v$ are
simply the chain rule applied to $n = N\circ\sigma$.

\subsubsection*{Example: explicit matrix of $Dn$ on the unit sphere}

Consider the unit sphere $S^{2}\subset\mathbb{R}^{3}$ with the usual spherical
coordinates
\[
\sigma(u,v) = (\sin u\cos v,\ \sin u\sin v,\ \cos u),
\qquad 0<u<\pi,\ 0<v<2\pi.
\]
The Gauss map of $S^{2}$ is simply the identity
\[
N(p)=p,\qquad p\in S^{2},
\]
so in these coordinates
\[
n(u,v) := N(\sigma(u,v)) = \sigma(u,v).
\]

\paragraph{Tangent vectors and metric.}
We have
\[
\sigma_{u} = (\cos u\cos v,\ \cos u\sin v,\ -\sin u),\qquad
\sigma_{v} = (-\sin u\sin v,\ \sin u\cos v,\ 0),
\]
so
\[
E = \sigma_{u}\cdot\sigma_{u} = 1,\quad
F = \sigma_{u}\cdot\sigma_{v} = 0,\quad
G = \sigma_{v}\cdot\sigma_{v} = \sin^{2}u.
\]
Thus the first fundamental form is
\[
I = du^{2} + \sin^{2}u\,dv^{2},
\]
and an orthonormal tangent frame is given by
\[
e := \sigma_{u},\qquad
f := \frac{\sigma_{v}}{\sin u}\quad (u\in(0,\pi)).
\]

\paragraph{Differential of the Gauss map.}
Since $n=\sigma$, we have
\[
n_{u} = \sigma_{u} = e,\qquad
n_{v} = \sigma_{v}.
\]
The differential $Dn$ at the point $\sigma(u,v)$ is characterized by
\[
Dn_{\sigma(u,v)}(\sigma_{u}) = n_{u},\qquad
Dn_{\sigma(u,v)}(\sigma_{v}) = n_{v}.
\]

We now express $Dn$ in the orthonormal basis $\{e,f\}$ of the tangent plane.
First,
\[
Dn(e) = Dn(\sigma_{u}) = n_{u} = e.
\]
Next, noting that $f = \sigma_{v}/\sin u$ and $u$ is constant when we vary
$v$,
\[
Dn(f)
= Dn\Bigl(\frac{1}{\sin u}\,\sigma_{v}\Bigr)
= \frac{1}{\sin u}\,Dn(\sigma_{v})
= \frac{1}{\sin u}\,n_{v}
= \frac{1}{\sin u}\,\sigma_{v}
= f.
\]

Therefore, relative to the orthonormal basis $\{e,f\}$,
\[
Dn|_{T_{\sigma(u,v)}S^{2}} =
\begin{pmatrix}
	1 & 0\\
	0 & 1
\end{pmatrix}.
\]

\paragraph{Shape operator and Gauss curvature.}
By definition the shape operator is
\[
S = -\,Dn|_{T\Sigma},
\]
so in the basis $\{e,f\}$ we have
\[
S =
\begin{pmatrix}
	-1 & 0\\
	0 & -1
\end{pmatrix}.
\]
Its eigenvalues are both $-1$, and hence the Gauss curvature is
\[
K = \det S = (-1)\cdot(-1) = 1.
\]

This agrees with the intrinsic formula for an allowable parametrization from
Problem~E2: here
\[
G(u,v)=\sin^{2}u,\quad \sqrt{G}=\sin u,\quad
(\sqrt{G})_{uu} = -\sin u,
\]
so
\[
\kappa = -\frac{(\sqrt{G})_{uu}}{\sqrt{G}}
= -\frac{-\sin u}{\sin u} = 1.
\]
Thus the determinant of the shape operator $S=-Dn$ coincides with the
intrinsic Gauss curvature.

\subsubsection*{Example: $Dn$ in stereographic coordinates on $S^2$}

Consider the stereographic parametrization of the unit sphere
(from the north pole) given by
\[
\sigma : \mathbb{R}^2 \longrightarrow S^2 \setminus \{(0,0,1)\},
\qquad
\sigma(x,y)
=
\left(
\frac{2x}{x^2 + y^2 + 1},
\frac{2y}{x^2 + y^2 + 1},
\frac{x^2 + y^2 - 1}{x^2 + y^2 + 1}
\right).
\]
Set $r^2 = x^2 + y^2$ and $d = r^2 + 1$ for brevity.

\paragraph{First fundamental form.}
Compute the partial derivatives
\[
\sigma_x = \frac{\partial \sigma}{\partial x},\qquad
\sigma_y = \frac{\partial \sigma}{\partial y}.
\]
A straightforward calculation shows
\[
\sigma_x \cdot \sigma_x
= \frac{4}{(1 + x^2 + y^2)^2},\qquad
\sigma_y \cdot \sigma_y
= \frac{4}{(1 + x^2 + y^2)^2},\qquad
\sigma_x \cdot \sigma_y = 0.
\]
Thus
\[
E = \sigma_x\cdot\sigma_x
= \frac{4}{(1 + r^2)^2},\quad
F = 0,\quad
G = \sigma_y\cdot\sigma_y
= \frac{4}{(1 + r^2)^2},
\]
and the first fundamental form is
\[
I
= \frac{4}{(1 + x^2 + y^2)^2}\,(dx^2 + dy^2).
\]
In particular, this chart is conformal: the metric is a scalar multiple
of the Euclidean metric on $\mathbb{R}^2$.

\paragraph{Orthonormal tangent frame.}
Since $E = G$ and $F = 0$, the coordinate vectors $\sigma_x$ and $\sigma_y$
are orthogonal and have equal length
\[
\|\sigma_x\|^2 = \|\sigma_y\|^2
= \frac{4}{(1 + r^2)^2}.
\]
We obtain an orthonormal tangent frame by normalizing:
\[
e := \frac{\sigma_x}{\|\sigma_x\|},\qquad
f := \frac{\sigma_y}{\|\sigma_y\|},\qquad
n := e\times f.
\]
Then $\{e,f\}$ is an orthonormal basis of $T_{\sigma(x,y)}S^2$ and $n$ is
the outward unit normal.

\paragraph{Gauss map and its differential.}
On the unit sphere, the Gauss map is simply
\[
N(p) = p,\qquad p\in S^2.
\]
Therefore in these coordinates
\[
n(x,y) := N(\sigma(x,y)) = \sigma(x,y),
\]
hence
\[
n_x = \sigma_x,\qquad n_y = \sigma_y.
\]
The differential of the Gauss map at the point $\sigma(x,y)$ is
\[
Dn_{\sigma(x,y)} : T_{\sigma(x,y)}S^2 \longrightarrow T_{\sigma(x,y)}S^2.
\]
Since $N$ is the identity map on $S^2$, its differential is the identity on
each tangent space:
\[
Dn_{\sigma(x,y)} = \mathrm{Id}_{T_{\sigma(x,y)}S^2}.
\]

In particular,
\[
Dn(e) = e,\qquad Dn(f) = f.
\]
Thus, relative to the orthonormal basis $\{e,f\}$, the matrix of $Dn$ is
\[
Dn|_{T_{\sigma(x,y)}S^2}
=
\begin{pmatrix}
	1 & 0\\
	0 & 1
\end{pmatrix}.
\]

\paragraph{Shape operator and Gauss curvature.}
The shape operator is defined by
\[
S = -\,Dn|_{T S^2}.
\]
Hence, in the orthonormal frame $\{e,f\}$,
\[
S
=
\begin{pmatrix}
	-1 & 0\\
	0 & -1
\end{pmatrix}.
\]
Its eigenvalues are both $-1$, so the Gauss curvature is
\[
K = \det S = (-1)\cdot(-1) = 1.
\]

This agrees with the fact that the unit sphere has constant Gauss curvature
$K \equiv 1$.  We see explicitly that, even in stereographic coordinates where
the metric is
\[
I = \frac{4}{(1 + x^2 + y^2)^2}\,(dx^2 + dy^2),
\]
the differential of the Gauss map restricted to the tangent plane has
matrix $-I_{2}$ (up to sign) in an orthonormal frame, and its determinant
gives the curvature.

\subsection*{Comparison of two charts on $S^2$ and the differential of the Gauss map}

We compare two different parametrizations of the unit sphere $S^2\subset\mathbb{R}^3$:
spherical (polar) coordinates and stereographic coordinates.  In both cases we
compute the first fundamental form, an orthonormal frame, the differential of
the Gauss map, and the Gauss curvature.

\subsubsection*{1. Spherical (polar) coordinates}

Consider
\[
\sigma(u,v) = (\sin u\cos v,\ \sin u\sin v,\ \cos u),
\qquad 0<u<\pi,\ 0<v<2\pi.
\]
We have
\[
\sigma_u = (\cos u\cos v,\ \cos u\sin v,\ -\sin u),\qquad
\sigma_v = (-\sin u\sin v,\ \sin u\cos v,\ 0),
\]
so
\[
E = \sigma_u\cdot\sigma_u = 1,\quad
F = \sigma_u\cdot\sigma_v = 0,\quad
G = \sigma_v\cdot\sigma_v = \sin^2 u,
\]
and the first fundamental form is
\[
I_{\mathrm{pol}} = du^2 + \sin^2 u\,dv^2.
\]

An orthonormal tangent frame is
\[
e := \sigma_u,\qquad
f := \frac{\sigma_v}{\sin u},\qquad
n := e\times f.
\]

On the unit sphere the Gauss map is the identity,
$N(p)=p$, hence $n(u,v):=N(\sigma(u,v))=\sigma(u,v)$ and
\[
n_u = \sigma_u,\qquad n_v = \sigma_v.
\]
The differential $Dn_{\sigma(u,v)}$ restricted to the tangent plane satisfies
\[
Dn(e) = e,\qquad Dn(f) = f,
\]
so in the orthonormal basis $\{e,f\}$,
\[
Dn|_{T_{\sigma(u,v)}S^2}
=
\begin{pmatrix}
	1 & 0\\
	0 & 1
\end{pmatrix}.
\]
The shape operator is $S=-\,Dn|_{T S^2}$, hence
\[
S =
\begin{pmatrix}
	-1 & 0\\
	0 & -1
\end{pmatrix},
\qquad
K = \det S = 1.
\]

\subsubsection*{2. Stereographic coordinates}

Now consider stereographic projection from the north pole.  With
$r^2 = x^2 + y^2$ we define
\[
\sigma(x,y)
=
\left(
\frac{2x}{r^2 + 1},
\frac{2y}{r^2 + 1},
\frac{r^2 - 1}{r^2 + 1}
\right),
\qquad (x,y)\in\mathbb{R}^2.
\]
A direct computation shows
\[
\sigma_x\cdot\sigma_x
= \frac{4}{(1+r^2)^2},\qquad
\sigma_y\cdot\sigma_y
= \frac{4}{(1+r^2)^2},\qquad
\sigma_x\cdot\sigma_y = 0.
\]
Thus
\[
E = G = \frac{4}{(1+r^2)^2},\qquad F = 0,
\]
and
\[
I_{\mathrm{stereo}}
= \frac{4}{(1+x^2 + y^2)^2}\,(dx^2 + dy^2),
\]
so this chart is conformal.

We obtain an orthonormal tangent frame by normalizing the coordinate vectors:
\[
e := \frac{\sigma_x}{\|\sigma_x\|},\qquad
f := \frac{\sigma_y}{\|\sigma_y\|},\qquad
n := e\times f.
\]
Again, the Gauss map is $N(p)=p$, so
\[
n(x,y) := N(\sigma(x,y)) = \sigma(x,y),\qquad
n_x = \sigma_x,\ n_y = \sigma_y.
\]
Since $N$ is the identity on $S^2$, its differential is the identity on each
tangent space:
\[
Dn_{\sigma(x,y)} = \mathrm{Id}_{T_{\sigma(x,y)}S^2}.
\]
In particular,
\[
Dn(e) = e,\qquad Dn(f) = f,
\]
so in the orthonormal basis $\{e,f\}$ we again have
\[
Dn|_{T_{\sigma(x,y)}S^2}
=
\begin{pmatrix}
	1 & 0\\
	0 & 1
\end{pmatrix},
\qquad
S = -\,Dn
=
\begin{pmatrix}
	-1 & 0\\
	0 & -1
\end{pmatrix},
\qquad
K = \det S = 1.
\]

\subsubsection*{3. Comparison}

In spherical coordinates the metric is
\[
I_{\mathrm{pol}} = du^2 + \sin^2 u\,dv^2,
\]
while in stereographic coordinates it is
\[
I_{\mathrm{stereo}}
= \frac{4}{(1+x^2 + y^2)^2}\,(dx^2 + dy^2).
\]
These two expressions look very different: one is adapted to meridians and
parallels, the other is conformal to the Euclidean plane.  Nevertheless, in
each chart we can construct an orthonormal frame $(e,f)$ and compute the
matrix of the differential of the Gauss map $Dn|_{T S^2}$.  In both cases this
matrix is $-I_{2}$ for the shape operator, hence
\[
K = \det S = 1
\]
everywhere on $S^2$.  This illustrates that Gauss curvature is intrinsic:
different coordinate systems and different expressions for the first
fundamental form all encode the same geometric quantity $K$.
