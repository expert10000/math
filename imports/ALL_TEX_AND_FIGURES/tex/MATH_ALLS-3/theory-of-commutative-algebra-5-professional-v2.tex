\section{Problem 1 --- Localization of \texorpdfstring{$\operatorname{Hom}$}{Hom} for finitely presented modules}

\textbf{Exact statement.}
Let \(A\) be a ring, \(S \subseteq A\) a multiplicatively closed subset, and \(N,M\) \(A\)-modules with \(M\) finitely presented, i.e. there exists an exact sequence
\[
A^{n_1}\longrightarrow A^{n_2}\longrightarrow M\longrightarrow 0
\]
for some nonnegative integers \(n_1,n_2\).

Show that there is an isomorphism of \(S^{-1}A\)-modules
\[
S^{-1}\!\operatorname{Hom}_A(M,N)\;\cong\;\operatorname{Hom}_{S^{-1}A}(S^{-1}M,S^{-1}N).
\]

\textbf{Solution.}
Take a finite presentation of \(M\):
\[
A^{n_1}\xrightarrow{u} A^{n_2}\xrightarrow{v} M\to 0.
\]

Apply the contravariant functor \(\operatorname{Hom}_A(-,N)\). Since \(\operatorname{Hom}\) is left exact, we obtain an exact sequence
\[
0\to \operatorname{Hom}_A(M,N)\to \operatorname{Hom}_A(A^{n_2},N)\to \operatorname{Hom}_A(A^{n_1},N).
\]
Because \(\operatorname{Hom}_A(A^r,N)\cong N^r\), this becomes
\[
0\to \operatorname{Hom}_A(M,N)\to N^{n_2}\xrightarrow{u^*} N^{n_1}.
\]

Now localize at \(S\). Localization is exact, so
\[
0\to S^{-1}\operatorname{Hom}_A(M,N)\to S^{-1}(N^{n_2})\xrightarrow{S^{-1}u^*} S^{-1}(N^{n_1}).
\]
Since localization commutes with finite direct sums,
\[
S^{-1}(N^{n_i})\cong (S^{-1}N)^{n_i},
\]
hence
\[
0\to S^{-1}\operatorname{Hom}_A(M,N)\to (S^{-1}N)^{n_2}\to (S^{-1}N)^{n_1}.
\]

On the other hand, localizing the presentation of \(M\) gives
\[
(S^{-1}A)^{n_1}\xrightarrow{S^{-1}u}(S^{-1}A)^{n_2}\to S^{-1}M\to 0.
\]

Apply \(\operatorname{Hom}_{S^{-1}A}(-,S^{-1}N)\). Again by left exactness,
\[
0\to \operatorname{Hom}_{S^{-1}A}(S^{-1}M,S^{-1}N)\to
\operatorname{Hom}_{S^{-1}A}((S^{-1}A)^{n_2},S^{-1}N)\to
\operatorname{Hom}_{S^{-1}A}((S^{-1}A)^{n_1},S^{-1}N).
\]
Using
\[
\operatorname{Hom}_{S^{-1}A}((S^{-1}A)^r,S^{-1}N)\cong (S^{-1}N)^r,
\]
we obtain
\[
0\to \operatorname{Hom}_{S^{-1}A}(S^{-1}M,S^{-1}N)\to
(S^{-1}N)^{n_2}\to (S^{-1}N)^{n_1}.
\]

The map
\[
(S^{-1}N)^{n_2}\to (S^{-1}N)^{n_1}
\]
in this sequence is induced by the same matrix \(u\), now viewed over \(S^{-1}A\). Therefore both
\[
S^{-1}\operatorname{Hom}_A(M,N)
\quad\text{and}\quad
\operatorname{Hom}_{S^{-1}A}(S^{-1}M,S^{-1}N)
\]
identify with the kernel of the same map
\[
(S^{-1}N)^{n_2}\to (S^{-1}N)^{n_1}.
\]
Hence they are canonically isomorphic:
\[
S^{-1}\!\operatorname{Hom}_A(M,N)\;\cong\;\operatorname{Hom}_{S^{-1}A}(S^{-1}M,S^{-1}N).
\]

An explicit formula for the isomorphism is
\[
\frac{f}{s}\;\longmapsto\;\left(\frac{m}{t}\mapsto \frac{f(m)}{st}\right).
\]

\section{Problem 2 --- UFDs are normal; normalization of \texorpdfstring{$k[x,y]/(y^2-x^3)$}{k[x,y]/(y\string^2-x\string^3)}}

\subsection*{(a) A UFD is integrally closed}

\textbf{Exact statement.}
Show that a UFD (Unique Factorization Domain) is integrally closed in its field of fractions, i.e. is normal.

\textbf{Solution.}
Let \(A\) be a UFD, and let \(K=\operatorname{Frac}(A)\). Take \(z\in K\) integral over \(A\). We must show that \(z\in A\).

Write
\[
z=\frac{a}{b}
\]
with \(a,b\in A\), \(b\neq 0\), and \(\gcd(a,b)=1\).

Since \(z\) is integral over \(A\), it satisfies a monic equation
\[
z^n+c_{n-1}z^{n-1}+\cdots+c_1z+c_0=0,
\qquad c_i\in A.
\]
Multiplying by \(b^n\), we get
\[
a^n+c_{n-1}a^{n-1}b+c_{n-2}a^{n-2}b^2+\cdots+c_1ab^{n-1}+c_0b^n=0.
\]
Thus
\[
a^n=-b\big(c_{n-1}a^{n-1}+c_{n-2}a^{n-2}b+\cdots+c_1ab^{n-2}+c_0b^{n-1}\big),
\]
so \(b\mid a^n\).

Let \(p\) be any irreducible divisor of \(b\). Then \(p\mid a^n\), hence \(p\mid a\). This contradicts \(\gcd(a,b)=1\). Therefore \(b\) has no irreducible divisors, so \(b\) is a unit. Hence
\[
z=\frac{a}{b}\in A.
\]

Therefore every element of \(\operatorname{Frac}(A)\) integral over \(A\) already lies in \(A\). So \(A\) is integrally closed, i.e. normal.

\subsection*{(b) The cusp ring \texorpdfstring{$k[x,y]/(y^2-x^3)$}{k[x,y]/(y\string^2-x\string^3)}}

\textbf{Exact statement.}
Let \(k\) be a field and
\[
A=k[x,y]/(y^2-x^3).
\]
Is \(A\) normal? If not, what is the integral closure of \(A\) in its field of fractions?

\textbf{Solution.}
Define a homomorphism
\[
\varphi:k[x,y]\to k[t],\qquad x\mapsto t^2,\quad y\mapsto t^3.
\]
Since
\[
(t^3)^2-(t^2)^3=t^6-t^6=0,
\]
the ideal \((y^2-x^3)\) is contained in \(\ker\varphi\). In fact,
\[
\ker\varphi=(y^2-x^3),
\]
so
\[
A\cong k[t^2,t^3]\subseteq k[t].
\]

Thus \(A\) is the cusp ring \(k[t^2,t^3]\).

Its field of fractions is
\[
\operatorname{Frac}(A)=k(t),
\]
because
\[
t=\frac{t^3}{t^2}=\frac{y}{x}\in \operatorname{Frac}(A).
\]

Now \(t\) is integral over \(A\), since it satisfies the monic equation
\[
T^2-x=0
\]
with coefficients in \(A\). But \(t\notin A=k[t^2,t^3]\). Hence \(A\) is not normal.

Since
\[
A\subseteq k[t]\subseteq k(t),
\]
and \(k[t]\) is a PID, hence a UFD, hence integrally closed, the integral closure of \(A\) in its field of fractions is
\[
\overline{A}=k[t].
\]

Therefore \(A\) is not normal, and its integral closure in \(\operatorname{Frac}(A)\) is \(k[t]\).	


\section{Problem 3 --- A subring whose complement is multiplicatively closed is integrally closed in the ambient ring}

\textbf{Exact statement.}
Let \(A\subseteq B\) be a subring of \(B\), with the property that the complement
\[
B\setminus A
\]
is multiplicatively closed. Show that \(A\) is integrally closed in \(B\).

\textbf{Solution.}
We must show that if \(b\in B\) is integral over \(A\), then \(b\in A\).

Assume, for contradiction, that \(b\notin A\). Then \(b\in B\setminus A\). Since \(B\setminus A\) is multiplicatively closed, every positive power
\[
b,\;b^2,\;b^3,\dots
\]
also lies in \(B\setminus A\).

Since \(b\) is integral over \(A\), it satisfies a monic equation
\[
b^n+a_{n-1}b^{n-1}+\cdots+a_1b+a_0=0,
\qquad a_i\in A.
\]
Choose such an equation with \(n\) minimal.

Rewrite it as
\[
b\bigl(b^{n-1}+a_{n-1}b^{n-2}+\cdots+a_1\bigr)=-a_0\in A.
\]
Set
\[
c:=b^{n-1}+a_{n-1}b^{n-2}+\cdots+a_1\in B.
\]

If \(c\in B\setminus A\), then both \(b\) and \(c\) lie in the multiplicatively closed set \(B\setminus A\), so
\[
bc\in B\setminus A.
\]
But
\[
bc=-a_0\in A,
\]
a contradiction. Hence \(c\in A\).

Therefore
\[
b^{n-1}+a_{n-1}b^{n-2}+\cdots+a_2b+(a_1-c)=0,
\]
which is a monic equation of degree \(n-1\) with coefficients in \(A\) satisfied by \(b\). This contradicts the minimality of \(n\), unless \(n=1\).

If \(n=1\), then
\[
b+a_0=0,
\]
so \(b=-a_0\in A\), again a contradiction.

Hence \(b\in A\). Therefore \(A\) is integrally closed in \(B\).

\section{Problem 4 --- More forms of Hilbert's Nullstellensatz}

Let
\[
A=k[x_1,\dots,x_n],
\]
where \(k\) is algebraically closed. For an ideal \(I\subseteq A\), define
\[
Z(I):=\{(a_1,\dots,a_n)\in k^n\mid f(a_1,\dots,a_n)=0\text{ for all }f\in I\},
\]
and for a subset \(X\subseteq k^n\), define
\[
I(X):=\{f\in A\mid f(a_1,\dots,a_n)=0\text{ for all }(a_1,\dots,a_n)\in X\}.
\]

\subsection*{(a) Proper ideals have a zero}

\textbf{Exact statement.}
Show that if \(I\neq A\), then \(Z(I)\) is non-empty.

\textbf{Solution.}
Since \(I\neq A\), it is contained in some maximal ideal \(\mathfrak m\subset A\).

By the weak Hilbert Nullstellensatz, every maximal ideal of \(k[x_1,\dots,x_n]\) has the form
\[
\mathfrak m=(x_1-a_1,\dots,x_n-a_n)
\]
for some \((a_1,\dots,a_n)\in k^n\).

Because \(I\subseteq \mathfrak m\), every polynomial in \(I\) vanishes at \((a_1,\dots,a_n)\). Hence
\[
(a_1,\dots,a_n)\in Z(I),
\]
so
\[
Z(I)\neq\varnothing.
\]

\subsection*{(b) The strong Nullstellensatz}

\textbf{Exact statement.}
Show that
\[
I(Z(I))=\sqrt I.
\]

\textbf{Solution.}
We prove both inclusions.

First, let \(f\in \sqrt I\). Then \(f^m\in I\) for some \(m\ge 1\). If \(a\in Z(I)\), then every element of \(I\) vanishes at \(a\), hence
\[
f(a)^m=0.
\]
Since \(k\) is a field, this implies \(f(a)=0\). Therefore \(f\in I(Z(I))\), and so
\[
\sqrt I\subseteq I(Z(I)).
\]

Now let \(f\in I(Z(I))\). We must show that \(f\in \sqrt I\).

Introduce a new variable \(y\), and consider the ideal
\[
J=(I,\;1-yf)\subseteq k[x_1,\dots,x_n,y].
\]
We claim that
\[
Z(J)=\varnothing.
\]
Indeed, if \((a_1,\dots,a_n,b)\in Z(J)\), then \((a_1,\dots,a_n)\in Z(I)\), so \(f(a_1,\dots,a_n)=0\) because \(f\in I(Z(I))\). But also \(1-yf\in J\), hence
\[
1-b\,f(a_1,\dots,a_n)=0,
\]
which becomes \(1=0\), impossible.

Thus \(Z(J)=\varnothing\). By part (a), the contrapositive implies that
\[
J=k[x_1,\dots,x_n,y].
\]
Hence there exist
\[
g\in I\cdot k[x_1,\dots,x_n,y],\qquad h\in k[x_1,\dots,x_n,y]
\]
such that
\[
1=g+h(1-yf).
\]

Now pass to the fraction field \(k(x_1,\dots,x_n)\) and substitute \(y=1/f\). Then
\[
1=g(x_1,\dots,x_n,1/f).
\]
Write
\[
g=\sum_j p_j(x,y)\,g_j(x),
\qquad g_j(x)\in I.
\]
After substitution \(y=1/f\), each \(p_j(x,1/f)\) is a rational function whose denominator is some power of \(f\). Multiplying by a sufficiently large power \(f^N\), we obtain
\[
f^N=\sum_j q_j(x)\,g_j(x)\in I
\]
for some polynomials \(q_j(x)\in k[x_1,\dots,x_n]\).

Thus \(f^N\in I\), so \(f\in \sqrt I\). Therefore
\[
I(Z(I))\subseteq \sqrt I.
\]

Combining the two inclusions, we conclude that
\[
I(Z(I))=\sqrt I.
\]

\section{Problem 5 --- Maximal local subrings of a field and valuation rings}

\textbf{Exact statement.}
Let \((A,\mathfrak m)\), \((B,\mathfrak n)\) be two local rings. We say that \(B\) \emph{dominates} \(A\) if \(A\) is a subring of \(B\) and
\[
\mathfrak m=A\cap \mathfrak n,
\]
or equivalently, \(\mathfrak m\subseteq \mathfrak n\).

Let \(K\) be a field and let \(\Sigma\) be the set of all local subrings of \(K\). If \(\Sigma\) is ordered by the relation of domination, show that \(\Sigma\) has maximal elements and that \(A\in\Sigma\) is maximal if and only if \(A\) is a valuation ring of \(K\).

\textbf{Solution.}
Recall that a subring \(V\subseteq K\) is a valuation ring of \(K\) if for every \(x\in K^\times\),
\[
x\in V \quad\text{or}\quad x^{-1}\in V.
\]

\textbf{Step 1: Existence of maximal elements.}
We apply Zorn's lemma. Let
\[
\{(A_i,\mathfrak m_i)\}_{i\in I}
\]
be a chain in \(\Sigma\), ordered by domination. Define
\[
A:=\bigcup_{i\in I} A_i,
\qquad
\mathfrak m:=\bigcup_{i\in I}\mathfrak m_i.
\]

Since the \(A_i\) form an increasing chain of subrings of \(K\), their union \(A\) is again a subring of \(K\). The set \(\mathfrak m\) is an ideal of \(A\): if \(x,y\in \mathfrak m\), then they lie in some common \(A_j\), hence \(x-y\in \mathfrak m_j\subseteq \mathfrak m\); similarly, if \(a\in A\) and \(x\in \mathfrak m\), then both lie in some \(A_j\), so \(ax\in \mathfrak m_j\subseteq \mathfrak m\).

Now let \(u\in A\setminus \mathfrak m\). Then \(u\in A_j\) for some \(j\), and \(u\notin \mathfrak m_j\). Since \(A_j\) is local, \(u\) is a unit in \(A_j\), hence in \(A\). Therefore every element outside \(\mathfrak m\) is a unit, so \(A\) is a local ring with maximal ideal \(\mathfrak m\). It dominates every \(A_i\).

Thus every chain in \(\Sigma\) has an upper bound, and Zorn's lemma gives maximal elements.

\medskip

\textbf{Step 2: A maximal element is a valuation ring.}
Assume \((A,\mathfrak m)\in \Sigma\) is maximal. Let \(x\in K^\times\). We show that
\[
x\in A \quad\text{or}\quad x^{-1}\in A.
\]

Suppose, to the contrary, that
\[
x\notin A,\qquad x^{-1}\notin A.
\]

We first show that \(\mathfrak m A[x]\neq A[x]\). Otherwise,
\[
1=a_0+a_1x+\cdots+a_nx^n
\qquad (a_i\in \mathfrak m).
\]
Since \(a_0\in \mathfrak m\), the element
\[
u:=1-a_0
\]
is a unit in \(A\). Hence
\[
x(a_1+a_2x+\cdots+a_nx^{n-1})=u,
\]
so
\[
x^{-1}=u^{-1}(a_1+a_2x+\cdots+a_nx^{n-1})\in A[x].
\]

Let \(y=x^{-1}\). Then
\[
y=u^{-1}(a_1+a_2y^{-1}+\cdots+a_n y^{-(n-1)}).
\]
Multiplying by \(y^{n-1}\), we get
\[
y^n-u^{-1}a_1y^{n-1}-u^{-1}a_2y^{n-2}-\cdots-u^{-1}a_n=0.
\]
So \(x^{-1}\) is integral over \(A\).

Therefore \(A[x^{-1}]\) is integral over \(A\). By lying over, there exists a prime ideal \(\mathfrak p\subseteq A[x^{-1}]\) with
\[
\mathfrak p\cap A=\mathfrak m.
\]
Then
\[
B:=A[x^{-1}]_{\mathfrak p}
\]
is a local subring of \(K\) dominating \(A\), and it strictly contains \(A\), since \(x^{-1}\notin A\). This contradicts the maximality of \(A\).

Hence \(\mathfrak m A[x]\) is proper. Choose a maximal ideal \(\mathfrak n\) of \(A[x]\) containing \(\mathfrak m A[x]\). Then
\[
B:=A[x]_{\mathfrak n}
\]
is a local subring of \(K\) dominating \(A\), and it contains \(x\). Since \(x\notin A\), this is again a strict extension of \(A\), contradicting maximality.

Therefore for every \(x\in K^\times\),
\[
x\in A \quad\text{or}\quad x^{-1}\in A.
\]
So \(A\) is a valuation ring.

\medskip

\textbf{Step 3: A valuation ring is maximal.}
Now assume that \(A\subseteq K\) is a valuation ring, and let \(B\in\Sigma\) dominate \(A\). We show that \(B=A\).

Take \(x\in B\). If \(x\notin A\), then since \(A\) is a valuation ring, we must have
\[
x^{-1}\in A.
\]
Because \(B\) dominates \(A\), its maximal ideal \(\mathfrak n\) satisfies
\[
\mathfrak m=A\cap \mathfrak n.
\]
Now \(x^{-1}\) cannot lie in \(\mathfrak n\), for otherwise
\[
1=xx^{-1}\in \mathfrak n,
\]
impossible. Hence
\[
x^{-1}\in A\setminus \mathfrak m.
\]
So \(x^{-1}\) is a unit in \(A\), and therefore \(x\in A\).

Thus every element of \(B\) lies in \(A\), so \(B\subseteq A\). Since \(A\subseteq B\), we get
\[
A=B.
\]

Therefore \(A\) is maximal in \(\Sigma\).

\medskip

Hence \(\Sigma\) has maximal elements, and a local subring \(A\subseteq K\) is maximal for domination if and only if \(A\) is a valuation ring of \(K\).

\section{Problem 6 --- The ordered value group of a valuation ring}

\textbf{Exact statement.}
Let \(A\) be a valuation ring of a field \(K\). Let \(U\) be the group of invertible elements of \(A\); this is a subgroup of
\[
K^\times=K\setminus\{0\},
\]
so that
\[
\Gamma:=K^\times/U
\]
is also a group.

\textbf{(a)} For \(\bar x,\bar y\in \Gamma\) represented by \(x,y\in K^\times\), write
\[
\bar x\ge \bar y \quad\text{if}\quad xy^{-1}\in A.
\]
Show that this defines a total ordering on \(\Gamma\) which respects the group structure, i.e. if \(\bar x\ge \bar y\), then \(\bar x\bar z\ge \bar y\bar z\).

\textbf{(b)} Let
\[
v:K^\times\to \Gamma
\]
be the quotient map. Show that
\[
v(x+y)\ge \min(v(x),v(y))
\]
for all \(x,y\in K^\times\) with \(x+y\neq 0\).

\textbf{Solution.}

\textbf{(a)} Define
\[
\bar x\ge \bar y \iff xy^{-1}\in A,
\]
where \(\bar x=xU\) and \(\bar y=yU\).

We first show that this is well defined. Suppose \(x'=xu\) and \(y'=yv\) with \(u,v\in U\). Then
\[
x'(y')^{-1}=xu(v^{-1}y^{-1})=xy^{-1}\cdot uv^{-1}.
\]
Since \(uv^{-1}\in U\subseteq A\) and also \((uv^{-1})^{-1}\in A\), multiplying by \(uv^{-1}\) does not affect membership in \(A\). Hence
\[
xy^{-1}\in A \iff x'(y')^{-1}\in A.
\]
So the relation is independent of the chosen representatives.

Reflexivity is immediate, since
\[
xx^{-1}=1\in A.
\]

For antisymmetry, assume \(\bar x\ge \bar y\) and \(\bar y\ge \bar x\). Then
\[
xy^{-1}\in A
\quad\text{and}\quad
yx^{-1}=(xy^{-1})^{-1}\in A.
\]
Therefore \(xy^{-1}\) is a unit of \(A\), so \(xy^{-1}\in U\). Hence
\[
xU=yU,
\]
that is, \(\bar x=\bar y\).

For transitivity, if \(\bar x\ge \bar y\) and \(\bar y\ge \bar z\), then
\[
xy^{-1}\in A,\qquad yz^{-1}\in A,
\]
so
\[
xz^{-1}=(xy^{-1})(yz^{-1})\in A.
\]
Thus \(\bar x\ge \bar z\).

For totality, let \(\bar x,\bar y\in \Gamma\). Since \(A\) is a valuation ring and \(xy^{-1}\in K^\times\), either
\[
xy^{-1}\in A
\quad\text{or}\quad
(xy^{-1})^{-1}=yx^{-1}\in A.
\]
So either \(\bar x\ge \bar y\) or \(\bar y\ge \bar x\).

Thus \(\ge\) is a total ordering on \(\Gamma\).

Finally, it respects the group structure: if \(\bar x\ge \bar y\), then \(xy^{-1}\in A\). For any \(\bar z\in \Gamma\),
\[
(xz)(yz)^{-1}=xy^{-1}\in A,
\]
hence
\[
\bar x\bar z\ge \bar y\bar z.
\]

\medskip

\textbf{(b)} Let
\[
v:K^\times\to \Gamma,\qquad v(x)=xU
\]
be the quotient map.

Take \(x,y\in K^\times\) with \(x+y\neq 0\). Since the order on \(\Gamma\) is total, either
\[
v(x)\ge v(y)
\quad\text{or}\quad
v(y)\ge v(x).
\]

Assume first that
\[
v(x)\ge v(y).
\]
By definition this means
\[
xy^{-1}\in A.
\]
Then
\[
(x+y)y^{-1}=xy^{-1}+1\in A,
\]
because both \(xy^{-1}\) and \(1\) lie in \(A\). Hence
\[
v(x+y)\ge v(y).
\]
Since in this case
\[
\min(v(x),v(y))=v(y),
\]
we get
\[
v(x+y)\ge \min(v(x),v(y)).
\]

The other case is symmetric. Therefore for all \(x,y\in K^\times\) with \(x+y\neq 0\),
\[
v(x+y)\ge \min(v(x),v(y)).
\]

\section{Problem 7 --- Constructing a valuation ring from a valuation}

\textbf{Exact statement.}
Let \(K\) be a field, \(\Gamma\) a totally ordered abelian group (written additively). A valuation of \(K\) with values in \(\Gamma\) is a map
\[
v:K^\times\to \Gamma
\]
such that:

\textbf{(a)}
\[
v(xy)=v(x)+v(y)
\]
for all \(x,y\in K^\times\);

\textbf{(b)}
\[
v(x+y)\ge \min(v(x),v(y))
\]
whenever \(x+y\neq 0\).

Show that
\[
A:=\{x\in K^\times\mid v(x)\ge 0\}\cup\{0\}
\]
is a valuation ring of \(K\), with maximal ideal
\[
\mathfrak m:=\{x\in K^\times\mid v(x)>0\}\cup\{0\}.
\]

\textbf{Solution.}
We prove first that \(A\) is a subring of \(K\).

By definition,
\[
0\in A.
\]
Also,
\[
v(1)=v(1\cdot 1)=v(1)+v(1),
\]
hence
\[
v(1)=0,
\]
so
\[
1\in A.
\]

Now let \(x,y\in A\).

For multiplication, if one of \(x,y\) is zero then \(xy=0\in A\). If both are nonzero, then
\[
v(x)\ge 0,\qquad v(y)\ge 0,
\]
so
\[
v(xy)=v(x)+v(y)\ge 0.
\]
Hence
\[
xy\in A.
\]

For addition, if \(x+y=0\), then clearly \(x+y\in A\). If \(x+y\neq 0\), then
\[
v(x+y)\ge \min(v(x),v(y))\ge 0,
\]
because \(v(x)\ge 0\) and \(v(y)\ge 0\). Thus
\[
x+y\in A.
\]

Therefore \(A\) is a subring of \(K\).

\medskip

Next we show that \(A\) is a valuation ring. Let \(t\in K^\times\). Since \(\Gamma\) is totally ordered, either
\[
v(t)\ge 0
\quad\text{or}\quad
v(t)<0.
\]
If \(v(t)\ge 0\), then \(t\in A\). If \(v(t)<0\), then
\[
v(t^{-1})=-v(t)>0,
\]
so in particular \(v(t^{-1})\ge 0\), hence \(t^{-1}\in A\).

Thus for every nonzero \(t\in K\),
\[
t\in A \quad\text{or}\quad t^{-1}\in A.
\]
So \(A\) is a valuation ring of \(K\).

\medskip

Now we identify the units of \(A\). Let \(x\in A\setminus\{0\}\).

If \(v(x)=0\), then
\[
v(x^{-1})=-v(x)=0,
\]
so \(x^{-1}\in A\). Hence \(x\) is a unit of \(A\).

Conversely, if \(x\) is a unit of \(A\), then \(x^{-1}\in A\). Thus
\[
v(x)\ge 0
\quad\text{and}\quad
v(x^{-1})=-v(x)\ge 0.
\]
Therefore
\[
v(x)=0.
\]

So the units of \(A\) are exactly the nonzero elements \(x\) with \(v(x)=0\). It follows that the nonunits are exactly the elements \(x\neq 0\) with \(v(x)>0\), together with \(0\).

Define
\[
\mathfrak m:=\{x\in K^\times\mid v(x)>0\}\cup\{0\}.
\]
We verify that \(\mathfrak m\) is an ideal of \(A\).

If \(x,y\in \mathfrak m\) and \(x+y\neq 0\), then
\[
v(x)>0,\qquad v(y)>0,
\]
so
\[
v(x+y)\ge \min(v(x),v(y))>0.
\]
Hence \(x+y\in \mathfrak m\). Also \(0\in \mathfrak m\), and if \(x+y=0\), then \(x+y\in \mathfrak m\) trivially.

If \(a\in A\) and \(x\in \mathfrak m\) are nonzero, then
\[
v(a)\ge 0,\qquad v(x)>0,
\]
hence
\[
v(ax)=v(a)+v(x)>0,
\]
so \(ax\in \mathfrak m\).

Thus \(\mathfrak m\) is an ideal of \(A\). It is proper because
\[
v(1)=0,
\]
so
\[
1\notin \mathfrak m.
\]

Finally, every element of \(A\setminus \mathfrak m\) has valuation \(0\), hence is a unit. Therefore \(\mathfrak m\) is the unique maximal ideal of \(A\).

So
\[
A:=\{x\in K^\times\mid v(x)\ge 0\}\cup\{0\}
\]
is a valuation ring of \(K\), and its maximal ideal is
\[
\mathfrak m:=\{x\in K^\times\mid v(x)>0\}\cup\{0\}.
\]
	
\section{Problem 8 --- Group rings of ordered abelian groups and the induced valuation}

\textbf{Exact statement.}
Let \(k\) be a field, and \(\Gamma\) any totally ordered abelian group. Let
\[
A:=k[\Gamma]
\]
denote the group ring of \(\Gamma\), i.e. \(k[\Gamma]\) is the \(k\)-vector space with basis \(\{z^\gamma\}_{\gamma\in\Gamma}\) and multiplication determined by
\[
z^\gamma\cdot z^{\gamma'}=z^{\gamma+\gamma'}.
\]

Show that \(A\) is an integral domain.

Define
\[
v_0:A\setminus\{0\}\to \Gamma
\]
by
\[
v_0\!\left(\sum_{i\in I}\alpha_i z^{\gamma_i}\right)=\min\{\gamma_i\mid i\in I\},
\]
where \(I\) is a finite index set and \(\alpha_i\in k\setminus\{0\}\) for each \(i\). Show that \(v_0\) satisfies conditions (a) and (b) of Question 7.

Now let \(K\) be the field of fractions of \(A\). Show that \(v_0\) can be extended to
\[
v:K^\times\to \Gamma
\]
so that \(v\) is a valuation with value group \(\Gamma\).

\textbf{Solution.}
Let
\[
f=\sum_{i=1}^r \alpha_i z^{\gamma_i}\neq 0,\qquad
g=\sum_{j=1}^s \beta_j z^{\delta_j}\neq 0,
\]
with all \(\alpha_i,\beta_j\neq 0\). Since the supports are finite and \(\Gamma\) is totally ordered, the minima
\[
\gamma_0:=\min_i\gamma_i,\qquad \delta_0:=\min_j\delta_j
\]
exist.

We first show that \(A\) is an integral domain. In the product
\[
fg=\sum_{i,j}\alpha_i\beta_j z^{\gamma_i+\delta_j},
\]
every exponent satisfies
\[
\gamma_i+\delta_j\ge \gamma_0+\delta_0,
\]
because \(\gamma_i\ge \gamma_0\) and \(\delta_j\ge \delta_0\), and the order on \(\Gamma\) is compatible with addition. Moreover, the equality
\[
\gamma_i+\delta_j=\gamma_0+\delta_0
\]
forces \(\gamma_i=\gamma_0\) and \(\delta_j=\delta_0\). Hence the coefficient of the lowest term \(z^{\gamma_0+\delta_0}\) is exactly
\[
\alpha_{i_0}\beta_{j_0}\neq 0
\]
for the unique indices \(i_0,j_0\) with \(\gamma_{i_0}=\gamma_0\) and \(\delta_{j_0}=\delta_0\). Therefore \(fg\neq 0\). So \(A\) is an integral domain.

Next we compute \(v_0(fg)\). From the previous paragraph, the lowest exponent in \(fg\) is \(\gamma_0+\delta_0\). Therefore
\[
v_0(fg)=\gamma_0+\delta_0=v_0(f)+v_0(g).
\]
So \(v_0\) satisfies condition (a).

Now let \(f,g\in A\setminus\{0\}\) and assume \(f+g\neq 0\). No exponent smaller than
\[
\min(v_0(f),v_0(g))
\]
can appear in \(f+g\), since all exponents occurring in \(f+g\) already occur in \(f\) or in \(g\). Hence
\[
v_0(f+g)\ge \min(v_0(f),v_0(g)).
\]
So \(v_0\) satisfies condition (b).

Now let \(K=\operatorname{Frac}(A)\). For
\[
x=\frac{f}{g}\in K^\times,\qquad f,g\in A\setminus\{0\},
\]
define
\[
v(x):=v_0(f)-v_0(g).
\]

We check that this is well defined. If
\[
\frac{f}{g}=\frac{f'}{g'},
\]
then
\[
fg'=f'g.
\]
Applying \(v_0\) and using multiplicativity,
\[
v_0(f)+v_0(g')=v_0(fg')=v_0(f'g)=v_0(f')+v_0(g),
\]
hence
\[
v_0(f)-v_0(g)=v_0(f')-v_0(g').
\]
So \(v\) is well defined.

It is multiplicative:
\[
v\!\left(\frac{f}{g}\cdot\frac{h}{\ell}\right)
=v\!\left(\frac{fh}{g\ell}\right)
=v_0(fh)-v_0(g\ell)
=(v_0(f)+v_0(h))-(v_0(g)+v_0(\ell))
=v(f/g)+v(h/\ell).
\]

For the inequality, let
\[
x=\frac{f}{g},\qquad y=\frac{h}{\ell},
\]
with \(x+y\neq 0\). Then
\[
x+y=\frac{f\ell+hg}{g\ell},
\]
so
\[
v(x+y)=v_0(f\ell+hg)-v_0(g\ell).
\]
Using the inequality for \(v_0\),
\[
v_0(f\ell+hg)\ge \min(v_0(f\ell),v_0(hg)),
\]
hence
\[
v(x+y)\ge \min(v_0(f)+v_0(\ell),\,v_0(h)+v_0(g))-(v_0(g)+v_0(\ell))
=\min(v(x),v(y)).
\]

Thus \(v:K^\times\to \Gamma\) is a valuation. Finally, for every \(\gamma\in\Gamma\),
\[
v(z^\gamma)=\gamma,
\]
so the value group is exactly \(\Gamma\).

\section{Problem 9 --- The image of the line \texorpdfstring{$1+x+y=0$}{1+x+y=0} under the valuation map}

\textbf{Exact statement.}
Take \(\Gamma=\mathbb R\) in the previous problem. Consider the ideal
\[
I=(1+x+y)\subseteq K[x,y],
\]
and let \(Z(I)\subseteq K^2\) be as defined in Question 4. Determine the image of
\[
Z(I)\cap (K^\times)^2
\]
under the map
\[
(K^\times)^2\to \mathbb R^2,\qquad (a,b)\mapsto (v(a),v(b)).
\]

\textbf{Solution.}
Let
\[
(a,b)\in Z(I)\cap (K^\times)^2.
\]
Then
\[
1+a+b=0.
\]
Set
\[
\alpha:=v(a),\qquad \beta:=v(b).
\]
The valuations of the three nonzero terms \(1,a,b\) are
\[
0,\alpha,\beta.
\]

We claim that the minimum of \(0,\alpha,\beta\) must be attained at least twice. Indeed, if one of the three were strictly smaller than the other two, say
\[
\alpha<\min(0,\beta),
\]
then from
\[
a=-(1+b)
\]
we would get
\[
v(a)=v(1+b)\ge \min(v(1),v(b))=\min(0,\beta)>\alpha,
\]
a contradiction. The same argument applies to the other possibilities.

Therefore necessarily
\[
(\alpha,\beta)\in
\{(\lambda,\lambda)\mid \lambda\le 0\}
\cup
\{(0,\mu)\mid \mu\ge 0\}
\cup
\{(\mu,0)\mid \mu\ge 0\}.
\]

Now we show that these points occur.

If \(\mu\ge 0\), let
\[
b=z^\mu,\qquad a=-1-z^\mu.
\]
Then
\[
1+a+b=0,
\qquad
v(b)=\mu,
\qquad
v(a)=\min(0,\mu)=0.
\]
So \((0,\mu)\) lies in the image.

Similarly, taking
\[
a=z^\mu,\qquad b=-1-z^\mu,
\]
we get \((\mu,0)\) in the image.

If \(\lambda<0\), take
\[
a=z^\lambda,\qquad b=-1-z^\lambda.
\]
Then
\[
1+a+b=0,
\qquad
v(a)=\lambda,
\qquad
v(b)=\min(0,\lambda)=\lambda.
\]
So \((\lambda,\lambda)\) lies in the image.

If the field \(k\) contains nonzero elements \(c,d\) such that
\[
1+c+d=0
\]
(for example any field with more than two elements), then taking \(a=c\), \(b=d\) gives the point \((0,0)\).

Hence in the standard situation the image is the tropical line
\[
\{(\lambda,\lambda)\mid \lambda\le 0\}
\cup
\{(0,\mu)\mid \mu\ge 0\}
\cup
\{(\mu,0)\mid \mu\ge 0\}.
\]

\medskip

If \(k=\mathbb F_2\), then \((0,0)\) is not attained. Indeed, if \(v(a)=v(b)=0\), then the residues \(\bar a,\bar b\in k^\times\) satisfy
\[
1+\bar a+\bar b=0,
\]
but in \(\mathbb F_2^\times=\{1\}\) this becomes
\[
1+1+1=1\neq 0.
\]
So over \(k=\mathbb F_2\), the image is the same union of three rays with the vertex removed.

\section{Problem 10 --- When is a valuation ring Noetherian?}

\textbf{Exact statement.}
Let \(K\) be a field, \(A\subseteq K\) a valuation ring of \(K\). Show that \(A\) is Noetherian if and only if either \(A=K\) or the value group of \(A\) is \(\mathbb Z\). In this case we say \(A\) is a discrete valuation ring.

\textbf{Solution.}
Let \(\Gamma\) be the value group of \(A\), written additively.

We first assume that \(A\) is Noetherian. If \(A=K\), there is nothing to prove. So assume \(A\neq K\). Then its maximal ideal \(\mathfrak m\) is nonzero.

We use the fact that every finitely generated ideal in a valuation ring is principal. Indeed, let
\[
I=(x_1,\dots,x_n).
\]
Choose an index \(j\) such that \(v(x_j)\) is minimal among the \(v(x_i)\). Since ideals in a valuation ring are linearly ordered, each \(x_i\in (x_j)\), hence
\[
I=(x_j).
\]

Since \(A\) is Noetherian, \(\mathfrak m\) is finitely generated, hence principal:
\[
\mathfrak m=(t)
\]
for some \(t\in A\) with \(v(t)>0\). Set
\[
\delta:=v(t).
\]
We claim that \(\delta\) is the smallest positive element of \(\Gamma\).

Let \(\gamma>0\) in \(\Gamma\), and choose \(x\in K^\times\) with \(v(x)=\gamma\). Then \(x\in \mathfrak m\), so since \(\mathfrak m=(t)\), we can write
\[
x=ta
\]
with \(a\in A\). Therefore
\[
\gamma=v(x)=v(t)+v(a)\ge v(t)=\delta.
\]
So \(\delta\) is minimal positive.

Now let \(\gamma>0\). Choose \(n\in\mathbb Z_{\ge 0}\) such that
\[
n\delta\le \gamma < (n+1)\delta.
\]
Then
\[
0\le \gamma-n\delta<\delta.
\]
By minimality of \(\delta\), this forces
\[
\gamma-n\delta=0,
\]
so
\[
\gamma=n\delta.
\]
Hence every positive element of \(\Gamma\) is an integer multiple of \(\delta\), and therefore
\[
\Gamma=\mathbb Z\delta\cong \mathbb Z.
\]

Conversely, suppose either \(A=K\) or \(\Gamma\cong \mathbb Z\). If \(A=K\), then \(A\) is a field, hence Noetherian.

So assume \(\Gamma\cong \mathbb Z\). Choose \(t\in A\) with
\[
v(t)=1
\]
after identifying \(\Gamma\) with \(\mathbb Z\). Let \(x\in A\setminus\{0\}\). Then \(v(x)=n\ge 0\), so
\[
v(x/t^n)=0.
\]
Hence \(x/t^n\) is a unit of \(A\), which means
\[
x=ut^n
\]
for some unit \(u\in A^\times\).

Thus every nonzero principal ideal is of the form
\[
(x)=(t^n).
\]
Now let \(I\subseteq A\) be a nonzero ideal. The set of values
\[
\{v(x)\mid x\in I,\ x\neq 0\}\subseteq \mathbb Z_{\ge 0}
\]
has a least element \(m\). Choose \(x\in I\) with \(v(x)=m\). Then for every \(y\in I\),
\[
v(y)\ge m,
\]
so \(y\in (x)\). Hence
\[
I=(x)=(t^m).
\]
Therefore every ideal of \(A\) is principal, so \(A\) is Noetherian.

We conclude that a valuation ring \(A\subseteq K\) is Noetherian if and only if either \(A=K\) or its value group is \(\mathbb Z\). In the non-field case this is exactly a discrete valuation ring.

\section{1. Problem 1 theory: finitely presented modules, localization, and \texorpdfstring{$\operatorname{Hom}$}{Hom}}

\subsection*{1.1. Modules}

Let \(A\) be a ring. An \emph{\(A\)-module} is an abelian group \(M\) equipped with scalar multiplication
\[
A\times M \to M,\qquad (a,m)\mapsto am,
\]
satisfying the usual distributive, associative, and unit axioms.

Examples:
\begin{itemize}
	\item \(A^n\),
	\item ideals \(I\subseteq A\),
	\item quotient modules \(A/I\),
	\item vector spaces when \(A\) is a field.
\end{itemize}

Modules are the natural generalization of vector spaces where scalars come from a ring instead of a field.

\subsection*{1.2. Finitely generated and finitely presented modules}

An \(A\)-module \(M\) is called \emph{finitely generated} if there exists a surjective homomorphism
\[
A^n \twoheadrightarrow M
\]
for some integer \(n\ge 0\).

It is called \emph{finitely presented} if there exists an exact sequence
\[
A^{n_1}\longrightarrow A^{n_2}\longrightarrow M\longrightarrow 0
\]
for some integers \(n_1,n_2\ge 0\).

Thus finitely presented means:

\begin{itemize}
	\item \(M\) has finitely many generators, and
	\item the relations among those generators are also generated by finitely many relations.
\end{itemize}

So finitely presented is stronger than finitely generated in general.

If \(A\) is Noetherian, then for many natural classes of modules the distinction becomes less serious, but in general it matters a lot.

\subsection*{1.3. Presentations and relation matrices}

A presentation
\[
A^m \xrightarrow{\varphi} A^n \to M \to 0
\]
means that \(M\) is obtained from the free module \(A^n\) by quotienting out the image of \(\varphi\). In coordinates, \(\varphi\) can often be represented by a matrix over \(A\). So finitely presented modules are exactly those that can be described by finitely sized matrices.

This is one of the reasons why they are so important in algebra and geometry.

\subsection*{1.4. Localization of rings}

Let \(S\subseteq A\) be a multiplicatively closed subset. The localization \(S^{-1}A\) is the ring obtained by formally inverting the elements of \(S\).

An element of \(S^{-1}A\) is written as
\[
\frac{a}{s},\qquad a\in A,\ s\in S.
\]

Typical examples:
\begin{itemize}
	\item \(\mathbb{Q}=S^{-1}\mathbb{Z}\), where \(S=\mathbb{Z}\setminus\{0\}\),
	\item \(A_{\mathfrak p}\), where \(S=A\setminus \mathfrak p\),
	\item \(k[x]_x\), where powers of \(x\) are inverted.
\end{itemize}

Intuitively, localization means:
\begin{quote}
	``ignore information away from \(S\) and force elements of \(S\) to become invertible.''
\end{quote}

\subsection*{1.5. Localization of modules}

If \(M\) is an \(A\)-module, then its localization \(S^{-1}M\) is defined by fractions
\[
\frac{m}{s},\qquad m\in M,\ s\in S.
\]
This becomes an \(S^{-1}A\)-module.

Important facts:
\begin{itemize}
	\item localization is an exact functor,
	\item localization commutes with finite direct sums,
	\item a presentation of \(M\) localizes to a presentation of \(S^{-1}M\).
\end{itemize}

\subsection*{1.6. The \texorpdfstring{$\operatorname{Hom}$}{Hom} functor}

For \(A\)-modules \(M,N\), the set of \(A\)-linear maps \(M\to N\) is denoted
\[
\operatorname{Hom}_A(M,N).
\]

If \(A\) is commutative, this is naturally an \(A\)-module.

Important special cases:
\[
\operatorname{Hom}_A(A,N)\cong N,
\qquad
\operatorname{Hom}_A(A^n,N)\cong N^n.
\]

The functor \(\operatorname{Hom}_A(-,N)\) is \emph{contravariant} and \emph{left exact}. That is exactly why kernels appear in Problem 1.

\subsection*{1.7. Why finite presentation matters in Problem 1}

The isomorphism
\[
S^{-1}\operatorname{Hom}_A(M,N)\cong \operatorname{Hom}_{S^{-1}A}(S^{-1}M,S^{-1}N)
\]
is not true for arbitrary \(M\). The crucial hypothesis is that \(M\) is finitely presented.

Why? Because if \(M\) has a finite presentation
\[
A^{n_1}\to A^{n_2}\to M\to 0,
\]
then \(\operatorname{Hom}_A(M,N)\) can be realized as the kernel of a map
\[
N^{n_2}\to N^{n_1},
\]
and localization preserves such kernels because it is exact and commutes with finite direct sums.

So Problem 1 is really teaching the principle:
\begin{quote}
	finite presentation is the correct finiteness condition that allows \(\operatorname{Hom}\) to commute with localization.
\end{quote}

\section{2. Problems 2--3 theory: integral dependence, integrally closed rings, and normalization}

\subsection*{2.1. Integral elements}

Let \(A\subseteq B\) be rings. An element \(b\in B\) is called \emph{integral over \(A\)} if it satisfies a monic polynomial equation
\[
b^n+a_{n-1}b^{n-1}+\cdots+a_1b+a_0=0
\qquad\text{with }a_i\in A.
\]

This is the ring-theoretic analog of being algebraic over a field.

Examples:
\begin{itemize}
	\item every element of \(A\) is integral over \(A\),
	\item algebraic integers are exactly the elements integral over \(\mathbb{Z}\).
\end{itemize}

\subsection*{2.2. Integral extensions}

If every element of \(B\) is integral over \(A\), then \(B\) is called an \emph{integral extension} of \(A\).

Integral extensions behave in many ways like algebraic field extensions. In particular, prime ideals behave well:
\begin{itemize}
	\item primes in \(B\) lie over primes in \(A\),
	\item chains of prime ideals can be compared between \(A\) and \(B\),
	\item much of the geometry of \(\operatorname{Spec}(B)\) maps in a controlled way onto \(\operatorname{Spec}(A)\).
\end{itemize}

\subsection*{2.3. Integrally closed domains}

Let \(A\) be an integral domain with fraction field \(K\). Then \(A\) is called \emph{integrally closed} if every element of \(K\) that is integral over \(A\) already lies in \(A\).

Equivalently, the integral closure of \(A\) inside \(K\) is just \(A\) itself.

A domain that is integrally closed is also called \emph{normal}.

\subsection*{2.4. Integral closure and normalization}

The \emph{integral closure} of \(A\) in \(K=\operatorname{Frac}(A)\) is the set
\[
\overline{A}
=
\{x\in K \mid x \text{ is integral over } A\}.
\]

This is again a ring, and is called the \emph{normalization} of \(A\).

Geometrically, normalization often removes singularities of affine varieties in a controlled way.

\subsection*{2.5. UFDs are integrally closed}

A \emph{unique factorization domain} is a domain in which every nonzero nonunit factors uniquely into irreducibles, up to order and units.

A fundamental theorem states:
\begin{quote}
	Every UFD is integrally closed.
\end{quote}

The proof idea is very important. If
\[
z=\frac{a}{b}\in \operatorname{Frac}(A)
\]
is integral over \(A\), with \(a,b\in A\) coprime, then from the monic equation satisfied by \(z\) one deduces
\[
b\mid a^n.
\]
Using unique factorization and the assumption that \(a\) and \(b\) are coprime, one concludes that \(b\) must be a unit. Therefore \(z\in A\).

So unique factorization gives a strong divisibility control, and that is exactly what makes the ring integrally closed.

Examples of UFDs:
\[
\mathbb{Z},\qquad k[x],\qquad k[x_1,\dots,x_n].
\]

\subsection*{2.6. The cusp ring \texorpdfstring{$k[x,y]/(y^2-x^3)$}{k[x,y]/(y\string^2-x\string^3)}}

The ring
\[
A=k[x,y]/(y^2-x^3)
\]
is one of the standard first examples of a non-normal domain.

It is isomorphic to
\[
k[t^2,t^3]\subseteq k[t]
\]
via
\[
x\mapsto t^2,\qquad y\mapsto t^3.
\]

Geometrically, this is the coordinate ring of the cusp
\[
y^2=x^3.
\]

Now \(t\) belongs to the fraction field \(k(t)\), and it is integral over \(k[t^2,t^3]\) because it satisfies
\[
T^2-x=0.
\]
But \(t\notin k[t^2,t^3]\).

So the cusp ring is not normal.

Its normalization is
\[
k[t].
\]

Geometrically, the map
\[
\operatorname{Spec}(k[t])\to \operatorname{Spec}(k[t^2,t^3])
\]
parametrizes the cusp and resolves the singular behavior of the coordinate ring.

\subsection*{2.7. Problem 3: multiplicatively closed complements}

Problem 3 gives a ring-theoretic criterion for integral closedness.

Suppose \(A\subseteq B\) and \(B\setminus A\) is multiplicatively closed. This means:
\begin{quote}
	if two elements lie outside \(A\), then their product also lies outside \(A\).
\end{quote}

That is a very strong closure property.

Now suppose \(b\in B\setminus A\) were integral over \(A\). Then \(b\) would satisfy a monic equation
\[
b^n+a_{n-1}b^{n-1}+\cdots+a_0=0.
\]
Rearranging it, one can isolate a product of two elements outside \(A\) and force it to lie in \(A\), contradicting the multiplicative closure of the complement.

So Problem 3 teaches a subtle but useful principle:
\begin{quote}
	strong multiplicative separation between \(A\) and \(B\setminus A\) can force \(A\) to be integrally closed in \(B\).
\end{quote}

\section{3. Problem 4 theory: polynomial rings, algebraic sets, radicals, and Nullstellensatz}

\subsection*{3.1. Polynomial rings as coordinate rings}

Let
\[
A=k[x_1,\dots,x_n]
\]
for a field \(k\).

This ring is the coordinate ring of affine \(n\)-space
\[
\mathbb{A}_k^n.
\]

A polynomial
\[
f(x_1,\dots,x_n)\in k[x_1,\dots,x_n]
\]
can be evaluated at a point
\[
(a_1,\dots,a_n)\in k^n.
\]

Thus polynomials are simultaneously:
\begin{itemize}
	\item algebraic expressions inside a ring,
	\item functions on affine space.
\end{itemize}

That double role is the basis of affine algebraic geometry.

\subsection*{3.2. Zero sets of ideals}

If \(I\subseteq A\) is an ideal, define
\[
Z(I)=\{a\in k^n \mid f(a)=0\text{ for all }f\in I\}.
\]

This is the set of common zeros of all polynomials in \(I\).

So an ideal gives a system of polynomial equations, and \(Z(I)\) is the geometric set of solutions.

\subsection*{3.3. Vanishing ideals of subsets}

If \(X\subseteq k^n\), define
\[
I(X)=\{f\in A\mid f(a)=0\text{ for all }a\in X\}.
\]

This is an ideal of \(A\). It records all polynomial equations satisfied by every point of \(X\).

Thus we have two operations:
\[
I \mapsto Z(I),\qquad X \mapsto I(X).
\]

These are the two basic directions of the algebra--geometry dictionary.

\subsection*{3.4. Radical ideals}

For an ideal \(I\subseteq A\), its \emph{radical} is
\[
\sqrt{I}
=
\{f\in A\mid f^m\in I\text{ for some }m\ge 1\}.
\]

An ideal \(I\) is called \emph{radical} if
\[
I=\sqrt{I}.
\]

Geometric meaning:
\[
Z(I)=Z(\sqrt{I}).
\]
Indeed, over a field,
\[
f(a)^m=0 \iff f(a)=0.
\]

So the geometry of the zero set only depends on the radical of the ideal, not on nilpotent thickening.

\subsection*{3.5. Weak Hilbert Nullstellensatz}

If \(k\) is algebraically closed and \(I\neq A\), then
\[
Z(I)\neq \varnothing.
\]

Equivalently, every maximal ideal of
\[
k[x_1,\dots,x_n]
\]
has the form
\[
(x_1-a_1,\dots,x_n-a_n)
\]
for some point \((a_1,\dots,a_n)\in k^n\).

This says:
\begin{quote}
	maximal ideals correspond exactly to \(k\)-rational points of affine \(n\)-space.
\end{quote}

\subsection*{3.6. Strong Hilbert Nullstellensatz}

The strong form says
\[
I(Z(I))=\sqrt{I}.
\]

This is one of the most important theorems in algebraic geometry.

It means:
\begin{itemize}
	\item start with equations \(I\),
	\item take their common zero set \(Z(I)\),
	\item then collect all polynomials vanishing on that set.
\end{itemize}

You recover exactly the radical of \(I\).

So affine algebraic sets over an algebraically closed field correspond precisely to radical ideals.

\subsection*{3.7. Geometric meaning of Problem 4}

Problem 4 is the precise statement that affine algebraic geometry is controlled by polynomial rings and radical ideals.

It is the algebraic foundation of the correspondence
\[
\{\text{affine algebraic subsets of }k^n\}
\longleftrightarrow
\{\text{radical ideals of }k[x_1,\dots,x_n]\}.
\]

This is the starting point of the functorial viewpoint
\[
X \longleftrightarrow k[X].
\]

\section{4. Problem 5 theory: local rings, domination, and valuation rings}

\subsection*{4.1. Local rings}

A ring \(A\) is called \emph{local} if it has a unique maximal ideal \(\mathfrak m\).

Equivalent characterizations:
\begin{itemize}
	\item the nonunits of \(A\) form an ideal,
	\item that ideal is the unique maximal ideal.
\end{itemize}

Examples:
\[
k[x]_{(x)},\qquad \mathbb{Z}_{(p)},\qquad \text{a field}.
\]

Local rings model what happens ``near one point.''

\subsection*{4.2. Domination of local rings}

If \((A,\mathfrak m)\) and \((B,\mathfrak n)\) are local rings, then \(B\) is said to \emph{dominate} \(A\) if
\begin{itemize}
	\item \(A\subseteq B\), and
	\item \(\mathfrak m=A\cap \mathfrak n\).
\end{itemize}

Intuitively, \(B\) is a larger local ring sitting over the same local point.

This is the natural order relation among local subrings of a field used in Problem 5.

\subsection*{4.3. Valuation rings}

Let \(K\) be a field. A subring \(V\subseteq K\) is called a \emph{valuation ring} if for every nonzero \(x\in K\),
\[
x\in V \quad\text{or}\quad x^{-1}\in V.
\]

This means that every nonzero element of the field can be compared with the ring: either the element is ``small enough'' to lie in the ring, or its inverse is.

Equivalent viewpoint:
the ideals of a valuation ring are totally ordered by inclusion.

So valuation rings are extremely rigid local rings.

Examples:
\begin{itemize}
	\item fields,
	\item \(\mathbb{Z}_{(p)}\),
	\item \(k[[t]]\),
	\item more exotic higher-rank valuation rings.
\end{itemize}

\subsection*{4.4. Maximal local subrings of a field}

Problem 5 shows that if one orders local subrings of a field \(K\) by domination, then maximal elements exist and are exactly the valuation rings.

This is conceptually very important.

Why should maximal local subrings be valuation rings? Because if \(x\) and \(x^{-1}\) were both outside the ring, then one could enlarge the ring by adjoining one of them and localizing appropriately. That would create a strictly larger local ring dominating the original one, contradicting maximality.

So valuation rings are precisely the ``largest possible'' local subrings of a field.

This gives a structural interpretation of valuation rings independent of formulas or value groups.

\section{5. Problems 6--7 theory: value groups and abstract valuations}

\subsection*{5.1. The unit group and the quotient \texorpdfstring{$K^\times/U$}{K*/U}}

Let \(A\) be a valuation ring of a field \(K\), and let
\[
U=A^\times
\]
be the group of units of \(A\).

Then the quotient
\[
\Gamma=K^\times/U
\]
measures nonzero elements of \(K\) up to multiplication by units.

Two elements of \(K^\times\) define the same class in \(\Gamma\) exactly when they differ by a unit of \(A\), that is, when they have the same ``size'' from the valuation point of view.

\subsection*{5.2. Order on the value group}

One defines an order on \(\Gamma\) by
\[
\overline{x}\ge \overline{y}
\quad\Longleftrightarrow\quad
xy^{-1}\in A.
\]

This gives \(\Gamma\) the structure of a totally ordered abelian group.

Thus every valuation ring naturally determines:
\begin{itemize}
	\item a totally ordered abelian group \(\Gamma\),
	\item a quotient map
	\[
	v:K^\times\to \Gamma,
	\]
	which behaves like a valuation.
\end{itemize}

\subsection*{5.3. Abstract valuations}

More generally, let \(K\) be a field and \(\Gamma\) a totally ordered abelian group written additively.

A map
\[
v:K^\times\to \Gamma
\]
is called a \emph{valuation} if it satisfies:
\begin{align*}
	v(xy)&=v(x)+v(y),\\
	v(x+y)&\ge \min(v(x),v(y))
	\qquad\text{whenever }x+y\neq 0.
\end{align*}

Often one extends \(v\) by defining
\[
v(0)=+\infty,
\]
but the problems here do not require that notation.

\subsection*{5.4. Meaning of the valuation inequality}

The inequality
\[
v(x+y)\ge \min(v(x),v(y))
\]
is the non-Archimedean triangle inequality written additively.

It says:
\begin{quote}
	the sum of two elements cannot have value smaller than the smaller of the two values.
\end{quote}

Cancellation may make the valuation larger, but never smaller.

This is one of the deepest algebraic differences between valuation theory and ordinary absolute value geometry.

\subsection*{5.5. Recovering the valuation ring from the valuation}

Given a valuation
\[
v:K^\times\to \Gamma,
\]
define
\[
A=\{x\in K^\times\mid v(x)\ge 0\}\cup\{0\}.
\]

Then \(A\) is a valuation ring of \(K\).

Its maximal ideal is
\[
\mathfrak m=\{x\in K^\times\mid v(x)>0\}\cup\{0\}.
\]

Its units are exactly the elements with value \(0\).

So one passes back and forth naturally between:
\begin{itemize}
	\item valuation rings,
	\item valuations,
	\item ordered abelian value groups.
\end{itemize}

This explains the statement in Problem 7 that one speaks interchangeably of valuations and valuation rings.

\section{6. Problem 8 theory: ordered group rings and constructing valuations}

\subsection*{6.1. Group rings}

Let \(\Gamma\) be an abelian group and \(k\) a field. The group ring
\[
k[\Gamma]
\]
is the \(k\)-vector space with basis \(\{z^\gamma\}_{\gamma\in\Gamma}\), where multiplication is determined by
\[
z^\gamma z^\delta = z^{\gamma+\delta}.
\]

So an element of \(k[\Gamma]\) is a finite sum
\[
\sum_{\gamma\in\Gamma}\alpha_\gamma z^\gamma,
\qquad
\alpha_\gamma\in k.
\]

Examples:
\begin{itemize}
	\item if \(\Gamma=\mathbb{Z}\), then
	\[
	k[\Gamma]\cong k[z,z^{-1}],
	\]
	the Laurent polynomial ring;
	\item if \(\Gamma=\mathbb{N}\), one gets the ordinary polynomial ring \(k[z]\), though strictly speaking \(\mathbb{N}\) is a monoid rather than a group.
\end{itemize}

\subsection*{6.2. Why order matters}

If \(\Gamma\) is totally ordered, then every nonzero finite sum
\[
\sum \alpha_i z^{\gamma_i}
\]
has a smallest exponent.

That plays the role of the lowest term of a polynomial.

When two such sums are multiplied, the lowest exponent of the product is the sum of the lowest exponents of the two factors. Because the order is total, that lowest term cannot cancel accidentally.

This is why \(k[\Gamma]\) becomes an integral domain.

\subsection*{6.3. The lowest-term map}

Define
\[
v_0\left(\sum \alpha_i z^{\gamma_i}\right)=\min\{\gamma_i\}.
\]

This is the analogue of the order of vanishing of a power series, or the least exponent occurring in a Laurent polynomial.

Then one proves:
\begin{align*}
	v_0(fg)&=v_0(f)+v_0(g),\\
	v_0(f+g)&\ge \min(v_0(f),v_0(g)).
\end{align*}

So \(v_0\) already behaves like a valuation on the integral domain \(A=k[\Gamma]\).

\subsection*{6.4. Extension to the field of fractions}

Once \(A\) is known to be a domain, it has a fraction field
\[
K=\operatorname{Frac}(A).
\]

Then the valuation extends by the standard formula
\[
v\left(\frac{f}{g}\right)=v_0(f)-v_0(g).
\]

This is the natural extension principle:
a multiplicative order function on a domain extends to its fraction field by subtraction.

The resulting valuation has value group exactly \(\Gamma\), because
\[
v(z^\gamma)=\gamma.
\]

So Problem 8 constructs valued fields with prescribed ordered abelian value group.

This is a very important existence theorem in valuation theory.

\bigskip

\section{7. Problem 9 theory: tropicalization of the equation \texorpdfstring{$1+x+y=0$}{1+x+y=0}}

\subsection*{7.1. Tropicalization}

Given a valued field \((K,v)\), one can send a point
\[
(a_1,\dots,a_n)\in (K^\times)^n
\]
to its value vector
\[
(v(a_1),\dots,v(a_n)).
\]

Applying this to algebraic sets produces piecewise-linear subsets of \(\mathbb{R}^n\) or \(\Gamma^n\), called tropicalizations.

This is one of the central ideas of tropical geometry.

\subsection*{7.2. The equation \texorpdfstring{$1+x+y=0$}{1+x+y=0}}

Suppose
\[
1+a+b=0.
\]
Then the three nonzero terms have valuations
\[
0,\qquad v(a),\qquad v(b).
\]

A basic principle of valuation theory says:
\begin{quote}
	in any relation \(u_1+\cdots+u_r=0\), the smallest valuation among the \(u_i\) must occur at least twice.
\end{quote}

Indeed, if one term had strictly smaller valuation than all the others, then it could not be cancelled by their sum.

For the equation
\[
1+a+b=0,
\]
this forces the minimum of
\[
0,\ v(a),\ v(b)
\]
to occur at least twice.

That gives exactly the tropical line:
\[
\{(\lambda,\lambda)\mid \lambda\le 0\}
\;\cup\;
\{(0,\mu)\mid \mu\ge 0\}
\;\cup\;
\{(\mu,0)\mid \mu\ge 0\}.
\]

\subsection*{7.3. Why Problem 9 matters}

Problem 9 is the first clear appearance of a major modern principle:
\begin{quote}
	classical polynomial equations become piecewise-linear balancing conditions after applying a valuation.
\end{quote}

So valuation theory does not just produce local rings; it also produces combinatorial geometry.

This is the doorway to tropical algebraic geometry.

\bigskip

\section{8. Problem 10 theory: Noetherian valuation rings and discrete valuation rings}

\subsection*{8.1. Noetherian rings}

A ring \(A\) is called \emph{Noetherian} if every ideal of \(A\) is finitely generated.

Equivalent condition:
every ascending chain of ideals
\[
I_1\subseteq I_2\subseteq I_3\subseteq \cdots
\]
stabilizes.

This is one of the main finiteness conditions of commutative algebra.

\subsection*{8.2. Ideals in valuation rings}

Valuation rings have a very special structure: their ideals are totally ordered by inclusion.

This immediately gives a useful fact:
\begin{quote}
	every finitely generated ideal in a valuation ring is principal.
\end{quote}

Indeed, if
\[
I=(x_1,\dots,x_n),
\]
choose one generator with smallest value. Then all the others belong to the ideal generated by that one, so \(I\) is principal.

Thus valuation rings are not always principal ideal rings, but finitely generated ideals behave much more simply than in a general ring.

\subsection*{8.3. Why Noetherian forces discreteness}

Suppose \(A\) is a valuation ring and Noetherian, and \(A\neq K\). Then its maximal ideal \(\mathfrak m\) is finitely generated, hence principal:
\[
\mathfrak m=(t).
\]

Then \(v(t)\) is the smallest positive value in the value group. Once such a smallest positive element exists, every positive value must be an integer multiple of it. Hence the value group is isomorphic to
\[
\mathbb{Z}.
\]

So Noetherian valuation rings are forced to have discrete value group.

\subsection*{8.4. Discrete valuation rings}

A \emph{discrete valuation ring} (DVR) is a valuation ring whose value group is \(\mathbb{Z}\), equivalently a non-field local principal valuation ring.

Typical examples:
\[
k[[t]],\qquad \mathbb{Z}_{(p)}.
\]

If \(\pi\) is an element of valuation \(1\), called a \emph{uniformizer}, then every nonzero element \(x\) of the field has the form
\[
x=u\pi^n
\]
with \(u\) a unit and \(n\in \mathbb{Z}\).

Every nonzero ideal is of the form
\[
(\pi^n).
\]

So a DVR is the cleanest possible nontrivial valuation ring.

\subsection*{8.5. Meaning of Problem 10}

Problem 10 identifies exactly which valuation rings belong to the classical Noetherian world:
\begin{itemize}
	\item either the valuation ring is already a field,
	\item or it is a discrete valuation ring.
\end{itemize}

So among all valuation rings, the Noetherian ones are precisely the simplest rank-one discrete ones.

This is why DVRs occur constantly in:
\begin{itemize}
	\item algebraic geometry,
	\item number theory,
	\item local algebra,
	\item divisor theory.
\end{itemize}

\section{9. Compact dictionary for Problems 1--10}

\subsection*{9.1. Ring-theoretic notions}

\begin{itemize}
	\item \textbf{Integral domain:} a commutative ring with \(1\) and no zero divisors.
	\item \textbf{Fraction field:} the field of fractions of a domain.
	\item \textbf{UFD:} a unique factorization domain.
	\item \textbf{Integral over \(A\):} satisfies a monic polynomial with coefficients in \(A\).
	\item \textbf{Integrally closed:} every element of the fraction field integral over the ring already lies in the ring.
	\item \textbf{Normal domain:} an integrally closed domain.
	\item \textbf{Normalization:} the integral closure in the fraction field.
	\item \textbf{Local ring:} a ring with exactly one maximal ideal.
	\item \textbf{Noetherian ring:} a ring in which every ideal is finitely generated.
\end{itemize}

\subsection*{9.2. Module-theoretic notions}

\begin{itemize}
	\item \textbf{Finitely generated module:} a quotient of \(A^n\) for some \(n\).
	\item \textbf{Finitely presented module:} the cokernel of a map \(A^{n_1}\to A^{n_2}\).
	\item \textbf{\(\operatorname{Hom}_A(M,N)\):} the module of \(A\)-linear maps \(M\to N\).
	\item \textbf{Localization:} the process of inverting a multiplicatively closed set \(S\).
\end{itemize}

\subsection*{9.3. Geometric notions}

\begin{itemize}
	\item \textbf{\(Z(I)\):} the common zero set of an ideal \(I\).
	\item \textbf{\(I(X)\):} the ideal of all polynomials vanishing on a subset \(X\subseteq k^n\).
	\item \textbf{Radical \(\sqrt{I}\):} all \(f\) such that \(f^m\in I\) for some \(m\).
	\item \textbf{Nullstellensatz:} over an algebraically closed field,
	\[
	I(Z(I))=\sqrt{I}.
	\]
\end{itemize}

\subsection*{9.4. Valuation-theoretic notions}

\begin{itemize}
	\item \textbf{Valuation ring:} a subring \(A\subseteq K\) such that for every \(x\in K^\times\),
	\[
	x\in A \quad\text{or}\quad x^{-1}\in A.
	\]
	\item \textbf{Domination:} a local ring extension preserving maximal ideals.
	\item \textbf{Value group:} \(K^\times/A^\times\), or abstractly an ordered abelian group \(\Gamma\).
	\item \textbf{Valuation:} a map
	\[
	v:K^\times\to \Gamma
	\]
	satisfying
	\[
	v(xy)=v(x)+v(y),\qquad v(x+y)\ge \min(v(x),v(y)).
	\]
	\item \textbf{Maximal ideal of a valuation ring:}
	\[
	\mathfrak m=\{x\mid v(x)>0\}\cup\{0\}.
	\]
	\item \textbf{DVR:} a valuation ring with value group \(\mathbb{Z}\).
\end{itemize}

\section{10. Final conceptual summary}

Problems 1--10 form a coherent chapter in commutative algebra.

\begin{itemize}
	\item \textbf{Problem 1} shows how finite algebraic data behaves well under localization.
	\item \textbf{Problems 2--3} explain when a ring already contains all integral elements that ought to belong to it.
	\item \textbf{Problem 4} builds the algebra--geometry dictionary using ideals and zero sets.
	\item \textbf{Problems 5--7} show that valuation rings are the extremal local subrings of a field, controlled by ordered groups.
	\item \textbf{Problem 8} constructs valuations explicitly from ordered algebra.
	\item \textbf{Problem 9} shows how valuations turn polynomial equations into piecewise-linear geometry.
	\item \textbf{Problem 10} identifies the Noetherian case: fields and discrete valuation rings.
\end{itemize}

Thus the chapter moves through the chain
\[
\text{module theory}
\;\to\;
\text{integrality}
\;\to\;
\text{algebraic geometry}
\;\to\;
\text{valuation theory}
\;\to\;
\text{discrete and tropical structure}.
\]

This is exactly one of the central roads from classical commutative algebra into modern algebraic geometry.

\chapter{Solved Problems 11--17}

\section{Problem 11 --- A direct product ring gives projective modules that are not free}

\textbf{Exact statement.}
Let
\[
A=A_1\times A_2
\]
be a direct product ring. Show that the \(A\)-modules
\[
M_1=A_1\times 0,\qquad M_2=0\times A_2
\]
are projective. As they are not free, this shows that projective modules need not be free.

\textbf{Solution.}
Recall that an \(A\)-module is projective if it is a direct summand of a free \(A\)-module.

As an \(A\)-module over itself,
\[
A=A_1\times A_2.
\]
But clearly
\[
A=(A_1\times 0)\oplus (0\times A_2)=M_1\oplus M_2.
\]
Indeed, every element \((a_1,a_2)\in A\) can be written uniquely as
\[
(a_1,a_2)=(a_1,0)+(0,a_2),
\]
with \((a_1,0)\in M_1\) and \((0,a_2)\in M_2\).

Since \(A\) is a free \(A\)-module of rank \(1\), and \(M_1,M_2\) are direct summands of \(A\), it follows that both \(M_1\) and \(M_2\) are projective \(A\)-modules.

Equivalently, if we set
\[
e_1=(1,0),\qquad e_2=(0,1)\in A,
\]
then
\[
e_1^2=e_1,\qquad e_2^2=e_2,\qquad e_1+e_2=1,\qquad e_1e_2=0,
\]
and
\[
M_1=Ae_1,\qquad M_2=Ae_2.
\]
Thus \(M_1\) and \(M_2\) are ideals generated by idempotents, hence direct summands of \(A\), so again projective.

Now assume \(A_1\neq 0\) and \(A_2\neq 0\). We show that \(M_1\) and \(M_2\) are not free.

First consider
\[
M_1=A_1\times 0.
\]
Its annihilator is
\[
\operatorname{Ann}_A(M_1)=0\times A_2.
\]
Indeed, an element \((b_1,b_2)\in A\) annihilates \(M_1\) if and only if
\[
(b_1,b_2)(a_1,0)=(b_1a_1,0)=(0,0)
\]
for all \(a_1\in A_1\), which forces \(b_1=0\). Hence
\[
\operatorname{Ann}_A(M_1)=0\times A_2.
\]
Since \(A_2\neq 0\), this annihilator is nonzero.

On the other hand, any nonzero free \(A\)-module \(A^n\) has annihilator \(0\): if
\[
xA^n=0,
\]
then in particular \(x\cdot (1,0,\dots,0)=0\), hence \(x=0\).

Therefore \(M_1\) cannot be free.

Similarly,
\[
\operatorname{Ann}_A(M_2)=A_1\times 0,
\]
which is nonzero because \(A_1\neq 0\). So \(M_2\) is not free either.

Hence \(M_1\) and \(M_2\) are projective \(A\)-modules which are not free. This proves that projective modules need not be free.	
	
	
\section{Problem 12 --- A free resolution over \texorpdfstring{$A=k[x,y]/(xy)$}{A = k[x,y]/(xy)} and computation of \texorpdfstring{$\operatorname{Ext}$}{Ext} and \texorpdfstring{$\operatorname{Tor}$}{Tor}}

\textbf{Exact statement.}
Let
\[
A=k[x,y]/(xy),\qquad \mathfrak m=(x,y).
\]
Construct a resolution of \(A/\mathfrak m\) by finitely generated free modules. Use this resolution to calculate
\[
\operatorname{Ext}_A^i(A/\mathfrak m,A)
\quad\text{and}\quad
\operatorname{Tor}_i^A(A/\mathfrak m,A/\mathfrak m).
\]

\textbf{Solution.}
Set
\[
k:=A/\mathfrak m.
\]
We construct a free resolution of \(k\).

There is a natural surjection
\[
A\twoheadrightarrow k,
\]
whose kernel is \(\mathfrak m=(x,y)\). Hence we begin with
\[
A^2 \xrightarrow{d_1} A \to k \to 0,
\qquad
d_1(a,b)=ax+by.
\]
Thus \(d_1\) is represented by
\[
d_1=\begin{bmatrix}x&y\end{bmatrix}.
\]

We next compute \(\ker(d_1)\). A useful fact in the ring \(A=k[x,y]/(xy)\) is
\[
\operatorname{Ann}_A(x)=(y),
\qquad
\operatorname{Ann}_A(y)=(x).
\]
Indeed, modulo \((y)\) the ring becomes \(k[x]\), in which the image of \(x\) is not a zero divisor, so \(ax=0\) forces \(a\in (y)\); similarly for \(y\).

Now suppose
\[
ax+by=0.
\]
Reducing modulo \((y)\), we get
\[
\bar a\,x=0 \quad\text{in } A/(y)\cong k[x].
\]
Hence \(\bar a=0\), so \(a\in (y)\). Similarly, reducing modulo \((x)\), we obtain \(b\in (x)\). Therefore
\[
\ker(d_1)=(y)\times (x).
\]

Thus \(\ker(d_1)\) is the image of the map
\[
d_2:A^2\to A^2,
\qquad
d_2(u,v)=(yu,xv),
\]
represented by
\[
d_2=
\begin{bmatrix}
	y&0\\
	0&x
\end{bmatrix}.
\]

Now compute \(\ker(d_2)\). If
\[
d_2(u,v)=(0,0),
\]
then
\[
yu=0,\qquad xv=0,
\]
so
\[
u\in \operatorname{Ann}_A(y)=(x),
\qquad
v\in \operatorname{Ann}_A(x)=(y).
\]
Hence
\[
\ker(d_2)=(x)\times (y),
\]
which is the image of
\[
d_3:A^2\to A^2,
\qquad
d_3(u,v)=(xu,yv),
\]
represented by
\[
d_3=
\begin{bmatrix}
	x&0\\
	0&y
\end{bmatrix}.
\]

Repeating this argument gives a \(2\)-periodic free resolution
\[
\cdots \xrightarrow{d_4} A^2 \xrightarrow{d_3} A^2 \xrightarrow{d_2} A^2 \xrightarrow{d_1} A \to k \to 0,
\]
where
\[
d_{2n}=
\begin{bmatrix}
	y&0\\
	0&x
\end{bmatrix},
\qquad
d_{2n+1}=
\begin{bmatrix}
	x&0\\
	0&y
\end{bmatrix}
\quad (n\ge 1),
\]
and
\[
d_1=\begin{bmatrix}x&y\end{bmatrix}.
\]

So a free resolution of \(A/\mathfrak m\) is
\[
\cdots \to A^2
\xrightarrow{\left[\begin{smallmatrix}y&0\\0&x\end{smallmatrix}\right]}
A^2
\xrightarrow{\left[\begin{smallmatrix}x&0\\0&y\end{smallmatrix}\right]}
A^2
\xrightarrow{\left[\begin{smallmatrix}y&0\\0&x\end{smallmatrix}\right]}
A^2
\xrightarrow{\left[\begin{smallmatrix}x&y\end{smallmatrix}\right]}
A
\to A/\mathfrak m \to 0.
\]

\medskip

\textbf{Computation of \(\operatorname{Ext}_A^i(k,A)\).}

Apply \(\operatorname{Hom}_A(-,A)\). Using
\[
\operatorname{Hom}_A(A,A)\cong A,\qquad \operatorname{Hom}_A(A^2,A)\cong A^2,
\]
we obtain the cochain complex
\[
0\to A \xrightarrow{\delta^0} A^2 \xrightarrow{\delta^1} A^2 \xrightarrow{\delta^2} A^2 \xrightarrow{\delta^3}\cdots
\]
where
\[
\delta^0(a)=(xa,ya),
\]
and then the differentials alternate as
\[
\delta^1=
\begin{bmatrix}
	y&0\\
	0&x
\end{bmatrix},
\qquad
\delta^2=
\begin{bmatrix}
	x&0\\
	0&y
\end{bmatrix},
\qquad
\delta^3=
\begin{bmatrix}
	y&0\\
	0&x
\end{bmatrix},
\ \dots
\]

First,
\[
\operatorname{Ext}_A^0(k,A)=\ker(\delta^0).
\]
So we need all \(a\in A\) such that
\[
xa=0,\qquad ya=0.
\]
Thus
\[
a\in (y)\cap (x).
\]
But in \(A=k[x,y]/(xy)\),
\[
(x)\cap (y)=0,
\]
since every common multiple is divisible by \(xy=0\). Hence
\[
\operatorname{Ext}_A^0(k,A)=0.
\]

Next,
\[
\operatorname{Ext}_A^1(k,A)=\ker(\delta^1)/\operatorname{im}(\delta^0).
\]
Now
\[
\ker(\delta^1)=\{(a,b)\in A^2\mid ya=0,\ xb=0\}=(x)\times (y),
\]
and
\[
\operatorname{im}(\delta^0)=\{(xa,ya)\mid a\in A\}.
\]
Therefore
\[
\operatorname{Ext}_A^1(k,A)
=
\frac{(x)\times (y)}{\{(xa,ya)\mid a\in A\}}.
\]
This quotient is one-dimensional over \(k=A/\mathfrak m\), hence
\[
\operatorname{Ext}_A^1(k,A)\cong k.
\]

For \(i\ge 2\), the kernels and images agree:
\[
\ker(\delta^2)=(y)\times (x)=\operatorname{im}(\delta^1),
\]
\[
\ker(\delta^3)=(x)\times (y)=\operatorname{im}(\delta^2),
\]
and similarly thereafter. Thus
\[
\operatorname{Ext}_A^i(k,A)=0
\qquad\text{for all }i\ge 2.
\]

So
\[
\operatorname{Ext}_A^i(A/\mathfrak m,A)\cong
\begin{cases}
	0,& i=0,\\[1mm]
	A/\mathfrak m,& i=1,\\[1mm]
	0,& i\ge 2.
\end{cases}
\]

\medskip

\textbf{Computation of \(\operatorname{Tor}_i^A(k,k)\).}

Now tensor the free resolution with \(k=A/\mathfrak m\). Since \(x,y\in \mathfrak m\), they act as zero on \(k\). Therefore every differential becomes zero after tensoring with \(k\). So the resulting complex is
\[
\cdots \to k^2 \xrightarrow{0} k^2 \xrightarrow{0} k^2 \xrightarrow{0} k \to 0.
\]

Hence its homology is just each term itself:
\[
\operatorname{Tor}_0^A(k,k)\cong k,
\]
and for every \(i\ge 1\),
\[
\operatorname{Tor}_i^A(k,k)\cong k^2.
\]

Therefore
\[
\operatorname{Tor}_i^A(A/\mathfrak m,A/\mathfrak m)\cong
\begin{cases}
	A/\mathfrak m,& i=0,\\[1mm]
	(A/\mathfrak m)^2,& i\ge 1.
\end{cases}
\]

\section{Problem 13 --- The localized Veronese subring is Cohen--Macaulay}

\textbf{Exact statement.}
Let
\[
S=k[x^3,x^2y,xy^2,y^3]\subseteq k[x,y],
\]
and let
\[
\mathfrak m=(x^3,x^2y,xy^2,y^3)\subseteq S.
\]
Show that \(S_{\mathfrak m}\) is Cohen--Macaulay by exhibiting an explicit regular sequence.

\medskip

\textit{Remark.} The intended dimension here is
\[
\dim S_{\mathfrak m}=2,
\]
which is the correct value for this ring.

\textbf{Solution.}
We will show that
\[
x^3,\ y^3
\]
is an \(S_{\mathfrak m}\)-regular sequence. Since \(\dim S_{\mathfrak m}=2\), this will imply that \(S_{\mathfrak m}\) is Cohen--Macaulay.

\medskip

Set
\[
R:=k[x^3,y^3]\cong k[u,v].
\]
We first claim that
\[
S=R\oplus R\cdot x^2y \oplus R\cdot xy^2
\]
as an \(R\)-module.

Indeed, every monomial in \(S\) has the form \(x^i y^j\) with \(i,j\ge 0\) and \(i+j\equiv 0\pmod 3\). According to the residue class of \(i\) modulo \(3\), one gets exactly one of the following possibilities:
\[
x^i y^j=(x^3)^a(y^3)^b,
\qquad
x^i y^j=(x^3)^a(y^3)^b\cdot x^2y,
\qquad
x^i y^j=(x^3)^a(y^3)^b\cdot xy^2.
\]
Thus every element of \(S\) lies in
\[
R\oplus R x^2y \oplus R xy^2.
\]
These three summands are linearly independent over \(R\), since they correspond to distinct residue classes modulo \(3\). Hence
\[
S=R\oplus R x^2y \oplus R xy^2.
\]

Now let
\[
\mathfrak n=(x^3,y^3)\subseteq R.
\]
After localizing, this gives
\[
S_{\mathfrak m}
=
R_{\mathfrak n}\oplus R_{\mathfrak n}x^2y \oplus R_{\mathfrak n}xy^2.
\]
So \(S_{\mathfrak m}\) is a free \(R_{\mathfrak n}\)-module of rank \(3\).

\medskip

We now show that \(x^3,y^3\) is a regular sequence on \(S_{\mathfrak m}\).

First, \(x^3\) is a non-zero-divisor on \(R_{\mathfrak n}\), because
\[
R_{\mathfrak n}\cong k[u,v]_{(u,v)}
\]
is a domain. Therefore multiplication by \(x^3\) is injective on each summand of
\[
S_{\mathfrak m}
=
R_{\mathfrak n}\oplus R_{\mathfrak n}x^2y \oplus R_{\mathfrak n}xy^2,
\]
hence on all of \(S_{\mathfrak m}\). So \(x^3\) is a non-zero-divisor on \(S_{\mathfrak m}\).

Next, consider the quotient modulo \(x^3\). Using the decomposition above,
\[
S_{\mathfrak m}/x^3S_{\mathfrak m}
\cong
(R_{\mathfrak n}/(x^3)) \oplus (R_{\mathfrak n}/(x^3))x^2y \oplus (R_{\mathfrak n}/(x^3))xy^2.
\]
But
\[
R_{\mathfrak n}/(x^3)
\cong
k[y^3]_{(y^3)},
\]
which is a domain. Hence multiplication by \(y^3\) is injective on \(R_{\mathfrak n}/(x^3)\), and therefore injective on each of the three summands above. It follows that multiplication by \(y^3\) is injective on
\[
S_{\mathfrak m}/x^3S_{\mathfrak m}.
\]
So \(y^3\) is a non-zero-divisor modulo \(x^3\).

Thus
\[
x^3,\ y^3
\]
is a regular sequence in \(S_{\mathfrak m}\).

Finally, since
\[
\dim S_{\mathfrak m}=2,
\]
we obtain
\[
\operatorname{depth}(S_{\mathfrak m})\ge 2.
\]
But depth is always at most dimension, so
\[
\operatorname{depth}(S_{\mathfrak m})=2=\dim S_{\mathfrak m}.
\]
Therefore
\[
\boxed{S_{\mathfrak m}\text{ is Cohen--Macaulay}.}
\]

\section{Problem 14(a) --- Characterization of minimal homomorphisms over a local ring}

\textbf{Exact statement.}
Let \((A,\mathfrak m)\) be a local ring, let \(k=A/\mathfrak m\), and let
\[
f:M\to N
\]
be an \(A\)-module homomorphism between finitely generated modules. We say that \(f\) is \emph{minimal} if the induced map
\[
f\otimes \mathrm{id}_k:\ M\otimes_A k \to N\otimes_A k
\]
is an isomorphism.

Show that \(f\) is minimal if and only if \(f\) is surjective and
\[
\ker f\subseteq \mathfrak m M.
\]

\textbf{Solution.}
For any \(A\)-module \(L\), there is a natural isomorphism
\[
L\otimes_A k \cong L/\mathfrak m L.
\]
Hence the map
\[
f\otimes k:\ M\otimes_A k \to N\otimes_A k
\]
identifies with the induced map
\[
\bar f:\ M/\mathfrak m M \to N/\mathfrak m N.
\]
Thus \(f\) is minimal if and only if \(\bar f\) is an isomorphism.

We now prove the equivalence.

\medskip

\textbf{(\(\Rightarrow\)) Assume \(f\) is minimal. Then \(f\) is surjective and \(\ker f\subseteq \mathfrak m M\).}

Since \(\bar f\) is surjective, we have
\[
N=f(M)+\mathfrak m N.
\]
Because \(N\) is finitely generated and \(A\) is local, Nakayama's lemma implies
\[
N=f(M).
\]
So \(f\) is surjective.

Now consider the exact sequence
\[
0\to \ker f \to M \xrightarrow{f} N \to 0.
\]
Tensoring with \(k\) gives a right exact sequence
\[
(\ker f)\otimes_A k \to M\otimes_A k \xrightarrow{f\otimes k} N\otimes_A k \to 0.
\]
Since \(f\otimes k\) is an isomorphism, its kernel is zero, so the image of
\[
(\ker f)\otimes_A k \to M\otimes_A k
\]
is zero.

Under the identifications
\[
(\ker f)\otimes_A k \cong \ker f/\mathfrak m(\ker f),
\qquad
M\otimes_A k \cong M/\mathfrak m M,
\]
this map has image
\[
(\ker f+\mathfrak m M)/\mathfrak m M.
\]
Since the image is zero, it follows that
\[
\ker f\subseteq \mathfrak m M.
\]

Thus minimality implies that \(f\) is surjective and \(\ker f\subseteq \mathfrak m M\).

\medskip

\textbf{(\(\Leftarrow\)) Assume \(f\) is surjective and \(\ker f\subseteq \mathfrak m M\). Then \(f\) is minimal.}

Since \(f\) is surjective, the induced map
\[
\bar f:\ M/\mathfrak m M \to N/\mathfrak m N
\]
is also surjective.

It remains to prove injectivity. Let \(m\in M\) satisfy
\[
\bar f(m+\mathfrak m M)=0.
\]
Then
\[
f(m)\in \mathfrak m N.
\]
Because \(f\) is surjective, we have
\[
\mathfrak m N=f(\mathfrak m M).
\]
So there exists \(m'\in \mathfrak m M\) such that
\[
f(m)=f(m').
\]
Hence
\[
m-m'\in \ker f.
\]
By assumption,
\[
m-m'\in \mathfrak m M.
\]
Since also \(m'\in \mathfrak m M\), it follows that
\[
m\in \mathfrak m M.
\]
Therefore
\[
m+\mathfrak m M=0.
\]
So \(\bar f\) is injective.

Hence \(\bar f\) is an isomorphism, and therefore
\[
f\otimes_A k
\]
is an isomorphism. Thus \(f\) is minimal.

\medskip

We conclude that
\[
\boxed{
f \text{ is minimal }
\iff
f \text{ is surjective and } \ker f\subseteq \mathfrak m M.
}
\]

\section{Problem 14(b) --- Existence of a minimal map from a free module}

\textbf{Exact statement.}
Let \((A,\mathfrak m)\) be a local ring, let \(k=A/\mathfrak m\), and let \(M\) be a finitely generated \(A\)-module. Show that there exists a minimal homomorphism
\[
f:F\to M
\]
with \(F\) free.

\textbf{Solution.}
We want to construct a free \(A\)-module \(F\) and an \(A\)-linear map
\[
f:F\to M
\]
such that
\[
f\otimes_A k:\ F\otimes_A k \to M\otimes_A k
\]
is an isomorphism.

Since
\[
M\otimes_A k \cong M/\mathfrak m M,
\qquad
F\otimes_A k \cong F/\mathfrak m F,
\]
it is enough to construct \(f\) so that the induced map
\[
\bar f:\ F/\mathfrak m F \to M/\mathfrak m M
\]
is an isomorphism.

Because \(M\) is finitely generated over \(A\), the quotient
\[
M/\mathfrak m M
\]
is a finite-dimensional vector space over
\[
k=A/\mathfrak m.
\]
Choose a \(k\)-basis
\[
\overline{m}_1,\dots,\overline{m}_r
\]
of \(M/\mathfrak m M\), and choose lifts
\[
m_1,\dots,m_r\in M.
\]

Now let
\[
F=A^r
\]
with basis \(e_1,\dots,e_r\), and define an \(A\)-linear map
\[
f:F\to M
\]
by
\[
f(e_i)=m_i
\qquad (1\le i\le r).
\]

Reducing modulo \(\mathfrak m\), we obtain a \(k\)-linear map
\[
\bar f:\ F/\mathfrak m F \to M/\mathfrak m M.
\]
Since
\[
F/\mathfrak m F \cong k^r
\]
with basis \(\bar e_1,\dots,\bar e_r\), the map \(\bar f\) sends \(\bar e_i\) to \(\overline{m}_i\). By construction, \(\overline{m}_1,\dots,\overline{m}_r\) form a basis of \(M/\mathfrak m M\). Therefore \(\bar f\) is an isomorphism.

Hence
\[
f\otimes_A k
\]
is an isomorphism, so \(f\) is minimal by definition.

\medskip

If one wishes, one can also reformulate this using part (a). Since \(\bar f\) is surjective,
\[
M=f(F)+\mathfrak m M.
\]
By Nakayama's lemma,
\[
M=f(F),
\]
so \(f\) is surjective. Then part (a) implies
\[
\ker f\subseteq \mathfrak m F.
\]
Thus \(f\) is minimal in the equivalent sense of part (a) as well.

\medskip

Therefore for every finitely generated \(A\)-module \(M\), there exists a minimal homomorphism
\[
f:F\to M
\]
with \(F\) free.

\section{Problem 14(c) --- Vanishing of the induced maps on \texorpdfstring{$\operatorname{Ext}$}{Ext} for a minimal presentation}

\textbf{Exact statement.}
Let \((A,\mathfrak m)\) be a local ring, \(k=A/\mathfrak m\), and suppose
\[
0\longrightarrow K \xrightarrow{\,f\,} F \xrightarrow{\,g\,} M \longrightarrow 0
\]
is exact, with \(g\) minimal and \(K,F\) free. Show that the homomorphisms
\[
f_*:\operatorname{Ext}_A^i(k,K)\longrightarrow \operatorname{Ext}_A^i(k,F)
\]
induced by \(f\) vanish for all \(i\).

\medskip

\textit{Note.} If \(K=A^n\), \(F=A^m\), and \(f\) is represented by a matrix \((c_{ij})\), then \(f_*\) is represented by the same matrix, now viewed as a map
\[
(\operatorname{Ext}_A^i(k,A))^n\to (\operatorname{Ext}_A^i(k,A))^m.
\]

\textbf{Solution.}
Since the sequence
\[
0\to K \xrightarrow{f} F \xrightarrow{g} M \to 0
\]
is exact and \(g\) is minimal, part (a) implies that \(g\) is surjective and
\[
\ker g\subseteq \mathfrak m F.
\]
But \(\operatorname{im}(f)=\ker g\), so
\[
\operatorname{im}(f)\subseteq \mathfrak m F.
\]

Choose bases so that
\[
K\cong A^n,\qquad F\cong A^m.
\]
Then \(f\) is represented by an \(m\times n\) matrix
\[
(c_{ij}).
\]
Since \(\operatorname{im}(f)\subseteq \mathfrak m F\), each column of this matrix lies in \(\mathfrak m A^m\), hence every entry satisfies
\[
c_{ij}\in \mathfrak m.
\]

We now claim that for every \(i\ge 0\),
\[
\mathfrak m\cdot \operatorname{Ext}_A^i(k,A)=0.
\]
Indeed, because \(A\) is commutative, the \(A\)-module structure on \(\operatorname{Ext}_A^i(k,A)\) may be viewed either from the second argument \(A\) or from the first argument \(k\), and these two actions agree. But if \(a\in \mathfrak m\), then multiplication by \(a\) on
\[
k=A/\mathfrak m
\]
is the zero map. Hence multiplication by \(a\) on \(\operatorname{Ext}_A^i(k,A)\) is also zero. Therefore
\[
\mathfrak m\cdot \operatorname{Ext}_A^i(k,A)=0.
\]

Now by the note, the induced homomorphism
\[
f_*:\operatorname{Ext}_A^i(k,K)\to \operatorname{Ext}_A^i(k,F)
\]
is represented by the same matrix \((c_{ij})\), now acting as
\[
(\operatorname{Ext}_A^i(k,A))^n\to (\operatorname{Ext}_A^i(k,A))^m.
\]
Since every \(c_{ij}\in \mathfrak m\), and \(\mathfrak m\) annihilates \(\operatorname{Ext}_A^i(k,A)\), each entry acts as zero. Thus the whole matrix defines the zero map.

Therefore
\[
f_*=0
\]
for every \(i\ge 0\).

\medskip

Hence
\[
\boxed{
	f_*:\operatorname{Ext}_A^i(k,K)\longrightarrow \operatorname{Ext}_A^i(k,F)
	\text{ is the zero map for all }i.
}
\]	

\section{Problem 15 --- Existence of minimal free resolutions over a Noetherian local ring}

\textbf{Exact statement.}
Let \((A,\mathfrak m)\) be a local ring, \(k=A/\mathfrak m\), and let
\[
F^\bullet \to M \to 0
\]
be a free resolution of a finitely generated \(A\)-module \(M\) by finitely generated free modules, with maps
\[
f_i:F_{i+1}\to F_i.
\]
We say this free resolution is \emph{minimal} if all the maps
\[
f_i\otimes \mathrm{id}_k:\ F_{i+1}\otimes_A k \to F_i\otimes_A k
\]
are zero.

Show that if \(A\) is Noetherian, then minimal resolutions exist.

\textbf{Solution.}
We construct a minimal free resolution inductively, using Problem 14(b).

\medskip

Since \(M\) is finitely generated, Problem 14(b) gives a minimal homomorphism
\[
\varepsilon_0:F_0\to M
\]
with \(F_0\) finitely generated free. By Problem 14(a), \(\varepsilon_0\) is surjective and
\[
\ker(\varepsilon_0)\subseteq \mathfrak m F_0.
\]
Set
\[
K_1:=\ker(\varepsilon_0).
\]
Because \(A\) is Noetherian and \(F_0\) is finitely generated, \(F_0\) is a Noetherian \(A\)-module, so \(K_1\) is finitely generated.

Now apply Problem 14(b) to \(K_1\). There exists a minimal homomorphism
\[
\varepsilon_1:F_1\to K_1
\]
with \(F_1\) finitely generated free. Composing with the inclusion \(K_1\hookrightarrow F_0\), we get
\[
f_1:F_1\to F_0.
\]
Since \(\varepsilon_1\) is surjective,
\[
\operatorname{im}(f_1)=K_1=\ker(\varepsilon_0),
\]
so
\[
F_1\xrightarrow{f_1}F_0\xrightarrow{\varepsilon_0}M\to 0
\]
is exact.

Moreover,
\[
\operatorname{im}(f_1)=K_1\subseteq \mathfrak m F_0,
\]
hence
\[
f_1\otimes_A k=0.
\]

Now set
\[
K_2:=\ker(f_1).
\]
Again \(K_2\) is finitely generated, since it is a submodule of the finitely generated free module \(F_1\). By Problem 14(b), there exists a minimal homomorphism
\[
\varepsilon_2:F_2\to K_2
\]
with \(F_2\) finitely generated free. Composing with the inclusion \(K_2\hookrightarrow F_1\), we obtain
\[
f_2:F_2\to F_1.
\]
Then
\[
\operatorname{im}(f_2)=K_2=\ker(f_1),
\]
so the complex remains exact.

To see that
\[
f_2\otimes_A k=0,
\]
observe that \(\varepsilon_1:F_1\to K_1\) is minimal, so by Problem 14(a),
\[
\ker(\varepsilon_1)\subseteq \mathfrak m F_1.
\]
But \(\ker(\varepsilon_1)=\ker(f_1)=K_2\). Therefore
\[
\operatorname{im}(f_2)=K_2\subseteq \mathfrak m F_1,
\]
hence
\[
f_2\otimes_A k=0.
\]

The same construction now repeats inductively.

Suppose we have already constructed an exact complex
\[
F_n\xrightarrow{f_n}F_{n-1}\xrightarrow{f_{n-1}}\cdots\xrightarrow{f_1}F_0\xrightarrow{\varepsilon_0}M\to 0
\]
with each \(F_i\) finitely generated free and each
\[
f_i\otimes_A k=0.
\]
Set
\[
K_{n+1}:=\ker(f_n).
\]
Since \(A\) is Noetherian and \(F_n\) is finitely generated, \(K_{n+1}\) is finitely generated. By Problem 14(b), there exists a minimal homomorphism
\[
\varepsilon_{n+1}:F_{n+1}\to K_{n+1}
\]
with \(F_{n+1}\) finitely generated free. Composing with the inclusion \(K_{n+1}\hookrightarrow F_n\), define
\[
f_{n+1}:F_{n+1}\to F_n.
\]
Then
\[
\operatorname{im}(f_{n+1})=K_{n+1}=\ker(f_n),
\]
so exactness continues.

Also, since \(\varepsilon_n:F_n\to K_n\) is minimal, Problem 14(a) gives
\[
\ker(\varepsilon_n)\subseteq \mathfrak m F_n.
\]
But \(\ker(\varepsilon_n)=\ker(f_n)=K_{n+1}\), so
\[
\operatorname{im}(f_{n+1})=K_{n+1}\subseteq \mathfrak m F_n.
\]
Hence
\[
f_{n+1}\otimes_A k=0.
\]

Thus by induction we obtain an exact free resolution
\[
\cdots \to F_2\to F_1\to F_0\to M\to 0
\]
such that every differential becomes zero after tensoring with \(k=A/\mathfrak m\). This is exactly a minimal free resolution.

\medskip

Therefore, if \(A\) is Noetherian, minimal free resolutions exist.

\section{Problem 16 --- Projective dimension, minimal free resolutions, and vanishing of \texorpdfstring{$\operatorname{Tor}$}{Tor}}

\textbf{Exact statement.}
Let \((A,\mathfrak m)\) be a Noetherian local ring, let
\[
k=A/\mathfrak m,
\]
and let \(M\) be a finitely generated \(A\)-module. We define the projective dimension \(\operatorname{pd}_A M\) of \(M\) to be the smallest \(n\) for which there exists a free resolution
\[
0\to F_n\to \cdots \to F_0\to M\to 0.
\]
Note that this need not be finite. Show that if \(\operatorname{pd}_A M\) is finite, then it is the length of every minimal free resolution of \(M\). Further, \(\operatorname{pd}_A M\) is the smallest integer \(i\) for which
\[
\operatorname{Tor}_{i+1}^A(k,M)=0.
\]

\textbf{Solution.}
Assume
\[
n:=\operatorname{pd}_A M<\infty.
\]

Since \(A\) is Noetherian and local, Problem 15 shows that \(M\) admits a minimal free resolution
\[
\cdots \longrightarrow F_2 \xrightarrow{d_2} F_1 \xrightarrow{d_1} F_0 \longrightarrow M \longrightarrow 0
\]
by finitely generated free modules. By definition of minimality,
\[
d_i\otimes_A k=0
\qquad\text{for all }i.
\]
Hence after tensoring with \(k=A/\mathfrak m\), the complex
\[
\cdots \longrightarrow F_2\otimes_A k \longrightarrow F_1\otimes_A k \longrightarrow F_0\otimes_A k \longrightarrow 0
\]
has zero differentials. Therefore its homology in degree \(i\) is simply
\[
\operatorname{Tor}_i^A(k,M)\cong F_i\otimes_A k
\qquad\text{for all }i\ge 0.
\]
Since each \(F_i\) is finitely generated free, we have
\[
F_i\otimes_A k=0 \iff F_i=0.
\]

Now, because \(n=\operatorname{pd}_A M\), there exists some free resolution of length \(n\),
\[
0\to G_n\to G_{n-1}\to \cdots \to G_0\to M\to 0.
\]
Tensoring this resolution with \(k\) shows that
\[
\operatorname{Tor}_i^A(k,M)=0
\qquad\text{for all }i>n.
\]
Using the minimal free resolution, this implies
\[
F_i\otimes_A k=0
\qquad\text{for all }i>n,
\]
hence
\[
F_i=0
\qquad\text{for all }i>n.
\]
So every minimal free resolution has length at most \(n\).

To see that the length is exactly \(n\), note that if \(F_n=0\), then the minimal free resolution would already terminate at \(F_{n-1}\), producing a free resolution of \(M\) of length at most \(n-1\). This contradicts the definition of \(n=\operatorname{pd}_A M\). Therefore
\[
F_n\neq 0.
\]
Hence every minimal free resolution has length exactly \(n\).

Finally, since
\[
\operatorname{Tor}_i^A(k,M)\cong F_i\otimes_A k,
\]
we have
\[
\operatorname{Tor}_{i+1}^A(k,M)=0
\iff
F_{i+1}=0.
\]
If the minimal free resolution has length \(n\), then
\[
F_n\neq 0,\qquad F_{n+1}=F_{n+2}=\cdots =0.
\]
Therefore
\[
\operatorname{Tor}_n^A(k,M)\neq 0,
\qquad
\operatorname{Tor}_{n+1}^A(k,M)=0.
\]
So the smallest integer \(i\) such that
\[
\operatorname{Tor}_{i+1}^A(k,M)=0
\]
is precisely
\[
i=n=\operatorname{pd}_A M.
\]

Thus, if \(\operatorname{pd}_A M<\infty\), then it is the length of every minimal free resolution of \(M\), and moreover
\[
\boxed{
	\operatorname{pd}_A M
	=
	\min\{\,i\ge 0 \mid \operatorname{Tor}_{i+1}^A(k,M)=0\,\}.
}
\]
	
\section{Problem 17 --- Auslander--Buchsbaum formula}

\textbf{Exact statement.}
Let \(A\) be a Noetherian local ring and \(M\) a finitely generated \(A\)-module with
\[
\operatorname{pd}_A M<\infty.
\]
Show that
\[
\operatorname{pd}_A M+\operatorname{depth}(M)=\operatorname{depth}(A).
\]

\textbf{Solution.}
This is the Auslander--Buchsbaum formula.

We use the standard characterization of depth:
\[
\operatorname{depth}(N)
=
\min\{\,i\ge 0 \mid \operatorname{Ext}_A^i(k,N)\neq 0\,\},
\qquad k=A/\mathfrak m,
\]
for every nonzero finitely generated \(A\)-module \(N\).

Set
\[
s:=\operatorname{depth}(A),
\qquad
n:=\operatorname{pd}_A M.
\]
We prove that
\[
n+\operatorname{depth}(M)=s
\]
by induction on \(n\).

\medskip

\textbf{Case \(n=0\).}
If \(\operatorname{pd}_A M=0\), then \(M\) is projective. Since \(A\) is local and \(M\) is finitely generated, \(M\) is free. Thus
\[
M\cong A^r
\]
for some \(r>0\). Therefore
\[
\operatorname{Ext}_A^i(k,M)\cong \bigl(\operatorname{Ext}_A^i(k,A)\bigr)^r
\]
for all \(i\), and hence
\[
\operatorname{depth}(M)=\operatorname{depth}(A)=s.
\]
So
\[
\operatorname{pd}_A M+\operatorname{depth}(M)=0+s=s=\operatorname{depth}(A).
\]

\medskip

\textbf{Case \(n=1\).}
Assume \(\operatorname{pd}_A M=1\). Take a minimal free presentation
\[
0\longrightarrow K \xrightarrow{f} F \xrightarrow{g} M \longrightarrow 0
\]
with \(K\) and \(F\) free. Since \(K\) and \(F\) are free,
\[
\operatorname{depth}(K)=\operatorname{depth}(F)=s.
\]

By Problem 14(c), the induced maps
\[
f_*:\operatorname{Ext}_A^i(k,K)\to \operatorname{Ext}_A^i(k,F)
\]
are zero for all \(i\). Hence the long exact \(\operatorname{Ext}\)-sequence breaks into short exact sequences
\[
0\to \operatorname{Ext}_A^i(k,F)
\to \operatorname{Ext}_A^i(k,M)
\to \operatorname{Ext}_A^{i+1}(k,K)
\to 0
\]
for all \(i\ge 0\).

Now if \(i<s-1\), then
\[
\operatorname{Ext}_A^i(k,F)=0
\quad\text{and}\quad
\operatorname{Ext}_A^{i+1}(k,K)=0,
\]
so
\[
\operatorname{Ext}_A^i(k,M)=0.
\]
For \(i=s-1\), we have
\[
\operatorname{Ext}_A^{s-1}(k,F)=0,
\qquad
\operatorname{Ext}_A^s(k,K)\neq 0,
\]
hence
\[
\operatorname{Ext}_A^{s-1}(k,M)\neq 0.
\]
Therefore
\[
\operatorname{depth}(M)=s-1.
\]
So
\[
\operatorname{pd}_A M+\operatorname{depth}(M)=1+(s-1)=s=\operatorname{depth}(A).
\]

\medskip

\textbf{Induction step.}
Assume the formula holds for all finitely generated modules of projective dimension \(n-1\), and let \(\operatorname{pd}_A M=n>1\).

Take the beginning of a minimal free resolution:
\[
0\longrightarrow K \longrightarrow F \longrightarrow M \longrightarrow 0,
\]
where \(F\) is free and
\[
\operatorname{pd}_A K=n-1.
\]

By the induction hypothesis,
\[
\operatorname{depth}(K)=s-(n-1).
\]
Set
\[
t:=\operatorname{depth}(K)=s-n+1.
\]
Since \(n>1\), we have
\[
t<s.
\]
Also \(F\) is free, so
\[
\operatorname{depth}(F)=s.
\]

Apply \(\operatorname{Hom}_A(k,-)\) to the exact sequence
\[
0\to K\to F\to M\to 0.
\]
For each \(i\), the resulting long exact sequence contains
\[
\operatorname{Ext}_A^i(k,F)\to \operatorname{Ext}_A^i(k,M)\to \operatorname{Ext}_A^{i+1}(k,K)\to \operatorname{Ext}_A^{i+1}(k,F).
\]

Since \(\operatorname{depth}(F)=s>t\), we know that
\[
\operatorname{Ext}_A^j(k,F)=0
\qquad\text{for all }j\le t.
\]
Therefore, for \(i<t-1\), both outer terms vanish:
\[
\operatorname{Ext}_A^i(k,F)=0,
\qquad
\operatorname{Ext}_A^{i+1}(k,F)=0.
\]
Because also \(i+1<t\), we have
\[
\operatorname{Ext}_A^{i+1}(k,K)=0.
\]
Hence
\[
\operatorname{Ext}_A^i(k,M)=0
\qquad\text{for all }i<t-1.
\]

Now take \(i=t-1\). Since \(t<s\), we still have
\[
\operatorname{Ext}_A^{t-1}(k,F)=0,
\qquad
\operatorname{Ext}_A^{t}(k,F)=0.
\]
So the long exact sequence gives an isomorphism
\[
\operatorname{Ext}_A^{t-1}(k,M)\cong \operatorname{Ext}_A^t(k,K).
\]
But
\[
\operatorname{Ext}_A^t(k,K)\neq 0
\]
by the definition of \(t=\operatorname{depth}(K)\). Therefore
\[
\operatorname{Ext}_A^{t-1}(k,M)\neq 0.
\]
It follows that
\[
\operatorname{depth}(M)=t-1=s-n.
\]

Thus
\[
\operatorname{pd}_A M+\operatorname{depth}(M)
=
n+(s-n)=s=\operatorname{depth}(A).
\]

This completes the induction.

\medskip

Hence for every finitely generated \(A\)-module \(M\) of finite projective dimension,
\[
\boxed{
	\operatorname{pd}_A M+\operatorname{depth}(M)=\operatorname{depth}(A).
}
\]

\chapter{Theory and Definitions for Problems 11--17}

\section{0. Big picture: what Problems 11--17 are about}

Problems 11--17 form a coherent block of \emph{homological commutative algebra}.

The progression is:

\begin{itemize}
	\item \textbf{Problem 11:} projective modules need not be free.
	\item \textbf{Problem 12:} explicit free resolutions, and computations of \(\operatorname{Ext}\) and \(\operatorname{Tor}\).
	\item \textbf{Problem 13:} regular sequences and Cohen--Macaulay rings.
	\item \textbf{Problem 14:} minimal maps over a local ring.
	\item \textbf{Problem 15:} minimal free resolutions exist.
	\item \textbf{Problem 16:} projective dimension can be read from minimal resolutions and from vanishing of \(\operatorname{Tor}\).
	\item \textbf{Problem 17:} projective dimension and depth are tied together by the Auslander--Buchsbaum formula.
\end{itemize}

So the thread is
\[
\text{projective/free modules}
\;\to\;
\text{free resolutions}
\;\to\;
\operatorname{Ext},\operatorname{Tor}
\;\to\;
\text{regular sequences and depth}
\;\to\;
\text{minimal resolutions}
\;\to\;
\text{projective dimension}
\;\to\;
\text{Auslander--Buchsbaum}.
\]

This is one of the main roads into modern local algebra.

\bigskip

\section{1. Problem 11 theory: projective modules, free modules, and direct summands}

\subsection*{1.1. Free modules}

Let \(A\) be a ring. An \(A\)-module \(F\) is called \emph{free} if it has a basis, equivalently if
\[
F\cong A^{(I)}
\]
for some index set \(I\). In the finitely generated case, this means
\[
F\cong A^n
\]
for some integer \(n\ge 0\).

Free modules are the module-theoretic analog of vector spaces with a chosen basis.

\subsection*{1.2. Projective modules}

An \(A\)-module \(P\) is called \emph{projective} if it satisfies the lifting property:

whenever
\[
M\twoheadrightarrow N
\]
is a surjective homomorphism and
\[
P\to N
\]
is any homomorphism, there exists a lift
\[
P\to M.
\]

A fundamental equivalent characterization is:
\[
P \text{ is projective }
\iff
P \text{ is a direct summand of a free module.}
\]

That is, there exists a free module \(F\) and an \(A\)-module \(Q\) such that
\[
F\cong P\oplus Q.
\]

Thus projective modules are exactly the pieces of free modules.

\subsection*{1.3. Free versus projective}

Every free module is projective, but the converse is false in general.

Problem 11 gives a very basic example. If
\[
A=A_1\times A_2,
\]
then as an \(A\)-module,
\[
A=(A_1\times 0)\oplus(0\times A_2).
\]
So the modules
\[
M_1=A_1\times 0,\qquad M_2=0\times A_2
\]
are direct summands of the free \(A\)-module \(A\), hence projective.

However, if \(A_1\neq 0\) and \(A_2\neq 0\), they are not free. The key obstruction is that they have nonzero annihilators, while a nonzero free module has annihilator \(0\).

So Problem 11 teaches an important principle:
\begin{quote}
	projective means ``summand of free,'' not ``has a basis.''
\end{quote}

\subsection*{1.4. Idempotents and projective modules}

An element \(e\in A\) is called an \emph{idempotent} if
\[
e^2=e.
\]

Such elements naturally give direct sum decompositions:
\[
A=Ae\oplus A(1-e).
\]

Hence ideals generated by idempotents are projective.

In Problem 11, for
\[
A=A_1\times A_2,
\]
the elements
\[
e_1=(1,0),\qquad e_2=(0,1)
\]
are orthogonal idempotents, and
\[
M_1=Ae_1,\qquad M_2=Ae_2.
\]

This is one of the main sources of projective modules in elementary algebra.

\bigskip

\section{2. Problem 12 theory: free resolutions, syzygies, \texorpdfstring{$\operatorname{Ext}$}{Ext}, and \texorpdfstring{$\operatorname{Tor}$}{Tor}}

\subsection*{2.1. Why resolutions are needed}

Modules are often complicated, but free modules are easy to understand. So one studies a module \(M\) by replacing it with a chain of free modules:
\[
\cdots \to F_2\to F_1\to F_0\to M\to 0.
\]
This is called a \emph{free resolution} of \(M\).

The point is that the structure of \(M\) is encoded in the maps between the free modules.

\subsection*{2.2. Exactness and syzygies}

A sequence
\[
\cdots \to F_2\xrightarrow{d_2}F_1\xrightarrow{d_1}F_0\to M\to 0
\]
is a free resolution if:
\begin{itemize}
	\item each \(F_i\) is free,
	\item the sequence is exact, i.e.
	\[
	\operatorname{im}(d_{i+1})=\ker(d_i)
	\]
	at each step.
\end{itemize}

The kernels that appear are called \emph{syzygy modules}. Concretely:
\begin{itemize}
	\item first syzygies are relations among generators,
	\item second syzygies are relations among relations,
	\item and so on.
\end{itemize}

\subsection*{2.3. The singular ring \texorpdfstring{$A=k[x,y]/(xy)$}{A = k[x,y]/(xy)}}

The ring
\[
A=k[x,y]/(xy)
\]
is a standard example of a singular affine ring. Geometrically, it is the coordinate ring of the union of the two coordinate axes:
\[
xy=0.
\]

This ring is reduced but not a domain, because
\[
x\cdot y=0.
\]

That presence of zero divisors is exactly why its homological behavior is interesting.

\subsection*{2.4. Periodic resolutions}

In Problem 12, the residue field
\[
k=A/\mathfrak m,\qquad \mathfrak m=(x,y),
\]
admits a free resolution that becomes \(2\)-periodic after the first step.

This is characteristic of many hypersurface singularities: after some stage, resolutions often become periodic.

That phenomenon is a deep feature of singular rings.

\subsection*{2.5. The functor \texorpdfstring{$\operatorname{Tor}$}{Tor}}

Given \(A\)-modules \(M,N\), the groups
\[
\operatorname{Tor}_i^A(M,N)
\]
measure the failure of tensor product to be exact.

Construction:
\begin{itemize}
	\item take a free resolution \(F_\bullet\to M\),
	\item tensor it with \(N\),
	\item define
	\[
	\operatorname{Tor}_i^A(M,N)=H_i(F_\bullet\otimes_A N).
	\]
\end{itemize}

The first term is
\[
\operatorname{Tor}_0^A(M,N)\cong M\otimes_A N.
\]
Higher \(\operatorname{Tor}\) detects hidden relations between \(M\) and \(N\).

In Problem 12, after tensoring with \(k\), all differentials vanish because \(x\) and \(y\) act by zero on \(k\). This makes the calculation very transparent.

\subsection*{2.6. The functor \texorpdfstring{$\operatorname{Ext}$}{Ext}}

Given \(A\)-modules \(M,N\), the groups
\[
\operatorname{Ext}_A^i(M,N)
\]
measure the failure of \(\operatorname{Hom}_A(-,N)\) to be exact in the first variable.

Construction:
\begin{itemize}
	\item take a free resolution \(F_\bullet\to M\),
	\item apply \(\operatorname{Hom}_A(-,N)\),
	\item define
	\[
	\operatorname{Ext}_A^i(M,N)=H^i(\operatorname{Hom}_A(F_\bullet,N)).
	\]
\end{itemize}

One has
\[
\operatorname{Ext}_A^0(M,N)=\operatorname{Hom}_A(M,N).
\]

The group \(\operatorname{Ext}_A^1(M,N)\) classifies extensions
\[
0\to N\to E\to M\to 0,
\]
and higher \(\operatorname{Ext}\) measures deeper homological complexity.

\subsection*{2.7. Meaning of Problem 12}

Problem 12 teaches several important principles at once:
\begin{itemize}
	\item singular rings often give infinite resolutions,
	\item those resolutions can be periodic,
	\item \(\operatorname{Ext}\) and \(\operatorname{Tor}\) are computed from resolutions,
	\item homological invariants detect singularity very clearly.
\end{itemize}

So Problem 12 is a prototype of explicit local homological algebra.

\bigskip

\section{3. Problem 13 theory: regular sequences, depth, and Cohen--Macaulay rings}

\subsection*{3.1. Non-zero-divisors}

Let \(M\) be an \(A\)-module. An element \(x\in A\) is a \emph{non-zero-divisor on \(M\)} if
\[
xm=0 \implies m=0
\]
for all \(m\in M\).

Equivalently, the multiplication map
\[
M\xrightarrow{\cdot x} M
\]
is injective.

This is the first building block of regular sequences.

\subsection*{3.2. Regular sequences}

Let \((A,\mathfrak m)\) be local and \(M\) an \(A\)-module. A sequence
\[
x_1,\dots,x_r\in \mathfrak m
\]
is called an \emph{\(M\)-regular sequence} if:
\begin{itemize}
	\item \(x_1\) is a non-zero-divisor on \(M\),
	\item \(x_2\) is a non-zero-divisor on \(M/x_1M\),
	\item \(x_3\) is a non-zero-divisor on \(M/(x_1,x_2)M\),
	\item and so on.
\end{itemize}

When \(M=A\), one simply says “regular sequence in \(A\).”

A regular sequence is the algebraic analog of independent parameters.

\subsection*{3.3. Depth}

The \emph{depth} of a finitely generated module \(M\) over a Noetherian local ring \((A,\mathfrak m)\) is the maximal length of an \(M\)-regular sequence contained in \(\mathfrak m\).

A fundamental homological characterization is
\[
\operatorname{depth}(M)
=
\min\{\,i\ge 0 \mid \operatorname{Ext}_A^i(k,M)\neq 0\,\},
\qquad k=A/\mathfrak m.
\]

This equivalence is one of the most important bridges between commutative algebra and homological algebra.

\subsection*{3.4. Cohen--Macaulay rings}

A Noetherian local ring \(A\) is called \emph{Cohen--Macaulay} if
\[
\operatorname{depth}(A)=\dim A.
\]

More generally, a finitely generated module \(M\) is Cohen--Macaulay if
\[
\operatorname{depth}(M)=\dim M.
\]

This means the module or ring has as many independent regular parameters as its dimension allows.

Cohen--Macaulay rings are homologically “as good as possible” relative to their dimension.

\subsection*{3.5. Problem 13 and semigroup rings}

The ring
\[
S=k[x^3,x^2y,xy^2,y^3]
\]
is a semigroup ring, also viewed as a Veronese-type subring of \(k[x,y]\).

Problem 13 asks one to prove that \(S_{\mathfrak m}\) is Cohen--Macaulay by exhibiting an explicit regular sequence, typically
\[
x^3,\ y^3.
\]

So the structure of the proof is:
\begin{itemize}
	\item produce a regular sequence of length \(2\),
	\item know that the local dimension is \(2\),
	\item conclude that depth equals dimension.
\end{itemize}

This is one of the most concrete ways to prove Cohen--Macaulayness.

\bigskip

\section{4. Problem 14 theory: minimal maps over local rings}

\subsection*{4.1. Why local rings simplify module theory}

Over a local ring \((A,\mathfrak m)\), the quotient
\[
M/\mathfrak m M
\]
is a vector space over
\[
k=A/\mathfrak m.
\]

This quotient measures the minimal number of generators of \(M\). Indeed, by Nakayama's lemma, a set of elements generates \(M\) if and only if their images generate \(M/\mathfrak mM\).

So studying a module modulo \(\mathfrak m\) is extremely powerful.

\subsection*{4.2. Minimal homomorphisms}

A homomorphism
\[
f:M\to N
\]
between finitely generated \(A\)-modules is called \emph{minimal} if
\[
f\otimes_A k: M\otimes_A k \to N\otimes_A k
\]
is an isomorphism.

Since
\[
M\otimes_A k\cong M/\mathfrak mM,
\]
this means exactly that
\[
M/\mathfrak mM \to N/\mathfrak mN
\]
is an isomorphism.

Thus a minimal map is one that becomes exactly an isomorphism after modding out by \(\mathfrak m\).

\subsection*{4.3. Characterization of minimal surjections}

Problem 14(a) proves the key criterion
\[
f \text{ minimal }
\iff
f \text{ is surjective and } \ker f\subseteq \mathfrak m M.
\]

This means:
\begin{itemize}
	\item the map is onto,
	\item its kernel is invisible modulo \(\mathfrak m\).
\end{itemize}

So the kernel contains only non-minimal relations.

\subsection*{4.4. Minimal free covers}

Problem 14(b) shows that every finitely generated module \(M\) admits a minimal surjection
\[
F\to M
\]
from a finitely generated free module \(F\).

Construction:
\begin{itemize}
	\item choose a basis of the \(k\)-vector space \(M/\mathfrak mM\),
	\item lift those basis elements to \(M\),
	\item map a free basis to those lifts.
\end{itemize}

This is exactly the algebraic version of choosing a minimal generating set.

\subsection*{4.5. Why part (c) matters}

Problem 14(c) shows that in a minimal presentation
\[
0\to K\xrightarrow{f}F\to M\to 0
\]
with \(K,F\) free, the induced maps on \(\operatorname{Ext}\)
\[
f_*:\operatorname{Ext}_A^i(k,K)\to \operatorname{Ext}_A^i(k,F)
\]
vanish.

The reason is that the matrix of \(f\) has entries in \(\mathfrak m\), and \(\mathfrak m\) acts trivially on \(\operatorname{Ext}_A^i(k,A)\).

This technical fact is exactly what later feeds into the proof of Auslander--Buchsbaum.

\bigskip

\section{5. Problem 15 theory: minimal free resolutions}

\subsection*{5.1. Minimal free resolutions}

A free resolution
\[
\cdots \to F_2\to F_1\to F_0\to M\to 0
\]
over a local ring is called \emph{minimal} if every differential becomes zero after tensoring with \(k=A/\mathfrak m\):
\[
d_i\otimes_A k=0.
\]

Equivalently, each image lies inside the maximal ideal multiple of the previous free module:
\[
\operatorname{im}(d_i)\subseteq \mathfrak m F_{i-1}.
\]

This means there are no unit coefficients in the matrices of the differentials, so nothing can be simplified further.

\subsection*{5.2. Existence over Noetherian local rings}

Problem 15 proves that minimal free resolutions exist when \(A\) is Noetherian local and \(M\) is finitely generated.

Why Noetherianity matters:
at each stage one forms a kernel
\[
K_i=\ker(F_i\to F_{i-1}),
\]
and one needs that kernel to be finitely generated so that one can choose a minimal surjection from a finitely generated free module onto it.

Noetherianity guarantees exactly that.

\subsection*{5.3. Why minimal resolutions are special}

A general free resolution may contain many unnecessary free summands. Minimal free resolutions remove all such redundancy.

Over a local ring, minimal free resolutions are essentially canonical: the ranks of the free modules are invariants of the module.

These ranks are the \emph{Betti numbers} of the module.

So minimal resolutions are the most efficient and informative free resolutions.

\bigskip

\section{6. Problem 16 theory: projective dimension and vanishing of \texorpdfstring{$\operatorname{Tor}$}{Tor}}

\subsection*{6.1. Projective dimension}

The \emph{projective dimension} of an \(A\)-module \(M\), denoted
\[
\operatorname{pd}_A M,
\]
is the smallest length of a finite projective resolution of \(M\). Over a local ring, finitely generated projective modules are free, so one may use finite free resolutions.

Thus
\[
\operatorname{pd}_A M=n
\]
means there exists a free resolution
\[
0\to F_n\to \cdots \to F_0\to M\to 0,
\]
and no shorter free resolution exists.

This measures how far \(M\) is from being free.

\subsection*{6.2. Minimal resolutions and \texorpdfstring{$\operatorname{Tor}$}{Tor}}

If
\[
F_\bullet\to M\to 0
\]
is a minimal free resolution, then after tensoring with
\[
k=A/\mathfrak m,
\]
all differentials become zero. Hence
\[
\operatorname{Tor}_i^A(k,M)\cong F_i\otimes_A k.
\]

This is one of the most important computational principles in local algebra.

It means:
\begin{itemize}
	\item \(\operatorname{Tor}_i^A(k,M)\) records the \(i\)-th free module in the minimal resolution,
	\item the dimension
	\[
	\beta_i(M)=\dim_k \operatorname{Tor}_i^A(k,M)
	\]
	is the \(i\)-th Betti number.
\end{itemize}

So \(\operatorname{Tor}\) literally measures the size of the minimal resolution.

\subsection*{6.3. Vanishing of \texorpdfstring{$\operatorname{Tor}$}{Tor} detects the length}

If the minimal free resolution has length \(n\), then
\[
F_n\neq 0,\qquad F_{n+1}=F_{n+2}=\cdots=0.
\]

Since
\[
\operatorname{Tor}_i^A(k,M)\cong F_i\otimes_A k,
\]
this implies
\[
\operatorname{Tor}_n^A(k,M)\neq 0,
\qquad
\operatorname{Tor}_{n+1}^A(k,M)=0.
\]

Thus
\[
\operatorname{pd}_A M
=
\min\{\,i\ge 0\mid \operatorname{Tor}_{i+1}^A(k,M)=0\,\}.
\]

That is exactly the statement proved in Problem 16.

\subsection*{6.4. Meaning of Problem 16}

Problem 16 says that if projective dimension is finite, then it can be recognized in two equivalent ways:
\begin{itemize}
	\item as the length of the minimal free resolution,
	\item as the first place where \(\operatorname{Tor}\) vanishes.
\end{itemize}

So projective dimension is visible both from resolutions and from derived functors.

\bigskip

\section{7. Problem 17 theory: depth and the Auslander--Buchsbaum formula}

\subsection*{7.1. Depth revisited}

For a finitely generated module \(M\) over a Noetherian local ring,
\[
\operatorname{depth}(M)
\]
measures how many independent non-zero-divisors one can find on \(M\).

Equivalently,
\[
\operatorname{depth}(M)
=
\min\{\,i\ge 0\mid \operatorname{Ext}_A^i(k,M)\neq 0\,\}.
\]

So depth may be understood either through regular sequences or through homological algebra.

\subsection*{7.2. Why free modules have full depth}

If \(F\cong A^r\) is a nonzero free module, then
\[
\operatorname{Ext}_A^i(k,F)\cong \bigl(\operatorname{Ext}_A^i(k,A)\bigr)^r,
\]
so
\[
\operatorname{depth}(F)=\operatorname{depth}(A).
\]

Thus free modules have the same depth as the ring itself.

This is one reason free modules behave as homologically perfect objects.

\subsection*{7.3. Auslander--Buchsbaum formula}

If \(M\) is a finitely generated module over a Noetherian local ring \(A\) and has finite projective dimension, then
\[
\operatorname{pd}_A M+\operatorname{depth}(M)=\operatorname{depth}(A).
\]

This is the \emph{Auslander--Buchsbaum formula}.

It is one of the most important structure theorems in local algebra.

Interpretation:
\begin{itemize}
	\item the larger the projective dimension,
	\item the smaller the depth must be,
\end{itemize}
and the two quantities balance exactly to give the depth of the ring.

\subsection*{7.4. Heuristic meaning}

Think of \(\operatorname{depth}(A)\) as the total homological room available in the ring.

A module \(M\) with finite projective dimension uses up part of that room in building its free resolution. The amount used is \(\operatorname{pd}_A M\). What remains is \(\operatorname{depth}(M)\).

That is why
\[
\operatorname{pd}_A M+\operatorname{depth}(M)=\operatorname{depth}(A).
\]

\subsection*{7.5. Why Problem 14(c) enters the proof}

The proof for projective dimension \(1\) uses a minimal presentation
\[
0\to K\to F\to M\to 0
\]
with \(K,F\) free.

Problem 14(c) says the induced maps on \(\operatorname{Ext}\) vanish, so the long exact sequence in \(\operatorname{Ext}\) simplifies dramatically. This makes it possible to locate the first nonvanishing \(\operatorname{Ext}\) of \(M\).

Then induction completes the proof for higher projective dimension.

So Problem 17 is not isolated: it is the culmination of Problems 14--16.

\bigskip

\section{8. Compact dictionary for Problems 11--17}

\subsection*{8.1. Module-theoretic notions}

\begin{itemize}
	\item \textbf{Free module:} a module with a basis, isomorphic to \(A^n\) in the finitely generated case.
	\item \textbf{Projective module:} a direct summand of a free module.
	\item \textbf{Annihilator of \(M\):}
	\[
	\operatorname{Ann}_A(M)=\{a\in A\mid aM=0\}.
	\]
\end{itemize}

\subsection*{8.2. Resolution-theoretic notions}

\begin{itemize}
	\item \textbf{Free resolution:}
	\[
	\cdots\to F_2\to F_1\to F_0\to M\to 0.
	\]
	\item \textbf{Syzygy:} a kernel appearing in a resolution.
	\item \textbf{Minimal free resolution:} a free resolution whose differentials become zero after tensoring with \(k=A/\mathfrak m\).
\end{itemize}

\subsection*{8.3. Homological notions}

\begin{itemize}
	\item \textbf{\(\operatorname{Ext}_A^i(M,N)\):} the cohomology obtained by applying \(\operatorname{Hom}_A(-,N)\) to a free resolution of \(M\).
	\item \textbf{\(\operatorname{Tor}_i^A(M,N)\):} the homology obtained by tensoring a free resolution of one variable with the other.
	\item \textbf{Projective dimension:}
	\[
	\operatorname{pd}_A M
	\]
	= minimal length of a finite free/projective resolution.
\end{itemize}

\subsection*{8.4. Local algebra notions}

\begin{itemize}
	\item \textbf{Regular sequence:} a sequence of successive non-zero-divisors.
	\item \textbf{Depth:}
	maximal length of a regular sequence in \(\mathfrak m\), equivalently
	\[
	\operatorname{depth}(M)=\min\{i\mid \operatorname{Ext}_A^i(k,M)\neq 0\}.
	\]
	\item \textbf{Cohen--Macaulay local ring:}
	\[
	\operatorname{depth}(A)=\dim A.
	\]
\end{itemize}

\subsection*{8.5. Central formulas}

\begin{itemize}
	\item In a minimal free resolution,
	\[
	\operatorname{Tor}_i^A(k,M)\cong F_i\otimes_A k.
	\]
	\item For finite projective dimension,
	\[
	\operatorname{pd}_A M
	=
	\min\{\,i\ge 0\mid \operatorname{Tor}_{i+1}^A(k,M)=0\,\}.
	\]
	\item Auslander--Buchsbaum:
	\[
	\operatorname{pd}_A M+\operatorname{depth}(M)=\operatorname{depth}(A).
	\]
\end{itemize}

\section{9. Final conceptual summary}

Problems 11--17 develop a major package of local homological algebra.

\begin{itemize}
	\item \textbf{Problem 11} separates projective modules from free modules.
	\item \textbf{Problem 12} shows how explicit resolutions let us compute \(\operatorname{Ext}\) and \(\operatorname{Tor}\).
	\item \textbf{Problem 13} introduces regular sequences and Cohen--Macaulayness.
	\item \textbf{Problem 14} explains minimal generators and minimal presentations over local rings.
	\item \textbf{Problem 15} builds minimal free resolutions.
	\item \textbf{Problem 16} shows that projective dimension is encoded by minimal resolutions and by \(\operatorname{Tor}\).
	\item \textbf{Problem 17} ties everything together in the Auslander--Buchsbaum formula.
\end{itemize}

So this block gives the central philosophy:
\[
\text{minimal free resolution}
\Longrightarrow
\text{Betti numbers / Tor}
\Longrightarrow
\text{projective dimension}
\Longrightarrow
\text{depth formula}.
\]

This is one of the foundational packages behind modern commutative algebra, singularity theory, and algebraic geometry.
