\chapter{Differential Calculus}
\chapteropeningart{Change becomes measurable}{Limits and derivatives turn local change into a precise mathematical language.}
\begin{roadmapbox}
Functions $\longrightarrow$ limits $\longrightarrow$ derivatives $\longrightarrow$ partial derivatives $\longrightarrow$ gradient and directional derivatives.
\end{roadmapbox}
\learningobjectives
\begin{itemize}
\item understand functions and limits;
\item derive and interpret ordinary and partial derivatives;
\item use total differentials and the operator $\nabla$;
\item calculate gradients and directional derivatives;
\item connect differential calculus to motion, fields, and optimization.
\end{itemize}
