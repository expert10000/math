\chapter{Vector Calculus}
\chapteropeningart{Local structure and global laws}{Vector fields are analyzed locally by divergence, curl, and the Laplacian, and globally by line, surface, and volume integrals.}
\begin{roadmapbox}
Vector fields $\longrightarrow$ divergence and curl $\longrightarrow$ Laplacian $\longrightarrow$ integral operators $\longrightarrow$ Green, Stokes, and Gauss.
\end{roadmapbox}
\learningobjectives
\begin{itemize}
\item interpret vector fields geometrically;
\item calculate divergence, curl, and Laplacians;
\item understand line, surface, and volume integrals;
\item connect differential and integral descriptions through Green's, Stokes', and Gauss' theorems;
\item recognize these structures in electromagnetism, fluid dynamics, and quantum mechanics.
\end{itemize}
