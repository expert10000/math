\section{Worked Example}

\begin{examplebox}
A force \(\mathbf F=(10,0,0)\,\mathrm N\) acts on a \(2\,\mathrm{kg}\) mass.
Find the acceleration.
\end{examplebox}

\begin{solutionbox}
Newton's second law gives
\[
\mathbf a=\frac{\mathbf F}{m}
=\frac{(10,0,0)}{2}
=(5,0,0)\,\mathrm{m/s^2}.
\]
\end{solutionbox}
