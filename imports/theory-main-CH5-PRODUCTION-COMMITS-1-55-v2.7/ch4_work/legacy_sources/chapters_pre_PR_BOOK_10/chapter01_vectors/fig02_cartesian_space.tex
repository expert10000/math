% Right-handed Cartesian coordinates in three dimensions
\begin{tikzpicture}[scale=1.0]
  \coordinate (O) at (0,0);
  \coordinate (X) at (5.2,0);
  \coordinate (Y) at (-2.2,-1.5);
  \coordinate (Z) at (0,4.3);
  \draw[axis] (O)--(X) node[right] {$x$};
  \draw[axis] (O)--(Y) node[below left] {$y$};
  \draw[axis] (O)--(Z) node[above] {$z$};
  \node[below right] at (O) {$O$};
  \coordinate (P) at (3.4,2.8);
  \draw[construction] (P)--(3.4,0);
  \draw[construction] (P)--(.7,-1.0);
  \draw[construction] (.7,-1.0)--(O);
  \draw[vector] (O)--(P) node[midway,left,label] {$\mathbf r$};
  \fill[HighlightGold] (P) circle (2.2pt) node[above right,label] {$P(x,y,z)$};
  \draw[ForceRed,line width=1pt,->] (1.2,.25) arc[start angle=10,end angle=305,x radius=.65,y radius=.32];
  \node[font=\scriptsize,text=ForceRed] at (1.75,.55) {right-handed};
\end{tikzpicture}
