\begin{tikzpicture}[scale=.78]
\begin{scope}[xshift=-3.6cm]
  \fill[visualred] (0,0) circle (2.4pt);
  \foreach \a in {0,30,...,330}{
    \foreach \r in {1.0,2.0}{
      \pgfmathsetmacro{\len}{.58/\r}
      \draw[vector secondary,line width=.8pt]
        ({\r*cos(\a)},{\r*sin(\a)})--++({\len*cos(\a)},{\len*sin(\a)});
    }
  }
  \node[below] at (0,-2.65) {outward field};
\end{scope}
\begin{scope}[xshift=3.6cm]
  \fill[visualblue] (0,0) circle (2.4pt);
  \foreach \a in {0,30,...,330}{
    \foreach \r in {1.0,2.0}{
      \pgfmathsetmacro{\len}{.58/\r}
      \draw[vector secondary,line width=.8pt]
        ({\r*cos(\a)},{\r*sin(\a)})--++({-\len*cos(\a)},{-\len*sin(\a)});
    }
  }
  \node[below] at (0,-2.65) {inward field};
