\section*{Worked Examples}

\begin{workedexample}{Magnitude and direction}
Given $\mathbf v=(3,4)$, find its magnitude and unit vector.
\end{workedexample}
\begin{solution}
\[
\|\mathbf v\|=\sqrt{3^2+4^2}=5,
\qquad
\hat{\mathbf v}=\frac{\mathbf v}{\|\mathbf v\|}
=\left(\frac35,\frac45\right).
\]
\end{solution}

\begin{workedexample}{Projection}
Let $\mathbf a=(3,2)$ and $\mathbf b=(4,0)$. Find
$\operatorname{proj}_{\mathbf b}\mathbf a$.
\end{workedexample}
\begin{solution}
\[
\operatorname{proj}_{\mathbf b}\mathbf a
=
\frac{\mathbf a\cdot\mathbf b}{\|\mathbf b\|^2}\mathbf b
=
\frac{12}{16}(4,0)
=
(3,0).
\]
\end{solution}

\begin{workedexample}{Torque}
A force $\mathbf F=(0,12,0)\,\mathrm N$ acts at
$\mathbf r=(0.5,0,0)\,\mathrm m$. Find the torque.
\end{workedexample}
\begin{solution}
\[
\boldsymbol\tau=\mathbf r\times\mathbf F
=(0,0,6)\,\mathrm{N\,m}.
\]
\end{solution}

\section*{Exercises}

\subsection*{Basic}
\begin{enumerate}
  \item Find the magnitude of $(5,12)$.
  \item Normalize $(2,-2,1)$.
  \item Add $(3,1)$ and $(-2,4)$ geometrically and algebraically.
  \item Find the displacement from $A=(1,2)$ to $B=(6,-1)$.
  \item Determine whether $(1,2)$ and $(2,-1)$ are orthogonal.
  \item Find the angle between $(1,0)$ and $(1,1)$.
  \item Project $(4,3)$ onto the $x$-axis.
  \item Compute $(1,2,3)\times(0,1,0)$.
  \item Find the area of the parallelogram generated by $(2,0,0)$ and $(0,3,0)$.
  \item Reflect $(4,-2)$ across the $y$-axis.
\end{enumerate}

\subsection*{Intermediate}
\begin{enumerate}
  \setcounter{enumi}{10}
  \item Resolve a force of magnitude $20\,\mathrm N$ at $35^\circ$ above the horizontal.
  \item Find the scalar and vector projection of $(3,4)$ onto $(1,1)$.
  \item Show that the dot product is invariant under rotation.
  \item Rotate $(2,1)$ counterclockwise by $60^\circ$.
  \item Determine the volume generated by $(1,0,1)$, $(0,2,0)$, and $(2,0,0)$.
  \item Decide whether three given vectors form a right-handed basis.
  \item Derive the matrix for reflection across the line $y=x$.
  \item Express a point in both Cartesian and polar coordinates.
  \item Compare active rotation of a vector with passive rotation of the basis.
  \item Construct a local tangent-normal basis for the curve $y=x^2$ at $x=1$.
\end{enumerate}

\subsection*{Advanced}
\begin{enumerate}
  \setcounter{enumi}{20}
  \item Prove the vector triple-product identity.
  \item Derive Rodrigues' rotation formula.
  \item Show that every orthogonal matrix has determinant $\pm1$.
  \item Explain geometrically why reflections reverse orientation.
  \item Derive the change-of-basis matrix between two orthonormal bases.
  \item Show that translation is not a linear map in ordinary vector coordinates.
  \item Represent a planar rigid motion using homogeneous coordinates.
  \item Analyze the field $\mathbf F=(-y,x)$ and sketch several integral curves.
  \item Compare Cartesian, cylindrical, and spherical bases at the same point.
  \item Prove that the scalar triple product is invariant under cyclic permutation.
\end{enumerate}
