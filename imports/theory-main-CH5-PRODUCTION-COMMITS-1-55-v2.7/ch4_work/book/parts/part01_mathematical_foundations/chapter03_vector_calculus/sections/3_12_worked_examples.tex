\section{Integrated Worked Examples}
\label{sec:ch3-worked-examples}

The following examples form a continuous problem-solving sequence.  Each calculation is paired with a geometric or physical interpretation so that the operators are not reduced to formal symbol manipulation.

\subsection{Fields and differential operators}

\begin{workedexample}
\textbf{Radial inverse-square field.}\par
\textbf{Problem.} For $\mathbf F(\mathbf r)=k\mathbf r/r^3$ with $r\ne0$, determine its magnitude and direction.
\begin{solution}
Since $|\mathbf r|=r$, one has $|\mathbf F|=|k|/r^2$.  For $k>0$ the field points radially outward; for $k<0$ it points inward.

\textbf{Interpretation.} This is the geometric form shared by ideal point-source gravitational and electrostatic fields.  The singular point must be excluded from ordinary differentiation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Gradient and equipotential curves.}\par
\textbf{Problem.} Let $\phi(x,y)=x^2+2y^2$. Find $\nabla\phi$ and describe its relation to the level curve through $(1,1)$.
\begin{solution}
$\nabla\phi=(2x,4y)$, hence $\nabla\phi(1,1)=(2,4)$. The level curve is $x^2+2y^2=3$. Its tangent vectors are orthogonal to $(2,4)$ at $(1,1)$.

\textbf{Interpretation.} The gradient points across equipotential curves in the direction of greatest increase.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Divergence of an expanding flow.}\par
\textbf{Problem.} Compute $\nabla\cdot\mathbf v$ for $\mathbf v=(ax,ay,az)$.
\begin{solution}
$\nabla\cdot\mathbf v=a+a+a=3a$.

\textbf{Interpretation.} Every small material volume expands at fractional rate $3a$ when $a>0$, and contracts when $a<0$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{An incompressible shear.}\par
\textbf{Problem.} For $\mathbf v=(\gamma y,0,0)$, calculate divergence and curl.
\begin{solution}
$\nabla\cdot\mathbf v=0$, while
\[
\nabla\times\mathbf v=(0,0,-\gamma).
\]

\textbf{Interpretation.} The flow preserves volume but possesses local rotational tendency. Divergence and curl measure independent aspects of a field.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Rigid rotation.}\par
\textbf{Problem.} Let $\mathbf v=\boldsymbol\Omega\times\mathbf r$ with constant $\boldsymbol\Omega$. Show that $\nabla\times\mathbf v=2\boldsymbol\Omega$.
\begin{solution}
Choosing $\boldsymbol\Omega=\Omega\hat{\mathbf z}$ gives $\mathbf v=(-\Omega y,\Omega x,0)$. Direct differentiation yields $\nabla\times\mathbf v=(0,0,2\Omega)$; rotational covariance gives the general result.

\textbf{Interpretation.} Fluid vorticity is twice the local angular-velocity vector for rigid rotation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Laplacian of a Gaussian.}\par
\textbf{Problem.} Find $\nabla^2 e^{-\alpha r^2}$ in three dimensions.
\begin{solution}
For a radial function, $\nabla^2f=r^{-2}\frac{d}{dr}(r^2f')$. With $f'=-2\alpha r e^{-\alpha r^2}$,
\[
\nabla^2e^{-\alpha r^2}=(4\alpha^2r^2-6\alpha)e^{-\alpha r^2}.
\]

\textbf{Interpretation.} Near the peak the Laplacian is negative; farther away it changes sign, encoding local curvature relative to neighboring values.
\end{solution}
\end{workedexample}

\subsection{Line integrals and potentials}

\begin{workedexample}
\textbf{Work along a straight path.}\par
\textbf{Problem.} Evaluate $\int_C\mathbf F\cdot d\mathbf r$ for $\mathbf F=(y,x)$ from $(0,0)$ to $(1,2)$ along the straight segment.
\begin{solution}
Use $\mathbf r(t)=(t,2t)$, $0\le t\le1$. Then $\mathbf F=(2t,t)$ and $d\mathbf r=(1,2)dt$, so
\[
\int_0^1(2t+2t)dt=2.
\]

\textbf{Interpretation.} Since $\mathbf F=\nabla(xy)$, the result also equals $xy|_{(1,2)}-xy|_{(0,0)}=2$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Path independence.}\par
\textbf{Problem.} Find the work done by $\mathbf F=(2x,2y,2z)$ from $A=(1,0,0)$ to $B=(0,2,1)$.
\begin{solution}
$\mathbf F=\nabla(x^2+y^2+z^2)$. Therefore
\[
W=(0^2+2^2+1^2)-(1^2+0+0)=4.
\]

\textbf{Interpretation.} A conservative field converts a potentially difficult path integral into an endpoint calculation.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Circulation around a circle.}\par
\textbf{Problem.} Compute $\oint_C(-y,x,0)\cdot d\mathbf r$ around the circle $x^2+y^2=R^2$, counterclockwise.
\begin{solution}
With $\mathbf r=(R\cos t,R\sin t,0)$, the dot product is $R^2dt$. Hence
\[
\oint_C\mathbf F\cdot d\mathbf r=\int_0^{2\pi}R^2dt=2\pi R^2.
\]

\textbf{Interpretation.} The result is twice the enclosed area, consistent with curl $(0,0,2)$ and Stokes' theorem.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{A field that is locally but not globally conservative.}\par
\textbf{Problem.} For $\mathbf F=(-y/(x^2+y^2),x/(x^2+y^2))$ on the punctured plane, evaluate circulation around the unit circle.
\begin{solution}
On $\mathbf r=(\cos t,\sin t)$, $\mathbf F=(-\sin t,\cos t)$ equals $\mathbf r'(t)$. Thus $\oint_C\mathbf F\cdot d\mathbf r=2\pi$.

\textbf{Interpretation.} The curl vanishes away from the origin, but the domain is not simply connected. Topology prevents a single-valued global potential.
\end{solution}
\end{workedexample}

\subsection{Surface and volume integrals}

\begin{workedexample}
\textbf{Flux through a plane.}\par
\textbf{Problem.} Find the flux of constant $\mathbf F=(1,2,3)$ through the rectangle $0\le x\le2$, $0\le y\le1$ in the plane $z=0$, oriented upward.
\begin{solution}
$d\mathbf S=\hat{\mathbf z}\,dxdy$, so $\mathbf F\cdot d\mathbf S=3dxdy$. The flux is $3(2)(1)=6$.

\textbf{Interpretation.} Only the component normal to the surface contributes.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Flux through a sphere.}\par
\textbf{Problem.} Compute the outward flux of $\mathbf F=\mathbf r$ through a sphere of radius $R$.
\begin{solution}
On the sphere, $\mathbf F\cdot\hat{\mathbf n}=R$ and the area is $4\pi R^2$. Therefore $\Phi=4\pi R^3$.

\textbf{Interpretation.} Gauss' theorem gives the same result because $\nabla\cdot\mathbf F=3$ and the enclosed volume is $4\pi R^3/3$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Mass from a nonuniform density.}\par
\textbf{Problem.} A ball of radius $R$ has density $\rho(r)=\rho_0(1-r/R)$. Find its mass.
\begin{solution}
Using spherical shells,
\[
M=4\pi\rho_0\int_0^R\left(1-\frac rR\right)r^2dr
=\frac{\pi}{3}\rho_0R^3.
\]

\textbf{Interpretation.} Volume integration accumulates a local material property over the body.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Cylindrical Jacobian.}\par
\textbf{Problem.} Evaluate the volume of a cylinder $0\le r\le a$, $0\le z\le h$.
\begin{solution}
The volume element is $dV=r\,dr\,d\theta\,dz$, hence
\[
V=\int_0^h\int_0^{2\pi}\int_0^a r\,dr\,d\theta\,dz=\pi a^2h.
\]

\textbf{Interpretation.} The factor $r$ is the Jacobian that accounts for the widening of angular coordinate cells.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Moment of inertia of a uniform disk.}\par
\textbf{Problem.} Find $I_z$ for a disk of radius $R$, surface density $\sigma$.
\begin{solution}
\[
I_z=\int r^2\,dm=\int_0^{2\pi}\int_0^R r^2\sigma r\,dr\,d\theta
=\frac12(\pi R^2\sigma)R^2=\frac12MR^2.
\]

\textbf{Interpretation.} The extra $r^2$ weights matter farther from the axis more strongly.
\end{solution}
\end{workedexample}

\subsection{Integral theorems}

\begin{workedexample}
\textbf{Green's theorem for circulation.}\par
\textbf{Problem.} Evaluate $\oint_C(-y^3\,dx+x^3\,dy)$ around the unit circle counterclockwise.
\begin{solution}
Green's theorem gives
\[
\iint_D\left(3x^2+3y^2\right)dA
=3\int_0^{2\pi}\int_0^1r^2r\,dr\,d\theta=\frac{3\pi}{2}.
\]

\textbf{Interpretation.} Boundary circulation equals the accumulated local rotation over the enclosed region.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Stokes' theorem on a disk.}\par
\textbf{Problem.} For $\mathbf F=(-y,x,z)$, calculate circulation around $x^2+y^2=R^2$, $z=0$.
\begin{solution}
$\nabla\times\mathbf F=(0,0,2)$. With upward normal,
\[
\oint_C\mathbf F\cdot d\mathbf r=\iint_S2\,dA=2\pi R^2.
\]

\textbf{Interpretation.} Any surface spanning the same oriented boundary gives the same result, provided the field is regular throughout it.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Gauss' theorem for a cube.}\par
\textbf{Problem.} Find the total outward flux of $\mathbf F=(x^3,y^3,z^3)$ through the cube $[-a,a]^3$.
\begin{solution}
$\nabla\cdot\mathbf F=3(x^2+y^2+z^2)$. Symmetry and separation give
\[
\Phi=3\left[3\left(\int_{-a}^ax^2dx\right)(2a)^2\right]=24a^5.
\]

\textbf{Interpretation.} A six-face surface calculation is replaced by one volume integral.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Electrostatic Gauss law.}\par
\textbf{Problem.} A spherically symmetric charge density is constant, $\rho$, inside radius $R$. Find the electric field for $r<R$.
\begin{solution}
Symmetry gives $\mathbf E=E(r)\hat{\mathbf r}$. The enclosed charge is $Q_r=4\pi\rho r^3/3$. Gauss' law yields
\[
E(r)4\pi r^2=\frac{Q_r}{\varepsilon_0},\qquad
\mathbf E=\frac{\rho r}{3\varepsilon_0}\hat{\mathbf r}.
\]

\textbf{Interpretation.} Inside a uniform sphere the field grows linearly because the enclosed charge grows as $r^3$ while area grows as $r^2$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Continuity equation check.}\par
\textbf{Problem.} Let $\rho(x,t)=\rho_0e^{-t}e^{-x}$ and $j(x,t)=\rho_0e^{-t}e^{-x}$. Verify one-dimensional conservation.
\begin{solution}
$\partial_t\rho=-\rho_0e^{-t-x}$ and $\partial_xj=-\rho_0e^{-t-x}$, so $\partial_t\rho+\partial_xj=-2\rho_0e^{-t-x}\ne0$. Therefore the proposed pair is not conserved. Choosing $j=-\rho_0e^{-t-x}$ instead gives $\partial_t\rho+\partial_xj=0$.

\textbf{Interpretation.} Conservation constrains density and current together; neither may be chosen independently.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Diffusion of a Fourier mode.}\par
\textbf{Problem.} Verify that $u(x,t)=A e^{-Dk^2t}\cos(kx)$ solves $\partial_tu=D\partial_x^2u$.
\begin{solution}
$\partial_tu=-Dk^2u$, while $\partial_x^2u=-k^2u$. Hence both sides equal $-Dk^2u$.

\textbf{Interpretation.} Fine spatial structure (large $k$) decays faster, which is why diffusion smooths profiles.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Harmonic potential.}\par
\textbf{Problem.} Show that $\phi=x^2-y^2$ is harmonic and determine its field $\mathbf E=-\nabla\phi$.
\begin{solution}
$\nabla^2\phi=2-2=0$, and $\mathbf E=(-2x,2y)$.

\textbf{Interpretation.} The potential has saddle curvature but zero average curvature. In a charge-free electrostatic region, such harmonic potentials are admissible.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Vector identity check.}\par
\textbf{Problem.} Verify $\nabla\cdot(\nabla\times\mathbf A)=0$ for $\mathbf A=(yz,zx,xy)$.
\begin{solution}
Directly, $\nabla\times\mathbf A=(0,0,0)$, so its divergence vanishes. More generally, equality of mixed partial derivatives makes the divergence of every smooth curl zero.

\textbf{Interpretation.} This identity is the mathematical reason magnetic fields written as curls automatically satisfy $\nabla\cdot\mathbf B=0$.
\end{solution}
\end{workedexample}

\begin{workedexample}
\textbf{Helmholtz-style reconstruction insight.}\par
\textbf{Problem.} Suppose a smooth field in a simply connected region has both zero divergence and zero curl and decays at infinity. What can be concluded?
\begin{solution}
Zero curl implies $\mathbf F=\nabla\phi$. Zero divergence then gives $\nabla^2\phi=0$. With decay at infinity and no singularities, the relevant uniqueness result gives $\phi$ constant, so $\mathbf F=0$.

\textbf{Interpretation.} Divergence, curl, boundary behavior, and topology jointly determine a field; local differential data alone may not suffice without global conditions.
\end{solution}
\end{workedexample}
