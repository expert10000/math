\chapter{Vector Calculus and Physical Fields}
\label{ch:vector-calculus}
\chapteropeningart{Local structure and global laws}{Fields vary point by point, while integrals accumulate their effects over curves, surfaces, and volumes.}
\begin{roadmapbox}
Scalar and vector fields $\longrightarrow$ gradient, divergence, curl, and Laplacian $\longrightarrow$ line, surface, and volume integrals $\longrightarrow$ Green, Stokes, and Gauss $\longrightarrow$ conservation laws and Maxwell's equations.
\end{roadmapbox}
\learningobjectives
\begin{itemize}
\item interpret scalar and vector fields geometrically and physically;
\item calculate and explain gradient, divergence, curl, and Laplacians;
\item construct line, surface, and volume integrals from parametrizations;
\item distinguish circulation, flux, density, and accumulated quantity;
\item connect differential and integral descriptions through Green's, Stokes', and Gauss' theorems;
\item recognize the same structures in fluid dynamics, gravitation, electromagnetism, and quantum mechanics.
\end{itemize}
