\chapter{Differential Calculus}
\label{ch:differential-calculus}
\chapteropeningart{Change becomes measurable}{Limits and derivatives turn local change into a precise mathematical language.}

\begin{roadmapbox}
\textbf{Dependence} (functions) $\longrightarrow$ \textbf{approach} (limits) $\longrightarrow$ \textbf{local change} (derivatives) $\longrightarrow$ \textbf{many-variable change} (partials and differentials) $\longrightarrow$ \textbf{geometric change} (gradients and directions).
\end{roadmapbox}

\learningobjectives
\begin{itemize}
\item distinguish domains, codomains, ranges, and scalar- or vector-valued functions;
\item use limits to define continuity and instantaneous rates of change;
\item derive and interpret ordinary derivatives, partial derivatives, and total differentials;
\item apply the one-variable and multivariable chain rules;
\item compute gradients and directional derivatives and interpret them geometrically;
\item use Jacobian matrices to describe local linear transformations;
\item connect differential calculus to motion, fields, optimization, and quantum mechanics.
\end{itemize}

\begin{bookinsight}[How this chapter builds on Chapter~\ref{ch:mathematical-language-of-physics}]
Chapter~\ref{ch:mathematical-language-of-physics} established vectors, bases, inner products, and linear transformations. This chapter adds change. The derivative of a scalar function is a local slope; the derivative of a vector-valued function is a tangent vector; the Jacobian is a local linear map; and the gradient converts the differential of a scalar field into a vector through the Euclidean inner product.
\end{bookinsight}
