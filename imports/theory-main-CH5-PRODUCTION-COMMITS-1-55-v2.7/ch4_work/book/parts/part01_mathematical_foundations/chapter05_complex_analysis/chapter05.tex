\chapter{Complex Analysis: Geometry, Analyticity, and Physical Phase}
\label{chap:05-complex-analysis}
\chapteropeningart{The geometry of phase}{Complex numbers turn oscillation into rotation and unite algebra, geometry, and analysis in a single plane.}

\begin{roadmapbox}
Why real numbers are not enough $\longrightarrow$ the complex plane and polar form $\longrightarrow$ multiplication as rotation and scaling $\longrightarrow$ Euler's formula and roots $\longrightarrow$ complex functions as plane mappings $\longrightarrow$ limits, derivatives, analyticity, and the Cauchy--Riemann equations.
\end{roadmapbox}

\learningobjectives
\begin{itemize}
\item perform algebraic operations with complex numbers and interpret them geometrically;
\item move fluently between Cartesian, polar, and exponential forms;
\item explain why multiplication combines scaling with rotation;
\item derive and use Euler's formula and de Moivre's theorem;
\item interpret a complex function as a mapping between planes;
\item define complex differentiability and derive the Cauchy--Riemann equations;
\item connect complex phase to oscillations, waves, circuits, and quantum states.
\end{itemize}
