\chapter{Mathematical Language of Physics}
\label{ch:mathematical-language-of-physics}

\chapteropeningart
{From measurement to quantum states}
{Physics becomes predictive when observations are organized by mathematical structures. We begin with quantities and geometry, then advance toward the vector spaces and operators used in quantum mechanics.}

\chapterquote{``The book of Nature is written in the language of mathematics.''}{Galileo Galilei}

\begin{roadmapbox}
Physical quantities $\longrightarrow$ geometry and vectors $\longrightarrow$
coordinates and vector algebra $\longrightarrow$ products and projections
$\longrightarrow$ bases and abstract vector spaces.
\end{roadmapbox}

\learningobjectives
\begin{itemize}
\item explain why mathematical modeling and experimental testing are inseparable in physics;
\item distinguish a physical quantity from its numerical representation and unit;
\item interpret measurement as comparison with a standard and recognize the role of uncertainty;
\item distinguish scalars from vectors and interpret vectors geometrically;
\item use coordinates, vector algebra, products, projections, bases, and dimension;
\item recognize how geometric vectors lead toward abstract state vectors in quantum mechanics.
\end{itemize}
