\newglossaryentry{vector}{
  name=vector,
  description={A quantity characterized by magnitude and direction}
}
\newglossaryentry{projection}{
  name=projection,
  description={The component of one vector or state along a chosen direction or subspace}
}
\newglossaryentry{wavefunction}{
  name=wavefunction,
  description={A complex-valued state amplitude used to describe a quantum system}
}
\newacronym{qhe}{QHE}{quantum Hall effect}
\newglossaryentry{scalar}{
  name=scalar,
  description={A quantity represented by a single value once its unit is specified and invariant under a rotation of coordinates}
}
\newglossaryentry{basis}{
  name=basis,
  description={A linearly independent spanning set that provides unique coordinates for every vector in a vector space}
}
\newglossaryentry{norm}{
  name=norm,
  description={A nonnegative function that measures the size of a vector and satisfies homogeneity and the triangle inequality}
}
\newglossaryentry{innerproduct}{
  name=inner product,
  description={A scalar-valued pairing that defines lengths, angles, orthogonality, and projections}
}
\newglossaryentry{linearindependence}{
  name=linear independence,
  description={The property that no vector in a set can be written as a linear combination of the others}
}
\newglossaryentry{lineartransformation}{
  name=linear transformation,
  description={A map that preserves vector addition and scalar multiplication}
}
\newglossaryentry{eigenvector}{
  name=eigenvector,
  description={A nonzero vector whose direction is preserved by a linear transformation, up to multiplication by an eigenvalue}
}
\newglossaryentry{orthogonalprojection}{
  name=orthogonal projection,
  description={The closest vector in a chosen line or subspace, obtained by requiring the residual to be orthogonal to that subspace}
}

% Chapter 2: Differential Calculus
\newglossaryentry{function}{
  name=function,
  description={A rule assigning each element of a domain exactly one element of a codomain}
}
\newglossaryentry{limit}{
  name=limit,
  description={The value approached by a function as its argument approaches a specified point or regime}
}
\newglossaryentry{continuity}{
  name=continuity,
  description={The property that sufficiently small changes of the input produce arbitrarily small changes of the output}
}
\newglossaryentry{derivative}{
  name=derivative,
  description={The unique best first-order linear approximation to a differentiable function at a point}
}
\newglossaryentry{partialderivative}{
  name=partial derivative,
  description={The rate of change of a multivariable function with respect to one coordinate while the other coordinates are held fixed}
}
\newglossaryentry{differential}{
  name=differential,
  description={The linear map that gives the first-order change of a differentiable function}
}
\newglossaryentry{gradient}{
  name=gradient,
  description={The vector representing the differential of a scalar field through the inner product and pointing toward steepest increase}
}
\newglossaryentry{directionalderivative}{
  name=directional derivative,
  description={The instantaneous rate of change of a function in a specified unit direction}
}
\newglossaryentry{jacobian}{
  name=Jacobian matrix,
  description={The matrix representing the derivative of a vector-valued function in chosen coordinates}
}
\newglossaryentry{frechetderivative}{
  name=Fr\'{e}chet derivative,
  description={A coordinate-independent derivative defined by a linear approximation whose remainder is negligible compared with the input displacement}
}
\newglossaryentry{chainrule}{
  name=chain rule,
  description={The composition law stating that the derivative of a composite map is the composition of the derivatives}
}
\newglossaryentry{tangentplane}{
  name=tangent plane,
  description={The affine plane determined by the first-order linearization of a differentiable surface at a point}
}
\newglossaryentry{levelset}{
  name=level set,
  description={The set of points at which a scalar function has one fixed value}
}


\newglossaryentry{vectorfield}{name={vector field},description={a function assigning a vector to every point of a domain}}

\newglossaryentry{divergence}{name={divergence},description={the scalar field measuring local outward source strength of a vector field}}

\newglossaryentry{curl}{name={curl},description={the vector field measuring local circulation density and its oriented axis}}

\newglossaryentry{laplacian}{name={Laplacian},description={the second-order operator given by the divergence of the gradient}}

\newglossaryentry{lineintegral}{name={line integral},description={an integral accumulating a scalar or vector field along a parametrized curve}}

\newglossaryentry{surfaceintegral}{name={surface integral},description={an integral accumulating density or flux over a parametrized surface}}

\newglossaryentry{volumeintegral}{name={volume integral},description={an integral accumulating a density throughout a region}}

\newglossaryentry{greenstheorem}{name={Green's theorem},description={the planar boundary theorem relating circulation or flux to an area integral}}

\newglossaryentry{stokestheorem}{name={Stokes' theorem},description={the theorem relating circulation around a boundary curve to curl flux through a spanning surface}}

\newglossaryentry{divergencetheorem}{name={divergence theorem},description={the theorem equating outward flux through a closed surface with integrated divergence inside}}

\newglossaryentry{conservativefield}{name={conservative field},description={a vector field representable as the gradient of a scalar potential}}

\newglossaryentry{continuityequation}{name={continuity equation},description={a local conservation law relating density change to current divergence}}

% Chapter 4: Ordinary Differential Equations and Dynamical Systems
\newglossaryentry{ordinarydifferentialequation}{name={ordinary differential equation},description={an equation relating an unknown function of one independent variable to one or more of its derivatives}}
\newglossaryentry{initialvalueproblem}{name={initial-value problem},description={a differential equation together with the state specified at an initial time}}
\newglossaryentry{directionfield}{name={direction field},description={a geometric display of the slope prescribed by a first-order differential equation at each point of the independent-dependent-variable plane}}

\newglossaryentry{rungekutta}{name={Runge--Kutta method},description={a family of one-step numerical integration methods that combine several slope evaluations to achieve higher-order accuracy}}
